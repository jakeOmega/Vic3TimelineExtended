# ----------------------------------------------------------------------------
# PLAYER-FACING SURFACE: the policy dashboard widget
# (gui/journal_entry_widgets/banking_dashboard_widget.gui). Every banking button
# below carries `is_ai = yes` in its `visible`, so the vanilla button grid shows
# nothing to a human: the buttons exist for the AI, which picks policies through
# their `ai_chance` (confirmed in play testing, 2026-09-18). Do not remove them,
# and do not drop the is_ai line without restoring a player surface.
# ----------------------------------------------------------------------------
# The on/off "Raise Policy Rate" pair that used to open this file is GONE
# (monetary_policy_design.md §18). The policy rate is now a continuously set
# country variable (te_policy_rate), and the AI steers it through its mandate
# formula in common/scripted_effects/te_monetary_effects.txt — the old button's
# frenzy/boom/recession weights live on as that formula's cycle_lean term (§6).
# Where a weight below used to read `banking_tool_rate_hike_active`, it now reads
# `banking_stance_is_tight`: the same question ("is the central bank leaning
# against the cycle right now?"), asked of the rate instead of a toggle.

cb_open_market_ops = {
	name = "CB_OPEN_MARKET_OPS"
	desc = "CB_OPEN_MARKET_OPS_DESC"
	visible = { is_ai = yes banking_is_market_economy = yes banking_tool_omo_active = no }

	# `possible` now gates this on the policy rate sitting at its floor
	# (banking_possible_cb_open_market_ops, monetary_policy_design.md §11), so
	# these weights no longer have to ask whether easing is the right direction —
	# at the floor it always is. They answer the remaining question: how badly
	# does the economy need more than a zero rate can give?
	#
	# The old `banking_stance_is_tight -> -30` line is dropped because it is now
	# unreachable: tight means the rate is at least 1pp above the neutral rate,
	# the neutral rate is clamped at 1 or above, and the floor test caps the rate
	# at 0.01. Even a stance gap one month stale cannot bridge that — the rate
	# drifts by at most 2/3 of a point a month.
	#
	# No weight asks about the currency regime either. OMO is fiat/digital only
	# (monetary_policy_design.md §0.1 ruling of 2026-09-20), and `possible`
	# refuses the button outright off-regime — an `ai_chance` term for it would
	# be scoring a click the AI can never make.
	#
	# §11's rule is "use it at the floor in recession OR DEFLATION". The floor
	# is `possible`; recession is the panic and downturn pair below
	# (banking_cycle_is_recession is exactly those two); deflation is the phase-2
	# half, and it is a separate reason to buy assets rather than a symptom of
	# the cycle — prices can fall while the cycle reads stable, and that is the
	# case a rate at its floor cannot answer on its own. te_inflation is read
	# behind has_variable because it does not exist until a country's first
	# monthly monetary update, and against te_mon_band_edge_deflation rather
	# than a literal 0, so the tool and the §9.2 deflation band agree on what
	# deflation is.
	ai_chance = {
		value = 0
		# Core: the deeper the hole, the more worth buying assets out of it.
		if = { limit = { banking_cycle_is_panic = yes } add = 70 }
		if = { limit = { banking_cycle_is_downturn = yes } add = 45 }
		if = { limit = { banking_cycle_is_stagnation = yes } add = 25 }
		if = { limit = { banking_cycle_risk_falling = yes } add = 15 }

		# Deflation: falling prices at a rate that can go no lower.
		if = {
			limit = {
				has_variable = te_inflation
				var:te_inflation < te_mon_band_edge_deflation
			}
			add = 40
		}

		# Law flavor and resources, only while a core term above is live
		# (banking_ai_core_*, banking_policy_triggers.txt).
		if = {
			limit = { banking_ai_core_cb_open_market_ops = yes }
			# an independent central bank is the one most willing to reach past
			# the rate for the balance sheet.
			if = { limit = { has_law = law_type:law_central_bank_independence } add = 15 }
			if = { limit = { banking_points_med = yes } add = 10 }
			if = { limit = { banking_points_low = yes } add = 5 }
		}

		# Avoid when hot: a zero rate during a frenzy is already a bubble waiting
		# to happen, and the tool now adds +1.5 bubble pressure a month.
		if = { limit = { banking_cycle_is_frenzy = yes } add = -40 }
		if = { limit = { banking_cycle_bubble_risk_high = yes } add = -30 }

		# And not while prices are already running: QE is worth +1.0pp of
		# standing inflation pressure (§9.1, §11), which is the last thing a
		# country outside the comfort band needs.
		if = {
			limit = {
				has_variable = te_inflation
				var:te_inflation >= te_mon_band_edge_elevated
			}
			add = -35
		}

		# (Optional) Fiscal stress: conservative on big asset buys if highly stressed
		if = { limit = { banking_fiscal_stress_high = yes } add = -15 }
	}


	possible = { banking_possible_cb_open_market_ops = yes }
	effect = { banking_effect_cb_open_market_ops = yes }
}

cb_disable_open_market_ops = {
	name = "CB_DISABLE_OPEN_MARKET_OPS"
	desc = "CB_DISABLE_OPEN_MARKET_OPS_DESC"
	visible = { is_ai = yes banking_tool_omo_active = yes }
	# The enable side is gated to the rate floor and now also scores deflation;
	# nothing on this side could see inflation at all, so a country that eased
	# into a price surge had no reason to stop until its cycle turned. The
	# inflation weight below is that reason: while the tool is on it is adding
	# +1.0pp of standing pressure (§9.1, §11), which is a cost the phase-1
	# weights were written before there was any such thing.
	ai_chance = {
		value = 0
		if = { limit = { banking_cycle_is_expansion = yes } add = 10 }
		if = { limit = { banking_cycle_is_boom = yes } add = 25 }
		if = { limit = { banking_cycle_is_frenzy = yes } add = 55 }
		if = { limit = { banking_cycle_bubble_risk_high = yes } add = 30 }
		if = { limit = { banking_stance_is_tight = yes } add = 35 } # stop easing while the rate is tight
		if = {
			limit = {
				has_variable = te_inflation
				var:te_inflation >= te_mon_band_edge_elevated
			}
			add = 40
		}
	}
	possible = { banking_tool_omo_active = yes }
	effect = { banking_effect_cb_disable_open_market_ops = yes }
}

# Capital controls are the trilemma's third corner (monetary_policy_design.md
# §15.4): they damp the carry term of the exchange rate, the gold-flow gap and
# the overvaluation drain to a quarter, at the price of te_capital_controls_fatigue
# for every PEACETIME month held. So the AI reaches for them in an external
# crisis with something to defend, and drops them once the crisis is over —
# te_mon_in_external_crisis is the one definition shared with the fatigue
# counter's freeze, so "when the AI uses controls" and "when controls are
# forgiven" cannot drift apart. (The two FX tools that used to sit above this
# one were deleted in phase 4: a weak or strong currency is an outcome now.)
cb_capital_controls_outflow = {
	name = "CB_CAPITAL_CONTROLS_OUT"
	desc = "CB_CAPITAL_CONTROLS_OUT_DESC"
	visible = { is_ai = yes banking_is_market_economy = yes banking_tool_capital_controls_active = no }

	ai_chance = {
		value = 0
		# THE CRISIS RULE IS THE MONETARY LAYER'S. Under the simplified game rule
		# there is no exchange rate, no gold flow and no peg, so every input below
		# reads its neutral value and te_mon_in_external_crisis is permanently no —
		# the AI would never reach for controls at all. That branch falls back on
		# the cycle rule this button carried before phase 1: controls are what a
		# panic calls for.
		if = {
			limit = { te_mon_full_system = yes }
			# Core: an external crisis AND something for controls to protect — a
			# currency in flight, a draining vault, or a peg losing confidence (the
			# same <= 40 the crisis trigger uses).
			if = {
				limit = {
					te_mon_in_external_crisis = yes
					OR = {
						te_mon_fx_index_display < 85
						AND = {
							te_mon_has_gold_flows = yes
							has_variable = te_gold_flow
							var:te_gold_flow < 0
							te_mon_bank_gold_scaled < 0.3
						}
						AND = {
							te_mon_has_gold_flows = yes
							has_variable = te_peg_confidence
							var:te_peg_confidence <= 40
						}
					}
				}
				add = 70
			}
			# A crisis with nothing external at stake: a weak case.
			else_if = {
				limit = { te_mon_in_external_crisis = yes }
				add = 15
			}
		}
		else = {
			if = { limit = { banking_cycle_is_panic = yes } add = 60 }
			if = { limit = { banking_cycle_is_downturn = yes } add = 35 }
			if = { limit = { banking_cycle_risk_falling = yes } add = 15 }
		}

		# Law flavor and resources, only while a core term above is live. Under
		# the full system with no external crisis this whole block is 0, so a
		# country with nothing to defend never puts controls on — before the
		# gate, a universal-banking AI at 3-5 points scored 10-15 here every
		# month of the century and toggled controls on and off through it.
		if = {
			limit = { banking_ai_core_cb_capital_controls_outflow = yes }
			if = { limit = { has_law = law_type:law_prudential_narrow_banking } add = 15 }
			if = { limit = { has_law = law_type:law_directed_credit_development_banks } add = 10 }
			if = { limit = { banking_points_med = yes } add = 10 }
			if = { limit = { banking_points_low = yes } add = 5 }
		}

		# Avoid when hot: nothing is leaving.
		if = { limit = { banking_cycle_is_frenzy = yes } add = -40 }
		if = { limit = { banking_cycle_bubble_risk_high = yes } add = -20 }
	}

	possible = { banking_possible_cb_capital_controls_outflow = yes }
	effect = { banking_effect_cb_capital_controls_outflow = yes }
}

cb_disable_capital_controls_outflow = {
	name = "CB_DISABLE_CAPITAL_CONTROLS_OUT"
	desc = "CB_DISABLE_CAPITAL_CONTROLS_OUT_DESC"
	visible = { is_ai = yes banking_tool_capital_controls_active = yes }
	ai_chance = {
		value = 0
		# Out of the crisis: lift them, and more urgently the longer the
		# fatigue counter has run (+10 a peacetime year held, to +50).
		#
		# The simplified game rule has no external crisis to be out of and no
		# fatigue counter, so it takes the pre-phase-1 rule instead: lift them as
		# the cycle recovers. Paired with the enable side's fallback above.
		if = {
			limit = { te_mon_full_system = yes }
			if = {
				limit = { te_mon_in_external_crisis = no }
				add = 40
				add = {
					value = te_mon_controls_fatigue_steps
					multiply = 10
				}
			}
			if = { limit = { banking_cycle_is_boom = yes } add = 10 }
			if = { limit = { banking_cycle_is_frenzy = yes } add = 20 }
		}
		else = {

cb_reserve_requirements = {
	name = "CB_RESERVE_REQUIREMENTS"
	desc = "CB_RESERVE_REQUIREMENTS_DESC"
	visible = { is_ai = yes banking_is_market_economy = yes banking_tool_reserve_requirements_active = no }

	ai_chance = {
		value = 0
		# Core: the countercyclical buffer's — lean on a boom, not on mild expansion
		if = { limit = { banking_cycle_is_expansion = yes } add = 10 }
		if = { limit = { banking_cycle_is_boom = yes } add = 30 }
		if = { limit = { banking_cycle_is_frenzy = yes } add = 35 }
		if = { limit = { banking_cycle_bubble_risk_rising = yes } add = 15 }

		# Law flavor and resources, only while a core term above is live
		if = {
			limit = { banking_ai_core_cb_reserve_requirements = yes }
			# the early-era lean: preferred while the buffer is still locked
			if = { limit = { modifier:country_can_use_countercyclical_buffer_bool = no } add = 15 }
			if = { limit = { has_law = law_type:law_prudential_narrow_banking } add = 15 }
			if = { limit = { has_law = law_type:law_universal_banking_light_prudence } add = 10 }
			if = { limit = { has_law = law_type:law_free_mutual_banking } add = 5 }
			if = { limit = { banking_points_low = yes } add = 10 }  # cheap, so attractive when points are scarce
		}

		# Avoid in recession/panic
		if = { limit = { banking_cycle_is_recession = yes } add = -60 }

		# Defer to the buffer, the stronger lean, whenever the buffer could be
		# bought instead. Reserve requirements are the lean BEFORE the buffer's
		# tech, and a second lean BESIDE a running buffer, never a weaker stand-in
		# for it: with the buffer's weights and nothing else, the AI split its
		# leaning between the two, and at 3 points a fiat or digital country
		# crashed 1.5 times a century more (banking_cycle_simulation.md §11).
		if = {
			limit = {
				banking_tool_buffer_active = no
				banking_possible_cb_countercyclical_buffer = yes
			}
			multiply = 0
		}
	}

	possible = { banking_possible_cb_reserve_requirements = yes }
	effect = { banking_effect_cb_reserve_requirements = yes }
}
