# Global Warming mitigation buttons
gw_carbon_tax_button = {
	name = "GW_CARBON_TAX"
	desc = "GW_CARBON_TAX_DESC"

	visible = {
		NOT = { je:je_global_warming = { has_modifier = carbon_tax_modifier } }
		# Emissions reduction only applies for the market leader; restrict button accordingly
		market_capital.owner = THIS
	}

	ai_chance = {
		value = 0
		if = {
			limit = { temperature_anomaly_display >= 1.0 }
			add = 30
		}
		if = {
			limit = { temperature_anomaly_display >= 2.0 }
			add = 30
		}
		if = {
			limit = {
				has_law = law_type:law_laissez_faire
			}
			add = -50
		}
	}

	possible = { gw_possible_carbon_tax = yes }

	effect = { gw_effect_carbon_tax = yes }
}

gw_remove_renewable_investment_button = {
	name = "GW_REMOVE_RENEWABLE_INVESTMENT"
	desc = "GW_REMOVE_RENEWABLE_INVESTMENT_DESC"

	visible = {
		je:je_global_warming = { has_modifier = renewable_investment_modifier }
		market_capital.owner = THIS
		NOT = { has_variable = has_emissions_reduction_treaty }
	}

	ai_chance = {
		value = 0
		# AI removes renewable investment only when warming is negligible
		# (warming below threshold = less urgency to maintain green policies)
		if = {
			limit = { temperature_anomaly_display < 0.3 }
			add = 60
		}
		# AI also more likely to remove if warming isn't worsening significantly
		if = {
			limit = {
				has_law = law_type:law_laissez_faire
				temperature_anomaly_display < 1.0
			}
			add = 30
		}
	}

	possible = { gw_possible_remove_renewable_investment = yes }

	effect = { gw_effect_remove_renewable_investment = yes }
}
