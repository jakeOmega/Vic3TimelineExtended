# Global Warming mitigation buttons
#
# ----------------------------------------------------------------------------
# PLAYER-FACING SURFACE: the climate dashboard widget
# (gui/journal_entry_widgets/global_warming_widget.gui). Every button below
# carries `is_ai = yes` in its `visible`, so the vanilla button grid shows
# nothing to a human: the buttons exist for the AI, which picks policies through
# their `ai_chance`. Do not remove them, and do not drop the is_ai line without
# restoring a player surface -- a scripted GUI has its own, separate AI hook
# (ai_is_valid/ai_chance/ai_frequency) and the widget's handlers all set
# `ai_is_valid = { always = no }`, so deleting a scripted_button would silently
# remove that option from the AI with no log line and no test that catches it.
#
# Each button's `possible` and `effect` bodies live in
# common/scripted_triggers/global_warming_triggers.txt and
# common/scripted_effects/global_warming_effects.txt, which the widget's scripted
# GUIs call as well. Change a policy's eligibility or effect in the HELPER, never
# here and never in the scripted GUI.
#
# `visible` is deliberately left as it was otherwise, inconsistencies included
# (some rows hid themselves before the warming threshold, others showed greyed).
# The widget normalises that by showing all eight rows always and letting
# `is_valid` carry the anomaly and authority gates as readable conditions, so it
# is intentionally more forthcoming than this grid ever was.
# ----------------------------------------------------------------------------
gw_carbon_tax_button = {
	name = "GW_CARBON_TAX"
	desc = "GW_CARBON_TAX_DESC"

	visible = {
		is_ai = yes
		NOT = { je:je_global_warming = { has_modifier = carbon_tax_modifier } }
		# Emissions reduction only applies for the market leader; restrict button accordingly
		market_capital.owner = THIS
	}

	ai_chance = {
		value = 0
		if = {
			limit = { temperature_anomaly_display >= 1.0 }
			add = 30
		}
		if = {
			limit = { temperature_anomaly_display >= 2.0 }
			add = 30
		}
		if = {
			limit = {
				has_law = law_type:law_laissez_faire
			}
			add = -50
		}
	}

	possible = { gw_possible_carbon_tax = yes }

	effect = { gw_effect_carbon_tax = yes }
}

gw_remove_renewable_investment_button = {
	name = "GW_REMOVE_RENEWABLE_INVESTMENT"
	desc = "GW_REMOVE_RENEWABLE_INVESTMENT_DESC"

	visible = {
		is_ai = yes
		je:je_global_warming = { has_modifier = renewable_investment_modifier }
		market_capital.owner = THIS
		NOT = { has_variable = has_emissions_reduction_treaty }
	}

	ai_chance = {
		value = 0
		# AI removes renewable investment only when warming is negligible
		# (warming below threshold = less urgency to maintain green policies)
		if = {
			limit = { temperature_anomaly_display < 0.3 }
			add = 60
		}
		# AI also more likely to remove if warming isn't worsening significantly
		if = {
			limit = {
				has_law = law_type:law_laissez_faire
				temperature_anomaly_display < 1.0
			}
			add = 30
		}
	}

	possible = { gw_possible_remove_renewable_investment = yes }

	effect = { gw_effect_remove_renewable_investment = yes }
}
