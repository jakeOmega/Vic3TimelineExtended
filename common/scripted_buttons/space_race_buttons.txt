# Space Race Per-JE Scripted Buttons
# 4 buttons per milestone x 8 milestones = 32 buttons
# Each milestone has independent funding level and approach selection

# ----------------------------------------------------------------------------
# PLAYER-FACING SURFACE: the milestone panel widget
# (gui/journal_entry_widgets/space_race_widget.gui). Every button below carries
# `is_ai = yes` in its `visible`, so the vanilla button grid shows nothing to a
# human: the buttons exist for the AI, which picks its approach and funding
# level through their `ai_chance`. Do not remove them, and do not drop the
# is_ai line without restoring a player surface.
#
# Nothing here holds a rule of its own any more. `visible`, `possible` and
# `effect` all delegate to the shared helpers that the widget's scripted GUIs
# also call (common/scripted_triggers/space_race_triggers.txt,
# common/scripted_effects/space_race_effects.txt), so the AI's grid and the
# player's panel cannot disagree. Change an approach or funding rule in the
# helper; the `ai_chance` weights below are the only per-button content left.
# ----------------------------------------------------------------------------


# ======= SUBORBITAL Buttons =======

sr_btn_safe_suborbital = {
	name = "SR_SAFE_APPROACH"
	desc = "SR_SAFE_APPROACH_DESC"

	visible = {
		is_ai = yes
		sr_controls_shown = { MILESTONE = suborbital }
	}

	ai_chance = {
		value = 50
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = -20
		}
	}

	possible = { sr_possible_safe = { MILESTONE = suborbital } }

	effect = { sr_effect_safe = { MILESTONE = suborbital } }
}

sr_btn_ambitious_suborbital = {
	name = "SR_AMBITIOUS_APPROACH"
	desc = "SR_AMBITIOUS_APPROACH_DESC"

	visible = {
		is_ai = yes
		sr_controls_shown = { MILESTONE = suborbital }
	}

	ai_chance = {
		value = 30
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = 30
		}
		if = {
			limit = {
				any_country = {
					NOT = { THIS = ROOT }
					has_variable = sr_active_suborbital
				}
			}
			add = 20
		}
	}

	possible = { sr_possible_ambitious = { MILESTONE = suborbital } }

	effect = { sr_effect_ambitious = { MILESTONE = suborbital } }
}
