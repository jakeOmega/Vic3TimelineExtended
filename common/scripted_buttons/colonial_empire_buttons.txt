# ----------------------------------------------------------------------------
# PLAYER-FACING SURFACE: the colonial stability widget
# (gui/journal_entry_widgets/colonial_empire_widget.gui). Every button below
# carries `is_ai = yes` in its `visible`, so the vanilla button grid shows
# nothing to a human: the buttons exist for the AI, which picks policies and
# decolonization decisions through their `ai_chance`. Do not remove them, and do
# not drop the is_ai line without restoring a player surface.
#
# `possible` and `effect` delegate to colonial_empire_possible_* /
# colonial_empire_effect_* (common/scripted_triggers/colonial_empire_triggers.txt,
# common/scripted_effects/decolonization.txt), which the widget's scripted GUIs
# in common/scripted_guis/colonial_empire_sguis.txt call as well -- so the AI's
# grid and the player's widget can never drift apart. Change a gate or an action
# in the helper, never here.
# ----------------------------------------------------------------------------

# ------ Invest in Colonial Development ------
ce_invest_in_development = {
	name = "CE_INVEST_IN_DEVELOPMENT"
	desc = "CE_INVEST_IN_DEVELOPMENT_DESC"

	visible = {
		is_ai = yes
		NOT = { has_modifier = colonial_development_investment_modifier }
	}

	ai_chance = {
		value = 0
		if = {
			limit = { colonial_state_count >= 1 }
			add = 30
		}
		if = {
			limit = { colonial_low_acceptance_count >= 2 }
			add = 20
		}
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = 10
		}
	}

	possible = {
		colonial_empire_possible_invest = yes
	}

	effect = {
		colonial_empire_effect_invest = yes
	}
}

# ------ Release Colonial Territory (one region at a time) ------
ce_release_colonial_territory = {
	name = "CE_RELEASE_COLONIAL_TERRITORY"
	desc = "CE_RELEASE_COLONIAL_TERRITORY_DESC"

	# No player-facing conditions: this button is AI-only, and the widget's
	# decision row carries the human's version of it.
	visible = {
		is_ai = yes
	}

	ai_chance = {
		value = 0
		if = {
			limit = { colonial_gp_condemners_count >= 3 }
			add = 15
		}
		if = {
			limit = { colonial_low_acceptance_count >= 3 }
			add = 20
		}
		# Less likely for great powers
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = -20
		}
	}

	possible = {
		colonial_empire_possible_release_territory = yes
	}

	effect = {
		colonial_empire_effect_release_territory = yes
	}
}

# ------ Planned Full Decolonization ------
ce_planned_decolonization = {
	name = "CE_PLANNED_DECOLONIZATION"
	desc = "CE_PLANNED_DECOLONIZATION_DESC"

	# No player-facing conditions: this button is AI-only, and the widget's
	# decision row carries the human's version of it.
	visible = {
		is_ai = yes
	}

	ai_chance = {
		value = 0
		# AI does this under moderate pressure now (lowered thresholds)
		if = {
			limit = { colonial_gp_condemners_count >= 3 }
			add = 30
		}
		if = {
			limit = { colonial_low_acceptance_count >= 3 }
			add = 30
		}
		if = {
			limit = { NOT = { country_rank >= rank_value:great_power } }
			add = 30
		}
		# Never if things are going well
		if = {
			limit = { colonial_high_acceptance_count >= 2 }
			add = -100
		}
	}

	possible = {
		colonial_empire_possible_planned_decolonization = yes
	}

	effect = {
		colonial_empire_effect_planned_decolonization = yes
	}
}

# ------ Round Table Conference ------
# A negotiated transition for ONE colony at a time. Costs authority + prestige
# up front, but produces a high-acceptance independent country with the
# Commonwealth Legacy modifier — friendly diplomatic relations, technological
# cooperation, and no infamy hit. Distinct from "Release Colonial Territory"
# (which is a unilateral grant) and "Planned Full Decolonization" (which
# liquidates the entire empire). The Round Table is for the colony you want
# to leave on good terms while keeping the rest.
ce_round_table_conference = {
	name = "CE_ROUND_TABLE_CONFERENCE"
	desc = "CE_ROUND_TABLE_CONFERENCE_DESC"

	# No player-facing conditions: this button is AI-only, and the widget's
	# decision row carries the human's version of it.
	visible = {
		is_ai = yes
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				colonial_high_acceptance_count >= 1
				colonial_gp_condemners_count >= 2
			}
			add = 30
		}
		# Intelligentsia leadership pushes for negotiated transitions
		if = {
			limit = {
				ig:ig_intelligentsia ?= {
					leader = { has_ideology = ideology:ideology_anti_colonialist }
				}
			}
			add = 20
		}
	}

	possible = {
		colonial_empire_possible_round_table = yes
	}

	effect = {
		colonial_empire_effect_round_table = yes
	}
}
