# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). je_te_map_host's buttons: one
# per map, each writing the selector its map_color branches on, and Close. Root: the entry's owner, the player's
# country. The AI never holds the entry (only a player's button adds it), so every ai_chance is 0.

te_jmap_pick_turmoil = {
	name = "te_jmap_pick_turmoil"
	desc = "te_jmap_pick_turmoil_desc"
	visible = { always = yes }
	selected = { var:te_jmap_sel ?= 1 }
	possible = { always = yes }
	ai_chance = { value = 0 }
	effect = {
		te_jmap_set_selector = { SEL = 1 }
	}
}
