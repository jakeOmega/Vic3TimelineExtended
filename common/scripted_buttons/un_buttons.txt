# ============================================================================
# UNITED NATIONS SCRIPTED BUTTONS
# Founding, membership, request, policy, and GP influence buttons
# ============================================================================
#
# THE SIX PROPOSAL BUTTONS ARE NOW THIN. un_request_peacekeepers,
# un_request_humanitarian_aid, un_propose_condemn, un_propose_expulsion,
# un_propose_mandate and un_propose_sanctions keep their name, desc and
# ai_chance here and delegate `visible` / `possible` / `effect` to
# common/scripted_triggers/un_propose_triggers.txt and
# common/scripted_effects/un_propose_effects.txt. The chamber widget's "Table a
# Resolution" section calls the very same triggers and effects, so a human
# proposing from the journal entry and an AI proposing through the button grid
# run identical script. Edit the shared file, never a copy.
#
# The buttons are deliberately NOT hidden for human players now that the chamber
# can propose: they are the AI's only proposal path, and hiding a scripted
# button from humans alone has no verified mechanism in this engine.
#
# The ten conventions (human rights, ICC, NPT, climate, pandemic, refugee,
# heritage, decolonization, space, law of the sea) deliberately have NO button.
# A human tables one from the chamber (ops 7-16 of un_chamber_propose_sgui);
# the AI reaches them only through the docket's offers (un_docket_effects.txt),
# which pace the floor. A button here would give the AI an ai_chance path that
# skips that pacing.

# ---- 0. FOUND THE UNITED NATIONS ----
# A Great Power with intergovernmental_organizations tech creates the UN.
# Any recognized independent country can then join.

un_found_button = {
	name = "UN_FOUND"
	desc = "UN_FOUND_DESC"

	# After a dissolution the same button convenes a founding conference
	# instead (redesign phase 2, un_ladder_effects.txt): hidden while one sits,
	# and greyed out until the refounding cooldown has run.
	visible = {
		NOT = { has_global_variable = un_founded }
		NOT = { un_conference_in_session = yes }
		has_technology_researched = intergovernmental_organizations
	}

	possible = {
		NOT = { has_global_variable = un_founded }
		NOT = { un_conference_in_session = yes }
		has_technology_researched = intergovernmental_organizations
		country_rank >= rank_value:great_power
		is_subject = no
		custom_tooltip = {
			text = un_refounding_cooldown_tt
			NOT = { has_global_variable = un_refounding_cooldown }
		}
	}

	ai_chance = {
		value = 20
		if = { limit = { has_law = law_type:law_isolationism } add = -60 }
		if = { limit = { has_law = law_type:law_humanitarian_regulations } add = 20 }
		if = { limit = { has_law = law_type:law_limited_war } add = 40 }
	}

	effect = {
		if = {
			limit = { un_founding_is_refounding = yes }
			custom_tooltip = {
				text = "UN_CONVENE_CONFERENCE_TT"
				un_conference_open = yes
			}
		}
		else = {
			custom_tooltip = {
				text = "UN_FOUND_DESC"
				# Cost: prestige and bureaucracy
				add_modifier = {
					name = un_founding_cost_modifier
					days = normal_modifier_time
					is_decaying = yes
				}
				# UN authority is a single GLOBAL value; born here at founding (50),
				# then moving toward the equilibrium its members give it
				# (un_authority_effects.txt). The founder becomes first member, HQ
				# host, founding member and, as a great power, a permanent member;
				# a 5-year founding window seats the next great powers to join
				# (un_found_organisation, un_ladder_effects.txt).
				un_found_organisation = { AUTHORITY = un_founding_authority }
				# Fire the charter event for all other recognized independent countries
				un_charter_invitations = yes
			}
		}
	}
}

# ---- 1b-bis. ASSESSED DUES (redesign phase 5, §7.2) ----
# Every member pays a share of its GDP set by the UN's tier
# (un_dues_values.txt). A member may withhold it: the money is saved, the
# funding pillar falls with the withholders' share of the members' GDP, the
# record takes a standing loss, and after 24 months in arrears the member loses
# its vote (Article 19). Paying again settles everything owed at once.

un_withhold_dues_button = {
	name = "UN_WITHHOLD_DUES"
	desc = "UN_WITHHOLD_DUES_DESC"

	# Our dues are our overlord's while our representation is suspended (§0.8).
	visible = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		NOT = { un_representation_suspended = yes }
		NOT = { un_dues_is_withholding = yes }
	}

	possible = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		NOT = { un_representation_suspended = yes }
		NOT = { un_dues_is_withholding = yes }
		custom_tooltip = {
			text = un_dues_assessed_tt
			un_dues_assessed = yes
		}
	}

	ai_chance = {
		value = 0
		# An isolationist or an underminer has no wish to pay for the order.
		if = { limit = { has_law = law_type:law_isolationism } add = 15 }
		if = {
			limit = { je:je_united_nations ?= { has_modifier = un_undermine_order_cost } }
			add = 20
		}
		# The levy bites: 0.4% of GDP at Strong, 1% at Supranational.
		if = { limit = { un_tier_at_least_strong = yes } add = 5 }
		if = { limit = { un_tier_is_supranational = yes } add = 10 }
		# A treasury in trouble looks for savings.
		if = { limit = { in_default = yes } add = 25 }
		else_if = { limit = { scaled_debt > 0.75 } add = 10 }
		# A champion of the order pays; so does anyone who values the vote.
		if = {
			limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
			add = -50
		}
		if = { limit = { is_un_permanent_member = yes } add = -10 }
		if = { limit = { has_law = law_type:law_humanitarian_regulations } add = -10 }
	}

	effect = {
		custom_tooltip = {
			text = un_dues_withhold_effect_tt
			un_dues_withhold = yes
		}
	}
}

# ---- 1c-bis. TABLE A CHARTER REFORM ----
# Redesign phase 2. The charter caps UN authority at 70; Reform I raises the
# ceiling to 85 and Reform II to 100. A reform may be tabled only once
# authority has sat within 5 points of the ceiling for 24 months running;
# it needs a two-thirds supermajority of all members, and any permanent
# member can veto it outright. Gates and effect are shared with the
# chamber's Propose row (un_propose_triggers.txt / un_propose_effects.txt).

un_propose_charter_reform_button = {
	name = "UN_PROPOSE_CHARTER_REFORM"
	desc = "UN_PROPOSE_CHARTER_REFORM_DESC"

			limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }

		}

# ---- 1d. PROPOSE AN AUTHORIZED MILITARY MANDATE ----
# A binding, vetoable resolution licensing ONE member to recover ONE state
# region it claims from ONE state the Assembly has already censured. Carrying
# it issues a mandate container (common/scripted_effects/un_mandate_effects.txt);
# the mandate is what makes te_un_mandate_restore_state selectable, and free.
#
# The button picks the case, rather than offering a selection UI: a scripted
# button's effect cannot prompt, and the alternative — an event listing N
# candidate pairs — buys a choice the eligibility rules have already narrowed
# to a handful. un_mandate_select_case picks the highest-scoring target
# (un_mandate_case_score) and its largest claimed state; the same pick is
# rendered in the chamber widget's "Mandates in Force" panel *before* the
# button is pressed, so the player sees which case they would be tabling.
#
# Cooldown: deliberately NOT un_resolution_topic_on_cooldown. That check is
# topic-wide — one member's mandate vote would bar every other member for five
# years, which would leave the feature dead on a busy map. The throttle is
# per-country instead: un_mandate_proposal_cooldown (5 years) plus the shared
# un_request_cooldown that every propose button already takes.
un_propose_mandate_button = {
	name = "UN_PROPOSE_MANDATE"
	desc = "UN_PROPOSE_MANDATE_DESC"

	visible = { un_propose_mandate_available = yes }

	possible = { un_propose_mandate_possible = yes }

	ai_chance = {
		value = 0
		# An AI only reaches for force through the Assembly when the Assembly
		# is worth something and it cannot simply take the land cheaply.
		if = { limit = { global_var:un_authority >= 60 } add = 20 }
		if = { limit = { infamy >= 25 } add = 20 }
		if = { limit = { country_rank >= rank_value:major_power } add = 10 }
		if = { limit = { has_law = law_type:law_isolationism } add = -40 }
		if = { limit = { has_law = law_type:law_total_war } add = -20 }
	}

	effect = { un_propose_mandate_effect = yes }
}

	name = "UN_LIFT_SANCTIONS"
	desc = "UN_LIFT_SANCTIONS_DESC"

	visible = {
		has_modifier = un_sanctions_enforcer_modifier
	}

	possible = {
		has_modifier = un_sanctions_enforcer_modifier
	}

	ai_chance = {
		value = 0
		if = { limit = { global_var:un_authority < 30 } add = 20 }
	}

	effect = {
		custom_tooltip = {
			text = "UN_LIFT_SANCTIONS_DESC"
			remove_modifier = un_sanctions_enforcer_modifier
			# The UN's record below names the sanctioned country when there is
			# one. The button lifts every regime in force, so with several it

			# line. One booking either way: gen_un_button_descs.py prints a

				limit = {
					NOT = {
						any_country = {
							count >= 2
							has_modifier = un_sanctions_target_modifier
						}
					}
				}
				random_country = {
					limit = { has_modifier = un_sanctions_target_modifier }
					save_scope_as = un_sanctions_lifted_target
				}
			}
			every_country = {
				limit = { has_modifier = un_sanctions_target_modifier }
				remove_modifier = un_sanctions_target_modifier
				# The embargoes the Assembly imposed go with the regime
				# (redesign phase 5, un_teeth_effects.txt).
				un_teeth_lift_embargoes = yes
			}
			# Walking away from a regime the Assembly imposed costs its credibility,
			# weighted by the enforcer's weight in world affairs.
			un_ledger_actor_entry_on = { PILLAR = credibility POINTS = -1 REASON = 7 SUBJECT = scope:un_sanctions_lifted_target }
		}
	}
