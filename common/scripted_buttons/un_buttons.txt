# ============================================================================
# UNITED NATIONS SCRIPTED BUTTONS
# Founding, membership, request, policy, and GP influence buttons
# ============================================================================
#
# THE SIX PROPOSAL BUTTONS ARE NOW THIN. un_request_peacekeepers,
# un_request_humanitarian_aid, un_propose_condemn, un_propose_expulsion,
# un_propose_mandate and un_propose_sanctions keep their name, desc and
# ai_chance here and delegate `visible` / `possible` / `effect` to
# common/scripted_triggers/un_propose_triggers.txt and
# common/scripted_effects/un_propose_effects.txt. The chamber widget's "Table a
# Resolution" section calls the very same triggers and effects, so a human
# proposing from the journal entry and an AI proposing through the button grid
# run identical script. Edit the shared file, never a copy.
#
# The buttons are deliberately NOT hidden for human players now that the chamber
# can propose: they are the AI's only proposal path, and hiding a scripted
# button from humans alone has no verified mechanism in this engine.

# ---- 0. FOUND THE UNITED NATIONS ----
# A Great Power with intergovernmental_organizations tech creates the UN.
# Any recognized independent country can then join.

un_found_button = {
	name = "UN_FOUND"
	desc = "UN_FOUND_DESC"

	visible = {
		NOT = { has_global_variable = un_founded }
		has_technology_researched = intergovernmental_organizations
	}

	possible = {
		NOT = { has_global_variable = un_founded }
		has_technology_researched = intergovernmental_organizations
		country_rank >= rank_value:great_power
		is_subject = no
	}

	ai_chance = {
		value = 20
		if = { limit = { has_law = law_type:law_isolationism } add = -60 }
		if = { limit = { has_law = law_type:law_humanitarian_regulations } add = 20 }
		if = { limit = { has_law = law_type:law_limited_war } add = 40 }
	}

	effect = {
		custom_tooltip = {
			text = "UN_FOUND_DESC"
			# Cost: prestige and bureaucracy
			add_modifier = {
				name = un_founding_cost_modifier
				days = normal_modifier_time
				is_decaying = yes
			}
			# Establish the UN
			set_global_variable = un_founded
			# UN authority is a single GLOBAL value; born here at founding (neutral 50).
			if = {
				limit = { NOT = { has_global_variable = un_authority } }
				set_global_variable = { name = un_authority value = 50 }
			}
			# 5-year founding window: any GP that joins during this window becomes
			# a permanent member (until the 5 seats fill). After it expires, new
			# permanent members can only be created via the expulsion-promotion vote.
			set_global_variable = { name = un_founding_window_active days = normal_modifier_time }
			# Founder becomes first member, HQ host, and founding member
			je:je_united_nations ?= { add_modifier = { name = un_member_modifier } }
			je:je_united_nations ?= { add_modifier = { name = un_founding_member_modifier } }
			je:je_united_nations ?= { add_modifier = { name = un_headquarters_modifier } }
			set_global_variable = { name = un_hq_country value = ROOT }
			# GP founder gets Security Council seat AND permanent membership
			if = {
				limit = { country_rank >= rank_value:great_power }
				je:je_united_nations ?= { add_modifier = { name = un_security_council_modifier } }
				je:je_united_nations ?= { add_modifier = { name = un_permanent_member_modifier } }
			}
			# Fire charter event for all other recognized independent countries
			every_country = {
				limit = {
					NOT = { this = ROOT }
					is_subject = no
					NOT = { is_country_type = decentralized }
				}
				trigger_event = { id = un_events.1 }
			}
		}
	}
}

# ---- 1. JOIN / LEAVE UN ----

un_join_button = {
	name = "UN_JOIN"
	desc = "UN_JOIN_DESC"

	visible = {
		has_global_variable = un_founded
		NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
	}

	possible = {
		has_global_variable = un_founded
		NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		is_subject = no
		NOT = { is_country_type = decentralized }
		NOT = { has_modifier = un_expelled_cooldown }
		NOT = { has_modifier = un_withdrawal_penalty_modifier }
	}

	ai_chance = {
		value = 50
		if = { limit = { country_rank >= rank_value:great_power } add = 30 }
		if = { limit = { country_rank >= rank_value:major_power } add = 15 }
		if = { limit = { has_law = law_type:law_isolationism } add = -80 }
		if = { limit = { has_law = law_type:law_humanitarian_regulations } add = 20 }
		if = { limit = { has_law = law_type:law_limited_war } add = 20 }
		if = { limit = { has_law = law_type:law_total_war } add = -40 }
	}

	effect = {
		custom_tooltip = {
			text = "UN_JOIN_DESC"
			je:je_united_nations ?= { add_modifier = { name = un_member_modifier } }
			# GPs joining during the founding window become permanent members
			# (and pick up the SC seat that goes with the permanent role) so long
			# as the 5-seat cap isn't already full.
			if = {
				limit = {
					country_rank >= rank_value:great_power
					has_global_variable = un_founding_window_active
					un_permanent_seat_count < 5
				}
				je:je_united_nations ?= { add_modifier = { name = un_security_council_modifier } }
				je:je_united_nations ?= { add_modifier = { name = un_permanent_member_modifier } }
			}
			# Bind the joiner to any conventions already ratified globally.
			un_apply_ratified_conventions = yes
		}
	}
}

			# Remove permanent treaty commitments
			je:je_united_nations ?= { remove_modifier = un_human_rights_declaration_modifier }
			remove_modifier = un_human_rights_reservations_modifier
			je:je_united_nations ?= { remove_modifier = un_nonproliferation_modifier }

			je:je_united_nations ?= { remove_modifier = un_refugee_program_modifier }
			je:je_united_nations ?= { remove_modifier = un_space_partnership_modifier }

			# is demolished, so the new host can build the only one

		# without charging for them separately, so this is the only charge for
		# leaving. Wrapped so the variable write is not tooltip noise.
		custom_tooltip = {
			text = un_standing_leave_tt
			un_standing_loss = { AMOUNT = un_standing_leave_loss REASON = 28 }
		}
	}

# ---- 1c. PROPOSE EXPULSION OF A PERMANENT MEMBER ----
# Any UN member can call a 2/3 supermajority vote to strip a permanent
# member's seat. Triggered against a permanent member that has recently
# vetoed a binding resolution (carries un_veto_isolation_modifier — i.e.,
# is freshly visible as a veto abuser).

un_propose_expulsion_button = {
	name = "UN_PROPOSE_EXPULSION"
	desc = "UN_PROPOSE_EXPULSION_DESC"

	visible = { un_propose_expulsion_available = yes }

	possible = { un_propose_expulsion_possible = yes }

	ai_chance = {
		value = 0
		# Strong pull when a rival permanent member just vetoed
		if = {
			limit = {
				any_country = {
					NOT = { this = ROOT }
					is_un_permanent_member = yes
					has_modifier = un_veto_isolation_modifier
					has_diplomatic_pact = { who = ROOT type = rivalry }
				}
			}
			add = 25
		}
		# Less interesting when our own UN authority is weak
		if = { limit = { has_global_variable = un_authority global_var:un_authority < 30 } add = -15 }
	}

	effect = { un_propose_expulsion_effect = yes }
}

# ---- 1d. PROPOSE AN AUTHORIZED MILITARY MANDATE ----
# A binding, vetoable resolution licensing ONE member to recover ONE state
# region it claims from ONE state the Assembly has already censured. Carrying
# it issues a mandate container (common/scripted_effects/un_mandate_effects.txt);
# the mandate is what makes te_un_mandate_restore_state selectable, and free.
#
# The button picks the case, rather than offering a selection UI: a scripted
# button's effect cannot prompt, and the alternative — an event listing N
# candidate pairs — buys a choice the eligibility rules have already narrowed
# to a handful. un_mandate_select_case picks the highest-scoring target
# (un_mandate_case_score) and its largest claimed state; the same pick is
# rendered in the chamber widget's "Mandates in Force" panel *before* the
# button is pressed, so the player sees which case they would be tabling.
#
# Cooldown: deliberately NOT un_resolution_topic_on_cooldown. That check is
# topic-wide — one member's mandate vote would bar every other member for five
# years, which would leave the feature dead on a busy map. The throttle is
# per-country instead: un_mandate_proposal_cooldown (5 years) plus the shared
# un_request_cooldown that every propose button already takes.
un_propose_mandate_button = {
	name = "UN_PROPOSE_MANDATE"
	desc = "UN_PROPOSE_MANDATE_DESC"

	visible = { un_propose_mandate_available = yes }

	possible = { un_propose_mandate_possible = yes }

	ai_chance = {
		value = 0
		# An AI only reaches for force through the Assembly when the Assembly
		# is worth something and it cannot simply take the land cheaply.
		if = { limit = { global_var:un_authority >= 60 } add = 20 }
		if = { limit = { infamy >= 25 } add = 20 }
		if = { limit = { country_rank >= rank_value:major_power } add = 10 }
		if = { limit = { has_law = law_type:law_isolationism } add = -40 }
		if = { limit = { has_law = law_type:law_total_war } add = -20 }
	}

	effect = { un_propose_mandate_effect = yes }
}
