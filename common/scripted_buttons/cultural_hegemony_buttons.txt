# ============================================================================
# CULTURAL HEGEMONY — journal-entry buttons (AI-only surface)
# ============================================================================
# These ten buttons are the AI's ONLY path into the cultural-hegemony policy
# system: the AI picks a policy through their `ai_chance` blocks, so never delete
# a `scripted_button = …` line from je_cultural_hegemony.txt.
#
# Every button carries `is_ai = yes` in its `visible`, so the vanilla journal-entry
# button grid shows nothing to a human. The human surface is the custom widget
# `gui/journal_entry_widgets/cultural_hegemony_widget.gui`, whose controls call
# `ch_policy_sgui` (common/scripted_guis/cultural_hegemony_sguis.txt).
#
# Neither surface holds its own logic. Each button's eligibility lives in
# `ch_possible_<button>` (common/scripted_triggers/cultural_hegemony_triggers.txt)
# and its body in `ch_effect_<button>`
# (common/scripted_effects/cultural_hegemony_effects.txt), and the scripted GUI
# asks and calls exactly the same two helpers — so what the AI does and what the
# widget does cannot drift apart. Change a cost, a gate or an applied modifier in
# the HELPER, never here.
# ============================================================================

ch_increase_program_funding_button = {
	name = "CH_INCREASE_PROGRAM_FUNDING"
	desc = "CH_INCREASE_PROGRAM_FUNDING_DESC"

	# Stays visible while the JE is up; gating moved to `possible` so the
	# button greys out with a tooltip when the player lacks the Ministry of
	# Culture or has no funding cap yet, instead of disappearing silently.
	visible = {
		is_ai = yes
		has_journal_entry = je_cultural_hegemony
	}

	ai_chance = {
		value = 0
		if = {
			limit = { has_variable = ch_total var:ch_total >= 10 }
			add = 20
		}
		if = {
			limit = { has_variable = ch_total var:ch_total < 5 }
			add = 10
		}
	}

	possible = {
		ch_possible_increase_program_funding = yes
	}

	effect = {
		ch_effect_increase_program_funding = yes
	}
}

ch_disable_world_exposition_button = {
	name = "CH_DISABLE_WORLD_EXPOSITION"
	desc = "CH_DISABLE_WORLD_EXPOSITION_DESC"

	visible = {
		is_ai = yes
		has_journal_entry = je_cultural_hegemony
		je:je_cultural_hegemony = { has_modifier = ch_world_exposition }
	}

	# The no-ministry branch below was the AI's escape hatch from an outreach
	# programme it could no longer justify: unlike the other three programmes,
	# the journal entry's monthly pulse never stripped this one when
	# law_ministry_of_culture was repealed, so the AI had to click it off itself.
	# The pulse now strips it too (see je_cultural_hegemony.txt), which makes this
	# branch belt-and-braces rather than the only path — it still fires in the
	# same tick the law goes away, ahead of the next pulse.
	ai_chance = {
		value = 0
		if = {
			limit = { NOT = { has_law = law_type:law_ministry_of_culture } }
			add = 100
		}
	}

	possible = {
		ch_possible_disable_world_exposition = yes
	}

	effect = {
		ch_effect_disable_world_exposition = yes
	}
}
