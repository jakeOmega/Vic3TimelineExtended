# No-First-Strike Pledge - a formal commitment never to initiate use of
# nuclear weapons. Boosts diplomacy but limits military options.
# It is the legal codification of the deterrence doctrine's No First Use
# (nd_doctrine = 1, #430): while it is in force the doctrine is held there
# (nd_weekly_update; nd_init_posture for an arsenal built under it), and
# leaving No First Use repudiates it and strikes it from the law
# (nd_set_doctrine → nd_strike_nfu_amendment). No on_activate / on_deactivate:
# the weekly binding covers every way in, and a hook on removal would loop.
#
# An AI may repeal it only under a militarist regime (#448). Before that it
# was permanent for an AI: the weekly binding holds the doctrine, the AI
# posture review leaves doctrine alone while the amendment is on file, and
# nothing else takes it off. ai_will_revoke is a permission, not an order
# (vanilla's Austrian amendments use it to stop one ruler revoking), so a
# militarist AI may repeal it, and leaving No First Use afterwards still pays
# the repudiation.
amendment_no_first_strike = {
	parent = law_limited_war

	allowed_laws = {
		law_humanitarian_regulations
		law_limited_war
	}

	modifier = {
		country_diplomatic_reputation_add = 5
	}

	# No nuclear weapons in this campaign, nothing to pledge away. Amendments
	# have no `visible`; `possible` is the only gate.
	possible = {
		has_game_rule = nuclear_weapons_enabled
		has_technology_researched = nuclear_weapons
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_trade_unions
		}
	}

	ai_will_revoke = {
		nd_regime_is_militarist = yes
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}

# ======================================================================
#  CENTRAL BANK MANDATES (#799)
#
#  Under central bank independence the bank's mandate is a statute:
#  one of these acts on law_central_bank_independence. They carry no
#  modifier; what they do is the mandate. te_monetary_update_target
#  (common/scripted_effects/te_monetary_effects.txt) reads the act and
#  moves the bank to its mandate 12 months later, AI or player alike,
#  unless the country is on a convertible gold standard, where the bank
#  defends the peg. Spec: docs/systems/monetary_policy_design.md §6.
#
#  WAYS IN. te_mandate.1 asks an independent bank with no act to pass
#  one (when the law passes, on an old save, after a repeal);
#  banking_cycle_events.16 option c passes the Dual Mandate Act; and an
#  interest group may sponsor either while the law is debated.
#
#  CHANGING ONE is repealing it in the law panel (the engine's gate: an
#  interest group in government that opposes the parent law, and its side
#  out-clouting the other), after which te_mandate.1 asks again.
#
#  THE PARENTS are chosen for who backs and who can repeal each act.
#  Inflation Targeting takes its stances from the Gold Standard: the
#  savers and creditors of monetary_policy_design.md §13 (Landowners,
#  Petite Bourgeoisie) back it, and the debtors (Trade Unions, Rural Folk)
#  oppose it. The Dual Mandate takes them from Directed Credit, credit
#  steered toward jobs: the labour and populist ideologies back it, the
#  market-liberal ones oppose it. Each act's named sponsor is the issue's:
#  Industrialists for inflation targeting, Trade Unions for the dual
#  mandate.
# ======================================================================

# Inflation Targeting Act - the bank answers for prices alone (mandate 4).
amendment_cb_inflation_targeting = {
	parent = law_gold_standard

	allowed_laws = {
		law_central_bank_independence
	}

	# The mandates exist only under the full monetary system.
	possible = {
		te_mon_full_system = yes
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_industrialists
			AND = {
				exists = scope:approval
				scope:approval > 0
			}
		}
	}

	# Only a powerful opponent in government repeals it: each repeal is a
	# mandate change lenders price, and te_mandate.1 then asks again.
	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			is_powerful = yes
			law_stance = {
				law = law_type:law_gold_standard
				value < neutral
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}

# Civilian Oversight - civilian review boards hear complaints against the
# police, public or private, and can discipline officers and licensed
# providers (New York's Civilian Complaint Review Board, 1953 and 1966). It
# modifies a police force rather than replacing one, so it is an amendment,
# not a law. Distinct from Guaranteed Liberties (what home affairs may
# investigate), Dedicated Police (a professional force) and Restorative
# Justice (what follows a conviction): this is who the street-level force
# answers to. Militarized Police is not an allowed law: a force built to carry
# out the government's will does not answer to the public.
amendment_civilian_police_oversight = {
	parent = law_guaranteed_liberties

	allowed_laws = {

# ======================================================================
#  INHERITANCE
#  Spec: docs/superpowers/specs/2026-10-08-inheritance-laws-design.md §2
#  No inheritance law stands for taxing estates, so each amendment takes
#  its interest-group stances from the law elsewhere whose supporters
#  backed it in history (vanilla does this for 18 of its 67). Most work
#  through the wealth-concentration score: its target reads them
#  (te_inh_concentration_target, te_inheritance_values.txt), so their
#  _desc loc is where that effect is told. Ways in: the debate events
#  inheritance_events.10-.13 and the petitions amendment_petition_events
#  .17-.20. A law's amendments leave with it; the debate events offer to
#  carry one over to the incoming law.

# Estate Duties - a progressive duty on estates: Britain 1894, the US
# federal estate tax 1916, both with war or naval spending behind them.
# No standing revenue: it lowers the concentration target by 25. Estate
# taxes raise about 0.5% of OECD tax revenue today, and the tax-code spec
# keeps the dividend tax from being relabelled one. Not on a law that has
# the Birthright Endowment, which levies the duty itself.
amendment_estate_duties = {
	parent = law_graduated_taxation

	allowed_laws = {
		law_primogeniture
		law_partible
		law_equal_inheritance
		law_free_testation
	}

	possible = {
		has_technology_researched = political_agitation
		NOT = {
			active_law:lawgroup_inheritance ?= {
				has_amendment = amendment_type:amendment_birthright_endowment
			}
		}
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_intelligentsia
			AND = {
				exists = scope:approval
				scope:approval > 0
			}
		}
	}

	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			law_stance = {
				law = law_type:law_graduated_taxation
				value < neutral
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}

# Birthright Endowment - Paine's Agrarian Justice (1797): a duty on estates
# pays every young adult a capital stake. It says nothing about who gets
# the untaxed rest, so it fits every law with private estates. It replaces
# Estate Duties instead of stacking on it, because the duty is what pays
# for it: one amendment both levies the duty and pays the stake, and
# repealing it removes both. Two would break when the duty was repealed:
# `possible` is read only when an amendment is added, so an Endowment that
# required the duty would outlive it. Lowers the concentration target by
# 30. Parent Old Age Pension: Paine paired the stake at 21 with a pension
# from 50. An Endowment paid from general revenue (Britain's Child Trust
# Fund, 2005) is a welfare programme, not inheritance policy.
amendment_birthright_endowment = {
	parent = law_old_age_pension

	allowed_laws = {
		law_primogeniture
		law_partible
		law_equal_inheritance
		law_free_testation
	}

	modifier = {
		state_pop_qualifications_mult = 0.1
		state_education_access_add = 0.05
		state_laborers_investment_pool_efficiency_mult = 0.1
		state_farmers_investment_pool_efficiency_mult = 0.1
		state_shopkeepers_investment_pool_efficiency_mult = 0.1
	}

	# Deliberately silent on Estate Duties: the swap
	# (te_inh_swap_duties_for_endowment) adds the Endowment before it
	# removes the duty, and te_inh_tidy_amendments drops a duty that ends
	# up beside it.
	possible = {
		has_technology_researched = universal_basic_income
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_trade_unions
			AND = {
				exists = scope:approval
				scope:approval > 0
			}
		}
	}

	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			law_stance = {
				law = law_type:law_old_age_pension
				value < neutral
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}
