# No-First-Strike Pledge - a formal commitment never to initiate use of
# nuclear weapons. Boosts diplomacy but limits military options.
# It is the legal codification of the deterrence doctrine's No First Use
# (nd_doctrine = 1, #430): while it is in force the doctrine is held there
# (nd_weekly_update; nd_init_posture for an arsenal built under it), and
# leaving No First Use repudiates it and strikes it from the law
# (nd_set_doctrine → nd_strike_nfu_amendment). No on_activate / on_deactivate:
# the weekly binding covers every way in, and a hook on removal would loop.
#
# An AI may repeal it only under a militarist regime (#448). Before that it
# was permanent for an AI: the weekly binding holds the doctrine, the AI
# posture review leaves doctrine alone while the amendment is on file, and
# nothing else takes it off. ai_will_revoke is a permission, not an order
# (vanilla's Austrian amendments use it to stop one ruler revoking), so a
# militarist AI may repeal it, and leaving No First Use afterwards still pays
# the repudiation.
amendment_no_first_strike = {
	parent = law_limited_war

	allowed_laws = {
		law_humanitarian_regulations
		law_limited_war
	}

	modifier = {
		country_diplomatic_reputation_add = 5
	}

	# No nuclear weapons in this campaign, nothing to pledge away. Amendments
	# have no `visible`; `possible` is the only gate.
	possible = {
		has_game_rule = nuclear_weapons_enabled
		has_technology_researched = nuclear_weapons
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_trade_unions
		}
	}

	ai_will_revoke = {
		nd_regime_is_militarist = yes
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}

# Spoils System - ordinary administrative posts go to the governing
# faction's supporters ("to the victor belong the spoils"). It fits the 1836
# start: civil-service reform came in 1883. Attaches to the two vanilla laws
# that fill posts by appointment or election; Hereditary posts are not the
# government's to give, and the mod's bureaucracy laws staff theirs with
# contractors, delegates or machines. Modelled on vanilla's
# amendment_preserved_bureaucratic_caste (sponsor +15% political strength).
# Distinct from decree_political_patronage, a state decree that doubles the
# state's pop political strength.
# Spec: docs/superpowers/specs/2026-10-01-bureaucracy-laws-design.md
amendment_spoils_system = {
	parent = law_appointed_bureaucrats

	allowed_laws = {
		law_appointed_bureaucrats
		law_elected_bureaucrats
	}

	# Loyalty, not competence, fills the tax offices.
	modifier = {
		state_tax_capacity_mult = -0.1
	}

	sponsor_modifier = {
		interest_group_pol_str_mult = 0.15
	}

	possible = {
		always = yes

# delegates the whole force and so is not an allowed law. Its parent is

# providers (New York's Civilian Complaint Review Board, 1953 and 1966). It

# out the government's will does not answer to the public.
