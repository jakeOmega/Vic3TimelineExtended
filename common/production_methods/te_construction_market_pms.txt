# Construction-market subsystem — the Construction Site's production methods,
# and the direct-construction twins free_market_construction_rule swaps in.
#
# te_construction_market_site is the engine's construction camp
# (common/defines/te_construction_market_defines.txt): its inputs are bought
# only for the construction points actually spent, by the treasury or the
# investment pool according to which queue spends them.
#
# The game rule picks which methods can exist, through vanilla's rule-flag PM
# lock (common/game_rules/extra_game_rules.txt, free_market_construction_rule):
# `disable_<pm>` makes a method impossible to activate, `force_<pm>` makes it
# the only one its group can hold. No script switches methods.
#
#   rule enabled (market, the default)       rule disabled (direct)
#   ---------------------------------------  --------------------------------------
#   pm_te_construction_market_base          pm_te_direct_construction_<tier>
#     1 construction good -> 1 point          a Construction Sector tier's goods
#     (script places and sizes the sites)     -> its construction as points; the
#                                             site is the buildable sector
#   pm_maintenance (54 building types)      pm_no_maintenance (forced)
#   pm_disney_world, pm_generic_industrial_ pm_<same>_direct (forced): the same
#     city, pm_generic_monument_to_industry   recipe without the construction good
#   pm_principle_engineering_and_logistics  pm_principle_engineering_and_logistics
#     (+ its no-effect fallback)              _direct: points instead of the good
#
# Every method that a setting disables is hidden when unavailable, so each game
# shows only its own family.
#
# pm_te_direct_construction_<tier> copies the Construction Sector tier of the
# same name in extra_pms.txt: the same goods inputs, employment, state
# construction bonus and mortality, and its construction-good output as
# country_construction_add. When a sector tier changes, change its twin too.
# A point therefore costs what a unit of the construction good costs to make,
# as vanilla's sectors price a point (pm_wooden_buildings: 2 points for 25
# fabric + 75 wood, 1,000 per point at base prices; here 1 point for 10 fabric
# + 40 wood, the same 1,000).

pm_te_construction_market_base = {
	texture = "gfx/interface/icons/production_method_icons/wooden_buildings.dds"

	is_default = yes
	is_hidden_when_unavailable = yes

	country_modifiers = {
		workforce_scaled = {
			country_construction_add = 1
		}
	}

	building_modifiers = {
		workforce_scaled = {
			goods_input_construction_add = 1
		}

		level_scaled = {
			building_employment_laborers_add = 10
		}
	}
}
