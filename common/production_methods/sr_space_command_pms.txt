# ============================================================================
# SOLAR SYSTEM COLONIZATION — Space Command's methods (fleets design § 5)
# ============================================================================
# pmg_space_command_yards (sr_space_command_pmgs.txt). Each method carries the
# fleet strength a level builds a month as a workforce-scaled state modifier,
# state_sr_ship_construction_add (script-only: the engine consumes nothing; it
# still renders in the method's tooltip and the state's modifiers), which
# sr_sc_rate sums over the states the country controls. One field, no
# building-scoped copy, which the game would list a second time.
#
#   method              gate                              strength/level/mo  Launch Cap./wk  Adv. Mat./wk  electricity/wk
#   Orbital Shipyards   space_militarization              0.5                500             10            -
#   Fusion Yards        fusion_power                      1.0                600             15            -
#   Torch Yards         compact_fusion_reactors           1.5                600             15            200
#   Antimatter Yards    antimatter_production             2.0                700             20            400
#
# Every method also carries one identical maintenance line, unscaled (so not
# multiplied by levels), of six goods no method otherwise uses: 20 Electronic
# Components, 20 Wired and 20 Wireless Telecommunication Gear, 10 Software, 20
# Robotics and 40 Chemicals a week, about 8,400 at base prices, run at the
# building's occupancy as any goods line is. The fleets' upkeep multiplies only
# those (sr_fleet_upkeep_goods, Space Race UI pass design § 10), so it follows
# neither the method nor Space Command's levels: nothing to gain by switching
# down or downsizing once the fleet is built.
#
# Antimatter Yards needs only its technology: a method can't require another
# building (unlocking_production_methods reads the same building's other
# groups), so the Antimatter Containment Facility gates the drive
# (sr_launch_has_antimatter), not the yards. Every level employs 500 officers,
# 2,000 engineers and 2,500 machinists.
# ============================================================================

pm_sr_space_command_orbital_shipyards = {
	texture = "gfx/interface/icons/production_method_icons/cat_rocket_gold_p0.dds"

	unlocking_technologies = {
		space_militarization
	}

	building_modifiers = {
		workforce_scaled = {
			goods_input_launch_capacity_add = 500
			goods_input_advanced_materials_add = 10
		}

		level_scaled = {
			building_employment_officers_add = 500
			building_employment_engineers_add = 2000
			building_employment_machinists_add = 2500
		}

		unscaled = {
			goods_input_electronic_components_add = 20
			goods_input_telephones_add = 20
			goods_input_radios_add = 20
			goods_input_digital_assets_add = 10
			goods_input_robotics_add = 20
			goods_input_fertilizer_add = 40
		}
	}

	state_modifiers = {
		workforce_scaled = {
			state_sr_ship_construction_add = 0.5
		}
	}
}
