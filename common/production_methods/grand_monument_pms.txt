#==================================================
# Grand Monument dedications
#==================================================
# THE RATCHET -- why every PM below lists itself in unlocking_production_methods.
#
# The Clausewitz engine has no "lock this PM once chosen" feature: PM gating is
# limited to techs / laws / principles / company categories / identities / other PMs,
# and none of those can differentiate two monuments in two different states. So the
# lock is built out of unlocking_production_methods, which is an OR over its list:
#
#   pm_monument_undedicated  unlocks on { itself }
#   every dedication         unlocks on { pm_monument_undedicated, itself }
#
# Fresh build  -> Undedicated is active (is_default), so all dedications are available.
# After a pick -> only the chosen PM's own self-reference is satisfied; Undedicated and
#                 the seven siblings are all unreachable. The choice is permanent, and
#                 re-flavouring means demolishing and rebuilding.
#
# Every PM carries is_hidden_when_unavailable so the group shows exactly one row at a
# time rather than a column of greyed-out dead ends.
#
# Do NOT add replacement_if_valid to any of these -- it auto-swaps PMs and would
# defeat the ratchet.
#
# The dedication is normally set by the ceremony event (monument_events.2) via
# activate_production_method; the manual building-panel path works too.
#
#--------------------------------------------------
# SCALING -- why national effects are unscaled and local ones are level_scaled.
#
# This building is repeatable in EVERY state, so anything country-wide stacks across
# every monument in the country. A level_scaled country modifier would therefore be
# unbounded: 20 monuments x 20 levels x -0.01 hits the engine's -100% clamp and the
# effect becomes free. So:
#
#   state_* and building_*_throughput_add  -> level_scaled (genuinely local to the state)
#   interest_group_* and country_*         -> unscaled     (flat per monument, bounded
#                                              by how many states you own)
#
# Economics, identical across every dedication so the choice is about flavour only:
#   goods_output_tourism_add = 1   level_scaled  (100/mo per level at base price)
#   maintenance comes from pmg_maintenance       (50/mo per level)
#   building_tourism_industry_throughput_add = 0.01 level_scaled, i.e. +1%/level
#
# Goods I/O is level_scaled rather than workforce_scaled on purpose: employment here
# is unscaled (flat, level-independent), so a workforce ratio measured against a
# constant headcount would behave unlike every other building -- and would collapse to
# zero output if the unscaled employment block failed to register. level_scaled goods
# I/O is already proven by pm_maintenance.

# Entry point and fallback: a monument nobody has yet decided the meaning of.
# Still a tourist draw, just a generic one. Also what a player is left with if
# they dismiss the dedication ceremony event.
pm_monument_undedicated = {
	texture = "gfx/interface/icons/production_method_icons/cat_star_gold_p0.dds"
	is_hidden_when_unavailable = yes
	is_default = yes

	unlocking_production_methods = {
		pm_monument_undedicated
	}

	state_modifiers = {
		level_scaled = {
			building_tourism_industry_throughput_add = 0.01
		}
	}

	building_modifiers = {
		level_scaled = {
			goods_output_tourism_add = 1
		}

		# Unscaled: a monument needs a caretaker staff, not a workforce that
		# grows with its height. Flat regardless of level.
		unscaled = {
			building_employment_laborers_add = 200
			building_employment_clerks_add = 50
		}
	}
}
