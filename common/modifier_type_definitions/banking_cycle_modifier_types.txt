# ============================================================================
# Monetary policy — the two risk-premium tiers (monetary_policy_design.md §7.1)
# ============================================================================
# `rate paid = policy rate + risk premium`, and the premium is computed in two
# tiers rather than one floored sum: a mature great power's standing sits well
# below the floor, so a single floor would silently swallow a panic, every tool
# and the whole debt-load term while their tooltips still promised "+2.0%".
#
#   te_premium_structural = max( floor , 100 x modifier:country_credit_standing_add )
#   te_premium_cyclical   = 100 x modifier:country_risk_premium_add + computed terms
#
# Both carry vanilla's interest scale (0.01 = 1pp), so there is one unit
# convention across the mod and te_monetary_monthly_update reads them back as
# 100 x modifier:X. JE-scope grants still feed the country-scope `modifier:`
# read — the same path banking_cycle_advance_variables already uses for
# country_finance_momentum_monthly_add.

# Structural — what the country *is*: capital-market access, rank, finance
# techs, national-bank institution level, laissez-faire, gold / central-bank
# credibility, company prosperity. Floored at 0.5pp (0.25pp under central bank
# independence); standing beyond the floor is wasted by design.
country_credit_standing_add = {
	color = bad
	percent = yes
	decimals = 1
	script_only = yes
	game_data = { ai_value = 0 }
}

# Cyclical — what is *happening*: cycle phases, banking tools, crash
# interventions, event outcomes, bankruptcy, debt load, monetisation. Added
# *after* the structural floor, so a crisis always bites.
country_risk_premium_add = {
	color = bad
	percent = yes
	decimals = 1
	script_only = yes
	game_data = { ai_value = 0 }
}

# ============================================================================
# Monetary policy — the two inflation-pressure channels (§9.1, §9.4, §11)
# ============================================================================
# "Pressure" is a LEVEL, not an increment: how far above expectations inflation
# settles while the source lasts. Step 6 of te_monetary_monthly_update sums both
# types into the §9.1 pressure term, read back the same way as the premium tiers
# — `100 x modifier:X`, because both carry vanilla's interest scale
# (0.01 = 1pp), so one unit convention covers the whole monetary system.
#
# TWO TYPES, NOT ONE. §9.4 put wage pressure on country_inflation_pressure_add,
# which §11 also uses for devaluation and event modifiers — and then the §9.4
# real-wage dividend could not tell a labour law from a currency crisis, and
# would have paid a country for debasing its money. So labour and welfare laws
# grant country_wage_pressure_add, everything else grants
# country_inflation_pressure_add, and only the wage half backs the dividend.
#
# BOTH ARE DELIBERATELY VISIBLE. Unlike the hidden stance gap, these are what
# the player *chose* — a labour law, a devaluation, a monetisation step — so
# each law and event modifier states its own contribution on its own tooltip.
# The pressure SUM is still hidden state (ruling P7): it carries the stance term
# and the random walk, and nothing may print it.

# Everything that is not a wage: currency devaluation and stabilisation event
# modifiers (§11), crisis outcomes, and any later source of standing price
# pressure that is not a labour cost.
country_inflation_pressure_add = {
	color = bad
	percent = yes
	decimals = 1
	script_only = yes
	game_data = { ai_value = 0 }
}

# Labour and welfare laws only (§9.4). Kept apart from the type above so the
# real-wage dividend can pay for wage-led demand without paying for inflation
# created any other way. Negative under laws that suppress wages
# (no workers' rights, combination acts, anti-strike laws).
country_wage_pressure_add = {
	color = bad
	percent = yes
	decimals = 1
	script_only = yes
	game_data = { ai_value = 0 }
}

# ============================================================================
# Monetary policy — the regime rules, carried on the laws that grant them
# ============================================================================
# Each of these replaces a rule that used to be a law check buried in script and
# a sentence of localization restating it. Putting the number on the law means
# the law's own tooltip states it, the engine renders the breakdown, and a later
# law, technology or principle can move the same dial without anyone editing
# te_monetary_effects.txt.
#
# Consumed ONLY by the named script values in
# common/script_values/te_monetary_script_values.txt — te_mon_drift_step,
# te_mon_target_min, te_mon_standing_floor and te_mon_forecast_error_bound.
# Nothing reads `modifier:` on these anywhere else.

# How much faster than the baseline third of a point a month the policy rate

# to hold transmits a rate change in weeks rather than seasons.

	game_data = { ai_value = 0 }
