# Flag aliases for the mod's major-formation India / Indonesia / China tags.
# BHA reuses vanilla BHT artwork; NUS reuses vanilla IDN artwork; ZHO restates
# vanilla CHI's definitions, so a formed China flies the Qing, republican,
# communist or other flag its government calls for. These matter because if no
# flag_definition exists for a tag the engine falls back to looking up the tag
# string as a COA_KEY (which doesn't exist for BHA / NUS / ZHO and would
# produce a random flag). The other formables' flags are coats of arms named
# after their tags: common/coat_of_arms/coat_of_arms/te_formable_countries.txt.

BHA = {
	flag_definition = {
		coa = BHT
		subject_canton = BHT
		priority = 1
	}
}

	flag_definition = {
		coa = IDN
		subject_canton = IDN
		priority = 1
	}
}
