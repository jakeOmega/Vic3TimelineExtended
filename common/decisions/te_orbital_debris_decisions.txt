# Switches the Space Program's debris removal group to the best method the
# country has (te_debris_activate_best_removal): Fusion Sweepers with
# Muon-Catalyzed Fusion Reactors, else the Laser Broom with Directed Energy
# Weapons, else Active Debris Removal (Reusable Rocketry). Shown while that
# method is not the one running (te_debris_best_removal_off), so it also
# upgrades a running method when a better one arrives. A player can still
# switch by hand in the building panel, and switch removal off there.
#
# Why a decision: every removal method only costs goods, and under the
# default most_profitable the AI keeps None, the least loss; what
# most_productive (the group's setting) does with cost-only methods is
# unobserved. The mission decisions (extra_decisions.txt) are the precedent.
#
# ai_chance (spec § 6.4): only an AI that uses orbit, with the index at
# Cluttered (25) or more, more readily at Hazardous (50), not in default, and
# whose weekly surplus covers the method's bill (net_fixed_income, the
# affordability reader the mission decisions use). ai_chance is AI-only by
# nature, so nothing here is an is_ai rule.
# ============================================================================
te_decision_debris_removal = {
	is_shown = {
		sr_space_race_on = yes
		has_technology_researched = reusable_rocketry
		sr_has_space_program = yes
		te_debris_best_removal_off = yes
	}

	# Every vanilla decision carries a possible; this one has nothing to wait
	# for beyond what shows it.
	possible = {
		sr_has_space_program = yes
	}

	when_taken = {
		custom_tooltip = {
			text = te_decision_debris_removal_tt
			te_debris_activate_best_removal = yes
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				te_debris_index >= te_debris_tier_cluttered
				te_debris_uses_orbit = yes
				in_default = no
				net_fixed_income > te_debris_best_removal_bill
			}
			add = 50
			if = {
				limit = { te_debris_index >= te_debris_tier_hazardous }
				add = 50
			}
		}
	}
}
