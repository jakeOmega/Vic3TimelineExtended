# ============================================================================
# Plan a New Capital (docs/superpowers/specs/2026-10-08-demographics-design.md §5.1's lever; mod_systems.md
# § Demographics, "The urban pattern's effects and the Planned Capital decision")
# Washington, Canberra, Brasília, Abuja and Astana: a seat of government built away from the country's great city. The
# decision opens a choice of up to three sites (te_demog_capital.1); the chosen site becomes the capital at once, the
# treasury pays 3% of a year's GDP, the five years of building cost 100 authority, and the new capital grows under a
# development modifier for 20 years. Another waits 50 years.
#
# Full only, like the census's other effects (the release precondition: nothing changes outside Full). It reads the
# urban pattern, which only the census computes, and it moves a capital, which changes a game.
#
# Gates. The pattern is Primate or Dominant (25% primacy or more, where the pattern's effects begin). The base game's
# own rules for moving a capital (its Move Capital map interaction): at peace, in no diplomatic play, not moved
# recently, and the site incorporated and a homeland of a primary culture (te_demog_pc_candidate). Steel-Frame
# Buildings (era 3: Canberra was laid out in 1912), shown from its prerequisite Modern Sewerage. The authority for the
# building. The money is not gated: a treasury short of it borrows, as for any other expense. One site event at a time
# (te_dg_pc_pending, set here and cleared by every option); each site's option checks the site, peace and the cooldown
# again when it is chosen, and the event lapses to Not Now.
#
# set_capital takes a state region's key, never a scope (gen_planned_capital.py's header), so the move dispatches on the
# site's region (te_demog_pc_set_capital, generated). Whether set_capital also brings the base game's Moved Capital
# Recently penalty is unverified: the planned_capital_moved self-check logs it (docs/testing/demographics-playtest.md).
#
# ai_chance: te_demog_pc_ai_wants, the same conditions for every country (no is_ai): a Primate pattern at 0.7 strength
# or more (42.5% primacy) in a country of eight census states or more whose largest city is under Migration Crowding,
# out of default, with the money in the treasury and authority to spare. The weight is the siblings' 100 (the Strategic
# Reserve and the space missions): ai_chance weighs a decision against the others possible, so the rule alone decides.
# ============================================================================
te_decision_plan_new_capital = {
	is_shown = {
		te_demog_effects_run = yes
		has_variable = te_dg_urban_label
		has_technology_researched = modern_sewerage
		is_revolutionary = no
		NOT = { is_country_type = decentralized }
	}

	possible = {
		has_technology_researched = steel_frame_buildings
		custom_tooltip = {
			text = te_demog_pc_tt_pattern
			var:te_dg_urban_label <= 1
		}
		custom_tooltip = {
			text = te_demog_pc_tt_peace
			is_at_war = no
			is_diplomatic_play_committed_participant = no
		}
		custom_tooltip = {
			text = te_demog_pc_tt_moved_recently
			NOT = { has_modifier = moved_capital_recently }
		}
		custom_tooltip = {
			text = te_demog_pc_tt_cooldown
			te_demog_pc_years_since >= te_demog_k_pc_cooldown_years
		}
		custom_tooltip = {
			text = te_demog_pc_tt_authority
			authority >= te_demog_k_pc_authority
		}
		custom_tooltip = {
			text = te_demog_pc_tt_site
			any_scope_state = { te_demog_pc_candidate = yes }
		}
		custom_tooltip = {
			text = te_demog_pc_tt_pending
			NOT = { has_variable = te_dg_pc_pending }
		}
	}

	# te_dg_pc_pending: the site event is open. Every option of te_demog_capital.1 clears it, and the event lapses to Not
	# Now long before the year runs out, so the decision can't open a second event, and pay twice, while one is open.
	when_taken = {
		custom_tooltip = te_demog_pc_tt_choose_site
		hidden_effect = {
			set_variable = { name = te_dg_pc_pending value = yes days = 365 }
		}
		trigger_event = { id = te_demog_capital.1 popup = yes }
	}

	ai_chance = {
		value = 0
		if = {
			limit = { te_demog_pc_ai_wants = yes }
			add = 100
		}
	}
}
