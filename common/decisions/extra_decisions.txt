# Available to liberal-democratic colonial empires holding the Neocolonialism law.
# Costs authority: mandate_system_transition_modifier, -100 easing over five

# ============================================================================
mandate_system_decision = {
	is_shown = {
		has_journal_entry = je_colonial_empire
		has_technology_researched = decolonization
		# One-shot (docs/archive/decolonization_review.md § 6): the conversion
		# is permanent, and without this a retake refreshed the 12-month
		# stability boost at will. Kept in a variable, as the Imperial Federation
		# Acts keep imperial_federation_taken, because a revolution's winner
		# inherits variables but no modifiers; the modifier check covers saves
		# from before the variable.
		NOT = { has_variable = mandate_system_taken }
		NOT = { has_modifier = mandate_system_modifier }
	}

	possible = {
		# Decisions are country-scoped — scope:journal_entry is only auto-saved
		# inside JE-internal blocks. Use the je:<key> scope link instead (same
		# pattern as je_colonial_empire's own monthly pulse).
		je:je_colonial_empire = {
			"scripted_bar_progress(colonial_stability_bar)" >= 65
		}
		# Hard requirement: neocolonialism law (the mod-added "informal empire"
		# Colonial Affairs law gated on the decolonization tech)
		has_law = law_type:law_neocolonialism
		# Broad-franchise DP — selling this requires legitimacy
		OR = {
			has_law = law_type:law_universal_suffrage
			has_law = law_type:law_census_voting
			has_law = law_type:law_wealth_voting
		}

		NOT = { has_law = law_type:law_outlawed_dissent }

			name = mandate_system_modifier

		}

		}

		value = 50
	}

# ============================================================================
# Establish a Strategic Reserve (strategic_reserve_system.md §2, §7)
# Places the Strategic Reserve Hub in the capital at once, for free; the
# reserve's expense is what it buys and stores. The building's own
# construction path is unchanged.
#
# Why a decision: in an AI-only observer run to 2015, after the hub's ai_value
# gained +10000 above st_res_ai_hub_min_gdp (PR #705), 1 of 157 countries built
# one. The construction AI does not pick it, and je_strategic_reserve cannot
# host a build button, because its possible needs a hub to exist.
#
# It is also how a reserve that lost its hub (to the rebels of a civil war,
# or to a foreign power) gets one back: the journal entry stays open without
# a hub and keeps its stock, unless the hub was captured.
#
# The one-per-country rule is the hub's own (is_shown: no hub anywhere).
# create_building checks the unlocking technology since 1.13.9; whether it
# checks the building's possible is unverified, so st_res_hub_relocating is
# raised around the call as st_res_relocate_hub_to_capital_effect does, and
# the hub's one-per-country guard accepts it.
#
# ai_chance mirrors the hub's ai_value shape (common/buildings/strategic_reserve.txt):
# great and major powers, more at peace, and any country above
# st_res_ai_hub_min_gdp whether at war or not. A smaller country gets 0, as
# it gets no ai_value.
# ============================================================================
te_decision_establish_strategic_reserve = {
	is_shown = {
		has_technology_researched = logistics
		NOT = { is_country_type = decentralized }
		# The rebels of a running revolution found no reserve of their own.
		# The journal entry's activation would give them a reserve's variables
		# at zero stock, and if they won, those zeros would beat the loser's
		# stockpile in the variable merge: the winner continues the nation
		# (#460-#465). Same rule as Suborbital Flight's is_shown_when_inactive
		# (je_space_race.txt); te_decision_found_space_program has it too.
		is_revolutionary = no
		NOT = {
			any_scope_building = { is_building_type = building_strategic_reserve_hub }
		}
	}

	possible = {
		custom_tooltip = {
			text = te_decision_tt_has_capital
			exists = capital
		}
	}

	when_taken = {
		custom_tooltip = {
			text = te_decision_tt_create_strategic_reserve_hub
			set_variable = { name = st_res_hub_relocating value = 1 }
			capital = {
				create_building = {
					building = building_strategic_reserve_hub
					level = 1
				}
				# The holder stamp the weekly pulse would write. Without it,
				# a hub lost in its first week counts as captured: a civil
				# war breaking out that week would demolish it. A stamp on a
				# capital the hub failed to reach is inert, as
				# st_res_on_state_owner_change acts only where a hub stands.
				set_variable = { name = st_res_hub_holder value = ROOT }
			}
			remove_variable = st_res_hub_relocating
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { is_great_or_major_power = yes }
			add = 50
			if = {
				limit = { is_at_war = no }
				add = 50
			}
		}
		if = {
			limit = { gdp > st_res_ai_hub_min_gdp }
			add = 100
		}
	}
}

# ============================================================================
# Begin the next space mission (mod_systems.md § Space Race)
# One decision per mission. Each switches the Space Program to that mission's
# production method; a player can still switch by hand in the building panel.
#
# Why: in a 2073 player game every AI had every technology, yet none got
# beyond Orbital Flight. Every later milestone needs a costlier mission, and
# under most_profitable the AI kept the least-loss one, Earth Orbit
# (pmg_base_building_space_program is now most_productive too).
#
# Each mission's production method carries every lower mission's
# country_sr_*_program_bool, so switching up never stops a running milestone.
# The mission -> milestone map (je_space_race.txt `possible` blocks):
#   pm_earth_orbit              Suborbital Flight, Orbital Flight
#   pm_moon_mission             Moon Landing, Deep-Space Probe
#   pm_mars_mission             Moon Base, Mars Landing
#   pm_solar_colonization       Solar System Colonization, stages 1-2 (and
#                               every stage once a colony is held)
#   pm_deep_space_exploration   Solar System Colonization, stages 3-5, for a
#                               country holding no colony
#   pm_interstellar_mission     Interstellar Probe
#
# A mission is offered once its technology is researched, while the programme
# runs the mission below it (Mars or above for the last three), and becomes
# possible for a great or major power (every milestone needs the rank) once
# the milestones the missions below it serve are complete:
# Moon after Orbital Flight; Mars after the Moon Landing and the Deep-Space
# Probe; Solar Colonization and Deep Space after the Moon Base and the Mars
# Landing. The Interstellar Mission waits only for the Interstellar Probe's own
# prerequisites (Deep-Space Probe, Mars Landing), from Mars, Solar
# Colonization or Deep Space: colonization never "completes" for one country
# while worlds are left, and the Interstellar Mission carries its bools.
# Deep Space Exploration is offered only to a country it would help: inner
# worlds all claimed (sr_colonization_stage >= 3) and no colony held, since a
# colony holder keeps colonizing on Solar Colonization.
#
# ai_chance: the same gates as founding, against the increase in the weekly
# bill (sr_mission_bill_increase_<mission>, space_race_values.txt).
# test_space_program_decisions.py holds the techs, bools and milestones to the
# production methods and the journal entries.
# ============================================================================
te_decision_space_mission_moon = {
	is_shown = {
		has_game_rule = space_race_enabled
		has_technology_researched = space_exploration
		sr_has_space_program = yes
		modifier:country_sr_earth_orbit_program_bool = yes
		NOT = { modifier:country_sr_moon_mission_program_bool = yes }
	}

	possible = {
		# Every milestone's possible needs the rank (je_space_race.txt; only
		# a colony holder's Solar System Colonization runs without it), so
		# below it a mission would be paid for and never used.
		custom_tooltip = {
			text = sr_unlock_rank_tt
			is_great_or_major_power = yes
		}
		custom_tooltip = {
			text = te_decision_tt_completed_orbital
			has_variable = sr_completed_orbital
		}
	}

	when_taken = {
		custom_tooltip = {
			text = te_decision_tt_begin_moon_mission
			activate_production_method = {
				building_type = building_space_program
				production_method = pm_moon_mission
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				sr_ai_should_participate = yes
				in_default = no
				net_fixed_income > sr_mission_bill_increase_moon
			}
			add = 100
		}
	}
}

# ============================================================================
# Restructure the Public Debt (#800, mod_systems.md § Banking Cycle)
# A country whose debt the markets price (banking_sovereign_debt_is_priced,
# where the debt-load premium starts) or that is in default opens talks with
# its creditors, and chooses the haircut there (banking_debt_events.1): 10%,
# 25%, or 50% in default only. Before #800 the only deliberate relief was the
# Sovereign Debt Scare's option C, an event draw. The talks may be broken off;
# opening them again waits a year (banking_restructuring_talks), and a deal
# waits five or ten years for the next (banking_debt_restructured).
#
# ai_chance: banking_restructuring_ai_value, in default or with the debt at
# three quarters of the credit limit and still growing. The haircut itself is
# the talks' option weights.
# ============================================================================
te_decision_restructure_debt = {
	is_shown = {
		banking_restructuring_shown = yes
	}

	possible = {
		banking_restructuring_possible = yes
	}

	when_taken = {
		custom_tooltip = {
			text = te_decision_tt_restructure_debt
			set_variable = { name = banking_restructuring_talks value = yes days = 365 }
		}
		trigger_event = { id = banking_debt_events.1 popup = yes }
	}

	ai_chance = {
		value = banking_restructuring_ai_value
	}
}
