adopt_local_culture = {
	is_shown = {
		NOT = {
			any_scope_state = {
				is_homeland_of_country_cultures = ROOT
			}
		}
	}

	possible = {
		is_subject = no
		has_technology_researched = decolonization
		capital = {
			any_scope_culture = {
				has_homeland = PREV
			}
		}
	}

	when_taken = {
		every_primary_culture = {
			ROOT = {
				remove_primary_culture = PREV
			}
		}
		capital  = {
			every_scope_culture = {
				limit = {
					has_homeland = PREV
				}
				ROOT = {
					add_primary_culture = PREV
				}
			}
		}
	}

	ai_chance = {
		value = 100
	}
}

# ============================================================================
# Imperial Federation Act — Iron Fist Path (capstone for Iron-Fist hold strategy)
# Available only when:
#   - colonial_solidified_months >= 36 (3 years sustained at ≥90 bar)
#   - country has explicitly committed to formal empire (Colonial Affairs)
#   - laws across 4 lawgroups all match the Iron-Fist profile
#   - country is GP and has ≥4 colonial states
# Effect: completes je_colonial_empire via flag, triggers victory event 300,
# applies permanent imperial_federation_modifier.
# ============================================================================
imperial_federation_act_iron_fist = {
	is_shown = {
		has_journal_entry = je_colonial_empire
	}

	possible = {
		# Sustained Solidified band
		var:colonial_solidified_months >= 36
		# Power and scale
		country_rank >= rank_value:great_power
		colonial_state_count >= 4
		# Colonial Affairs commitment
		OR = {
			has_law = law_type:law_colonial_exploitation
			has_law = law_type:law_colonial_resettlement
		}
		# Iron-Fist Minority Rights
		OR = {
			has_law = law_type:law_minority_rights_violent_hostility
			has_law = law_type:law_minority_rights_ghettoization
			has_law = law_type:law_minority_rights_discrimination
			has_law = law_type:law_minority_rights_cultural_assimilation
		}
		# Hard-line Citizenship
		OR = {
			has_law = law_type:law_national_supremacy
			has_law = law_type:law_ethnostate
			has_law = law_type:law_racial_segregation
		}
		# Speech suppression
		OR = {
			has_law = law_type:law_outlawed_dissent
			has_law = law_type:law_censorship
		}
		# Authoritarian DP
		OR = {
			has_law = law_type:law_autocracy
			has_law = law_type:law_oligarchy
			has_law = law_type:law_single_party_state
			has_law = law_type:law_landed_voting
		}
		# Costs (gated)
		authority > 300
		prestige >= 100
	}

	when_taken = {
		add_modifier = {
			name = imperial_federation_modifier
			days = -1
		}
		set_variable = imperial_federation_taken
		set_variable = imperial_federation_iron_fist
	}

	ai_chance = {
		value = 100
	}
}

# ============================================================================
# Mandate System Decision — managed-decline option for the broad-franchise empire
# Converts the colonial JE into informal economic empire (mandate_system_modifier)
# while allowing the player to keep collecting some of the benefits.
# Available to liberal-democratic colonial empires holding the Neocolonialism law.
# Costs authority: mandate_system_transition_modifier, -100 easing over five
# years, the decaying shape peaceful_decolonization_modifier gives the Round
# Table and a unilateral release. (The original one-shot add_authority = -100
# was dropped in e1fe52ae: add_authority is not a Vic3 effect.)
# ============================================================================
mandate_system_decision = {
	is_shown = {
		has_journal_entry = je_colonial_empire
		has_technology_researched = decolonization
		# One-shot (docs/archive/decolonization_review.md § 6): the conversion
		# is permanent, and without this a retake refreshed the 12-month
		# stability boost at will. Kept in a variable, as the Imperial Federation
		# Acts keep imperial_federation_taken, because a revolution's winner
		# inherits variables but no modifiers; the modifier check covers saves
		# from before the variable.
		NOT = { has_variable = mandate_system_taken }
		NOT = { has_modifier = mandate_system_modifier }
	}

	possible = {
		# Decisions are country-scoped — scope:journal_entry is only auto-saved
		# inside JE-internal blocks. Use the je:<key> scope link instead (same
		# pattern as je_colonial_empire's own monthly pulse).
		je:je_colonial_empire = {
			"scripted_bar_progress(colonial_stability_bar)" >= 65
		}
		# Hard requirement: neocolonialism law (the mod-added "informal empire"
		# Colonial Affairs law gated on the decolonization tech)
		has_law = law_type:law_neocolonialism
		# Broad-franchise DP — selling this requires legitimacy
		OR = {
			has_law = law_type:law_universal_suffrage
			has_law = law_type:law_census_voting
			has_law = law_type:law_wealth_voting
		}
		# Cannot have outlawed dissent — selling negotiated transition needs press
		NOT = { has_law = law_type:law_outlawed_dissent }
		authority > 100
	}

	when_taken = {
		set_variable = mandate_system_taken
		add_modifier = {
			name = mandate_system_modifier
			days = -1
		}
		# The cost: selling the transition spends authority, easing over five years
		add_modifier = {
			name = mandate_system_transition_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		# Push bar up — the world recognizes the negotiated mode
		add_modifier = {
			name = colonial_stability_positive_event
			months = 12
		}
	}

	ai_chance = {
		value = 50
	}
}

# ============================================================================
# Resume the Nuclear Programme (the nuclear taboo, nuclear_taboo_effects.txt)
# A country that dismantled its arsenal (nd_taboo_renounced) may take the bomb
# up again. The breakout — a ledger entry against the taboo — is booked when
# it is armed again, whatever the route (nd_taboo_country_monthly).
# ============================================================================
nd_taboo_resume_programme = {
	is_shown = {
		has_modifier = nd_taboo_renounced
	}

	possible = {
		has_game_rule = nuclear_weapons_enabled
	}

	when_taken = {
		# A renouncer without the standing to run a programme (I2, final-review.md):
		# Resume still ends the renunciation and pays its price, but gains no
		# programme. Trigger-gated custom_tooltip, the form nuclear_crisis_effects.txt
		# uses for a conditional explanatory line (e.g. nd_crisis_open's
		# dispute-kind branches).
		if = {
			limit = { nd_taboo_has_programme_standing = no }
			custom_tooltip = { text = nd_taboo_tt_resume_no_standing }
		}
		nd_taboo_resume = yes
	}

	ai_chance = {
		value = 0
		if = {
			limit = { nd_taboo_ai_would_resume = yes }
			add = 100
		}
	}
}
