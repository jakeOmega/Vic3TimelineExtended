# rename reaches the panel. rt_law_effects is the hover on the law line: the
# law's flavor description followed by the same effect lines its enactment
# preview shows (on_enact in common/laws/resource_transition_laws.txt), so the
# rules read the same before and after enactment.

	type = country
	random_valid = no

	text = {
		trigger = { has_law = law_type:law_managed_fossil_phaseout }
		localization_key = law_managed_fossil_phaseout
	}
	text = {
		trigger = { has_law = law_type:law_fossil_expansion_moratorium }
		localization_key = law_fossil_expansion_moratorium

	text = {
		trigger = { always = yes }
		localization_key = law_unrestricted_extraction
	}
}

		localization_key = law_unrestricted_extraction_desc
