# ============================================================================
# Orbital debris — customizable localization (Global Commons § 6.1)
# ============================================================================
# te_debris_tier: the index's tier, for text only (spec § 6.1). Read by the
# Mission Control row (te_sr_debris_row, te_orbital_debris_widget.gui), its
# hover, and space_race_events.31. The thresholds are the script values
# te_debris_tier_cluttered / _hazardous and te_debris_kessler_threshold; the
# index reads globals only, so the branches are safe for any country.
# ============================================================================

te_debris_tier = {
	type = country
	random_valid = no

	text = {
		trigger = { te_debris_index >= te_debris_kessler_threshold }
		localization_key = te_debris_tier_kessler
	}
	text = {
		trigger = { te_debris_index >= te_debris_tier_hazardous }
		localization_key = te_debris_tier_hazardous
	}
	text = {
		trigger = { te_debris_index >= te_debris_tier_cluttered }
		localization_key = te_debris_tier_cluttered
	}
	text = {
		trigger = { always = yes }
		localization_key = te_debris_tier_clear
	}
}

# ============================================================================
# The Space Race tab's Orbit section (te_sr_sec_orbit, space_race_widget.gui):
# the state words of two rows, and the Moratorium row's hover. Every branch
# reads the country's own modifiers and buildings or globals, so they are safe
# for any country, and every target key is plain text: the hovers' numbers are
# in their outer keys (te_orbit_satellites_tt, on te_asat_disp_satellites).
# ============================================================================

# Our Satellites: whether a strike has disrupted our orbital network
# (te_orbital_network_disrupted, te_asat_strike_effect), and whether we have
# anything a strike could hit (te_asat_has_orbital_assets).
te_asat_satellites_state = {
	type = country
	random_valid = no

	text = {
		trigger = { has_modifier = te_orbital_network_disrupted }
		localization_key = te_asat_satellites_disrupted
	}
	text = {
		trigger = { te_asat_has_orbital_assets = yes }
		localization_key = te_asat_satellites_intact
	}
	text = {
		trigger = { always = yes }
		localization_key = te_asat_satellites_none
	}
}

# The Moratorium on Destructive Anti-Satellite Tests: not adopted, or in force
# (UNODA), with us bound by it or not (te_asat_moratorium_party).
te_asat_moratorium_state = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_global_variable = un_agency_unoda } }
		localization_key = te_asat_moratorium_not_adopted
	}
	text = {
		trigger = { te_asat_moratorium_party = yes }
		localization_key = te_asat_moratorium_bound
	}
	text = {
		trigger = { always = yes }
		localization_key = te_asat_moratorium_not_bound
	}
}
