# The tab (Task 3). A row's stance on the law being enacted (te_pa_r_stance_enacting, -2 to 2; Task 1's
# te_pa_gen_stance_enacting, which can fall between whole points: the labor laws' middle steps weigh half).

	type = container
	random_valid = no
	text = {
		trigger = { var:te_pa_r_stance_enacting ?= 0 }
		localization_key = te_pa_stance_none
	}
	text = {
		trigger = {
			has_variable = te_pa_r_stance_enacting
			var:te_pa_r_stance_enacting <= -1.5
		}
		localization_key = te_pa_stance_against_strongly
	}
	text = {
		trigger = {
			has_variable = te_pa_r_stance_enacting
			var:te_pa_r_stance_enacting < 0
		}
		localization_key = te_pa_stance_against
	}
	text = {
		trigger = {
			has_variable = te_pa_r_stance_enacting
			var:te_pa_r_stance_enacting >= 1.5
		}
		localization_key = te_pa_stance_for_strongly
	}
	text = {
		trigger = {
			has_variable = te_pa_r_stance_enacting
			var:te_pa_r_stance_enacting > 0
		}
		localization_key = te_pa_stance_for
	}
	text = {
		localization_key = te_pa_stance_none
