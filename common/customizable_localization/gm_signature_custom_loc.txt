# Whether its signature counts (§1's table): upheld on its parent, nationally
# and locally; heritage, contested or not upheld on its parent, locally only;
# on another dedication, dormant.
gm_sig_status = {
	type = state
	random_valid = no

	text = {
		trigger = { gm_state_sig_counts = yes }
		localization_key = gm_sig_status_counts
	}
	text = {
		trigger = { gm_state_sig_on_parent = yes }
		localization_key = gm_sig_status_local
	}
	text = {
		localization_key = gm_sig_status_dormant
	}
}

# The label the row title, the state card's title print in place of the plain
# name (§3): with a signature, the name as a hover span whose tooltip says what
# it is a unique dedication of, what it adds and whether it counts. Read through
# `State` only (the GUI's rows and card); events print gm_dedication_name.
gm_dedication_label = {
	type = state
	random_valid = no

	text = {
		trigger = { has_variable = gm_signature }
		localization_key = gm_sig_label
	}
	text = {
		localization_key = gm_sig_label_plain
	}
}
