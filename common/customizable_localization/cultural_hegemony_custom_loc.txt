# ============================================================================
# Cultural Hegemony — branchy one-liners for the journal-entry widget
# ============================================================================
# Two readings that used to be a chain of `triggered_desc` entries on the journal
# entry: the country's influence tier (six branches) and the world hegemon's
# exported political model (fifteen). Both are read from the widget as
# [JournalEntry.GetCountry.GetCustom('<key>')].
#
# Neither block owns a threshold or a classification. The tier reads `ch_tier`,
# whose only derivation site is `ch_set_display_state`
# (common/scripted_effects/cultural_hegemony_effects.txt), where 2/5/10/15/25
# live. The exported model reads `global_var:ch_rank_1_ideology`, the rank-1
# country's `var:ch_model`, whose only derivation site is
# `ch_set_political_model` in the same file — the 1..15 codes below are that
# effect's (its header has the table; 13-15 are the newer models).
# ============================================================================

ch_tier_text = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_variable = ch_tier } }
		localization_key = ch_tier_negligible
	}
	text = {
		trigger = { var:ch_tier >= 5 }
		localization_key = ch_tier_hegemon
	}
	text = {
		trigger = { var:ch_tier >= 4 }
		localization_key = ch_tier_major
	}
	text = {
		trigger = { var:ch_tier >= 3 }
		localization_key = ch_tier_significant
	}
	text = {
		trigger = { var:ch_tier >= 2 }
		localization_key = ch_tier_moderate
	}
	text = {
		trigger = { var:ch_tier >= 1 }
		localization_key = ch_tier_minor
	}
	text = {
		localization_key = ch_tier_negligible
	}
}

# Whether the exported model is ours, so the alignment block can say "our model"
# or name the hegemon instead of implying the player leads the world.
ch_exported_model_owner = {
	type = country
	random_valid = no

	text = {
		trigger = {
			has_global_variable = ch_rank_1
			global_var:ch_rank_1 ?= ROOT
		}
		localization_key = ch_model_owner_us
	}
	text = {
		trigger = {
			has_global_variable = ch_rank_1
			exists = global_var:ch_rank_1
		}
		localization_key = ch_model_owner_them
	}
	text = {
		localization_key = ch_model_owner_unknown
	}
}
