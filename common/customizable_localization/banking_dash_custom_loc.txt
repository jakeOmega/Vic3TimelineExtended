# ============================================================================
# Banking dashboard — banded readings
# ============================================================================
# The cycle's momentum and bubble pressure are deliberately imperfect
# information: the player reads a bar and a band, never the number behind it.
# These two blocks turn the hidden variables into that band for the Current
# Conditions rows (gui/journal_entry_widgets/banking_dashboard_widget.gui).
# The band edges reuse thresholds the script already acts on (momentum +-3 is
# banking_cycle_bubble_risk_rising / banking_cycle_risk_falling, and +-5 the
# ends of banking_cycle_momentum_bar and the deepest crash tier's momentum;
# bubble 15 / 30 / 50 are crash-roll steps in banking_cycle_effects.txt).
# Momentum has seven bands, so that its two open-ended ones, Freefall and
# Overheating, are past the bar's ends: in the cycle simulator momentum is
# there in under 1% of months, where +-3 was crossed in up to a quarter, and
# a monthly-change range open on one side (banking_disp_*_change_shape) is
# that rare too. The overview's icon
# codes and the monthly-change ranges (banking_overview_display_values.txt)
# band momentum and bubble pressure at the same edges; keep them in step.
# ============================================================================

banking_dash_momentum_band = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_variable = finance_cycle_momentum } }
		localization_key = banking_dash_band_unknown
	}
	text = {
		trigger = { var:finance_cycle_momentum >= 5 }
		localization_key = banking_dash_momentum_band_overheating
	}
	text = {
		trigger = { var:finance_cycle_momentum >= 3 }
		localization_key = banking_dash_momentum_band_surging
	}
	text = {
		trigger = { var:finance_cycle_momentum >= 1 }
		localization_key = banking_dash_momentum_band_rising
	}
	text = {
		trigger = { var:finance_cycle_momentum > -1 }
		localization_key = banking_dash_momentum_band_steady
	}
	text = {
		trigger = { var:finance_cycle_momentum > -3 }
		localization_key = banking_dash_momentum_band_falling
	}
	text = {
		trigger = { var:finance_cycle_momentum > -5 }
		localization_key = banking_dash_momentum_band_collapsing
	}
	text = {
		localization_key = banking_dash_momentum_band_freefall
	}
}

# Monetary stance band (monetary_policy_design.md §8)
# ----------------------------------------------------------------------------
# Returns the NAME of the stance band and nothing else — no gap, no rate, no
# arithmetic. The gap behind the band is hidden state (§8, "The rule"): what the
# player chooses is shown exactly, what the economy is doing is banded. This is
# the only thing allowed to render any part of te_mon_stance_gap.
#
# Deliberately named generically rather than banking_dash_*: the banking
# journal entry's momentum tooltip is its first consumer, and the monetary
# dashboard's own stance row is meant to call the same block rather than
# duplicate the ladder.
#
# te_mon_stance_band is written by te_monetary_update_stance
# (common/scripted_effects/te_monetary_effects.txt) and is 3 for every country
# without a policy dial, which is why bankless and command economies read
# Neutral here. The has_variable branch covers a country whose first monthly
# monetary update has not run yet.
te_mon_stance_band_name = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_variable = te_mon_stance_band } }
		localization_key = banking_dash_band_unknown
	}
	text = {
		trigger = { var:te_mon_stance_band <= 1 }
		localization_key = banking_dash_stance_band_very_loose
	}
	text = {
		trigger = { var:te_mon_stance_band <= 2 }
		localization_key = banking_dash_stance_band_loose
	}
	text = {
		trigger = { var:te_mon_stance_band <= 3 }
		localization_key = banking_dash_stance_band_neutral
	}
	text = {
		trigger = { var:te_mon_stance_band <= 4 }

	}
	text = {
		localization_key = banking_dash_stance_band_very_tight
	}
}

# ----------------------------------------------------------------------------
# What the band is WORTH, in the cycle's own units (monetary_policy_design.md §8)
# ----------------------------------------------------------------------------
# "Loose" on its own does not tell a player whether it means a tenth of a point
# of momentum or ten. These two blocks give each band its order of magnitude
# without giving up the gap.
#
# WHERE THE NUMBERS COME FROM. The band is set from the estimated gap
# (te_monetary_update_stance): band 1 at or below -2, band 2 above that to
# -0.75, band 3 inside +/-0.75, band 4 up to 2, band 5 at or above 2. The push
# is te_mon_stance_momentum_add (-0.125 per pp) and te_mon_stance_bubble_add
# (-0.75 per pp), both on the gap CLAMPED to -2 .. +4
# (te_mon_stance_gap_clamped; the loose bound was halved from -4 in #371). So
# each band's range is its threshold pair times those two rates, band 5 runs
# out at the tight clamp, and band 1 sits at the loose clamp, which is also
# band 2's far edge:
#
#   band   gap            momentum/month      bubble/month
#   1      <= -2 (at -2)  +0.25 (the most)    +1.5 (the most)
#   2      -2 .. -0.75    +0.1  .. +0.25      +0.5 .. +1.5
#   3      +/-0.75        under 0.1 either way   under 0.5 either way
#   4      0.75 .. 2      -0.1  .. -0.25      -0.5 .. -1.5
#   5      >= 2 (to 4)    -0.25 .. -0.5       -1.5 .. -3
#
# THESE ARE NOT EXACT AND MUST NOT READ AS IF THEY WERE. The band is drawn from
# the bank's ESTIMATE of the gap and the push from the TRUE gap, and the two
# differ by the forecasting error (te_mon_forecast_error_bound, +/-1.5pp, or
# +/-0.5pp under central bank independence). Every consumer says so once, which
# is why each phrase here is hedged with "roughly" and never carries a decimal
# the player could difference against.
#
# If §21's -0.125 / -0.75 rates or the band thresholds are ever retuned, the
# table above and the five pairs of phrases below move with them — that is the
# cost of stating a magnitude at all, and it is cheaper than a player who
# cannot tell a lever from a label.
te_mon_stance_momentum_effect = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_variable = te_mon_stance_band } }
		localization_key = banking_dash_band_unknown
	}
	text = {
		trigger = { var:te_mon_stance_band <= 1 }
		localization_key = banking_dash_stance_mom_very_loose
	}
	text = {
		trigger = { var:te_mon_stance_band <= 2 }
		localization_key = banking_dash_stance_mom_loose
	}
	text = {
		trigger = { var:te_mon_stance_band <= 3 }
		localization_key = banking_dash_stance_mom_neutral
	}
	text = {
		trigger = { var:te_mon_stance_band <= 4 }
		localization_key = banking_dash_stance_mom_tight
	}
	text = {
		localization_key = banking_dash_stance_mom_very_tight
	}
}

# Why this country has no policy dial (§4, §5.2, §5.3). One sentence per cause,
# because "you cannot set a rate" without a reason is the kind of dead end the
# dashboard exists to avoid. The order matches te_monetary_set_regime's: a
# command economy administers the rate whatever its currency law says, and a
# country with no national bank has no dial whatever its currency law says.
#
# DOLLARISATION COMES FIRST (ruling P9), ahead of every law-based cause. A
# country that abandoned its currency in a hyperinflation usually still has
# law_national_bank and law_fiat_currency on the books, so every branch below
# would miss it and the fall-through would tell it that its central bank sets
# the rate — which is the one thing that is no longer true.
#
# COMMODITY MONEY LEFT THIS LIST on 2026-09-21 (§5.4's open question). With a
# national bank it now holds the narrow dial and the widget hides this line
# outright; without one it is caught by the no-bank branch above, which is the
# binding cause and the one that names the way out. So a commodity-money branch
# here would be unreachable from either side, and its loc key went with it —
# te_mon_dial_range_note below is where the regime gets its sentence now.
#
# The last branch is the fall-through for a country that DOES hold the dial; the
# widget hides this line for those, so it should never be seen.
te_mon_no_dial_reason = {
	type = country
	random_valid = no

	text = {
		trigger = { te_mon_is_dollarised = yes }
		localization_key = banking_dash_mon_why_dollarised
	}
	text = {
		trigger = { te_mon_is_command = yes }
		localization_key = banking_dash_mon_why_command
	}
	# [P5] Anchored (§15A.1), above the bank test: a country with a national
	# bank and a treaty peg has no dial BECAUSE of the peg, and the way out it
	# should be told about is the treaty's, not a law's. Each key names the
	# anchor through Var('te_mon_anchor'), which te_mon_is_anchored guarantees.
	text = {
		trigger = {
			te_mon_is_anchored = yes
			var:te_mon_anchor_kind = 3
		}
		localization_key = banking_dash_mon_why_board
	}
	text = {
		trigger = {
			te_mon_is_anchored = yes
			var:te_mon_anchor_kind = 2
		}
		localization_key = banking_dash_mon_why_union
	}
	text = {
		trigger = { te_mon_is_anchored = yes }
		localization_key = banking_dash_mon_why_peg
	}
	text = {
		trigger = { NOT = { has_law_or_variant = law_type:law_national_bank } }
		localization_key = banking_dash_mon_why_no_bank
	}
	text = {
		trigger = { has_law_or_variant = law_type:law_decentralized_cryptocurrency }
		localization_key = banking_dash_mon_why_crypto
	}
	text = {
		localization_key = banking_dash_mon_why_has_dial
	}
}

# ----------------------------------------------------------------------------
# Phase 2 — may this country print money to pay its bills? (§11)
# ----------------------------------------------------------------------------
# ONE LINE, PHRASED AS A CONDITION, naming the single binding cause. The
# monetisation stepper's is_valid renders it through banking_mon_tt_can_monetise
# beside a tick or a cross (common/scripted_guis/te_monetary_sguis.txt, ops
# 7/8), so every branch has to read correctly in both states — "You have a
# national bank to print the money through" works with either mark, while "You
# have no national bank" would not.
#
# THE BRANCH ORDER IS A PRIORITY ORDER, not te_mon_can_monetise's clause order:
# a country can fail several of these at once, and this picks which one the
# player is told about. Dollarisation comes first because it is the one cause
# that does not appear in the law list at all. Then the command economy, which
# settles the question before any monetary law is consulted; then the missing
# bank, because there is no institution to give the order to; then the
# currency; and independence last, because it is the only one of the five that
# can be undone without changing what the country's money is.
#
# The fall-through is reached exactly when te_mon_can_monetise holds — every
# clause of that trigger has a branch above — so it is the line that renders
# with a tick.
te_mon_monetise_condition = {
	type = country
	random_valid = no

	text = {
		trigger = { te_mon_is_dollarised = yes }
		localization_key = banking_dash_mon_monetise_why_dollarised
	}
	text = {
		trigger = { te_mon_is_command = yes }
		localization_key = banking_dash_mon_monetise_why_command
	}
	text = {
		trigger = { NOT = { has_law_or_variant = law_type:law_national_bank } }
		localization_key = banking_dash_mon_monetise_why_no_bank
	}
	text = {
		trigger = {
			NOR = {
				has_law_or_variant = law_type:law_fiat_currency
				has_law_or_variant = law_type:law_digital_currency
			}
		}
		localization_key = banking_dash_mon_monetise_why_backed
	}
	text = {
		trigger = { te_mon_is_cbi = yes }
		localization_key = banking_dash_mon_monetise_why_cbi
	}
	text = {
		localization_key = banking_dash_mon_monetise_why_allowed
	}
}


# Phase 3 (§12.3): the word beside the peg-confidence figure. The figure is
# displayed exactly — it is built from the displayed gap, the treasury and the
# debt, none of which carries the neutral rate — so this is a reading aid, not a
# band standing in for a hidden number. Thresholds: the crisis fires at
# te_mon_peg_crisis_threshold (20), and a resumed or devalued peg restarts at
# te_mon_peg_devalue_confidence (50), which is why 50 is still only "Watched".
# The Doubted / Watched bands are te_mon_peg_doubted_below and
# te_mon_peg_watched_below, which the monthly recovery is keyed on too.
#
# Suspension is tested first: confidence is frozen while convertibility is
# suspended, so its figure says nothing and the word says why. Defending comes
# next because the +40 it bought would otherwise read as calm. A deep slump
# (2026-10-05) comes after Breaking: it is WHY a healthy vault's confidence is
# falling, which none of the band words says, and it holds the recovery the
# bands are keyed on, so for as long as it lasts the band is not the news.
te_mon_peg_state = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_variable = te_peg_confidence } }
		localization_key = banking_dash_band_unknown
	}
	text = {
		trigger = { te_mon_peg_is_suspended = yes }
		localization_key = banking_dash_mon_peg_state_suspended
	}
	text = {
		trigger = { te_mon_peg_is_defending = yes }
		localization_key = banking_dash_mon_peg_state_defending
	}
	text = {
		trigger = { var:te_peg_confidence <= te_mon_peg_crisis_threshold }
		localization_key = banking_dash_mon_peg_state_breaking
	}
	text = {
		trigger = { te_mon_peg_in_deep_slump = yes }
		localization_key = banking_dash_mon_peg_state_slump
	}
	text = {
		trigger = { var:te_peg_confidence < te_mon_peg_doubted_below }
		localization_key = banking_dash_mon_peg_state_doubted
	}
	text = {
		trigger = { var:te_peg_confidence < te_mon_peg_watched_below }
		localization_key = banking_dash_mon_peg_state_watched
	}
	text = {
		localization_key = banking_dash_mon_peg_state_trusted
	}
}

# The Peg Confidence tooltip's suspension clock (2026-10-05). Step 10 stops the
# counter at 1 and asks te_peg.3, and the country stays suspended until it is
# answered — up to the event's three months — so "1 months left" would sit there
# that long. With the question out it says the decision is due instead.
te_mon_peg_suspension_note = {
	type = country

			has_variable = te_peg_resume_asked
		}
		localization_key = banking_dash_mon_peg_suspension_due
	}
	text = {
		localization_key = banking_dash_mon_peg_suspension_left
	}
}


# ----------------------------------------------------------------------------
# What kind of range the dial has, and what pays for it (§5.1, §5.4)
# ----------------------------------------------------------------------------
# The Rate Target tooltip prints the regime's floor and ceiling from
# te_mon_target_min / te_mon_target_max, then this block says in a sentence why
# they sit where they do. It used to be one hard-coded #lore line about the gold
# standard, which was true of the only regime that had an interesting range —
# and became wrong the moment commodity money got a band of its own.
#
# ORDER. Suspended gold is tested before gold, because a suspension is exactly
# the case where the printed range (0-25) contradicts the law's name. Commodity
# money is next, then gold, then the two flat-range regimes share the
# fall-through: a fiat or digital player is told the range is theirs to use,
# which is the whole difference the ladder is selling.
#
# NOT GATED ON te_mon_display_ready. The tooltip that calls this is, and no
# branch here reads a variable — every trigger is a law or a law-plus-clock
# test, so a country that has never pulsed still gets a true sentence.
te_mon_dial_range_note = {
	type = country
	random_valid = no

	text = {
		trigger = {
			has_law_or_variant = law_type:law_gold_standard
			te_mon_peg_is_suspended = yes
		}
		localization_key = banking_dash_mon_range_gold_suspended
	}
	text = {
		trigger = { te_mon_has_narrow_dial = yes }
		localization_key = banking_dash_mon_range_commodity
	}
	text = {
		trigger = { te_mon_is_on_gold = yes }
		localization_key = banking_dash_mon_range_gold
	}
	text = {
		localization_key = banking_dash_mon_range_free
	}
}

# §0.12, ruling N1 — one sentence appended to the Delegation tooltip's first
