# ============================================================================
# GLOBAL WARMING — branchy display text
# ============================================================================
# Read from je_global_warming's `progress_desc` as
# [ROOT.GetCountry.GetCustom('<key>')] and from
# gui/journal_entry_widgets/global_warming_widget.gui as
# [JournalEntry.GetCountry.GetCustom('<key>')].
#
# THE SEVERITY LADDER IS DEFINED EXACTLY ONCE, in the script value
# gw_disp_tier_code (common/script_values/global_warming_values.txt), 0
# negligible to 6 apocalyptic. The two ladders below compare that code, and the
# widget's overview picks its tier icon by it, so the sentence, the word and the
# icon cannot drift. The ladder used to be six `triggered_desc` blocks inside the
# journal entry's `status_desc`, and then temperature thresholds here; the
# sentence keys are unchanged. The thresholds match the journal entry's own
# scale and the five warming-threshold events in global_warming_events_on_action
# (0.5, 1, 2, 3, 4 C); change them in all three places or in none.
#
# `gw_disp_tier_code` is read as a bare script value on the left of a
# comparison, which is the documented form for using a script value as a
# trigger. It is O(1) (a few comparisons of one global-variable read), so these
# are safe to evaluate per frame.
# ============================================================================

# The full tier sentence: the tooltip of the overview's tier icon. It was the
# journal entry's status line until the overview took its place (2026-09-29).
gw_severity_text = {
	type = country
	random_valid = no

	text = {
		trigger = { gw_disp_tier_code >= 6 }
		localization_key = je_global_warming_status_apocalyptic
	}
	text = {
		trigger = { gw_disp_tier_code >= 5 }
		localization_key = je_global_warming_status_catastrophic
	}
	text = {
		trigger = { gw_disp_tier_code >= 4 }
		localization_key = je_global_warming_status_severe
	}
	text = {
		trigger = { gw_disp_tier_code >= 3 }
		localization_key = je_global_warming_status_significant
	}
	text = {
		trigger = { gw_disp_tier_code >= 2 }
		localization_key = je_global_warming_status_moderate
	}
	text = {
		trigger = { gw_disp_tier_code >= 1 }
		localization_key = je_global_warming_status_slight
	}
	text = {
		trigger = { always = yes }
		localization_key = je_global_warming_status_negligible
	}
}

# Sign and colour for the yearly temperature change.
#
# Deliberately NOT `te_yearly_temp_change_prefix` in zzz_extra_custom_loc.txt:
# that one's triggers read `global_greenhouse_gas_emissions_script_value_display`,
# which iterates every state in the world, and reusing it here would re-import
# per frame exactly the cost this widget exists to remove. This reads the
# guarded yearly snapshot instead.
gw_trend_prefix = {
	type = country
	random_valid = no

	text = {
		trigger = { gw_yearly_change_display < 0 }
		localization_key = gw_trend_prefix_cooling
	}
	text = {
		trigger = { gw_yearly_change_display > 0 }
		localization_key = gw_trend_prefix_warming
	}
	text = {
		trigger = { always = yes }
		localization_key = gw_trend_prefix_steady
	}
}
