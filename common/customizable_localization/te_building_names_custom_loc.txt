# ============================================================================
# ERA BUILDING NAMES — customizable localization
# ============================================================================
# Buildings whose name fits only part of the timeline. The building's name key
# (in localization/english/replace/timeline_extended_override_l_english.yml for
# a base-game building, te_buildings_l_english.yml for a mod building) reads
#
#   [SelectLocalization(GetPlayer.IsValid, '<building>_name_player', '<building>_name_<first>')]
#
# and <building>_name_player (te_buildings_l_english.yml) calls the block below
# on the player. So the name follows the *player's* technology: every country's
# buildings of that type read the same name, and each multiplayer client sees
# the name for its own tech. The building's own owner or production methods
# cannot be used, because the engine looks up a building type's name with no
# building in context. The GetPlayer.IsValid guard covers screens that render
# with no player, such as game concepts opened from the main menu; there the
# first-era name shows.
#
# Each block opens with the Era Building Names game rule's disabled setting,
# which gives the building the one name it had before the era names (the rule's
# comment in common/game_rules/extra_game_rules.txt lists them). Screens with no
# player show the first-era name under either setting. The eras follow, latest
# first. The first valid entry wins, and the trigger-less last entry is the name
# before the first switch. Each switch is the technology that unlocks the
# production method the new name describes, and a name that must also cover
# buildings still on older methods says so (the Skyport). Base-game
# names keep the base game's wording and number.
#
# To add a building: give it a block here (opening with the rule entry), a
# <building>_name_player key and one key per name in te_buildings_l_english.yml,
# point its name key at the SelectLocalization line above, and turn any other
# loc that spells the name out (its throughput modifier, its base production
# method group) into a $<building>$ splice so it follows along.

# Arts Academy: Sound Film (Mass Media), Digital Streaming (Digital
# Entertainment), Neurostimulation (Brain-Computer Interfaces).
te_building_art_academy_name = {
	type = country
	random_valid = no

	text = {
		trigger = { has_game_rule = era_building_names_disabled }
		localization_key = building_art_academy_name_creative
	}
	text = {
		trigger = { has_technology_researched = brain_computer_interfaces }
		localization_key = building_art_academy_name_experience
	}
	text = {
		trigger = { has_technology_researched = digital_entertainment }
		localization_key = building_art_academy_name_creative
	}
	text = {
		trigger = { has_technology_researched = mass_media }
		localization_key = building_art_academy_name_entertainment
	}
	text = {
		trigger = { always = yes }
		localization_key = building_art_academy_name_academy
	}
}
