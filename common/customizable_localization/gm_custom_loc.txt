# The monument's name in the journal entry's row. Its targets read their data through

	random_valid = no

		trigger = { has_variable = gm_name }

		trigger = { gm_state_namesake_is = { NAMESAKE = city } }

		trigger = { gm_state_namesake_is = { NAMESAKE = state } }
		localization_key = gm_name_state_row
	}
	text = {
		trigger = {
			gm_state_namesake_is = { NAMESAKE = ruler }
			has_variable = gm_name_char
			exists = var:gm_name_char
		}
		localization_key = gm_name_ruler_row
	}
	text = {
		trigger = {
			gm_state_namesake_is = { NAMESAKE = honours }

		}
		localization_key = gm_name_honours_row
	}
	text = {
		trigger = {
			gm_state_namesake_is = { NAMESAKE = occasion }
