# ============================================================================
# GRAND MONUMENTS — trophies: customizable localization
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-12-grand-monument-trophies-design.md §1.
# A trophy keeps the name of the monument it came from: the state's name family
# (gm_monument_name_row, gm_custom_loc.txt) read from the trophy's own records
# (gm_tr_*, gm_trophy_effects.txt). Customizable-localization targets inherit
# their caller's context, so these serve ScriptContainer rows only. The form,
# piece and landmark branches are pinned to the registry's tables by
# test_grand_monument_trophies.py.

# The monument's name. First valid wins: a typed name, the namesake, else its
# form and the state it came from.
gm_trophy_name = {
	type = container
	random_valid = no

	text = {
		trigger = { has_variable = gm_tr_name }
		localization_key = gm_tr_name_custom
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:landmark has_variable = gm_tr_landmark }
		localization_key = gm_tr_name_landmark
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:city exists = var:gm_tr_origin }
		localization_key = gm_tr_name_city
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:state exists = var:gm_tr_origin }
		localization_key = gm_tr_name_state
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:ruler has_variable = gm_tr_name_char exists = var:gm_tr_name_char }
		localization_key = gm_tr_name_ruler
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:honours }
		localization_key = gm_tr_name_honours
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:victory has_variable = gm_tr_name_enemy_cap exists = var:gm_tr_name_enemy_cap }
		localization_key = gm_tr_name_victory
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:defeat }
		localization_key = gm_tr_name_defeat
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:centenary }
		localization_key = gm_tr_name_centenary
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:space_orbital }
		localization_key = gm_tr_name_space_orbital
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:space_moon_landing }
		localization_key = gm_tr_name_space_moon_landing
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:space_moon_base }
		localization_key = gm_tr_name_space_moon_base
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:space_mars_landing }
		localization_key = gm_tr_name_space_mars_landing
	}
	text = {
		trigger = { has_variable = gm_tr_namesake var:gm_tr_namesake = flag:occasion has_variable = gm_tr_name_occasion var:gm_tr_name_occasion = flag:space_interstellar_probe }
		localization_key = gm_tr_name_space_interstellar_probe
	}
	text = {
		trigger = { has_variable = gm_tr_namesake has_variable = gm_tr_name_year }
		localization_key = gm_tr_name_year
	}
	# A landmark that lost its historical name keeps its look: name it by that.
	text = {
		trigger = { has_variable = gm_tr_landmark }
		localization_key = gm_tr_name_landmark
	}
	text = {
		trigger = { exists = var:gm_tr_origin }
		localization_key = gm_tr_name_from
	}
	text = {
		localization_key = gm_tr_name_form_only
	}
}

# The row's title: the name, and what of it was taken when it was not the whole.

	type = container
	random_valid = no

	text = {
		trigger = { has_variable = gm_tr_share var:gm_tr_share = flag:whole }
		localization_key = gm_tr_title_whole
	}
	text = {
		localization_key = gm_tr_title_part
	}
}
