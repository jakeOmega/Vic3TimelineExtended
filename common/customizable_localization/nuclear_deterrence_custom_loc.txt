# ============================================================================
# NUCLEAR DETERRENCE — customizable localization
# ============================================================================
# The words the posture and crisis panels (and a few crisis events) choose
# between. Design: docs/systems/nuclear_crisis_design.md (§0 is what shipped).
#
# CONTEXT. Every block here is called from widget loc
# ([JournalEntry.GetCountry.GetCustom('…')]) or event loc
# ([SCOPE.sCountry('…').GetCustom('…')] / [ROOT.GetCountry.GetCustom('…')]),
# never from a journal entry's status_desc, and every target key is plain text
# with no country accessor — the numbers are in the calling key
# (gui_modding_guide.md gotcha #20).
#
# GUARDS. random_valid = no evaluates every branch until one matches, and a
# non-nuclear crisis party reaches these blocks without the posture variables,
# so every variable read is paired with has_variable and every block ends in an
# unconditional fallback.
# ============================================================================

# A readiness change under way: empty when readiness already matches its target.
nd_readiness_moving = {
	type = country
	random_valid = no
	text = {
		trigger = {
			has_variable = nd_readiness
			has_variable = nd_readiness_target
			var:nd_readiness_target = 0
			NOT = { var:nd_readiness = 0 }
		}
		localization_key = nd_readiness_moving_0
	}
	# Up to Routine from Recessed is no stand-down.
	text = {
		trigger = {
			has_variable = nd_readiness
			has_variable = nd_readiness_target
			var:nd_readiness_target = 1
			var:nd_readiness = 0
		}
		localization_key = nd_readiness_moving_1_up
	}
	text = {
		trigger = {
			has_variable = nd_readiness
			has_variable = nd_readiness_target
			var:nd_readiness_target = 1
			NOT = { var:nd_readiness = 1 }
		}
		localization_key = nd_readiness_moving_1
	}
	text = {
		trigger = {
			has_variable = nd_readiness
			has_variable = nd_readiness_target
			var:nd_readiness_target = 2

		}
		localization_key = nd_readiness_moving_2
