# ============================================================================
# Strategic Reserve — Customizable Localization
# ============================================================================
# Both families read ONE variable: `st_res_<good>_last_status`, written by
# st_res_set_good_status_effect (common/scripted_effects/st_res_effects.txt).
# Nothing here re-derives a reserve's state from rates or stockpiles, so the
# label the player reads and the state the bookkeeping decided can never drift.
#
#   0 Idle | 1 Storing | 2 Withdrawing | 3 Storing (flow-capped)
#   4 Withdrawing (flow-capped) | 5 Full | 6 Empty | 7 Understaffed | 8 No hub
#
# `st_res_<good>_mode_text`   -> the short Storing / Withdrawing / Idle /
#                                Blocked label shown in the inventory row.
# `st_res_<good>_reason_text` -> the one-sentence explanation of why the actual
#                                weekly movement differs from the configured
#                                rate, shown in that row's tooltip.
#
# Every branch is `has_variable`-guarded because customizable_localization is
# evaluated for countries that have never opened the journal entry, and an
# unguarded `var:` read on an unset variable raises script-system errors. A save
# from before these variables existed therefore falls through to Idle / no
# explanation until the first weekly tick writes a real status.
# ============================================================================

st_res_grain_mode_text = {
	type = country
	random_valid = no

	text = {
		trigger = {
			has_variable = st_res_grain_last_status
			OR = {
				var:st_res_grain_last_status = 1
				var:st_res_grain_last_status = 3
			}
		}
		localization_key = st_res_mode_storing
	}
	text = {
		trigger = {
			has_variable = st_res_grain_last_status
			OR = {
				var:st_res_grain_last_status = 2
				var:st_res_grain_last_status = 4
			}
		}
		localization_key = st_res_mode_withdrawing
	}
	text = {
		trigger = {
			has_variable = st_res_grain_last_status
			var:st_res_grain_last_status >= 5
		}
		localization_key = st_res_mode_blocked
	}
	text = {
		trigger = { always = yes }
		localization_key = st_res_mode_idle
	}
}
