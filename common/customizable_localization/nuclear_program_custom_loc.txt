# ============================================================================
# NUCLEAR PROGRAMME — customizable localization
# ============================================================================
# The branchy text that used to be spelled out on je_nuclear_program: its
# eleven-line status_desc and the four-clause je_nuclear_program_reason wall.
#
# TWO RULES, both of which have bitten this repo:
#
# 1. CONTEXT. A target key reached from a journal entry's status_desc / desc is
#    built in the ENTRY's script scope, where the country root is
#    `ROOT.GetCountry`; a target key reached from a .gui `text =` / `tooltip =`
#    is built in a GUI datacontext, where it is `JournalEntry.GetCountry`
#    (gui_modding_guide.md gotcha #20). No root is right for both. So:
#      * nuclear_program_status_line is called ONLY from
#        je_nuclear_program_status_line, and its targets carry NO country
#        accessor at all — every number in that line is appended by the calling
#        key, not by the branch.
#      * every other block here is called ONLY from
#        gui/journal_entry_widgets/nuclear_program_widget.gui, so its targets may
#        use JournalEntry.GetCountry. DO NOT call them from a desc.
#
# 2. GUARDS. je_nuclear_program carries is_shown_when_inactive, so these blocks
#    are evaluated for countries whose entry never activated and which have none
#    of the programme's variables. `random_valid = no` means every `text` trigger
#    is evaluated until one matches, so an unguarded `var:` read in an early
#    branch would log "Failed to fetch variable" on every frame for those
#    countries (gotcha #14). Every variable read below is therefore paired with
#    `has_variable`, and the three branches that describe a country with no
#    programme come first and test modifiers, the tech and a scripted trigger
#    live — the entry may be deactivated, with the weekly pulse stopped and
#    nuclear_program_last_status stale, or active for an armed power or a
#    crisis party that has no programme at all (je_nuclear_program also
#    carries posture and crises; see its header).
#
# nuclear_program_last_status is written in exactly one place,
# nuclear_program_refresh_state_effect. Nothing here re-derives it.
#   0 authorised but unfunded   1 first device under development
#   2 in series production      3 frozen by a pause treaty
#   4 held at our arsenal ceiling 5 being dismantled
# ============================================================================

# The journal entry's whole status line. Called from
# je_nuclear_program_status_line only; accessor-free branches.
nuclear_program_status_line = {
	type = country
	random_valid = no

	# Renounced the bomb of its own accord (Dismantle the Arsenal, the nuclear
	# taboo): comes first, ahead of the disarmed branch below (I3, final-review.md)
	# — without it, a country that chose to give up its arsenal would be told,
	# permanently and in red, that a settlement or the NPT did it for them.
	text = {
		trigger = { has_modifier = nd_taboo_renounced }
		localization_key = nuclear_program_status_renounced
	}
	# A disarmament settlement (or the UN's NPT modifier) makes the entry's
	# `possible` fail, so the entry deactivates and the pulse stops: read the
	# modifier, not the status code. A country that renounced of its own accord
	# (nd_taboo_renounced, above) never falls through to this branch.
	text = {
		trigger = { modifier:country_nuclear_disarmament_bool = yes }
		localization_key = nuclear_program_status_disarmed
	}
	# No programme: the entry is inactive, or active for an armed power or a
	# crisis party. The same triggers nuclear_program_has_programme asks, so
	# the line and the programme panel can never disagree.
	text = {
		trigger = {
			NOT = { has_technology_researched = nuclear_weapons }
		}
		localization_key = nuclear_program_status_no_science
	}
	text = {
		trigger = { nuclear_program_can_run_programme = no }
		localization_key = nuclear_program_status_ineligible
	}
	text = {
		trigger = {
			has_variable = nuclear_program_last_status
			var:nuclear_program_last_status = 5
		}
		localization_key = nuclear_program_status_dismantling
	}
	text = {
		trigger = {
			has_variable = nuclear_program_last_status
			var:nuclear_program_last_status = 4
		}
		localization_key = nuclear_program_status_held
	}
	text = {
		trigger = {
			has_variable = nuclear_program_last_status
			var:nuclear_program_last_status = 3
		}
		localization_key = nuclear_program_status_paused
	}
	text = {
		trigger = {
			has_variable = nuclear_program_last_status
			var:nuclear_program_last_status = 0
		}
		localization_key = nuclear_program_status_unfunded
	}
	text = {
		trigger = {
			has_variable = nuclear_program_last_status
			var:nuclear_program_last_status = 1
		}
		localization_key = nuclear_program_status_developing
	}
	text = {
		trigger = {
			has_variable = nuclear_program_last_status
			var:nuclear_program_last_status = 2
		}
		localization_key = nuclear_program_status_producing
	}
	# A save from before the status code existed, or the week between activation
	# and the first refresh.
	text = {
		trigger = { always = yes }
		localization_key = nuclear_program_status_generic
	}
}

# The nuclear taboo, first on the status line (nuclear_taboo_values.txt). Once
# it exists the entry is active for every country, and for most of them the
# taboo is the whole story. Accessor-free: the score is a global, read in the
# targets with GetGlobalVariable, which needs no country root; the bands are
# the live score's (nd_taboo_line_*), not the last one announced.
nd_taboo_status_line = {
	type = country
	random_valid = no

	text = {
		trigger = { NOT = { has_global_variable = nd_taboo } }
		localization_key = nuke_line_empty
	}
	text = {
		trigger = { nd_taboo_value < nd_taboo_line_fragile }
		localization_key = nd_taboo_status_1
	}
	text = {
		trigger = { nd_taboo_value < nd_taboo_line_established }
		localization_key = nd_taboo_status_2
	}
	text = {
		trigger = { nd_taboo_value < nd_taboo_line_strong }
		localization_key = nd_taboo_status_3
	}
	text = {
		trigger = { nd_taboo_value < nd_taboo_line_absolute }
		localization_key = nd_taboo_status_4
	}
	text = {
		trigger = { always = yes }
		localization_key = nd_taboo_status_5
	}
}

# ---------------------------------------------------------------------------
# WIDGET-ONLY BLOCKS — targets below may use JournalEntry.GetCountry, so none of
# them may ever be called from a status_desc, desc or scripted-button desc.
# ---------------------------------------------------------------------------

# What is setting the warhead production rate. Shown under the rate row.
nuclear_program_rate_note = {
	type = country
	random_valid = no

	text = {
		trigger = { has_modifier = nd_taboo_programme_held }
		localization_key = nuclear_program_rate_note_held
	}
	text = {
		# Live modifier, not status code 3, for the same reason the status line
		# reads its blockers live: between a pause article entering force and the
		# next weekly pulse the status code is still stale, while
		# nuclear_program_display_months_to_next already tests the modifier — so
		# reading the code here would have this row quote a rate in the same frame
		# the row below it said "not progressing".
		trigger = { modifier:country_nuclear_program_pause_bool = yes }
		localization_key = nuclear_program_rate_note_paused
	}
	text = {
		trigger = {
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding = 0
		}
		localization_key = nuclear_program_rate_note_unfunded
	}
	text = {
		trigger = {
			has_variable = nuclear_weapons_program_first_nuke_done
			var:nuclear_weapons_program_first_nuke_done > 0
		}
		localization_key = nuclear_program_rate_note_series
	}
	text = {
		trigger = { always = yes }
		localization_key = nuclear_program_rate_note_prototype
	}
}

# The foreign line — assistance, sabotage (covert slice 6), or both —
# appended after the rate note. Separate block because customizable
# localization picks exactly one branch, and the foreign line is orthogonal to
# which phase the programme is in.
nuclear_program_aid_note = {
	type = country
	random_valid = no

	# Covert slice 6: country_nuclear_program_progress_mult also carries
	# covert_nuclear_sabotage, so while it runs the number is the NET foreign
	# effect, not assistance alone. These two come first because customizable
	# localization takes the first branch that holds.
	text = {
		trigger = {
			has_modifier = covert_nuclear_sabotage
			modifier:country_receiving_nuclear_program_aid_bool = yes
		}
		localization_key = nuclear_program_aid_note_sabotaged
	}
	text = {
		trigger = { has_modifier = covert_nuclear_sabotage }
		localization_key = nuclear_program_sabotage_note
	}
	text = {
		trigger = { modifier:country_receiving_nuclear_program_aid_bool = yes }
		localization_key = nuclear_program_aid_note_active
	}
	text = {
		trigger = { always = yes }
		localization_key = nuke_line_empty
	}
}

# "Next warhead in" — the estimate, or a dash when nothing is accruing.
# nuclear_program_display_months_to_next is 0 in exactly that case.
nuclear_program_next_warhead = {
	type = country
	random_valid = no

	text = {
		trigger = {
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding >= 1
			NOT = { modifier:country_nuclear_program_pause_bool = yes }
			NOT = { modifier:country_nuclear_disarmament_bool = yes }
		}
		localization_key = nuclear_program_next_warhead_estimate
	}
	text = {
		trigger = { always = yes }
		localization_key = nuclear_program_next_warhead_stalled
	}
}

# Appended after the first-device year. Separate block for the same reason as
# the assistance line: exactly one branch of a block can be chosen, and being
# first in the world is orthogonal to whether the year was recorded.
nuclear_program_world_first_note = {
	type = country
	random_valid = no

	text = {
		trigger = { has_variable = is_world_first_nuclear_power }
		localization_key = nuclear_program_world_first_note_active
	}
	text = {
		trigger = { always = yes }
		localization_key = nuke_line_empty
	}
}

# "First device" — the year, or that it predates the record, or that there is
# none yet. nuclear_program_first_device_year is only written from the weekly
# pulse's first-device branch, so a programme that finished its first device
# before this panel shipped has the flag but not the year.
nuclear_program_first_device = {
	type = country
	random_valid = no

	text = {
		trigger = {
			has_variable = nuclear_weapons_program_first_nuke_done
			var:nuclear_weapons_program_first_nuke_done > 0
			has_variable = nuclear_program_first_device_year
		}
		localization_key = nuclear_program_first_device_year_known
	}
	text = {
		trigger = {
			has_variable = nuclear_weapons_program_first_nuke_done
			var:nuclear_weapons_program_first_nuke_done > 0
		}
		localization_key = nuclear_program_first_device_year_unknown
	}
	text = {
		trigger = { always = yes }
		localization_key = nuclear_program_first_device_none
	}
}

# ---- The arsenal: a ceiling, or dismantling it (the nuclear taboo) -----------

nd_taboo_arsenal_status = {
	type = country
	random_valid = no

	text = {
		trigger = { nd_taboo_is_dismantling = yes }
		localization_key = nd_taboo_arsenal_dismantling
	}
	text = {
		trigger = {
			has_variable = nd_treaty_ceiling
			OR = {
				NOT = { has_variable = nd_warhead_ceiling }
				var:nd_treaty_ceiling < var:nd_warhead_ceiling
			}
		}
		localization_key = nd_taboo_arsenal_treaty
	}
	text = {
		trigger = { has_variable = nd_warhead_ceiling }
		localization_key = nd_taboo_arsenal_ceiling
	}
	text = {
		trigger = { always = yes }
		localization_key = nd_taboo_arsenal_free
	}
}
