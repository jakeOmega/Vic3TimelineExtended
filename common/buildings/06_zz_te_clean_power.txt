# ============================================================================
# Power plants the mod adds: hydro, nuclear, renewable, fusion
# ============================================================================
# This file's name sets where its buildings appear in the building list,
# whose default sort is the order building definitions load: file names in
# ASCII order, vanilla's and the mod's in one list. 06_zz_te_clean_power.txt sorts
# right after vanilla's 06_urban_center.txt, so these buildings follow its Power Plant.
# Keep them bare definitions, not REPLACE_OR_CREATE: a directive applies
# after every bare definition, so the entry it creates may land at the end.
# Rule: docs/guides/scripting_best_practices.md § Load Order (probed
# 2026-10-26), "Building list order"; test_building_list_order.py.
# ============================================================================

building_hydro_plant = {
	building_group = bg_clean_power
	icon = "gfx/interface/icons/building_icons/building_hydro_plant.dds"
	city_type = city
	levels_per_mesh = -1

	production_method_groups = {
		pmg_base_building_hydro_power
		pmg_power_automation
		pmg_maintenance_heavy
	}

	unlocking_technologies = {
		public_works_programs
	}

	required_construction = construction_cost_very_high

	has_max_level = yes

	# On a river, or in a state whose great dam is built. Each dam wonder raises
	# its state's Hydro Plant cap and throughput, and Nevada, where the Hoover
	# Dam stands, is not on a river in the engine's map data, so with on_river
	# alone that wonder did nothing (2026-11-18). has_building, not
	# any_scope_building: a dam counts once a level is built, not while queued.
	# test_hydro_plant_sites.py holds this list to the buildings whose methods
	# grant state_building_hydro_plant_max_level_add.
	possible = {
		custom_tooltip = {
			text = building_hydro_plant_possible_tt
			OR = {
				on_river = yes
				has_building = building_wonder_hoover_dam
				has_building = building_wonder_aswan_high_dam
				has_building = building_wonder_itaipu_dam
				has_building = building_wonder_three_gorges_dam
			}
		}
	}
	ownership_type = self
	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}
