# ============================================================================
# extra_buildings.txt: the mod's INJECTs and REPLACEs of vanilla buildings, and
# the space buildings and megastructures.
# ============================================================================
# A REPLACE keeps the vanilla building's place in the building list and an
# INJECT never moves one, so those can live here. A building the mod adds lands
# where its file loads: this file loads after every vanilla building file, so
# only the space and megastructure block, which closes the list, is defined
# here. A new building belongs in its sector's NN_zz_te_*.txt file, after its
# vanilla peers (test_building_list_order.py checks it). Rule and file map:
# docs/guides/scripting_best_practices.md § Load Order (probed 2026-10-26).
# ============================================================================

	city_type = city
	levels_per_mesh = -1

	# Era 10 with Regolith Processing; Asteroid Mining (era 11) brings the
	# Asteroid Mine method (lunar colonization design § 6)

		in_situ_resource_utilization
	}

	production_method_groups = {
		pmg_base_building_space_mine
		pmg_launch_to_orbit_building_space_mine	# Global Commons § 1.2: era-12 logistics
		pmg_transport_in_space_building_space_mine
		pmg_maintenance_intensive
	}

	required_construction = construction_cost_space_mine
	# Simple and full colonies (colony modules design § 9): room for levels comes
	# from the near-Earth asteroids (with Asteroid Mining) and the colonies held,
	# and a level past it works at -75% (-50% until 2026-10-31). No private investor builds past the
	# room, and no AI government; a player's government can, at the penalty.
	# Under races only both always pass (sr_ebase_has_room).
	can_build_government = {
		owner ?= { sr_ebase_may_build_government = yes }
	}
	can_build_private = {
		owner ?= { sr_ebase_may_build_private = yes }
	}
	ownership_type = self
	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}
