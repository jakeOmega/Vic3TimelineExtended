# ============================================================================
# Strategic Reserve Buildings
# ============================================================================
# Two buildings:
#   - building_strategic_reserve_hub: capital-only, controlled from the JE.
#     Each PMG has a single reserve PM with base 1-unit input and output, and
#     scripted building modifiers choose direction and scale actual flow.
#   - building_strategic_reserve_silo: buildable in any state, passive. Only
#     provides additional capacity. Useless without a hub, but a hub alone
#     caps out quickly so silos are the way to scale reserves up.
# ============================================================================

# ---------------- HUB ----------------
building_strategic_reserve_hub = {
	building_group = bg_public_infrastructure
	city_type = city
	levels_per_mesh = -1

	has_max_level = yes

	# A standing national asset: neither the player nor the AI can downsize or
	# demolish it. The one way to lose it is to lose its state to another
	# nation (st_res_on_state_owner_change demolishes a captured hub; script's
	# remove_building is unaffected by this flag).
	downsizeable = no

	unlocking_technologies = {
		logistics
	}

	production_method_groups = {
		pmg_st_res_hub
	}

	icon = "gfx/interface/icons/building_icons/building_government_administration.dds"
	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"

	required_construction = construction_cost_very_high

	possible = {
		is_capital = yes
		# One hub per country. A hub cannot be demolished, so a country that
		# moves its capital keeps the old one (it works from any state) rather
		# than adding a second: every st_res effect reads "the" hub through
		# random_scope_building. Max level is 1, so this never blocks an
		# expansion either.
		custom_tooltip = {
			text = st_res_hub_one_per_country_tt
			owner = {
				NOT = {
					any_scope_building = { is_building_type = building_strategic_reserve_hub }
				}
			}
		}
	}

	# GOVERNMENT_BUILDING_BASE_VALUE (1000) is what a government building with
	# no ai_value gets; this one used to script 50, or 200 at war, which the AI
	# read as nearly worthless. 2000 is vanilla's "higher base value" tier
	# (naval administration). A depot is built for the next war, so peacetime
	# counts double: one begun during a war finishes too late to fill before
	# the peace, when barracks already take every round the market makes.
	ai_value = {
		value = 0
		if = {
			limit = {
				owner = {
					is_great_or_major_power = yes
				}
			}
			add = 1000
			if = {
				limit = {
					owner = { is_at_war = no }
				}
				add = 1000
			}
		}
	}
}

# ---------------- SILO ----------------
# Passive capacity-only building. One PMG with a single idle PM that adds
# capacity for all three goods at each level. Expandable up to a high cap
# so large empires can pile up reserves.
building_strategic_reserve_silo = {
	building_group = bg_public_infrastructure
	city_type = city
	levels_per_mesh = 3

	unlocking_technologies = {
		logistics
	}

	production_method_groups = {
		pmg_st_res_silo_capacity
	}

	icon = "gfx/interface/icons/building_icons/building_government_administration.dds"

	required_construction = construction_cost_high

	# Storage is worth building when the reserve is running out of room: at
	# the government-building default (1000) while any good stands at 75% of
	# capacity (st_res_reserve_filling_up), and next to nothing otherwise. A
	# flat value with no such signal would have the AI raise silos without
	# end, since, unlike a bureaucracy or a barracks, nothing in code tells
	# it how much storage it needs. A war drains the reserve, so the value
	# falls on its own while one is on.
	ai_value = {
		value = 0
		if = {
			limit = {
				owner = {
					any_scope_building = {
						is_building_type = building_strategic_reserve_hub
					}
				}
			}
			add = 25
			if = {
				limit = {
					owner = { st_res_reserve_filling_up = yes }
				}
				add = 1000
			}
		}
	}
}
