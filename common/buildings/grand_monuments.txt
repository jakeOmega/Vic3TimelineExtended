#==================================================
# Grand Monuments
#==================================================
# A monument a regime raises to what it stands for: its crown, republic,
# revolution, leader, faith or nation. Expensive to build, cheap to run,
# repeatable (one per state, levels uncapped). Its level is its GRANDEUR; every
# effect grows with grandeur along the doubling curve, and national effects
# come from national totals on the Monuments journal entry, never from each
# building separately.
# Design: docs/superpowers/specs/2026-09-27-grand-monument-rework-design.md.
#
# Group is bg_grand_monuments, NOT bg_monuments -- see common/building_groups/
# extra_building_groups.txt for why that separation is load-bearing.

building_grand_monument = {
	building_group = bg_grand_monuments
	icon = "gfx/interface/icons/building_icons/building_grand_monument.dds"
	city_type = city
	levels_per_mesh = -1

	expandable = yes
	# Neither the player nor the AI can downsize or demolish a monument from
	# the building panel. To the AI's economic code it is pure upkeep with no
	# output, and AI countries were demolishing theirs. Every way a monument
	# goes or shrinks is script's, through remove_building, which this flag
	# does not block: Pull Down and Rededicate on a contested monument
	# (gm_state_tear_down / gm_state_rededicate), a trophy taking, and a
	# commission's rebuild. Every vanilla monument and the Strategic Reserve
	# hub set the same flag.
	downsizeable = no
	unique = no
	ai_nationalization_desire = 0

	production_method_groups = {
		pmg_monument_dedication
		pmg_maintenance
	}

	required_construction = construction_cost_grand_monument

	# A finished monument takes no more levels of its own: more found a new
	# monument beside it (spec 2026-10-16 §3.2), except while it is contested,
	# a ceremony or naming is under way, script is rebuilding it, or six
	# monuments stand (gm_state_can_expand, state scope). Monuments are
	# government-funded; private investors are held to the same gate.
	can_build_government = {
		gm_state_can_expand = yes
	}
	can_build_private = {
		gm_state_can_expand = yes
	}

	# The game rule (spec §7).
	possible = {
		custom_tooltip = {
			text = gm_possible_rule_tt
			gm_system_enabled = yes
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"

	# Evaluated in STATE scope: ROOT is the state, owner reaches the country.
	# owner ?= : the AI can ask while a state's owner is briefly invalid
	# ("Event target link 'owner' returned an invalid object").
	ai_value = {
		value = 0

		# Only interesting to countries with money to spare.
		if = {
			limit = {
				owner ?= { is_great_or_major_power = yes }
			}
			add = 25
		}

		# The Tourism Industry throughput pays off where one exists.
		if = {
			limit = { has_building = building_tourism_industry }
			add = 50
		}

		# A regime short of legitimacy is the one that raises monuments (§7).
		if = {
			limit = {
				owner ?= { government_legitimacy < 40 }
			}
			add = 25
		}

		# Never at the expense of the war effort.
		if = {
			limit = {
				owner ?= { is_at_war = yes }
			}
			add = -100
		}

		# Every monument already standing here: found new ones elsewhere first
		# (spec 2026-10-16 §3.2).
		add = {
			value = gm_state_monument_count
			multiply = -25
		}

		# Never in hard times: a level finished then costs legitimacy (§5).
		if = {
			limit = {
				owner ?= { gm_country_hard_times = yes }
			}
			add = -100
		}
	}
}
