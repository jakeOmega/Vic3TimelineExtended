#==================================================
# Grand Monuments
#==================================================
# A monument a regime raises to what it stands for: its crown, republic,
# revolution, leader, faith or nation. Expensive to build, cheap to run,
# repeatable. Every effect grows with grandeur along the doubling curve, and
# national effects come from national totals on the Monuments journal entry,
# never from each building separately.
# Design: docs/superpowers/archive/specs/2026-09-27-grand-monument-rework-design.md.
#
# The FINISHED monument (sites spec 2026-10-18 §3.1): always level 1, made only
# by script, at dedication, on its dedication. Its size lives in its records
# (gm_levels, gm_start_grandeur, gm_age); a player builds a
# building_grand_monument_site (grand_monument_site.txt) and dedicates it. A
# state holds one; the monuments beside it are containers
# (gm_monument_effects.txt).
#
# Group is bg_grand_monuments, NOT bg_monuments -- see common/building_groups/
# extra_building_groups.txt for why that separation is load-bearing.

building_grand_monument = {
	building_group = bg_grand_monuments
	icon = "gfx/interface/icons/building_icons/building_grand_monument.dds"
	city_type = city
	levels_per_mesh = -1

	# Script makes it (create_building, as vanilla makes the Manila cathedral's
	# ruins); no player or AI builds, expands, downsizes or demolishes it. Every
	# way a monument goes is script's, through remove_building, which these
	# flags do not block: Pull Down, a whole taking, the last monument here
	# gone. Every vanilla monument and the Strategic Reserve hub set
	# downsizeable = no.
	buildable = no
	expandable = no
	downsizeable = no
	unique = no
	ai_nationalization_desire = 0

	production_method_groups = {
		pmg_monument_dedication
		pmg_maintenance
	}

	required_construction = construction_cost_grand_monument

	# The game rule (spec §7).
	possible = {
		custom_tooltip = {
			text = gm_possible_rule_tt
			gm_system_enabled = yes
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}
