# ============================================================================
# SOLAR SYSTEM COLONIZATION — Space Command (fleets design § 5)
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-26-solar-colonization-fleets-design.md.
# A government building, under Full Colonies: a nation's first is built where
# the Space Program stands, and the one it has expands where it is (possible
# below; one a nation, sr_sc_one_per_nation). Its methods (sr_space_command_pms.txt) carry the fleet strength a
# level builds a month as a workforce-scaled state modifier
# (state_sr_ship_construction_add), which sr_sc_rate sums over the states the
# country controls; the month's construction joins the home fleet at Earth
# (sr_sc_month, sr_fleet_effects.txt). The fleets' upkeep is a modifier on
# this building that multiplies its maintenance line (sr_fl_upkeep_refresh).
#
# bg_government, not bg_military: vanilla's bg_military is the parent of the
# army and navy groups (barracks and naval bases, is_military = yes, tied to
# formations and conscription) and has no building of its own; the mod's
# Military Base is bg_government too. An AI government builds no further than
# sr_sc_may_build_government allows (2 + its colonies, at most 10 levels).
# ============================================================================

building_space_command = {
	building_group = bg_government
	icon = "gfx/interface/icons/building_icons/building_space_command.dds"
	city_type = city
	levels_per_mesh = -1

	unlocking_technologies = {
		space_militarization
	}

	production_method_groups = {
		pmg_space_command_yards
	}

	potential = {
		sr_colonies_full_enabled = yes
	}

	# A new Space Command needs the Space Program's state, and a nation runs one
	# (owner, 2026-10-31): it's refused while the owner has one of a level or
	# more anywhere, with the reason in the build menu, so moving the capital
	# doesn't build one the sweep would dismantle (sr_sc_one_per_nation, which
	# stays for annexation). One it has expands wherever it stands, an annexed
	# one included (review of #1287, I2): otherwise an AI that kept an annexed
	# one would be held to its levels for good, and keep queuing new ones in
	# its capital that the sweep removes. Building possible is state scope.
	possible = {
		trigger_if = {
			limit = { has_building = building_space_command }
			always = yes
		}
		trigger_else = {
			has_building = building_space_program
			custom_tooltip = {
				text = sr_sc_one_per_nation_tt
				owner ?= {
					NOT = {
						any_scope_building = {
							is_building_type = building_space_command
							level >= 1
						}
					}
				}
			}
		}
	}

	required_construction = construction_cost_very_high

	can_build_government = {
		owner ?= { sr_sc_may_build_government = yes }
	}

	can_build_private = {
		always = no
	}

	ai_value = {
		value = 2000
		if = {
			limit = {
				owner ?= {
					OR = {
						is_at_war = yes
						sr_has_space_colony = yes
					}
				}
			}
			add = 2000
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_military.dds"
}
