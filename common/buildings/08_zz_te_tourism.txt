# ============================================================================
# National Park and Tourism Industry
# ============================================================================
# This file's name sets where its buildings appear in the building list,
# whose default sort is the order building definitions load: file names in
# ASCII order, vanilla's and the mod's in one list. 08_zz_te_tourism.txt sorts
# right after vanilla's 08_monuments.txt, so these buildings follow its monuments.
# Keep them bare definitions, not REPLACE_OR_CREATE: a directive applies
# after every bare definition, so the entry it creates may land at the end.
# Rule: docs/guides/scripting_best_practices.md § Load Order (probed
# 2026-10-26), "Building list order"; test_building_list_order.py.
# ============================================================================

building_national_park = {
	building_group = bg_parks
	icon = "gfx/interface/icons/building_icons/building_national_park.dds"
	city_type = city
	levels_per_mesh = -1

	has_max_level = yes

	should_auto_expand = {
		occupancy >= 0.8
	}

	production_method_groups = {
		pmg_base_building_national_park
	}

	required_construction = construction_cost_very_high

	can_build_government = {
		always = yes
	}

	can_build_private = {
		always = no
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"

	ai_value = {
		value = 1000
		# Strongly preferred under environmental governance
		if = {
			limit = { owner = { has_law = law_type:law_ministry_of_the_environment } }
			add = 4000
		}
		# Strongly preferred when this state is heavily polluted
		if = {
			limit = { state_region = { pollution_amount > 150 } }
			add = 4000
		}
		# Modest pull when polluted at all
		if = {
			limit = {
				state_region = { pollution_amount > 50 }
				NOT = { state_region = { pollution_amount > 150 } }
			}
			add = 2000
		}
		# Tourism synergy: parks drive tourism revenue where the tourism industry exists
		if = {
			limit = { has_building = building_tourism_industry }
			add = 1500
		}
		# Wartime countries deprioritize parks
		if = {
			limit = { owner = { is_at_war = yes } }
			add = -2000
		}
	}
}

building_tourism_industry = {
	building_group = bg_arts
	icon = "gfx/interface/icons/building_icons/building_tourism_industry.dds"
	city_type = city

	ai_nationalization_desire = 0.5 # AI generally does not want to privatize tourism industries

	unlocking_technologies = {
		romanticism
	}

	production_method_groups = {
		pmg_tourism
		pmg_maintenance
	}

	required_construction = construction_cost_very_high
	ownership_type = self

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_heavy_industry.dds"
}
