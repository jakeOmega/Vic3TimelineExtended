# ============================================================================
# Industries the mod adds: synthetics and electrics variants, then high technology
# ============================================================================
# This file's name sets where its buildings appear in the building list,
# whose default sort is the order building definitions load: file names in
# ASCII order, vanilla's and the mod's in one list. 01_zz_te_industry.txt sorts
# right after vanilla's 01_industry.txt, so these buildings follow its industries (after Munitions Plants).
# Keep them bare definitions, not REPLACE_OR_CREATE: a directive applies
# after every bare definition, so the entry it creates may land at the end.
# Rule: docs/guides/scripting_best_practices.md § Load Order (probed
# 2026-10-26), "Building list order"; test_building_list_order.py.
# ============================================================================

building_synthetics_plant_opium = {
	building_group = bg_biotechnology
	icon = "gfx/interface/icons/building_icons/building_synthetics_plant_opium.dds"
	city_type = city
	levels_per_mesh = -1

	# Base-game Pharmaceuticals (era 2). The early methods lose to opium
	# plantations; see the ladder comment in extra_pms.txt.
	unlocking_technologies = {
		pharmaceuticals
	}

	production_method_groups = {
		pmg_synthetic_opium
		pmg_automation_building_chemical_plant
		pmg_maintenance
		pmg_carbon_capture_synthetic_opium
	}

	required_construction = construction_cost_very_high

	possible = {
		is_industrial_production_allowed = yes
	}
	# Drugs are the opium good: a ban on it (Qing's Opium Crisis) stops these
	# as it stops opium plantations.
	can_build_government = {
		owner = {
			NOT = { is_banning_goods = g:opium }
		}
	}
	can_build_private = {
		owner = {
			NOT = { is_banning_goods = g:opium }
		}
	}
	ownership_type = self
	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}
