# Repeatable "Grand Monument" buildings (see common/buildings/grand_monuments.txt).
# Deliberately a TOP-LEVEL group with NO parent_group: two existing systems iterate
# bg_monuments and assume its members are unique one-off wonders --
#   * tourism.txt / tourism_throughput_from_monuments  (+0.25 per bg_monuments building)
#   * cultural_hegemony_script_values.txt / cultural_pull_from_monuments (+3 each, uncapped)
# An infinitely repeatable member would break both, so Grand Monuments stay outside
# bg_monuments and contribute to those systems on their own capped terms instead.
# NOT is_government_funded: the treasury paying the whole goods input would wreck the
# deliberately-long tourism payback.
bg_grand_monuments = {
	category = urban

	lens = development

	always_possible = yes

	economy_of_scale = no

	urbanization = 10
	infrastructure_usage_per_level = 1

	foreign_investment_ai_factor = 0
}
