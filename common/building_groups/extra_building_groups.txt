# The clean generators: hydro, nuclear, fusion and renewable plants. A child of
# bg_power, so every bg_power modifier (infrastructure usage, throughput, the
# electrification credit's construction line, collectivization) still reaches
# them, and only the conventional Power Plant stays in bg_power alone. It exists
# so a modifier can target the clean plants without the coal and oil one:
# construction speed has no per-building form, only a per-group one
# (building_group_<bg>_construction_efficiency_add), which Renewable Investment
# uses. The properties below are the ones every vanilla child group restates
# (bg_light_industry: lens, urbanization, economy_of_scale_ai_factor,
# infrastructure_usage_per_level), at bg_power's values, so the plants are
# built, spread and loaded onto infrastructure exactly as before.
REPLACE_OR_CREATE:bg_clean_power = {
	parent_group = bg_power

	lens = development

	urbanization = 10
	economy_of_scale_ai_factor = 1.5
	infrastructure_usage_per_level = 2

	construction_efficiency_modifier = yes
}

# Repeatable "Grand Monument" buildings (see common/buildings/grand_monuments.txt).
# Deliberately a TOP-LEVEL group with NO parent_group: two existing systems iterate
# bg_monuments and assume its members are unique one-off wonders --
#   * tourism.txt / tourism_bonus_from_monuments  (+10% output per bg_monuments building, more for the 1836 wonders)
#   * cultural_hegemony_script_values.txt / cultural_pull_from_monuments (+3 each, uncapped)
# An infinitely repeatable member would break both, so Grand Monuments stay outside
# bg_monuments and contribute to those systems on their own capped terms instead.
#
# is_government_funded, like bg_government (Government Administration) and the mod's own
# bg_parks: a monument is raised by the state, not by investors. The treasury pays the
# pmg_maintenance goods input and no private or foreign investor can put money into one.
# That deepens the intended construction *sink* -- the upkeep is now a permanent line on
# the budget rather than something the building earns back on its own books -- which is
# the point: this is civic spending, not an investment.
bg_grand_monuments = {
	category = urban

	lens = development

	always_possible = yes

	is_government_funded = yes

	economy_of_scale = no

	urbanization = 10
	infrastructure_usage_per_level = 1

	foreign_investment_ai_factor = 0
}
