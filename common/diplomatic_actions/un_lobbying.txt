# ============================================================================
# RESOLUTION LOBBYING — pledges ("Seek a Vote Commitment", for or against)
# ============================================================================
# docs/superpowers/specs/2026-09-27-un-vote-lobbying-design.md §4. Any member
# may ask any other member with a vote to pledge its vote on the resolution in
# session, for it or against it, and pays with an obligation owed to the
# member that pledges. Two actions, one per direction, sharing every rule but
# the side and the flipped weight of the member's ties to the target.
#
# Diplomatic actions rather than journal-entry buttons, for three reasons that
# are all engine capabilities rather than preferences:
#
#   1. A scripted button cannot prompt for a free choice of country. A
#      diplomatic action is targeted through the ordinary diplomacy UI, so the
#      asker picks whichever member it wants. (The chamber's Delegations rows
#      also offer the ask for an AI member, resolved at once through
#      would_accept_diplomatic_action — the same accept_score as here.)
#   2. `requires_approval = yes` gives a HUMAN recipient the engine's own
#      accept/decline request — the design requirement "human-controlled
#      recipients receive a choice" — with no popup event of our own.
#   3. The `ai.accept_score` block is itemised: every `add` carries a `desc`,
#      and the engine renders the whole list as the acceptance tooltip. That is
#      the requirement "display the main reasons for support or opposition
#      without guaranteeing an uncertain vote", for free and always in step
#      with the numbers the AI actually uses.
#
# Structure copied from vanilla's pactless, approval-requiring action
# game/common/diplomatic_actions/03_violate_sovereignty.txt (no `pact` block =
# no pact is created, per game/common/diplomatic_actions/diplomatic_action.md).
# Conventions (groups, custom_tooltip'd `possible` clauses, loc key families)
# follow the mod's own covert-warfare actions in
# common/diplomatic_actions/covert_operations.txt.
#
# SCOPES, which flip between blocks and are easy to get wrong:
#   selectable                  ROOT = us, no target at all
#   potential / possible        ROOT = the asker, scope:target_country = the
#                               prospective recipient
#   accept_effect/decline_effect  same as possible
#   ai.will_propose             ROOT = the AI asker, scope:target_country =
#                               the recipient
#   ai.accept_score             ROOT = the AI RECIPIENT, scope:actor = the
#                               asker (diplomatic_action.md: "the AI country
#                               is always root")
#
# THE WINDOW. A member can be asked until it votes. An AI member votes at
# month 9, 10 or 11 of the session (un_vote_ai_draws), so there is most of a
# year to ask it; a human member can be asked until it casts its ballot.
#
# The influence campaigns (§3) are the two pacts at the end of this file.
# ============================================================================

# ---- Pledge FOR ------------------------------------------------------------------
un_secure_commitment_action = {
	groups = {
		general
	}

	requires_approval = yes
	show_confirmation_box = yes
	show_effect_in_tooltip = yes
	is_hostile = no
	should_notify_third_parties = no

	# The engine's own "spend an obligation the target owes us to force this
	# through" is switched OFF. This deal is the mirror of that — WE take on the
	# obligation — and letting both run at once would let an asker cash in one
	# favour to buy another, which is not a trade anybody could read.
	can_use_obligations = no # REVIEWED 2026-09-18: diplomatic-action field, not an effect/trigger — vanilla violate_sovereignty uses it (game/common/diplomatic_actions/03_violate_sovereignty.txt:8). effect_trigger_validity_audit has no schema for this entity type.

	# Asker-side only: evaluated once for us, not per target, so it is the
	# right place for the structural gates. Any member may ask (spec §4).
	selectable = {
		has_game_rule = united_nations_enabled
		has_global_variable = un_active_resolution
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		NOT = { un_representation_suspended = yes }
		global_var:un_active_resolution ?= { has_tag = un_res_voting }
	}

	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			# A member without a vote has none to promise: Article 19, or
			# suspended representation (§0.8).
			NOT = { un_dues_vote_suspended = yes }
			NOT = { un_representation_suspended = yes }
		}
	}

	# Every clause carries its own custom_tooltip so a greyed-out action says
	# which rule it is failing, the way the covert-warfare actions do. Written
	# out in both actions rather than behind a scripted trigger, so each
	# clause renders as its own line.
	possible = {
		custom_tooltip = {
			text = un_lobby_not_the_target_tt
			global_var:un_active_resolution ?= {
				NOT = {
					AND = {
						has_variable = un_res_target
						var:un_res_target ?= scope:target_country
					}
				}
			}
		}
		custom_tooltip = {
			text = un_lobby_has_not_voted_tt
			global_var:un_active_resolution ?= {
				NOT = { un_lobby_has_voted = { COUNTRY = scope:target_country } }
			}
		}
		custom_tooltip = {
			text = un_lobby_not_pledged_tt
			global_var:un_active_resolution ?= {
				NOT = { un_pledge_made_by = { COUNTRY = scope:target_country } }
			}
		}
		custom_tooltip = {
			text = un_lobby_not_yet_approached_tt
			scope:target_country = {
				NOT = { un_pledge_asked_by = { ASKER = ROOT } }
			}
		}
		custom_tooltip = {
			text = un_lobby_under_cap_tt
			un_pledge_under_cap = yes
		}
		custom_tooltip = {
			text = un_lobby_no_obligation_owed_tt
			NOT = { owes_obligation_to = scope:target_country }
		}
		is_revolutionary = no
		scope:target_country = { is_revolutionary = no }
	}

	accept_effect = {
		custom_tooltip = {
			text = un_lobby_commitment_struck_tt
		}
		un_lobby_accept_commitment = { LIST = yes DIR = 1 }
	}

	decline_effect = { # REVIEWED 2026-09-18: diplomatic-action field, not an effect/trigger — vanilla violate_sovereignty uses it (game/common/diplomatic_actions/03_violate_sovereignty.txt:66). effect_trigger_validity_audit has no schema for this entity type.
		# The ask is still booked: one ask per asker per member per
		# resolution, and a refusal is an answer.
		un_lobby_record_attempt = yes
	}

	ai = {
		# Rarely. An obligation — above all a great power's — is one of the
		# strongest things a country can hand another, so the AI offers one
		# only for its own resolution (spec §5: the proposer lobbies for), and
		# more readily when it has a real stake in it: its own military
		# mandate, or a binding resolution aimed at one of its rivals.
		evaluation_chance = {
			value = 0
			if = {
				limit = { un_lobby_ai_side_for = yes }
				add = 0.02
				if = {
					limit = {
						global_var:un_active_resolution ?= {
							OR = {
								has_tag = un_topic_military_mandate
								AND = {
									un_resolution_is_binding = yes
									has_variable = un_res_target
									var:un_res_target ?= {
										has_diplomatic_pact = { who = ROOT type = rivalry }
									}
								}
							}
						}
					}
					add = 0.03
				}
			}
		}

		# Only toward a plausible recipient close to the line: a member whose
		# lean is between leaning against and undecided, which a +100 pledge
		# carries. No is_ai test: these are the same rules a human asker plays
		# by, they just also decide whether the AI bothers.
		will_propose = {
			NOT = { has_diplomatic_pact = { who = scope:target_country type = rivalry } }
			has_diplomatic_relevance = scope:target_country
			NOT = { scope:target_country.relations:root < relations_threshold:cordial }
			scope:target_country = { un_lobby_true_lean_in = { LOW = -25 HIGH = 9 } }
		}

		propose_score = {
			value = 10
		}

		# ROOT is the RECIPIENT here and scope:actor is the asker. Positive
		# means accept. Each factor is a line in the acceptance tooltip the
		# recipient sees.
		accept_score = {
			value = 0

			# Pledging a vote away in advance is a real cost: the deal starts
			# under water and the obligation has to lift it out.
			add = {
				desc = "UN_LOBBY_ACCEPT_BASE"
				value = -20
			}

			# What is being offered. The engine's own diplomatic obligation,
			# redeemable at a moment of our choosing.
			add = {
				desc = "UN_LOBBY_ACCEPT_OBLIGATION"
				value = 40
			}

			# Relations with the country asking.
			add = {
				desc = "UN_LOBBY_ACCEPT_RELATIONS"
				if = {
					limit = { "relations(scope:actor)" < relations_threshold:cordial }
					value = -30
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:cordial
						"relations(scope:actor)" < relations_threshold:amicable
					}
					value = 10
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:amicable
						"relations(scope:actor)" < relations_threshold:friendly
					}
					value = 25
				}
				if = {
					limit = { "relations(scope:actor)" >= relations_threshold:friendly }
					value = 40
				}
			}

			# A rival does not do favours.
			if = {
				limit = { has_diplomatic_pact = { who = scope:actor type = rivalry } }
				add = {
					desc = "UN_LOBBY_ACCEPT_RIVALRY"
					value = -60
				}
			}

			# Our ties to the country the resolution is aimed AT — skipped for aid and
			# peacekeeping requests, whose "target" is the country being helped (a
			# rival of the requester is not eager to help it). Written from
			# the target's side with ROOT as the other end, because these three
			# relationships are symmetric and that avoids chaining a scope
			# through a container variable inside a script value.
			if = {
				limit = {
					global_var:un_active_resolution ?= {
						NOT = { un_resolution_target_is_beneficiary = yes }
						var:un_res_target ?= { has_treaty_alliance_with = { TARGET = ROOT } }
					}
				}
				add = {
					desc = "UN_LOBBY_ACCEPT_TARGET_ALLY"
					value = -60
				}
			}
			else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						NOT = { un_resolution_target_is_beneficiary = yes }
						var:un_res_target ?= { is_in_same_power_bloc = ROOT }
					}
				}
				add = {
					desc = "UN_LOBBY_ACCEPT_TARGET_BLOC"
					value = -30
				}
			}
			else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						NOT = { un_resolution_target_is_beneficiary = yes }
						var:un_res_target ?= { has_diplomatic_pact = { who = ROOT type = rivalry } }
					}
				}
				add = {
					desc = "UN_LOBBY_ACCEPT_TARGET_RIVAL"
					value = 25
				}
			}

			# The asker's record in the organisation. The same tier triggers the
			# Assembly's own vote lean reads, so a member that has built a record
			# finds its requests easier to place — and a disgraced one finds them
			# harder.
			if = {
				limit = { scope:actor = { un_standing_tier_exemplary = yes } }
				add = {
					desc = "UN_LOBBY_ACCEPT_STANDING"
					value = 15
				}
			}
			else_if = {
				limit = { scope:actor = { un_standing_tier_respected = yes } }
				add = {
					desc = "UN_LOBBY_ACCEPT_STANDING"
					value = 8
				}
			}
			else_if = {
				limit = { scope:actor = { un_standing_tier_poor = yes } }
				add = {
					desc = "UN_LOBBY_ACCEPT_STANDING"
					value = -8
				}
			}
			else_if = {
				limit = { scope:actor = { un_standing_tier_disgraced = yes } }
				add = {
					desc = "UN_LOBBY_ACCEPT_STANDING"
					value = -15
				}
			}
		}
	}
}

# ---- Campaign: lobby for (spec §3) -------------------------------------------
# A one-sided pact, like vanilla's Improve Relations: our delegation works on
# an AI member for as long as we keep it running. Each full month moves the
# member's lean 3 points in favour, up to 15; campaigns one way on the same
# member stack to 20, and opposing ones net out (un_lobby_effects.txt,
# § CAMPAIGNS). While it runs we read the member's true position in the
# chamber. It ends when the member votes, when the resolution closes, or when
# we cancel it — and a cancelled campaign's gain goes with it.
#
# Cost: a flat 100 influence. The spec asked for 75 / 100 / 150 by the
# member's rank if a pact's cost could be a script value; every vanilla pact
# cost is a literal and the engine's reading of anything else cannot be
# checked outside the game, so the fallback it named is used.
#
# The AI never takes this action through the engine: its campaigns are chosen,
# started and ended in script each month (un_lobby_ai_monthly), which creates
# the pact directly. evaluation_chance 0 keeps the engine from proposing it,
# and will_break always-no keeps the engine from breaking one script started.
un_lobby_for_action = {
	texture = "gfx/interface/icons/diplomatic_action_icons/un_lobby_for_action.dds"
	groups = {
		general
	}

	requires_approval = no
	show_confirmation_box = yes
	show_effect_in_tooltip = yes
	is_hostile = no
	should_notify_third_parties = no

	selectable = {
		un_lobby_campaign_selectable = yes
	}

	# Only AI members: a human member decides for itself.
	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			is_ai = yes
			NOT = { is_country_type = decentralized }
			je:je_united_nations ?= { has_modifier = un_member_modifier }
		}
	}

	possible = {
		custom_tooltip = {
			text = un_lobby_campaign_member_votes_tt
			scope:target_country = {
				NOT = { un_dues_vote_suspended = yes }
				NOT = { un_representation_suspended = yes }
			}
		}
		custom_tooltip = {
			text = un_lobby_not_the_target_tt
			global_var:un_active_resolution ?= {
				NOT = {
					AND = {
						has_variable = un_res_target
						var:un_res_target ?= scope:target_country
					}
				}
			}
		}
		custom_tooltip = {
			text = un_lobby_has_not_voted_tt
			global_var:un_active_resolution ?= {
				NOT = { un_lobby_has_voted = { COUNTRY = scope:target_country } }
			}
		}
		custom_tooltip = {
			text = un_lobby_campaign_not_both_ways_tt
			NOT = { has_diplomatic_pact = { who = scope:target_country type = un_lobby_against_action } }
		}
	}

	accept_effect = {
		custom_tooltip = un_lobby_campaign_for_effect_tt
		hidden_effect = {
			un_lobby_campaign_start = { DIR = for }
		}
	}

	pact = {
		cost = 100
		is_two_sided_pact = no
		show_in_outliner = yes

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = un_lobby_campaign_maintain_tt
					un_lobby_campaign_live_ok = yes
				}
			}
		}

		manual_break_effect = {
			un_lobby_campaign_end = { DIR = for }
		}
		auto_break_effect = {
			un_lobby_campaign_end = { DIR = for }
		}
	}

	ai = {
		evaluation_chance = {
			value = 0
		}
		will_propose = { always = no }
		will_break = { always = no }
		propose_score = {
			value = 0
		}
	}
}

# ---- Campaign: lobby against (spec §3) -------------------------------------------
# A one-sided pact, like vanilla's Improve Relations: our delegation works on
# an AI member for as long as we keep it running. Each full month moves the
# member's lean 3 points against, up to 15; campaigns one way on the same
# member stack to 20, and opposing ones net out (un_lobby_effects.txt,
# § CAMPAIGNS). While it runs we read the member's true position in the
# chamber. It ends when the member votes, when the resolution closes, or when
# we cancel it — and a cancelled campaign's gain goes with it.
#
# Cost: a flat 100 influence. The spec asked for 75 / 100 / 150 by the
# member's rank if a pact's cost could be a script value; every vanilla pact
# cost is a literal and the engine's reading of anything else cannot be
# checked outside the game, so the fallback it named is used.
#
# The AI never takes this action through the engine: its campaigns are chosen,
# started and ended in script each month (un_lobby_ai_monthly), which creates
# the pact directly. evaluation_chance 0 keeps the engine from proposing it,
# and will_break always-no keeps the engine from breaking one script started.
un_lobby_against_action = {
	texture = "gfx/interface/icons/diplomatic_action_icons/un_lobby_against_action.dds"
	groups = {
		general
	}

	requires_approval = no
	show_confirmation_box = yes
	show_effect_in_tooltip = yes
	is_hostile = no
	should_notify_third_parties = no

	selectable = {
		un_lobby_campaign_selectable = yes
	}

	# Only AI members: a human member decides for itself.
	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			is_ai = yes
			NOT = { is_country_type = decentralized }
			je:je_united_nations ?= { has_modifier = un_member_modifier }
		}
	}

	possible = {
		custom_tooltip = {
			text = un_lobby_campaign_member_votes_tt
			scope:target_country = {
				NOT = { un_dues_vote_suspended = yes }
				NOT = { un_representation_suspended = yes }
			}
		}
		custom_tooltip = {
			text = un_lobby_not_the_target_tt
			global_var:un_active_resolution ?= {
				NOT = {
					AND = {
						has_variable = un_res_target
						var:un_res_target ?= scope:target_country
					}
				}
			}
		}
		custom_tooltip = {
			text = un_lobby_has_not_voted_tt
			global_var:un_active_resolution ?= {
				NOT = { un_lobby_has_voted = { COUNTRY = scope:target_country } }
			}
		}
		custom_tooltip = {
			text = un_lobby_campaign_not_both_ways_tt
			NOT = { has_diplomatic_pact = { who = scope:target_country type = un_lobby_for_action } }
		}
	}

	accept_effect = {
		custom_tooltip = un_lobby_campaign_against_effect_tt
		hidden_effect = {
			un_lobby_campaign_start = { DIR = against }
		}
	}

	pact = {
		cost = 100
		is_two_sided_pact = no
		show_in_outliner = yes

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = un_lobby_campaign_maintain_tt
					un_lobby_campaign_live_ok = yes
				}
			}
		}

		manual_break_effect = {
			un_lobby_campaign_end = { DIR = against }
		}
		auto_break_effect = {
			un_lobby_campaign_end = { DIR = against }
		}
	}

	ai = {
		evaluation_chance = {
			value = 0
		}
		will_propose = { always = no }
		will_break = { always = no }
		propose_score = {
			value = 0
		}
	}
}
