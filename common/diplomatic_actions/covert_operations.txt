# ============================================================================
# COVERT WARFARE — Covert Operation Diplomatic Actions
# ============================================================================
# These diplomatic actions function like increase_relations: togglable pacts
# that persist until the player breaks them or conditions change.
# Core effects are applied via pact modifiers (auto-active while pact exists).
# Complex state/movement-scoped effects supplement pact modifiers via JE pulse.
# All ops auto-cancel when funding drops to 0 (requirement_to_maintain).
#
# LAUNCH PREVIEW: each accept_effect shows what its operation does, at base
# strength (established, priority 1). show_as_tooltip renders the same static
# modifiers covert_ops_apply_all_phase_effects applies, unscaled and with no
# duration; nothing in it runs. It lists every field, script_only ones
# included. The <type>_extra_tt lines after it state what the engine cannot
# list: modifiers on movements or states, and conditional parts.
# covert_op_phase_warning_tt then says how phase and priority scale all of it.
# In game the scope:target_country block renders after the root-level lines
# even though it is scripted before them, so no line here may say "above" or
# "below". test_covert_launch_preview.py keeps the preview in step with the
# pulse.
#
# TECHNOLOGY GATES: seven operations need a technology before they can be
# launched, the first custom_tooltip in their possible block. The tech is
# chosen so the operations that can break a great power (financial
# subversion, destabilization, regime change) sit deep in the tree, where a
# small country can't rush them before the great powers have built their own
# services. Launch only, like the Tradecraft unlocks. test_covert_tech_gates.py
# holds the table and checks the tooltips and descriptions against it.
#
# AI DESIGN:
# - Low evaluation_chance (0.02–0.05) — strategic long-term decisions, not
#   re-evaluated every tick. Prevents spammy on/off toggling.
# - will_propose guards: country rank, financial health (in_default = no),
#   attitude/rivalry requirement, IC advantage for some ops.
# - propose_score scales with rivalry, rank, ruler personality.
# - Wartime ops have higher eval chance — urgency justifies faster decisions.
# - Ruler personality: aggressive rulers more likely, cautious less.
# - Target size: upkeep scales with the operator's GDP, so an operation whose
#   payoff scales with the target's size takes covert_ai_target_worth_the_upkeep
#   (covert_warfare_triggers.txt): no much weaker target without money to burn.
#   Espionage, nuclear programme sabotage and secure loose material don't take
#   it; test_covert_ai_targets.py holds the two lists.
# - Breaking: an operation pays nothing for six months, so every action spells
#   out will_break (none is left to the engine's default). The AI keeps an
#   operation until it is fully operational (covert_ai_op_settled), then ends
#   it only once its reason has gone, judged more loosely than at launch:
#   hostility (covert_ai_hostility_gone), nothing left to steal, no milestone
#   of our own. Wartime operations, influence campaign and secure loose
#   material are never called off by choice; their requirement_to_maintain
#   ends them. Cultivate assets also stops once its network reaches
#   covert_cultivate_ai_net_done. Launch-only tests (free slot, intelligence
#   capacity, target size, debt) never break anything.
# ============================================================================

# ---- PEACETIME: Election Interference ----
# Undermine a target's political stability through covert influence campaigns
covert_election_interference_action = {
	texture = "gfx/interface/icons/diplomatic_action_icons/covert_election_interference_action.dds"
	groups = {
		general
	}
	requires_approval = no
	show_confirmation_box = yes
	show_effect_in_tooltip = yes
	is_hostile = yes
	should_notify_third_parties = no

	accept_effect = {
		custom_tooltip = {
			text = covert_op_preview_header_tt
		}
		show_as_tooltip = {
			scope:target_country = {
				add_modifier = { name = covert_election_interference }
			}
		}
		custom_tooltip = {
			text = covert_op_phase_warning_tt
		}
		custom_tooltip = {
			text = covert_op_detection_warning_tt
		}
		covert_op_start = { TYPE = election_interference DEFENSE_MOD = country_covert_defense_ideological_add CODE = 0 }
	}

	potential = {
		has_game_rule = covert_warfare_enabled
	}

	possible = {
		custom_tooltip = {
			text = covert_op_tech_election_interference_tt
			has_technology_researched = television_broadcasting
		}
		custom_tooltip = {
			text = iw_funding_not_zero_tt
			has_variable = iw_funding_level
			var:iw_funding_level >= 1
		}
		custom_tooltip = {
			text = covert_op_capacity_election_tt
			covert_ops_type_below_cap = { TYPE = covert_election_interference_action }
		}
		custom_tooltip = {
			text = iw_has_available_slot_tt
			covert_operations_available_slots >= 1
		}
		custom_tooltip = {
			text = iw_target_has_elections_tt
			scope:target_country = {
				OR = {
					has_law = law_type:law_landed_voting
					has_law = law_type:law_wealth_voting
					has_law = law_type:law_census_voting
					has_law = law_type:law_universal_suffrage
				}
			}
		}
		custom_tooltip = {
			text = iw_no_duplicate_target_tt
			NOT = { has_diplomatic_pact = { who = scope:target_country type = covert_election_interference_action } }
		}
		custom_tooltip = {
			text = iw_target_not_decentralized_tt
			scope:target_country = { covert_action_valid_target = yes }
		}
		custom_tooltip = {
			text = iw_no_truce_tt
			NOT = { has_truce_with = scope:target_country }
		}

		custom_tooltip = {
			text = iw_no_intel_sharing_tt
			NOT = {
				any_scope_treaty = {
					binds = scope:target_country
					any_scope_article = { has_type = intelligence_sharing_pact }
				}
			}
		}
		custom_tooltip = {
			text = iw_no_good_relations_tt
			NOT = { scope:target_country.relations:root >= relations_threshold:amicable }
		}
	}

	pact = {
		cost = 0
		is_two_sided_pact = no
		show_in_outliner = yes

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = iw_funding_not_zero_tt
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
				# Target must still have elections
				custom_tooltip = {
					text = iw_target_has_elections_tt
					scope:target_country = {
						OR = {
							has_law = law_type:law_landed_voting
							has_law = law_type:law_wealth_voting
							has_law = law_type:law_census_voting
							has_law = law_type:law_universal_suffrage
						}
					}
				}
			}
		}


		# Allow operation to continue during war; do not auto-break on war start

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = iw_no_truce_tt
					NOT = { has_truce_with = scope:target_country }
				}
			}
		}
		manual_break_effect = { covert_op_end = { TYPE = election_interference } }
		auto_break_effect = { covert_op_end = { TYPE = election_interference } }
	}

	ai = {
		evaluation_chance = {
			value = 0.03
			if = {
				limit = { ruler_is_aggressive = yes }
				add = 0.02
			}
			if = {
				limit = { ruler_is_cautious = yes }
				multiply = 0.5
			}
		}

		will_propose = {
			covert_operations_available_slots >= 1
			has_variable = iw_funding_level
			var:iw_funding_level >= 1
			in_default = no
			country_rank >= rank_value:major_power
			OR = {
				has_diplomatic_pact = { who = scope:target_country type = rivalry }
				has_attitude = { who = scope:target_country attitude = antagonistic }
				has_attitude = { who = scope:target_country attitude = domineering }
			}
			intelligence_capacity_total >= scope:target_country.intelligence_capacity_total
			# Much weaker targets only with money to burn
			covert_ai_target_worth_the_upkeep = { TARGET = scope:target_country }
		}

		# Kept until fully operational, then until the hostility is gone
		will_break = {
			covert_ai_op_settled = { TYPE = election_interference TARGET = scope:target_country }
			covert_ai_hostility_gone = { TARGET = scope:target_country }
		}

		propose_score = {
			value = 5
			if = {
				limit = { has_diplomatic_pact = { who = scope:target_country type = rivalry } }
				add = 10
			}
			if = {
				limit = { country_rank >= rank_value:great_power }
				add = 5
			}
			if = {
				limit = { ruler_is_aggressive = yes }
				add = 5
			}
		}
	}
}

			covert_ai_target_worth_the_upkeep = { TARGET = scope:target_country }
		}

		# Called off once the network is strong, whatever its age (the operation
		# pays from its first month), or once fully operational if the hostility
		# is gone
		will_break = {
			OR = {
				NOT = { covert_net_weaker_than = { TARGET = scope:target_country STRENGTH = covert_cultivate_ai_net_done } }
				AND = {
					covert_ai_op_settled = { TYPE = cultivate_assets TARGET = scope:target_country }
					covert_ai_hostility_gone = { TARGET = scope:target_country }
				}
			}
		}

		propose_score = {
			value = 5
			# Feed the network another operation there already benefits from.
			if = {
				limit = {
					has_variable_list = iw_ops
					any_in_list = {
						variable = iw_ops
						var:iw_target ?= scope:target_country
					}
				}
				add = 5
			}
		}
	}
}

# SECURE LOOSE MATERIAL (nuclear_crisis_design.md §0.10 step 5)
# ----------------------------------------------------------------------------
# Hunts, inside the target, for the nuclear warheads missing from its arsenal
# line (nd_ar_loose): lost when its arsenal, or one that passed to it, changed
# hands. It does nothing to the target. Once established, each month it may
# find one (covert_secure_material_chance); the warhead is destroyed and
# leaves the loose pool, the operator is told, and the target never is. Mild
# tier: every service looks for loose material. Ends by itself when the
# target's line has nothing left to find.
covert_secure_material_action = {
	texture = "gfx/interface/icons/diplomatic_action_icons/covert_secure_material_action.dds"
	groups = {
		general
	}
	requires_approval = no
	show_confirmation_box = yes
	show_effect_in_tooltip = yes
	is_hostile = yes
	should_notify_third_parties = no

	accept_effect = {
		custom_tooltip = {
			text = covert_secure_material_extra_tt
		}
		custom_tooltip = {
			text = covert_op_detection_warning_tt
		}
		covert_op_start = { TYPE = secure_material DEFENSE_MOD = country_covert_defense_military_add CODE = 13 }
	}

	potential = {
		has_game_rule = covert_warfare_enabled
		has_game_rule = nuclear_weapons_enabled
	}

	possible = {
		custom_tooltip = {
			text = iw_funding_not_zero_tt
			has_variable = iw_funding_level
			var:iw_funding_level >= 1
		}
		custom_tooltip = {
			text = covert_op_capacity_secure_material_tt
			covert_ops_type_below_cap = { TYPE = covert_secure_material_action }
		}
		custom_tooltip = {
			text = iw_has_available_slot_tt
			covert_operations_available_slots >= 1
		}
		custom_tooltip = {
			text = iw_no_duplicate_target_tt
			NOT = { has_diplomatic_pact = { who = scope:target_country type = covert_secure_material_action } }
		}
		custom_tooltip = {
			text = iw_target_not_decentralized_tt
			scope:target_country = { covert_action_valid_target = yes }
		}
		custom_tooltip = {
			text = iw_no_truce_tt
			NOT = { has_truce_with = scope:target_country }
		}
		# Allowed during war: no has_war guard
		custom_tooltip = {
			text = iw_no_intel_sharing_tt
			NOT = {
				any_scope_treaty = {
					binds = scope:target_country
					any_scope_article = { has_type = intelligence_sharing_pact }
				}
			}
		}
		custom_tooltip = {
			text = iw_target_has_loose_material_tt
			scope:target_country = { nd_loose_has_line = yes }
		}
	}

	pact = {
		cost = 0
		is_two_sided_pact = no
		show_in_outliner = yes

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = iw_funding_not_zero_tt
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = iw_target_not_decentralized_tt
					scope:target_country = { covert_action_valid_target = yes }
				}
			}
		}

		# Nothing left to find: the operation ends by itself.
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = iw_target_has_loose_material_tt
					scope:target_country = { nd_loose_has_line = yes }
				}
			}
		}

		# Allowed during war: do not auto-break when war starts

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = iw_no_truce_tt
					NOT = { has_truce_with = scope:target_country }
				}
			}
		}
		manual_break_effect = { covert_op_end = { TYPE = secure_material } }
		auto_break_effect = { covert_op_end = { TYPE = secure_material } }
	}

	ai = {
		evaluation_chance = {
			value = 0.05
		}

		will_propose = {
			scope:target_country = { nd_loose_has_line = yes }
			country_rank >= rank_value:major_power
			covert_operations_available_slots >= 1
			has_variable = iw_funding_level
			var:iw_funding_level >= 1
			in_default = no
		}

		# Never called off by choice: it ends when nothing is left to find
		# (requirement_to_maintain)
		will_break = { always = no }

		propose_score = {
			value = 5
			# A warhead in a neighbour's hands is a warhead near us.
			if = {
				limit = { is_adjacent_to_country = scope:target_country }
				add = 5
			}
		}
	}
}
