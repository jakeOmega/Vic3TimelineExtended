nuke_diplo_action = {
	groups= {
		general
	}
	requires_approval = no

	state_selection = second_required
	second_state_list = second_country

	show_confirmation_box = yes

	potential = {
		has_modifier = nuclear_power
		NOT = { modifier:country_nuclear_disarmament_bool = yes }
	}

	possible = {
		var:nuclear_weapon_stockpile > 0
		# Doctrine (je_nuclear_deterrence): when do we say we will use them?
		# The law gates below stay and apply on top — a strike needs both.
		# See docs/systems/nuclear_crisis_design.md §0.2.
		custom_tooltip = {
			text = nd_tt_doctrine_permits_strike
			nd_doctrine_permits_strike = { ENEMY = scope:target_country }
		}
		custom_tooltip = {
			text = nd_tt_pledge_permits_strike
			nd_pledge_permits_strike = { ENEMY = scope:target_country }
		}
		OR = {
			# Normal: can use if no humanitarian/limited war laws
			NOT = {
				OR = {
					has_law = law_type:law_humanitarian_regulations
					has_law = law_type:law_limited_war
				}
			}
			# Retaliation: already been nuked overrides war laws
			any_scope_state = {
				OR = {
					has_modifier = nuclear_strike_aftermath
					has_modifier = tactical_nuke_aftermath
				}
			}
			# Existential threat overrides war laws
			any_scope_war = {
				is_target_of_wargoal_in_war = ROOT
				OR = {
					has_war_goal = make_protectorate
					has_war_goal = make_dominion
					has_war_goal = make_tributary
					has_war_goal = make_personal_union
					has_war_goal = make_crown_land
					has_war_goal = make_chartered_company
					has_war_goal = annex_country
					has_war_goal = te_reunify_country
				}
			}
		}
	}

	second_state_trigger = {
		exists = scope:country
		owner = {
			has_war_with = scope:country
		}
	}

	accept_effect = {
		save_scope_as = attacking_country
		if = {
			limit = { exists = scope:second_state }
			scope:second_state = {
				save_scope_as = target_state
				nuclear_first_strike = yes
			}
			# Pledges, doctrine breaches, crises, guarantors (§11): once per launch.
			nd_record_nuclear_use = { VICTIM = scope:target_country }
		}
	}

	is_hostile = yes

	ai = {
		# Whether the AI uses the bomb at all is nd_ai_nuclear_use_justified in
		# will_propose (doctrine, survival, retaliation, perceived retaliation,
		# a reserve, a six-month pause); this only sets how often it asks.
		# docs/systems/nuclear_crisis_design.md §0.2 and §9.
		evaluation_chance = {
			value = 0.1 # base 10% chance to evaluate
			if = {
				limit = {
					exists = scope:target_country
					nd_enemy_threatens_existence = { ENEMY = scope:target_country }
				}
				multiply = 7 # existential threats make the AI seriously consider nuclear use
			}
			if = {
				limit = {
					exists = scope:target_country
					nd_was_struck_by = { ENEMY = scope:target_country }
				}
				multiply = 5 # retaliation is strongly favoured
			}
			if = {
				limit = { nd_doctrine_warfighting = yes }
				multiply = 3
			}
			else_if = {
				limit = { nd_doctrine_compellence = yes }
				multiply = 1.5
			}
			if = {
				limit = {
					ruler_is_aggressive = yes
				}
				multiply = 3
			}
			if = {
				limit = {
					exists = ruler
					ruler = { has_trait = ruthless }
					NOT = { ruler = { has_trait = cautious } }
				}
				multiply = 2
			}
			if = {
				limit = {
					ruler_is_cautious = yes
				}
				multiply = 0.25 # very reluctant but not impossible in existential situations
			}
			max = 1
		}

		will_select_as_second_state = {
			always = yes
		}

		# The old gate read scope:country. Vanilla never uses that name for the
		# target in an AI block (scope:target_country, 600+ uses); inside the
		# state triggers it is the actor. So the gate most likely never passed
		# and the AI never struck first (unverified in-game). The
		# doctrine-aware judgement replaces it and names the target properly.
		will_propose = {
			has_war_with = scope:target_country
			nd_ai_nuclear_use_justified = { ENEMY = scope:target_country }
		}

		propose_score = {
			value = 0
			if = {
				limit = { exists = scope:second_state }
				add = {
					scope:second_state = {
						value = nuclear_industrial_strike_target_score
					}
				}
			}
			# A war against a declared rival is one the AI is readier to escalate
			if = {
				limit = {
					any_scope_war = {
						is_war_participant = scope:target_country
						is_at_war_with_rival = ROOT
					}
				}
				multiply = 1.5
			}
		}
	}
}


tactical_nuke_diplo_action = {
	groups= {
		general
	}
	requires_approval = no

	state_selection = second_required
	second_state_list = second_country

	show_confirmation_box = yes

	potential = {
		has_modifier = nuclear_power
		NOT = { modifier:country_nuclear_disarmament_bool = yes }
	}

	possible = {
		has_technology_researched = tactical_nuclear_weapons
		has_variable = nuclear_weapon_stockpile
		var:nuclear_weapon_stockpile > 0
		# Doctrine and pledges, exactly as for the strategic strike: a limited
		# use is still a use (§3.1).
		custom_tooltip = {
			text = nd_tt_doctrine_permits_strike
			nd_doctrine_permits_strike = { ENEMY = scope:target_country }
		}
		custom_tooltip = {
			text = nd_tt_pledge_permits_strike
			nd_pledge_permits_strike = { ENEMY = scope:target_country }
		}
		OR = {
			# Normal: can use if no limited war law
			NOT = {
				has_law = law_type:law_limited_war
			}
			# Retaliation: already been nuked overrides war laws
			any_scope_state = {
				OR = {
					has_modifier = nuclear_strike_aftermath
					has_modifier = tactical_nuke_aftermath
				}
			}
			# Existential threat overrides limited war law
			any_scope_war = {
				is_target_of_wargoal_in_war = ROOT
				OR = {
					has_war_goal = make_protectorate
					has_war_goal = make_dominion
					has_war_goal = make_tributary
					has_war_goal = make_personal_union
					has_war_goal = make_crown_land
					has_war_goal = make_chartered_company
					has_war_goal = annex_country
					has_war_goal = te_reunify_country
				}
			}
		}
	}

	second_state_trigger = {
		exists = scope:country
		owner = {
			has_war_with = scope:country
		}
		OR = {
			has_building = building_barrack
			has_building = building_naval_administration
			has_building = building_naval_fortification
			has_building = building_naval_logistics_center
			has_building = building_conscription_center
		}
	}

	accept_effect = {
		save_scope_as = attacking_country
		if = {
			limit = { exists = scope:second_state }
			scope:second_state = {
				save_scope_as = target_state
			}
			# The strike itself, shared with the incident chains
			# (nuclear_deterrence_effects.txt). It used to name the victim as
			# scope:country, which vanilla uses for the actor, not the target.
			nd_tactical_strike_resolve = yes
			nd_record_nuclear_use = { VICTIM = scope:target_country }
		}
	}

	is_hostile = yes

	ai = {
		# As for the strategic strike: nd_ai_nuclear_use_justified decides,
		# this sets how often the AI asks. Tactical weapons are considered more
		# readily.
		evaluation_chance = {
			value = 0.25 # tactical nukes considered more readily than strategic
			if = {
				limit = {
					exists = scope:target_country
					nd_enemy_threatens_existence = { ENEMY = scope:target_country }
				}
				multiply = 3
			}
			if = {
				limit = {
					exists = scope:target_country
					nd_was_struck_by = { ENEMY = scope:target_country }
				}
				multiply = 3
			}
			if = {
				limit = { nd_doctrine_warfighting = yes }
				multiply = 2
			}
			if = {
				limit = {
					ruler_is_aggressive = yes
				}
				multiply = 2
			}
			if = {
				limit = {
					ruler_is_cautious = yes
				}
				multiply = 0.25 # reluctant but not impossible
			}
			max = 1
		}

		will_select_as_second_state = {
			OR = {
				has_building = building_barrack
				has_building = building_naval_administration
				has_building = building_naval_fortification
				has_building = building_naval_logistics_center
				has_building = building_conscription_center
			}
		}

		will_propose = {
			has_war_with = scope:target_country
			nd_ai_nuclear_use_justified = { ENEMY = scope:target_country }
		}

		propose_score = {
			value = 0
			if = {
				limit = { exists = scope:second_state }
				scope:second_state = {
					if = {
						limit = {
							has_building = building_barrack
						}
						add = {
							value = b:building_barrack.level
							divide = 100
						}
					}
					if = {
						limit = {
							has_building = building_naval_administration
						}
						add = {
							value = b:building_naval_administration.level
							divide = 100
						}
					}
					if = {
						limit = {
							has_building = building_naval_fortification
						}
						add = {
							value = b:building_naval_fortification.level
							divide = 100
						}
					}
					if = {
						limit = {
							has_building = building_naval_logistics_center
						}
						add = {
							value = b:building_naval_logistics_center.level
							divide = 100
						}
					}
					if = {
						limit = {
							has_building = building_conscription_center
						}
						add = {
							value = b:building_conscription_center.level
							divide = 100
						}
					}
					if = {
						limit = { owner = { has_modifier = nuclear_power }}
						divide = 1000 # near-zero priority against nuclear-armed targets
					}
				}
			}
			# A war against a declared rival is one the AI is readier to escalate
			if = {
				limit = {
					any_scope_war = {
						is_war_participant = scope:target_country
						is_at_war_with_rival = ROOT
					}
				}
				multiply = 1.5
			}
		}
	}
}
