# ======================================================================
#  Subjugate (force_become_subject) — per-identity rebalance
# ======================================================================
#
# Vanilla gates this action behind identity_sovereign_empire (via
# power_bloc_leader_can_make_subjects_bool) and a flat 5x prestige check.
# This rebalance opens it to all identities but raises the baseline cost
# to 20x prestige and feeds identity-specific bonuses (read from
# country_subjugation_*_strength_add modifiers seeded on each identity's
# leader_modifier) through the te_subjugation_strength script_value.
#
# Now that any bloc can subjugate, the action also charges infamy on
# success, a fifth of the target's annexation figure (see the change_infamy
# block below). Sovereign Empire keeps a flavor edge via the dormant vanilla bool
# power_bloc_leader_can_make_subjects_bool — repurposed as a 50% infamy
# discount flag (the bool is still granted by the vanilla SE identity's
# power_bloc_modifier; relocalized in te_modifiers_l_english.yml).
# If a vanilla patch renames or removes that bool, the SE discount
# silently breaks — re-check on every vanilla bump.
#
# accept_effect, potential, and ai blocks were copied byte-identical from
# vanilla 31_power_bloc_force_become_subject.txt — re-diff on every Vic3
# patch. Vanilla version copied: 1.10 (Steam, 2026-05). Last re-diffed:
# 1.15.0 (2026-11-05): ported 1.15's can-be-subjugated tooltip in `possible`
# and the AI's regional-objective lines (ai_is_regional_objective_subjugate_target,
# from 1.14.x, missed then); otherwise byte-identical apart from the documented
# intentional changes. The change_infamy block is mod-only; everything
# else above it remains vanilla-shaped.
#
# See common/script_values/extra_script_values.txt :: te_subjugation_strength
# for the full scoring formula.

REPLACE:force_become_subject = {
	groups = {
		power_bloc_leader
	}
	show_in_lens = no

	texture = "gfx/interface/icons/diplomatic_action_icons/force_become_subject.dds"

	selectable = {
		is_power_bloc_leader = yes
		# Vanilla power_bloc_leader_can_make_subjects_bool gate intentionally dropped —
		# all bloc identities can now select this action; identity-conditional
		# strength gating happens in `possible` via te_subjugation_strength.
	}

	potential = {
		power_bloc ?= {
			scope:target_country.power_bloc ?= this
		}
	}

	possible = {
		scope:target_country = {
			is_subject = no
			country_rank < rank_value:unrecognized_major_power
			is_diplomatic_play_committed_participant = no
		}

		# vanilla 1.15
		custom_tooltip = {
			text = DIPLOMATIC_PACT_REQUIREMENT_CAN_BE_SUBJUGATED
			scope:target_country = { modifier:country_cannot_be_subjugated_bool ?= no }
		}

		power_bloc = {
			current_cohesion_percentage >= 0.5
		}
		scope:target_country = {
			tenure_in_current_power_bloc_years >= 5
		}

		# Replaces vanilla's flat `prestige >= target.prestige * 5` with a
		# scored gate. See te_subjugation_strength for the full formula. The
		# tooltip shows the current score and its parts (prestige, identity
		# bonus, flat bonus) through the te_subjugation_* display script values.
		custom_tooltip = {
			text = te_subjugation_strength_sufficient_tt
			te_subjugation_strength >= 1.0
		}
	}

	accept_effect = {
		if = {
			limit = { is_country_type = unrecognized }
			create_diplomatic_pact = {
				country = scope:target_country
				type = tributary
			}
		}
		else_if = {
			limit = {
				scope:target_country = {
					is_country_type = colonial
				}
			}
			create_diplomatic_pact = {
				country = scope:target_country
				type = dominion
			}
		}
		else = {
			create_diplomatic_pact = {
				country = scope:target_country
				type = protectorate
			}
		}
		create_unidirectional_truce = { country = scope:target_country months = 60 }
		hidden_effect = { scope:target_country = { add_liberty_desire = 30 } } # Starts at a higher LD level

		power_bloc = {
			add_cohesion_percent = {
				value = scaled_cohesion_cost_subjugation
			}
		}

		# Infamy: a fifth of what annexing the target would cost, the target's
		# country_infamy_value (vanilla 1.14+ common/script_values/
		# 00_infamy_values.txt, the figure 00_annex_country.txt charges; root is
		# the aggressor here as there). It is a fifth because subjugation leaves
		# the target nominally sovereign (owner, 2026-11-12, #387). Capped at
		# INFAMY_THRESHOLD_INFAMOUS (25), so it can never alone tip the leader
		# into "Infamous"; from a target worth 125 (a mid-sized country in the
		# 1.14.5 migration's worked cases) the cap decides. Sovereign Empire
		# halves it via the repurposed power_bloc_leader_can_make_subjects_bool
		# flag still granted by the vanilla SE identity. Vanilla's subject
		# multipliers are left out: the target can't be a subject (possible).
		# Read before the pact would be the same: the value has no subject term.
		# No vanilla diplomatic action reads country_infamy_value (stake colonial
		# claim reads state_infamy_value), so the once-a-game
		# force_subject_infamy self-check says whether it resolves here
		# (play-test list docs/testing/vanilla-1-15-playtest.md).
		change_infamy = {
			value = {
				value = 0

				scope:target_country = {
					add = country_infamy_value
				}

				# Subjugate is 20% as bad as annex.
				multiply = 0.2

				# Sovereign Empire discount via repurposed dormant bool.
				if = {
					limit = { modifier:power_bloc_leader_can_make_subjects_bool = yes }
					multiply = 0.5
				}

				min = 0
				max = 25
			}
		}
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = te_fbs_infamy_checked } }
				set_global_variable = { name = te_fbs_infamy_checked value = yes }
				if = {
					limit = {
						scope:target_country = { country_infamy_value > 0 }
					}
					debug_log = "TE_SELF_CHECK OK force_subject_infamy country=[THIS.GetCountry.GetTagName] (Subjugate read the target's country_infamy_value above 0 in a diplomatic action) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN force_subject_infamy country=[THIS.GetCountry.GetTagName] (Subjugate read the target's country_infamy_value as 0 in a diplomatic action, so it charged no infamy; take the fallback in docs/testing/vanilla-1-15-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
		}
	}

	ai = {
		evaluation_chance = {
			value = 0.025 # AI shouldn't use this very often
		}

		will_propose = {
			power_bloc ?= { current_cohesion_percentage >= 0.6 }
			OR = {
				has_attitude = {
					who = scope:target_country
					attitude = domineering
				}
				power_bloc ?= { current_cohesion_percentage >= 0.8 }
				ai_is_regional_objective_subjugate_target = { TARGET = scope:target_country }
			}
			OR = {
				AND = {
					has_attitude = {
						who = scope:target_country
						attitude = domineering
					}
					power_bloc ?= { current_cohesion_percentage >= 0.8 }
				}
				scope:target_country = {
					is_country_type = unrecognized
				}
				ai_is_regional_objective_subjugate_target = { TARGET = scope:target_country }
			}
		}

		propose_score = {
			value = 10

			if = {
				limit = {
					ai_is_regional_objective_subjugate_target = { TARGET = scope:target_country }
				}
				add = 20
			}
		}
	}
}
