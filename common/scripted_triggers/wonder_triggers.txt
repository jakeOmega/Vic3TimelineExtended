# State scope: true if this state is in at least one recognized continental region.
# States not covered (e.g. Oceania) return false.
# custom_tooltip keeps the tooltip to one line. Until 2026-10-03 each test was
# `any_state_in_<region> = { this = prev }`, a walk over the region's states
# per test and per call (the Continental Union HQ's potential, 231 us a call in
# a late-game script profile); is_in_geographic_region is the engine's lookup
# for the same thing (short_key <region> = geographic_region_<region>).
state_is_in_recognized_continent = {
	custom_tooltip = {
		text = STATE_IS_IN_RECOGNIZED_CONTINENT_TT
		OR = {
			is_in_geographic_region = geographic_region_africa
			is_in_geographic_region = geographic_region_europe
			is_in_geographic_region = geographic_region_north_america
			is_in_geographic_region = geographic_region_south_america
			is_in_geographic_region = geographic_region_east_asia
			is_in_geographic_region = geographic_region_south_east_asia
			is_in_geographic_region = geographic_region_india
			is_in_geographic_region = geographic_region_middle_east
			is_in_geographic_region = geographic_region_central_asia
		}
	}
}

# State scope: true if no state in the same continental region already
# has a building of type $BUILDING_TYPE$.
# Gate on state_is_in_recognized_continent first — unrecognized states
# pass vacuously (all NOT blocks succeed because no continent matches).
# custom_tooltip wraps the body for the same reason as above.
continent_has_no_building_of_type = {
	custom_tooltip = {
		text = CONTINENT_HAS_NO_BUILDING_OF_TYPE_TT
		NOT = {
			AND = {
				is_in_geographic_region = geographic_region_africa
				any_state_in_africa = { any_scope_building = { is_building_type = $BUILDING_TYPE$ } }
			}
		}
		NOT = {
			AND = {
				is_in_geographic_region = geographic_region_europe
				any_state_in_europe = { any_scope_building = { is_building_type = $BUILDING_TYPE$ } }
			}
		}
		NOT = {
			AND = {
				is_in_geographic_region = geographic_region_north_america
				any_state_in_north_america = { any_scope_building = { is_building_type = $BUILDING_TYPE$ } }
			}
		}
		NOT = {
			AND = {
				is_in_geographic_region = geographic_region_south_america
				any_state_in_south_america = { any_scope_building = { is_building_type = $BUILDING_TYPE$ } }
			}
		}
		NOT = {
			AND = {
				is_in_geographic_region = geographic_region_east_asia
				any_state_in_east_asia = { any_scope_building = { is_building_type = $BUILDING_TYPE$ } }

			}

		NOT = {
			AND = {
				is_in_geographic_region = geographic_region_central_asia
				any_state_in_central_asia = { any_scope_building = { is_building_type = $BUILDING_TYPE$ } }
			}
		}
	}
}

# State scope: standard "unique per owner" potential guard.
# True if this state already has the building OR the owner doesn't have one anywhere.
# Two kinds of caller: the UN and power-bloc HQs (unique = no), where this
# guard is what limits the owner to one; and the build-anywhere wonders (LHC,
# ISS, ITER, Svalbard, LIGO, Peace Palace, Palais des Nations), which also set
# unique = yes, so the engine already allows only one in the WORLD and this
# guard adds nothing beyond it.
building_unique_per_owner_potential = {
	OR = {
		any_scope_building = {
			is_building_type = $BUILDING_TYPE$
		}
		NOT = {
			owner = {
				any_scope_building = {
					is_building_type = $BUILDING_TYPE$
				}
			}
		}
	}
}
