# AI coverage of the mod's own content (#1822). docs/systems/mod_systems/ai.md § AI Coverage of the Mod's Content.

# State scope. The state has a building that makes Electricity, built or under construction: vanilla's coal and
# oil plant, the mod's four plants, or a Solar Receiver. Vanilla's power plant asks this of itself alone (the NOR
# in its ai_value); the mod widens it so that one plant of any kind ends the first-plant bonus for all of them.
# test_ai_building_values.py fails if a building of the power group is missing here.
te_state_has_a_power_plant = {
	OR = {
		has_building = building_power_plant
		has_building = building_hydro_plant
		has_building = building_nuclear_plant
		has_building = building_renewable_energy_plant
		has_building = building_fusion_plant
		has_building = building_solar_receiver
		any_scope_building = {
			is_under_construction = yes
			OR = {
				is_building_type = building_power_plant
				is_building_type = building_hydro_plant
				is_building_type = building_nuclear_plant
				is_building_type = building_renewable_energy_plant
				is_building_type = building_fusion_plant
				is_building_type = building_solar_receiver
			}
		}
	}
}

# Country scope. The country's bureaucracy can carry one more institution: it has the spare bureaucracy that level 1
# of a new institution takes (te_ai_new_institution_bureaucracy), and the institutions it already has are not
# heading for a shortage as they grow (approaching_bureaucracy_shortage, the engine's own forecast). Vanilla asks
# the pair `bureaucracy >= 0` and `approaching_bureaucracy_shortage = no` for "the bureaucracy is in order"
# (je_tsarpoints_bureaucratic_reform_completed); this adds the new institution's own cost to the first.
te_ai_can_staff_new_institution = {
	approaching_bureaucracy_shortage = no
	bureaucracy >= te_ai_new_institution_bureaucracy
}

# Country scope. Enacting a law that carries $INSTITUTION$ would found an institution the bureaucracy can't carry.
# False while the country already has the institution: a law that only changes what an existing institution does
# (Old Age Pension to Universal Basic Income, one Ministry of Labor to the other) adds no institution, and its
# bureaucracy is a running cost the country pays already. Every law of the mod's that carries an institution
# reads this in its ai_enact_weight_modifier (test_ai_institution_laws.py).
te_ai_law_brings_unstaffed_institution = {
	NOT = { has_institution = $INSTITUTION$ }
	te_ai_can_staff_new_institution = no
}
