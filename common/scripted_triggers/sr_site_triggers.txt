# Design: docs/superpowers/specs/2026-10-25-solar-colonization-holders-and-cession-design.md.

# The Space Race rule has four settings (design § 10): off, races only (the
# setting keeps its old name, space_race_enabled, so saves from before keep
# today's colonies), simple colonies and full colonies. Each setting has one
# flag, named after it. Read the rule through these gates, never through
# has_game_rule = space_race_enabled, which is races only alone:
# test_solar_sites.py fails on a bare one anywhere else. Any scope.
sr_space_race_on = {
	OR = {
		has_game_rule = space_race_enabled
		has_game_rule = space_race_colonies_simple
		has_game_rule = space_race_colonies_full
	}
}

# Colonies change hands by treaty and war (simple and full).
sr_colonies_contested_enabled = {
	OR = {
		has_game_rule = space_race_colonies_simple
		has_game_rule = space_race_colonies_full
	}
}

# Fleets (full only; later slices).
sr_colonies_full_enabled = {
	has_game_rule = space_race_colonies_full
}

# Country scope. A country a dead holder's colonies can pass to (design § 6).
sr_site_can_hold = {
	is_country_alive = yes
	NOT = { is_country_type = decentralized }
}
