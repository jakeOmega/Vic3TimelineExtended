# Global Warming — shared eligibility triggers.
#
# Each `gw_possible_<button>` holds the body that used to live inline in the
# matching scripted_button's `possible` block in
# common/scripted_buttons/global_warming_buttons.txt. Both the journal-entry
# button and the widget's scripted GUI
# (common/scripted_guis/global_warming_sguis.txt) call these, so a click through
# either surface is gated by exactly the same rules.
#
# The nine policy modifiers live on the JOURNAL ENTRY, not the country, so
# "is it active" is always `je:je_global_warming ?= { has_modifier = X }` --
# a country-scope has_modifier cannot see them.
#
# EVERY CLAUSE IS PHRASED AS A CONDITION, NOT A COMPLAINT. The widget renders
# these through `ScriptedGui.IsValidTooltip`, which prints each line with a tick
# when satisfied and a cross when not, so "Leads its own market" reads correctly
# in both states where "Cannot do this" would not. `produced_authority` is
# deliberately left unwrapped: the engine's own rendering carries the live
# number, which a custom_tooltip would hide.
#
# Every country controls its own nine national policies. Market leadership
# gates only fossil-fuel tariffs; members can see their leader's tariff status.

# ---------------------------------------------------------------------------
# "Is this policy in force for this country?" — the single definition, used by
# the widget's row status, its controls' `is_shown`, and the display counters.
# `?=` because a country whose journal entry never activated has no instance.
# ---------------------------------------------------------------------------
gw_policy_carbon_tax_active = {
	je:je_global_warming ?= { has_modifier = carbon_tax_modifier }
}

gw_policy_renewable_investment_active = {

# Does this country lead its own market? Used for the role icon and tariffs.
# A country that has lost every state stays alive while its war goes on, with
# no market capital; unguarded, the panel's market-leader checks logged an
# error every time they rendered for one.
gw_is_market_leader = {
	exists = market_capital.owner
	market_capital.owner = THIS
}

# Treaty obligations belong to the source country and prevent its national
# mitigation repeals. Adaptation and tariffs remain voluntary.
gw_bound_by_emissions_treaty = {
	has_variable = has_emissions_reduction_treaty
}

# Countries with a positive annual snapshot rank independently of leadership.
gw_is_ranked_emitter = {
	has_variable = gw_disp_country_emis
	var:gw_disp_country_emis > 0
}

# Have the yearly emissions figures been taken yet? False on a brand-new game
# before the first yearly state pulse and on a save upgraded mid-year, which is
# when the widget shows "figures update each January" instead of a row of
# misleading zeroes. Both halves are checked because the market snapshot and the
# world re-base land on different pulses.
gw_has_yearly_figures = {
	has_global_variable = gw_g_global_emis
	market_capital ?= {
		owner ?= { has_variable = gw_disp_market_emis }
	}
}

# What je_global_warming needs to activate, its `possible`, as one tooltip
# line: the checklist on the Market panel's greyed Climate tab
# (te_market_global_warming_tab_unlock_sgui's is_valid,
# te_system_tab_sguis.txt). Wrapped because the engine would print the bare
# comparison under the script value's key. The test inside is the entry's
# `possible` word for word, and MarketTabTest in test_global_warming_layout.py
# fails if the two drift apart. Scope: country.
gw_entry_unlocked = {
	custom_tooltip = {
		text = gw_entry_unlock_tt
		OR = {
			temperature_anomaly_display >= 0.1
			rt_restrictive_transition_law = yes
		}
	}
}

# ---------------------------------------------------------------------------
# Event gates shared by an event and the monthly roll that fires it, so the
# two cannot drift apart. A roll whose own gate is narrower than the event's
# silently wastes the event for every country it leaves out.
# ---------------------------------------------------------------------------

# The Barren Harvest (environmentalism_events.5): a state growing staple food.
# The five grain farms and livestock ranches; subsistence farms are left out on
# purpose, or every state would qualify.
# Scope: state
gw_state_has_staple_farm = {
	any_scope_building = {
		OR = {
			is_building_type = building_wheat_farm
			is_building_type = building_rye_farm
			is_building_type = building_rice_farm
			is_building_type = building_maize_farm
			is_building_type = building_millet_farm
			is_building_type = building_livestock_ranch
		}
	}
}

}
