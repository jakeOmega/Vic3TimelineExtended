# Country scope. Public Schools could be started: Empiricism, not held, and neither law vanilla's disallowing_laws
# names held (that list names laws one by one, so a State Religion variant does not block it).
te_alw_public_schools_open = {
	has_technology_researched = empiricism
	NOT = { has_law_or_variant = law_type:law_public_schools }
	NOT = { has_law = law_type:law_state_religion }
	NOT = { has_law = law_type:law_serfdom }
}

# Country scope. Compulsory Primary School could be started: Human Rights, not held, no No Schools or Terakoya.
te_alw_compulsory_primary_open = {
	has_technology_researched = human_rights
	NOT = { has_law_or_variant = law_type:law_compulsory_primary_school }
	NOT = { has_law = law_type:law_no_schools }
	NOT = { has_law = law_type:law_terakoya }
}

# Country scope. Child labour could be ended with the law the weight is on: Labor Movement, Child Labor Allowed held.
te_alw_end_child_labor_open = {
	has_technology_researched = labor_movement
	has_law_or_variant = law_type:law_child_labor_allowed
}

# Country scope. Nothing else lifts one of the three laws: vanilla's own +500 for a schools law under No Schools,
# and its +750 for the law a petition, an imposition, the Social Revolution's list or a law commitment names
# (ai_has_enact_weight_modifier_journal_entries, which reads scope:law and so can't be asked here).
te_alw_clean = {
	NOT = { has_law_or_variant = law_type:law_no_schools }
	NOT = { has_journal_entry = je_government_petition }
	NOT = { has_journal_entry = je_law_imposition }
	NOT = { has_journal_entry = je_british_dictate_law }
	NOT = { has_journal_entry = je_social_revolution }
	NOT = { has_law_commitment = law_type:law_public_schools }
	NOT = { has_law_commitment = law_type:law_compulsory_primary_school }
	NOT = { has_law_commitment = law_type:law_restricted_child_labor }
}

# Country scope. At least one of the three could be started.
te_alw_a_target_open = {
	OR = {
		te_alw_public_schools_open = yes
		te_alw_compulsory_primary_open = yes
		te_alw_end_child_labor_open = yes
	}
}

# Country scope. The cohort: research at its innovation cap (the weight applies), clean, one of the three open.
te_alw_cohort = {
	te_ai_research_at_innovation_cap = yes
	te_alw_clean = yes
	te_alw_a_target_open = yes
}

# Country scope. The control: the cap read works and research is under it (the weight does not apply), clean, one
# of the three open.
te_alw_control = {
	te_ai_build_pass_innovation_cap > 0
	te_ai_research_at_innovation_cap = no
	te_alw_clean = yes
	te_alw_a_target_open = yes
}

# Country scope, at on_law_enactment_started. The law just started is one of the three, under the condition its
# weight asks for (Restricted Child Labor only from Child Labor Allowed).
te_alw_started_public_schools = {
	is_enacting_law = law_type:law_public_schools
}
te_alw_started_compulsory_primary = {
	is_enacting_law = law_type:law_compulsory_primary_school
}
te_alw_started_end_child_labor = {
	is_enacting_law = law_type:law_restricted_child_labor
	has_law_or_variant = law_type:law_child_labor_allowed
}
te_alw_started_a_target = {
	OR = {
		te_alw_started_public_schools = yes
		te_alw_started_compulsory_primary = yes
		te_alw_started_end_child_labor = yes
	}
}
