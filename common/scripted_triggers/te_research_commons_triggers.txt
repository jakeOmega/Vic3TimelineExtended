# ============================================================================
# RESEARCH COMMONS (identity_research_commons): triggers
# ============================================================================
# docs/systems/mod_systems/power_blocs.md § Research Commons; design:
# docs/superpowers/specs/2026-11-16-power-bloc-identities-design.md § 6.
# Script reads the identity through has_identity, as the monetary union reads
# has_principle; the identity's marker (power_bloc_research_pool_bool) only
# labels its tooltip.
# ============================================================================

# Country scope: the country is in a power bloc whose identity is the
# Research Commons, as its leader or as a member.
te_rc_in_commons = {
	is_in_power_bloc = yes
	power_bloc ?= { has_identity = identity:identity_research_commons }
}

# Country scope: the country leads a Research Commons.
te_rc_leads_commons = {
	is_power_bloc_leader = yes
	power_bloc ?= { has_identity = identity:identity_research_commons }
}

# Country scope: a member that contributes (in a Commons, not its leader).
te_rc_is_contributor = {
	is_power_bloc_leader = no
	te_rc_in_commons = yes
}

# Country scope: a former non-leader member (te_rc_member, set by the leader's
# pulse) that left a Commons that still stands: it is in no Research Commons
# now, and the leader it served (te_rc_leader) still exists and still leads one
# (coordinator's ruling, 2026-11-16). A dissolved bloc, a leader that is gone,
# and a bloc that changed identity put no penalty on their former members. The
# leader read is guarded apart (scripting_best_practices.md, the has_variable
# bullet).
te_rc_has_left_commons = {
	has_variable = te_rc_member
	te_rc_in_commons = no
	has_variable = te_rc_leader
	trigger_if = {
		limit = { exists = var:te_rc_leader }
		var:te_rc_leader = { te_rc_leads_commons = yes }
	}
	trigger_else = {
		always = no
	}
}

# Country scope (the would-be giver): Technology Controls closes Science Aid
# and its upgrade to $TARGET$, a country outside the giver's bloc. Read in
# science_aid's and science_aid_2's possible (extra_treaty_articles.txt), where
# root is the source party. The principle's power_bloc_modifier carries the
# boolean at every tier.
te_rc_tech_controls_bars_aid = {
	power_bloc ?= { modifier:power_bloc_technology_controls_bool = yes }
	NOT = { is_in_same_power_bloc = $TARGET$ }
}

# Country scope: leads a bloc whose Technology Controls tier opens the
# Technology Embargo (III and up).
te_rc_can_embargo = {
	is_power_bloc_leader = yes
	power_bloc ?= { modifier:power_bloc_can_use_technology_embargo_bool = yes }
}
