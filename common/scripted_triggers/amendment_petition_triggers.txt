# ======================================================================
#  AMENDMENT PETITIONS — scripted triggers
#
#  Rare yearly events (events/amendment_petition_events.txt) in which an
#  interest group asks for one of the mod's amendments on a law that is
#  already in force. Pool: common/on_actions/amendment_petition_on_actions.txt.
#  Design notes: docs/systems/mod_systems.md § Amendment Petitions.
#
#  Country scope throughout.
# ======================================================================

# ----------------------------------------------------------------------
#  Frequency
# ----------------------------------------------------------------------

# One petition at a time: every petition sets te_ap_recent_petition for
# five years in its `immediate` (te_ap_start_petition), whatever the answer.
te_ap_window_open = {
	NOT = { has_variable = te_ap_recent_petition }
}

# ----------------------------------------------------------------------
#  A replacement law is being enacted in the group
#
#  A petition amends the law in force. While another law of the same group
#  is being enacted, the amendment would leave with the outgoing law, and
#  the enactment events (extra_law_events.28–34, .87–.90,
#  inheritance_events.10–.13) already offer amendments to the incoming one. Each list names every law in the group,
#  variants included (law_crownland_diets); a law added to a group later
#  belongs here too.
# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
#  The law in force qualifies
#
#  One per amendment: a law the amendment allows is in force, it doesn't
#  carry the amendment yet, and no law of its group is being enacted. The
#  host laws are the amendment's `allowed_laws`
#  (common/amendments/extra_amendments.txt); keep the two in step. Also the
#  events' cancellation_trigger, so a petition still open when the law
#  changes or the amendment arrives some other way closes.
# ----------------------------------------------------------------------

		has_law = law_type:law_state_led_language_reform

	active_law:lawgroup_language_policy ?= {
		NOT = { has_amendment = amendment_type:amendment_minority_script_preservation }
	}
	NOT = { te_ap_enacting_in_language_policy = yes }
}

# Estate Duties: a law with private estates (te_inh_has_private_estates_law,
# te_inheritance_triggers.txt). Its `possible` keeps it off a law with the
# Birthright Endowment, which levies the duty itself.
te_ap_law_ok_estate_duties = {
	te_inh_has_private_estates_law = yes
	active_law:lawgroup_inheritance ?= {
		NOR = {
			has_amendment = amendment_type:amendment_estate_duties
			has_amendment = amendment_type:amendment_birthright_endowment
		}
	}
	NOT = { te_ap_enacting_in_inheritance = yes }
}

# ----------------------------------------------------------------------
#  Situations
#
#  Shared reasons a petition is made now rather than at any time. Each
#  event adds its own (an election campaign, a war, low legitimacy).
# ----------------------------------------------------------------------

# Hard times: the banking cycle in panic or downturn (only while its journal
# entry runs; banking_cycle_is_recession reads its variables), a slide down
# the power rankings, or a default.
te_ap_economy_faltering = {
	OR = {
		AND = {
			has_journal_entry = je_banking_cycle
			has_variable = finance_cycle_value
			banking_cycle_is_recession = yes
		}
		is_losing_power_rank = yes
		in_default = yes
	}
}

# Disorder in the streets: turmoil across the country or a large radical
# population.
te_ap_unrest = {
	OR = {
		country_turmoil > 0.2
		radical_population_fraction > 0.15
	}
}

# A cultural minority is pressing its case: a separatist movement is
# growing, or a cultural minority movement draws real support.
te_ap_minority_restive = {
	OR = {
		has_any_secessionists_growing = yes
		any_political_movement = {
			is_political_movement_type = movement_cultural_minority
			political_movement_support >= 0.05
		}
	}
}

# A large minority whose loyalty is in question: at least a fifth of the
# population outside the primary cultures, and the minority restive.
te_ap_minority_large_and_restive = {
	primary_cultures_percent_country = {
		target_country = ROOT
		value < 0.8
	}
	te_ap_minority_restive = yes
}

# Nativist pressure: a cultural majority movement draws real support, or
# under seven in ten people belong to the primary cultures.
te_ap_nativist_pressure = {
	OR = {
		any_political_movement = {
			is_political_movement_type = movement_cultural_majority
			political_movement_support >= 0.05
		}
		primary_cultures_percent_country = {
			target_country = ROOT
			value < 0.7
		}
	}
}

# Organised labour on the march: a labor movement draws real support.
te_ap_labor_agitation = {

# Interest group scope: carries enough weight to force the issue.
te_ap_ig_strong = {
	OR = {

	}
}

# The No-First-Strike Pledge binds the deterrence doctrine to No First Use
# while it is in force (nd_weekly_update, nuclear_deterrence_effects.txt). A
# petition offers it only where that binding changes nothing: no arsenal
# posture yet (an arsenal built later starts under the pledge) or a doctrine
# already at No First Use.
te_ap_no_first_strike_doctrine_fits = {
	OR = {
		NOT = { has_variable = nd_doctrine }
		AND = {
			has_variable = nd_doctrine
			var:nd_doctrine = 1
		}
	}
}

# A rival on either side of a rivalry holds nuclear weapons.
te_ap_rival_is_armed = {
	OR = {
		any_rival_country = { nd_is_armed = yes }
		any_rivaling_country = { nd_is_armed = yes }
	}
}
