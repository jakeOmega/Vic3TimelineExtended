# ---- Skins ---------------------------------------------------------------------------
# No gm_skin reads as generic (a monument older than skins, before first sight).
gm_state_skin_is = {
	OR = {
		AND = {
			has_variable = gm_skin
			var:gm_skin = flag:$SKIN$
		}
		AND = {
			NOT = { has_variable = gm_skin }
			always = $GENERIC$
		}
	}
}

# The five civic-progress dedications are buildings with a form of their own.
gm_state_has_building_form = {
	OR = {
		gm_state_has_pm = { PM = pm_monument_artistic }
		gm_state_has_pm = { PM = pm_monument_naturalist }
		gm_state_has_pm = { PM = pm_monument_scientific }
		gm_state_has_pm = { PM = pm_monument_industrial }
		gm_state_has_pm = { PM = pm_monument_athletic }
	}
}

# A dedicated monument that takes a monument's form (not a building's, not a
# faith's own building).
gm_state_takes_monument_forms = {
	gm_state_is_dedicated = yes
	gm_state_has_building_form = no
	gm_state_has_faith_skin = no
}

# ---- The commission's name -----------------------------------------------------------
# This monument answers the owner's open commission: of its dedication, and in
# its state if it names one (a just-dedicated monument fits, so the status test
# of gm_state_com_counts is not repeated here).
gm_state_answers_commission = {
	owner = { has_variable = gm_com_open }
	gm_state_has_dedication_flag = { VAR = gm_com_dedication }
	OR = {
		owner = {
			NOT = { has_variable = gm_com_state }
		}
		AND = {
			owner.var:gm_com_state ?= { save_temporary_scope_as = gm_tmp_com_state }
			this = scope:gm_tmp_com_state
		}
	}
}

# The commission's form fits this monument's skin.
gm_state_com_form_fits = {
	owner = { has_variable = gm_com_form }
	OR = {
		AND = { owner = { var:gm_com_form = flag:column } gm_state_form_fits_column = yes }
		AND = { owner = { var:gm_com_form = flag:arch } gm_state_form_fits_arch = yes }
		AND = { owner = { var:gm_com_form = flag:obelisk } gm_state_form_fits_obelisk = yes }
		AND = { owner = { var:gm_com_form = flag:statue } gm_state_form_fits_statue = yes }
		AND = { owner = { var:gm_com_form = flag:gate } gm_state_form_fits_gate = yes }
		AND = { owner = { var:gm_com_form = flag:hall } gm_state_form_fits_hall = yes }
		AND = { owner = { var:gm_com_form = flag:tower } gm_state_form_fits_tower = yes }
		AND = { owner = { var:gm_com_form = flag:memorial } gm_state_form_fits_memorial = yes }
		AND = { owner = { var:gm_com_form = flag:mausoleum } gm_state_form_fits_mausoleum = yes }
	}
}
