# Communal Child-Rearing's measures by who governs (owner, the review of #989): on its own the law allows Child
# Allowance, Parental Leave, Family Housing (the 1920s house-communes, the kibbutz's dining halls) and Free Contraception
# (Kollontai, the kibbutzim). Under a council or an anarchist commune it also allows Sex Education (the anarchist free
# schools, Emma Goldman's birth-control lectures); under a party state or an autocrat that outlaws dissent, Honours for
# Large Families and Restricted Contraception (the state raises the children it orders born: Lebensborn, Decree 770).
# Birth Limit and the Crash Campaign stay Population Control Measures' alone. A change to the government's laws ends a
# measure they no longer allow (te_demog_ms_law_check runs on every law activated).
te_demog_ms_communal_utopia = {
	has_law = law_type:law_communal_child_rearing
	OR = {
		has_law = law_type:law_council_republic
		has_law = law_type:law_anarchy
	}
}
te_demog_ms_communal_dystopia = {
	has_law = law_type:law_communal_child_rearing
	OR = {
		has_law = law_type:law_single_party_state
		has_law_or_variant = law_type:law_autocracy
	}
	has_law = law_type:law_outlawed_dissent
}

# ---- The term (§3.3) ------------------------------------------------------------------------------------------------
# Country: measure $M$ has run its five years, so it may be switched off (te_demog_ms_years_<measure> reads -1 while it
# is off). The Crash Campaign ends itself then (te_demog_ms_monthly).
te_demog_ms_term_run = {
	te_demog_ms_years_$M$ >= 5
}

# Country: the Crash Campaign ended less than ten years ago and can't return yet.
te_demog_ms_crash_campaign_barred = {
	has_variable = te_dg_ms_crash_campaign_end
	te_demog_ms_years_since_crash_campaign < 10
}

# ---- The AI (§3.6) ----------------------------------------------------------------------------------------------------
# The laws' weights (every game, owner call 7: ai_enact_weight_modifier in extra_laws.txt) and the measures' yearly
# pulse (Full, te_demog_ms_ai_choose) read the same conditions, which any country has; is_ai = yes only decides whom
# the pulse picks for. Where the census runs, its figures add to the walk's.

# Country: the population shrank over the last year by more than a tenth of a percent, by the yearly walk's figures
# (every game). The deadband keeps a country near zero growth from swinging between the two laws' weights each year.
te_demog_ms_ai_shrinking = {
	te_demog_ms_pop_trend < -0.001
}

# Country: the population shrinks fast, more than half a percent a year.
te_demog_ms_ai_shrinking_fast = {
	te_demog_ms_pop_trend < -0.005
}

# Country: the census counts many old people, 15% or more aged 65 and over (0 where the census doesn't run).
te_demog_ms_ai_old = {
	te_demog_ms_old_share >= 0.15
}

# Country: a quarter or more aged 65 and over.
te_demog_ms_ai_very_old = {
	te_demog_ms_old_share >= 0.25
}

# Country: a youth bulge, more than 18% of the people aged 20 to 29 (the census; §4.3's measure).
te_demog_ms_ai_youth = {
	te_demog_ms_youth_share > 0.18
}

# Country: half its people or more live in states Migration Crowding presses at its ceiling (te_demog_ms_crowded_share).
te_demog_ms_ai_crowded = {
	te_demog_ms_crowded_share >= 0.5
}

# Country: a reason to want more births: the population shrinks or the old are many.
te_demog_ms_ai_pronatal = {
	OR = {
		te_demog_ms_ai_shrinking = yes
		te_demog_ms_ai_old = yes
	}
}

# Country: the pro-natalist target is far off: the population shrinks fast or a quarter are old.
te_demog_ms_ai_pronatal_far = {
	OR = {
		te_demog_ms_ai_shrinking_fast = yes
		te_demog_ms_ai_very_old = yes
	}
}

# Country: a reason to want fewer births: famine, crowding or a youth bulge.
te_demog_ms_ai_popcontrol = {
	OR = {
		has_famine = yes
		te_demog_ms_ai_crowded = yes
		te_demog_ms_ai_youth = yes
	}
}

# Country: the population-control target is far off: famine with crowding or a youth bulge.
te_demog_ms_ai_popcontrol_far = {
	has_famine = yes
	OR = {
		te_demog_ms_ai_crowded = yes
		te_demog_ms_ai_youth = yes
	}
}

# Country: the treasury can take on a money upkeep (one a pulse: the pulse's local te_dg_ms_money counts the money
# measures it started, te_demog_ms_ai_choose), and still carry one.
te_demog_ms_ai_money_start = {
	in_default = no
	scaled_debt < 0.25
	local_var:te_dg_ms_money < 1
}
te_demog_ms_ai_money_keep = {
	in_default = no
	scaled_debt < 0.5
}

# Country: legitimacy high enough for a coercive measure (Restricted Contraception, Birth Limit's one child, the Crash
# Campaign: §3.6, "only when a target is far off and legitimacy is high").
te_demog_ms_ai_legitimate = {
	government_legitimacy >= 75
}
