# ============================================================================
# TABLING A RESOLUTION — SHARED GATES
# ============================================================================
# The six topics a member can put before the General Assembly on its own
# initiative, each with the two gates its journal-entry scripted button used to
# carry inline:
#
#   un_propose_<key>_available   the button's `visible` — whether the topic is
#                                on the table at all for us right now (there is
#                                a case, we are not already enforcing it, ...)
#   un_propose_<key>_possible    the button's `possible` — every cost, cooldown,
#                                lock and standing gate, with the custom_tooltip
#                                lines that explain a refusal
#
# Extracted word for word out of common/scripted_buttons/un_buttons.txt so the
# button and the chamber widget's "Table a Resolution" section cannot drift:
# the button `possible = { un_propose_condemn_possible = yes }`, and
# un_chamber_propose_sgui's `is_valid` asks the same trigger. The AI still
# proposes through the buttons' ai_chance, so both callers must stay honest.
#
# Charter reform joined them in redesign phase 2 as the seventh (the reform
# topic, un_propose_reform_*): it is also still raised by its proposer event,
# un_events.6, under the same gates.
#
# THE TEN CONVENTIONS (human rights, ICC, NPT, climate, pandemic, refugee,
# heritage, decolonization, space, law of the sea) are at the bottom of the
# file. The docket raises them for everyone through the proposer events in
# events/un_events.txt; a human member may also table one from the chamber on
# its own motion. They have no scripted button: the AI reaches them only
# through the docket, which paces the floor (un_docket_effects.txt), and a
# button with an ai_chance would let it skip that pacing. Each has, besides
# the two gates above,
#
#   un_propose_<key>_in_force    the question is settled: its agency founded,
#                                or (decolonization) its regime standing. The
#                                chamber hides the Propose control.
#   un_propose_<key>_qualifies   the per-country conditions the docket also
#                                asks before offering it (technology, rank,
#                                not already a party, no recent refusal):
#                                un_docket_may_propose_<key> calls it, so the
#                                two paths cannot drift.
#
# and its global conditions (cooldown, authority) are the docket's own
# un_docket_topic_open_<key>, called here rather than restated.
# ============================================================================

# Condemnation of aggression — binding, simple majority, names a rival that is
# prosecuting a war of its own making.
# Scope: the proposing country
un_propose_condemn_available = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	any_country = {
		has_diplomatic_pact = { who = ROOT type = rivalry }
		is_at_war = yes
		is_diplomatic_play_initiator = yes
		# Grounds (redesign phase 3): its record must support a censure.
		un_case_supports = { TOPIC = condemn }
	}
}

# Scope: the proposing country
un_propose_condemn_possible = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	custom_tooltip = {
		text = un_no_active_vote_tt
		NOT = { has_global_variable = un_vote_active }
	}
	custom_tooltip = {
		text = un_no_resolution_cooldown_tt
		NOT = { un_resolution_topic_on_cooldown = { TOPIC = condemn } }
	}
	NOT = { has_modifier = un_ga_resolution_modifier }
	NOT = { has_modifier = un_request_cooldown }
	custom_tooltip = {
		text = un_recess_ai_tt
		un_may_table_now = yes
	}
	custom_tooltip = {
		text = un_article_19_table_tt
		NOT = { un_dues_vote_suspended = yes }
	}
	custom_tooltip = {
		text = un_representation_table_tt
		NOT = { un_representation_suspended = yes }
	}
	custom_tooltip = {
		text = un_grounds_condemn_tt
		any_country = {
			has_diplomatic_pact = { who = ROOT type = rivalry }
			is_at_war = yes
			is_diplomatic_play_initiator = yes
			un_case_supports = { TOPIC = condemn }
		}
	}
}

# Authorized military mandate — binding, simple majority, and the only topic
# whose proposal also names a territory. Gates are the fullest of the six.
# Scope: the proposing country
un_propose_mandate_available = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	has_global_variable = un_authority
	NOT = { un_mandate_has_live_mandate = yes }
	un_mandate_case_exists = yes
}

# Scope: the proposing country
un_propose_mandate_possible = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	custom_tooltip = {
		text = un_no_active_vote_tt
		NOT = { has_global_variable = un_vote_active }
	}
	custom_tooltip = {
		text = un_mandate_no_live_mandate_tt
		NOT = { un_mandate_has_live_mandate = yes }
	}
	custom_tooltip = {
		text = un_mandate_authority_tt
		has_global_variable = un_authority
		global_var:un_authority >= 40
	}
	custom_tooltip = {
		text = un_mandate_cooldown_tt
		NOT = { has_variable = un_mandate_proposal_cooldown }
	}
	# Standing floor, set at the Disgraced boundary (20). A member that has
	# done nothing at all sits at the neutral 50, so this bars only a
	# country whose record the Assembly has already wrecked — it cannot
	# make the mandate unreachable for an ordinary member.
	#
	# Deliberately NOT added to un_mandate_actor_in_good_standing: that
	# trigger is also read by un_mandate_still_valid, so a floor there
	# would VOID a mandate already in the field the moment the actor's
	# standing dipped — deleting a war the player is fighting over a score
	# movement. The gate belongs on the proposal, not on the authorization.
	custom_tooltip = {
		text = un_mandate_standing_tt
		un_standing_at_least = { VALUE = un_standing_mandate_floor }
	}
	custom_tooltip = {
		text = un_mandate_case_tt
		un_mandate_case_exists = yes
	}
	NOT = { has_modifier = un_ga_resolution_modifier }
	NOT = { has_modifier = un_request_cooldown }
	custom_tooltip = {
		text = un_recess_ai_tt
		un_may_table_now = yes
	}
	custom_tooltip = {
		text = un_article_19_table_tt
		NOT = { un_dues_vote_suspended = yes }
	}
	custom_tooltip = {
		text = un_representation_table_tt
		NOT = { un_representation_suspended = yes }
	}
}

# ---- International Refugee Resolution (UNHCR) ----
# The one convention that names a country: the one the displaced are fleeing
# (un_propose_refugee_select_target). It accuses nobody.

# A country the displaced could be fleeing: at war, or with a state in misery.
# Call it directly inside a country iterator entered from the proposer, so
# that PREV is the proposer.
# Scope: the candidate source
un_propose_refugee_source = {
	NOT = { this = PREV }
	NOT = { is_country_type = decentralized }
	OR = {
		is_at_war = yes
		any_scope_state = { average_sol < 6 }
	}
}

# The graver case, preferred when there is one: famine, or a war that has
# devastated a state. Same calling rule as un_propose_refugee_source.
# Scope: the candidate source
un_propose_refugee_grave_source = {
	NOT = { this = PREV }
	NOT = { is_country_type = decentralized }
	OR = {
		any_scope_state = { has_famine = yes }
		AND = {
			is_at_war = yes
			any_scope_state = { devastation > un_docket_war_devastation }
		}
	}
}

# Scope: any
un_propose_refugee_in_force = {
	has_global_variable = un_agency_unhcr
}

# Scope: the prospective proposer
un_propose_refugee_qualifies = {
	custom_tooltip = {
		text = un_propose_not_party_tt
		NOT = { je:je_united_nations ?= { has_modifier = un_refugee_program_modifier } }
	}
	custom_tooltip = {
		text = un_propose_not_refused_tt
		NOT = { has_modifier = un_refugee_closed_modifier }
	}
	custom_tooltip = {
		text = un_propose_refugee_grounds_tt
		any_country = { un_propose_refugee_source = yes }
	}
}

# Scope: the proposing country

	NOT = { un_propose_refugee_in_force = yes }
