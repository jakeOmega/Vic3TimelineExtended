# ======================================================================
#  Grand Monument Scripted Triggers
#  Country scope. Used by the dedication ceremony (monument_events.2) to
#  gate which dedications a country may raise.
# ======================================================================
# These two triggers carve lawgroup_governance_principles for the dedication
# ceremony: monument_government_is_crowned gates To the Crown (pm_monument_crown)
# and excludes To the Leader (a crowned ruler's statue is To the Crown, not To
# the Leader); monument_government_is_republican gates To the Republic
# (pm_monument_republic). The neutral wording is its own dedication, To the
# Nation (pm_monument_civic), which every government -- crowned, republican or
# neither -- may raise regardless of these two triggers.
#
# has_law_or_variant throughout: variant laws (parent = law_X) are tag- and
# condition-restricted skins of their parent and must count as the parent here.

# True if the head of state wears a crown (or the equivalent).
monument_government_is_crowned = {
	OR = {
		has_law_or_variant = law_type:law_monarchy
		has_law_or_variant = law_type:law_social_monarchy
		has_law_or_variant = law_type:law_chiefdom
	}
}

# True if the head of state is chosen rather than born to it.
# Deliberately excludes law_corporate_state, law_theocracy, law_neocameralism and
# law_colonial_administration -- all of them are "the republic" only by courtesy.
# Those governments cannot raise To the Republic; every government -- these
# included -- can raise To the Nation instead.
monument_government_is_republican = {
	OR = {
		has_law_or_variant = law_type:law_presidential_republic
		has_law_or_variant = law_type:law_parliamentary_republic
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_direct_democracy
	}
}

gm_system_enabled = {

}

# Country scope. The Timeline Extended window's Grand Monuments tab checklist
# (te_window_grand_monuments_tab_unlock_sgui's is_valid, rendered by

gm_entry_unlocked = {

			has_variable = gm_com_offered
			gm_tr_country_has_any = yes
		}
	}
}

# Country scope. A personality cult: Autocracy or a Single-Party State under a
# head of state who is not crowned (a crowned ruler's statue is To the Crown).
gm_can_raise_leader_monument = {
	OR = {
		has_law_or_variant = law_type:law_autocracy
		has_law_or_variant = law_type:law_single_party_state
	}
	monument_government_is_crowned = no
}

# Country scope. Hard times: finishing a monument level now causes vanity
# backlash (§5). finance_cycle_value exists only once the banking journal
# entry has run, so the recession read is guarded.
gm_country_hard_times = {
	OR = {
		in_default = yes
		has_famine = yes
		AND = {
			has_variable = finance_cycle_value
			banking_cycle_is_recession = yes
		}
	}
}

# ======================================================================
#  A monument's kind, fit and status (spec §1, §2.3, §3)
# ======================================================================
# State scope. Everything a monument must remember lives on its STATE
# (buildings hold no variables; one Grand Monument per state):
#   gm_raised_by      the country that dedicated it (regime, ruler, faith)
#   gm_honoree        To the Leader: the ruler it honours
#   gm_honoree_ig     To the Leader: that ruler's IG, as a flag
#   gm_faith          Grand Shrine: the religion it honours
#   gm_skin           flag: the form it takes (§1.1)
#   gm_seen           its records have been written
#   gm_contested      months it has stood contested (§4)
#   gm_heritage       preserved as heritage (§4.3)
#   gm_base_ig / gm_supporter_ig   IG flags recorded at contest (§4.1)
#   gm_landmark       flag: a monument standing in 1836, its key (phase 4,
#                     gm_landmark_effects.txt); locks its skin and form
# A missing record reads as "does not fit", never as "fits".

gm_state_has_pm = {
	b:building_grand_monument ?= { has_active_production_method = $PM$ }
}

# A rebuilt undedicated monument stands on the engine's copy of its siblings'
# dedication until its own method reads back (gm_restore_pm = undedicated,
# gm_state_mark_restore_pm): undedicated meanwhile, so no sum, first sight or
# contest takes the copy for a dedication.
gm_state_is_dedicated = {
	has_building = building_grand_monument
	NOT = { gm_state_has_pm = { PM = pm_monument_undedicated } }
	NOT = {
		AND = {
			has_variable = gm_restore_pm
			var:gm_restore_pm = flag:undedicated
		}
	}
}

# Dedicate (owner, 2026-10-18): a construction site waits as long as its owner
# likes, and its Dedicate button (gm_dedicate_sgui) opens the ceremony (sites
# spec 2026-10-18 §3.1).
gm_state_awaits_dedication = {
	gm_system_enabled = yes
	gm_state_has_site = yes
}

# The AI has no button: its ceremony asks once nothing more is queued on its
# construction site, when its monuments were finished before the button.
# Whether the site still reads under construction inside its last level's
# on_building_built is untested; the refresh's check
# (gm_state_offer_missed_ceremony) asks within the month either way.
gm_state_ai_dedicates_now = {
	owner = { is_ai = yes }
	gm_state_has_site = yes
	gm_state_site_level >= 1
	b:building_grand_monument_site ?= { is_under_construction = no }
}

# ---- Landmarks (phase 4, spec 2026-10-12 §3.2) ----------------------------------
# A landmark keeps its skin and form, and so its picture, while it stands:
# through contest, heritage, Rededicate, conquest and civil war. Pull Down or
# demolition ends it (gm_clear_state).
gm_state_is_landmark = {
	has_variable = gm_landmark
}

# Its historical name stands: Rename, Name It and a completed opportunity leave it
# alone. A rededication names it again (gm_state_unveil), and the namesake
# choice can give the historical name back (monument_events.25).
gm_state_has_historical_name = {
	gm_state_is_landmark = yes
	gm_state_namesake_is = { NAMESAKE = landmark }
	NOT = { has_variable = gm_name }
}

gm_state_kind_regime = {
	OR = {
		gm_state_has_pm = { PM = pm_monument_crown }
		gm_state_has_pm = { PM = pm_monument_republic }
		gm_state_has_pm = { PM = pm_monument_revolution }
	}
}

gm_state_kind_ruler = {
	gm_state_has_pm = { PM = pm_monument_leader }
}

gm_state_kind_faith = {
	gm_state_has_pm = { PM = pm_monument_religious }
}

# Regime, ruler or faith: a message that can fall. Everything else is timeless.
gm_state_is_bound = {
	OR = {
		gm_state_kind_regime = yes
		gm_state_kind_ruler = yes
		gm_state_kind_faith = yes
	}
}

# The country that raised it still owns it. Not true of a conquered monument,
# whatever the conqueror's laws (plan decision 2).
gm_state_raised_by_owner = {
	has_variable = gm_raised_by
	var:gm_raised_by ?= { save_temporary_scope_as = gm_tmp_raiser }
	owner = scope:gm_tmp_raiser
}

# The honoured character still rules. A dead honoree fails the `?=`.
gm_state_leader_fits = {
	has_variable = gm_honoree
	var:gm_honoree ?= { save_temporary_scope_as = gm_tmp_honoree }
	owner = {
		ruler ?= { this = scope:gm_tmp_honoree }
	}
}

# The state religion is still the one it honours, and not under State Atheism.
gm_state_shrine_fits = {
	has_variable = gm_faith
	owner = {
		NOT = { has_law_or_variant = law_type:law_state_atheism }
	}
	var:gm_faith ?= { save_temporary_scope_as = gm_tmp_faith }
	owner = { religion = scope:gm_tmp_faith }
}

# Its message still stands. Read live every month from the owner's laws,
# religion and ruler; never cached.
gm_state_message_fits = {
	OR = {
		gm_state_is_bound = no
		AND = {
			gm_state_raised_by_owner = yes
			OR = {
				AND = {
					gm_state_has_pm = { PM = pm_monument_crown }
					owner = { monument_government_is_crowned = yes }
				}
				AND = {
					gm_state_has_pm = { PM = pm_monument_republic }
					owner = { monument_government_is_republican = yes }
				}
				AND = {
					gm_state_has_pm = { PM = pm_monument_revolution }
					owner = { gm_government_is_revolutionary = yes }
				}
				AND = {
					gm_state_kind_ruler = yes
					gm_state_leader_fits = yes
				}
				AND = {
					gm_state_kind_faith = yes
					gm_state_shrine_fits = yes
				}
			}
		}
	}
}

gm_state_is_contested = {
	has_variable = gm_contested
}

# A monument founded beside another (gm_newcomer, spec 2026-10-16 §3.2) is
# dedicated, finished and named, with nothing under way: the settle check may
# give the state's first place back to a grander one (gm_state_settle_newcomer).
gm_state_newcomer_settled = {
	has_variable = gm_newcomer
	gm_state_is_dedicated = yes
	has_variable = gm_seen
	has_variable = gm_levels
	OR = {
		has_variable = gm_namesake
		has_variable = gm_name
	}
	NOT = { has_variable = gm_unveil_pending }
	NOT = { has_variable = gm_name_choice_open }
	NOT = { has_variable = gm_skin_axis }
	NOT = { has_variable = gm_being_named }
	NOT = { has_variable = gm_restore_pm }
	NOT = { has_variable = gm_ceremony_pending }
	NOT = { has_variable = gm_rebuilding }
}

# Its years add to its grandeur this month (spec 2026-10-16 §2): dedicated and
# recorded, not contested (a heritage monument ages), and not while its method
# is being put back, a ceremony asks or a rebuild is under way. An undedicated
# monument is still being built and never ages.
gm_state_ages = {
	gm_state_is_dedicated = yes
	has_variable = gm_seen
	gm_state_is_contested = no
	NOT = { has_variable = gm_restore_pm }
	NOT = { has_variable = gm_ceremony_pending }
	NOT = { has_variable = gm_rebuilding }
}

gm_state_is_heritage = {
	has_variable = gm_heritage
}

gm_state_status_fits = {

	gm_state_is_contested = no

# §2.3 standing grandeur (prestige, cultural pull): every monument not contested.
gm_state_counts_standing = {

# State Ceremonial's venues (policy spec 2026-10-23 § 4.2): every monument not
# contested, heritage included, so a new regime can hold its events in the old
# king's palace (owner). A To History monument counts at half (spec § 4.5,
# gm_sum_venues).
gm_state_counts_venue = {
	has_building = building_grand_monument
	gm_state_is_contested = no
	OR = {
		gm_state_is_dedicated = yes
		gm_state_is_heritage = yes
	}
}

# §1.1: the dedications that take heritage skins. The Grand Shrine takes faith

gm_state_takes_heritage_skin = {
	OR = {
		gm_state_kind_regime = yes

}

	}
}

gm_ledger_idle = {
	OR = {
		NOT = { has_variable = $VAR$ }
		AND = {
			var:$VAR$ < 0.5

		}
	}
}

	has_variable = gm_dedication

		AND = { var:gm_dedication = flag:war_memorial gm_state_has_pm = { PM = pm_monument_war_memorial } }

		AND = { var:gm_dedication = flag:naturalist gm_state_has_pm = { PM = pm_monument_naturalist } }

# State scope. A finished monument's building went while its records stand

	NOT = { has_variable = gm_carried_off }

# State scope. A finished building no record describes, from no older save and

gm_state_stray_building = {
	has_global_variable = gm_sites_marked
	has_building = building_grand_monument
	NOT = { has_variable = gm_migrated }
	NOT = { has_variable = gm_premigration }
	NOT = { has_variable = gm_rebuilding }
	gm_state_orphan_in_region = no
}
