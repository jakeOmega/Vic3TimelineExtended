# Private Actors, the trust and the foundation: the triggers (docs/systems/mod_systems/private_actors.md § The Trust;
# spec §§ 4.6, 5.5, 6.1-6.4). Company scope unless noted. Each verb is a te_pa_can_<verb> trigger here (legal,
# affordable, applicable) and a te_pa_do_<verb> effect in te_pa_trust_effects.txt; te_pa_act's lists name them.

# ---------------------------------------------------------------------------
# The antitrust law and the cartel (§ 6.2)
# ---------------------------------------------------------------------------

# No cartel at all: private ownership has ended, or the antitrust law is Command or Cooperative Economy.
te_pa_cartel_banned = {
	owner = { te_pa_dominance_illegal = yes }
}

# Its sector is infrastructure or a utility (§ 6.2's Regulated Utilities row: Task 1's features).
te_pa_cartel_utility = {
	OR = {
		var:te_pa_g_f_infrastructure ?= 1
		var:te_pa_g_f_power ?= 1
	}
}

# The antitrust law in force lets the firm hold a cartel in the open: Guilds and Chartered Monopolies, Freedom of
# Contract, Regulated Utilities outside infrastructure and utilities, or Dirigiste Policy for a Favored firm; or it
# holds a monopoly charter (Task 4b, § 6.5). Under
# any other (Antitrust Enforcement, Regulated Utilities in its own sectors, Dirigiste Policy for a firm not Favored)
# a cartel is illegal and forms in secret.
te_pa_cartel_lawful = {
	te_pa_cartel_banned = no
	OR = {
		# A monopoly charter (§ 6.5) makes the cartel the state's own grant, open and lawful under any of these laws.
		has_variable = te_pa_charter
		owner = { has_law = law_type:law_guilds_chartered_monopolies }
		owner = { has_law = law_type:law_freedom_of_contract }
		AND = {
			owner = { has_law = law_type:law_regulated_utilities }
			te_pa_cartel_utility = no
		}
		AND = {
			owner = { has_law = law_type:law_dirigisme }
			var:te_pa_posture ?= 1
		}
	}
}

# Regulated Utilities caps the cartel at te_pa_cartel_ru_cap outside infrastructure and utilities.
te_pa_cartel_ru_capped = {
	owner = { has_law = law_type:law_regulated_utilities }
	te_pa_cartel_utility = no
}

# Country scope. The antitrust law has a regulator to capture (§ 6.2's column).
te_pa_law_has_regulator = {
	OR = {
		has_law = law_type:law_trust_busting
		has_law = law_type:law_regulated_utilities
		has_law = law_type:law_dirigisme
	}
}

# Country scope. A regulator is bought half again as fast (§ 6.4): Contracted Administration, or the Spoils System.
te_pa_reg_fast = {
	OR = {
		has_law = law_type:law_contracted_administration
		active_law:lawgroup_bureaucracy ?= { has_amendment = amendment_type:amendment_spoils_system }
	}
}

# The firm holds a cartel the state knows of: open, or exposed.
te_pa_holds_known_cartel = {
	te_pa_v_cartel > 0
	var:te_pa_cartel_known ?= 1
}

# A law being enacted would outlaw its cartel (§ 6.4's stall): one that bans any cartel (Command or Cooperative
# Economy, Command Economy, Cooperative Ownership), or, while the cartel is lawful, one under which it would not be.
te_pa_law_threatens_cartel = {
	owner = { enacting_any_law = yes }
	OR = {
		owner = {
			OR = {
				is_enacting_law = law_type:law_command_cooperative_economy
				is_enacting_law = law_type:law_command_economy
				is_enacting_law = law_type:law_cooperative_ownership
			}
		}
		AND = {
			te_pa_cartel_lawful = yes
			OR = {
				owner = { is_enacting_law = law_type:law_trust_busting }
				AND = {
					owner = { is_enacting_law = law_type:law_regulated_utilities }
					te_pa_cartel_utility = yes
				}
				AND = {
					owner = { is_enacting_law = law_type:law_dirigisme }
					NOT = { var:te_pa_posture ?= 1 }
				}
			}
		}
	}
}

# A Favored firm's cartel is immune from suit while the antitrust law carries National Champion Exemption (§ 6.7).
# Task 4b's suit reads it.
te_pa_cartel_immune = {
	var:te_pa_posture ?= 1
	owner = {
		active_law:lawgroup_antitrust ?= { has_amendment = amendment_type:amendment_national_champion_exemption }
	}
}

# ---------------------------------------------------------------------------
# The verbs (te_pa_act's lists, te_pa_trust_effects.txt)
# ---------------------------------------------------------------------------

# The firm holds AMOUNT (a script value, pounds) in funds, and AMOUNT is above 0.
te_pa_can_afford = {
	$AMOUNT$ > 0
	te_pa_v_funds >= $AMOUNT$
}

# Form a cartel (§ 6.1 stage 3): it holds none, its goal is Dominance or A Protected Market, and its share is at the
# line; any law but one that bans cartels outright (an illegal one forms in secret); and no court has broken one of
# its cartels in the last ten years.
te_pa_can_form_cartel = {
	te_pa_v_cartel <= 0
	OR = {
		var:te_pa_goal ?= 5
		var:te_pa_goal ?= 2
	}
	te_pa_cartel_banned = no
	te_pa_v_share >= te_pa_cartel_threshold_value
	# A suit that broke its cartel bars a new one for ten years (§ 6.6, te_pa_case_verdict).
	te_pa_wait_over = { VAR = te_pa_cartel_barred_until }
}

# Capture the regulator (§ 6.4): it holds a cartel and no regulator, the law has one, and it has a quarter's
# Connections to begin.
te_pa_can_capture_reg = {
	te_pa_v_cartel > 0
	te_pa_v_reg <= 0
	owner = { te_pa_law_has_regulator = yes }
	te_pa_v_conn >= te_pa_reg_conn_quarter
}

# Stall the law (§ 6.4): te_pa_can_stall, widened by Task 5a to any law a firm is against, and the quarter's
# te_pa_can_stall_cartel are te_pa_politics_triggers.txt's.

# Expand (§ 6.1 stage 1: it builds in its sector): below the cartel's share, private ownership stands, the
# investment pool holds money, and its seat can take a level of one of its building types.
te_pa_can_expand = {
	te_pa_v_share < te_pa_cartel_threshold_value
	owner = {
		te_pa_private_ownership_ended = no
		investment_pool > 0
	}
	te_pa_gen_can_expand = yes
}

# The cartel's firm (an entry of the country list te_pa_cartels), with scope:te_pa_firm a buyer at its turn: it is
# another firm, holds a secret cartel, and makes one of the buyer's two inputs.
te_pa_cartel_sells_to_buyer = {
	var:te_pa_cartel_known ?= 0
	te_pa_cartel_makes_buyer_input = yes
}

# The cartel's firm (an entry of te_pa_cartels), with scope:te_pa_firm a buyer: it is another firm, holds a cartel,
# and its output is one of the buyer's two inputs. Secret or known: the routine verb reads a secret one
# (te_pa_cartel_sells_to_buyer), the petition a known one (te_pa_cartel_suable_by_buyer).
te_pa_cartel_makes_buyer_input = {
	NOT = { this = scope:te_pa_firm }
	te_pa_v_cartel > 0
	has_variable = te_pa_g_out_good
	var:te_pa_g_out_good > 0
	OR = {
		AND = {
			var:te_pa_g_out_good >= te_pa_buyer_in1
			var:te_pa_g_out_good <= te_pa_buyer_in1
		}
		AND = {
			var:te_pa_g_out_good >= te_pa_buyer_in2
			var:te_pa_g_out_good <= te_pa_buyer_in2
		}
	}
}

# The cartel's firm, with scope:te_pa_firm a buyer at its turn: a cartel the state knows of and could sue (te_pa_sa_suable:
# known, illegal under the law in force, not immune, no case under way) that makes one of the buyer's inputs.
te_pa_cartel_suable_by_buyer = {
	te_pa_cartel_makes_buyer_input = yes
	te_pa_sa_suable = yes
}

# Pay to expose a cartel (§ 6.3, a buyer's routine verb): its goal is Cheaper Inputs, a secret cartel in its country
# makes one of its inputs, and it can pay.
te_pa_can_press_cartel = {
	var:te_pa_goal ?= 1
	te_pa_can_afford = { AMOUNT = te_pa_buyer_cost_value }
	owner = {
		has_variable_list = te_pa_cartels
		any_in_list = {
			variable = te_pa_cartels
			te_pa_cartel_sells_to_buyer = yes
		}
	}
}

# Endow a foundation (§ 5.5; checked each turn after the goal's verb): it has none, its Standing is under the line
# its Paternalism sets, it has the founding cost and a year's accrual besides, private ownership stands, and some
# incorporated home state holds no foundation yet.
te_pa_can_endow = {
	te_pa_v_found <= 0
	te_pa_v_stand < te_pa_found_stand_line
	te_pa_v_funds >= te_pa_found_spare_value
	owner = {
		te_pa_private_ownership_ended = no
		any_scope_state = {
			is_incorporated = yes
			NOT = { has_variable = te_pa_st_found_str }
		}
	}
}

# ---------------------------------------------------------------------------
# The trust's petitions (Task 4b; the frame: te_pa_moves_effects.txt)
# ---------------------------------------------------------------------------

# Country scope. The antitrust law lets a firm ask for a monopoly charter (§ 6.2's first row): Guilds and Chartered
# Monopolies. § 6.2 gives Freedom of Contract a legal cartel but no charter.
te_pa_charter_law = {
	has_law = law_type:law_guilds_chartered_monopolies
}

# Kind charter (§ 6.5), at its turn: a cartel at strength 80 under a law that allows monopolies, or any cartel of a
# Favored firm, declares. The cartel must be one the state knows of: a major move is never secret (§ 4.6), so a firm
# with a secret cartel doesn't announce it by asking. It holds no charter yet, and five years have passed since it was
# last refused or left unanswered.
te_pa_can_declare_charter = {
	te_pa_move_open = yes
	te_pa_holds_known_cartel = yes
	NOT = { has_variable = te_pa_charter }
	OR = {
		AND = {
			te_pa_v_cartel >= te_pa_cartel_mark_2
			owner = { te_pa_charter_law = yes }
		}
		var:te_pa_posture ?= 1
	}
	te_pa_wait_over = { VAR = te_pa_charter_again }
}

# Kind suit (§ 6.3's major move), at a buyer's turn: it seeks Cheaper Inputs with that goal's score past
# te_pa_suit_petition_score and Connections past te_pa_suit_petition_conn, a cartel the state could sue makes one of
# its inputs, and five years have passed since it was last refused or left unanswered. (§ 6.3's other form, a
# petition to change the antitrust law, waits for Task 5's petition for a law.)
te_pa_can_declare_suit = {
	te_pa_move_open = yes
	var:te_pa_goal ?= 1
	var:te_pa_s_inputs >= te_pa_suit_petition_score
	te_pa_v_conn >= te_pa_suit_petition_conn
	te_pa_wait_over = { VAR = te_pa_suit_again }
	owner = {
		has_variable_list = te_pa_cartels
		any_in_list = {
			variable = te_pa_cartels
			te_pa_cartel_suable_by_buyer = yes
		}
	}
}

# Country scope. With the rule on, one of its firms holds a cartel the state knows of: "Too Large to Break"
# (amendment_petition_events.4, § 6.7) is then the largest such cartel's firm's petition (te_pa_save_largest_cartel).
te_pa_holds_known_cartels = {
	te_pa_on = yes
	has_variable_list = te_pa_cartels
	any_in_list = {
		variable = te_pa_cartels
		te_pa_holds_known_cartel = yes
	}
}

# The firm's foundation's state is still its home country's.
te_pa_found_state_kept = {
	var:te_pa_found_state ?= {
		exists = owner
		owner = scope:te_pa_home
	}
}
