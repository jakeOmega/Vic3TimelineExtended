# ============================================================================
# UN RESOLUTION TRIGGERS
# Checks on General Assembly resolutions. Each resolution is a script
# container; see common/scripted_effects/un_vote_effects.txt for its tags,
# variables and lifecycle.
# ============================================================================

# Whether $TOPIC$ can't be proposed yet because a resolution on it closed
# recently. A closed resolution carries un_res_cooldown_months, counted down
# from 60 (120 for expulsion) by un_resolution_cooldowns_tick, and stays in
# un_resolution_history at least that long.
# Parameters: $TOPIC$ = topic key (condemn, human_rights, sanctions, ...)
# Scope: any
un_resolution_topic_on_cooldown = {
	has_global_variable_list = un_resolution_history
	any_in_global_list = {
		variable = un_resolution_history
		has_tag = un_topic_$TOPIC$
		has_variable = un_res_cooldown_months
	}
}

# Whether the resolution is *binding* (vetoable). The 6 binding topics are
# sanctions, peacekeeping_request, icc, condemn, reform and military_mandate;
# all others are recommendatory and cannot be vetoed.
#
# military_mandate and reform are the two with NO graduated form: a veto on an
# authorization to use force blocks it outright, because there is no coherent
# weaker version of "you may go to war over this" (un_vote.2 applies no effect
# for either). See docs/systems/journal_entry_systems.md § Military mandates.
# Scope: resolution container
un_resolution_is_binding = {
	OR = {
		has_tag = un_topic_sanctions
		has_tag = un_topic_peacekeeping_request
		has_tag = un_topic_icc
		has_tag = un_topic_condemn
		has_tag = un_topic_reform
		has_tag = un_topic_military_mandate
	}
}

# Whether the resolution is decided by a two-thirds supermajority of ALL
# current members (un_vote_supermajority_passed) rather than a simple majority
# of the votes cast: the motion to strip a permanent seat, and (since redesign
# phase 2) a charter reform. un_resolution_decide and the chamber's tally and
# projection lines all read this one trigger.
# Scope: resolution container
un_resolution_needs_supermajority = {
	OR = {
		has_tag = un_topic_expulsion
		has_tag = un_topic_reform
	}
}

# Whether the resolution is PUNITIVE: it punishes a named country on the
# strength of that country's record, so it needs grounds (the target's case
# strength must reach the topic's threshold, un_res_case_threshold) and it is
# weighed on the grounds in every member's lean (redesign phase 3).
# Scope: resolution container
un_resolution_is_punitive = {
	OR = {
		has_tag = un_topic_condemn
		has_tag = un_topic_sanctions
		has_tag = un_topic_military_mandate
	}
}

# Whether the resolution ACCUSES its target: the punitive topics, and the
# motion to strip a permanent seat. Allies and rivals of the target, and the
# target itself, weigh these as such (un_lean_target, un_lean_self). The
# motion needs no grounds from the dossier: its case is the veto abuse its
# proposal gate already requires.
# Scope: resolution container
un_resolution_accuses_target = {
	OR = {
		un_resolution_is_punitive = yes
		has_tag = un_topic_expulsion
	}
}

# Whether ROOT is on the resolution's $LIST$ of voters.
# Parameters: $LIST$ = un_res_yes | un_res_no
# Scope: resolution container; ROOT = the country asked about
un_resolution_has_vote = {
	has_variable_list = $LIST$
	is_target_in_variable_list = { name = $LIST$ target = ROOT }
}

# Whether ROOT may still cast a ballot on scope:un_resolution. This is
# un_vote.1's own `trigger`, word for word, pulled out so the shared cast
# effects (common/scripted_effects/un_vote_cast_effects.txt) can re-assert it
# at the moment the option is actually pressed. A delayed trigger_event
# re-evaluates the event trigger when it FIRES, so a popup queued before the
# player voted in the chamber never appears; what this guards is the popup that
# had already fired and was left sitting in the notification list.
# Article 19 (redesign phase 5): a member two years in arrears on its dues has
# lost its vote until it pays. §0.8: a member whose representation is
# suspended has no vote either.
# Scope: the voting country; expects scope:un_resolution
un_vote_can_cast_ballot = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	NOT = { un_dues_vote_suspended = yes }
	NOT = { un_representation_suspended = yes }
	scope:un_resolution ?= {
		has_tag = un_res_voting
		NOT = { un_resolution_has_vote = { LIST = un_res_yes } }
		NOT = { un_resolution_has_vote = { LIST = un_res_no } }
	}
}

# The same question asked of the OPEN resolution rather than of a saved scope.
# A scripted GUI's is_shown / is_valid cannot save a scope, so the chamber's
# vote controls have to reach global_var:un_active_resolution directly.
# un_lobby_has_voted covers both ballot lists, a veto included.
# Scope: the voting country
un_chamber_ballot_open = {
	NOT = { un_dues_vote_suspended = yes }
	NOT = { un_representation_suspended = yes }
	has_global_variable = un_active_resolution
	global_var:un_active_resolution ?= {
		has_tag = un_res_voting
		NOT = { un_lobby_has_voted = { COUNTRY = ROOT } }
	}
}

# Whether ROOT tabled the open resolution. Checked before "already voted" in the
# chamber's vote controls, because un_resolution_open books the proposer into
# un_res_yes — without this the proposer would be told it had already voted
# rather than that its vote is the proposal.
# Scope: country
un_chamber_is_proposer = {
	has_global_variable = un_active_resolution
	global_var:un_active_resolution ?= { var:un_res_proposer ?= ROOT }
}

# Whether the AI should consider vetoing scope:un_resolution. Used by the
# un_vote.1 veto option's ai_chance: only veto a resolution that would pass
# even with our "no". Our vote isn't cast yet when this is evaluated, so that
# means a margin of at least 2 (support - oppose - 1 >= 1).
# Scope: voting country
un_veto_would_change_outcome = {
	scope:un_resolution ?= { un_res_margin >= 2 }
}

# Whether this resolution's "target" is who it HELPS rather than who it is aimed
# at: a request for aid or peacekeepers names the requester as its target.
# Anything that reasons "the target is my rival, so I like this" must exclude
# these, or a rival of the requester counts as eager to help it.
# Scope: resolution container
un_resolution_target_is_beneficiary = {
	OR = {
		has_tag = un_topic_aid_request
		has_tag = un_topic_peacekeeping_request
	}
}

# Whether the resolution is a CONVENTION: something each member ratifies for
# itself, so that it binds only the members that accept it
# (un_vote_apply_treaty_modifier gives each of them its modifier). A no-voter
# who refuses it in un_vote.3 simply stays outside it. The pledge topics
# (un_resolution_target_is_beneficiary) are the other kind a member can opt out
# of: refusing them means contributing nothing. Every other topic — a
# condemnation, sanctions, a mandate, an expulsion, a charter reform,
# decolonization, and an ICC statute vetoed down to a censure — is a decision
# of the whole Assembly that stands whatever one member says, so refusing it is
# a public denunciation and nothing more; un_vote.3 says so.
# Scope: resolution container
un_resolution_is_convention = {
	OR = {
		has_tag = un_topic_human_rights
		has_tag = un_topic_npt
		has_tag = un_topic_climate
		has_tag = un_topic_heritage
		has_tag = un_topic_pandemic
		has_tag = un_topic_refugee
		has_tag = un_topic_space
		has_tag = un_topic_law_of_sea
		has_tag = un_topic_physical_protection
		AND = {
			has_tag = un_topic_icc
			NOT = { has_tag = un_res_vetoed }
		}
	}
}

# The three kinds un_vote.3 tells apart for a member that voted against a
# resolution that carried, one trigger each so the event can name the kind as a
# parameter (un_vote_3_we_opposed_passed). Exactly one holds for any resolution.
# Scope: resolution container
un_resolution_kind_is_convention = {
	un_resolution_is_convention = yes
}
un_resolution_kind_is_pledge = {
	NOT = { un_resolution_is_convention = yes }
	un_resolution_target_is_beneficiary = yes
}
un_resolution_kind_is_decision = {
	NOT = { un_resolution_is_convention = yes }
	NOT = { un_resolution_target_is_beneficiary = yes }
}

# Whether ROOT voted against scope:un_resolution, the resolution carried, and
# it is of kind $KIND$ (convention, pledge or decision): the case in which
# un_vote.3 offers to accept it after all or to refuse, and says what refusing
# would mean.
# Parameters: $KIND$ = convention | pledge | decision
# Scope: the country (un_vote.3's ROOT); expects scope:un_resolution
un_vote_3_we_opposed_passed = {
	scope:un_resolution ?= {
		has_tag = un_res_passed
		un_resolution_has_vote = { LIST = un_res_no }
		un_resolution_kind_is_$KIND$ = yes
	}
}

# The resolution archive holds at least $COUNT$ resolutions: row $COUNT$ - 1 of
# the chamber's Voting Details has one to show (un_chamber_history_details_row_sgui).
# Scope: any
un_chamber_history_has_entry = {
	has_global_variable_list = un_resolution_history
	any_in_global_list = {
		variable = un_resolution_history
		count >= $COUNT$
		has_tag = un_resolution
	}
}
