# The AI conscript refresh's triggers (#1877 trap 5; design docs/superpowers/specs/2026-11-18-ai-conscript-refresh-design.md).
# The pass itself: common/scripted_effects/te_ai_conscript_refresh_effects.txt.

# Country scope: an AI country at peace, whose conscripts nobody needs this month. Not at war, in no diplomatic play
# (committed or undecided: a play's escalation raises conscripts), and no formation mobilized, so that no raised
# battalion is deleted and no country is left short as a war starts.
te_ai_conscript_refresh_may = {
	is_ai = yes
	is_at_war = no
	is_diplomatic_play_committed_participant = no
	is_diplomatic_play_undecided_participant = no
	NOT = { any_military_formation = { is_mobilized = yes } }
}

# State scope: the state's conscription center holds at least one stale battalion.
te_ai_conscript_center_stale = {
	has_building = building_conscription_center
	b:building_conscription_center = {
		any_combat_unit = { te_ai_conscript_unit_stale = yes }
	}
}
