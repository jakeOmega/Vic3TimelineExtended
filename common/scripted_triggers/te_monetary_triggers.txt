# ============================================================================
# THE GAME RULE: full monetary policy, or the simplified banking system?
# ============================================================================
# `banking_system_rule` has three settings (common/game_rules/extra_game_rules.txt).
# This trigger answers the only question the monetary layer ever asks of it:
# is the whole of phases 1-5 live?
#
#   banking_system_enabled     yes — the dial, inflation, FX, arrangements
#   banking_system_simplified  no  — the Banking Cycle journal entry without
#                                    the monetary layer on top of it
#   banking_system_disabled    no  — no journal entry either
#
# WHAT STILL RUNS WHEN THIS IS `no`. The passive half of the rate stack, and
# only that: step 0 (init), 0b (modifier home), 1 (regime code), 4 (the derived
# rate — world reference rate + the bankless spread), 5 (the risk premium), 7
# (the rate paid) and 9 (writing it into the engine). It has to: vanilla's flat
# 20% base, the rank multipliers, laissez-faire and the five finance techs are
# cancelled by unconditional INJECTs that no game rule can reach (§16.1), so
# with nothing in their place every country would borrow free. Every OTHER step
# is skipped, and the variables those steps own keep the neutral values step 0
# initialises them to — te_inflation 0, te_fx_index 100, te_mon_anchor_kind 0,
# te_mon_stance_gap 0 — so a consumer that reads one gets a sane answer instead
# of a missing-variable error. That is the contract: skipped steps do not write,
# and everything downstream reads the initialised value.
#
# READS ONLY A GAME RULE, so it works from any scope — country, journal entry,
# treaty article, power-bloc principle, GUI. Same idiom as
# te_aptitude_traits_enabled (common/scripted_triggers/misc_triggers.txt).
#
# A GAME RULE CANNOT CHANGE MID-CAMPAIGN, so nothing ever has to strip a
# modifier this gate turned off: under the simplified setting the modifiers of
# the advanced steps are never applied in the first place.
te_mon_full_system = {
	has_game_rule = banking_system_enabled
}


# Does this country set its own policy rate?
#
# A dial requires a national bank AND a monetary regime that leaves a rate to
# set. Without one, te_policy_rate is derived rather than chosen (§4): bankless
# and crypto countries get world rate + the bankless spread, and a command
# economy gets the administered rate.
#
# COMMODITY MONEY IS IN, BUT ONLY WITH A BANK, AND ONLY NARROWLY (owner
# decision, 2026-09-21, closing §5.4's open question). The Banque de France did
# set a discount rate on bimetallic money in 1836, so a commodity-money country
# that has built a national bank holds a dial — one restricted to a band around
# the world rate (one under, three over), which is te_mon_has_narrow_dial below. Commodity money
# WITHOUT a bank still has no dial at all, and this trigger's national-bank line
# is the whole of that rule.
#
# lawgroup_national_bank holds exactly two laws (law_no_national_bank,
# law_national_bank), so the positive test is the whole of it. Central bank
# independence and state-owned banking live in lawgroup_financial_regulation, a
# separate group — a CBI country still has law_national_bank and still has a
# dial; it just does not steer it by hand (§6).
#
# has_law_or_variant rather than has_law throughout: variant laws
# (`parent = law_X`) are tag-restricted skins of the same law and must count.
#
# DOLLARISATION (phase 2, ruling P9). A country that abandoned its own currency
# in a hyperinflation has no dial left whatever its laws still say: the rate is
# whoever's money it now uses. The flag is set by the dollarise option of
# te_inflation.1 (events/te_inflation_events.txt) and cleared by
# te_banking_law_change_cleanup, on either of its two exits (a different
# lawgroup_monetary_policy law, or entering a command economy).
# te_mon_is_dollarised below owns the guarded read, so a country that has never
# pulsed — and every save from before phase 2 — reads exactly as it did before.
te_mon_has_dial = {
	# THE GAME RULE FIRST (banking_system_simplified): with the monetary layer
	# off, nobody holds a dial, and every consumer of this trigger — the update's
	# step 2/4 branch, the dashboard block, the OMO gate, the stance row, the
	# union principles' ai_weight — falls to its no-dial side from this one line.
	te_mon_full_system = yes
	has_law_or_variant = law_type:law_national_bank
	OR = {
		has_law_or_variant = law_type:law_commodity_money
		has_law_or_variant = law_type:law_gold_standard
		has_law_or_variant = law_type:law_fiat_currency
		has_law_or_variant = law_type:law_digital_currency
	}
	NOT = { has_law_or_variant = law_type:law_command_economy }
	NOT = { te_mon_is_dollarised = yes }
	# [P5] ANCHORED (§15A.1): a country that has tied its money to another's —
	# treaty peg, bloc currency, currency board — imports that country's rate.
	# te_mon_is_anchored reads STORED state only (it must: the anchor's validity
	# is "the anchor has a dial", which is this trigger again), and step 1c of
	# the monthly update keeps the stored pair checked.
	NOT = { te_mon_is_anchored = yes }
}

# Does this country have a STANCE — a real rate that can be the wrong one? (§8,
# §15A.1) Every dial country, and every anchored one: an anchored country "keeps
# its own inflation, neutral rate, stance gap and cycle. This is the whole
# point" — the imported rate is wrong whenever the two cycles diverge, and the
# gap is the measure of it. So the three sites in the monthly update that used
# to gate hidden state on te_mon_has_dial — the inflation noise walk, the two
# neutral-rate walks, the gap itself — gate on this instead (§0.8 ruling G1).
# The OTHER no-dial countries are unchanged: their derived rate is world +
# expected + spread, a constant real gap that means nothing, and it stays 0.
te_mon_has_stance = {
	OR = {
		te_mon_has_dial = yes
		te_mon_is_anchored = yes
	}
}

# Is this country's dial the NARROW one? (§5.1, §0.6) Commodity money buys a
# discount rate, not a monetary policy: convertibility into metal at a fixed
# weight lets the bank make credit one point cheaper than the world and up to
# three points dearer, and nothing more. The band itself lives in
# te_mon_target_min / te_mon_target_max, which is the single place it is stated;
# this trigger only answers whose band it is.
#
# THE BAND IS NOT THE REGIME'S WHOLE DISCIPLINE ANY MORE (ruling R7, issue
# #351). §0.6 R5 said it was, and a band on the RATE turned out to say nothing
# about the QUANTITY of money: te_mon_pressure_commodity_specie is the
# price-specie channel that answers for the price level, and it is gated on the
# LAW, not on this trigger — a commodity country without a bank has no dial to
# narrow and is disciplined all the same.
#
# FIVE SETTINGS, NOT TWENTY-SIX, AND LOPSIDED. Both bounds hang off one
# whole-number centre (var:te_mon_commodity_centre), so the dial always offers
# exactly centre-1 to centre+3. Cheap money is what metal forbids; dear money
# was always available, and was the period's answer to a panic.
#
# READ BY: the two range branches, the stepper's floor/ceiling tooltips
# (common/scripted_guis/te_monetary_sguis.txt), te_mon_dial_range_note, and
# te_mon_rate_is_discretionary, which excludes these countries from the
# world-rate average for the same reason it excludes a peg defender.
te_mon_has_narrow_dial = {
	te_mon_has_dial = yes
	has_law_or_variant = law_type:law_commodity_money
}

# Has the world rate moved far enough from the commodity band's centre that
# step 0 should re-round it? Hysteresis, so a world rate hovering at x.5 does
# not walk the band (and with it a target sitting on the floor) up and down.
te_mon_commodity_centre_is_stale = {
	has_variable = te_mon_commodity_centre
	has_variable = te_mon_world_rate_now
	OR = {
		te_mon_commodity_centre_offset >= te_mon_commodity_centre_hysteresis
		te_mon_commodity_centre_offset <= te_mon_commodity_centre_hysteresis_neg
	}
}

# Central bank independence (lawgroup_financial_regulation). Binds the target to
# the mandate formula, halves the structural premium floor to 0.25pp and narrows
# the neutral-rate estimation error to +/-0.5 (§6, §7, §21).
te_mon_is_cbi = {
	has_law_or_variant = law_type:law_central_bank_independence
}

# Command economy (vanilla lawgroup_economic_system). The rate is administered,
# there is no dial, and the stance channel is switched off (§4, §8).
te_mon_is_command = {
	has_law_or_variant = law_type:law_command_economy
}

# Has this country given up its own currency? (ruling P9) Set by the dollarise
# option of te_inflation.1 and cleared by te_banking_law_change_cleanup, either
# when the country next enacts a monetary-policy law (which pays the
# new-currency price) or when it enters a command economy (which does not).
#
# Guarded, so a country that has never pulsed — and every save from before phase
# 2 — reads no. THE ONE PLACE THE QUESTION IS ASKED: the dashboard's four
# cause-specific custom-loc blocks (te_mon_no_dial_reason,
# te_mon_inflation_band_name, _cost and te_mon_monetise_condition in
# common/customizable_localization/banking_dash_custom_loc.txt), every one of
# which has to tell a dollarised country from a bankless one, and — since the
# final review — te_mon_has_dial above and te_mon_can_monetise below, which used
# to carry their own copy of the guarded read. `NOT = { te_mon_is_dollarised =
# yes }` is exactly equivalent to that copy: NOT(has_variable AND var = 1) is the
# same statement as "if the variable is there, it is not 1".
te_mon_is_dollarised = {
	has_variable = te_mon_dollarised
	var:te_mon_dollarised = 1
}

# State-owned banking (lawgroup_financial_regulation). The weakest credibility
# anchor (§5.2, §9.1). Ruling R2 ships the premium part only in phase 1.
#
# It lost its phase-1 caller when the +0.5pp structural premium moved onto the
# law itself as country_credit_standing_add = 0.005, where the law tooltip
# states it. Phase 2 gave it one again: te_mon_credibility_c reads it for the
# §9.1 expectations anchor, where state-owned banking is the weakest c there is
# (0.15) and outranks delegation — a bank the treasury owns cannot promise
# anything the treasury might want undone.
te_mon_is_state_banking = {
	has_law_or_variant = law_type:law_state_owned_banking
}


# --- Who has inflation, and on what terms (§9, §11, §14) --------------------

# Does this country run the inflation loop at all? Everyone except a command
# economy (ruling P6, §14: repressed inflation is a system nobody has designed
# yet, so a planned economy holds pi and expected at 0 and is excluded from the
# bands, the premium terms and the cost-push basket).
#
# NOTE WHO IS *IN*. Bankless and crypto countries all run inflation, even
# though neither has a dial: §4 makes them pay the world rate plus THEIR OWN
# expected inflation plus the bankless spread, so their expectations have to be
# computed. Their stance gap stays 0, which simply drops the loose-money term
# out of §9.1's pressure for them. (Commodity money with a bank now holds the
# narrow dial and so does have a stance — a band one point under the world
# rate and three over is a small channel, but it is not a closed one.)
te_mon_has_inflation = {
	te_mon_is_command = no
}

# Metallic money — gold standard or commodity money (§5.1). Convertibility is
# the anchor: expectations are pinned (credibility c = 1), the inflation target
# is 0 rather than 2, and §9.1's regime pull drags core inflation back toward
# zero, which is the deflation bias of a hard-money regime.
#
# CRYPTO IS NOT METALLIC. law_decentralized_cryptocurrency gets a stronger pull
# (-(pi_core + 1)) and none of the anchoring, because nothing redeems it; that
# branch belongs to step 6, not here.
#
# A SUSPENDED GOLD STANDARD IS NOT METALLIC (phase 3, §12.3). Convertibility is
# the whole of the anchor, so a country that has suspended it keeps the law and
# loses the pin: it "acts as fiat" — anchor 2, ordinary credibility c, no regime
# pull — until the suspension runs out. te_mon_is_on_gold owns that test.
te_mon_is_metallic = {
	OR = {
		te_mon_is_on_gold = yes
		has_law_or_variant = law_type:law_commodity_money
	}
}

# May this country monetise its deficit? (§11) Printing money to pay the
# government needs a currency the state can create (fiat or digital), a bank to
# create it through, and a government still able to give that bank the order —
# so central bank independence rules it out, and so does a command economy,
# which has its own `ce_*` plan finance. A dollarised country has no currency of
# its own left to print (ruling P9); te_mon_is_dollarised owns that guarded read,
# so a country that has never pulsed reads as it did before phase 2. It also
# means step 1b of the monthly update zeroes te_monetisation_level by itself the
# month after a country dollarises — the dollarise option does not have to.
#
# The regime test is the same pair law_fiat_currency / law_digital_currency that
# country_can_create_unbacked_money_bool gates open-market operations with. It
# is written out rather than read from the bool because monetisation is a
# TREASURY act and OMO is a CENTRAL BANK act: a later law could plausibly permit
# one and forbid the other, and sharing the gate would hide that.
te_mon_can_monetise = {
	has_law_or_variant = law_type:law_national_bank
	OR = {
		has_law_or_variant = law_type:law_fiat_currency
		has_law_or_variant = law_type:law_digital_currency
	}
	te_mon_is_cbi = no
	te_mon_is_command = no
	NOT = { te_mon_is_dollarised = yes }
	# [P5] Nor may an anchored country (§15A.1): it is not their currency to
	# print. Step 1b zeroes the level by itself the month after, as it does for
	# a dollarised one.
	NOT = { te_mon_is_anchored = yes }
}


# --- Policy stance ----------------------------------------------------------
#
# te_mon_stance_gap is HIDDEN STATE (§8): these two triggers exist so AI weights
# and gating can react to the stance without any script ever printing the gap.
# They must never be used to build a player-visible number.
#
# Both are guarded on has_variable, because the gap does not exist until a
# country's first monthly monetary update — and because the JE pulse and the
# country update may run in either order, the value read may be one month old.
# A country with no dial has a gap of exactly 0, so it is neither tight nor
# loose, which is the intended answer.

# Real rate at least 1pp above the (hidden) neutral rate.
banking_stance_is_tight = {
	has_variable = te_mon_stance_gap
	var:te_mon_stance_gap >= 1
}

# Real rate at least 1pp below the (hidden) neutral rate.
# KEEP: no caller in phase 1 (only _is_tight is weighted today). It is the other
# half of this pair's contract surface for phases 2-3, where loose money feeds
# inflation and the easing-side AI weights; a cleanup pass should leave it.
banking_stance_is_loose = {
	has_variable = te_mon_stance_gap
	var:te_mon_stance_gap <= -1
}


# --- The dial's controls (§3, §5.1, §6) -------------------------------------
#
# Everything the dashboard's monetary block greys a control on. Each of these is
# asked from the scripted GUI's `is_valid`, wrapped in a `custom_tooltip` there
# so IsValidTooltip renders it with a tick or a cross; the triggers themselves
# stay plain conditions so they can also be read by script.
#
# All of them are guarded: a country that has never run a monetary pulse has
# none of these variables, and an unguarded `var:` read resolves to 'none'.

# Has the player handed the dial to the central bank? (§6) The toggle is free
# and reversible for any dial country that is not independent; central bank
# independence is permanently mandate-bound and does not use this variable,
# which is why te_mon_mandate_binds below asks for either.
te_mon_is_delegated = {
	has_variable = te_mon_delegated
	var:te_mon_delegated = 1
}

# Does the mandate actually drive the target? Only then is picking one a
# decision rather than a preference stored for later.
te_mon_mandate_binds = {
	OR = {
		te_mon_is_delegated = yes
		te_mon_is_cbi = yes
	}
}

# Is there room left to step the target down / up inside the regime's range?
# The bounds are te_mon_target_min / te_mon_target_max
# (common/script_values/te_monetary_script_values.txt), the same two values
# te_monetary_clamp_target re-clamps with every month — so the stepper can never
# offer a step the next pulse would undo.
te_mon_target_can_step_down = {
	has_variable = te_policy_rate_target
	var:te_policy_rate_target > te_mon_target_min
}

te_mon_target_can_step_up = {
	has_variable = te_policy_rate_target
	var:te_policy_rate_target < te_mon_target_max
}

# The same pair for the monetisation stepper (§11). The ceiling is
# te_mon_monetisation_max, which both step effects and step 1b of the monthly
# update clamp against, so the stepper can never offer a step the next pulse
# would undo — exactly the arrangement the rate target has above.
#
# WHETHER THE COUNTRY MAY MONETISE AT ALL is te_mon_can_monetise, a separate
# question with its own one-line answer (te_mon_monetise_condition). These two
# only say whether there is room left in the range.
te_mon_monetise_can_step_down = {
	has_variable = te_monetisation_level
	var:te_monetisation_level > 0
}

te_mon_monetise_can_step_up = {
	has_variable = te_monetisation_level
	var:te_monetisation_level < te_mon_monetisation_max
}

# Has this country's first monthly monetary update run? Everything the dashboard
# reads is written by that pulse, and a `.gui` read of a missing variable logs
# "Failed to fetch variable" every frame (gui_modding_guide.md gotcha 14/23), so
# the whole monetary block hides behind this one question. The list is the
# displayed contract of §16.2, not every variable the update owns —
# te_mon_era_base_now is on it because the World Reference Rate row reads it,
# and because it is the first thing step 0 writes, so a country that has it has
# run a pulse.
#
# THE LAST FOUR ARE PHASE 2. The Rate You Pay tooltip breaks the figure down as
# market yield + premium - inflation (§10), the block's own Inflation and
# Expected Inflation rows print two of those, and the monetisation stepper
# prints its level. A save made under phase 1 has every variable above this line
# and none of these until its first pulse on the new version, which is exactly
# the case this gate exists for.
#
# te_inflation_band and te_mon_dollarised are deliberately NOT here: every
# dashboard read of either goes through a customizable_localization block that
# carries its own has_variable branch, so they need no second gate.
te_mon_display_ready = {
	has_variable = te_mon_era_base_now
	has_variable = te_policy_rate
	has_variable = te_policy_rate_target
	has_variable = te_premium_structural
	has_variable = te_premium_cyclical
	has_variable = te_rate_paid_pts
	has_variable = te_mon_stance_band
	has_variable = te_market_yield
	has_variable = te_inflation
	has_variable = te_inflation_expected
	has_variable = te_monetisation_level
}

# Is the policy rate at the floor of its regime's range? This is the gate on
# open market operations (§11, and monetary_policy_design.md's "the floor tool"):
# a bank that can still cut does not buy assets, because cutting is cheaper and
# reversible and does the same job.
#
# ONLY FIAT AND DIGITAL EVER ASK. OMO is fiat/digital only (§0.1 ruling of
# 2026-09-20 — a convertible currency cannot create money to buy bonds), and
# banking_possible_cb_open_market_ops checks the regime before it reaches this
# trigger. So the floor here is 0 under fiat and -3 under digital, which has no
# cash to hoard and so three more points of cutting room; no gold-standard band
# floor is in the picture at all.
#
# The bound comes from te_mon_floor_threshold, which reads te_mon_target_min —
# so the law check that used to sit here is gone, and
# country_policy_rate_floor_add on law_digital_currency is the single statement
# of the rule. The threshold carries one point of slack (0.01) because
# te_policy_rate is a drifting fractional value that need not land exactly on
# the bound.
te_mon_policy_rate_at_floor = {
	has_variable = te_policy_rate
	var:te_policy_rate <= te_mon_floor_threshold
}


# --- Phase 3: the world rate, gold flows and the peg (§12) ------------------
#
# ONE TEST FOR "ON GOLD". The gold standard is a law; being on gold is the law
# AND a convertible currency. §12.3's suspension keeps the law and drops the
# convertibility for five years, on the dollarisation precedent (ruling P9: a
# variable overriding what the law-derived regime says). Every monetary test
# that means "the peg binds" — the metallic anchor, peg defence, gold flows, the
# 0-15 range — goes through te_mon_is_on_gold; a raw has_law_or_variant on
# law_gold_standard is right only where the LAW is meant (the regime code, the
# crisis's own gate, the dashboard's suspended read-out).

# Has this country suspended convertibility? Guarded, so a country that has
# never pulsed and every save from before phase 3 read "no".
te_mon_peg_is_suspended = {
	has_variable = te_peg_suspended_months
	var:te_peg_suspended_months > 0
}

te_mon_is_on_gold = {
	has_law_or_variant = law_type:law_gold_standard
	te_mon_peg_is_suspended = no
}

# Do gold flows run for this country? (§12.2: "gold-standard countries with a
# national bank only".) A gold country without a bank pays the world rate, so
# its gap and its drain are zero by construction; te_mon_has_dial carries the
# bank, the command-economy and the dollarisation tests.
te_mon_has_gold_flows = {
	te_mon_is_on_gold = yes
	te_mon_has_dial = yes
}

# §12.3's switch from moving gold to moving confidence: the BANK'S vault under
# a tenth of its limit. (The first version also counted any treasury debt,
# because the vault was the treasury; under the hybrid model the government's
# books have nothing to do with whether the bank can pay out gold.)
te_mon_peg_under_pressure = {
	te_mon_bank_gold_scaled < te_mon_peg_reserve_floor
}

# May the treasury put gold into the bank right now? Gold flows, a counted
# vault with room in it, and the step's worth of cash actually in hand — a
# recapitalisation on credit would be the government borrowing to lend to
# itself. One gate for the dashboard button and the AI's rule in step 10.
te_mon_bank_can_recapitalise = {
	te_mon_has_gold_flows = yes
	has_variable = te_bank_gold_seeded
	var:te_bank_gold_seeded = 1
	te_mon_bank_recap_amount > 0
	gold_reserves >= te_mon_bank_recap_amount
}

# Is this country's Defend option (§12.3) still running? While it is, the
# target's floor is the world rate + 4 (te_mon_target_min).
te_mon_peg_is_defending = {
	has_variable = te_peg_defend_months
	var:te_peg_defend_months > 0
}

# Is the peg-defence mandate what actually sets this country's target? The
# mandate variable alone is not enough — a player on a manual target may have
# peg defence stored as a preference for later. Mirrors the mandate branch of
# te_monetary_update_target: AI is always delegated, and a bank without the
# journal entry is auto-delegated (to price stability, so never to this).
te_mon_is_on_peg_defence = {
	te_mon_is_on_gold = yes
	has_variable = te_mon_mandate
	var:te_mon_mandate = 3
	OR = {
		is_player = no
		te_mon_mandate_binds = yes
	}
}

# Is this country's policy rate DISCRETIONARY — chosen, rather than derived
# from the world rate? (§12.1) Only these great powers enter the world-rate
# average: a no-dial rate is world + spread and a peg defender's target IS the
# world rate, so averaging either would define the world rate in terms of
# itself and ratchet it up by the spread every month.
#
# A NARROW DIAL IS NOT DISCRETIONARY EITHER, for exactly that reason: its band
# is world-1 to world+3, so a commodity-money great power in the average would
# be helping to set the number its own band is measured from. It gets a rate it
# chooses inside the band and no vote on where the band sits.
te_mon_rate_is_discretionary = {
	te_mon_has_dial = yes
	te_mon_has_narrow_dial = no
	te_mon_is_on_peg_defence = no
	has_variable = te_policy_rate
	has_variable = te_inflation_expected
}


# --- Where the managed modifiers live ---------------------------------------
# Is $NAME$ on this country's monetary modifier HOME — the banking journal entry
# if te_mon_mod_home says so, the country otherwise? The read-side twin of
# te_mon_mod_add / _remove (common/scripted_effects/te_monetary_effects.txt,
# "WHERE THE MODIFIERS LIVE"). te_mon_mod_home is written by step 0b of every
# pulse before anything asks; the has_journal_entry guard is for a caller
# outside the update, where the entry may have gone since.
te_mon_mod_has = {
	trigger_if = {
		limit = {
			has_variable = te_mon_mod_home
			var:te_mon_mod_home = 1
			has_journal_entry = je_banking_cycle
		}
		je:je_banking_cycle = { has_modifier = $NAME$ }
	}
	trigger_else = {
		has_modifier = $NAME$
	}
}


# ============================================================================
# Phase 4 — exchange rates and capital controls (§15)
# ============================================================================

# Is this country in the kind of trouble that excuses capital controls? (§15.4)
# ONE shared definition, read by the fatigue counter (frozen in a crisis) and by
# the two capital-controls buttons' ai_chance, so "when the AI reaches for
# controls" and "when controls are forgiven" cannot drift apart.
#
# Peg confidence is tested at <= 40, deliberately ABOVE the crisis event's own
# <= 20: emergency controls are forgiven before the crisis breaks, which is
# when they are any use.
te_mon_in_external_crisis = {
	OR = {
		is_at_war = yes
		te_mon_in_financial_crisis = yes
	}
}

# [6a] The same trouble MINUS the war (§15C.1, ruling H2). One consumer reads
# this narrower name: the swap line's draw. Capital-controls forgiveness and the
# AI's controls rule keep the wider definition — a war excuses shutting the
# door, but a swap line is a central bank's liquidity instrument, not war
# finance, and `is_at_war` is the one leg a player can switch on and sustain at
# will (§15B.3's multi-provider farm). Precedent for one consumer re-reading a
# narrower trigger: §0.8 ruling G15's te_mon_capital_controls_in_force.
te_mon_in_financial_crisis = {
	OR = {
		AND = {
			has_variable = finance_cycle_value
			OR = {
				banking_cycle_is_panic = yes
				banking_cycle_is_downturn = yes
			}
		}
		AND = {
			te_mon_has_gold_flows = yes
			has_variable = te_peg_confidence
			var:te_peg_confidence <= 40
		}
		AND = {
			has_variable = te_inflation_band
			var:te_inflation_band >= 6
		}
	}
}

# Does this country's index follow its shadow? (§15.2) Everything that is
# neither metal (par: the NOMINAL parity is fixed, so there is no trade edge to
# give) nor a command economy (inconvertible; excluded like §14's others).
# te_mon_is_metallic is already suspension-aware through te_mon_is_on_gold
# (ruling Q4), so "suspend convertibility -> floats" needs no branch here.
te_mon_fx_floats = {
	te_mon_is_command = no
	te_mon_is_metallic = no
}

# Is a metallic index still walking back to par after §12.3's Devalue?
te_mon_fx_is_devalued = {
	has_variable = te_fx_devalued_months
	var:te_fx_devalued_months > 0
}

# Is overvaluation past the tolerance at which it drains a gold peg? Also read
# by step 10's heal, so that the +1 a healthy peg earns holds instead of netting
# the drain away (ruling Q10's "under pressure", extended).
#
# Judged on te_mon_fx_overvaluation_for_peg, NOT the raw variable: the share of
# overvaluation a low rate caused is excluded, because gold flows already price
# it (§0.7 ruling F4).
te_mon_fx_is_overvalued = {
	has_variable = te_mon_overvaluation
	te_mon_fx_overvaluation_for_peg > te_mon_fx_overvaluation_tolerance
}
