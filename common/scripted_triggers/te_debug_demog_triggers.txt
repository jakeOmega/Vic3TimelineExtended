# The demographics self-checks' triggers (phase 2 step 1; te_debug_demog_wf_selfchecks in
# common/scripted_effects/te_debug_demog_effects.txt, te_debug_demog.6 and .7 in events/te_debug_demog_events.txt).

# THIS = a country: its criminal-justice law holds Community Service or Mandatory Minimums, the two amendments whose
# working-adult lines phase 2 step 1 moved to script-only types, each with a men's participation input beside it
# (common/amendments/extra_amendments.txt). One law holds at most one of them: their allowed laws don't overlap.
te_debug_demog_wf_amendment_held = {
	active_law:lawgroup_criminal_justice ?= {
		OR = {
			has_amendment = amendment_type:amendment_community_service
			has_amendment = amendment_type:amendment_mandatory_minimums
		}
	}
}

# THIS = a country: its Trade Unions' loyal trait (Solidarity, or the Bourse du Travail in France) is on at its plain
# size: approval at loyal, clout neither marginal (the defines' MARGINAL_IG_TRAIT_PENALTY halves a trait) nor powerful
# (POWERFUL_IG_TRAIT_BONUS doubles it), and no Ministry of Labour law carries the same participation input.
te_debug_demog_wf_ig_trait_plain = {
	ig:ig_trade_unions ?= {
		ig_approval >= define:NPolitics|APPROVAL_THRESHOLD_LOYAL
		ig_clout > define:NPolitics|MARGINAL_IG_THRESHOLD
		ig_clout < define:NPolitics|POWERFUL_IG_CUTOFF
	}
	NOT = { has_law = law_type:law_pro_labor_ministry_of_labor }
	NOT = { has_law = law_type:law_pro_capital_ministry_of_labor }
}

# Calibration step 3's self-checks (te_debug_demog_step3_selfchecks, te_debug_demog.40 and .41; owner, 2026-10-12).
# THIS = a pop working a subsistence farm whose farm reads the famine reserves line (the walk's read below 0): no company
# owns a subsistence farm, so the read holds no executive's trait, and the trait is the key's only other carrier.
te_debug_demog_fr_farm_pop = {
	is_employed = yes
	exists = workplace
	workplace = { is_building_group = bg_subsistence_agriculture }
	te_demog_pop_country_line_walk_read < -0.001
}

# THIS = a country whose Landowners are powerful (the engine's own test, with its cutoff; their traits count double:
# POWERFUL_IG_TRAIT_BONUS) and whose capital has a subsistence-farm pop reading the line at its farm: the side the saves
# measured. The census's own read (the owner's) is left out of the arming, so a country read of 0 logs BROKEN instead of
# keeping the check silent.
te_debug_demog_fr_armed = {
	ig:ig_landowners ?= { is_powerful = yes }
	capital ?= {
		any_scope_pop = { te_debug_demog_fr_farm_pop = yes }
	}
}

# THIS = a building: a mine of bg_mining itself, the group whose key the check reads (not the mod's
# bg_discoverable_mining under it), where working conditions apply at half their size or more
# (te_debug_demog_wc_mine_scale: Workplace Safety below three), so a table switched off can't read as OK.
te_debug_demog_wc_mine = {
	is_building_group = bg_mining
	te_debug_demog_wc_mine_scale >= 0.5
}

# THIS = a state in a state region the map files give the mild malaria trait (te_demog_malaria_level_listed 1), the trait
# quinine's research switches off (disabling_technologies).
te_debug_demog_mal_listed_mild = {
	te_demog_malaria_level_listed > 0
	te_demog_malaria_level_listed < 2
}
