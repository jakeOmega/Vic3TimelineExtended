# Single-source of the three conditions the nuclear-programme surfaces ask.
# Each body is the verbatim text that used to sit inline on the journal entry
# or on one of its two scripted buttons; the button and the widget's scripted
# GUI both call the helper, so the two surfaces cannot drift apart.
# See docs/systems/journal_entry_systems.md, "Nuclear Programme Widget".

# Country scope: can this country sustain a nuclear weapons programme at all?
# The body of je_nuclear_program's `possible`. Also read by the programme's
# status line (common/customizable_localization/nuclear_program_custom_loc.txt),
# which has to explain a *deactivated* entry — the entry drops back to inactive
# the moment this goes false, because it carries can_deactivate = yes.
nuclear_program_can_run_programme = {
	OR = {
		country_rank >= rank_value:great_power
		AND = {
			country_rank = rank_value:major_power
			has_technology_researched = ICBMs
		}
		modifier:country_receiving_nuclear_program_aid_bool = yes
	}
	NOT = {
		modifier:country_nuclear_disarmament_bool = yes
	}
}

# Country scope: may funding be stepped up? Body of
# increase_funding_nuclear_program's `possible`. Deliberately carries no
# custom_tooltip: the engine renders these three clauses with a tick or a cross
# each, which is both more informative than a hand-written line and impossible
# to let drift away from the real condition.
nuclear_program_possible_increase_funding = {
	country_innovation > nuclear_weapon_program_funding_step
	NOT = {
		modifier:country_nuclear_disarmament_bool = yes
	}
	NOT = {
		modifier:country_nuclear_program_pause_bool = yes
	}
}

# Country scope: may funding be stepped down? Body of
# decrease_funding_nuclear_program's `possible`. Reads the variable unguarded,
# exactly as the button did: every caller is gated on the entry being active,
# which is the only state in which the variable exists (the entry's `immediate`
# creates it).
nuclear_program_possible_decrease_funding = {
	var:nuclear_weapons_program_funding > 0
}

# Country scope: is this country's nuclear weapons programme funded and
# running? A nuclear power building its stockpile qualifies too. The gate on
# covert_nuclear_sabotage_action (covert slice 6), whose modifier slows exactly
# this progress. Same test the programme's own display values use
# (nuclear_program_display_months_to_next, extra_script_values.txt).
nuclear_program_is_proliferating = {
	has_variable = nuclear_weapon_program_progress
	has_variable = nuclear_weapons_program_funding
	var:nuclear_weapons_program_funding >= 1
	NOT = { modifier:country_nuclear_program_pause_bool = yes }
	NOT = { modifier:country_nuclear_disarmament_bool = yes }
}

# Country scope: True if scope:country has existential war goals against ROOT
# (any subjugation war goal, or outright annexation incl. the mod's te_reunify_country)
# Requires scope:country to be set (the enemy)
enemy_has_existential_war_goal = {
	scope:country = {
		OR = {
			war_side_has_war_goal_of_type_against = { type = make_protectorate target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_dominion target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_tributary target = ROOT }
			# 1.14 subjugation goals
			war_side_has_war_goal_of_type_against = { type = make_personal_union target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_crown_land target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_chartered_company target = ROOT }
			war_side_has_war_goal_of_type_against = { type = annex_country target = ROOT }
			war_side_has_war_goal_of_type_against = { type = te_reunify_country target = ROOT }
		}
	}
}

# Country scope: True if scope:country wants to annex or subjugate ROOT
# (excludes tributary and personal union - only the most threatening goals)
enemy_has_subjugation_war_goal = {
	scope:country = {
		OR = {
			war_side_has_war_goal_of_type_against = { type = annex_country target = ROOT }
			war_side_has_war_goal_of_type_against = { type = te_reunify_country target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_protectorate target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_dominion target = ROOT }
			# 1.14 subjugation goals
			war_side_has_war_goal_of_type_against = { type = make_crown_land target = ROOT }
			war_side_has_war_goal_of_type_against = { type = make_chartered_company target = ROOT }
		}
	}
}

# Country scope: True if the country's ruler has aggressive/reckless personality traits
ruler_is_aggressive = {
	exists = ruler
	ruler = {
		OR = {
			has_trait = reckless
			has_trait = wrathful
			has_trait = war_criminal
		}
	}
}

# Country scope: True if the country's ruler has cautious/honorable personality traits
ruler_is_cautious = {
	exists = ruler
	ruler = {
		OR = {
			has_trait = cautious
			has_trait = honorable
		}
	}
}

# Country scope: True if this specific enemy (scope:country) previously nuked us
was_nuked_by_enemy = {
	exists = scope:country
	has_variable = nuked_by_country
	var:nuked_by_country = scope:country
}

# Country scope, ENEMY = the enemy country (scope:country in nuke.txt,
# scope:target_country in covert_operations.txt). True if ROOT is losing the
# war it shares with $ENEMY$ on the battlefield: most size-weighted battles in
# that war lost (gated on enough significant battles -- with none fought the
# fraction reads 0, i.e. "losing"), or a quarter of ROOT's own territory under
# enemy occupation. Replaces the 1.14 war-support proxy, which also fires for
# a merely unpopular war.
# NOTE: enemy_side_occupation is how much of the ENEMY's land we hold (high =
# winning); enemy_occupation is our own occupied share.
is_losing_war_against = {
	OR = {
		enemy_occupation >= 0.25
		any_scope_war = {
			is_war_participant = $ENEMY$
			num_significant_battles >= 5
			ROOT = {
				size_weighted_won_battles_fraction = { target = PREV value < 0.35 }
			}
		}
	}
}
