# ============================================================================
# GRAND MONUMENTS — opportunities: scripted triggers
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-12-grand-monument-opportunities-design.md
# (v2's commissions, reworked: 2026-10-05-grand-monuments-v2-design.md §2).
# Players read "opportunity"; the script keeps v2's gm_com_* names. The record
# is country variables (gm_commission_effects.txt lists them); the dedication
# and the asking interest group are flags, never scopes, so a civil war's
# winner can read them.

# ---- What a country may raise -------------------------------------------------
# Country scope. The dedication's gate, the same test the ceremony's option
# makes (monument_events.2): an opportunity is only offered, and only stays
# open, for a dedication the country could dedicate today.
# test_grand_monument_registry.py pins each body to its ceremony option.
gm_country_can_raise = {
	gm_country_can_raise_$KEY$ = yes
}

gm_country_can_raise_crown = {
	monument_government_is_crowned = yes
}

gm_country_can_raise_republic = {
	monument_government_is_republican = yes
}

gm_country_can_raise_revolution = {
	gm_government_is_revolutionary = yes
}

gm_country_can_raise_leader = {
	gm_can_raise_leader_monument = yes
}

gm_country_can_raise_religious = {
	NOT = { has_law_or_variant = law_type:law_state_atheism }
}

gm_country_can_raise_civic = {
	always = yes
}

gm_country_can_raise_war_memorial = {
	always = yes
}

gm_country_can_raise_artistic = {
	has_technology_researched = romanticism
}

gm_country_can_raise_naturalist = {
	has_technology_researched = romanticism
}

gm_country_can_raise_scientific = {
	has_technology_researched = empiricism
}

gm_country_can_raise_industrial = {
	has_technology_researched = marketing_research
	NOT = { has_law_or_variant = law_type:law_industry_banned }
}

gm_country_can_raise_athletic = {
	has_technology_researched = television_broadcasting
}

# Country scope. The open opportunity's dedication can still be raised; when
# it can't (a crown falls), the opportunity lapses at no cost.
gm_country_com_gate_holds = {
	has_variable = gm_com_dedication
	OR = {
		AND = { var:gm_com_dedication = flag:crown gm_country_can_raise_crown = yes }
		AND = { var:gm_com_dedication = flag:republic gm_country_can_raise_republic = yes }
		AND = { var:gm_com_dedication = flag:revolution gm_country_can_raise_revolution = yes }
		AND = { var:gm_com_dedication = flag:leader gm_country_can_raise_leader = yes }
		AND = { var:gm_com_dedication = flag:religious gm_country_can_raise_religious = yes }
		AND = { var:gm_com_dedication = flag:civic gm_country_can_raise_civic = yes }
		AND = { var:gm_com_dedication = flag:war_memorial gm_country_can_raise_war_memorial = yes }
		AND = { var:gm_com_dedication = flag:artistic gm_country_can_raise_artistic = yes }
		AND = { var:gm_com_dedication = flag:naturalist gm_country_can_raise_naturalist = yes }
		AND = { var:gm_com_dedication = flag:scientific gm_country_can_raise_scientific = yes }
		AND = { var:gm_com_dedication = flag:industrial gm_country_can_raise_industrial = yes }
		AND = { var:gm_com_dedication = flag:athletic gm_country_can_raise_athletic = yes }
	}
}

# ---- Which monuments count (§2.3) ---------------------------------------------
# State scope. This monument's dedication is the one the owner's variable $VAR$
# names (a dedication flag, e.g. gm_com_dedication).
gm_state_has_dedication_flag = {
	owner = { has_variable = $VAR$ }
	OR = {
		AND = { owner = { var:$VAR$ = flag:crown } gm_state_has_pm = { PM = pm_monument_crown } }
		AND = { owner = { var:$VAR$ = flag:republic } gm_state_has_pm = { PM = pm_monument_republic } }
		AND = { owner = { var:$VAR$ = flag:revolution } gm_state_has_pm = { PM = pm_monument_revolution } }
		AND = { owner = { var:$VAR$ = flag:leader } gm_state_has_pm = { PM = pm_monument_leader } }
		AND = { owner = { var:$VAR$ = flag:religious } gm_state_has_pm = { PM = pm_monument_religious } }
		AND = { owner = { var:$VAR$ = flag:civic } gm_state_has_pm = { PM = pm_monument_civic } }
		AND = { owner = { var:$VAR$ = flag:war_memorial } gm_state_has_pm = { PM = pm_monument_war_memorial } }
		AND = { owner = { var:$VAR$ = flag:artistic } gm_state_has_pm = { PM = pm_monument_artistic } }
		AND = { owner = { var:$VAR$ = flag:naturalist } gm_state_has_pm = { PM = pm_monument_naturalist } }
		AND = { owner = { var:$VAR$ = flag:scientific } gm_state_has_pm = { PM = pm_monument_scientific } }
		AND = { owner = { var:$VAR$ = flag:industrial } gm_state_has_pm = { PM = pm_monument_industrial } }
		AND = { owner = { var:$VAR$ = flag:athletic } gm_state_has_pm = { PM = pm_monument_athletic } }
	}
}

# State scope. This monument counts toward the owner's opportunity: it fits
# (dedicated, not contested, not heritage, its message stands), it is of the
# opportunity's dedication, and it stands in the opportunity's state if it
# names one. A monument contested after acceptance drops out; the progress count
# floors at zero (gm_com_progress).
gm_state_com_counts = {
	owner = { has_variable = gm_com_dedication }
	gm_state_status_fits = yes
	gm_state_has_dedication_flag = { VAR = gm_com_dedication }
	OR = {
		owner = {
			NOT = { has_variable = gm_com_state }
		}
		AND = {
			owner.var:gm_com_state ?= { save_temporary_scope_as = gm_tmp_com_state }
			this = scope:gm_tmp_com_state
		}
	}
}

# State scope. A fixed-state opportunity can be met here: the state holds no
# monument, an undedicated one (the ceremony will offer the dedication), or a
# fitting one of the opportunity's dedication. One Grand Monument per state, so
# anywhere else the opportunity falls back to "anywhere" (gm_com_fix_to_capital).
gm_state_can_host_commission = {
	OR = {
		NOT = { has_building = building_grand_monument }
		gm_state_is_dedicated = no
		AND = {
			gm_state_status_fits = yes
			gm_state_has_dedication_flag = { VAR = gm_com_dedication }
		}
	}
}

# ---- Pacing (spec §1) -----------------------------------------------------------
# Country scope. Any rank (owner, 2026-10-12: with nothing lost by letting one
# pass, a minor power may take one too; it gets longer, gm_com_time_months, and
# the AI below major power lets them pass, gm_com_ai_declines), with nothing
# open or on offer, not a side in a civil war (the winner is judged after it;
# scripting_best_practices.md § "What a Civil War's Winner Inherits"), and a
# country that can build. Not while a gifted monument's dedication is being
# put back (gm_restore_pm, spec §3.1): its method may read as another
# dedication for a month, and a baseline taken then would miscount. A return
# to Undedicated (flag:undedicated: the engine's copy, a monument founded
# beside another, a rebuilt undedicated one) does not hold it: such a monument
# reads as undedicated throughout (gm_state_is_dedicated), so it counts toward
# no baseline before the switch or after.
gm_com_can_offer = {
	gm_system_enabled = yes
	NOT = { has_variable = gm_com_offered }
	NOT = { has_variable = gm_com_open }
	NOT = { has_variable = te_cw_role }
	is_revolutionary = no
	NOT = { is_country_type = decentralized }
	any_scope_state = { is_incorporated = yes }
	NOT = {
		any_scope_state = {
			has_variable = gm_restore_pm
			NOT = { var:gm_restore_pm = flag:undedicated }
		}
	}
}

# State scope, a monument at its dedication ceremony (monument_events.2): it
# can answer the owner's open opportunity, being in the opportunity's state if
# it names one. The AI's ceremony then takes the opportunity's dedication for
# certain (gm_state_ceremony_for_commission +100, every other choice
# gm_state_ceremony_not_for_commission x0): one monument per state, so another
# dedication here would leave the opportunity unmeetable.
gm_state_ceremony_answers_commission = {
	owner = {
		has_variable = gm_com_open
		has_variable = gm_com_dedication
	}
	OR = {
		owner = {
			NOT = { has_variable = gm_com_state }
		}
		AND = {
			owner.var:gm_com_state ?= { save_temporary_scope_as = gm_tmp_com_state }
			this = scope:gm_tmp_com_state
		}
	}
}

# State scope, a monument at its ceremony: the opportunity's dedication is
# offered as the first option and the default (usability §3): it can answer
# the open opportunity, and the dedication can still be raised.
gm_state_ceremony_offers_opportunity = {
	gm_state_ceremony_answers_commission = yes
	owner = { gm_country_com_gate_holds = yes }
}

gm_state_ceremony_for_commission = {
	gm_state_ceremony_answers_commission = yes
	owner = { var:gm_com_dedication = flag:$KEY$ }
}

gm_state_ceremony_not_for_commission = {
	gm_state_ceremony_answers_commission = yes
	owner = {
		NOT = { var:gm_com_dedication = flag:$KEY$ }
	}
}

# ---- The AI (v2 §2.6) -----------------------------------------------------------
# Country scope. The AI lets an opportunity pass below major power (five
# levels are 5,000 construction, beyond a minor power's means within a few
# years: owner, 2026-10-05), at war, in hard times, with more than a year of
# government construction queued, or in debt.
gm_com_ai_declines = {
	OR = {
		NOT = { country_rank >= rank_value:major_power }
		is_at_war = yes
		gm_country_hard_times = yes
		construction_queue_government_duration >= 52
		scaled_debt > 0
	}
}

# ---- Life cycle (spec §2) -----------------------------------------------------
# Country scope. The open opportunity can no longer be met: its dedication's
# gate has fallen, or its state is no longer ours.
gm_com_has_lapsed = {
	has_variable = gm_com_open
	OR = {
		gm_country_com_gate_holds = no
		AND = {
			has_variable = gm_com_state
			NOT = { exists = var:gm_com_state }
		}
		AND = {
			has_variable = gm_com_state
			save_temporary_scope_as = gm_tmp_com_holder
			var:gm_com_state ?= {
				NOT = { owner = scope:gm_tmp_com_holder }
			}
		}
		# A historical opportunity's holder condition (signatures spec §6.1).
		gm_com_hist_cond_holds = no
	}
}

# ---- The boons (spec §3.2) -------------------------------------------------------
# Country scope. No boon is running. Part of gm_ledgers_idle, so the journal
# entry and the country pulse stay while one runs.
gm_boons_idle = {
	NOT = { has_variable = gm_boon_reign }
	NOT = { has_variable = gm_boon_regime }
	NOT = { has_variable = gm_boon_great_power }
	NOT = { has_variable = gm_boon_victory }
	NOT = { has_variable = gm_boon_defeat }
	NOT = { has_variable = gm_boon_centenary }
	NOT = { has_variable = gm_boon_ruler_death }
	NOT = { has_variable = gm_boon_space }
	NOT = { has_variable = gm_boon_faith }
	NOT = { has_variable = gm_boon_tech_artistic }
	NOT = { has_variable = gm_boon_tech_scientific }

# ---- Usability (opportunities spec, usability §1-§5) ---------------------------
# Country scope. The open opportunity's moment is $SRC$, or, once it has
# closed, the last one seized was (gm_com_seized_source): the boon's name and
# effect read through it, so monument_events.28 can name the boon after the
# record is gone.
gm_com_source_is = {
	OR = {
		AND = {
			has_variable = gm_com_source
			var:gm_com_source = flag:$SRC$
		}
		AND = {
			NOT = { has_variable = gm_com_source }
			has_variable = gm_com_seized_source
			var:gm_com_seized_source = flag:$SRC$
		}
	}
}

# Country scope. One of our Grand Monuments has levels under construction:
# Raise It waits, since script cannot count queued levels and a second press
# would queue the same levels twice.
gm_com_monument_under_construction = {
	any_scope_building = {
		is_building_type = building_grand_monument_site
		is_under_construction = yes
	}
}

# Country scope. Raise It can queue the missing levels (gm_com_raise_sgui).
# gm_com_raise_pending covers the moment between a press and its event.
gm_com_can_raise = {
	has_variable = gm_com_open
	custom_tooltip = {
		text = gm_com_raise_building_tt
		gm_com_monument_under_construction = no
		NOT = { has_variable = gm_com_raise_pending }
	}
	custom_tooltip = {
		text = gm_com_raise_nowhere_tt
		has_variable = gm_com_build_where
	}
}

# ---- Historical opportunities' holders (signatures spec §6.1) ---------------------
# Country scope. Held when the offer is made and while it is open; when it
# stops holding, the opportunity lapses (gm_com_has_lapsed).
gm_com_hist_cond_none = {
	always = yes
}

gm_com_hist_cond_german = {
	any_primary_culture = { has_discrimination_trait = language_germanophone }
}

gm_com_hist_cond_sunni = {
	country_has_state_religion = rel:sunni
}

gm_com_hist_cond_republic = {
	monument_government_is_republican = yes
}

# Country scope. The open historical opportunity's condition (gm_com_hist_cond)
# still holds; true for any other opportunity.
gm_com_hist_cond_holds = {
	OR = {
		NOT = { has_variable = gm_com_hist_cond }
		AND = { var:gm_com_hist_cond = flag:none gm_com_hist_cond_none = yes }
		AND = { var:gm_com_hist_cond = flag:german gm_com_hist_cond_german = yes }
		AND = { var:gm_com_hist_cond = flag:sunni gm_com_hist_cond_sunni = yes }
		AND = { var:gm_com_hist_cond = flag:republic gm_com_hist_cond_republic = yes }
	}
}
