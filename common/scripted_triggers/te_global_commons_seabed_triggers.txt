# GLOBAL COMMONS — SEABED LICENCES: triggers
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-16-global-commons-design.md § 2.
# The UN's side (who claims in the reserved half, who is an ITLOS party, who
# the Antarctic Treaty bars) is in te_seabed_un_hooks.txt.

# Any scope. The Global Commons rule (§ 2.9). The system arrives with this rule,
# so the check names the setting that turns it on: a save started before the
# rule existed has neither setting and keeps the uncapped Deep-Sea Mine until
# the rule is switched on (game rules can change mid-game). When it is, the
# first recording grandfathers every standing level (§ 2.9).
te_global_commons_enabled = {
	has_game_rule = global_commons_enabled
}

# Country scope. A claimant (§ 2.3): Abyssal Plain Mining and a coastal state.
te_seabed_is_claimant = {
	has_technology_researched = abyssal_plain_mining
	any_scope_state = { is_coastal = yes }
}

# Country scope. Takes part in the yearly sharing: a claimant, or a holder of
# licences without one (a conqueror without the technology), whose holdings
# are frozen at what it holds.
te_seabed_takes_part = {
	OR = {
		te_seabed_is_claimant = yes
		AND = {
			has_variable = te_seabed_country
			any_scope_state = {
				is_coastal = yes
				te_seabed_state_records > 0
			}
		}
	}
}

# Country scope. Recorded monthly: a claimant, or any country that has held
# seabed records (te_seabed_country is set at its first recording, and on a
# state with records arriving).
te_seabed_records_monthly = {
	OR = {
		te_seabed_is_claimant = yes
		has_variable = te_seabed_country
	}
}

# Any scope. The regimes (§ 3.2), from global_var:te_seabed_regime (absent = Open).
te_seabed_regime_regulated = {
	has_global_variable = te_seabed_regime
	global_var:te_seabed_regime = 2
}

te_seabed_regime_scramble = {
	has_global_variable = te_seabed_regime
	global_var:te_seabed_regime = 3
}

# Country scope. The policy (§ 2.5), from te_seabed_policy (absent = Survey).
te_seabed_policy_survey = {
	te_seabed_policy_value = 1
}

te_seabed_policy_sponsor = {
	te_seabed_policy_value = 2
}

te_seabed_policy_unilateral = {
	te_seabed_policy_value = 3
}

# Country scope. The navy counts double: Unilateral Licensing, or the Scramble
# for everyone (§ 2.5, § 3.2). The two don't stack.
te_seabed_navy_assertive = {
	OR = {
		te_seabed_policy_unilateral = yes
		te_seabed_regime_scramble = yes
	}
}

# State scope. The state has seabed data to record: records, or a Deep-Sea
# Mine (finished or under construction).
te_seabed_state_has_data = {
	OR = {
		te_seabed_state_records > 0
		any_scope_building = { is_building_type = building_ocean_mine }
	}
}

# State scope. The next Deep-Sea Mine level here would take a licence from the
# state's own basin (the owner has unbuilt share there); otherwise a distant
# one, at the long-haul penalty. For the gate's tooltip only.
te_seabed_state_next_local = {
	te_seabed_state_owner_local_unbuilt > 0
}

# State scope: building_ocean_mine's licence gate (§ 2.6), read from its
# possible. With the rule on, the owner must have room; as for the Solar Power
# Receiver, a state with a level already under construction passes at 0 room,
# so construction under way is not halted by its own queued level. That second
# branch can't tell re-checking a queued level from queueing one more, so at
# room 0 it lets one extra level in; the recording leaves that level waiting
# unrecorded (te_seabed_record_country, review H1), room drops below 0 and both
# branches shut until a share pays for it. Whether the engine re-checks
# possible on construction under way at all is unread
# (docs/testing/global-commons-playtest.md): if it doesn't, the branch can go.
# The second and third lines are information: which licence the next level
# would take. Every room read sits behind a cheap test first (review M1): the
# unbuilt shares (a variable sum), or a Deep-Sea Mine under construction here,
# since te_seabed_room walks every building of every coastal state.
te_seabed_gate = {
	trigger_if = {
		limit = { te_global_commons_enabled = yes }
		custom_tooltip = {
			text = te_seabed_gate_room_tt
			OR = {
				owner = {
					te_seabed_unbuilt_total > 0
					te_seabed_room > 0
				}
				AND = {
					any_scope_building = {
						is_building_type = building_ocean_mine
						is_under_construction = yes
					}
					owner = { te_seabed_room >= 0 }
				}
			}
		}
		trigger_if = {
			limit = {
				owner = {
					te_seabed_unbuilt_total > 0
					te_seabed_room > 0
				}
			}
			trigger_if = {
				limit = { te_seabed_state_next_local = yes }
				custom_tooltip = {
					text = te_seabed_gate_next_local_tt
					always = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = te_seabed_gate_next_distant_tt
					always = yes
				}
			}
		}
	}
}

# Country scope (yearly sharing): shares a home basin with scope:te_seabed_uni,
# both having coastal states there, unincorporated ones included
# (te_seabed_home_<basin>, set by te_seabed_compute_weights).
te_seabed_shares_home_basin_with_uni = {
	OR = {
		AND = {
			has_variable = te_seabed_home_pacific
			scope:te_seabed_uni = { has_variable = te_seabed_home_pacific }
		}
		AND = {
			has_variable = te_seabed_home_atlantic
			scope:te_seabed_uni = { has_variable = te_seabed_home_atlantic }
		}
		AND = {
			has_variable = te_seabed_home_indian
			scope:te_seabed_uni = { has_variable = te_seabed_home_indian }
		}
		AND = {
			has_variable = te_seabed_home_arctic
			scope:te_seabed_uni = { has_variable = te_seabed_home_arctic }
		}
		AND = {
			has_variable = te_seabed_home_southern
			scope:te_seabed_uni = { has_variable = te_seabed_home_southern }
		}
	}
}

# Any scope: the yearly sharing is due (te_seabed_global_monthly). Once a
# year, from the first global monthly pulse of a new year, and in the month
# after a new claimant appears (te_seabed_reshare), so a country that has just
# researched Abyssal Plain Mining need not wait for January to build. Never
# before anyone can claim.
te_seabed_pass_due = {
	OR = {
		NOT = { has_global_variable = te_seabed_pass_year }
		AND = {
			has_global_variable = te_seabed_pass_year
			global_var:te_seabed_pass_year < te_history_current_year
		}
		has_global_variable = te_seabed_reshare
	}
	any_country = { te_seabed_is_claimant = yes }
}

# No scope: the sharing about to run is the year's first (seabed claims spec
# § 2.2): January's, or the first mid-year one of a year with no January pass.
# Read before te_seabed_yearly_pass sets te_seabed_pass_year, which caches it
# as the global flag te_seabed_first_of_year for its own steps.
te_seabed_first_sharing_of_year = {
	OR = {
		NOT = { has_global_variable = te_seabed_pass_year }
		AND = {
			has_global_variable = te_seabed_pass_year
			global_var:te_seabed_pass_year < te_history_current_year
		}
	}
}

# Country scope. Model: floor_in_force. A stored floor counts for nothing off
# Sponsor Contractors, in a Scramble, or for a country no longer a claimant;
# in the Southern Ocean also where the Antarctic Treaty bars it.
te_seabed_floor_void = {
	OR = {
		NOT = { te_seabed_policy_sponsor = yes }
		te_seabed_regime_scramble = yes
		NOT = { te_seabed_is_claimant = yes }
	}
}

te_seabed_floor_void_pacific = {
	te_seabed_floor_void = yes
}

te_seabed_floor_void_atlantic = {
	te_seabed_floor_void = yes
}

te_seabed_floor_void_indian = {
	te_seabed_floor_void = yes
}

te_seabed_floor_void_arctic = {
	te_seabed_floor_void = yes
}

te_seabed_floor_void_southern = {
	OR = {
		te_seabed_floor_void = yes
		te_seabed_southern_barred = yes
	}
}

# Country scope: the Seabed tab's policy buttons (te_global_commons_seabed_sguis.txt).
# A claimant may change to a policy it doesn't hold; one that holds influence
# needs that much available, as the UN's stances do.
# The five-year lock (seabed claims spec § 2.4): any change holds the policy
# for te_seabed_policy_lock_months. Scientific Survey stays open while our
# influence is in deficit; Unilateral Licensing while a rival in a home basin
# has switched to it since our last change. ROOT = the country choosing (the
# button's country, or the AI's yearly pulse).
te_seabed_can_choose_survey = {
	te_seabed_policy_buttons_open = yes
	custom_tooltip = {
		text = te_seabed_policy_not_current_tt
		NOT = { te_seabed_policy_survey = yes }
	}
	custom_tooltip = {
		text = te_seabed_policy_deficit_tt
		OR = {
			NOT = { te_seabed_policy_locked = yes }
			te_seabed_influence_deficit = yes
		}
	}
}

te_seabed_can_choose_sponsor = {
	te_seabed_policy_buttons_open = yes
	custom_tooltip = {
		text = te_seabed_policy_not_current_tt
		NOT = { te_seabed_policy_sponsor = yes }
	}
	custom_tooltip = {
		text = te_seabed_policy_locked_tt
		NOT = { te_seabed_policy_locked = yes }
	}
	custom_tooltip = {
		text = te_seabed_policy_influence_2_tt
		influence >= te_seabed_sponsor_cost_value
	}
}

te_seabed_can_choose_unilateral = {
	te_seabed_policy_buttons_open = yes
	custom_tooltip = {
		text = te_seabed_policy_not_current_tt
		NOT = { te_seabed_policy_unilateral = yes }
	}
	custom_tooltip = {
		text = te_seabed_policy_answer_tt
		OR = {
			NOT = { te_seabed_policy_locked = yes }
			te_seabed_unilateral_answer_open = yes
		}
	}
	custom_tooltip = {
		text = te_seabed_policy_influence_3_tt
		influence >= te_seabed_unilateral_cost_value
	}
}

# Country scope: changed policy less than te_seabed_policy_lock_months ago. A
# country that never changed (no date: the first choice, an old save) is free.
te_seabed_policy_locked = {
	has_variable = te_seabed_policy_changed
	te_seabed_policy_unlock_month_value > te_history_month_index
}

# Country scope: influence in deficit, the way out of a cost it can't carry.
te_seabed_influence_deficit = {
	influence < 0
}

# Country scope (ROOT = the same country): a country sharing a home basin with
# ROOT switched to Unilateral Licensing after ROOT's own last change, and
# holds it.
te_seabed_unilateral_answer_open = {
	has_variable = te_seabed_policy_changed
	any_country = {
		NOT = { this = root }
		has_variable = te_seabed_shared
		te_seabed_policy_unilateral = yes
		has_variable = te_seabed_policy_changed
		var:te_seabed_policy_changed > root.var:te_seabed_policy_changed
		te_seabed_shares_home_basin_with_root = yes
	}
}

# Country scope: shares a home basin with ROOT, both having coastal states

# ---- Leases (seabed claims spec § 4) ----

# Country scope, the article's possible (ROOT the lessor, scope:other_country
# the lessee). Model: can_lease. Two claimants; the lessor has own claims in
# basin $B$; the lessee holds no lease there but this lessor's (so a treaty
# holding the lease can be renegotiated) and has not relinquished the basin;
# in the Southern Ocean, neither side barred by the Antarctic Treaty.
te_seabed_can_lease = {
	custom_tooltip = {
		text = te_seabed_lease_lessor_tt
		te_seabed_is_claimant = yes
	}
	custom_tooltip = {
		text = te_seabed_lease_claims_tt
		te_seabed_own_claims_$B$ > 0
	}
	custom_tooltip = {
		text = te_seabed_lease_lessee_tt
		scope:other_country ?= {
			te_seabed_is_claimant = yes
			NOT = { has_variable = te_seabed_relinq_$B$ }
		}
	}
	custom_tooltip = {
		text = te_seabed_lease_one_tt
		scope:other_country ?= {
			OR = {
				NOT = { has_variable = te_seabed_lease_src_$B$ }
				var:te_seabed_lease_src_$B$ ?= { this = root }
			}
		}
	}
	trigger_if = {
		limit = { te_seabed_is_southern_$B$ = yes }
		custom_tooltip = {
			text = te_seabed_lease_antarctic_tt
			NOT = { te_seabed_southern_barred = yes }
			scope:other_country ?= {
				NOT = { te_seabed_southern_barred = yes }
			}
		}
	}
}

# The article's requirement_to_maintain (ROOT the article): both sides still
# claim, and in the Southern Ocean neither is barred. A lease whose article
# breaks ends at the next sharing (te_seabed_lease_shift_basin).
te_seabed_lease_maintained = {
	custom_tooltip = {
		text = te_seabed_lease_maintain_tt
		scope:source_country ?= { te_seabed_is_claimant = yes }
		scope:target_country ?= { te_seabed_is_claimant = yes }
	}
	trigger_if = {
		limit = { te_seabed_is_southern_$B$ = yes }
		custom_tooltip = {
			text = te_seabed_lease_antarctic_tt
			scope:source_country ?= {
				NOT = { te_seabed_southern_barred = yes }
			}
			scope:target_country ?= {
				NOT = { te_seabed_southern_barred = yes }
			}
		}
	}
}

# Country scope, the lessee, its lessor saved as scope:te_sb_lessor
# (te_seabed_lease_shift_basin). Model: lease_stands, less the article, which
# the caller found: both sides claim; in the Southern Ocean neither is barred.
te_seabed_lease_stands = {
	te_seabed_is_claimant = yes
	exists = scope:te_sb_lessor
	scope:te_sb_lessor = { te_seabed_is_claimant = yes }
	trigger_if = {
		limit = { te_seabed_is_southern_$B$ = yes }
		NOT = { te_seabed_southern_barred = yes }
		scope:te_sb_lessor = {
			NOT = { te_seabed_southern_barred = yes }
		}
	}
}

# ---- Subjects: Administer Seabed Claims (seabed claims spec § 5.2) ----

# Country scope, a subject, its overlord saved as scope:te_sb_over. The
# pact's terms without the pact itself (its accept_effect runs before the pact
# exists): both claim; in the Southern Ocean neither is barred.
te_seabed_pact_ok = {
	exists = scope:te_sb_over
	te_seabed_is_claimant = yes
	scope:te_sb_over = { te_seabed_is_claimant = yes }
	trigger_if = {
		limit = { te_seabed_is_southern_$B$ = yes }
		NOT = { te_seabed_southern_barred = yes }
		scope:te_sb_over = {
			NOT = { te_seabed_southern_barred = yes }
		}
	}
}

# Country scope, a subject, its overlord saved as scope:te_sb_over. Model:
# pact_stands. The pact is the record: it stands in basin $B$ while the
# overlord holds it, the country is still its direct subject and the terms
# hold (te_seabed_pact_ok).
te_seabed_pact_stands = {
	exists = scope:te_sb_over
	is_direct_subject_of = scope:te_sb_over
	has_diplomatic_pact = { who = scope:te_sb_over type = te_seabed_administer_claims }
	te_seabed_pact_ok = { B = $B$ }
}
