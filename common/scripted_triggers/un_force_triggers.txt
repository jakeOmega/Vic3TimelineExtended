# The Standing UN Force has been founded and the UN still stands. An
# institution, not a convention: nobody ratifies it, so its flag is
# un_inst_standing_force (§0.12 "Identifiers"), set by
# un_standing_force_on_carried and cleared by un_force_on_dissolve.
# Scope: any
un_force_in_force = {
	has_global_variable = un_inst_standing_force
}

# ---- The topic (the hooks un_propose_triggers.txt calls) ------------------------

# There is something to table: the Force has not been founded yet.
# Scope: the proposing country
un_standing_force_case_exists = {
	NOT = { un_force_in_force = yes }
}

# The topic's own gate: a major power moves it (a force is put together by the
# powers that can lend it soldiers).
# Scope: the proposing country
un_standing_force_can_table = {
	custom_tooltip = {
		text = un_standing_force_need_major_power_tt
		country_rank >= rank_value:major_power
	}
}

# ---- War goals against a mission's host -------------------------------------------

# A war goal just added to this play is barred: the Force stands, the UN is
# Supranational, the actor is a member, and it holds a war goal against the
# play's target, which hosts an active peacekeeping or stabilisation mission
# (un_mission_protected). A non-member is not barred; it pays the surcharge
# (un_teeth_war_goal_surcharge). The goal types are listed in
# un_force_holds_goal_against.
# Scope: diplomatic play; expects scope:actor (from on_wargoal_added)
un_force_war_goal_barred = {
	un_force_in_force = yes
	un_tier_is_supranational = yes
	exists = scope:actor
	scope:actor = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
	}
	target ?= {
		un_mission_protected = yes
		scope:actor = { un_force_holds_goal_against = { TARGET = prev } }
	}
}

# The scoped country holds, in a diplomatic play, a war goal of one of the
# types the Force bars against $TARGET$. The list is explicit because the
# engine names one war goal type at a time (un_mandate_has_prohibited_goal has
# the same shape); it is every type a play opens with in vanilla, the
# subjugation types of 1.14 and the mod's reunification, less the goals only a
# subject can hold (the actor is never one: un_teeth_war_goal_surcharge leaves
# subjects alone) and the UN mandate's own goal (a bound mandate's holder is
# left alone too). Keep it in step with un_force_strike_war_goals.
# Parameters: $TARGET$ = the protected host
# Scope: country (the actor)
un_force_holds_goal_against = {
	OR = {
