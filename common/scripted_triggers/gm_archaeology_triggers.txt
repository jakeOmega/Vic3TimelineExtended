# ============================================================================
# GRAND MONUMENTS — archaeology: scripted triggers
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-14-grand-monument-archaeology-design.md (issue #939),
# reworked by 2026-10-16-grand-monument-archaeology-rework-design.md (foreign
# digs by treaty: gm_arch_may_dig_ground, gm_arch_state_conceded_to).
# A buried site is a script container tagged gm_site (gm_archaeology_effects.txt).
# Its records, under the names a returned trophy's carry so the container
# custom localization names it (gm_trophy_custom_loc.txt) and the monument
# machinery places it unchanged (gm_monument_effects.txt):
#   gm_tr_landmark, gm_tr_skin, gm_tr_form, gm_tr_levels, gm_tr_dedication,
#   gm_tr_namesake, gm_tr_share, gm_tr_origin_region   the monument it will be
#   gm_site_tech, gm_site_vector                       what opens it (below)
#   gm_site_status                                     1 buried, 2 known,
#                                                      3 being dug, 4 unearthed
#   gm_site_sponsor, gm_site_method, gm_site_progress  the dig
# The technology gate (gm_arch_has_gate_for) is in gm_archaeology_gate_triggers.txt: one branch
# for each technology the table uses, kept in step with it by test_grand_monument_archaeology.py.

# ---- What a site is -----------------------------------------------------------------
# Container scope. A survey site is open to anyone with its technology as soon
# as it is buried; an incidental one only once it has been turned up (status 2),
# which a monthly chance does (gm_arch_site_monthly). Neither while it is dug.
gm_arch_site_open = {
	has_variable = gm_site_status
	OR = {
		var:gm_site_status = 2
		AND = {
			var:gm_site_status = 1
			var:gm_site_vector = flag:survey
		}
	}
}

# Container scope. Being dug now.
gm_arch_site_digging = {
	has_variable = gm_site_status
	var:gm_site_status = 3
}

# Container scope. $WHO$ is the country digging it.
gm_arch_digs_for = {
	gm_arch_site_digging = yes
	var:gm_site_sponsor ?= { this = $WHO$ }
}

# ---- Who may dig ----------------------------------------------------------------------------
# Container scope. $WHO$ (a country) holds a part of the ground the site lies
# under.
gm_arch_owns_ground = {
	var:gm_tr_origin_region ?= {
		any_scope_state = { owner = $WHO$ }
	}
}

# State scope. $WHO$ holds an in-force treaty article of type $ARTICLE$
# (archaeological_concession or excavation_permit) granted by this state's
# owner and naming this state. The treaty is the record: nothing is written on
# the state, so a treaty that ends by any road, or a state that changes hands,
# ends the right at once (te_country_concedes_state_article's shape;
# scripting_best_practices.md § "A Treaty Article's State-Side Effects Must Be
# Checked Against the Treaty").
gm_arch_state_conceded_to = {
	save_temporary_scope_as = gm_arch_cs_state
	exists = owner
	owner = {
		save_temporary_scope_as = gm_arch_cs_host
		NOT = { this = $WHO$ }
	}
	$WHO$ = {
		any_scope_treaty = {
			any_scope_article = {
				has_type = $ARTICLE$
				source_country = scope:gm_arch_cs_host
				input_state = scope:gm_arch_cs_state
			}
		}
	}
}

# State scope. $WHO$ may dig here by either article.
gm_arch_state_open_abroad = {
	OR = {
		gm_arch_state_conceded_to = { WHO = $WHO$ ARTICLE = archaeological_concession }
		gm_arch_state_conceded_to = { WHO = $WHO$ ARTICLE = excavation_permit }
	}
}

# Container scope. $WHO$ may dig the site's ground: it holds a part of it, or
# the right to dig a part another holds (rework spec § 3).
gm_arch_may_dig_ground = {
	var:gm_tr_origin_region ?= {
		any_scope_state = {
			OR = {
				owner = $WHO$
				gm_arch_state_open_abroad = { WHO = $WHO$ }
			}
		}
	}
}

# Container scope. $WHO$ would dig this site abroad, under the article
# $ARTICLE$ (and not on its own ground), for the open rows' terms line. A
# concession wins over a permit when both are in force.
gm_arch_would_dig_abroad_by = {
	NOT = { gm_arch_owns_ground = { WHO = $WHO$ } }
	var:gm_tr_origin_region ?= {
		any_scope_state = {
			gm_arch_state_conceded_to = { WHO = $WHO$ ARTICLE = $ARTICLE$ }
		}
	}
}

# Container scope. The site's own conditions for $WHO$: open, on ground it may
# dig, and its technology in hand. The methods add their own (below).
gm_arch_can_dig = {
	gm_system_enabled = yes
	has_tag = gm_site
	gm_arch_site_open = yes
	gm_arch_may_dig_ground = { WHO = $WHO$ }
	gm_arch_has_gate_for = { WHO = $WHO$ }
}

# The same, for a tooltip: each condition on its own line.
gm_arch_can_dig_tt = {
	custom_tooltip = {
		text = gm_arch_dig_open_tt
		gm_arch_site_open = yes
	}
	custom_tooltip = {
		text = gm_arch_dig_ground_tt
		gm_arch_may_dig_ground = { WHO = $WHO$ }
	}
	custom_tooltip = {
		text = gm_arch_dig_tech_tt
		gm_arch_has_gate_for = { WHO = $WHO$ }
	}
}

# ---- What a state offers a foreign sponsor (the articles' inputs) ------------------------------
# State scope. A site in this state's region is open to $WHO$: a survey site
# whose technology it has, or an incidental one already turned up. The
# articles name only such a state, so none is signed for nothing.
gm_arch_state_offers_site = {
	save_temporary_scope_as = gm_arch_os_state
	has_global_variable_list = gm_arch_sites
	any_in_global_list = {
		variable = gm_arch_sites
		gm_arch_site_open = yes
		gm_arch_has_gate_for = { WHO = $WHO$ }
		var:gm_tr_origin_region ?= {
			any_scope_state = { this = scope:gm_arch_os_state }
		}
	}
}

# ---- The three methods (spec § Methods) -----------------------------------------------------
# Country scope. Antiquarian & Dynamite needs Romanticism, Systematic
# Stratigraphy Empiricism, Forensic & Remote Sensing Computer Networks.
gm_arch_method_antiquarian_ok = {
	has_technology_researched = romanticism
}

gm_arch_method_systematic_ok = {
	has_technology_researched = empiricism
}

gm_arch_method_forensic_ok = {
	has_technology_researched = computer_networks
}

# Country scope. It makes at least the weekly innovation a dig by the method
# diverts (country_innovation is net of the digs it already runs), so a launch
# never takes its weekly innovation below zero (owner, 2026-10-16). The AI's
# own floor is four times this (gm_arch_ai_affords_<method>).
gm_arch_spares_antiquarian = {
	country_innovation >= gm_arch_innovation_antiquarian
}

gm_arch_spares_systematic = {
	country_innovation >= gm_arch_innovation_systematic
}

gm_arch_spares_forensic = {
	country_innovation >= gm_arch_innovation_forensic
}

# Container scope, with gm_site_order_method set: $WHO$ makes the weekly
# innovation the method ordered diverts (arch_events.10 checks again).
gm_arch_order_spared = {
	OR = {
		AND = {
			var:gm_site_order_method = flag:antiquarian
			$WHO$ = { gm_arch_spares_antiquarian = yes }
		}
		AND = {
			var:gm_site_order_method = flag:systematic
			$WHO$ = { gm_arch_spares_systematic = yes }
		}
		AND = {
			var:gm_site_order_method = flag:forensic
			$WHO$ = { gm_arch_spares_forensic = yes }
		}
	}
}

# ---- A country's side -------------------------------------------------------------------------
# Country scope. It digs, or a site waits for it (gm_arch_pass writes the
# counts): the journal entry stays for either.
gm_arch_country_has_any = {
	OR = {
		AND = {
			has_variable = gm_arch_dig_count
			var:gm_arch_dig_count > 0
		}
		AND = {
			has_variable = gm_arch_open_count
			var:gm_arch_open_count > 0
		}
	}
}

# Country scope. Nothing dug, nothing waiting, no cost or drain still on.
# Part of gm_ledgers_idle.
gm_arch_country_idle = {
	gm_arch_country_has_any = no
}

# Country scope, the host. A site on its ground is open to $WHO$: the
# articles' `possible`, walking the 35 sites rather than every state.
gm_arch_country_offers_site = {
	save_temporary_scope_as = gm_arch_os_host
	has_global_variable_list = gm_arch_sites
	any_in_global_list = {
		variable = gm_arch_sites
		gm_arch_site_open = yes
		gm_arch_has_gate_for = { WHO = $WHO$ }
		var:gm_tr_origin_region ?= {
			any_scope_state = { owner = scope:gm_arch_os_host }
		}
	}
}

# ---- The AI -------------------------------------------------------------------------------------
# Country scope. An AI digs only at peace and out of default, with at most two
# digs going (gm_arch_ai_decide).
gm_arch_ai_may_dig = {
	is_ai = yes
	is_at_war = no
	in_default = no
}

# Country scope, an AI: it can spare the weekly innovation a method diverts,
# making at least four times as much (gm_arch_ai_floor_<method>), and has the
# method's technology.
gm_arch_ai_affords_antiquarian = {
	gm_arch_method_antiquarian_ok = yes
	country_innovation >= gm_arch_ai_floor_antiquarian
}

gm_arch_ai_affords_systematic = {
	gm_arch_method_systematic_ok = yes
	country_innovation >= gm_arch_ai_floor_systematic
}

gm_arch_ai_affords_forensic = {
	gm_arch_method_forensic_ok = yes
	country_innovation >= gm_arch_ai_floor_forensic
}
