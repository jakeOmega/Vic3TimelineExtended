# ============================================================================
# CULTURAL HEGEMONY — shared button eligibility triggers
# ============================================================================
# Each `ch_possible_<button>` holds the body that used to live inline in the
# matching scripted_button's `possible` block. Both the journal-entry button and
# the `ch_policy_sgui` scripted GUI ask these, so what greys a control out
# cannot drift apart between the two surfaces.
#
# The `custom_tooltip` clauses are phrased as CONDITIONS, not complaints,
# because IsValidTooltip renders each of them with a tick or a cross.
#
# Scope: country (the journal entry's owner).
# ============================================================================

# What a country needs before je_cultural_hegemony activates: the entry's
# `possible` (its is_shown_when_inactive is the game rule alone). Also the
# checklist in the Society panel's greyed Hegemony tab
# (te_society_ch_tab_unlock_sgui's is_valid), which is why the all-countries
# test carries its own tooltip line.
ch_entry_unlocked = {
	custom_tooltip = {
		text = ch_unlock_any_mass_media_tt
		any_country = {
			has_technology_researched = mass_media
		}
	}
	has_technology_researched = romanticism
}

# ---- Widget row visibility (mirrors each scripted_button's `visible`) ----
# The enable / disable controls in the widget swap on exactly the conditions the
# journal-entry buttons swap on, so the pair can never both offer "Enable".

# Is the programme running? The one positive check per programme. A programme
# is a country variable, set by its ch_effect_<programme> and removed by its
# ch_effect_disable_<programme> or the entry's monthly pulse once the Ministry
# of Culture is gone; je_cultural_hegemony declares the programme's modifier
# (scaling_modifiers_while_active) at 1 while this is true. has_modifier cannot
# see a declared modifier, so nothing tests the modifier by name. A civil war's
# winner inherits the variables (but no modifiers), so it keeps the programmes.
# Read by: the entry's declarations, ch_shown_<programme> below (the
# journal-entry enable button's `visible` swap), the disable button's `visible`
# and `possible`, the two mutual exclusions, ch_prog_count
# (ch_set_display_state) and ch_active_<programme>_sgui's `is_shown`, which is
# what the widget's Enable / Disable pair reads. The entry test matches the
# declaration, which applies only while the entry is active.
ch_active_world_exposition = {
	has_journal_entry = je_cultural_hegemony
	has_variable = ch_world_exposition_on
}

ch_active_cultural_institutes = {
	has_journal_entry = je_cultural_hegemony
	has_variable = ch_cultural_institutes_on
}

ch_active_global_media_campaign = {
	has_journal_entry = je_cultural_hegemony
	has_variable = ch_global_media_campaign_on
}

ch_active_cultural_protectionism = {
	has_journal_entry = je_cultural_hegemony
	has_variable = ch_cultural_protectionism_on
}

# Is programme funding in force? je_cultural_hegemony declares
# ch_cultural_program_funding and ch_cultural_program_funding_cost while it is:
# the Ministry of Culture law and at least one funding step (the gate the old
# monthly re-add had).
ch_program_funding_in_force = {
	has_law = law_type:law_ministry_of_culture
	ch_program_funding_level_value >= 1
}

# Does this country trail the world's cultural hegemon? The Foreign Cultural
# Benchmark (cultural_hegemony_foreign_benchmark) is on it exactly then:
# je_cultural_hegemony declares the modifier at var:ch_benchmark_applied, which
# the entry's monthly pulse writes every month (cultural_hegemony_benchmark_mult,
# or 0 below 0.1). has_modifier cannot see a declared modifier: every reader of
# the benchmark (events, AI weights, the overview's alert, UN votes, the
# hegemon's baseline movement pressure) asks this instead.
# Scope: country.
ch_trails_hegemon = {
	has_journal_entry = je_cultural_hegemony
	has_variable = ch_benchmark_applied
	var:ch_benchmark_applied >= 0.1
}

# ============================================================================
# POLITICAL MODEL CLASSIFICATION
# ============================================================================
# Used by ch_set_political_model (common/scripted_effects/
# cultural_hegemony_effects.txt), the single place that sorts a country into
# the political-model family its culture exports.

# ---- Regressive laws ----
# Scope: country
# True when the country keeps any law that marks an unreformed state. A military
# junta only counts as developmentalist without these. Mirrors vanilla's
# country_has_laws_in_need_of_critical_modernization
# (game/common/scripted_triggers/00_ip4_victoria_scripted_triggers.txt) minus its
# two taxation laws, which linger well into the mid-game, plus hereditary
# bureaucrats, the mod's Feudal Contracts and the three slavery laws.
ch_has_regressive_laws = {
	OR = {
		has_law_or_variant = law_type:law_isolationism
		has_law_or_variant = law_type:law_extraction_economy
		has_law_or_variant = law_type:law_traditionalism
		has_law_or_variant = law_type:law_industry_banned
		has_law_or_variant = law_type:law_serfdom
		has_law_or_variant = law_type:law_peasant_levies
		has_law_or_variant = law_type:law_guild_system
		has_law_or_variant = law_type:law_hereditary_bureaucrats
		has_law_or_variant = law_type:law_feudal_contracts
		has_law_or_variant = law_type:law_slave_trade
		has_law_or_variant = law_type:law_legacy_slavery
		has_law_or_variant = law_type:law_debt_slavery
	}
}

# ---- Who leads the government ----
# Scope: country
# True when the interest group with the most clout in government holds a
# communist ideology (an interest group counts its leader's ideology among its
# own): the Trade Unions' Communist (1.15, under Factory Councils), a Communist
# leader (ideology_communist_leader since 1.15) or a Vanguardist one. The laws alone can't tell a communist one-party planned economy from a
# nationalist, technocratic or military one, so the Communist model's one-party
# branch asks who runs the party.
ch_government_led_by_communists = {
	any_interest_group = {
		is_in_government = yes
		is_strongest_ig_in_government = yes
		OR = {
			has_ideology = ideology:ideology_communist
			has_ideology = ideology:ideology_communist_leader
			has_ideology = ideology:ideology_vanguardist
		}
	}
}

# Scope: country
# True when that interest group holds the Fascist or Ethno-Nationalist ideology,
# the two that back both hard-line citizenship laws. Cultural Citizenship
# (National Supremacy) is a common starting law, Britain's among them, so under
# Autocracy it fits an absolute monarch or a caudillo as well as a fascist
# dictatorship; the Fascist model's Autocracy branch asks who runs the government.
ch_government_led_by_fascists = {
	any_interest_group = {
		is_in_government = yes
		is_strongest_ig_in_government = yes
		OR = {
			has_ideology = ideology:ideology_fascist
			has_ideology = ideology:ideology_ethno_nationalist
		}
	}
}

# Scope: country
# True when that interest group holds any left-wing ideology: the communist
# ones above (1.15's Communist took over from the Trade Unions' Socialist), the
# Trade Unions' baseline Proletarian, or a State Socialist leader (the Social
# Democrat until 1.15). A Command
# Economy run by the left counts as Socialist, and Cultural Citizenship in a
# one-party state run by the left is not Fascist.
ch_government_led_by_the_left = {
	any_interest_group = {
		is_in_government = yes
		is_strongest_ig_in_government = yes
		OR = {
			has_ideology = ideology:ideology_communist
			has_ideology = ideology:ideology_communist_leader
			has_ideology = ideology:ideology_vanguardist
			has_ideology = ideology:ideology_proletarian
			has_ideology = ideology:ideology_state_socialist
		}
	}
}

# ---- Switched to the hegemon's model ----
# Scope: country
# True when this rebuild's census moved the country onto the model the world's
# leading cultural power runs, from something else. Reads the census pair
# (var:ch_model_census / _prev, written only by ch_add_country_to_model_totals),
# never var:ch_model, which the pressure effect also rewrites between rebuilds.
# Drives the Domino event (cultural_hegemony.20).
ch_switched_to_hegemon_model = {
	has_variable = ch_model_census_prev
	has_variable = ch_model_census
	NOT = { var:ch_model_census = var:ch_model_census_prev }
	has_global_variable = ch_rank_1_ideology
	var:ch_model_census = global_var:ch_rank_1_ideology
	NOT = { global_var:ch_rank_1_ideology = 12 }
	exists = global_var:ch_rank_1
	NOT = { this = global_var:ch_rank_1 }
}
