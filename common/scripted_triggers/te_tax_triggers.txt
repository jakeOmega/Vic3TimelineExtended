# Legislated tax code: game-rule gates (common/game_rules/extra_game_rules.txt,
# te_tax_code_rule).
#
# Both read only a game rule, so they work from any scope. Every hook, pulse,
# event, effect entry point, GUI root and AI path of the tax code asks
# te_tax_code_on; customs legislation asks te_tax_customs_on as well.
#
# They are written positively on purpose. A save from before the rule existed
# has no value stored for it, so NOT = { has_game_rule = te_tax_code_disabled }
# would be true there and migrate the campaign. Do not rewrite these as
# negations, and do not add a negated check anywhere else.

# True under both enabled settings.
te_tax_code_on = {
	OR = {
		has_game_rule = te_tax_code_enabled
		has_game_rule = te_tax_code_enabled_customs
	}
}

# True only under the setting that also legislates customs.
te_tax_customs_on = {
	has_game_rule = te_tax_code_enabled_customs
}

# ---------------------------------------------------------------------------
# Customs authority (plan Task 15; docs/systems/tax_code_schema.md, "Customs
# schedule"). Country scope. Only the market owner sets tariff levels
# (SET_TARIFF_LEVEL_NOT_MARKET_OWNER); a junior partner of a customs union, a
# subject pact included, trades in its senior's market and may not.
# ---------------------------------------------------------------------------

# The country owns the market it trades in. market.owner is vanilla's own form
# (china_buttons.txt `market.owner = root`, 42_subjects_request_market_control.txt
# `market.owner = this`); is_junior_in_customs_union is the treaty articles'.
te_tax_owns_market = {
	exists = market
	market.owner = this
	is_junior_in_customs_union = no
}

# The code may hold this country's customs: the customs option, and its own market.
te_tax_customs_eligible = {
	te_tax_customs_on = yes
	te_tax_owns_market = yes
}

# The code holds this country's customs now: a code in force that has taken them
# up (te_tax_customs_held, written by the customs migration and cleared by
# te_tax_customs_revalidate when the market is lost), and the market is still
# its own. The customs records (te_tax_en_<d>_<good>) are read, synced and
# drafted only while this holds; a country that lost its market keeps them
# frozen until it gains one again, which migrates afresh.
te_tax_customs_authority = {
	te_tax_code_in_force = yes
	te_tax_customs_eligible = yes
	has_variable = te_tax_customs_held
	var:te_tax_customs_held = 1
}

# Country scope. Slot SLOT (a or b) may take a package from the bill record:
# the rule is on, the country holds the code's tokens, the slot is free, the
# bill has a due month after the current one, and every sunset in the bill
# falls at least one month after commencement (te_tax_gen_bill_sunsets_valid:
# the store writes _exp = due + offset, so _exp >= _due + 1 exactly when the
# offset is at least 1). te_tax_store_package checks it; Task 6's passage
# trigger repeats the sunset and slot checks for the player's tooltip.
te_tax_can_store_package = {
	te_tax_code_on = yes
	has_variable = te_tax_schema
	var:te_tax_p$SLOT$_on = 0
	has_variable = te_tax_bl_due
	var:te_tax_bl_due > te_history_month_index
	te_tax_gen_bill_sunsets_valid = yes
}

# Country scope. The collection writer already ran today (te_tax_sync_day, the
# day te_tax_sync_collection last wrote; final review A-I1): a second sync the
# same day defers to te_tax.4 the next day (te_tax_defer_sync).
te_tax_synced_today = {
	has_variable = te_tax_sync_day
	var:te_tax_sync_day >= te_tax_today
}

# Slot SLOT (a or b) holds a package awaiting commencement in a month before
# the one in the variable DUE (te_tax_dr_due or te_tax_bl_due). For the
# baselines in te_tax_generated_values.txt.

	has_variable = te_tax_p$SLOT$_on

	var:te_tax_p$SLOT$_state = 1
	has_variable = $DUE$
	var:te_tax_p$SLOT$_due < var:$DUE$

# Slot b's package commences before slot a's: an earlier month, or the same
# month and a lower approval sequence (the scheduler's order).
te_tax_slot_b_first = {
	has_variable = te_tax_pa_due
	has_variable = te_tax_pb_due
	has_variable = te_tax_pa_seq
	has_variable = te_tax_pb_seq
	OR = {
		var:te_tax_pb_due < var:te_tax_pa_due
		AND = {
			var:te_tax_pb_due = var:te_tax_pa_due
			var:te_tax_pb_seq < var:te_tax_pa_seq
		}
	}
}

# is always allowed. Naming one needs a regional relief of at least one band in

# scope:te_tax_st, is the country), and room: at most three states
# (any_in_list's count; variable_list_size's `target` key fails to load in

	custom_tooltip = { text = te_tax_tt_code_in_force te_tax_code_in_force = yes }
	trigger_if = {
		limit = { te_tax_draft_active = yes }
		custom_tooltip = { text = te_tax_tt_relief_state_given exists = scope:te_tax_st }

			text = te_tax_tt_relief_regional_in_draft

		}

					has_variable_list = te_tax_dr_relief_states

				text = te_tax_tt_relief_states_max

# The draft can become the bill: it changes something, commences within range,
# and every provision it changes is still on the baseline it was drafted on.
te_tax_draft_ready = {
	trigger_if = {
		limit = { te_tax_draft_active = yes }
		custom_tooltip = { text = te_tax_tt_draft_changes te_tax_gen_draft_touches_any = yes }
		custom_tooltip = {
			text = te_tax_tt_draft_due_range
			var:te_tax_dr_due >= te_tax_due_earliest
			var:te_tax_dr_due <= te_tax_due_latest
		}
		custom_tooltip = { text = te_tax_tt_draft_baseline te_tax_gen_draft_baseline_current = yes }
		# Traditionalism disallows the vanilla laws that tax wages and dividends
		# (Per-Capita, Proportional, Graduated); under it a bill may not either.
		# Here, so a revision is held to it as well as an introduction.
		# The draft's own index on the left: 0 where it changes the tax, or
		# untouched (-1) where existing law levies none (te_tax_dr_zero_limit_<key>
		# is -1 then, -2 otherwise).
		trigger_if = {
			limit = { has_law = law_type:law_traditionalism }
			custom_tooltip = {
				text = te_tax_tt_draft_traditionalism
				has_variable = te_tax_dr_wage
				has_variable = te_tax_dr_div
				OR = { var:te_tax_dr_wage = 0 var:te_tax_dr_wage <= te_tax_dr_zero_limit_wage }
				OR = { var:te_tax_dr_div = 0 var:te_tax_dr_div <= te_tax_dr_zero_limit_div }
			}
		}
		# A regional relief that relieves names at least one state; a relief of
		# none names none (te_tax_dr_relief_settle). The toggle allows at most
		# three, checked again here. Every named state must still be the
		# country's own and incorporated (one lost since it was named forces a
		# revision; inside the list PREV is the country).
		# Customs (plan Task 15): only a code that holds the country's customs may
		# change them; a draft that touches them is held to it at introduction and
		# at every revision (a market lost since the draft was written).
		trigger_if = {
			limit = { te_tax_gen_draft_touches_customs = yes }
			custom_tooltip = { text = te_tax_tt_customs_authority te_tax_customs_authority = yes }
		}
		trigger_if = {
			limit = { var:te_tax_dr_regrel >= 1 }
			custom_tooltip = {
				text = te_tax_tt_relief_states_named
				has_variable_list = te_tax_dr_relief_states
				any_in_list = { variable = te_tax_dr_relief_states exists = this }
			}
		}
		trigger_if = {
			limit = { has_variable_list = te_tax_dr_relief_states }
			custom_tooltip = {
				text = te_tax_tt_relief_states_at_most
				NOT = { any_in_list = { variable = te_tax_dr_relief_states count >= 4 exists = this } }
			}
			custom_tooltip = {
				text = te_tax_tt_relief_states_own
				NOT = {
					any_in_list = {
						variable = te_tax_dr_relief_states
						NAND = { is_incorporated = yes owner = PREV }
					}
				}
			}
		}
	}
	trigger_else = {
		custom_tooltip = { text = te_tax_tt_draft_open always = no }
	}
}

# The bill under debate is minor: at most two provisions changed, each by at
# most two index steps from existing law, and no good, relief or customs level. It needs
# te_tax_debate_days_minor days of debate instead of te_tax_debate_days_major.

	te_tax_bl_provisions <= 2

# Relief eligibility (docs/systems/tax_code_schema.md, "Civil wars and new
# countries"). State scope, from te_tax_relief_follow_owner
# (te_tax_civil_war_effects.txt), which saves first:
#   scope:te_tax_owner_new     the state's owner now
#   scope:te_tax_owner_holder  the country whose code named the state, from its
#                              te_tax_relief_holder stamp (absent: no stamp)
# True when that code went with the state, so its relief marks stay. Which test
# holds at a given moment depends on an engine order nobody has read (whether
# the transfers of an outbreak, a release or a reunification fire before or
# after the copy and the repair), so any one of them keeps the marks, as for
# the Strategic Reserve hub (st_res_hub_stays_within_nation):
#   - the new owner has no code yet: a released country or an uprising that is
#     about to receive its parent's, which keeps and restamps the marks;
#   - the new owner is the holder;
#   - one country definition: a revolution's rebels share the original's;
#   - var:te_cw_origin in either direction: the two sides of a civil war, at
#     the outbreak and at reunification (a dead loser still reads).
# No stamp and a new owner with a code is a capture: the marks go.
te_tax_relief_stays_with_code = {
	exists = scope:te_tax_owner_new
	OR = {
		scope:te_tax_owner_new = {
			OR = { NOT = { has_variable = te_tax_migrated } var:te_tax_migrated < 1 }
		}
		AND = {
			exists = scope:te_tax_owner_holder
			OR = {
				scope:te_tax_owner_new = { this = scope:te_tax_owner_holder }
				scope:te_tax_owner_new = { country_definition = scope:te_tax_owner_holder.country_definition }
				scope:te_tax_owner_new = { var:te_cw_origin ?= scope:te_tax_owner_holder }
				scope:te_tax_owner_holder = { var:te_cw_origin ?= scope:te_tax_owner_new }
			}
		}
	}
}

# State scope, a state in the enacted list te_tax_en_relief_states of
# scope:te_tax_country (te_tax_rebuild_relief_marks saves it). The list keeps
# the state by te_tax_relief_stays_with_code's rule, with the list's country
# as the holder: it is the country's own, or the country's civil-war
# counterpart holds it (one country definition, or var:te_cw_origin either
# way; inside scope:te_tax_country, PREV is the owner). So during a civil war
# both sides keep the states the other holds, and the winner's list still
# names them at reunification. A state owned by anyone else (captured, or
# released to a new country, which copies the list and keeps its own) leaves
# the list at the next rebuild; one without an owner too.
te_tax_relief_listed_state_kept = {
	exists = owner
	OR = {
		owner = scope:te_tax_country
		owner = { country_definition = scope:te_tax_country.country_definition }
		owner = { var:te_cw_origin ?= scope:te_tax_country }
		owner = { scope:te_tax_country = { var:te_cw_origin ?= prev } }
	}
}

# Policy obligations (plan Task 12; docs/systems/tax_code_schema.md, "Policy
# obligations"). Country scope. N is an obligation slot, 1 to 4
# (te_tax_o<N>_*); its payload is read only while its _on is 1, which the
# callers test first. Each kind's monthly condition reads the measure the
# promise names, never a value requested or scheduled:
#   1  an institution level: the delivered level (te_tax_obl_inst_level_<arg>,
#      institution:<x>.investment, 0 without the institution) reaches the
#      target, while the country has the institution (capability ledger row
#      28, pending P16). A bureaucracy deficit is not a condition of it: it
#      pauses the deadline (te_tax_obl_check_one);
#   2  the bureaucracy balance: no deficit (bureaucracy >= 0);
#   3  military wages: the level set now, sampled each month (no hook fires
#      on a change), at or below the promised one;
#   4  the fixed budget: a surplus (net_fixed_income > 0, vanilla's own read
#      of a healthy budget: construction is not a fixed expense, loan interest
#      is), delivered after te_tax_obl_surplus_months consecutive checks.
# ---------------------------------------------------------------------------

# Obligation slot N is in force: its bill has taken effect and it is being
# delivered (2) or maintained (3). Only such a promise can be renegotiated.
te_tax_obl_active = {
	has_variable = te_tax_o$N$_on
	var:te_tax_o$N$_on = 1
	OR = {
		var:te_tax_o$N$_state = 2
		var:te_tax_o$N$_state = 3
	}
}

# A maintained obligation N has failed as many consecutive monthly checks
# as its kind allows (te_tax_obl_grace_<kind>): it is breached. An outcome
# (kinds 2 and 4) gets a few months to recover; a level or an action (kinds 1
# and 3) none.
te_tax_obl_grace_reached = {
	trigger_if = {
		limit = { var:te_tax_o$N$_kind = 1 }
		var:te_tax_o$N$_fails >= te_tax_obl_grace_1
	}
	trigger_else_if = {
		limit = { var:te_tax_o$N$_kind = 2 }
		var:te_tax_o$N$_fails >= te_tax_obl_grace_2
	}
	trigger_else_if = {
		limit = { var:te_tax_o$N$_kind = 3 }
		var:te_tax_o$N$_fails >= te_tax_obl_grace_3
	}
	trigger_else_if = {
		limit = { var:te_tax_o$N$_kind = 4 }
		var:te_tax_o$N$_fails >= te_tax_obl_grace_4
	}
	trigger_else = { always = yes }
}

te_tax_obl_holds = {

		trigger_else = { always = no }
	}

# Offers and force-through (plan Task 13; docs/systems/tax_code_schema.md,

# Group IG's offer can be accepted: a bill is under debate, the group made an
# offer for this revision, it is not marginal and has not committed since, and
# the offer can still be carried out (te_tax_gen_offer_feasible: the bill still
# raises the tax a cut softens or adds the staple, relief has a band to add, or
# the promise is still feasible and could be recorded; false without an offer).
# No line sits inside a trigger_if that could pass for want of an offer.
te_tax_can_accept_offer = {
	custom_tooltip = { text = te_tax_tt_code_in_force te_tax_code_in_force = yes }
	trigger_if = {
		limit = { te_tax_bill_active = yes }
		custom_tooltip = {
			text = te_tax_tt_offer_made
			has_variable = te_tax_off_$IG$_kind
			has_variable = te_tax_off_$IG$_rev
			var:te_tax_off_$IG$_kind > 0
			var:te_tax_off_$IG$_rev = var:te_tax_bl_rev
		}
		custom_tooltip = {
			text = te_tax_tt_offer_not_marginal
			ig:ig_$IG$ ?= { ig_counts_as_marginal = no }
		}
		custom_tooltip = {
			text = te_tax_tt_offer_not_committed
			NAND = {
				has_variable = te_tax_com_$IG$
				has_variable = te_tax_com_$IG$_rev
				var:te_tax_com_$IG$ = 1
				var:te_tax_com_$IG$_rev = var:te_tax_bl_rev
			}
		}
		custom_tooltip = { text = te_tax_tt_offer_feasible te_tax_gen_offer_feasible = { IG = $IG$ } }
	}
	trigger_else = {
		custom_tooltip = { text = te_tax_tt_bill_open always = no }
	}
}

# A clause offer never empties the bill (plan Task 13): a cut of KEY leaves it
# changing something from existing law (the raise stays above it, or the bill
# changes another provision, a good or relief), and taking back an added staple
# leaves it changing a provision, relief or another good. Only real changes
# count, never a restatement of existing law: instruments and goods by their
# change from the baseline (te_tax_bl_changed_provisions, te_tax_bl_goods_changed),
# agricultural relief by its band change, regional relief by
# te_tax_bl_regrel_differs. Read while a bill is open.
te_tax_offer_cut_leaves_a_bill = {
	OR = {
		te_tax_bl_dstep_$KEY$ >= 2
		te_tax_bl_changed_provisions >= 2
		te_tax_bl_goods_changed >= 1
		NOT = { te_tax_bl_dstep_agrel = 0 }
		te_tax_bl_regrel_differs = yes
		# A customs level the bill changes (plan Task 15; 0 under the plain option).
		te_tax_bl_customs_changed >= 1
	}
}

te_tax_offer_untax_leaves_a_bill = {
	OR = {
		te_tax_bl_changed_provisions >= 1
		te_tax_bl_goods_changed >= 2
		NOT = { te_tax_bl_dstep_agrel = 0 }
		te_tax_bl_regrel_differs = yes
		# A customs level the bill changes (plan Task 15; 0 under the plain option).
		te_tax_bl_customs_changed >= 1
	}
}

# The bill's regional relief differs from existing law's in its month: another
# band, or at a band of 1 or more another set of named states.
te_tax_bl_regrel_differs = {
	has_variable = te_tax_bl_regrel
	var:te_tax_bl_regrel >= 0
	OR = {
		NOT = { var:te_tax_bl_regrel = te_tax_base_bl_regrel }
		AND = {
			var:te_tax_bl_regrel >= 1
			NOT = { te_tax_bl_relief_states_match = yes }
		}
	}
}

# The bill names the same regional-relief states as existing law in its month:
# an awaiting package's list that restates them before then (the last in
# commencement order), else the enacted list.
te_tax_bl_relief_states_match = {
	trigger_if = {
		limit = { te_tax_relief_base_from_a = { DUE = te_tax_bl_due } }
		te_tax_relief_lists_match = { BASE = te_tax_pa_relief_states }
	}
	trigger_else_if = {
		limit = { te_tax_relief_base_from_b = { DUE = te_tax_bl_due } }
		te_tax_relief_lists_match = { BASE = te_tax_pb_relief_states }
	}
	trigger_else = {
		te_tax_relief_lists_match = { BASE = te_tax_en_relief_states }
	}
}

# Every state on the bill's list te_tax_bl_relief_states is on the country list
# BASE, and every one on BASE that the country owns is on the bill's (the bill
# names only the country's own states, te_tax_draft_ready). A missing list is an
# empty one. Inside a list PREV is the country (te_tax_draft_ready's
# `owner = PREV`); inside owner = { } from there, PREV is the state (the
# te_tax_view_dr_relief_state form). A bill state another country has taken
# since counts as a difference.
te_tax_relief_lists_match = {
	NAND = {
		has_variable_list = te_tax_bl_relief_states
		any_in_list = {
			variable = te_tax_bl_relief_states
			NAND = {
				exists = owner
				owner = PREV
				owner = {
					has_variable_list = $BASE$
					is_target_in_variable_list = { name = $BASE$ target = PREV }
				}
			}
		}
	}
	NAND = {
		has_variable_list = $BASE$
		any_in_list = {
			variable = $BASE$
			exists = owner
			owner = PREV
			NOT = {
				owner = {
					has_variable_list = te_tax_bl_relief_states
					is_target_in_variable_list = { name = te_tax_bl_relief_states target = PREV }
				}
			}
		}
	}
}

# Group IG's promise accepted under this bill (te_tax_bl_prom_<IG>_*) is a live
# pending obligation in slot N: the same kind and arg, made to IG, for the
# bill's current revision. A promise counts in the promises reason only while
# it is live (te_tax_prom_<ig>), so one that could not be recorded, because
# the same promise is already pending or binding, adds nothing.
te_tax_offer_prom_in_slot = {
	has_variable = te_tax_o$N$_on
	has_variable = te_tax_o$N$_kind
	has_variable = te_tax_o$N$_rev
	var:te_tax_o$N$_on = 1
	var:te_tax_o$N$_state = 1
	var:te_tax_o$N$_kind = var:te_tax_bl_prom_$IG$_kind
	var:te_tax_o$N$_arg = var:te_tax_bl_prom_$IG$_arg
	var:te_tax_o$N$_ig = te_tax_ig_id_$IG$
	var:te_tax_o$N$_rev = var:te_tax_bl_rev
}

# ...and only maintains what already held when it was proposed (_maint_only,
# recorded by te_tax_obl_write with the test te_tax_obl_is_maintenance makes).
te_tax_offer_prom_maint_in_slot = {
	te_tax_offer_prom_in_slot = { IG = $IG$ N = $N$ }
	has_variable = te_tax_o$N$_maint_only
	var:te_tax_o$N$_maint_only = 1
}

te_tax_offer_prom_live = {
	has_variable = te_tax_bl_rev
	has_variable = te_tax_bl_prom_$IG$_kind
	var:te_tax_bl_prom_$IG$_kind > 0
	OR = {

		te_tax_offer_prom_in_slot = { IG = $IG$ N = 2 }

	var:te_tax_bl_prom_$IG$_kind > 0
	OR = {

# Force through: a bill short of the majority passage needs, with at least
# te_tax_force_share of the clout committed, an override capacity of 2 (the
# forced-law event's medium tier, read as the raw modifier sum) and the
# Authority the override costs, and every other passage condition of
# te_tax_can_pass, repeated line for line.
te_tax_can_force_through = {
	custom_tooltip = { text = te_tax_tt_code_in_force te_tax_code_in_force = yes }
	trigger_if = {
		limit = { te_tax_bill_active = yes }
		custom_tooltip = { text = te_tax_tt_force_short NOT = { te_tax_committed_share > te_tax_passage_share } }
		custom_tooltip = { text = te_tax_tt_force_share te_tax_committed_share >= te_tax_force_share }
		custom_tooltip = { text = te_tax_tt_force_capacity "modifier:country_legislative_override_capacity_add" >= 2 }
		custom_tooltip = { text = te_tax_tt_force_authority produced_authority > forced_law_through_event_authority_cost_medium }
		custom_tooltip = { text = te_tax_tt_pass_legitimacy legitimacy >= te_tax_passage_legitimacy }
		custom_tooltip = { text = te_tax_tt_pass_debate te_tax_debate_days_left <= 0 }
		custom_tooltip = { text = te_tax_tt_pass_due var:te_tax_bl_due > te_history_month_index }
		custom_tooltip = { text = te_tax_tt_pass_slot OR = { var:te_tax_pa_on = 0 var:te_tax_pb_on = 0 } }
		custom_tooltip = { text = te_tax_tt_pass_current te_tax_gen_bill_current = yes }
		custom_tooltip = { text = te_tax_tt_pass_sunsets te_tax_gen_bill_sunsets_valid = yes }
		custom_tooltip = { text = te_tax_tt_pass_changes te_tax_bill_changes_something = yes }
		custom_tooltip = {
			text = te_tax_tt_pass_store
			OR = {
				te_tax_can_store_package = { SLOT = a }
				te_tax_can_store_package = { SLOT = b }
			}
		}
	}
	trigger_else = {
		custom_tooltip = { text = te_tax_tt_bill_open always = no }
	}
}
