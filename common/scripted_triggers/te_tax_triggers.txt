# Legislated tax code: game-rule gates (common/game_rules/extra_game_rules.txt,
# te_tax_code_rule).
#
# Both read only a game rule, so they work from any scope. Every hook, pulse,
# event, effect entry point, GUI root and AI path of the tax code asks
# te_tax_code_on; customs legislation asks te_tax_customs_on as well.
#
# They are written positively on purpose. A save from before the rule existed
# has no value stored for it, so NOT = { has_game_rule = te_tax_code_disabled }
# would be true there and migrate the campaign. Do not rewrite these as
# negations, and do not add a negated check anywhere else.

# True under both enabled settings.
te_tax_code_on = {
	OR = {
		has_game_rule = te_tax_code_enabled
		has_game_rule = te_tax_code_enabled_customs
	}
}

# True only under the setting that also legislates customs.
te_tax_customs_on = {
	has_game_rule = te_tax_code_enabled_customs
}

# Country scope. Slot SLOT (a or b) may take a package from the bill record:
# the rule is on, the country holds the code's tokens, the slot is free, the
# bill has a due month after the current one, and every sunset in the bill
# falls at least one month after commencement (te_tax_gen_bill_sunsets_valid:
# the store writes _exp = due + offset, so _exp >= _due + 1 exactly when the
# offset is at least 1). te_tax_store_package checks it; Task 6's passage
# trigger repeats the sunset and slot checks for the player's tooltip.
te_tax_can_store_package = {
	te_tax_code_on = yes
	has_variable = te_tax_schema
	var:te_tax_p$SLOT$_on = 0
	has_variable = te_tax_bl_due
	var:te_tax_bl_due > te_history_month_index
	te_tax_gen_bill_sunsets_valid = yes
}

# ---------------------------------------------------------------------------
# Draft, bill and passage (docs/systems/tax_code_schema.md, "Draft, bill and
# passage"). Country scope. Each command te_tax_cmd_<c>
# (te_tax_bill_effects.txt) runs only when te_tax_can_<c> holds; a GUI button
# uses the same trigger as its is_valid, so every condition is a custom_tooltip
# line and the disabled tooltip lists each one that fails. A condition that
# reads a record's payload sits behind trigger_if on the record being open, so
# a closed record (payload removed) is never read.
# ---------------------------------------------------------------------------

# The code is in force: the rule is on and the country's code exists.
te_tax_code_in_force = {
	te_tax_code_on = yes
	has_variable = te_tax_schema
	has_variable = te_tax_migrated
	var:te_tax_migrated >= 1
}

# The journal entry (je_tax_code's `possible`) and the Budget panel's Tax Code
# tab (te_budget_tax_tab_unlock_sgui's `is_valid`) open once the code is in
# force, so the greyed tab's checklist and the entry cannot disagree. One
# custom_tooltip line: the checklist is IsValidTooltip, which would print the
# conditions inside te_tax_code_in_force without a heading of their own.
te_tax_entry_unlocked = {
	custom_tooltip = {
		text = te_tax_tt_code_in_force
		te_tax_code_in_force = yes
	}
}

# A draft is open (its payload exists exactly while it is).
te_tax_draft_active = {
	has_variable = te_tax_dr_on
	var:te_tax_dr_on = 1
	has_variable = te_tax_dr_due
}

# A bill is under debate (its payload exists exactly while it is).
te_tax_bill_active = {
	has_variable = te_tax_bl_on
	var:te_tax_bl_on = 1
	has_variable = te_tax_bl_due
}

# Slot SLOT (a or b) holds a package awaiting commencement in a month before
# the one in the variable DUE (te_tax_dr_due or te_tax_bl_due). For the
# baselines in te_tax_generated_values.txt.
te_tax_slot_awaits_before = {
	has_variable = te_tax_p$SLOT$_on
	var:te_tax_p$SLOT$_on = 1
	var:te_tax_p$SLOT$_state = 1
	has_variable = $DUE$
	var:te_tax_p$SLOT$_due < var:$DUE$
}

# Slot b's package commences before slot a's: an earlier month, or the same
# month and a lower approval sequence (the scheduler's order).
te_tax_slot_b_first = {
	has_variable = te_tax_pa_due
	has_variable = te_tax_pb_due
	has_variable = te_tax_pa_seq
	has_variable = te_tax_pb_seq
	OR = {
		var:te_tax_pb_due < var:te_tax_pa_due
		AND = {
			var:te_tax_pb_due = var:te_tax_pa_due
			var:te_tax_pb_seq < var:te_tax_pa_seq
		}
	}
}

# KEY: wage, div, land, head or cons. DIR: 0 down, 1 up, 2 zero, 3 highest,
# 4 leave the provision alone.
te_tax_can_draft_step = {
	custom_tooltip = { text = te_tax_tt_code_in_force te_tax_code_in_force = yes }
	trigger_if = {
		limit = {
			te_tax_draft_active = yes
			has_variable = te_tax_dr_$KEY$
		}
		te_tax_dr_step_ok_$DIR$ = { KEY = $KEY$ }
	}
	trigger_else = {
		custom_tooltip = { text = te_tax_tt_draft_open always = no }
	}
}

# The draft can become the bill: it changes something, commences within range,
# and every provision it changes is still on the baseline it was drafted on.
te_tax_draft_ready = {
	trigger_if = {
		limit = { te_tax_draft_active = yes }
		custom_tooltip = { text = te_tax_tt_draft_changes te_tax_gen_draft_touches_any = yes }
		custom_tooltip = {
			text = te_tax_tt_draft_due_range
			var:te_tax_dr_due >= te_tax_due_earliest
			var:te_tax_dr_due <= te_tax_due_latest
		}
		custom_tooltip = { text = te_tax_tt_draft_baseline te_tax_gen_draft_baseline_current = yes }
		# Traditionalism disallows the vanilla laws that tax wages and dividends
		# (Per-Capita, Proportional, Graduated); under it a bill may not either.
		# Here, so a revision is held to it as well as an introduction.
		trigger_if = {
			limit = { has_law = law_type:law_traditionalism }
			custom_tooltip = {
				text = te_tax_tt_draft_traditionalism
				te_tax_dr_eff_wage < 1
				te_tax_dr_eff_div < 1
			}
		}
	}
	trigger_else = {
		custom_tooltip = { text = te_tax_tt_draft_open always = no }
	}
}

# The bill under debate is minor: at most two provisions changed, each by at
# most two index steps from existing law, and no good or relief. It needs
# te_tax_debate_days_minor days of debate instead of te_tax_debate_days_major.
te_tax_bill_is_minor = {
	te_tax_bl_provisions <= 2
	te_tax_gen_bill_small_steps = yes
	NOT = { te_tax_gen_bill_touches_goods = yes }
	var:te_tax_bl_agrel < 0
	var:te_tax_bl_regrel < 0
}

# The open draft would make a minor bill: te_tax_bill_is_minor's rule on the
# draft's fields, so the workbench shows the class the bill would get
# (test_tax_code_sguis.py checks the two read the same). Read only while a draft
# is open (te_tax_draft_active), as it reads the draft's payload.
te_tax_draft_is_minor = {
	te_tax_dr_provisions <= 2
	te_tax_gen_draft_small_steps = yes
	NOT = { te_tax_gen_draft_touches_goods = yes }
	var:te_tax_dr_agrel < 0
	var:te_tax_dr_regrel < 0
}

# Passage under the fallback rule. The last line repeats the scheduler's own
# check (te_tax_can_store_package) so the trigger and the store cannot drift
# apart; it fails only together with a line above it.
te_tax_can_pass = {
	custom_tooltip = { text = te_tax_tt_code_in_force te_tax_code_in_force = yes }
	trigger_if = {
		limit = { te_tax_bill_active = yes }
		custom_tooltip = { text = te_tax_tt_pass_support te_tax_committed_share > te_tax_passage_share }
		custom_tooltip = { text = te_tax_tt_pass_legitimacy legitimacy >= te_tax_passage_legitimacy }
		custom_tooltip = { text = te_tax_tt_pass_debate te_tax_debate_days_left <= 0 }
		custom_tooltip = { text = te_tax_tt_pass_due var:te_tax_bl_due > te_history_month_index }
		custom_tooltip = { text = te_tax_tt_pass_slot OR = { var:te_tax_pa_on = 0 var:te_tax_pb_on = 0 } }
		custom_tooltip = { text = te_tax_tt_pass_current te_tax_gen_bill_current = yes }
		custom_tooltip = { text = te_tax_tt_pass_sunsets te_tax_gen_bill_sunsets_valid = yes }
		custom_tooltip = {
			text = te_tax_tt_pass_store
			OR = {
				te_tax_can_store_package = { SLOT = a }
				te_tax_can_store_package = { SLOT = b }
			}
		}
	}
	trigger_else = {
		custom_tooltip = { text = te_tax_tt_bill_open always = no }
	}
}

# SLOT: a or b. A package the scheduler held as missed (state 3: its due month
# passed without commencement) may commence next month instead, if that is at
# most te_tax_reschedule_max_delay months after the month it was approved for
# (te_tax_due_earliest - due0 <= 3, written due0 >= te_tax_due_earliest - 3)
# and no later bill has touched what it changes. A longer delay, or a package
# held as conflicting (state 2), changes the bargain: it can only be released
# and passed again as a new bill.
te_tax_can_package_reschedule = {
	custom_tooltip = { text = te_tax_tt_code_in_force te_tax_code_in_force = yes }
	custom_tooltip = {
		text = te_tax_tt_package_missed
		has_variable = te_tax_p$SLOT$_on
		var:te_tax_p$SLOT$_on = 1
		var:te_tax_p$SLOT$_state = 3
	}
	# The package's payload exists only while the slot is on.
	trigger_if = {
		limit = {
			has_variable = te_tax_p$SLOT$_on
			var:te_tax_p$SLOT$_on = 1
		}
		custom_tooltip = {
			text = te_tax_tt_package_delay
			has_variable = te_tax_p$SLOT$_due0
			var:te_tax_p$SLOT$_due0 >= te_tax_package_reschedule_floor
		}
		custom_tooltip = { text = te_tax_tt_package_unsuperseded te_tax_gen_package_unsuperseded_$SLOT$ = yes }
	}
}

# SLOT: a or b. A held package (state 2 conflicting or 3 missed) may be

te_tax_can_package_release = {

# Vanilla objectives (docs/systems/tax_code_schema.md, "Native controls and
# drift"). Country scope. Under the rule every country holds law_te_tax_code,
# so an objective that needs Per-Capita, Proportional or Graduated taxation
# could never be met; each of these says the code collects what that law
# would. The three vanilla journal entries that need one
# (te_tax_vanilla_je_overrides.txt) accept them in a custom_tooltip; the AI can
# read them too. Only the rule and the carrier gate them: a country still on a
# vanilla law answers with its law.
# ---------------------------------------------------------------------------

# Per-Capita: a wage tax and a head tax, as the law levies both.
te_tax_code_counts_as_per_capita = {
	te_tax_code_on = yes
	has_law = law_type:law_te_tax_code
	te_tax_view_en_head >= 1
	te_tax_view_en_wage >= 1
}

# Proportional: a wage tax of at least 10% and a dividend tax (the law's
# lowest level taxes wages at 10% and dividends at 2.5%).
te_tax_code_counts_as_proportional = {
	te_tax_code_on = yes
	has_law = law_type:law_te_tax_code
	te_tax_view_en_wage_rate >= 0.1
	te_tax_view_en_div_rate >= 0.025
}

# Graduated: a wage tax of at least 10% and dividends taxed at least as
# heavily as wages (the law's levels tax dividends at or above wages).
te_tax_code_counts_as_graduated = {
	te_tax_code_on = yes
	has_law = law_type:law_te_tax_code
	te_tax_view_en_wage_rate >= 0.1
	te_tax_view_en_div_rate >= te_tax_view_en_wage_rate
}

# Relief eligibility (docs/systems/tax_code_schema.md, "Civil wars and new
# countries"). State scope, from te_tax_relief_follow_owner
# (te_tax_civil_war_effects.txt), which saves first:
#   scope:te_tax_owner_new     the state's owner now
#   scope:te_tax_owner_holder  the country whose code named the state, from its
#                              te_tax_relief_holder stamp (absent: no stamp)
# True when that code went with the state, so its relief marks stay. Which test
# holds at a given moment depends on an engine order nobody has read (whether
# the transfers of an outbreak, a release or a reunification fire before or
# after the copy and the repair), so any one of them keeps the marks, as for
# the Strategic Reserve hub (st_res_hub_stays_within_nation):
#   - the new owner has no code yet: a released country or an uprising that is
#     about to receive its parent's, which keeps and restamps the marks;
#   - the new owner is the holder;
#   - one country definition: a revolution's rebels share the original's;
#   - var:te_cw_origin in either direction: the two sides of a civil war, at
#     the outbreak and at reunification (a dead loser still reads).
# No stamp and a new owner with a code is a capture: the marks go.
te_tax_relief_stays_with_code = {
	exists = scope:te_tax_owner_new
	OR = {
		scope:te_tax_owner_new = {
			OR = { NOT = { has_variable = te_tax_migrated } var:te_tax_migrated < 1 }
		}
		AND = {
			exists = scope:te_tax_owner_holder
			OR = {
				scope:te_tax_owner_new = { this = scope:te_tax_owner_holder }
				scope:te_tax_owner_new = { country_definition = scope:te_tax_owner_holder.country_definition }
				scope:te_tax_owner_new = { var:te_cw_origin ?= scope:te_tax_owner_holder }
				scope:te_tax_owner_holder = { var:te_cw_origin ?= scope:te_tax_owner_new }
			}
		}
	}
}
