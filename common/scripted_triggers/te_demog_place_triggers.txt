# DEMOGRAPHICS — where people live (phase 3): triggers for the urban pattern's effects and the Planned Capital decision

# State scope: its owner's largest city at the last census (te_dg_primate_state, te_demog_up_country), so the state the
# Primate City modifier goes to. A state that has changed hands since reads no.
te_demog_up_is_primate = {
	exists = owner
	owner = { has_variable = te_dg_primate_state }
	this = owner.var:te_dg_primate_state
}

# State scope: a site the Planned Capital decision may offer (te_demog_capital.1). The base game's own rules for a
# capital (its Move Capital map interaction: incorporated, a homeland of one of the owner's primary cultures), not the
# capital already, not the largest city (a planned capital is built away from it) and at least 1% of the owner's people.
te_demog_pc_candidate = {
	exists = owner
	is_incorporated = yes
	is_capital = no
	is_homeland_of_country_cultures = owner
	te_demog_has_census = yes
	NOT = { te_demog_up_is_primate = yes }
	state_population >= te_demog_pc_min_people
}

# Country scope: the decision's AI rule, the same conditions for every country (spec §11.2; no is_ai). A Primate pattern
# at half strength or more in a country of eight census states or more, out of default, with the money in the treasury
# and the authority to spare.
te_demog_pc_ai_wants = {
	has_variable = te_dg_up_s
	var:te_dg_up_s >= te_demog_k_pc_ai_strength
	has_variable = te_dg_up_states
	var:te_dg_up_states >= te_demog_k_pc_ai_states
	in_default = no
	gold_reserves > te_demog_pc_money_shown
	authority > te_demog_pc_ai_authority
}
