# Heir Education: which character ideologies a country allows its heir.
#
# heir_education_resolve_effect rolls the heir's ideology from weighted lists,
# and set_ideology does not check an ideology's country_trigger (vanilla's
# suffragist_events.8 sets ideology_reformer inside je_suffragists, which
# needs the feminism that Reformer's country_trigger forbids). Each roll whose
# ideology has a country_trigger is gated on the matching trigger below, so a
# roll never lands on an ideology the country cannot have.
#
# Each body is a copy of that ideology's country_trigger: from the REPLACE
# block in common/ideologies/modified.txt (generated by apply_ideologies.py),
# or from vanilla where the mod does not replace it. Country scope; `this` is
# the country. test_heir_education_ideologies.py fails if a copy drifts from
# its source or a gated roll is missing its gate.

heir_ed_country_allows_ideology_reformer = {
	has_technology_researched = egalitarianism
	NOT = {
		has_technology_researched = feminism
	}
}

heir_ed_country_allows_ideology_liberal_leader = {
	has_technology_researched = democracy
}

# The multi-child NOT below is vanilla's, kept verbatim so it means whatever
# the source means (scripting_best_practices.md, "Write NOR or NAND").
heir_ed_country_allows_ideology_radical = {
	has_technology_researched = egalitarianism
	# Invalid for pre-ACW USA whilst Jacksonian Democrats are relevant
	NOT = {
		AND = {
			exists = c:USA
			OR = {
				c:USA = this
				civil_war_origin_country ?= c:USA
			}
		}
		AND = {
			has_journal_entry = je_acw_countdown
			any_scope_character = {
				has_ideology = ideology:ideology_jacksonian_democrat
			}
		}
	}
}

heir_ed_country_allows_ideology_social_democrat = {
	has_technology_researched = socialism
}

heir_ed_country_allows_ideology_communist = {
	has_technology_researched = socialism
}

heir_ed_country_allows_ideology_vanguardist = {
	has_technology_researched = political_agitation
}

heir_ed_country_allows_ideology_humanitarian = {
	has_technology_researched = feminism
}

heir_ed_country_allows_ideology_royalist = {
	# no monarchists in countries with strong republican tradition
	# no generic monarchists in France
	NOR = {
		c:FRA ?= this
		AND = {
			any_interest_group = {
				has_ideology = ideology:ideology_republican_paternalistic
			}
			NOT = {
				has_variable = brz_enduring_empire
			}
		}
		has_variable = monarchy_stamped_out_var
	}
}

heir_ed_country_allows_ideology_traditionalist = {
	NOT = {
		c:FRA ?= this
	}
}

heir_ed_country_allows_ideology_ethno_nationalist = {
	has_technology_researched = pan-nationalism
}
