# Construction Cost Scaling's rule gates. Values: common/script_values/extra_script_values.txt (CONSTRUCTION COST
# SCALING). Effects: common/scripted_effects/te_construction_cost_effects.txt.
#
# te_small_economy_construction_rule has three strengths and an off setting. Every read of it is one of these
# triggers; nothing else names the rule's settings. Each reads only a game rule, so it works from any scope. All are
# named positively: a save from before the rule (no setting stored) reads as off.

# The rule is on, at any strength.
te_small_economy_construction_on = {
	OR = {
		has_game_rule = te_small_economy_construction_mild
		has_game_rule = te_small_economy_construction_standard
		has_game_rule = te_small_economy_construction_enabled
	}
}

# Mild: half the price with almost no GDP.
te_small_economy_construction_mild_on = {
	has_game_rule = te_small_economy_construction_mild
}

# Standard: a fifth of the price with almost no GDP.
te_small_economy_construction_standard_on = {
	has_game_rule = te_small_economy_construction_standard
}

# Strong: a tenth of the price with almost no GDP. The setting is named enabled because it was the rule's only "on"
# setting until 2026-11-16; the name is kept so a preset or a save made with it still means a tenth.
te_small_economy_construction_strong_on = {
	has_game_rule = te_small_economy_construction_enabled
}
