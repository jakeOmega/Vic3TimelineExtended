# Country scope. gm_policy is a flag (open, ceremonial, culture, education);
# a country that never chose one reads as the default, Cultural Promotion
# (STANDARD = yes). An older save's Mothballed or Standard reads as the default
# too until gm_policy_migrate drops it on the next refresh, so its button is
# never valid for nothing (policy spec 2026-10-23 § 6).
gm_policy_is = {
	OR = {
		AND = {
			has_variable = gm_policy
			var:gm_policy = flag:$P$
		}
		AND = {
			always = $STANDARD$
			OR = {
				NOT = { has_variable = gm_policy }
				var:gm_policy = flag:standard
				var:gm_policy = flag:mothballed
			}
		}
	}
}

# Country scope. The policy may change now: no change in the last 60 months
# (the first choice starts no wait; gm_policy_set).
gm_policy_can_change = {
	NOT = { has_variable = gm_policy_cooldown }
}
