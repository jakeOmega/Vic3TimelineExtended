# Banking policy dashboard — shared eligibility triggers.
#
# Each `banking_possible_<button>` holds the body that used to live inline in the
# matching scripted_button's `possible` block. Both the journal-entry button
# (common/scripted_buttons/) and the dashboard scripted GUI
# (common/scripted_guis/banking_dashboard_scripted_gui.txt) call these, so player
# clicks through either surface are gated by exactly the same rules.
# Disable-side eligibility reuses the `banking_tool_*_active` family in
# common/scripted_triggers/market_triggers.txt.

# Is the Banking Cycle system switched on at all? TRUE for both of the settings
# that keep the journal entry — `banking_system_enabled` and the simplified one
# — and false only for `banking_system_disabled`
# (common/game_rules/extra_game_rules.txt).
#
# THIS IS THE CYCLE'S GATE; te_mon_full_system
# (common/scripted_triggers/te_monetary_triggers.txt) is the MONETARY LAYER'S.
# The two are deliberately different questions: the simplified setting answers
# yes here and no there. Anything that belongs to the cycle — the journal entry,
# its tools, its history series — asks this one.
#
# Reads only a game rule, so it works from any scope.
te_banking_system_on = {
	OR = {
		has_game_rule = banking_system_enabled
		has_game_rule = banking_system_simplified
	}
}

# The "Raise Policy Rate" eligibility trigger used to sit above this one, and
# OMO used to exclude itself while that tool was active. Both are gone with the
# tool (monetary_policy_design.md §18).

# Open market operations are FIAT AND DIGITAL ONLY, and within those two, THE
# FLOOR TOOL (monetary_policy_design.md §5.1 and §11, and the §0.1 ruling of
# 2026-09-20). Two separate rules, and the tooltip names them one at a time:
#
#   1. REGIME. Creating money to buy bonds is precisely what a promise to redeem
#      the currency in metal forbids — the reserves walk out of the door. So
#      commodity money, the gold standard and crypto can never run OMO, at any
#      rate. The gold standard's own crisis tool is suspending convertibility,
#      and that arrives in phase 3 (§12.3). The gate is
#      country_can_create_unbacked_money_bool, granted by law_fiat_currency and
#      law_digital_currency in common/laws/extra_laws.txt — a declarative
#      modifier, so both laws advertise "Enables Open-Market Operations" on
#      their own tooltip instead of the rule living only in this file.
#
#   2. THE FLOOR, for the two regimes that pass rule 1. A central bank that can
#      still cut does not buy assets: cutting is cheaper, it is reversible, and
#      it does the same job. The floor is the regime's own minimum, not a flat
#      zero — 0 under fiat, -3 under digital, which has no cash to hoard and so
#      three more points of cutting room. te_mon_floor_threshold derives both
#      from country_policy_rate_floor_add rather than restating them.
#
# The floor line is wrapped in a trigger_if so a gold-standard player is told
# about convertibility and is NOT also shown a red line about a fiat floor that
# could never have applied. te_mon_has_dial stays in that branch because fiat
# plus a command economy, or fiat after the national bank is repealed, is a
# country with the regime and no dial to be at the bottom of (§4, §8).
#
# SCOPE. This runs in country scope from both surfaces — the scripted_button's
# `possible` and the dashboard scripted GUI's `is_valid` (scope = country) — which
# is why the points and lock checks read country modifiers directly.
# var:te_policy_rate resolves against the same country.
#
# GUARDED on has_variable: te_policy_rate does not exist until a country's first
# monthly monetary update, and an ungated `var:` read would resolve to 'none'.
# Before that first pulse the tool is simply unavailable, which is the safe
# answer. The guard lives inside te_mon_policy_rate_at_floor
# (common/scripted_triggers/te_monetary_triggers.txt).
banking_possible_cb_open_market_ops = {
	modifier:country_can_use_open_market_ops_bool = yes
	custom_tooltip = {
		text = BANKING_POINTS_4
		modifier:country_banking_intervention_max_add >= 4
	}
	custom_tooltip = {
		text = banking_button_locked_by_law_tt
		modifier:country_banking_lock_open_market_ops_bool = no
	}
	# BOTH OF THE MONETARY CONDITIONS BELONG TO THE MONETARY LAYER, so the
	# simplified setting of banking_system_rule drops them and open-market
	# operations revert to what they were before phase 1: the unlock bool, four
	# intervention points and the law lock, and nothing about the regime or the
	# rate. There is no policy rate to be at the
	# floor of when nobody holds a dial, so leaving the floor test in would lock
	# the tool out of a simplified game entirely.
	trigger_if = {
		limit = { te_mon_full_system = yes }
		custom_tooltip = {
			text = banking_omo_unbacked_money_tt
			modifier:country_can_create_unbacked_money_bool = yes
		}
		trigger_if = {
			limit = { modifier:country_can_create_unbacked_money_bool = yes }
			custom_tooltip = {
				text = banking_omo_rate_floor_tt
				te_mon_has_dial = yes
				te_mon_policy_rate_at_floor = yes
			}
		}
	}
}

# AI: "is there a cycle reason to reach for this tool at all?"
#
# Each cb_* enable button's ai_chance is a sum of CORE terms (the cycle state
# the tool answers), FLAVOUR terms (the financial-regulation law, an independent
# bank, a companion tool) and RESOURCE terms (how many intervention points are
# free). The flavour and resource terms are gated on the matching trigger
# below, because as unconditional adds on a base of 0 they scored a tool 5-25
# in a STABLE phase with no cycle reason at all, and the AI clicked it: a
# universal-banking AI parked the capital buffer and margin requirements for
# 40% of the century, which is what made two intervention points the safest
# budget in the game and three the most dangerous
# (docs/audits/banking_cycle_simulation.md F10). With the gate the budget curve
# is flat and the AI holds about half as many tools.
#
# Each trigger restates the conditions of its button's core terms; keep the
# two in step when a core term changes. The disable buttons are not gated:
# lifting a tool needs no reason beyond the phase it is lifted in.
# ----------------------------------------------------------------------------
banking_ai_core_cb_open_market_ops = {
	OR = {
		banking_cycle_is_panic = yes
		banking_cycle_is_downturn = yes
		banking_cycle_is_stagnation = yes
		banking_cycle_risk_falling = yes
		AND = {
			has_variable = te_inflation
			var:te_inflation < te_mon_band_edge_deflation
		}
	}
}

banking_ai_core_cb_countercyclical_buffer = {

		banking_cycle_is_expansion = yes

}

# AI: which directed-credit sector?
#
# Every directed-credit enable button scores the same weight,
# banking_dc_ai_weight — Infrastructure's ai_chance as it stood before the four
# new sectors, gated by banking_ai_core_cb_directed_credit_infrastructure — and
# the weight is SPLIT among the sectors the government favours: a new sector is
# a candidate when it has an affinity below and could be clicked right now; with
# n candidates each scores weight / n and Infrastructure scores 0, and with none
# Infrastructure scores the whole weight. So the AI reaches for directed credit
# exactly as often as it did with one sector (docs/audits/banking_cycle_simulation.md
# F11: it is the AI's riskiest tool), and picks the sector that suits whoever
# governs. The one deliberate increase is the Directed Credit law's second slot.
# ----------------------------------------------------------------------------
banking_dc_affinity_heavy_industry = {
	ig:ig_industrialists ?= { is_in_government = yes }
}

		ig:ig_landowners ?= { is_in_government = yes }

}

# Legacy cycle events that must answer to the dashboard (#428)
# ----------------------------------------------------------------------------
# banking_cycle_random_event_effect (common/scripted_effects/banking_cycle_effects.txt)
# draws events that offer a policy the dashboard owns, or narrate monetary
# state. Each random_list entry asks the triggers below in its `trigger`, so an
# event can never offer what the dashboard's own button would refuse, nor tell
# a story the system's state contradicts. Precedent: events 15 and 58, which
# defer to the monetary layer under te_mon_full_system (PR #416).

# banking_cycle_events.31 (the bank holiday). Option A enacts the dashboard's
# cb_bank_holiday through banking_effect_cb_bank_holiday, so the event is
# offered only when that button could be pressed: the holiday is not already
# running (the button's `visible`), and banking_possible_cb_bank_holiday holds
# whole — a downturn or panic, no five-year cooldown, and the two intervention
# points. The points line stays in because banking_bank_holiday itself carries
# country_banking_intervention_max_add = -2: the event cannot enact the tool
# without holding the points for its 90 days, and a country short of them would
# be driven below zero and locked out of every other tool until it expired.
# Option A asks again when it is picked, because the event stays open for months.
banking_can_enact_cb_bank_holiday = {
	banking_tool_bank_holiday_active = no
	banking_possible_cb_bank_holiday = yes
}

# banking_cycle_events.39 (the quantitative easing debate). Buying bonds with
# new money is open-market operations, so it never fires while OMO is already
# running. Under the full monetary system option A enacts cb_open_market_ops,
# and the debate needs that button's whole gate — fiat or digital money, a
# national bank with a dial, the policy rate at its floor, the unlock, four
# points (held by banking_open_market_ops' own modifier) and no law lock. Under
# the simplified rule there is no dial and no floor; the event keeps its old
# law and phase gates and its bespoke stimulus.
banking_qe_debate_possible = {
	banking_tool_omo_active = no
	trigger_if = {
		limit = { te_mon_full_system = yes }
		banking_possible_cb_open_market_ops = yes
	}
}

# banking_cycle_events.34 (the deposit insurance debate). Is a deposit guarantee
# already in force? The dashboard's Expanded Deposit Guarantee; a power-bloc
# banking union's common guarantee (tier 4 and up, on the currency's adopters —
# the te_mon_union_banking_union managed modifier, which only exists under the
# full monetary system); or the scheme a previous debate set up.
banking_deposit_guarantee_in_force = {
	OR = {
		banking_tool_deposit_active = yes
		te_mon_mod_has = { NAME = te_mon_union_banking_union }
		has_modifier = banking_event_deposit_insurance
	}
}

}

# banking_cycle_events.16 (market branch: the central bank governor pushes back
# on politicians who want cheaper money). There has to be a central bank whose
# rate is not the government's to set. Under the full monetary system that is a
# dial (te_mon_has_dial: a national bank and a regime with a rate) that the
# government has handed over — delegated from the dashboard, or bound by central
# bank independence (te_mon_mandate_binds). Under the simplified rule nobody
# holds a dial, so a national bank is all the event can ask for.
banking_cbi_debate_has_governor = {
	has_law_or_variant = law_type:law_national_bank
	trigger_if = {
		limit = { te_mon_full_system = yes }
		te_mon_has_dial = yes
		te_mon_mandate_binds = yes
	}
}

# banking_cycle_events.10 (the sovereign debt scare) and its planned and
# cooperative variants .110 / .160. Bondholders only take fright at a debt worth
# pricing: scaled_debt above 0.25 is where the rate stack's debt-load premium
# starts (te_mon_scaled_debt_premium, monetary_policy_design.md §7.6), under both
# the full and the simplified rule. A country already in default has no credit
# left to lose, which is what the scare threatens.
banking_sovereign_debt_is_priced = {
	scaled_debt > 0.25
	in_default = no
}

# Was this country reached by the current crisis wave — hit by contagion, or the
# crash that started it? banking_crisis_wave_hit marks every country the wave
# reached, whether its banks crashed or only took a scare;
# banking_crisis_wave_crashed marks only the ones that crashed (the Great
# Depression event, minor_events_timelineextended.8, reads that one). A wave in
# flight in a save from before the split has only _crashed on every country it
# reached, so this reads both.
banking_crisis_wave_reached = {
	OR = {
		has_variable = banking_crisis_wave_hit
		has_variable = banking_crisis_wave_crashed
	}
}
