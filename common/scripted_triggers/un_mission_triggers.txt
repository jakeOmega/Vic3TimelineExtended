# The scoped state already hosts an active mission (one per state at a time).
# Scope: state
un_mission_state_occupied = {
	var:un_msn_here ?= { has_tag = un_msn_active }
}

# The scoped country hosts an active peacekeeping or stabilisation mission: it
# is under UN protection, which the war-goal surcharge doubles from Strong
# (un_teeth_war_goal_surcharge).
# Scope: country
un_mission_protected = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		has_tag = un_msn_active
		OR = {
			has_tag = un_msn_peacekeeping
			has_tag = un_msn_stabilisation
		}
		var:un_msn_host ?= PREV
	}
}

# The scoped country contributes to an active mission hosted by
# scope:un_msn_intel_host: its covert networks there grow faster
# (covert_nets_sync; §7.4).
# Scope: country (the operator); expects scope:un_msn_intel_host
un_mission_contributes_in = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		has_tag = un_msn_active
		var:un_msn_host ?= scope:un_msn_intel_host
		is_target_in_variable_list = { name = un_msn_contributors target = PREV }
	}
}

# The scoped country hosts an active mission of $TYPE$ (peacekeeping | aid |
# stabilisation). A country hosting one cannot ask the Assembly for another.
# Scope: country
un_mission_hosts = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		has_tag = un_msn_active
		has_tag = un_msn_$TYPE$
		var:un_msn_host ?= PREV
	}
}

# A mission of $TYPE$ can be found or opened for the scoped host: it already
# hosts one, or it holds a state with no mission (un_mission_join_host). For
# the tooltips, which are shown before the effect runs.
# Scope: country (the host)
un_mission_can_join_host = {
	OR = {
		un_mission_hosts = { TYPE = $TYPE$ }
		any_scope_state = { NOT = { un_mission_state_occupied = yes } }
	}
}

# The host is at war, and not as the initiator of the play that started it.
# Scope: country (the host)
un_mission_host_attacked = {
	is_at_war = yes
	is_diplomatic_play_initiator = no
}

# The mission's host has been attacked since the mission opened: it is at war
# and not the initiator (un_mission_host_attacked), and the mission did not
# open into that war (un_msn_war_at_open, cleared by the host's first month
# of peace).
# Scope: mission container; expects scope:un_msn_host_now
un_mission_attacked_since_open = {
	NOT = { has_variable = un_msn_war_at_open }
	scope:un_msn_host_now = { un_mission_host_attacked = yes }
}

# The host is a failed or collapsing state (the docket's collapse test).
# Scope: country (the host)
un_mission_host_collapsing = {
	OR = {
		has_modifier = failed_state_modifier
		has_journal_entry = je_state_collapse
	}
}

# An aid mission's state has recovered: no famine, and its standard of living
# without the mission's help has reached the goal (un_mission_aid_goal).
# Scope: mission container
un_mission_aid_recovered = {
	var:un_msn_state ?= { NOT = { has_famine = yes } }
	un_mission_underlying_sol >= un_mission_aid_goal
}
