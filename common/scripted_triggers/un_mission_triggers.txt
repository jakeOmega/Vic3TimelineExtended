# The scoped state already hosts an active mission (one per state at a time).
# Scope: state
un_mission_state_occupied = {
	var:un_msn_here ?= { has_tag = un_msn_active }
}

# The scoped country hosts an active peacekeeping or stabilisation mission: it
# is under UN protection, which the war-goal surcharge doubles from Strong
# (un_teeth_war_goal_surcharge).
# Scope: country
un_mission_protected = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		has_tag = un_msn_active
		OR = {
			has_tag = un_msn_peacekeeping
			has_tag = un_msn_stabilisation
		}
		var:un_msn_host ?= PREV
	}
}

# The scoped country contributes to an active mission hosted by
# scope:un_msn_intel_host: its covert networks there grow faster
# (covert_nets_sync; §7.4).
# Scope: country (the operator); expects scope:un_msn_intel_host
un_mission_contributes_in = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		has_tag = un_msn_active
		var:un_msn_host ?= scope:un_msn_intel_host
		is_target_in_variable_list = { name = un_msn_contributors target = PREV }
	}
}

# The scoped country hosts an active mission of $TYPE$ (peacekeeping | aid |
# stabilisation). A country hosting one cannot ask the Assembly for another.
# Scope: country
un_mission_hosts = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		has_tag = un_msn_active
		has_tag = un_msn_$TYPE$
		var:un_msn_host ?= PREV
	}
}

# A mission of $TYPE$ can be found or opened for the scoped host: it already
# hosts one, or it holds a state with no mission (un_mission_join_host). For
# the tooltips, which are shown before the effect runs.
# Scope: country (the host)
un_mission_can_join_host = {
	OR = {
		un_mission_hosts = { TYPE = $TYPE$ }
		any_scope_state = { NOT = { un_mission_state_occupied = yes } }
	}
}

# The host is at war, and not as the initiator of the play that started it.
# Scope: country (the host)
un_mission_host_attacked = {
	is_at_war = yes
	is_diplomatic_play_initiator = no
}

# The mission's host has been attacked since the mission opened: it is at war
# and not the initiator (un_mission_host_attacked), and the mission did not
# open into that war (un_msn_war_at_open, cleared by the host's first month
# of peace).
# Scope: mission container; expects scope:un_msn_host_now
un_mission_attacked_since_open = {
	NOT = { has_variable = un_msn_war_at_open }
	scope:un_msn_host_now = { un_mission_host_attacked = yes }
}

# The host is a failed or collapsing state (the docket's collapse test).
# Scope: country (the host)
un_mission_host_collapsing = {
	OR = {
		has_modifier = failed_state_modifier
		has_journal_entry = je_state_collapse
	}
}

# An aid mission's state has recovered: no famine, and its standard of living
# without the mission's help has reached the goal (un_mission_aid_goal).
# Scope: mission container
un_mission_aid_recovered = {
	var:un_msn_state ?= { NOT = { has_famine = yes } }
	un_mission_underlying_sol >= un_mission_aid_goal
}

# ---- Joining at will -------------------------------------------------------------------
# Any major-power member can send a contingent to a mission in the field, and
# bring it home again: the chamber's Missions in the Field rows for a player
# (un_chamber_mission_join_sgui / un_chamber_mission_withdraw_sgui), and
# un_mission_ai_monthly for the AI. Both ask the triggers below, so the
# rules cannot drift between them. The writers are un_mission_volunteer and
# un_mission_withdraw (un_mission_effects.txt).

# The scoped country may send contingents at all: a member, of major-power
# rank (every other path into a mission asks the same, and strength counts
# heads up to three), and not under the Assembly's sanctions. Not a member
# whose representation is suspended (§0.8): sending troops is foreign policy,
# which its overlord conducts. Contingents it already has in the field stay.
# Scope: country
un_mission_volunteer_eligible = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	NOT = { un_representation_suspended = yes }
	country_rank >= rank_value:major_power
	NOT = { has_modifier = un_sanctions_target_modifier }
}

# $WHO$ is not the scoped mission's host.
# Scope: mission container
un_mission_not_hosted_by = {
	NOT = { var:un_msn_host ?= $WHO$ }
}

# $WHO$ is not at war with the scoped mission's host.
# Scope: mission container
un_mission_not_at_war_with_host = {
	NOT = { var:un_msn_host ?= { has_war_with = $WHO$ } }
}

# $WHO$ serves in the scoped mission.
# Scope: mission container
un_mission_served_by = {
	un_lobby_in_list = { LIST = un_msn_contributors COUNTRY = $WHO$ }
}

# $WHO$ brought its contingent home from the scoped mission, and so cannot
# send it back.
# Scope: mission container
un_mission_withdrawn_by = {
	un_lobby_in_list = { LIST = un_msn_withdrawn COUNTRY = $WHO$ }
}

# The scoped mission takes $WHO$ as a volunteer. $WHO$'s own eligibility
# (un_mission_volunteer_eligible) is asked separately.
# Scope: mission container
un_mission_accepts_volunteer = {
	has_tag = un_msn_active
	un_mission_not_hosted_by = { WHO = $WHO$ }
	un_mission_not_at_war_with_host = { WHO = $WHO$ }
	NOT = { un_mission_served_by = { WHO = $WHO$ } }
	NOT = { un_mission_withdrawn_by = { WHO = $WHO$ } }
}

# $WHO$ serves in the scoped mission at will, and pays for it
# (un_mission_service_cost): an active mission it joined through
# un_mission_volunteer and has not left. A peacekeeping or stabilisation
# mission is paid for by the peacekeeping programme when $WHO$ runs it.
# Scope: mission container
un_mission_charges = {
	has_tag = un_msn_active
	un_mission_served_by = { WHO = $WHO$ }
	un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = $WHO$ }
	NOT = {
		AND = {
			OR = {
				has_tag = un_msn_peacekeeping
				has_tag = un_msn_stabilisation
			}
			$WHO$ = { un_standing_program_active_peacekeeping = yes }
		}
	}
}

# ---- The register's slots ----------------------------------------------------------------
# Every active mission holds a display slot, 0 to 7 (un_mission_slot_count),
# the lowest free one when it opens (un_mission_assign_slot); it gives the slot
# up when it closes. The chamber shows one row per slot, and a row's controls
# name their mission by slot, because a trigger cannot hold a scope. A
# mission opened while all eight are taken waits for one to free up
# (un_mission_monthly), and is listed under the rows meanwhile.

# The scoped mission is active and holds slot $SLOT$.
# Scope: mission container
un_mission_in_slot = {
	has_tag = un_msn_active
	has_variable = un_msn_slot
	var:un_msn_slot = $SLOT$
}

# Some active mission holds slot $SLOT$.
# Scope: any
un_mission_slot_filled = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		un_mission_in_slot = { SLOT = $SLOT$ }
	}
}

# The scoped country may send a contingent to the mission in slot $SLOT$. For
# the chamber's Join control: every clause says why it fails.
# Scope: country
un_mission_slot_can_volunteer = {
	has_global_variable_list = un_mission_registry
	custom_tooltip = {
		text = un_mission_volunteer_need_member_tt
		je:je_united_nations ?= { has_modifier = un_member_modifier }
	}
	custom_tooltip = {
		text = un_mission_volunteer_need_represented_tt
		NOT = { un_representation_suspended = yes }
	}
	custom_tooltip = {
		text = un_mission_volunteer_need_rank_tt
		country_rank >= rank_value:major_power
	}
	custom_tooltip = {
		text = un_mission_volunteer_need_not_sanctioned_tt
		NOT = { has_modifier = un_sanctions_target_modifier }
	}
	custom_tooltip = {
		text = un_mission_volunteer_need_not_host_tt
		any_in_global_list = {
			variable = un_mission_registry
			un_mission_in_slot = { SLOT = $SLOT$ }
			un_mission_not_hosted_by = { WHO = ROOT }
		}
	}
	custom_tooltip = {
		text = un_mission_volunteer_need_peace_tt
		any_in_global_list = {
			variable = un_mission_registry
			un_mission_in_slot = { SLOT = $SLOT$ }
			un_mission_not_at_war_with_host = { WHO = ROOT }
		}
	}
	custom_tooltip = {
		text = un_mission_volunteer_need_not_withdrawn_tt
		any_in_global_list = {
			variable = un_mission_registry
			un_mission_in_slot = { SLOT = $SLOT$ }
			NOT = { un_mission_withdrawn_by = { WHO = ROOT } }
		}
	}
	# The whole rule, as the AI asks it: whatever the clauses above miss
	# (serving already, which hides the control) still fails here.
	un_mission_volunteer_eligible = yes
	any_in_global_list = {
		variable = un_mission_registry
		un_mission_in_slot = { SLOT = $SLOT$ }
		un_mission_accepts_volunteer = { WHO = ROOT }
	}
}

# The scoped country serves in the mission in slot $SLOT$, and so may bring
# its contingent home.
# Scope: country
un_mission_slot_served = {
	has_global_variable_list = un_mission_registry
	any_in_global_list = {
		variable = un_mission_registry
		un_mission_in_slot = { SLOT = $SLOT$ }
		un_mission_served_by = { WHO = ROOT }
	}
}
