# Country scope: je_covert_warfare's `possible` as one tooltip line: the
# checklist in the Military panel's greyed Covert tab
# (te_military_covert_tab_unlock_sgui's is_valid, te_system_tab_sguis.txt).
# One line because the test is an OR, and a bare OR prints there as an empty
# "All of these:" (gui_modding_guide.md gotcha #31, the UN tab). The OR is the
# entry's own, copied verbatim, since the entry itself is left as it is;
# MilitaryCovertTabTest in test_covert_layout.py fails if the two differ.
covert_warfare_entry_unlocked = {
	custom_tooltip = {
		text = covert_warfare_unlock_tt
		OR = {
			AND = {
				country_rank >= rank_value:great_power
				covert_operation_max_slots >= 4
			}
			AND = {
				country_rank = rank_value:major_power
				covert_operation_max_slots >= 3
			}
			AND = {
				NOT = { country_rank >= rank_value:major_power }
				covert_operation_max_slots >= 2
			}
		}
	}
}

# Check if a diplomatic pact is a covert operation pact (any type; the list is
# pinned by test_covert_op_registry).
# Scope: diplomatic_pact
is_covert_operation_pact = {
	OR = {
		is_diplomatic_action_type = covert_election_interference_action
		is_diplomatic_action_type = covert_financial_subversion_action
		is_diplomatic_action_type = covert_infrastructure_sabotage_action
		is_diplomatic_action_type = covert_comms_disruption_action
		is_diplomatic_action_type = covert_industrial_espionage_action
		is_diplomatic_action_type = covert_military_espionage_action
		is_diplomatic_action_type = covert_influence_campaign_action
		is_diplomatic_action_type = covert_ideological_subversion_action
		is_diplomatic_action_type = covert_destabilization_action
		is_diplomatic_action_type = covert_regime_change_action
		is_diplomatic_action_type = covert_nuclear_sabotage_action
		is_diplomatic_action_type = covert_space_espionage_action
		is_diplomatic_action_type = covert_cultivate_assets_action
		is_diplomatic_action_type = covert_secure_material_action
		is_diplomatic_action_type = covert_counterintel_support_action
		is_diplomatic_action_type = covert_incite_autonomy_action
		is_diplomatic_action_type = covert_sabotage_colony_action
	}
}

# A covert operation pact that works against its second country: every type
# but Counterintelligence Support, which protects a friend. The pact-scope
# twin of covert_op_is_hostile, for counts that read pacts rather than
# containers (a network's tier-2 count of operations against us).
# Scope: diplomatic_pact
is_hostile_covert_operation_pact = {
	is_covert_operation_pact = yes
	NOT = { is_diplomatic_action_type = covert_counterintel_support_action }
}

# Target eligibility shared by all covert actions. Decentralized countries have
# has_politics = no / has_economy = no -- no centralized apparatus to subvert --
# so they are never a valid covert target. Scope: country (prospective target).
covert_action_valid_target = {
	NOT = { is_country_type = decentralized }
}

# The one definition of a friend, for every covert rule that asks. Relations
# of at least cordial AND a tie (alliance, defensive pact, the same power
# bloc, or one the other's overlord or subject), or relations of at least
# amicable with no tie. A tie never counts on its own: two members of one
# bloc who dislike each other are not friends. Consumers: the attacker's
# backlash and the defender's backlash scaling, the AI's watch odds,
# Counterintelligence Support's possible and requirement_to_maintain, and the
# AI's launch and break checks for the six operations that used to be
# refused at good relations.
# The bloc tie is guarded on the country in scope being in one: two countries
# in none can otherwise read as being in the same one.
# `is_subject_of` also counts a subject's subject. It has no mirror, so the
# overlord direction enters $TARGET$ and reads PREV (the country in scope).
# Relations are symmetric, so the answer is the same from either side.
# Relations are read as $TARGET$.relations:this: the link's argument resolves
# in the scope the chain starts from (vanilla 12_military_assistance.txt,
# `scope:source_country.relations:this` inside an enemy iterator), so the
# trigger works from any country, not only ROOT. The backlash reads it from
# the defender's side with ROOT the attacker.
# Parameters: $TARGET$ = country (a scope: reference)
# Scope: country
covert_is_friend = {
	OR = {
		AND = {
			$TARGET$.relations:this >= relations_threshold:cordial
			OR = {
				has_treaty_alliance_with = { TARGET = $TARGET$ }
				has_treaty_defensive_pact_with = { TARGET = $TARGET$ }
				AND = {
					is_in_power_bloc = yes
					is_in_same_power_bloc = $TARGET$
				}
				is_subject_of = $TARGET$
				$TARGET$ = { is_subject_of = PREV }
			}
		}
		$TARGET$.relations:this >= relations_threshold:amicable
	}
}

# True while the country runs fewer operations of this type than
# covert_ops_max_per_type. That script value is the only place the cap lives:
# each operation is its own script container, so raising the cap needs no
# storage change. (`count >= <script value>` is vanilla's form, e.g.
# je_zaibatsu_min_company_count.)
# Parameters: $TYPE$ = diplomatic_action_type key (e.g. covert_election_interference_action)
# Scope: country
covert_ops_type_below_cap = {
	NOT = {
		any_scope_diplomatic_pact = {
			count >= covert_ops_max_per_type
			is_diplomatic_action_type = $TYPE$
			first_country = prev
		}
	}
}

# Operation phase checks. Phase 2 (establishing) starts at month 6, phase 3
# (fully operational) at month 12, measured on covert_op_effective_duration:
# months run plus the head start the target's network gave the operation at
# creation (slice 4). These two are the ONLY place the thresholds live.
# Scope: covert operation container
covert_op_is_established = {
	covert_op_effective_duration >= 6
}

covert_op_is_fully_operational = {
	covert_op_effective_duration >= 12
}

# True when the operation's priority is at least $N$. A container without
# iw_priority (a save made before slice 3, until the first monthly sync) reads
# as priority 1, so this is false for it.
# Parameters: $N$ = 2 or 3
# Scope: covert operation container
covert_op_priority_at_least = {
	has_variable = iw_priority
	var:iw_priority >= $N$
}

# True when the priority upkeep is charged at is at least $N$: the higher of
# the current priority and the one the running effects were applied at
# (iw_priority_applied, snapshot by covert_ops_apply_all_phase_effects).
# Parameters: $N$ = 2 or 3
# Scope: covert operation container
covert_op_charged_priority_at_least = {
	OR = {
		covert_op_priority_at_least = { N = $N$ }
		AND = {
			has_variable = iw_priority_applied
			var:iw_priority_applied >= $N$
		}
	}
}

# FUNDING CONTROLS — shared eligibility
# ============================================================================
# One definition per funding action, called by BOTH the journal entry's
# scripted button (`possible`) and the widget's scripted GUI (`is_valid`), so
# the two surfaces cannot drift. The custom_tooltip text is phrased as a
# condition, not a complaint, because ScriptedGui.IsValidTooltip renders it
# with a tick when it holds and a cross when it does not — and it doubles as
# the "you are here" indicator at either end of the ladder.
# Scope: country

covert_possible_increase_funding = {
	custom_tooltip = {
		text = iw_funding_not_max_tt
		OR = {
			NOT = { has_variable = iw_funding_level }
			var:iw_funding_level < iw_funding_level_max
		}
	}
}

covert_possible_decrease_funding = {
	custom_tooltip = {
		text = iw_funding_not_min_tt
		has_variable = iw_funding_level
		var:iw_funding_level > 0
	}
}

# ============================================================================
# STAND DOWN — shared eligibility
# ============================================================================
# Whether the operation a widget row is showing can be ended from that row.
#
# FAIL-CLOSED BY DESIGN. The row hands the target country to script through a
# saved scope, and this trigger requires BOTH that the scope resolved AND that we
# really run an operation of this exact type against that exact country. If the
# scope ever arrives unset — the one part of the mechanism with no vanilla
# precedent for a scripted GUI, see the .gui header — the control greys out and
# does nothing. It can never end an operation other than the one the row names.
# Parameters: $TYPE$ = short name (e.g. election_interference)
# Scope: operator country; scope:iw_tgt = the row's target country
covert_possible_stand_down = {
	custom_tooltip = {
		text = iw_stand_down_target_known_tt
		exists = scope:iw_tgt
	}
	custom_tooltip = {
		text = iw_stand_down_op_matches_tt
		has_variable_list = iw_ops
		any_in_list = {
			variable = iw_ops
			has_tag = iw_op_$TYPE$
			var:iw_target ?= scope:iw_tgt
		}
	}
}

# ============================================================================
# PRIORITY — shared eligibility
# ============================================================================
# Whether the operation a widget row is showing can have its priority changed.
# Called by the row stepper's scripted GUIs (is_valid) and by the AI's
# priority management, so the two cannot diverge.
#
# FAIL-CLOSED BY DESIGN, like covert_possible_stand_down. The row hands the
# operation's container to script as the saved scope iw_op; passing an object
# into a scripted GUI's saved_scopes has no vanilla precedent. If it arrives
# unset, or is anything but one of OUR running operations, the stepper greys
# out and does nothing.
# Scope: operator country; scope:iw_op = the operation container
covert_priority_op_is_ours = {
	custom_tooltip = {
		text = iw_priority_op_known_tt
		exists = scope:iw_op
	}
	custom_tooltip = {
		text = iw_priority_op_ours_tt
		exists = scope:iw_op
		has_variable_list = iw_ops
		is_target_in_variable_list = { name = iw_ops target = scope:iw_op }
		scope:iw_op ?= { has_tag = iw_op }
	}
}

covert_possible_priority_up = {
	covert_priority_op_is_ours = yes
	custom_tooltip = {
		text = iw_priority_not_max_tt
		scope:iw_op ?= {
			OR = {
				NOT = { has_variable = iw_priority }
				var:iw_priority < covert_op_priority_max
			}
		}
	}
	# Cultivate assets (slice 6) has no effect for priority to multiply, so it
	# stays at 1: raising it would only cost upkeep and detection. The AI
	# steps through this same trigger, so it cannot raise it either. Inside a
	# trigger_if so the line exists only on such a row: IsValidTooltip prints
	# every custom_tooltip with a tick or a cross, and a bare one would sit on
	# every other operation's stepper as a meaningless tick.
	trigger_if = {
		limit = { scope:iw_op ?= { has_tag = iw_op_cultivate_assets } }
		custom_tooltip = {
			text = iw_priority_cultivate_fixed_tt
			always = no
		}
	}
	# Tradecraft gate (slice 5): the step TO priority 3 needs an Established
	# agency. Only the step up is gated — an operation already at 3 stays
	# there when the score falls, and stepping down is never blocked. The AI
	# steps through this same trigger (covert_ai_manage_priorities).
	custom_tooltip = {
		text = iw_priority_3_needs_tradecraft_tt
		OR = {
			scope:iw_op ?= { NOT = { covert_op_priority_at_least = { N = 2 } } }
			covert_tradecraft_unlocks_priority_3 = yes
		}
	}
}

covert_possible_priority_down = {
	covert_priority_op_is_ours = yes
	custom_tooltip = {
		text = iw_priority_not_min_tt
		scope:iw_op ?= {
			has_variable = iw_priority
			var:iw_priority > 1
		}
	}
}

# ============================================================================
# EXPOSURE TIERS
# ============================================================================
# How hostile the world considers an operation when it is caught. This is THE
# tier table: the blowback script values, both custom-loc tier names and the
# pre-launch action descriptions all read these four triggers, so a new
# operation type is one code added to one OR list here and nowhere else.
#
# Parameters: $VAR$ = the NAME of a country variable holding an operation type
#             code — iw_burned_type_code on the operator while
#             covert_warfare.1 is open, iw_last_exposed_type on the target
#             afterwards. Each trigger guards the variable itself, because
#             reading one that does not exist is an engine error and custom
#             loc can render before anything has been exposed.
# Scope: country
# ============================================================================

# "Everyone spies" — stealing designs is not an attack on the state.
covert_code_tier_mild = {
	has_variable = $VAR$
	OR = {
		var:$VAR$ = 4
		var:$VAR$ = 5
		var:$VAR$ = 11
		var:$VAR$ = 12
		var:$VAR$ = 13
		# Counterintelligence Support: nominal. It is never caught, so no
		# exposure cost ever reads this; it sits here so every code has a tier.
		var:$VAR$ = 14
	}
}

# Meddling in how the target governs itself.
covert_code_tier_moderate = {
	has_variable = $VAR$
	OR = {
		var:$VAR$ = 0
		var:$VAR$ = 1
		var:$VAR$ = 6
		# Inciting colonial autonomy: a colony's own politics.
		var:$VAR$ = 15
	}
}

# Attacking the state itself. These are the ones third parties hear about.
covert_code_tier_severe = {
	has_variable = $VAR$
	OR = {
		var:$VAR$ = 7
		var:$VAR$ = 8
		var:$VAR$ = 9
		var:$VAR$ = 10
		# Colony sabotage: people killed in the target's colony.
		var:$VAR$ = 16
	}
}

# Operations usable once conflict with the target is already underway: an
# active war, or the diplomatic play that precedes one. Infamy is what's at
# stake, and how much depends on which of those two you're caught in — see
# covert_exposure_is_costless below.
covert_code_tier_war = {
	has_variable = $VAR$
	OR = {
		var:$VAR$ = 2
		var:$VAR$ = 3
	}
}

# A wartime operation caught once the war has actually started costs nothing:
# no infamy, no relations. Sabotage does not make the world angrier at a
# country that is already fighting. Caught during the diplomatic play that
# precedes the war it is the opposite — see covert_exposure_infamy_war_prewar.
#
# THE one place that question is answered: both blowback script values and
# both options of covert_warfare.1 read this.
# Scope: country (the operator, while covert_warfare.1 is open)
covert_exposure_is_costless = {
	covert_code_tier_war = { VAR = iw_burned_type_code }
	has_variable = iw_burned_at_war
	var:iw_burned_at_war = 1
}

# A political lobby that favours $TARGET$: lobby_pro_country with that target,
# or lobby_pro_overlord when it is the overlord of the lobby's country (the
# type a pro-country lobby swaps to once its country becomes a subject). THE
# one place that says which lobbies the backlash reaches: the attacker's home
# appeasement hit (covert_backlash_home) and the defender's discredited
# interest groups (covert_backlash_public) both ask it.
# Parameters: $TARGET$ = the country the lobby favours (a scope: reference)
# Scope: political_lobby
covert_backlash_pro_lobby_of = {
	OR = {
		is_political_lobby_type = lobby_pro_country
		is_political_lobby_type = lobby_pro_overlord
	}
	target = $TARGET$
}

# ============================================================================
# WATCHING A CAUGHT OPERATION
# ============================================================================
# A caught operation goes to its target first when the target runs a covert
# service: expose it (covert_warfare.1, as before) or let it run and watch it
# (covert_watch_start). The record shape and the flow: covert_warfare_effects.txt,
# § WATCHING A CAUGHT OPERATION.

# True when this country can decide what to do with an operation it caught:
# the service that would watch it, the same test covert_warfare.2's
# "Retaliate in kind" applies. Without it a catch goes straight to the exposure.
# Scope: country (the target of the caught operation)
covert_can_watch = {
	has_game_rule = covert_warfare_enabled
	has_journal_entry = je_covert_warfare
}

# One exposure in flight per attacker. covert_warfare.1 keeps the operation it
# is resolving on the attacker itself (iw_burned_type_code and its siblings),
# so a second one opened before the first closes would overwrite them. True
# from the moment an exposure is ordered (iw_expose_inflight, set by
# covert_exposure_fire before the event is triggered, since trigger_event may
# not run the event's immediate inside the calling effect) until the event's
# `after` clears both. While it holds, the detection roll waits a month and
# every other exposure of this attacker's operations waits its turn.
# Scope: country (the attacker)
covert_exposure_in_flight = {
	OR = {
		has_variable = iw_burned_type_code
		has_variable = iw_expose_inflight
	}
}

# True for an operation that works against its target: every type but
# Counterintelligence Support, which protects a friend and is never rolled for
# detection, caught or watched.
# Scope: covert operation container
covert_op_is_hostile = {
	NOT = { has_tag = iw_op_counterintel_support }
}

# True for the operation a watch record stands for: run against $DEFENDER$,
# of the type code $CODE$, and known to its target. The no-duplicate rule
# makes (type, target) unique per operator, so this picks at most one of the
# attacker's operations. $CODE$ is read in the operation's scope: pass a
# number or a country-variable chain (scope:<country>.var:<staged code>),
# never a container's own var: or PREV.
# Parameters: $DEFENDER$ = the watching country (a scope: reference)
#             $CODE$ = operation type code
# Scope: covert operation container
covert_watch_op_matches = {
	has_variable = iw_known
	var:iw_target ?= $DEFENDER$
	has_variable = iw_type_code
	var:iw_type_code = $CODE$
}

# True when $TARGET$ has completed at least one space race milestone this
# country has not. Milestone completion is public (world-first modifiers
# announce it), so this is not the progress omniscience the space race design
# forbids. The gate on covert_space_espionage_action — the space-race
# analogue of industrial espionage's tech-gap gate.
# Parameters: $TARGET$ = the country that may be ahead (scope:target_country
#             from an action; PREV from the console harness)
# Scope: country (the operator)
covert_target_ahead_in_space = {
	OR = {
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = suborbital }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = orbital }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = moon_landing }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = probe }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = moon_base }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = mars_landing }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = interstellar_probe }
		covert_target_ahead_in_space_on = { TARGET = $TARGET$ MILESTONE = solar_colonization }
	}
}

# One milestone of the above.
# Parameters: $TARGET$ as above; $MILESTONE$ = suborbital | orbital |
#             moon_landing | probe | moon_base | mars_landing |
#             interstellar_probe | solar_colonization
# Scope: country (the operator)
covert_target_ahead_in_space_on = {
	$TARGET$ = { has_variable = sr_completed_$MILESTONE$ }
	NOT = { has_variable = sr_completed_$MILESTONE$ }
}

# True unless this country already has a network of at least $STRENGTH$
# against $TARGET$. For the AI's cultivate-assets proposal: build where the
# network is thin, not where it is already strong.
# Parameters: $TARGET$ = country (a scope: reference), $STRENGTH$ = 0-100
# Scope: country (the operator)
covert_net_weaker_than = {
	NOT = {
		AND = {
			has_variable_list = iw_nets
			any_in_list = {
				variable = iw_nets
				var:iw_target ?= $TARGET$
				var:iw_net_strength >= $STRENGTH$
			}
		}
	}
}

# ============================================================================
# AI TARGET SELECTION
# ============================================================================

# True when this country's GDP is at least covert_ai_much_weaker_gdp_factor
# times $TARGET$'s.
# Parameters: $TARGET$ = country (a scope: reference)
# Scope: country (the operator)
covert_ai_target_much_weaker = {
	gdp >= {
		value = $TARGET$.gdp
		multiply = covert_ai_much_weaker_gdp_factor
	}
}

# Gold reserves at least covert_ai_spare_reserves_ratio full and a weekly
# surplus: the treasury can carry an operation it does not need.
# Scope: country (the operator)
covert_ai_has_money_to_burn = {
	scaled_gold_reserves >= covert_ai_spare_reserves_ratio
	net_fixed_income > 0
}

# The AI's launch test (will_propose) for the operations whose payoff scales
# with the target's size: meddling in its politics, economy or movements,
# building a network there, and sabotage in a war. Upkeep scales with our GDP,
# so against a much smaller country an operation costs more than it returns;
# the AI runs one there only with money to burn. Espionage (the payoff is our
# research or space progress), nuclear programme sabotage and secure loose
# material (the payoff is the threat removed) do not take this test.
# Parameters: $TARGET$ = country (a scope: reference)
# Scope: country (the operator)
covert_ai_target_worth_the_upkeep = {
	OR = {
		NOT = { covert_ai_target_much_weaker = { TARGET = $TARGET$ } }
		covert_ai_has_money_to_burn = yes
	}
}

# Every operation but the two wartime ones costs influence (its pact cost,
# covert_operations.txt § INFLUENCE). The AI launches one only when it can pay
# it and still holds covert_ai_influence_floor, more than any operation costs
# even a great power; an operation it calls off in deficit (will_break) frees
# less than that, so it is not relaunched at once.
# Parameters: $TYPE$ = the operation's diplomatic action
# Scope: country (the operator); scope:target_country = the target
covert_ai_can_spare_influence = {
	can_afford_diplomatic_action = { target = scope:target_country type = $TYPE$ }
	influence >= covert_ai_influence_floor
}

# ============================================================================
# AI: CALLING AN OPERATION OFF
# ============================================================================
# An operation does nothing for its first six months and reaches full strength
# at twelve, while its upkeep runs from the first month, so an AI that ends
# one early has paid for nothing. Every covert action's will_break is written
# out (none is left to the engine's default) and, where the AI may call the
# operation off at all, waits for covert_ai_op_settled first. After that it
# ends only once the operation's reason has gone, judged more loosely than
# will_propose judges it at launch (covert_ai_hostility_gone), so a target
# drifting in and out of the launch test does not make the AI stop and start.
# The four breakable operations that used to be refused at good relations
# also end once the target is a friend (covert_is_friend), after the same wait.
# The launch-only tests (free slot, intelligence capacity, target size, debt)
# are never break reasons.

# True when our $TYPE$ operation against $TARGET$ is fully operational
# (covert_op_is_fully_operational: twelve months, less the network's head
# start). A pact whose container is missing (made since the last monthly sync)
# reads as not settled, so it is kept.
# Parameters: $TYPE$ = short type name (election_interference, ...),
#             $TARGET$ = country (a scope: reference)
# Scope: country (the operator)
covert_ai_op_settled = {
	has_variable_list = iw_ops
	any_in_list = {
		variable = iw_ops
		has_tag = iw_op_$TYPE$
		var:iw_target ?= $TARGET$
		covert_op_is_fully_operational = yes
	}
}

# True when nothing hostile is left between us and $TARGET$: no rivalry either
# way, no antagonistic, belligerent or domineering attitude, and relations
# above poor. The launch test needs only our rivalry or an antagonistic or
# domineering attitude, so an operation outlasts the reason it was launched
# for while any of the rest holds.
# Parameters: $TARGET$ = country (a scope: reference)
# Scope: country (the operator, ROOT in a diplomatic action's ai block)
covert_ai_hostility_gone = {
	NOR = {
		has_diplomatic_pact = { who = $TARGET$ type = rivalry }
		$TARGET$ = { has_diplomatic_pact = { who = root type = rivalry } }
		has_attitude = { who = $TARGET$ attitude = antagonistic }
		has_attitude = { who = $TARGET$ attitude = belligerent }
		has_attitude = { who = $TARGET$ attitude = domineering }
		$TARGET$.relations:root <= relations_threshold:poor
	}
}
