# ============================================================================
# GRAND MONUMENTS — several monuments in one state: scripted triggers
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-13-multiple-monuments-per-state-design.md.

# State scope. A secondary monument stands here beside the primary. Not
# variable_list_size: its documented `target` key fails to load in 1.14, and a
# list keeps existing after its last entry is removed.
gm_state_has_secondary = {
	has_variable_list = gm_monuments_here
	any_in_list = { variable = gm_monuments_here exists = this }
}

# ---- A monument beside another: its own dedication (spec 2026-10-16 §4.1) -------------
# Container scope, a secondary monument (tag gm_monument), reading its records
# (gm_tr_dedication, gm_tr_raised_by, gm_tr_honoree, gm_tr_faith) and its
# state's owner through var:gm_tr_origin: the state triggers' twins
# (monument_triggers.txt). A missing record reads as "does not fit", never as
# "fits". test_grand_monument_registry.py pins them to DEDICATIONS.
gm_mon_is = {
	has_variable = gm_tr_dedication
	var:gm_tr_dedication = flag:$KEY$
}

gm_mon_kind_regime = {
	OR = {
		gm_mon_is = { KEY = crown }
		gm_mon_is = { KEY = republic }
		gm_mon_is = { KEY = revolution }
	}
}

gm_mon_kind_ruler = {
	gm_mon_is = { KEY = leader }
}

gm_mon_kind_faith = {
	gm_mon_is = { KEY = religious }
}

gm_mon_is_bound = {
	OR = {
		gm_mon_kind_regime = yes
		gm_mon_kind_ruler = yes
		gm_mon_kind_faith = yes
	}
}

# The country that raised it owns its state.
gm_mon_raised_by_owner = {
	has_variable = gm_tr_raised_by
	var:gm_tr_raised_by ?= { save_temporary_scope_as = gm_tmp_mon_raiser }
	var:gm_tr_origin ?= { owner = scope:gm_tmp_mon_raiser }
}

# The honoured character still rules its state's owner.
gm_mon_leader_fits = {
	has_variable = gm_tr_honoree
	var:gm_tr_honoree ?= { save_temporary_scope_as = gm_tmp_mon_honoree }
	var:gm_tr_origin ?= {
		owner = {
			ruler ?= { this = scope:gm_tmp_mon_honoree }
		}
	}
}

# The owner's state religion is still the one it honours, not under State
# Atheism.
gm_mon_shrine_fits = {
	has_variable = gm_tr_faith
	var:gm_tr_faith ?= { save_temporary_scope_as = gm_tmp_mon_faith }
	var:gm_tr_origin ?= {
		owner = {
			NOT = { has_law_or_variant = law_type:law_state_atheism }
			religion = scope:gm_tmp_mon_faith
		}
	}
}

# Its message still stands, read live from its state's owner.
gm_mon_message_fits = {
	OR = {
		gm_mon_is_bound = no
		AND = {
			gm_mon_raised_by_owner = yes
			OR = {
				AND = {
					gm_mon_is = { KEY = crown }
					var:gm_tr_origin ?= { owner = { monument_government_is_crowned = yes } }
				}
				AND = {
					gm_mon_is = { KEY = republic }
					var:gm_tr_origin ?= { owner = { monument_government_is_republican = yes } }
				}
				AND = {
					gm_mon_is = { KEY = revolution }
					var:gm_tr_origin ?= { owner = { gm_government_is_revolutionary = yes } }
				}
				AND = {
					gm_mon_kind_ruler = yes
					gm_mon_leader_fits = yes
				}
				AND = {
					gm_mon_kind_faith = yes
					gm_mon_shrine_fits = yes
				}
			}
		}
	}
}

gm_mon_is_contested = {
	has_variable = gm_tr_contested
}

gm_mon_is_heritage = {
	has_variable = gm_tr_heritage
}

# "Fits": its politics count. At home, dedicated and seen (gm_mon_first_sight
# has written its raiser), neither contested nor heritage, its message
# standing.
gm_mon_status_fits = {
	has_tag = gm_monument
	has_variable = gm_tr_dedication
	has_variable = gm_tr_seen
	gm_mon_is_contested = no
	gm_mon_is_heritage = no
	gm_mon_message_fits = yes
}

# Standing grandeur (prestige, cultural pull): every monument at home not
# contested.
gm_mon_counts_standing = {
	has_tag = gm_monument
	gm_mon_is_contested = no
}

		}
	}
}

# Container scope. A secondary monument's years add to its grandeur this month
# (spec 2026-10-16 §2): it stands at home, dedicated, not contested. A trophy
# abroad (tag gm_trophy) never ages.
gm_mon_ages = {
	has_tag = gm_monument
	has_variable = gm_tr_dedication
	has_variable = gm_tr_origin
	exists = var:gm_tr_origin
	NOT = { has_variable = gm_tr_contested }
}

# State scope. A construction site may be started or expanded here (sites spec
# 2026-10-18 §3.1): the site's can_build_government / can_build_private, so
# each reason shows in the build tooltip. Not while the monument here is
# contested (decide first), while a ceremony, unveiling or naming is under way
# (a ceremony removes the site), while script puts a method back, or once six
# monuments stand, the site among them (returns and archaeology may pass six;
# construction may not). Open while levels of the site are being built:
# closing the gate on them could cancel or stall them. Open while script
# creates or removes a monument's building (gm_rebuilding, one day): since
# 1.13.9 create_building checks some of a building's requirements, and whether
# that includes can_build_government is unverified (scripting_best_practices.md;
# the te_cm_placing_site pattern).
gm_state_can_build_site = {
	trigger_if = {
		limit = {
			NOT = { has_variable = gm_rebuilding }
			NOT = { b:building_grand_monument_site ?= { is_under_construction = yes } }
		}
		custom_tooltip = {
			text = gm_expand_contested_tt
			NOT = { gm_state_is_contested = yes }
		}
		custom_tooltip = {
			text = gm_expand_pending_tt
			NOT = { has_variable = gm_ceremony_pending }
			NOT = { has_variable = gm_unveil_pending }
			NOT = { has_variable = gm_name_choice_open }
			NOT = { has_variable = gm_being_named }
		}
		custom_tooltip = {
			text = gm_expand_rebuilding_tt
			NOT = { has_variable = gm_restore_pm }
			NOT = { has_variable = gm_gift }
		}
		custom_tooltip = {
			text = gm_expand_full_tt
			gm_state_monument_count < 6
		}
		# An older save's monument here is migrated first (gm_state_migrate).
		custom_tooltip = {
			text = gm_expand_migrating_tt
			OR = {
				NOT = { has_building = building_grand_monument }
				has_variable = gm_migrated
			}
		}
	}
}

# State scope. A secondary here may be rededicated (spec 2026-10-16 §4.3): it is
# swapped in as the state's monument first, so the one standing there now must
# be a finished, named dedicated monument with nothing under way.
gm_state_can_rededicate_beside = {
	gm_state_is_dedicated = yes
	has_variable = gm_seen
	has_variable = gm_levels
	NOT = { has_variable = gm_restore_pm }
	NOT = { has_variable = gm_ceremony_pending }
	NOT = { has_variable = gm_rebuilding }
	NOT = { has_variable = gm_unveil_pending }
	NOT = { has_variable = gm_name_choice_open }
	NOT = { has_variable = gm_skin_axis }
	NOT = { has_variable = gm_being_named }
}

# Container scope. Its dedication is the one its state's owner's variable $VAR$
# names (gm_state_has_dedication_flag's twin, gm_commission_triggers.txt).
gm_mon_has_dedication_flag = {
	var:gm_tr_origin ?= { owner = { has_variable = $VAR$ } }
	OR = {
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:crown } } gm_mon_is = { KEY = crown } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:republic } } gm_mon_is = { KEY = republic } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:revolution } } gm_mon_is = { KEY = revolution } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:leader } } gm_mon_is = { KEY = leader } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:religious } } gm_mon_is = { KEY = religious } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:civic } } gm_mon_is = { KEY = civic } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:war_memorial } } gm_mon_is = { KEY = war_memorial } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:artistic } } gm_mon_is = { KEY = artistic } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:naturalist } } gm_mon_is = { KEY = naturalist } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:scientific } } gm_mon_is = { KEY = scientific } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:industrial } } gm_mon_is = { KEY = industrial } }
		AND = { var:gm_tr_origin ?= { owner = { var:$VAR$ = flag:athletic } } gm_mon_is = { KEY = athletic } }
	}
}

# Container scope. This secondary counts toward its owner's open opportunity
# (spec 2026-10-16 §5), as gm_state_com_counts does for a state's own: it fits,
# it is of the opportunity's dedication, and it stands in the opportunity's
# state if that names one.
gm_mon_com_counts = {
	var:gm_tr_origin ?= { owner = { has_variable = gm_com_dedication } }
	gm_mon_status_fits = yes
	gm_mon_has_dedication_flag = { VAR = gm_com_dedication }
	var:gm_tr_origin ?= {
		OR = {
			owner = {
				NOT = { has_variable = gm_com_state }
			}
			AND = {
				owner.var:gm_com_state ?= { save_temporary_scope_as = gm_tmp_mon_com_state }
				this = scope:gm_tmp_mon_com_state
			}
		}
	}
}

# ---- Anniversaries (ageing spec 2026-10-16 §5) -------------------------------------
# Container scope. A monument beside another of dedication $KEY$ that fits and
# is a real landmark (grandeur 3 or more) holds anniversaries, as
# gm_state_holds_anniversary does for a state's own.
gm_mon_holds_anniversary = {
	gm_mon_status_fits = yes
	gm_mon_is = { KEY = $KEY$ }
	gm_mon_grandeur >= 3
}

# Country scope. Some monument of dedication $KEY$ holds anniversaries: a
# state's own or one beside another. The dispatcher and each anniversary ask
# this.
gm_country_holds_anniversary = {
	OR = {
		any_scope_state = {
			gm_state_holds_anniversary = { PM = pm_monument_$KEY$ }
		}
		AND = {
			has_variable_list = gm_monuments_held
			any_in_list = {
				variable = gm_monuments_held
				gm_mon_holds_anniversary = { KEY = $KEY$ }
			}
		}
	}
}

# ---- A trophy comes home (spec §4.1) -----------------------------------------------
# State scope, with scope:gm_tr_moving the returning trophy. What stands here
# is what the trophy is a piece of: the landmark it came from, or, with
# neither naming one, a monument of the same unnamed make.
gm_state_stands_for_moving = {
	OR = {
		AND = {
			has_variable = gm_landmark
			scope:gm_tr_moving = { has_variable = gm_tr_landmark }
			var:gm_landmark = scope:gm_tr_moving.var:gm_tr_landmark
		}
		AND = {
			NOT = { has_variable = gm_landmark }
			scope:gm_tr_moving = { NOT = { has_variable = gm_tr_landmark } }
		}
	}
}

# Container scope (the returning trophy, scope:gm_tr_moving). A piece of the
# primary monument standing in its origin: it rejoins it (the Quadriga on the
# Gate). A whole monument never does: it is a monument of its own.
gm_tr_rejoins_primary = {
	has_variable = gm_tr_share
	NOT = { var:gm_tr_share = flag:whole }
	var:gm_tr_origin ?= {
		has_building = building_grand_monument
		gm_state_stands_for_moving = yes
	}
}

# Container scope (scope:gm_tr_moving). A piece of a secondary monument
# standing in its origin, the same landmark: it rejoins that one.
gm_tr_rejoins_secondary = {
	has_variable = gm_tr_share
	NOT = { var:gm_tr_share = flag:whole }
	has_variable = gm_tr_landmark
	var:gm_tr_origin ?= {
		has_variable_list = gm_monuments_here
		any_in_list = {
			variable = gm_monuments_here
			has_variable = gm_tr_landmark
			var:gm_tr_landmark = scope:gm_tr_moving.var:gm_tr_landmark
		}
	}
}

# Container scope (scope:gm_tr_moving). A monument stands in its origin and
# the trophy is not a piece of it: it comes home as a monument of its own (the
# Luxor Obelisk beside the Obelisks of Karnak).
gm_tr_stands_apart = {
	# A building raised there today (gm_rebuilding) stands, whether or not it
	# reads back yet (sites spec 2026-10-18 §3.3).
	var:gm_tr_origin ?= {
		OR = {
			has_building = building_grand_monument
			has_variable = gm_rebuilding
		}
	}
	NOT = { gm_tr_rejoins_primary = yes }
	NOT = { gm_tr_rejoins_secondary = yes }
}

# ---- Looting one monument of a state (spec §4.2) -----------------------------------
# Container scope, a secondary monument. By its form, the share an occupier
# carries off (gm_mon_tr_share): the primary's lists again, on gm_tr_form
# (gm_state_tr_form_is_object / _piece, gm_trophy_triggers.txt).
# test_grand_monument_trophies.py pins both to TROPHY_SHARES.
gm_mon_form_is_object = {
	has_variable = gm_tr_form
	OR = {
		var:gm_tr_form = flag:column
		var:gm_tr_form = flag:obelisk
		var:gm_tr_form = flag:statue
		var:gm_tr_form = flag:stele
		var:gm_tr_form = flag:pillar
		var:gm_tr_form = flag:high_cross
		var:gm_tr_form = flag:rune_stone
		var:gm_tr_form = flag:menorah
		var:gm_tr_form = flag:sun_stone
		var:gm_tr_form = flag:stela
	}
}

gm_mon_form_is_piece = {
	has_variable = gm_tr_form
	OR = {
		var:gm_tr_form = flag:arch
		var:gm_tr_form = flag:gate
		var:gm_tr_form = flag:tower
		var:gm_tr_form = flag:bell_tower
		var:gm_tr_form = flag:round_tower
		var:gm_tr_form = flag:rock_cut
	}
}

# State scope. A secondary monument here can be carried off at all: the
# building is not being rebuilt, no ceremony is asking, nothing has been taken
# from the state in the last ten years. It asks nothing of the primary's
# dedication (gm_state_tr_takeable does): a returned artifact stands in a state
# whose own monument may be undedicated.
gm_state_tr_takeable_secondary = {
	gm_system_enabled = yes
	has_building = building_grand_monument
	NOT = { has_variable = gm_restore_pm }
	NOT = { has_variable = gm_ceremony_pending }
	NOT = { has_variable = gm_gift }
	NOT = { has_variable = gm_carried_off }
	NOT = { has_variable = gm_tr_cooldown }
}

# State scope. Some monument here can be carried off: the primary
# (gm_state_tr_takeable, which needs it dedicated and not a Grand Shrine) or
# one standing beside it. An occupier's monthly roll asks this, so a returned
# artifact beside an undedicated primary or a shrine can still be looted.
gm_state_tr_any_takeable = {
	OR = {
		gm_state_tr_takeable = yes
		AND = {
			gm_state_has_secondary = yes
			gm_state_tr_takeable_secondary = yes
		}
	}
}

# Container scope, a secondary monument. $WHO$ (a country) occupies its state
# in a war and may carry it off.
gm_mon_tr_can_take = {
	var:gm_tr_origin ?= {
		gm_state_tr_takeable_secondary = yes
		gm_state_tr_occupied_by = { WHO = $WHO$ }
	}
}

# The scripted GUI's check: each part with its reason.
gm_mon_tr_can_take_tt = {
	custom_tooltip = {
		text = gm_mon_sgui_takeable_tt
		var:gm_tr_origin ?= { gm_state_tr_takeable_secondary = yes }
	}
	custom_tooltip = {
		text = gm_mon_sgui_route_tt
		var:gm_tr_origin ?= { gm_state_tr_occupied_by = { WHO = $WHO$ } }
	}
}

# ---- Landmarks seeded beside another (spec §6) ---------------------------------------
# State scope. The landmark $KEY$ stands here, as the primary or as a secondary
# monument: the seeding places it once, and not again on a re-run.
gm_state_has_landmark = {
	OR = {
		AND = {
			has_variable = gm_landmark
			var:gm_landmark = flag:$KEY$
		}
		AND = {
			has_variable_list = gm_monuments_here
			any_in_list = {
				variable = gm_monuments_here
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:$KEY$
			}
		}
	}
}

# ---- A merged state (issue #936; gm_monument_effects.txt § "A merged state") --------
# Container scope, a shadow: its state is gone (its part of the region merged
# into another, or lost its last province).
gm_sh_is_orphan = {
	has_tag = gm_shadow
	has_variable = gm_tr_origin_region
	NOT = { exists = var:gm_tr_origin }
}

# Container scope, a shadow or a monument beside another: its state is gone and
# it stood in the state region $REGION$.
gm_sh_orphan_of = {
	has_variable = gm_tr_origin_region
	NOT = { exists = var:gm_tr_origin }
	var:gm_tr_origin_region ?= $REGION$
}

# State scope. A monument from another part of this state's region may be
# waiting to be adopted here (gm_state_adopt_orphans): an orphan in the region,
# and here a finished building no record describes or one the merge made
# taller than 1 (sites spec 2026-10-18 §3.7). Until it is settled nothing reads
# this building as a new dedication. An older save's building stands at its
# full height until monument_events.32 records it (gm_migrated), so its height
# says nothing of a merge then. The cheap reads first: the refresh, which
# tooltips walk, asks it of every monument.
gm_state_merge_pending = {
	has_building = building_grand_monument
	OR = {
		gm_state_unrecorded = yes
		AND = {
			has_variable = gm_migrated
			gm_state_level > 1
		}
	}
	gm_state_orphan_in_region = yes
}

# State scope. A shadow, or a monument that stood beside one, whose state is
# gone stood in this state's region.
gm_state_orphan_in_region = {
	state_region = { save_temporary_scope_as = gm_tmp_sh_region }
	OR = {
		AND = {
			has_global_variable_list = gm_shadows
			any_in_global_list = {
				variable = gm_shadows
				gm_sh_orphan_of = { REGION = scope:gm_tmp_sh_region }
			}
		}
		AND = {
			has_global_variable_list = gm_monuments
			any_in_global_list = {
				variable = gm_monuments
				gm_sh_orphan_of = { REGION = scope:gm_tmp_sh_region }
			}
		}
	}
}

# State scope. A building here that no record describes: no ceremony has asked
# about it and none is under way. A building a merge brought, or (for a day) a
# new one before its first ceremony.
gm_state_unrecorded = {
	has_building = building_grand_monument
	NOT = { has_variable = gm_seen }
	NOT = { has_variable = gm_ceremony_level }
	NOT = { has_variable = gm_ceremony_pending }
	NOT = { has_variable = gm_restore_pm }
	NOT = { has_variable = gm_newcomer }
}
