# ============================================================================
# NUCLEAR DETERRENCE AND CRISIS DIPLOMACY — scripted triggers
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0 is what shipped).
#
# Conventions for every trigger here:
#   * Country scope unless the header says otherwise.
#   * Every `var:nd_*` read is paired with `has_variable` — the journal entry
#     and the diplomatic actions are evaluated for countries that never ran the
#     posture initialiser.
#   * A parameter naming another country ($ENEMY$, $TARGET$, …) takes a scope:
#     `scope:target_country`, a saved scope, or `var:nd_crisis_opponent`.
#   * Never pass `PREV` or a bare `var:` as a parameter: parameters are
#     substituted textually, so either one silently changes meaning inside the
#     scope changes of the trigger it is passed into (a `var:` read inside
#     `any_scope_state` reads the STATE's variable). Save a temporary scope
#     first and pass that.
#   * Nothing reads ROOT unless the header says so. Where an existing trigger
#     that does (is_losing_war_against) is reused, the caller's ROOT is the
#     country being asked, which holds for the diplomatic actions, the journal
#     entry pulses and country events alike.
# ============================================================================

# ARSENAL AND JOURNAL ENTRY

# Does this country hold at least one warhead it is allowed to keep? The gate
# for upkeep, incidents, posture and every threat. Deliberately not the
# programme's eligibility (rank, ICBMs, aid): a demoted power keeps its arsenal,
# its upkeep and its accidents (§2).
nd_is_armed = {
	has_variable = nuclear_weapon_stockpile
	var:nuclear_weapon_stockpile >= 1
	NOT = { modifier:country_nuclear_disarmament_bool = yes }
}

# Has the posture initialiser run for this country?
nd_is_initialized = {
	has_variable = nd_doctrine
	has_variable = nd_readiness
	has_variable = nd_authority
}

# The No-First-Strike Pledge (amendment_no_first_strike, #430) is the legal
# codification of doctrine 1. In force: on a law we hold (it attaches to a
# rules-of-war law). On file: in force, or attached to the law we are
# enacting, where it will be in force if the law passes.
#   in force  → the doctrine is held at no first use (nd_weekly_update, and
#               nd_init_posture for an arsenal built under it)
#   on file   → leaving no first use strikes it (nd_set_doctrine)
nd_nfu_amendment_in_force = {
	any_active_law = { has_amendment = amendment_type:amendment_no_first_strike }
}

nd_nfu_amendment_on_file = {
	OR = {
		nd_nfu_amendment_in_force = yes
		currently_enacting_law ?= { has_amendment = amendment_type:amendment_no_first_strike }
	}
}

# DOCTRINE PERMISSION TO STRIKE — one gate for every launch entry point
# ----------------------------------------------------------------------------
# ENEMY = the country to be struck. Read by nuke_diplo_action and
# tactical_nuke_diplo_action (`possible` and AI `will_propose`), by the
# Monopoly Window and by every incident branch that can end in a deliberate
# launch, the government's order on an unconfirmed warning
# (nuclear_incident.1 option c, owner ruling 2026-09-26) included. Launches
# nobody in government ordered (launch on warning, delegated authority, the
# cut-off commander) deliberately do NOT consult it: that they can breach a
# doctrine is the point of §8.

# Did $ENEMY$ — or anyone on its side of a war with us — strike us, or a
# country we cover? Retaliation is permitted under every doctrine, including
# no first use.
nd_was_struck_by = {
	save_temporary_scope_as = nd_wsb_self
	OR = {
		# Outside war, permission is pair-specific and ends with the exchange.
		AND = {
			nd_in_peacetime_exchange_with = { ENEMY = $ENEMY$ }
			has_variable = nuked_by_country
			var:nuked_by_country ?= $ENEMY$
		}
		AND = {
			has_war_with = $ENEMY$
			OR = {
				AND = { has_variable = nuked_by_country var:nuked_by_country ?= $ENEMY$ }
				# A state of ours still carrying a strike's aftermath, struck by
				# $ENEMY$ or by a country fighting us in a war $ENEMY$ is in.
				# The state names its striker (nd_struck_by, set by
				# nuclear_industrial_strike and nuclear_tactical_strike). A
				# terror detonation names nobody, so its aftermath licenses
				# nothing; nor does a strike by a country we are not fighting,
				# or one from an earlier war (nd_country_monthly_cleanup).
				# The aftermath alone used to count, whoever caused it (#526).
				any_scope_state = {
					OR = { has_modifier = nuclear_strike_aftermath has_modifier = tactical_nuke_aftermath }
					var:nd_struck_by ?= {
						has_war_with = scope:nd_wsb_self
						any_scope_war = {
							is_war_participant = scope:nd_wsb_self
							is_war_participant = $ENEMY$
						}
					}
				}
				nd_covered_country_struck_by = { ENEMY = $ENEMY$ }
			}
		}
	}
}

# Did $ENEMY$ strike a country we cover? The struck country's
# nuked_by_country (set by both strike kinds) names its last striker. The
# caller (nd_was_struck_by) has checked that we are at war with $ENEMY$.
nd_covered_country_struck_by = {
	save_temporary_scope_as = nd_ccs_self
	OR = {
		any_direct_subject = {
			nd_under_an_umbrella = yes
			has_variable = nuked_by_country
			var:nuked_by_country ?= $ENEMY$
		}
		any_scope_treaty = {
			any_scope_article = {
				has_type = nuclear_guarantee
				source_country = scope:nd_ccs_self
				target_country = {
					has_variable = nuked_by_country
					var:nuked_by_country ?= $ENEMY$
				}
			}
		}
		# A country hosting our weapons: a strike on it struck them (Nuclear
		# Basing, rework spec §7.3).
		any_scope_treaty = {
			any_scope_article = {
				has_type = nuclear_basing
				source_country = scope:nd_ccs_self
				target_country = {
					has_variable = nuked_by_country
					var:nuked_by_country ?= $ENEMY$
				}
			}
		}
		# An unarmed co-belligerent in the war with $ENEMY$ (rework spec §8.3).
		any_cobelligerent_in_war = {
			nd_believed_armed = no
			has_war_with = $ENEMY$
			has_variable = nuked_by_country
			var:nuked_by_country ?= $ENEMY$
		}
	}
}

# Does a country we cover carry a fresh strike's aftermath? The Rules of War
# law yields to it as it does to a strike on our own cities
# (nd_war_law_exception).
nd_covered_country_struck = {
	save_temporary_scope_as = nd_ccs2_self
	OR = {
		any_direct_subject = {
			nd_under_an_umbrella = yes
			any_scope_state = {
				OR = {
					has_modifier = nuclear_strike_aftermath
					has_modifier = tactical_nuke_aftermath
				}
			}
		}
		any_scope_treaty = {
			any_scope_article = {
				has_type = nuclear_guarantee
				source_country = scope:nd_ccs2_self
				target_country = {
					any_scope_state = {
						OR = {
							has_modifier = nuclear_strike_aftermath
							has_modifier = tactical_nuke_aftermath
						}
					}
				}
			}
		}
		# An unarmed co-belligerent in one of our wars (rework spec §8.3).
		any_cobelligerent_in_war = {
			nd_believed_armed = no
			any_scope_state = {
				OR = {
					has_modifier = nuclear_strike_aftermath
					has_modifier = tactical_nuke_aftermath
				}
			}
		}
	}
}

# $ENEMY$'s side holds a subjugation or annexation goal against us. PREV inside
# the $ENEMY$ block is this country.
nd_enemy_threatens_existence = {
	$ENEMY$ = {
		OR = {
			war_side_has_war_goal_of_type_against = { type = annex_country target = PREV }
			war_side_has_war_goal_of_type_against = { type = te_reunify_country target = PREV }
			war_side_has_war_goal_of_type_against = { type = make_protectorate target = PREV }
			war_side_has_war_goal_of_type_against = { type = make_dominion target = PREV }
			war_side_has_war_goal_of_type_against = { type = make_tributary target = PREV }
			war_side_has_war_goal_of_type_against = { type = make_personal_union target = PREV }
			war_side_has_war_goal_of_type_against = { type = make_crown_land target = PREV }
			war_side_has_war_goal_of_type_against = { type = make_chartered_company target = PREV }
		}
	}
}

# One of our incorporated states is a war-goal target of $ENEMY$, and its side
# holds a goal that takes a state. Both: in a war is_target_of_wargoal also
# flags our capital under any country-level goal, a humiliation say (probed
# 2026-10-28, scripting_best_practices.md § "War Goals, Plays, Adjacency and
# Claims"). In a play it finds nothing, so this is a war test.
nd_core_threatened_by = {
	save_temporary_scope_as = nd_ct_self
	$ENEMY$ = {
		OR = {
			war_side_has_war_goal_of_type_against = { type = conquer_state target = scope:nd_ct_self }
			war_side_has_war_goal_of_type_against = { type = return_state target = scope:nd_ct_self }
			war_side_has_war_goal_of_type_against = { type = take_treaty_port target = scope:nd_ct_self }
			war_side_has_war_goal_of_type_against = { type = liberate_country target = scope:nd_ct_self }
		}
	}
	any_scope_state = {
		is_incorporated = yes
		is_target_of_wargoal = $ENEMY$
	}
}

# The scoped country is losing its war with $ENEMY$: a quarter of its land
# occupied, or, after five significant battles in a war with them, under 35 %
# of the battles won. is_losing_war_against's rule, with the loser read from
# this scope rather than ROOT.
nd_is_losing_war_to = {
	save_temporary_scope_as = nd_lw_self
	OR = {
		enemy_occupation >= 0.25
		any_scope_war = {
			is_war_participant = $ENEMY$
			num_significant_battles >= 5
			scope:nd_lw_self = {
				size_weighted_won_battles_fraction = { target = PREV value < 0.35 }
			}
		}
	}
}

# We defied-or-were-defied bookkeeping: $ENEMY$ rejected a public ultimatum of
# ours (nd_crisis_record_defiance). Compellence's own licence.
nd_was_defied_by = {
	has_variable_list = nd_defied_us
	is_target_in_variable_list = { name = nd_defied_us target = $ENEMY$ }
}

# Does our doctrine permit a strike on $ENEMY$ right now? Law gates
# (humanitarian regulations, limited war) are separate and stay on the actions.
nd_doctrine_permits_strike = {
	OR = {
		has_war_with = $ENEMY$
		AND = {
			nd_in_peacetime_exchange_with = { ENEMY = $ENEMY$ }
			nd_was_struck_by = { ENEMY = $ENEMY$ }
		}
	}
	OR = {
		# Retaliation: every doctrine.
		nd_was_struck_by = { ENEMY = $ENEMY$ }
		# Warfighting: any war.
		nd_doctrine_warfighting = yes
		# Compellence: a defied public ultimatum, or our crisis with them past
		# the warning stage.
		AND = {
			nd_doctrine_at_least_compellence = yes
			OR = {
				nd_was_defied_by = { ENEMY = $ENEMY$ }
				AND = {
					nd_is_crisis_owner = yes
					var:nd_crisis_opponent ?= $ENEMY$
					has_variable = nd_crisis_stage
					var:nd_crisis_stage >= 2
				}
			}
		}
		# Flexible first use: losing to them, or our core under threat.
		AND = {
			nd_doctrine_at_least_flexible = yes
			OR = {
				is_losing_war_against = { ENEMY = $ENEMY$ }
				nd_core_threatened_by = { ENEMY = $ENEMY$ }
			}
		}
		# Existential deterrence (and everything above it): national survival.
		AND = {
			nd_doctrine_nfu = no
			OR = {
				nd_enemy_threatens_existence = { ENEMY = $ENEMY$ }
				AND = {
					nd_core_threatened_by = { ENEMY = $ENEMY$ }
					is_losing_war_against = { ENEMY = $ENEMY$ }
				}
			}
		}
	}
}

# The Rules of War gate on a nuclear strike, shared by the strike actions
# (nuke.txt), every crisis option that can end in a strike (.4.g, .7.a,
# .20.b) and the Monopoly Window's first use (nuclear_incident.50.c), which
# used to skip it. Country scope. The ban is a law modifier, not a has_law
# test (2026-10-09), so the law tooltip shows it: Humanitarian Regulations
# carries country_strategic_nuclear_strikes_forbidden_bool and Limited War both
# booleans (nuclear_program_modifier_types.txt). Either is lifted while
# nd_war_law_exception holds. A threat the law forbids carrying out is a bluff
# (nd_threat_law_permits).
nd_war_law_permits_strategic_strike = {
	OR = {
		NOT = { modifier:country_strategic_nuclear_strikes_forbidden_bool = yes }
		nd_war_law_exception = yes
	}
}

nd_war_law_permits_tactical_strike = {
	OR = {
		NOT = { modifier:country_tactical_nuclear_strikes_forbidden_bool = yes }
		nd_war_law_exception = yes
	}
}

# Either lifts the law: we — or a country we cover — have been struck, or a war
# we are fighting carries an annexation-type war goal against us.
nd_war_law_exception = {
	OR = {
		any_scope_state = {
			OR = {
				has_modifier = nuclear_strike_aftermath
				has_modifier = tactical_nuke_aftermath
			}
		}
		nd_covered_country_struck = yes
		nd_fighting_for_existence = yes
	}
}

# A war we are fighting carries an annexation-type war goal against us. Country
# scope. The war-law exception's half that needs no strike, and part of the
# Total War AI weight's serious war (te_row_serious_war).
nd_fighting_for_existence = {
	save_temporary_scope_as = nd_wl_self
	any_scope_war = {
		is_target_of_wargoal_in_war = scope:nd_wl_self
		OR = {
			has_war_goal = make_protectorate
			has_war_goal = make_dominion
			has_war_goal = make_tributary
			has_war_goal = make_personal_union
			has_war_goal = make_crown_land
			has_war_goal = make_chartered_company
			has_war_goal = annex_country
			has_war_goal = te_reunify_country
		}
	}
}

# No First Use, and they have not struck us (retaliation is allowed under
# every doctrine).
nd_threat_bluff_nfu = {
	nd_doctrine_nfu = yes
	NOT = { nd_was_struck_by = { ENEMY = $TARGET$ } }
}

# The Rules of War law would let us carry it out: in a war, the strike gate

	OR = {
		nd_war_law_permits_strategic_strike = yes

# Existential deterrence has something to deter here: a country we guarantee,
# our core, or our existence.
nd_threat_existential_stake = {
	OR = {
		nd_dispute_guarantee_against = { TARGET = $TARGET$ }
		nd_core_threatened_by = { ENEMY = $TARGET$ }
		nd_enemy_threatens_existence_in_play = { ENEMY = $TARGET$ }

# Dispute 6 (rework spec §8.5): $TARGET$ struck us, or a country we cover by
# treaty or umbrella, outside a war in the last 24 months
# (nd_peacetime_struck_by, set by nd_record_nuclear_use). The grievance is the
# struck side's, and the concession its reparations.
nd_dispute_peacetime_strike_by = {
	save_temporary_scope_as = nd_dps_self
	OR = {
		var:nd_peacetime_struck_by ?= $TARGET$
		any_direct_subject = {
			nd_under_umbrella_of = { OVERLORD = scope:nd_dps_self }
			var:nd_peacetime_struck_by ?= $TARGET$
		}
		any_scope_treaty = {
			any_scope_article = {
				has_type = nuclear_guarantee
				source_country = scope:nd_dps_self
				target_country = { var:nd_peacetime_struck_by ?= $TARGET$ }
			}
		}
	}
}

# Dispute 7 (rework spec §8.5): $TARGET$ deploys next to us — its weapons
# based in a country adjacent to us (Nuclear Basing, §7.3), its nuclear
# guarantee over one, or its missile defense in a state next to ours. The
# articles are read live: the same ones the concession leaves, so a day after
# it the self-check sees the result. Adjacency is is_adjacent_to_country, not
# strategic adjacency, which a war goal would stretch to anyone (rulings R4).
# An overlord's umbrella over a subject is not a deployment: every armed
# overlord's subjects are under it by default (R2).
nd_dispute_deployment_by = {
	save_temporary_scope_as = nd_dd_self
	OR = {
		$TARGET$ = {
			any_scope_treaty = {
				any_scope_article = {
					OR = {
						has_type = nuclear_basing
						has_type = nuclear_guarantee
					}
					source_country = $TARGET$
					target_country = {
						NOT = { THIS = scope:nd_dd_self }
						is_adjacent_to_country = scope:nd_dd_self
					}
				}
			}
		}
		$TARGET$ = {
			any_scope_state = {
				nd_state_has_missile_defense = yes
				any_neighbouring_state = { owner = scope:nd_dd_self }
			}
		}
	}
}

# State scope: a military base here runs a Missile Defense method (the
# battery, or directed-energy point defense in the same slot).
nd_state_has_missile_defense = {
	any_scope_building = {
		is_building_type = building_military_base
		OR = {
			has_active_production_method = pm_missile_defense_battery
			has_active_production_method = pm_directed_energy_point_defense
		}
	}
}

# Dispute 8 (rework spec §8.5): $TARGET$ claims one of our incorporated
# states. A claim sits on the state region, so the claimer's has_claim is
# tested against the region, not has_claim_by on our part of a split state
# (the decentralized-claims sweep's pattern, extra_on_actions.txt).
nd_dispute_claims_by = {
	any_scope_state = {
		is_incorporated = yes
		state_region = { $TARGET$ = { has_claim = PREV } }
	}

# ---- The nuclear umbrella (umbrella/recessed/dead-hand spec §1) -------------
# A direct subject of an overlord believed armed is covered as if that
# overlord had signed a nuclear_guarantee for it, unless the overlord has
# withdrawn it (nd_withdraw_umbrella_action, a pact). Coverage by umbrella is
# coverage by guarantee: it is folded into nd_has_armed_guarantor_against,
# nd_is_guaranteed and nd_is_guaranteed_by, and the three effect loops that
# walk guarantee articles (nd_crisis_notify_guarantors, nd_record_nuclear_use,
# nd_guarantee_act_abandon) each have an umbrella branch beside the treaty one.
# test_nuclear_deterrence.TestUmbrellaCoverage fails if a new article reader
# appears anywhere else.

# Has our overlord withdrawn its umbrella from us? The pact exists only
# between an overlord and its direct subject (its requirement_to_maintain);
# we are its second country. A withdrawal we made from a subject of our own
# is not one made from us.
nd_umbrella_withdrawn = {
	save_temporary_scope_as = nd_uw_self
	any_scope_diplomatic_pact = {
		is_diplomatic_action_type = nd_withdraw_umbrella_action
		second_country = { this = scope:nd_uw_self }
	}
}

# Is this country under its (direct) overlord's umbrella? Not while the two
# are at war or on opposite sides of a play — a subject fighting for its
# freedom is not protected by the arsenal it is fighting, as a treaty
# guarantee lapses in a war between its parties.
nd_under_an_umbrella = {
	is_subject = yes
	save_temporary_scope_as = nd_ua_self
	overlord ?= {
		nd_believed_armed = yes
		NOT = { has_war_with = scope:nd_ua_self }
		NOT = { is_diplomatic_play_enemy_of = scope:nd_ua_self }
	}
	NOT = { nd_umbrella_withdrawn = yes }
}

# ...under $OVERLORD$'s in particular?
nd_under_umbrella_of = {
	is_direct_subject_of = $OVERLORD$
	nd_under_an_umbrella = yes
}

# The Armed Forces are professional officers and the Industrialists business,
# whatever their politics; every other group is militarist or restraint by
# its Rules of War stance, or has no view. A group's stance already folds in
# its leader's ideology, so a fascist-led Intelligentsia is militarist with no
# special case. Strongly approving is a full view (±2); approving is a mild
# one (±1).
#
# The Armed Forces' and Industrialists' LEAN reads their leader's own stance,
# not the group's: the Armed Forces' core ideology (patriotic) strongly
# approves Total War, so a group-stance lean would make every army hawkish and
# the officers' own view would never apply. A jingoist-led army leans
# militarist, a pacifist-led one restraint, a moderate-led one neither.
#
# The nd_stance_* tests work in interest-group or character scope (law_stance
# reads the base stance of either). law_stance compares one law at a time, so
# "its stance on Total War beats its stance on Limited War" is spelled out.
# law_stance can also return a value above strongly_approve (vanilla guards
# `value > strongly_approve`, "equals count"); it is never counted as approval.
# Every ideology that names lawgroup_rules_of_war names both laws, which these
# comparisons rely on.

nd_stance_militarist_full = {
	law_stance = {
		law = law_type:law_total_war
		value > approve
	}
	NOT = {
		law_stance = {
			law = law_type:law_total_war
			value > strongly_approve
		}
	}
	NOT = {
		law_stance = {
			law = law_type:law_limited_war
			value > approve
		}
	}
}

nd_stance_militarist_mild = {
	law_stance = {
		law = law_type:law_total_war
		value > neutral
	}
	NOT = {
		law_stance = {
			law = law_type:law_total_war
			value > approve
		}
	}
	NOT = {
		law_stance = {
			law = law_type:law_limited_war
			value > neutral
		}
	}
}

nd_stance_restraint_full = {
	law_stance = {
		law = law_type:law_limited_war
		value > approve
	}
	NOT = {
		law_stance = {
			law = law_type:law_limited_war
			value > strongly_approve
		}
	}
	NOT = {
		law_stance = {
			law = law_type:law_total_war
			value > approve
		}
	}
}

nd_stance_restraint_mild = {
	law_stance = {
		law = law_type:law_limited_war
		value > neutral
	}
	NOT = {
		law_stance = {
			law = law_type:law_limited_war
			value > approve
		}
	}
	NOT = {
		law_stance = {
			law = law_type:law_total_war
			value > neutral
		}
	}
}

nd_stance_militarist = {

# The 24-month cooldown after a crisis with the same country (anti-farming,
# §7). Waived in war, and after a crisis that ended in $TARGET$'s nuclear
# strike (outcome 7) while the strike's record stands: the strike is the
# grievance (dispute 6), not a crisis chosen to farm, as nuclear_incident.5's
# option c skips it for a recalled launch. Once a crisis over the strike has
# closed, its outcome replaces 7 and the cooldown holds again.
nd_crisis_cooldown_clear = {
	OR = {
		has_war_with = $TARGET$
		AND = {
			var:nd_crisis_last_outcome ?= 7
			var:nd_crisis_last_opponent ?= $TARGET$
			nd_dispute_peacetime_strike_by = { TARGET = $TARGET$ }
		}
		NOT = { has_variable = nd_crisis_cooldown_target }
		NOT = { var:nd_crisis_cooldown_target ?= $TARGET$ }
		NOT = { has_variable = nd_crisis_cooldown_months }
		var:nd_crisis_cooldown_months <= 0
	}
}

# Dispute 1: we are enemies in a diplomatic play (not yet a war) that $TARGET$

	}
}

	NOT = { has_war_with = $TARGET$ }

	}

# Could $GUARANTOR$ offer this country a nuclear guarantee against
# $PROLIFERATOR$'s nearly finished bomb (nuclear_weapon_events.18 option g,
# #804)? Read when the offer is made and again when it is answered (.25).
# Scope: the country offered it.
#   * a neighbour of the proliferator, not the proliferator's subject;
#   * one that needs it: no bomb of its own as the world sees it, and no
#     guarantee or umbrella already (guarantors do not add up);
#   * every gate a new guarantee passes in the article
#     (115_nuclear_guarantee.txt), read without a draft. possible's: the
#     guarantor armed, the two at peace, relations above Poor, not the
#     guarantor's direct subject (its umbrella covers those). can_ratify's:
#     no guarantee from it already (the line above), military ties in force
#     (an alliance or defensive pact, its Guarantee Independence of us, or a
#     shared power bloc), and outgunning us as the world sees it, which an
#     armed guarantor of a country not believed armed always does
#     (nd_believed_stockpile is 0). Keep these in step with the article;
#   * and the engine's own word on the treaty .25 creates
#     (nd_bp_can_treaty_guarantee, the Budapest offer's guarantee).
nd_proliferation_guarantee_candidate = {
	save_temporary_scope_as = nd_pg_beneficiary
	$GUARANTOR$ = { save_temporary_scope_as = nd_pg_guarantor }
	NOT = { this = scope:nd_pg_guarantor }
	NOT = { this = $PROLIFERATOR$ }
	NOT = { is_country_type = decentralized }
	any_neighbouring_state = { owner = $PROLIFERATOR$ }
	NOT = { is_subject_of = $PROLIFERATOR$ }
	nd_believed_armed = no
	nd_is_guaranteed = no
	NOT = { is_direct_subject_of = scope:nd_pg_guarantor }
	NOT = { has_war_with = scope:nd_pg_guarantor }
	scope:nd_pg_guarantor = {
		nd_is_armed = yes
		# A deployment conceded in a crisis bars a guarantee next to the
		# country conceded to, as the article's can_ratify does (dispute 7).
		NOT = { var:nd_crisis_deploy_bar ?= { is_adjacent_to_country = scope:nd_pg_beneficiary } }
		scope:nd_pg_beneficiary.relations:this > relations_threshold:poor
		OR = {
			any_scope_treaty = {
				binds = scope:nd_pg_beneficiary
				any_scope_article = {
					OR = {
						has_type = alliance
						has_type = defensive_pact
						AND = {
							has_type = guarantee_independence
							source_country = scope:nd_pg_guarantor
							target_country = scope:nd_pg_beneficiary
						}
					}
				}
			}
			AND = {
				is_in_power_bloc = yes
				is_in_same_power_bloc = scope:nd_pg_beneficiary
			}
		}
	}
	nd_bp_can_treaty_guarantee = { GUARANTOR = scope:nd_pg_guarantor STATE = scope:nd_pg_beneficiary }
}

# ----------------------------------------------------------------------------
# AI JUDGEMENT ON NUCLEAR USE (§9) — ROOT = the AI deciding
# ----------------------------------------------------------------------------
# Read by both strike actions' `will_propose`, on top of the doctrine and law
# gates in `possible`. Replaces the old universal severe-threat gate: a
# warfighting power that nobody can answer may use nuclear weapons in an
# ordinary war it is winning; a flexible one when it is losing; everyone for
# survival or retaliation. Perceived retaliation (their arsenal or an armed
# guarantor) and a six-month pause between first uses bound it.
nd_ai_nuclear_use_justified = {
	nd_is_armed = yes
	nd_reaches_country = { TARGET = $ENEMY$ }
	# Our own civil war: only when losing badly under laws that already
	# silence dissent — retaliation included (owner ruling 2026-09-25;
	# nuclear_custody_triggers.txt).
	OR = {
		NOT = { nd_is_civil_war_counterpart = { ENEMY = $ENEMY$ } }
		nd_ai_would_strike_own_people = yes
	}
	OR = {
		# Retaliation is always on the table.
		nd_was_struck_by = { ENEMY = $ENEMY$ }
		AND = {
			# Keep a warhead in reserve unless survival is at stake.
			OR = {
				var:nuclear_weapon_stockpile >= 2
				nd_enemy_threatens_existence = { ENEMY = $ENEMY$ }
			}
			NOT = { has_variable = nd_first_use_cooldown }
			# The nuclear taboo (nuclear_taboo_values.txt): at 70 or more,
			# first use only for national survival, whatever the doctrine.
			OR = {
				nd_taboo_value < nd_taboo_ai_survival_only_line
				nd_enemy_threatens_existence = { ENEMY = $ENEMY$ }
			}
			OR = {
				# National survival.
				nd_enemy_threatens_existence = { ENEMY = $ENEMY$ }
				AND = {
					nd_core_threatened_by = { ENEMY = $ENEMY$ }
					is_losing_war_against = { ENEMY = $ENEMY$ }
				}
				# Flexible first use: a war we are losing, against an enemy that
				# cannot answer — or with a reckless leader regardless.
				AND = {
					nd_doctrine_at_least_flexible = yes
					is_losing_war_against = { ENEMY = $ENEMY$ }
					OR = {
						NOT = { nd_faces_retaliation_from = { ENEMY = $ENEMY$ } }
						ruler_is_aggressive = yes
					}
				}
				# Compellence: a defied ultimatum, against an enemy that cannot answer.
				AND = {
					nd_doctrine_at_least_compellence = yes
					nd_was_defied_by = { ENEMY = $ENEMY$ }
					NOT = { nd_faces_retaliation_from = { ENEMY = $ENEMY$ } }
				}
				# Warfighting: any war against an enemy that cannot answer.
				AND = {
					nd_doctrine_warfighting = yes
					NOT = { nd_faces_retaliation_from = { ENEMY = $ENEMY$ } }
				}
			}
			# A cautious ruler only for survival — unless the world treats
			# the bomb as an ordinary weapon (the nuclear taboo below 30).
			OR = {
				ruler_is_cautious = no
				nd_enemy_threatens_existence = { ENEMY = $ENEMY$ }
				nd_taboo_value < nd_taboo_ai_loosen_line
			}
		}
	}
}

# ----------------------------------------------------------------------------
# AI JUDGEMENT ON OPENING A CRISIS (§9) — ROOT = the AI, $TARGET$ the country
# ----------------------------------------------------------------------------
# A warning is worth issuing when there is something the threat can move and
# the target is unlikely to answer in kind. Defending a beneficiary or our own
# survival needs no licence beyond an arsenal; demanding needs a hawkish
# doctrine or regime.
nd_ai_would_warn = {
	nd_is_armed = yes
	in_default = no
	OR = {
		# Protecting a country we guarantee.
		nd_dispute_guarantee_against = { TARGET = $TARGET$ }
		# A strike on us, or on a country we cover, outside a war (dispute 6).
		nd_dispute_peacetime_strike_by = { TARGET = $TARGET$ }
		# Our survival or our core is at stake in a play or war with them.
		AND = {
			OR = {
				nd_dispute_play_with = { TARGET = $TARGET$ }
				nd_dispute_war_with = { TARGET = $TARGET$ }
			}
			OR = {
				nd_enemy_threatens_existence = { ENEMY = $TARGET$ }
				nd_core_threatened_by = { ENEMY = $TARGET$ }
				$TARGET$ = { is_diplomatic_play_initiator = yes }
			}
		}
		# Coercion: a hawkish doctrine or regime against a target that cannot
		# answer, over a play or a war.
		AND = {
			OR = {
				nd_dispute_play_with = { TARGET = $TARGET$ }
				nd_dispute_war_with = { TARGET = $TARGET$ }
			}
			OR = {
				nd_doctrine_at_least_compellence = yes
				nd_regime_is_militarist = yes
			}
			NOT = { nd_faces_retaliation_from = { ENEMY = $TARGET$ } }
			OR = {
				nd_threat_backed = { TARGET = $TARGET$ }
				ruler_is_aggressive = yes
			}
		}
		# A rival's bomb, for a great power that has one.
		AND = {
			nd_dispute_proliferation_by = { TARGET = $TARGET$ }
			country_rank >= rank_value:great_power
			nd_doctrine_nfu = no
		}
		# Forces next to us, our land, or operations against us, from a rival
		# (rework spec §8.5; rulings R8: claims and covert operations are
		# common, so only a rival's). Only when nothing weightier is at stake
		# between us, since the crisis would then be about that, past its own
		# licences; and never as a bluff, which an Existential or No First
		# Use doctrine makes of these.
		AND = {
			has_diplomatic_pact = { who = $TARGET$ type = rivalry }
			OR = {
				nd_dispute_deployment_by = { TARGET = $TARGET$ }
				nd_dispute_claims_by = { TARGET = $TARGET$ }
				nd_dispute_covert_ops_by = { TARGET = $TARGET$ }
			}
			NOT = { nd_dispute_weightier_than_7 = { TARGET = $TARGET$ } }
			OR = {
				nd_threat_backed = { TARGET = $TARGET$ }
				nd_threat_uncertain = { TARGET = $TARGET$ }
			}
		}
		# An orbital battlestation going up, for a great power (rework spec
		# §9), against a country it is hostile to, when the crisis would be
		# about the battlestation (no weightier dispute, and none of 7–9,
		# which classify first) and isn't a bluff.
		AND = {
			country_rank >= rank_value:great_power
			nd_dispute_orbital_by = { TARGET = $TARGET$ }
			OR = {
				has_diplomatic_pact = { who = $TARGET$ type = rivalry }
				has_attitude = { who = $TARGET$ attitude = antagonistic }
				has_attitude = { who = $TARGET$ attitude = belligerent }
				has_attitude = { who = $TARGET$ attitude = domineering }
			}
			NOT = { nd_dispute_weightier_than_7 = { TARGET = $TARGET$ } }
			NOT = { nd_dispute_deployment_by = { TARGET = $TARGET$ } }
			NOT = { nd_dispute_claims_by = { TARGET = $TARGET$ } }
			NOT = { nd_dispute_covert_ops_by = { TARGET = $TARGET$ } }
			OR = {
				nd_threat_backed = { TARGET = $TARGET$ }
				nd_threat_uncertain = { TARGET = $TARGET$ }
			}
		}
	}
	# A cautious ruler warns only in defence.
	OR = {
		ruler_is_cautious = no
		nd_dispute_guarantee_against = { TARGET = $TARGET$ }
		nd_dispute_peacetime_strike_by = { TARGET = $TARGET$ }
		nd_enemy_threatens_existence = { ENEMY = $TARGET$ }
	}
	# Stakes (rework spec §8.2): not a threat whose demands alone would hold
	# the pressure on them below the line where an AI concedes. Last, since it
	# computes the pressure twice; not for a guarantee, whose demands are the
	# protégé's.
	OR = {
		nd_dispute_guarantee_against = { TARGET = $TARGET$ }
		nd_dispute_peacetime_strike_by = { TARGET = $TARGET$ }
		NOT = { nd_ai_stakes_block = { TARGET = $TARGET$ } }
	}
}

# here as temporary scopes; nd_ai_would_warn's callers (the opening actions'

}

# Either party: the crisis is over a dispute that can trade.

# A trade holds while each side holds the same grievance against the other:
# the dispute $ISSUER$'s record is about still holds against $TARGET$, and
# $TARGET$ holds the same code against $ISSUER$. Read when a trade is offered
# and again when it is accepted, since an offer can go stale before its answer.
# Neither side may hold that kind's ten-year bar from an earlier crisis (7, 9
# and 10; claims have none): a bar holds one country, so the trade's own bar
# would overwrite one held against a third country and free its holder.
nd_crisis_trade_holds = {
	exists = $ISSUER$
	exists = $TARGET$
	OR = {
		AND = {
			$ISSUER$ = {
				has_variable = nd_crisis_dispute
				var:nd_crisis_dispute = 7
			}
			$ISSUER$ = { nd_dispute_deployment_by = { TARGET = $TARGET$ } }
			$TARGET$ = { nd_dispute_deployment_by = { TARGET = $ISSUER$ } }
			NOT = { $ISSUER$ = { has_variable = nd_crisis_deploy_bar } }
			NOT = { $TARGET$ = { has_variable = nd_crisis_deploy_bar } }
		}
		AND = {
			$ISSUER$ = {
				has_variable = nd_crisis_dispute
				var:nd_crisis_dispute = 8
			}
			$ISSUER$ = { nd_dispute_claims_by = { TARGET = $TARGET$ } }
			$TARGET$ = { nd_dispute_claims_by = { TARGET = $ISSUER$ } }
		}
		AND = {
			$ISSUER$ = {
				has_variable = nd_crisis_dispute
				var:nd_crisis_dispute = 9
			}
			$ISSUER$ = { nd_dispute_covert_ops_by = { TARGET = $TARGET$ } }
			$TARGET$ = { nd_dispute_covert_ops_by = { TARGET = $ISSUER$ } }
			NOT = { $ISSUER$ = { has_variable = nd_crisis_covert_bar } }
			NOT = { $TARGET$ = { has_variable = nd_crisis_covert_bar } }
		}
		AND = {
			$ISSUER$ = {
				has_variable = nd_crisis_dispute
				var:nd_crisis_dispute = 10
			}
			$ISSUER$ = { nd_dispute_orbital_by = { TARGET = $TARGET$ } }
			$TARGET$ = { nd_dispute_orbital_by = { TARGET = $ISSUER$ } }
			NOT = { $ISSUER$ = { has_variable = nd_crisis_orbital_freeze } }
			NOT = { $TARGET$ = { has_variable = nd_crisis_orbital_freeze } }
		}
	}
}

# Either party, parties saved, in a deployment crisis: our half of a trade
# would leave a treaty that also holds an alliance or a defensive pact of ours
# (a treaty is left whole). The target's half is nd_concession_costs_an_alliance;
# the issuer's is the same read the other way. An AI keeps an ally.
nd_trade_costs_us_an_alliance = {
	nd_crisis_dispute_is = { D = 7 }
	exists = scope:nd_issuer
	exists = scope:nd_target
	OR = {
		AND = {
			THIS = scope:nd_target
			nd_concession_costs_an_alliance = yes
		}
		AND = {
			THIS = scope:nd_issuer
			any_scope_treaty = {
				any_scope_article = {
					OR = {
						has_type = nuclear_basing
						has_type = nuclear_guarantee
					}
					source_country = scope:nd_issuer
					target_country = {
						NOT = { THIS = scope:nd_target }
						is_adjacent_to_country = scope:nd_target
					}
				}
				any_scope_article = {
					OR = {
						has_type = alliance
						has_type = defensive_pact
					}
				}
			}
		}
	}
}

# The AI's answer to a trade (nuclear_crisis.8; rework spec §8.6, ruling R4),
# parties saved: a target accepts at the pressure where it would start to
# yield (nd_ai_concession_line), an issuer once the crisis is dangerous
# (nd_ai_trade_danger_line) and its threat has run to a first deadline
# (nd_ai_trade_min_weeks): a public ultimatum opens near danger 40, so danger
# alone let a target trade out in its first week. Each accepts only if it gives no more than it
# gets (nd_trade_balance_value) and loses no ally by its own half.
nd_ai_accepts_trade = {
	exists = scope:nd_issuer
	exists = scope:nd_target
	nd_crisis_trade_holds = { ISSUER = scope:nd_issuer TARGET = scope:nd_target }
	NOT = { nd_trade_costs_us_an_alliance = yes }
	OR = {
		AND = {
			THIS = scope:nd_target
			has_variable = nd_yield_pressure
			var:nd_yield_pressure >= nd_ai_concession_line
			nd_trade_balance_value <= 0
		}
		AND = {
			THIS = scope:nd_issuer
			has_variable = nd_crisis_danger
			var:nd_crisis_danger >= nd_ai_trade_danger_line
			var:nd_crisis_weeks >= nd_ai_trade_min_weeks
			nd_trade_balance_value >= 0
		}
	}
}

# Country scope, in a crisis: nd_crisis_trade_holds, the parties read from our
# own record (the panel and the deadline event, where no party scopes need be
# saved). A temporary scope saved inside a `?=` holds for the rest of the
# trigger (zz_te_war_support_injections.txt, wars of independence).
nd_crisis_trade_holds_for_us = {
	OR = {
		AND = {
			nd_is_crisis_owner = yes
			save_temporary_scope_as = nd_tr_issuer
			var:nd_crisis_opponent ?= { save_temporary_scope_as = nd_tr_target }
			nd_crisis_trade_holds = { ISSUER = scope:nd_tr_issuer TARGET = scope:nd_tr_target }
		}
		AND = {
			nd_is_crisis_target = yes
			save_temporary_scope_as = nd_tr_target
			var:nd_crisis_opponent ?= { save_temporary_scope_as = nd_tr_issuer }
			nd_crisis_trade_holds = { ISSUER = scope:nd_tr_issuer TARGET = scope:nd_tr_target }
		}
	}
}
