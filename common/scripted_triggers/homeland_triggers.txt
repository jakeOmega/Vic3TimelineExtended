# State scope; ROOT remains the state during candidate discovery.
te_homeland_region_unified = {
	NOT = {
		state_region = {
			any_scope_state = { NOT = { owner = ROOT.owner } }
		}
	}
}
te_homeland_system_enabled = {
	owner = { modifier:country_homelands_can_change_bool = yes }
}

# State scope. Whether a track applies here at all, whatever its thresholds:
# a primary culture without a homeland here could gain one; a non-primary
# culture with a homeland here could lose it.
te_homeland_creation_culture_present = {
	any_scope_culture = {
		is_primary_culture_of = ROOT.owner
		NOT = { has_homeland = ROOT }
	}
}
te_homeland_removal_culture_present = {
	any_scope_culture = {
		NOT = { is_primary_culture_of = ROOT.owner }
		has_homeland = ROOT
	}
}
# State scope. Removal can happen here at all: at the base removal threshold
# of 0% no culture's share can ever fall below it.
te_homeland_removal_threshold_open = {
	modifier:state_homeland_removal_threshold_add > 0
}

# Culture scope; ROOT = state.
te_homeland_creation_candidate = {
	is_primary_culture_of = ROOT.owner
	NOT = { has_homeland = ROOT }
	ROOT = {
		culture_percent_state = { target = PREV value >= homeland_culture_form_threshold }
	}
}
te_homeland_creation_has_candidate = {
	any_scope_culture = { te_homeland_creation_candidate = yes }
}
# Container scope; this also works when GUI evaluation makes ROOT the container.
te_homeland_creation_project_valid = {
	var:te_homeland_state = { owner = PREV.var:te_homeland_owner }
	var:te_homeland_culture = {
		is_primary_culture_of = PREV.var:te_homeland_state.owner
		NOT = { has_homeland = PREV.var:te_homeland_state }
		PREV.var:te_homeland_state = {
			culture_percent_state = { target = PREV value >= homeland_culture_form_threshold }
		}
	}
}

# Culture scope; ROOT = state.
# A culture merge's switched part is not a candidate while the state is also a homeland of the culture it
# switched into (te_cmg_switched_part_kept, generated; owner, 2026-11-07): the part keeps its homeland beside
# its successor's, and loses it with the successor's (te_cmg_remove_switched_part_homelands, called where
# te_homeland_removal_monthly_effect removes a homeland). Last, so the cheap tests prune first.
te_homeland_removal_candidate = {
	NOT = { is_primary_culture_of = ROOT.owner }
	has_homeland = ROOT
	ROOT = {
		culture_percent_state = { target = PREV value < homeland_culture_disappear_threshold }
	}
	NOT = { te_cmg_switched_part_kept = yes }
}
te_homeland_removal_has_candidate = {
	any_scope_culture = { te_homeland_removal_candidate = yes }
}
# Container scope; this also works when GUI evaluation makes ROOT the container.
te_homeland_removal_project_valid = {
	var:te_homeland_state = { owner = PREV.var:te_homeland_owner }
	var:te_homeland_culture = {
		NOT = { is_primary_culture_of = PREV.var:te_homeland_state.owner }
		has_homeland = PREV.var:te_homeland_state
		PREV.var:te_homeland_state = {
			culture_percent_state = { target = PREV value < homeland_culture_disappear_threshold }
		}
	}
}

# Container scope. Dispatch to the appropriate culture eligibility rule.
te_homeland_project_valid = {
	OR = {
		AND = { has_tag = te_homeland_creation te_homeland_creation_project_valid = yes }
		AND = { has_tag = te_homeland_removal te_homeland_removal_project_valid = yes }
	}
}

# Any scope. No state anywhere is a homeland of cu:$CULTURE$. Vanilla's Ukrainian
# and Byelorussian Cultural Renaissances open on fervor alone and involve the
# owners of the culture's homeland states (should_be_involved), so once the
# dynamic homelands above have removed every one, the entry opens and runs for
# no one. te_vanilla_cultural_renaissances.txt ends them on this (Workshop
# report, 2026-11-08).
te_homeland_none_left = {
	NOT = {
		any_country = {
			any_scope_state = {
				state_region = { is_homeland = cu:$CULTURE$ }
			}
		}
	}
}
