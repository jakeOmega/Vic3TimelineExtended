# Sophisticated `possible`/`ai_will_do` gating for the major-unification
# formables in common/country_formation/te_formable_countries.txt.
#
# The vanilla rank gate blocks minor formations once a country is at the
# formable's tier or above. Pan-region political unions (AFU/EUN/UNA/UNE/INM/
# EAF/DAR) are thematically formed by countries that are already powerful, so
# they are `is_major_formation = yes` to bypass that gate. The triggers
# below replace the rank gate with world-state conditions so the unification
# only fires when it would feel earned, not whenever the rank check passes.
#
# Three load-bearing conditions per regional union (EU/AFU/UNA/INM/EAF/DAR):
#   1. External rank pressure — at least one external great power that is
#      globally top-3, i.e. a real outside force the region would unify
#      *against*. (A rank-8 external GP is not real pressure.)
#   2. Bloc / alliance saturation — every major regional constituent is
#      already in ROOT's power bloc, alliance, defensive pact, or customs
#      union. Rivals don't unify; aligned partners do.
#   3. Absence of intra-region rivalry / active war between ROOT and any
#      major regional constituent.
#
# `is_major_<region>_power` is dynamic (capital state in the region +
# country_rank >= major_power), so the trigger composition survives
# formed countries and shifting borders.
#
# UNE (United Earth) gates differently — there is no external bloc to be
# outweighed by, and the WW3 system is optional content the user does not
# want to hard-require. Instead UNE asks for a global supermajority: ROOT
# leads a power bloc that contains essentially all great powers (≤1 GP
# remains outside), plus alignment / no-rivalry checks against all other
# great powers.

# === Constituency: "major <region> power" =====================================

is_major_european_power = {
	country_rank >= rank_value:major_power
	capital = { is_in_geographic_region = geographic_region_united_europe }
}

is_major_african_power = {
	country_rank >= rank_value:major_power
	capital = { is_in_geographic_region = geographic_region_united_africa }
}

is_major_north_american_power = {
	country_rank >= rank_value:major_power
	capital = { is_in_geographic_region = geographic_region_united_north_america }
}

is_major_intermarium_power = {
	country_rank >= rank_value:major_power
	capital = { is_in_geographic_region = geographic_region_intermarium }
}

is_major_east_african_power = {
	country_rank >= rank_value:major_power
	capital = { is_in_geographic_region = geographic_region_east_african_federation }
}

is_major_dar_al_islam_power = {
	country_rank >= rank_value:major_power
	capital = { is_in_geographic_region = geographic_region_dar_al_islam }
}

# === Shared helpers ============================================================

# Aligned with ROOT via power-bloc co-membership OR a treaty pact OR customs
# union. Subject relationships and ROOT itself short-circuit to aligned.
is_aligned_with_root = {
	OR = {
		this = ROOT
		is_subject_of = ROOT
		is_in_same_power_bloc = ROOT
		has_treaty_alliance_with = { TARGET = ROOT }
		has_treaty_defensive_pact_with = { TARGET = ROOT }
		is_in_customs_union_with = ROOT
	}
}

# In a rivalry or active war with ROOT (intra-region friction blocker).
is_at_friction_with_root = {
	OR = {
		has_war_with = ROOT
		has_diplomatic_pact = { who = ROOT type = rivalry }
	}
}

# Performance (2026-10-03 script profile: the AFU formation's possible cost
# 0.29 ms a call): each any_country below drops `count >= 1`, which means the
# same as a bare any_country and may make the engine count every country
# instead of stopping at the first match.
#
# External pressure is read from a global variable the world's monthly pulse
# sets (te_formable_refresh_external_pressure, te_formable_effects.txt), not
# walked per call: it does not depend on ROOT, and in the 2026-11-12 profile of
# a 1931 game the AFU formation's possible cost 0.47 ms a call, 1,890 calls in
# 33 days, and INM's and EAF's about 0.5 ms (docs/audits/late_game_performance.md;
# the profile does not say how much of that is this walk and how much the two
# below). A great power entering or leaving the top three takes effect at the
# next monthly pulse, which the tooltip says.

# === EU (European Union) ======================================================

eu_formation_external_pressure = {
	custom_tooltip = {
		text = eu_formation_external_pressure_tt
		has_global_variable = te_formable_pressure_eu
	}
}

eu_formation_bloc_aligned = {
	NOT = {
		any_country = {
			is_major_european_power = yes
			NOT = { is_aligned_with_root = yes }
		}
	}
}

eu_formation_no_internal_rivalry = {
	NOT = {
		any_country = {
			is_major_european_power = yes
			is_at_friction_with_root = yes
		}
	}
}

# === UNE (United Earth) =======================================================
# WW3 not used as a gate (optional content per user). The supermajority gate
# is what makes UNE-formation feel earned: ROOT leads a bloc that contains
# essentially every great power on earth.

# The four gates below name the great powers in their tooltips (list_scopes
# and show_trigger_per_scope, vanilla 1.15): each is a positive count over the
# great powers other than ROOT (`filter`, as vanilla's tsar buttons write it)
# rather than "no great power that fails" (vanilla lists only in positive
# counts; whether a NOT-wrapped count lists is untested). Each
# is exactly the NOT form it replaced; test_une_formation_gates.py pins that.
# A count over no great power at all is never asked: trigger_if passes when
# there is nothing to count, as the NOT form did.

une_formation_supermajority = {
	is_power_bloc_leader = yes
	# At most one great power exists outside ROOT's power bloc: at least all
	# the others but one are inside it (te_formation_gating_values.txt).
	trigger_if = {
		limit = {
			any_country = {
				country_rank >= rank_value:great_power
				NOT = { this = ROOT }
				count >= 2
			}
		}
		trigger_if = {
			limit = { country_rank >= rank_value:great_power }
			any_country = {
				filter = {
					country_rank >= rank_value:great_power
					NOT = { this = ROOT }
				}
				is_in_same_power_bloc = ROOT
				count >= une_supermajority_needed_gp_candidate
				list_scopes = yes
				show_trigger_per_scope = yes
			}
		}
		trigger_else = {
			any_country = {
				filter = {
					country_rank >= rank_value:great_power
					NOT = { this = ROOT }
				}
				is_in_same_power_bloc = ROOT
				count >= une_supermajority_needed_other_candidate
				list_scopes = yes
				show_trigger_per_scope = yes
			}
		}
	}
}

une_formation_bloc_aligned = {
	# Any great power not already in ROOT's bloc must be aligned via treaty
	# or customs union (i.e. friendly outliers, not rivals).
	trigger_if = {
		limit = {
			any_country = {
				country_rank >= rank_value:great_power
				NOT = { this = ROOT }
			}
		}
		any_country = {
			filter = {
				country_rank >= rank_value:great_power
				NOT = { this = ROOT }
			}
			OR = {
				is_in_same_power_bloc = ROOT
				is_aligned_with_root = yes
			}
			count = all
			list_scopes = yes
			show_trigger_per_scope = yes
		}
	}
}

une_formation_no_internal_rivalry = {
	trigger_if = {
		limit = {
			any_country = {
				country_rank >= rank_value:great_power
				NOT = { this = ROOT }
			}
		}
		any_country = {
			filter = {
				country_rank >= rank_value:great_power
				NOT = { this = ROOT }
			}
			NOT = { is_at_friction_with_root = yes }
			count = all
			list_scopes = yes
			show_trigger_per_scope = yes
		}
	}
}

# A world government needs a world order already in place. With the UN game
# rule on, that is the United Nations at the Supranational tier of UN
# authority (85+, which only Charter Reform II opens). With it off there is no
# UN to ask, so ROOT's bloc must stand in for it: no great power outside it,
# where une_formation_supermajority allows one aligned outsider. Also gates
# dp_unify_united_earth, so a unification play cannot form UNE around it.
# Whether the list shows inside the custom_tooltip is the C10 question
# (docs/audits/vanilla_1_15_prep.md); the logic is the same either way.
# Scope: country (ROOT is the candidate or the play's initiator)
une_formation_world_order = {
	trigger_if = {
		limit = { has_game_rule = united_nations_enabled }
		custom_tooltip = {
			text = une_formation_un_supranational_tt
			un_tier_is_supranational = yes
		}
	}
	trigger_else = {
		custom_tooltip = {
			text = une_formation_bloc_unanimity_tt
			trigger_if = {
				limit = {
					any_country = {
						country_rank >= rank_value:great_power
						NOT = { this = ROOT }
					}
				}
				any_country = {
					filter = {
						country_rank >= rank_value:great_power
						NOT = { this = ROOT }
					}
					is_in_same_power_bloc = ROOT
					count = all
					list_scopes = yes
					show_trigger_per_scope = yes
				}
			}
		}
	}
}

# === Vanilla's Turtle Island JE must not reach the unions =====================
# Vanilla je_unite_the_nations shows for any North American country with an
# indigenous-heritage primary culture while TIS does not exist, and its
# "Proclaim Turtle Island" button is a bare change_tag = TIS. A union formed
# from a Turtle Island (or holding Métis) qualifies, the tag change deletes the
# union, and the same country forms the union again: UNA and TIS alternated in
# play. UNA and UNE are the two unions that can hold a North American capital;
# the other five bind candidacy to a capital in their own region.
# Consumed by the REPLACE in common/journal_entries/te_vanilla_turtle_island.txt.
te_is_north_american_capable_union = {
	OR = {
		c:UNA ?= this
		c:UNE ?= this
	}
}
