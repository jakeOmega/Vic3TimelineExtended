# ============================================================================
# COLONIAL GEOGRAPHY — the separation test's sea crossings and fill links
#
# A colony is "geographically separate" from the country that rules it (UN
# General Assembly Resolution 1541 (XV), Principle IV), which the UN read as
# separated by sea. In the game: a state of ours not joined to our capital over
# land through states we own (is_separated_from_capital,
# colonial_empire_triggers.txt). No trigger answers that, so a monthly ring
# fill works it out and caches it on our states (ce_refresh_separation,
# common/scripted_effects/colonial_geography_effects.txt); these triggers are
# the fill's tests.
#
# STRAITS. Vanilla's map joins 26 pairs of states by a sea crossing
# (adjacencies.csv, read into scripts/analysis/demographics_state_adjacency.json
# under "strait"). A crossing joins two states only when both are homelands of
# our primary cultures: Constantinople reaches Anatolia, Tokyo Kyushu and
# Hokkaido, Copenhagen Jutland, Rome Sicily; Spanish Morocco, Ireland, Ryukyu
# and Sakhalin are separated. Whether the engine's any_neighbouring_state
# crosses a strait is unverified (the demographics self-check
# neighbouring_state_reads logs strait=0|1), so the fill does both halves and
# gives the same answer either way: it refuses a crossing the iterator may
# return unless both ends are our homelands, and adds a crossing between two
# homelands itself. test_colonial_geography.py pins the two lists below to the
# JSON's pairs.
#
# SCOPE: state.
# ============================================================================

# The fill's link from a marked neighbour (this) into scope:ce_strait_other, an
# unmarked state of ours: a land border, or a crossing between two of our
# homelands. scope:ce_fill_country is the country filling.
ce_fill_land_link = {
	OR = {
		NOT = { has_variable = ce_strait_end }
		scope:ce_strait_other = { NOT = { has_variable = ce_strait_end } }
		NOT = { ce_strait_pair_with_other = yes }
		AND = {
			is_homeland_of_country_cultures = scope:ce_fill_country
			scope:ce_strait_other = { is_homeland_of_country_cultures = scope:ce_fill_country }
		}
	}
}

# A crossing from one of our marked homelands into this unmarked homeland of
# ours (scope:ce_strait_other is this state, saved by the caller). The fill's
# own half of the strait rule, for an iterator that does not cross straits.
ce_fill_homeland_strait = {
	has_variable = ce_strait_end
	is_homeland_of_country_cultures = scope:ce_fill_country
	scope:ce_fill_country = {
		any_scope_state = {
			has_variable = ce_strait_end
			var:ce_linked ?= scope:ce_fill_country
			is_homeland_of_country_cultures = scope:ce_fill_country
			ce_strait_pair_with_other = yes
		}
	}
}

# A protectorate of ours borders our capital's land: one of its states
# (scope:ce_strait_other, saved by the caller) has a land neighbour of ours that
# the fill joined to the capital. A crossing never counts here: no protectorate
# state is a homeland of ours. Scope: state (the protectorate's).
ce_protectorate_state_borders_linked = {
	any_neighbouring_state = {
		owner = scope:ce_fill_country
		var:ce_linked ?= scope:ce_fill_country
		NOT = { ce_strait_pair_with_other = yes }
	}
}

# Whether a country's states need the monthly separation pass
# (ce_separation_step, colonial_geography_effects.txt): it holds a
# non-homeland state large enough to be a colony, or a protectorate.
# Scope: country.
ce_needs_separation = {
	NOT = { is_country_type = decentralized }
	exists = capital
	OR = {
		any_scope_state = {
			state_population >= colonial_territory_min_population
			NOT = { is_homeland_of_country_cultures = owner }
		}
		any_subject_or_below = {
			is_subject_type = subject_type_protectorate
		}
	}
}
