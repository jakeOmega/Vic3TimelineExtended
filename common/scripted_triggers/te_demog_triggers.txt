# The cohort model, the panel and the history run under Full and Display only.
# Both gates name the settings that run, never NOT = { ... demographics_disabled }:
# a save from before the rule existed has no value stored for it, and a negated
# check would read that as on and start the census in a running campaign (the
# tax code's rule, te_tax_triggers.txt, for the same reason). The only reads of
# the rule; everything else calls these two.
te_demog_cohorts_run = {
	OR = {
		has_game_rule = demographics_full
		has_game_rule = demographics_display_only
	}
}

# The once-a-game self-checks (docs/testing/demographics-playtest.md, "Self-checks") that read census-only inputs:
# on in every development build whatever the rule, since the inputs are there; a release sets it to always = no
# (scripts/release/comment_out_census_inputs.py's GATES), where the inputs are commented out and would read 0.
te_debug_demog_selfchecks_run = {
	always = yes
}

# The model's effects (phase 2) apply only under Full.
te_demog_effects_run = {
	has_game_rule = demographics_full
}

# Any scope: fast mode's census clock runs (te_debug_demog.1 options h-k, console only).
# While it does, te_demog_year reads the clock, the global monthly pulse steps every census,
# and the yearly state pulse keeps only inheritance and Wealth Concentration
# (te_demog_fast_effects.txt). Once started, the clock never stops.
te_demog_clock_on = {
	has_global_variable = te_demog_clock
}

# THIS = a country: it pays an old-age pension (phase 2 step 1). UBI and Post-Scarcity replace it and have no pension age.
te_demog_has_pension_law = {
	has_law = law_type:law_old_age_pension
}

# State scope: the state has a census (a seeded ring).
te_demog_has_census = {
	has_variable = te_dg_year
}

# Any scope: the census log is on (te_debug_demog.1 option b, console only). While it is,
# the country census writes a TE_DEMOG_CENSUS line and the player's capital a
# TE_DEMOG_PULSE line (te_debug_demog_effects.txt).
te_demog_census_log_on = {
	has_global_variable = te_demog_census_log
}

# Building scope: capital in §4.2's sense. The ownership buildings own themselves,
# Subsistence Farms read private and government buildings read state-owned, so all
# three are left out (docs/testing/demographics-probe-results-2026-10-08.md, Q2).
te_demog_is_capital_building = {
	NOR = {
		te_demog_is_ownership_building = yes
		is_building_group = bg_subsistence_agriculture
		is_building_group = bg_subsistence_ranching
		is_government_funded = yes
	}
}

# Building scope: where great fortunes are held (§4.2's weights).
te_demog_is_ownership_building = {
	OR = {
		is_building_type = building_manor_house
		is_building_type = building_financial_district
		is_building_type = building_company_headquarter
		is_building_type = building_company_regional_headquarter
	}
}

# Building scope: a rural workplace, for urban population (§5.1; plan Decisions).
te_demog_rural_workplace = {
	OR = {
		is_building_group = bg_agriculture
		is_building_group = bg_plantations
		is_building_group = bg_ranching
		is_building_group = bg_extraction
		is_building_group = bg_subsistence_agriculture
		is_building_group = bg_subsistence_ranching
	}
}

# Building scope: work that drew women as labour migrants (§2.5).
te_demog_female_job_workplace = {
	OR = {
		is_building_group = bg_light_industry
		is_building_group = bg_service
		is_building_group = bg_urban_facilities
	}
}

# THIS = a state: the engine's moved birth and mortality lines (te_demog_lines_refresh; owner, 2026-10-11) apply
# unless the census sets this state's births and deaths: Full, and its term in force (te_dg_mb, which a census step
# sets and Display only or Disabled clear). So under Full a state keeps vanilla's lines until its first census step.
te_demog_lines_apply = {
	OR = {
		NOT = { te_demog_effects_run = yes }
		NOT = { has_variable = te_dg_mb }
	}
}

# THIS = a state: the working-adult lines' pair applies (phase 2 step 1) unless the census sets this state's
# workforce: Full, and step 2's term in force (te_dg_mw). In step 1 nothing sets te_dg_mw, so it always applies.
te_demog_workforce_lines_apply = {
	OR = {
		NOT = { te_demog_effects_run = yes }
		NOT = { has_variable = te_dg_mw }
	}
}

# THIS = a state without Medical Outreach: the decree would change its next count. Some value is still far enough behind
# its target that the count, at the state's share of the gap as it stands, would leave it at least the snap short, where
# the decree's line takes it further. False at the targets, for a value the count puts on its target anyway, where the
# share of the gap is at its cap, and for a state without the census's medicine.
te_demog_medicine_behind = {
	te_demog_medicine_left_next >= te_demog_k_medicine_snap
}

# THIS = a state holding Medical Outreach: the decree still changes its next count, judged at the share of the gap
# without the decree's line, so a decree lasts exactly where it could be issued.
te_demog_medicine_behind_without_outreach = {
	te_demog_medicine_left_without_outreach >= te_demog_k_medicine_snap
}

# ---- Phase 2 step 2: the pension setting (te_demog_work_effects.txt) -------------------------------------------

# THIS = a country: its position follows an old-age threshold (Linked to Average or to Common Longevity).
te_demog_pension_is_linked = {
	has_variable = te_dg_pension_pos
	OR = {
		var:te_dg_pension_pos = te_demog_k_pension_pos_linked_average
		var:te_dg_pension_pos = te_demog_k_pension_pos_linked_common
	}
}

# THIS = a country: no five-year term runs (none chosen yet, or chosen te_demog_k_pension_term_years or more ago).
te_demog_pension_can_change = {
	te_demog_pension_term_left <= 0
}
