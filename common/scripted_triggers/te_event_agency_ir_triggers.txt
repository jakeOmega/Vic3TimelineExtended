# ============================================================================
# EVENT AGENCY (package ir) - scripted triggers
# ============================================================================
# "Give the accused country the choice first." Each event that used to pick a
# rival and tell the player that rival had acted (an arms buildup, a
# propaganda offensive, a spy, election meddling, a terror cell) now starts
# with a precursor event on the actor, which picks the victim and offers the
# act. Only if the actor chooses it does the victim's event fire.
#
# Each trigger below is the victim's old event trigger, mirrored onto the
# candidate victim, and is read in three places that must agree: the
# precursor's pool entry, the precursor's own `trigger`, and the precursor's
# `random_country` pick. A candidate the pick would reject can't satisfy the
# trigger either.
#
# Each precursor also sets a timed lockout variable on the victim it picks,
# in its own `immediate` (so the victim is off the table from the moment an
# actor is asked, and two actors can't pick it inside the delay before the
# victim's event). It stands in for the victim event's old `cooldown`, which
# a pick can't read; declining shortens it to a year.
#
# Scope: the candidate victim country. ROOT: the actor.
# ============================================================================

# international_relations_events.201 -> .1: a rival the actor's buildup is
# aimed at. The modifiers are .1's old trigger (a country still carrying the
# answer to a previous buildup isn't picked again); te_ea_ir_arms_race_lockout
# is .201's lockout.
te_ea_ir_arms_race_target = {
	NOT = { this = ROOT }
	has_diplomatic_pact = { who = ROOT type = rivalry }
	country_rank >= rank_value:major_power
	NOT = { is_country_type = decentralized }
	has_technology_researched = radar
	NOR = {
		has_modifier = arms_race_escalation_modifier
		has_modifier = diplomatic_overture_modifier
		has_modifier = diplomatic_overture_rebuffed_modifier
		has_modifier = arms_race_ignored_modifier
		has_variable = te_ea_ir_arms_race_lockout
	}
}

# international_relations_events.202 -> .4: a rival whose press the actor's

# international_relations_events.8: the other side of a real nuclear dispute.
# A nuclear-armed country opposing ROOT in a diplomatic play that is not yet a
# war, with neither party already in a deterrence crisis (a crisis belongs to
# the nuclear weapons journal entry, nuclear_crisis_effects.txt).
te_ea_ir_brinkmanship_opponent = {
	NOT = { this = ROOT }
	nd_is_armed = yes
	is_diplomatic_play_enemy_of = ROOT
	NOT = { has_war_with = ROOT }
	nd_crisis_parties_free = { TARGET = ROOT }
}

# society_technology_events.32 -> .14: a rival in the middle of an election
# campaign whose voters the actor's troll farms could reach. .14's old
# trigger (a real election with social media, not still carrying its answer
# to the last interference); te_ea_ir_meddling_lockout is .32's lockout (.14's
# old ten-year cooldown).
te_ea_ir_election_target = {
	NOT = { this = ROOT }
	has_diplomatic_pact = { who = ROOT type = rivalry }
	in_election_campaign = yes
	has_technology_researched = social_media
	NOT = { has_law = law_type:law_single_party_state }
	OR = {
		has_law = law_type:law_universal_suffrage
		has_law = law_type:law_census_voting
		has_law = law_type:law_wealth_voting
		has_law = law_type:law_landed_voting
	}
	NOR = {
		has_modifier = ste_election_interference_counter
		has_modifier = ste_election_interference_ignored
		has_variable = te_ea_ir_meddling_lockout
	}
}
