# ============================================================================
# EVENT AGENCY (package ir) - scripted triggers
# ============================================================================
# "Give the accused country the choice first." Each event that used to pick a
# rival and tell the player that rival had acted (an arms buildup, a
# propaganda offensive, a spy, election meddling, a terror cell) now starts
# with a precursor event on the actor, which picks the victim and offers the
# act. Only if the actor chooses it does the victim's event fire.
#
# Each trigger below is the victim's old event trigger, mirrored onto the
# candidate victim, and is read in three places that must agree: the
# precursor's pool entry, the precursor's own `trigger`, and the precursor's
# `random_country` pick. A candidate the pick would reject can't satisfy the
# trigger either.
#
# Scope: the candidate victim country. ROOT: the actor.
# ============================================================================

# international_relations_events.201 -> .1: a rival the actor's buildup is
# aimed at. The NOR list is .1's old trigger (a country still carrying the
# answer to a previous buildup isn't picked again).
te_ea_ir_arms_race_target = {
	NOT = { this = ROOT }
	has_diplomatic_pact = { who = ROOT type = rivalry }
	country_rank >= rank_value:major_power
	NOT = { is_country_type = decentralized }
	has_technology_researched = radar
	NOR = {
		has_modifier = arms_race_escalation_modifier
		has_modifier = diplomatic_overture_modifier
		has_modifier = diplomatic_overture_rebuffed_modifier
		has_modifier = arms_race_ignored_modifier
	}
}

# international_relations_events.202 -> .4: a rival whose press the actor's
# propaganda offensive targets. .4's old trigger (mass media, and not still
# carrying its answer to a previous campaign), plus the pool's rank gate.
te_ea_ir_propaganda_target = {
	NOT = { this = ROOT }
	has_diplomatic_pact = { who = ROOT type = rivalry }
	country_rank >= rank_value:major_power
	NOT = { is_country_type = decentralized }
	has_technology_researched = mass_media
	NOR = {
		has_modifier = propaganda_campaign_modifier
		has_modifier = propaganda_ignored_modifier
	}
}

# international_relations_events.203 -> .6: a rival the actor plants an agent
# in. .6's old trigger (cryptography for the counterintelligence that catches
# him; not still carrying its answer to the last spy), plus the pool's rank
# gate. te_ea_ir_spy_caught (set by .6) replaces .6's old event cooldown,
# which a pick cannot read.
te_ea_ir_espionage_target = {
	NOT = { this = ROOT }
	has_diplomatic_pact = { who = ROOT type = rivalry }
	country_rank >= rank_value:major_power
	NOT = { is_country_type = decentralized }
	has_technology_researched = cryptography
	NOR = {
		has_modifier = espionage_scandal_modifier
		has_modifier = espionage_prisoner_exchange_modifier
		has_variable = te_ea_ir_spy_caught
	}
}
