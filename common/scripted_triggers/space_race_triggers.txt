# Space Race Scripted Triggers

sr_has_space_program = {
	any_scope_building = {
		is_building_type = building_space_program
	}
}

# State scope: this state is where its owner's Space Program stands, the
# owner's capital (owner, 2026-11-03). Until then it was the market capital of
# a market the owner led, so a customs-union member could never found one and
# a leader that joined another market lost its program; nothing reads where
# the program stands, so the market gained nothing. Read by the building's
# possible and by the two removal sweeps (extra_on_actions.txt);
# te_decision_found_space_program places it in the capital. Moving the capital
# still loses the program (the sweep), as moving the market capital did.
# ?=: an invalid capital link fails the test, and the sweeps require the link
# to exist before removing anything.
sr_space_program_site = {
	owner.capital ?= THIS
}

# The country has run a milestone, so it may hold space-race state to repair:
# the gate of the monthly sync (space_race_on_actions.txt). A cheap test that
# no such country fails:
# - suborbital, the first milestone, runs with sr_active_suborbital set;
# - every other entry's `possible` needs a sr_completed_<m> that needs
#   sr_completed_suborbital before it, and nothing removes sr_completed_<m>;
# - the probe result and the colony records exist only at the end of that chain.
# Both are variables, so a revolution's winner inherits them together with the
# records the sync rebuilds from.
sr_space_race_participant = {
	OR = {
		has_variable = sr_active_suborbital
		has_variable = sr_completed_suborbital
	}
}

# The country holds one of the eight space race entries active.
# has_journal_entry counts active entries only. Scope: country
sr_holds_space_race_entry = {
	OR = {
		has_journal_entry = je_space_race_suborbital
		has_journal_entry = je_space_race_orbital
		has_journal_entry = je_space_race_moon_landing
		has_journal_entry = je_space_race_probe
		has_journal_entry = je_space_race_mars_landing
		has_journal_entry = je_space_race_interstellar_probe
		has_journal_entry = je_space_race_interstellar_results
		has_journal_entry = je_space_race_solar_colonization
	}
}

# The country is in the space race for good: it has run a milestone
# (sr_space_race_participant, variables a revolution's winner inherits), or it
# holds an entry (a colony given by treaty opens Solar System Colonization's
# without one). From then on the Timeline Extended window's Space Race tab
# stays open, with or without a milestone running
# (te_window_space_race_tab_sgui), and the Space Powers standings list it
# (sr_is_space_power) and refresh for it (sr_powers_refresh). Scope: country
sr_in_space_race = {
	OR = {
		sr_space_race_participant = yes
		sr_holds_space_race_entry = yes
	}
}

# Another country the Space Powers standings list: in the race, with a capital
# for its row to read (gotcha #11), and not ROOT, the viewer. Every one, not
# only the top three by ranking as the Rivals rows (sr_is_rival): a power
# outside them makes no progress, but what it achieved stands.
# Scope: the country tested; ROOT the viewer.
sr_is_space_power = {
	NOT = { THIS = ROOT }
	sr_in_space_race = yes
	exists = capital
}

# AI participation gate: only the top ~3 countries by prestige should pursue space race
# Players can always participate regardless
sr_ai_should_participate = {
	OR = {
		is_player = yes
		global_country_ranking <= 3
	}
}

# An AI that keeps up its colonies (colony management design § 3): a holder of
# one, whatever its rank, or a top-three space power. Gates the AI's emigration
# and module reviews, which until 2026-10-30 read sr_ai_should_participate alone,
# so a holder outside the top three let its colonies wither.
sr_ai_tends_colonies = {
	OR = {
		sr_has_space_colony = yes
		sr_ai_should_participate = yes
	}
}

# is_shown_when_inactive composer for follow-up space race milestones —
# milestones that gate on a prior milestone's completion variable plus a
# tech. Saves 3 lines per use across ~5 milestones.
#   $PREREQ$  — prior milestone token (e.g. suborbital, orbital, moon_landing)
#   $TECH$    — technology required (e.g. guided_missiles, space_exploration)
sr_milestone_followup_is_shown = {
	sr_space_race_on = yes
	has_variable = sr_completed_$PREREQ$
	has_technology_researched = $TECH$
}

}

# Shared gates for the milestone controls

# Each of the 28 space race journal-entry buttons and each op on the matching
# scripted GUI (common/scripted_guis/space_race_sguis.txt) reads its gate from
# here, so the AI's button grid and the player's widget can never disagree about
# when a control is offered or allowed. $MILESTONE$ is one of the seven tokens
# suborbital | orbital | moon_landing | probe | mars_landing |
# interstellar_probe | solar_colonization.
#
# The `possible` gates phrase their custom tooltips as CONDITIONS rather than
# complaints, because ScriptedGui.IsValidTooltip renders them with a tick when
# they pass and a cross when they do not.

sr_controls_shown = {
	has_variable = sr_active_$MILESTONE$
}

# Solar colonization's four controls carry two extra conditions: its journal
# entry deliberately stays alive in passive mode once every colony is claimed
# (it holds the colony modifiers) and while the production method is switched
# off, and in neither state may funding or approach be touched.
sr_solar_controls_shown = {
	sr_solar_colonization_running = yes
}

sr_possible_safe = {
	custom_tooltip = {
		text = "SR_NOT_ALREADY_SAFE"
		NOT = { has_variable = sr_safe_$MILESTONE$ }
	}
}

sr_possible_ambitious = {
	custom_tooltip = {
		text = "SR_NOT_ALREADY_AMBITIOUS"
		NOT = { has_variable = sr_ambitious_$MILESTONE$ }
	}
}

# Standard is the approach with neither variable: offered while one is set.
sr_possible_standard = {
	custom_tooltip = {
		text = "SR_NOT_ALREADY_STANDARD"
		OR = {
			has_variable = sr_safe_$MILESTONE$
			has_variable = sr_ambitious_$MILESTONE$
		}
	}
}

# The funding ceiling: the cap plus the ladder past it (sr_max_funding_level_soft;
# lunar colonization design § 5.1). The player's plus button and, through
# sr_possible_fund_up_within_cap, the AI's.
sr_possible_fund_up = {
	custom_tooltip = {
		text = "SR_MAX_FUNDING_NOT_REACHED"
		var:sr_funding_$MILESTONE$ < sr_max_funding_level_soft
	}
}

# The AI's increase-funding buttons (space_race_buttons.txt): the same ceiling,
# and the AI stays within the cap (owner, 2026-10-29, design § 15.6). Observer
# runs decide later whether a great power in a contested race should go past it.
sr_possible_fund_up_within_cap = {
	sr_possible_fund_up = { MILESTONE = $MILESTONE$ }
	custom_tooltip = {
		text = "SR_FUNDING_BELOW_CAP"
		var:sr_funding_$MILESTONE$ < sr_max_funding_level
	}
}

sr_possible_fund_down = {
	custom_tooltip = {
		text = "SR_MIN_FUNDING_NOT_REACHED"
		var:sr_funding_$MILESTONE$ > 0
	}
}

# Solar colonization is actually running: the journal entry is open, its
# production method is switched on, and colonies remain to claim. The same test
# as the colonization pulse's own gate — outside it the entry is open but frozen
# (passive mode, holding the colony modifiers), so an event must not narrate or
# move its progress. Also the gate of the entry's four controls.
# A finished programme runs again only when the country resumed it for the
# Moon (sr_colonization_resumed, te_decision_resume_solar_colonization), which
# keeps its completion record: lunar colonization 2b, owner 2026-10-29. And
# with the race over and a lunar world free, nothing runs without that choice
# (sr_colonization_by_choice).
sr_solar_colonization_running = {
	# The rule first: switched off mid-game, everything that reads this stops at
	# once, before the entry's invalid outcome ends it.
	sr_space_race_on = yes
	has_variable = sr_active_solar_colonization
	modifier:country_sr_solar_colonization_program_bool = yes
	OR = {
		NOT = { has_variable = sr_completed_solar_colonization }
		has_variable = sr_colonization_resumed
	}
	sr_solar_colonization_qualified = yes
	sr_colonization_by_choice = yes
}

# The race is over (every world was once claimed: sr_all_colonies_complete,
# kept as history) and a lunar world is free again, released since. Settling
# the Moon is then a choice (sr_colonization_by_choice). Read by that trigger,
# the entry's `possible` (to show its line only here), Begin Solar
# Colonization and Resume Solar System Colonization. Any scope.
sr_race_over_moon_free = {
	has_global_variable = sr_all_colonies_complete
	sr_sites_lunar_open = yes
}

# Colonizing the Moon after the race is over is a choice (owner, 2026-10-29).
# While sr_race_over_moon_free holds, a programme colonizes only once the
# country chose it: Resume Solar System Colonization or Begin Solar
# Colonization set sr_colonization_resumed. Otherwise a method carrying the
# colonization bool, which every power that finished the race runs, would
# start it on its own and pause its Commercial Space Industry. Read by the
# entry's `possible` and sr_solar_colonization_running (its pulse, billing,
# controls and the dividend's pause). Country scope.
sr_colonization_by_choice = {
	OR = {
		NOT = { sr_race_over_moon_free = yes }
		has_variable = sr_colonization_resumed
	}
}

# The milestone Solar System Colonization needs to colonize: the Moon Landing
# (lunar colonization design § 4; before 2026-10-29 the Moon Base and the Mars
# Landing). A country given a colony without it, by treaty or as a dead
# holder's heir, holds the entry and keeps its colony's modifiers, and colonizes
# nothing: its panel reads Idle and nothing is billed. Country scope.
sr_solar_colonization_qualified = {
	has_variable = sr_completed_moon_landing
}

# Worlds beyond the Moon (stage 1 and later) need the Mars Landing as well: a
# hard gate, the "physical prerequisite" exception to the soft-caps default
# (lunar colonization design §§ 1.5, 4). Every stage >= 1 world's
# sr_cf_targetable_<key> and draw branch tests it (gen_solar_sites.py), so a
# country given a world beyond the Moon by treaty keeps it and colonizes only
# the Moon until its own Mars Landing. Country scope.
sr_colonization_beyond_moon = {
	has_variable = sr_completed_mars_landing
}

# Whether a milestone's funding and approach are billed, and its funding adds
# to the programme's pace: the gate of its entry's three cost and funding
# declarations (je_space_race.txt; #1404), and of the panel's funding
# figures (space_race_display_values.txt). A running milestone is billed;
# the six single-goal milestones run exactly while sr_active_<m> is set,
# because losing their production method fails the entry. Solar colonization
# stays open without it (its colonies' modifiers live on the entry), and then
# it is not billed: the same test as its pulse and its controls (owner,
# 2026-09-29: billing it with the method off was a bug). Country scope.
sr_milestone_billed_suborbital = { has_variable = sr_active_suborbital }
sr_milestone_billed_orbital = { has_variable = sr_active_orbital }
sr_milestone_billed_moon_landing = { has_variable = sr_active_moon_landing }
sr_milestone_billed_probe = { has_variable = sr_active_probe }
sr_milestone_billed_mars_landing = { has_variable = sr_active_mars_landing }
sr_milestone_billed_interstellar_probe = { has_variable = sr_active_interstellar_probe }
sr_milestone_billed_solar_colonization = { sr_solar_colonization_running = yes }

# A rival on milestone $MILESTONE$, seen from ROOT: another country running it
# and actually moving it (sr_ai_should_participate — an AI outside the top
# three can hold sr_active_<m> and never tick). The Rivals list, its estimate
# rows and the monthly refresh that fills them all use this one test.
# Scope: the candidate country; ROOT: the viewer.
sr_is_rival = {
	NOT = { THIS = ROOT }
	has_variable = sr_active_$MILESTONE$
	sr_ai_should_participate = yes
}

# This country (the viewer) holds a covert network in $TARGET$ whose
# intelligence tier is at least $TIER$ (1: strength 50+, 2: 75+). Reads the
# covert system's own stored tier code, iw_net_intel_tier (covert_net_clamp
# writes it after every strength change), so the thresholds live in one place.
# The lookup works inside the list, as the covert "lookup rule" requires.
# Parameters: $TARGET$ = country (a scope: reference), $TIER$ = 1 or 2
# Scope: country (the viewer)
sr_has_intel_tier = {
	has_variable_list = iw_nets
	any_in_list = {
		variable = iw_nets
		var:iw_target ?= $TARGET$
		has_variable = iw_net_intel_tier
		var:iw_net_intel_tier >= $TIER$
	}
}

# At least one milestone is under way: exactly the milestones
# sr_boost_active_milestones pushes, so an event can offer progress when this
# holds and something else when it does not. The one test of "a milestone is
# running": it reads the per-milestone sr_active_<m> flags, which each journal
# entry sets on activation and clears on completion or failure. It replaced a
# shared sr_active_milestone flag that the completion of any one milestone
# cleared while others (the probe, the Moon Base, Mars) were still running (#480).
sr_has_running_milestone = {
	OR = {
		has_variable = sr_active_suborbital
		has_variable = sr_active_orbital
		has_variable = sr_active_moon_landing
		has_variable = sr_active_probe
		has_variable = sr_active_mars_landing
		has_variable = sr_active_interstellar_probe
		sr_solar_colonization_running = yes
	}
}

# The race is over for this country: every single-goal milestone done, every
# world claimed by someone, and no programme still running, so nothing is left
# to spend progress on. Its progress sources then pay the Commercial Space
# Industry dividend instead (sr_refresh_space_dividend). A country that reaches
# the end after the last world is gone qualifies as soon as its own milestones
# are done; it never opens Solar System Colonization, but its sources are just
# as idle. The Moon Base is no longer one of them (retired 2026-10-29): its
# record is a first lunar colony now, which a power may never found once the
# six lunar worlds are taken (lunar colonization design § 7). All variables,
# so a revolution's winner inherits the state. sr_all_colonies_complete is
# history: it stays when a lunar world is released after every world was
# claimed, so a country that had concluded keeps the dividend until it chooses
# to resume colonization for the Moon (owner, 2026-10-29;
# sr_colonization_resumed makes it running again).
sr_space_race_concluded = {
	has_variable = sr_completed_suborbital
	has_variable = sr_completed_orbital
	has_variable = sr_completed_moon_landing
	has_variable = sr_completed_probe
	has_variable = sr_completed_mars_landing
	has_variable = sr_completed_interstellar_probe
	has_global_variable = sr_all_colonies_complete
	NOT = { sr_has_running_milestone = yes }
}

# We hold at least one colony: every world held, whether founded, given by

# A MILESTONE STARTING AGAIN AFTER A PROGRAMME LOSS (2026-11-06)
#
# Country scope. sr_reentry_due_<m>: the re-entry pass (sr_reentry_monthly,
# space_race_effects.txt) may add je_space_race_<m> back. A lowered Space
# Program method or a lost building fails the milestone (its failure outcome),
# the engine deletes a concluded record three days later and never activates
# the type again (scripting_best_practices.md, "A finished record is
# deleted"), so the milestone would never come back by itself.
#
# Each trigger is what add_journal_entry needs, then the entry's
# is_shown_when_inactive and possible, which add_journal_entry skips, word for
# word (test_space_race_reentry.py holds them to je_space_race.txt):
#   - no record of the type, active or inactive: the add is refused while one
#     exists (journal_entry_manager.cpp), and every country holds an inactive
#     record of each milestone from the game's start until it concludes, so
#     these two tests end the trigger for nearly every country, cheaply;
#   - a country that holds entries (not decentralized) and is not the rebels
#     of a running revolution.
# A completed milestone is never due: its possible requires the absence of
# sr_completed_<m> (solar colonization's restates its own gates instead).
# ============================================================================
sr_reentry_due_suborbital = {
	NOT = { has_journal_entry = je_space_race_suborbital }
	NOT = { has_inactive_journal_entry = je_space_race_suborbital }
	NOT = { is_country_type = decentralized }
	is_revolutionary = no
	# je_space_race_suborbital's is_shown_when_inactive
	sr_space_race_on = yes
	has_technology_researched = rocketry
	is_great_or_major_power = yes
	is_revolutionary = no
	# its possible
	has_technology_researched = rocketry
	modifier:country_sr_earth_orbit_program_bool = yes
	is_great_or_major_power = yes
	NOT = { has_variable = sr_completed_suborbital }
}

# Solar colonization fails only while it holds no colony (its failure
# outcome), so this is the colony-less programme coming back once its method
# returns. A colony gained while the record is gone opens the entry already
# (sr_site_gained_base, sr_cf_open_entry); this trigger's has_journal_entry
# test then ends the pass's interest.
sr_reentry_due_solar_colonization = {
	NOT = { has_journal_entry = je_space_race_solar_colonization }
	NOT = { has_inactive_journal_entry = je_space_race_solar_colonization }
	NOT = { is_country_type = decentralized }
	is_revolutionary = no
	# je_space_race_solar_colonization's is_shown_when_inactive
	sr_space_race_on = yes
	OR = {
		sr_has_space_colony = yes
		AND = {
			has_variable = sr_completed_moon_landing
			has_technology_researched = cislunar_infrastructure
		}
	}
	# its possible
	OR = {
		custom_tooltip = {
			text = sr_has_existing_colonies_tt
			sr_has_space_colony = yes
		}
		AND = {
			has_variable = sr_completed_moon_landing
			modifier:country_sr_solar_colonization_program_bool = yes
			OR = {
				custom_tooltip = {
					text = sr_lunar_world_free_tt
					sr_sites_lunar_open = yes
				}
				custom_tooltip = {
					text = sr_outer_system_requires_deep_space_tt
					OR = {
						sr_colonization_inner_open = yes
						modifier:country_sr_deep_space_exploration_program_bool = yes
					}
				}
			}
			custom_tooltip = {
				text = sr_world_left_to_claim_tt
				sr_any_site_unclaimed = yes
			}
			trigger_if = {
				limit = { sr_race_over_moon_free = yes }
				custom_tooltip = {
					text = sr_colonization_by_choice_tt
					has_variable = sr_colonization_resumed
				}
			}
			is_great_or_major_power = yes
		}
	}
}
