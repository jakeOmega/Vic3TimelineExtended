# Space Race Scripted Triggers

sr_has_space_program = {
	any_scope_building = {
		is_building_type = building_space_program
	}
}

sr_is_pursuing_milestone = {
	has_variable = sr_active_milestone
}

# AI participation gate: only the top ~3 countries by prestige should pursue space race
# Players can always participate regardless
sr_ai_should_participate = {
	OR = {
		is_player = yes
		global_country_ranking <= 3
	}
}

sr_can_start_space_race = {
	has_technology_researched = rocketry
	is_great_or_major_power = yes
	NOT = { has_variable = sr_active_milestone }
}

# is_shown_when_inactive composer for follow-up space race milestones —
# milestones that gate on a prior milestone's completion variable plus a
# tech. Saves 3 lines per use across ~5 milestones.
#   $PREREQ$  — prior milestone token (e.g. suborbital, orbital, moon_landing)
#   $TECH$    — technology required (e.g. guided_missiles, space_exploration)
sr_milestone_followup_is_shown = {
	has_game_rule = space_race_enabled
	has_variable = sr_completed_$PREREQ$
	has_technology_researched = $TECH$
}

sr_has_failure_cooldown = {
	has_variable = sr_failure_cooldown
	var:sr_failure_cooldown > 0
}

has_space_company = {
	OR = {
		has_company = company_type:company_spacex
		has_company = company_type:company_lockheed_martin
		has_company = company_type:company_roscosmos
	}
}

# ---------------------------------------------------------------------------
# Shared gates for the milestone controls
# ---------------------------------------------------------------------------
# Each of the 32 space race journal-entry buttons and each op on the matching
# scripted GUI (common/scripted_guis/space_race_sguis.txt) reads its gate from
# here, so the AI's button grid and the player's widget can never disagree about
# when a control is offered or allowed. $MILESTONE$ is one of the eight tokens
# suborbital | orbital | moon_landing | probe | moon_base | mars_landing |
# interstellar_probe | solar_colonization.
#
# The `possible` gates phrase their custom tooltips as CONDITIONS rather than
# complaints, because ScriptedGui.IsValidTooltip renders them with a tick when
# they pass and a cross when they do not.

sr_controls_shown = {
	has_variable = sr_active_$MILESTONE$
}

# Solar colonization's four controls carry two extra conditions: its journal
# entry deliberately stays alive in passive mode once every colony is claimed
# (it holds the colony modifiers) and while the production method is switched
# off, and in neither state may funding or approach be touched.
sr_solar_controls_shown = {
	sr_solar_colonization_running = yes
}

sr_possible_safe = {
	custom_tooltip = {
		text = "SR_NOT_ALREADY_SAFE"
		NOT = { has_variable = sr_safe_$MILESTONE$ }
	}
}

sr_possible_ambitious = {
	custom_tooltip = {
		text = "SR_NOT_ALREADY_AMBITIOUS"
		NOT = { has_variable = sr_ambitious_$MILESTONE$ }
	}
}

sr_possible_fund_up = {
	custom_tooltip = {
		text = "SR_MAX_FUNDING_NOT_REACHED"
		var:sr_funding_$MILESTONE$ < sr_max_funding_level
	}
}

sr_possible_fund_down = {
	custom_tooltip = {
		text = "SR_MIN_FUNDING_NOT_REACHED"
		var:sr_funding_$MILESTONE$ > 0
	}
}

# ---------------------------------------------------------------------------
# State the space race events read
# ---------------------------------------------------------------------------

# Solar colonization is actually running: the journal entry is open, its
# production method is switched on, and colonies remain to claim. The same test
# as the colonization pulse's own gate — outside it the entry is open but frozen
# (passive mode, holding the colony modifiers), so an event must not narrate or
# move its progress. Also the gate of the entry's four controls.
sr_solar_colonization_running = {
	has_variable = sr_active_solar_colonization
	modifier:country_sr_solar_colonization_program_bool = yes
	NOT = { has_variable = sr_completed_solar_colonization }
}

# We have really claimed at least one colony (sr_establish_colony_effect counts
# only real claims). The same test society_technology_events.18 uses.
sr_has_space_colony = {
	has_variable = sr_colony_count
	var:sr_colony_count > 0
}

# Failure-event themes, keyed on the milestone that failed
# (scope:sr_setback_<milestone>, saved by the failure dispatchers in
# common/scripted_effects/space_race_effects.txt).
sr_setback_is_deep_space_probe = {
	OR = {
		exists = scope:sr_setback_probe
		exists = scope:sr_setback_interstellar_probe
	}
}

# Any failure-event marker at all. Without one the setback helpers fall back to
# the sr_failed_ flags (sr_resolve_setback_markers).
sr_has_setback_marker = {
	OR = {
		exists = scope:sr_setback_suborbital
		exists = scope:sr_setback_orbital
		exists = scope:sr_setback_moon_landing
		exists = scope:sr_setback_probe
		exists = scope:sr_setback_moon_base
		exists = scope:sr_setback_mars_landing
		exists = scope:sr_setback_interstellar_probe
		exists = scope:sr_setback_solar_colonization
	}
}

sr_setback_is_crewed_outpost = {
	OR = {
		exists = scope:sr_setback_moon_base
		exists = scope:sr_setback_mars_landing
		exists = scope:sr_setback_solar_colonization
	}
}

# Scope: a country. It has a space programme of its own: a milestone reached or
# one under way.
sr_is_spacefaring = {
	OR = {
		has_variable = sr_completed_suborbital
		has_variable = sr_active_milestone
	}
}

# Scope: a foreign country. A space power our relations with have really soured:
# a rivalry either way (has_diplomatic_pact without is_initiator matches both
# directions), or relations at cold or worse. space_race_events.26 ("When
# Politics Reaches Orbit") needs one to name. ROOT is our country.
sr_is_estranged_space_power = {
	NOT = { THIS = ROOT }
	sr_is_spacefaring = yes
	OR = {
		has_diplomatic_pact = { who = ROOT type = rivalry }
		relations:root <= relations_threshold:cold
	}
}

# Scope: a foreign country. A United Nations member that has reached orbit — a
# partner space_race_events.52's joint orbital station could really be built
# with. ROOT is our country.
sr_is_station_partner = {
	NOT = { THIS = ROOT }
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	has_variable = sr_completed_orbital
}

# Scope: the receiver of space_race_events.20. A milestone notification for
# $MILESTONE$ is waiting (sr_notify_milestone_achievement adds one entry per
# completion).
sr_has_milestone_notice = {
	has_variable_list = sr_notify_achievers_$MILESTONE$
	any_in_list = {
		variable = sr_notify_achievers_$MILESTONE$
		always = yes
	}
}
