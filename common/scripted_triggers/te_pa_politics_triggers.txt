# Private Actors, politics at home: the triggers (docs/systems/mod_systems/private_actors.md § Politics at Home;
# spec §§ 8.2, 8.3, 8.5, 8.7; plan Task 5). Company scope unless noted. Each verb is a te_pa_can_<verb> trigger here
# and a te_pa_do_<verb> effect in te_pa_politics_effects.txt; te_pa_act's lists name them. The numbers are
# te_pa_politics_values.txt's.

# ---------------------------------------------------------------------------
# The Electoral Finance law (§ 8.3's table), country scope
# ---------------------------------------------------------------------------

# No Campaign Finance Laws (a country without a vote), or none of the group's laws: patronage only, open, uncapped.
te_pa_ef_none = {
	NOR = {
		has_law = law_type:law_unregulated_donations
		has_law = law_type:law_donation_limits
		has_law = law_type:law_publicly_funded_elections
	}
}

# Unregulated Donations: open and uncapped.
te_pa_ef_open = {
	has_law = law_type:law_unregulated_donations
}

# Donation Limits: open up to a cap for each firm, the excess in secret.
te_pa_ef_limits = {
	has_law = law_type:law_donation_limits
}

# Publicly Funded Elections: private campaign money is illegal; in secret only, at half effect.
te_pa_ef_public = {
	has_law = law_type:law_publicly_funded_elections
}

# Campaign money buys momentum for a party (§ 8.3's first form): the country is in an election campaign, under an
# Electoral Finance law that has elections to buy. Probe 9's fallback (momentum_from_pulse BROKEN): make this
# always = no, and patronage is the one form everywhere.
te_pa_cm_momentum_form = {
	in_election_campaign = yes
	te_pa_ef_none = no
}

# ---------------------------------------------------------------------------
# The firm's law at stake (te_pa_stake_pick)
# ---------------------------------------------------------------------------

# It has a law at stake: one of the law table's, or the trade policy (te_pa_stake_trade).
te_pa_has_stake = {
	OR = {
		te_pa_v_stake_law > 0
		var:te_pa_stake_trade ?= 1
	}
}

# Its law at stake is the law its country is enacting (the side it takes there is te_pa_do_back_side's).
te_pa_stake_is_enacting = {
	te_pa_v_stake_law > 0
	te_pa_enacting_law_firm >= te_pa_v_stake_law
	te_pa_enacting_law_firm <= te_pa_v_stake_law
}

# It may spend campaign money at all under its country's law: under Publicly Funded Elections only in secret, which a
# cautious executive won't risk (te_pa_cm_secret_bold).
te_pa_cm_allowed = {
	owner = { te_pa_private_ownership_ended = no }
	trigger_if = {
		limit = { owner = { te_pa_ef_public = yes } }
		te_pa_v_bold >= te_pa_cm_secret_bold
	}
}

# ---------------------------------------------------------------------------
# The verbs (te_pa_act's lists, te_pa_politics_effects.txt)
# ---------------------------------------------------------------------------

# Stall the law (§ 8.3; Task 4a's, widened): a law being enacted that it is against, or one that would outlaw its
# cartel (§ 6.4); it has the Connections; a year has passed since its last stall; no firm has set this law back in
# the last year (one setback a year a law from all firms together); and the enactment can take another setback
# without failing outright (vanilla's guard, events/ministry_law_events.txt).
te_pa_can_stall = {
	owner = { enacting_any_law = yes }
	OR = {
		te_pa_gen_stance_enacting < 0
		AND = {
			te_pa_v_cartel > 0
			te_pa_law_threatens_cartel = yes
		}
	}
	te_pa_v_conn >= te_pa_stall_conn
	trigger_if = {
		limit = { has_variable = te_pa_stall_month }
		te_pa_stall_since >= te_pa_stall_months
	}
	owner = {
		te_pa_law_setback_open = yes
		enactment_setback_count < max_setbacks_minus_one
	}
}

# The quarter's stall (te_pa_trust_quarter): a cartel firm defends its cartel whatever its goal (§ 6.4), so a law
# that would outlaw it can start and end between two turns; any other firm stalls at its turn only.
te_pa_can_stall_cartel = {
	te_pa_v_cartel > 0
	te_pa_law_threatens_cartel = yes
	te_pa_can_stall = yes
}

# Speed the law (§ 8.3): a law being enacted that it is for; the Connections; a year since its last speed; no firm
# has sped this law in the last year (one progress a year a law from all firms together).
te_pa_can_speed = {
	owner = { enacting_any_law = yes }
	te_pa_gen_stance_enacting > 0
	te_pa_v_conn >= te_pa_speed_conn
	trigger_if = {
		limit = { has_variable = te_pa_speed_month }
		te_pa_speed_since >= te_pa_speed_months
	}
	owner = { te_pa_law_progress_open = yes }
}

# Country scope. No firm has set back the law being enacted in the last year (te_pa_stall_law and its month,
# te_pa_do_stall).
te_pa_law_setback_open = {
	te_pa_stall_law_recent < 1
}

# Country scope. No firm has sped the law being enacted in the last year.
te_pa_law_progress_open = {
	te_pa_speed_law_recent < 1
}

# Campaign money (§ 8.3) as a goal's verb: it has a law at stake that is not the law being enacted (its side on that
# one is te_pa_do_back_side's, after the goal's verb, so it is not paid for twice in a turn), the law lets it spend,
# and it can pay. Cheaper Inputs spends only where the trade policy taxes its imported inputs (§ 8.3's "a tariff on
# its input": the tariffs themselves are Task 5b's).
te_pa_can_campaign = {
	te_pa_has_stake = yes
	te_pa_stake_is_enacting = no
	trigger_if = {
		limit = { var:te_pa_goal ?= 1 }
		var:te_pa_stake_trade ?= 1
		var:te_pa_stake_sign ?= 1
	}
	te_pa_cm_allowed = yes
	te_pa_can_afford = { AMOUNT = te_pa_cm_cost_full_value }
}

# Back its side on the law being enacted (§ 8.2), after the goal's verb, like the foundation: a firm whose goal is A
# Friendly Law spends on its side every turn, a cartel firm against a law that would outlaw its cartel (§ 6.4) too;
# a firm with another goal only on a stance of weight 2, and half as much (te_pa_back_side_size).
te_pa_can_back_side = {
	owner = {
		enacting_any_law = yes
		has_variable = te_pa_side_law
		te_pa_side_law_enacting = yes
	}
	OR = {
		te_pa_gen_stance_enacting > 0
		te_pa_gen_stance_enacting < 0
	}
	OR = {
		var:te_pa_goal ?= 4
		te_pa_side_stance_size >= te_pa_side_weight_other
		AND = {
			te_pa_v_cartel > 0
			te_pa_law_threatens_cartel = yes
		}
	}
	te_pa_cm_allowed = yes
	te_pa_can_afford = { AMOUNT = te_pa_cm_cost_side_value }
}

# It backs its side in full (its goal, or its cartel at stake), not at half.
te_pa_back_side_full = {
	OR = {
		var:te_pa_goal ?= 4
		AND = {
			te_pa_v_cartel > 0
			te_pa_law_threatens_cartel = yes
		}
	}
}

# Country scope. The law its firms' sides are on (te_pa_side_law) is the one it is enacting now.
te_pa_side_law_enacting = {
	te_pa_side_law_gap < 0.5
}

# Company scope, with scope:te_pa_side_home its country settling its sides (te_pa_sides_settle): the firm spent on
# the side that won, for a law that passed or against one that did not.
te_pa_side_won = {
	OR = {
		AND = {
			var:te_pa_side_sign ?= 1
			scope:te_pa_side_home = { var:te_pa_side_passed ?= 1 }
		}
		AND = {
			var:te_pa_side_sign ?= -1
			scope:te_pa_side_home = { var:te_pa_side_passed ?= 0 }
		}
	}
}

# Company scope. Its side spending is on the law its country's sides are on now.
te_pa_side_on_current = {
	has_variable = te_pa_side_of
	te_pa_v_side_of >= te_pa_side_law_now
	te_pa_v_side_of <= te_pa_side_law_now
}

# ---------------------------------------------------------------------------
# The press (§ 8.5)
# ---------------------------------------------------------------------------

# Its country's speech law lets it hold a press: none may under Outlawed Dissent; under Censorship only a Favored or
# Tolerated firm; under Right of Assembly and Protected Speech any.
te_pa_press_allowed = {
	owner = { NOT = { has_law = law_type:law_outlawed_dissent } }
	trigger_if = {
		limit = { owner = { has_law = law_type:law_censorship } }
		NOT = { var:te_pa_posture ?= -1 }
	}
}

# It has a secret holding a press would keep out of the papers: a secret cartel, a captured regulator not yet
# exposed, or secret campaign money still at risk.
te_pa_has_secret = {
	OR = {
		AND = {
			te_pa_v_cartel > 0
			var:te_pa_cartel_known ?= 0
		}
		AND = {
			te_pa_v_reg > 0
			var:te_pa_reg_known ?= 0
		}
		te_pa_v_cm_secret > 0
	}
}

# Found a press (§ 8.5), an opportunistic verb after the goal's (and the trust's defence in the cartel firm's lists,
# § 6.4): it holds none, the speech law allows one, it has the founding cost and a year's accrual besides, it has a
# secret holding to hide or a law at stake, and five years have passed since a press of its was closed.
te_pa_can_found_press = {
	te_pa_v_press <= 0
	te_pa_press_allowed = yes
	owner = { te_pa_private_ownership_ended = no }
	te_pa_v_funds >= te_pa_press_spare_value
	OR = {
		te_pa_has_secret = yes
		te_pa_has_stake = yes
	}
	te_pa_wait_over = { VAR = te_pa_press_again }
}

# The government may close its press (§ 8.7): under Censorship (a speech law that allows it), or once it has won a
# suit against the firm within ten years (te_pa_cartel_barred_until, the court's bar on the cartel). Under Right of
# Assembly and Protected Speech the suit is the only way; under Outlawed Dissent no press is left to close.
te_pa_sa_can_close_press = {
	te_pa_v_press > 0
	OR = {
		owner = { has_law = law_type:law_censorship }
		te_pa_v_cartel_barred_until > te_history_month_index
	}
}

# Closing its press costs the government nothing (§ 8.5: under Censorship, a Restrained firm's).
te_pa_press_close_free = {
	owner = { has_law = law_type:law_censorship }
	var:te_pa_posture ?= -1
}

# The Close the Press button's test (te_pa_sguis.txt): country scope, the scripted GUI's root, with scope:te_pa_row the
# row the button sits on; one custom_tooltip a condition, as the other actions' are.
te_pa_sa_can_close_press_row = {
	te_pa_sa_row_ours = yes
	custom_tooltip = {
		text = te_pa_sa_has_press_tt
		scope:te_pa_row ?= { var:te_pa_r_firm ?= { te_pa_v_press > 0 } }
	}
	custom_tooltip = {
		text = te_pa_sa_press_closable_tt
		scope:te_pa_row ?= { var:te_pa_r_firm ?= { te_pa_sa_can_close_press = yes } }
	}
}

# The firm (an entry of the country list te_pa_presses), with scope:te_pa_expose_firm a firm whose secret holding is
# being rolled: its press presses on that firm (§ 8.5: the other side of its law, or a cartel that sells its input).
te_pa_press_hits = {
	te_pa_v_press > 0
	exists = scope:te_pa_expose_firm
	NOT = { this = scope:te_pa_expose_firm }
	OR = {
		# The two backed opposite sides of the law being enacted.
		AND = {
			te_pa_side_on_current = yes
			scope:te_pa_expose_firm = { te_pa_side_on_current = yes }
			te_pa_v_side_sign_product < 0
		}
		# The rolled firm holds a cartel that makes one of this firm's inputs.
		AND = {
			scope:te_pa_expose_firm = { te_pa_v_cartel > 0 }
			te_pa_press_x_out > 0
			OR = {
				AND = {
					te_pa_press_in1 >= te_pa_press_x_out
					te_pa_press_in1 <= te_pa_press_x_out
				}
				AND = {
					te_pa_press_in2 >= te_pa_press_x_out
					te_pa_press_in2 <= te_pa_press_x_out
				}
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Tariffs (Task 5b; § 8.1, § 8.8's Tax Code tie), country scope
# ---------------------------------------------------------------------------

# The trade policy lets an import tariff collect (vanilla's buttons' SET_TARIFF_LEVEL_NOT_ALLOWED): its maximum rate,
# read in the capital as te_bulk_tariff_policy_allows reads it. Free Trade has none, so a level set there is idle.
te_pa_tariffs_bite = {
	capital ?= { modifier:state_tariff_import_add > 0 }
}

# The government can set a tariff on a firm's petition: it owns its market (a junior partner of a customs union
# trades in its senior's), and the Tax Code's customs option does not legislate its tariffs. Under that option the
# code's monthly sync re-asserts its own levels, so a pledge defers to it (§ 8.8): no firm asks for a tariff, none is
# set, and a pledge already running lapses (te_pa_pledge_tariff_monthly).
te_pa_tariffs_settable = {
	NOT = { te_tax_customs_on = yes }
	te_tax_owns_market = yes
}

# ---------------------------------------------------------------------------
# The petition for a law or a tariff (Task 5b; § 8.4): kind law on the declared-move frame
# ---------------------------------------------------------------------------

# Kind law, at its turn (scope:te_pa_firm the firm): its goal's score is 70 or more and it holds 40 Connections, it
# holds no pledge from the government already, five years have passed since it was last refused or left unanswered,
# and a law or a tariff would serve its goal (te_pa_pt_can_ask).
te_pa_can_declare_law = {
	te_pa_move_open = yes
	te_pa_goal_score_value >= te_pa_pt_score
	te_pa_v_conn >= te_pa_pt_conn
	NOT = { has_variable = te_pa_pl_kind }
	te_pa_wait_over = { VAR = te_pa_law_again }
	te_pa_pt_can_ask = yes
}

# A law or a tariff that would serve the goal it holds (§ 8.4): Cheaper Inputs asks for a lower tariff on an input
# or, a buyer of a lawful cartel's good, for Antitrust Enforcement (§ 6.3); A Protected Market for a tariff on its
# output; A Friendly Law for the law from the table it could move to that favors it most. Labor Peace's labor law is
# Task 6's: its branch goes here (and in te_pa_pt_pick), ask 5.
te_pa_pt_can_ask = {
	OR = {
		AND = {
			var:te_pa_goal ?= 1
			OR = {
				te_pa_pt_ask_input_tariff = yes
				te_pa_pt_ask_antitrust = yes
			}
		}
		AND = {
			var:te_pa_goal ?= 2
			te_pa_pt_ask_output_tariff = yes
		}
		AND = {
			var:te_pa_goal ?= 4
			te_pa_pt_ask_law = yes
		}
		# Task 6: Labor Peace (goal 6) asks for a labor law here, ask 5 (te_pa_pt_pick's branch beside it).
	}
}

# A tariff collects on an input the market imports, and the government can lower it.
te_pa_pt_ask_input_tariff = {
	owner = { te_pa_tariffs_settable = yes }
	OR = {
		AND = {
			var:te_pa_g_in1_imported ?= 1
			te_pa_v_in1_tariff > 0
		}
		AND = {
			var:te_pa_g_in2_imported ?= 1
			te_pa_v_in2_tariff > 0
		}
	}
}

# The market imports what it makes, its import tariff is below the highest, and the government can raise it under a
# trade policy that lets it collect.
te_pa_pt_ask_output_tariff = {
	owner = {
		te_pa_tariffs_settable = yes
		te_pa_tariffs_bite = yes
	}
	te_pa_wants_output_tariff = yes
	te_pa_v_out_tariff_level < te_pa_tariff_top
}

# A law it could move to that favors it (te_pa_gen_law_pick's best), not the one being enacted (that one it speeds
# and backs its side on, § 8.2).
te_pa_pt_ask_law = {
	te_pa_v_law_best > 0
	has_variable = te_pa_g_law_best_law
	var:te_pa_g_law_best_law > 0
	NOR = {
		AND = {
			te_pa_enacting_law_firm >= var:te_pa_g_law_best_law
			te_pa_enacting_law_firm <= var:te_pa_g_law_best_law
		}
	}
}

# The buyers' other petition (§ 6.3): a cartel the state knows of, lawful under the antitrust law in force and
# holding no charter, makes one of its inputs, so it can't be sued; the country could move to Antitrust Enforcement,
# under which it would not be lawful, and is not enacting it already. scope:te_pa_firm is the buyer.
te_pa_pt_ask_antitrust = {
	owner = {
		te_pa_gen_law_open_law_trust_busting = yes
		NOT = { is_enacting_law = law_type:law_trust_busting }
		has_variable_list = te_pa_cartels
		any_in_list = {
			variable = te_pa_cartels
			te_pa_cartel_makes_buyer_input = yes
			te_pa_holds_known_cartel = yes
			te_pa_cartel_lawful = yes
			NOT = { has_variable = te_pa_charter }
		}
	}
}

# Country scope. Its petition asks for a law (ask 1, or the buyers' Antitrust Enforcement, 4), not a tariff.
te_pa_pt_asks_law = {
	OR = {
		var:te_pa_pt_ask ?= 1
		var:te_pa_pt_ask ?= 4
	}
}

# Country scope. It is enacting its petition's law.
te_pa_pt_enacting_its_law = {
	te_pa_v_pt_law > 0
	te_pa_gen_enacting_law_index >= te_pa_v_pt_law
	te_pa_gen_enacting_law_index <= te_pa_v_pt_law
}

# Country scope. Its petition's tariff can still be set at the answer: it owns its market and the Tax Code's customs
# option is off (te_pa_tariffs_settable); a raise needs a trade policy that lets it collect and no treaty barring a
# tariff on the good.
te_pa_pt_tariff_takeable = {
	te_pa_tariffs_settable = yes
	has_variable = te_pa_pt_good
	has_variable = te_pa_pt_level
	trigger_if = {
		limit = { var:te_pa_pt_ask ?= 2 }
		te_pa_tariffs_bite = yes
		NOT = { te_pa_gen_tariff_barred_n = { VAR = te_pa_pt_good } }
	}
}

# Company scope (scope:te_pa_firm the firm at its turn): it is the firm whose declared move is under way. Read in a
# trigger_if behind its guard, so its negation never reads an unset move.
te_pa_is_move_firm = {
	owner = {
		trigger_if = {
			limit = { has_variable = te_pa_move_firm }
			var:te_pa_move_firm = scope:te_pa_firm
		}
		trigger_else = {
			always = no
		}
	}
}

# Country scope. A petition's record is open (te_pa_pt_open): its sides are being counted, or its answer is due.
te_pa_pt_is_open = {
	has_variable = te_pa_pt_firm
}

# Country scope. A petition for a law or a tariff is under way: its move counts down, and its record is open (a new
# one can't open while an old record stands: the twelve-month gap outlasts the record's grace).
te_pa_pt_running = {
	has_variable = te_pa_move_firm
	var:te_pa_move_kind ?= 3
	has_variable = te_pa_pt_firm
}

# Company scope, with scope:te_pa_pt_petitioner the petitioner: the firm may spend on its country's petition this

# Company scope, with scope:te_pa_mv_firm the petitioner (an answer): a firm on the other side of the petition,

	te_pa_v_tf_law > 0
	te_pa_gen_enacting_law_index >= te_pa_v_tf_law
	te_pa_gen_enacting_law_index <= te_pa_v_tf_law

# The pledged tariff holds: for a raise (te_pa_pl_dir 1) the good's level is at least the pledged one and the trade
# policy still lets it collect; for a lower tariff, the level is at most the pledged one.
te_pa_pl_tariff_held = {
	trigger_if = {
		limit = { var:te_pa_pl_dir ?= 1 }
		owner = {
			te_pa_tariffs_bite = yes
			te_pa_gen_tariff_n >= te_pa_v_tf_level
		}
	}
	trigger_else = {
		owner = { te_pa_gen_tariff_n <= te_pa_v_tf_level }
	}
}

# The pledged tariff is set again rather than counted broken (as the Tax Code's sync re-asserts its levels): an AI
# government keeps its pledge so, and in the month after the set anyone's is tried again once (probe 14: a set the
# engine kept back, by its cooldown, is not the player's doing). Only the level has changed: a raise's trade policy
# still lets it collect.
te_pa_pl_tariff_reassert = {
	owner = { te_pa_tariffs_settable = yes }
	OR = {
		owner = { is_ai = yes }
		te_pa_pl_set_since < 2
	}
	trigger_if = {
		limit = { var:te_pa_pl_dir ?= 1 }
		owner = { te_pa_tariffs_bite = yes }
	}
}

# Its free speed on the pledged law is due: a year since the last, or none yet (its own count, not the firm's or the
# law's yearly limit, te_pa_can_speed).
te_pa_pl_speed_due = {
	te_pa_pl_speed_since >= te_pa_pl_speed_months
}
