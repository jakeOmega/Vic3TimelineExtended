# Event-agency package E2 ("misc") — scripted triggers.
# Each block names the event(s) that read it.

# ----------------------------------------------------------------------------
# te_lolr.1 / te_lolr.2 — who is calling the guarantee
# ----------------------------------------------------------------------------

# te_lolr.1 (ROOT = the guarantor, scope:te_lolr_ward saved). True when ROOT
# was called as a TREATY guarantor, the first branch of
# te_monetary_on_country_default; false when it was called as the leader of
# the ward's monetary union (the else branch), which signed no article.
te_ea_lolr_called_by_treaty = {
	scope:te_lolr_ward ?= {
		te_mon_lolr_is_effective = yes
		var:te_mon_lolr_guarantor ?= root
	}
}

# te_lolr.2 (ROOT = the ward). The guarantor paid.
te_ea_lolr_answer_honoured = {
	has_variable = te_lolr_answer
	var:te_lolr_answer = 1
}

# ----------------------------------------------------------------------------
# banking_cycle_events.45 / .68 — the bailout appeal
# ----------------------------------------------------------------------------

# Candidate scope (a distressed trading partner, beside
# banking_is_bailout_recipient). Not already deciding whether to ask, or just
# having decided: .68 sets te_ea_bailout_asked for a year when it fires, so a
# human in a long downturn is not asked every month by another donor's pulse.
te_ea_bailout_can_be_asked = {
	NOT = { has_variable = te_ea_bailout_asked }
}

# banking_cycle_events.69 (ROOT = the country that asked). Which answer came
# back: 1 rescue package, 2 smaller grant, 3 refusal.
te_ea_bailout_answer_is = {
	has_variable = te_ea_bailout_answer
	var:te_ea_bailout_answer = $ANSWER$
}

# ----------------------------------------------------------------------------
# movement_events_te.4 / .17 — a great power condemns our treatment of minorities
# ----------------------------------------------------------------------------

# The accused (country scope): what movement_events_te.4's trigger used to
# require of it — an active civil rights movement facing discriminatory law.
# Read by the dispatch, by .17 on scope:cr_condemned_country and by .4.
te_ea_cr_condemnable = {
	has_active_civil_rights_movement = yes
	OR = {
		NOT = { has_progressive_minority_law = yes }
		has_law = law_type:law_active_persecution
		has_law = law_type:law_no_womens_rights
	}
}

# A candidate critic (country scope; ROOT = the accused): a great power that
# knows the language of civil rights and protects its own minorities, and has
# not been asked in the past year (.17 sets te_ea_cr_critic_cd when it fires).
te_ea_cr_can_condemn_root = {
	NOT = { this = root }
	country_rank >= rank_value:great_power
	has_technology_researched = civil_rights_movement
	has_progressive_minority_law = yes
	NOT = { has_variable = te_ea_cr_critic_cd }
}

# ----------------------------------------------------------------------------
# nuclear_incident.10 / .11 — the exercise they mistook
# ----------------------------------------------------------------------------

# The crisis opponent of a country whose exercise was misread (country
# scope). A player that could raise its own alert decides whether to
# (nuclear_incident.11); .10 raises an AI opponent's alert itself, as it
# always did. Read by .10's immediate and by its desc.
te_ea_nd_opponent_decides_alert = {
	is_ai = no
	nd_is_armed = yes
	nd_readiness_locked = no
	has_variable = nd_readiness_target
	var:nd_readiness_target < 3
}
