# ============================================================================
# Monetary policy, phase 5 — international monetary arrangements: triggers
# ============================================================================
# Spec: docs/systems/monetary_policy_design.md §15A, as shipped in §0.8.
# Country scope except the explicitly marked article-validity helpers.
# ============================================================================


# Has this country tied its money to another's? (§15A.1)
#
# STORED STATE ONLY, DELIBERATELY. §15A.1 defines an anchored country as "kind >
# 0 and the anchor is valid", and the anchor is valid when it has a dial — but
# te_mon_has_dial asks THIS question of the anchor, so testing validity here
# would make the two triggers call each other across the chain of anchors.
# Validity is step 1c's job instead (te_monetary_update_anchor): it demotes the
# kind to 0 the month the anchor stops qualifying, BEFORE step 2 asks for the
# dial, so within a pulse the stored pair is always a checked one.
#
# That is also what forbids chains and cycles: an anchored country has no dial,
# and every gate that creates an anchor asks the anchor for one.
te_mon_is_anchored = {
	has_variable = te_mon_anchor_kind
	var:te_mon_anchor_kind > 0
	has_variable = te_mon_anchor
}

# Is the stored anchor still one? The monthly half of §15A.1's "validity":
# alive, and holding a dial of its own.
te_mon_anchor_is_valid = {
	has_variable = te_mon_anchor
	exists = var:te_mon_anchor
	var:te_mon_anchor = {
		NOT = { this = root }
		te_mon_has_dial = yes
	}
}

# Can THIS country serve as somebody's anchor? One name for the rule, so the
# article gates, the discovery and the validity check cannot drift apart.
te_mon_can_be_anchor = {
	te_mon_has_dial = yes
}

# Currency compatibility survives ratification: a peg is not currency reform.
# Country scope, OTHER is the anchor. Do not require the pegger's own dial:
# entering a peg deliberately removes it, and bankless countries can peg.
te_mon_peg_regime_fits = {
	te_mon_is_command = no
	te_mon_is_dollarised = no
	NOT = { te_mon_is_on_currency_board = yes }
	NOT = { te_mon_is_union_adopter = yes }
	OR = {
		has_law_or_variant = law_type:law_fiat_currency
		has_law_or_variant = law_type:law_digital_currency
		AND = {
			OR = {
				has_law_or_variant = law_type:law_commodity_money
				has_law_or_variant = law_type:law_gold_standard
			}
			$OTHER$ = {
				OR = {
					has_law_or_variant = law_type:law_commodity_money
					te_mon_is_on_gold = yes
				}
			}
		}
	}
}

# Finite treasury lending and debt supervision do not require money creation.
# Shared ongoing provider requirement; rank/GDP superiority is signing-only.
te_mon_backstop_provider_valid = {
	has_law_or_variant = law_type:law_national_bank
	te_mon_is_command = no
	in_default = no
}

# ARTICLE scope. Discovery must not wait for the engine to withdraw an invalid
# article before dropping its benefits, costs and (for swaps) settling its loan.
te_mon_peg_article_valid = {
	target_country = { te_mon_can_be_anchor = yes }
	source_country = { te_mon_peg_regime_fits = { OTHER = prev.target_country } }
}

te_mon_support_article_valid = {
	OR = {
		AND = {
			OR = {
				has_type = swap_line
				has_type = lender_of_last_resort
			}
			source_country = { te_mon_backstop_provider_valid = yes }
		}
		AND = {
			has_type = debt_receivership
			target_country = { te_mon_backstop_provider_valid = yes }
			source_country = { te_mon_is_command = no }
		}
	}
}

# --- 5a: reading the treaties from the pulse --------------------------------
# `any_scope_treaty` iterates IN-FORCE treaties only. The direction test reads
# the article's own source / target links; ROOT must be the country asking,
# which it is at every caller (the monthly pulse and te_monetary_internal.2).
# no_duplicate_treaty_article is NOT the model: its scope:treaty guard only
# resolves in can_ratify.

# Is ROOT the pegger in some in-force peg article? [6c] BOTH TYPES: an imposed
# peg (113) makes its source kind 1 exactly as a chosen one does — the terms
# differ, the anchored state does not (§15C.3, ruling C1). Every site that
# walks for a peg carries the same OR; they are listed in §15C.3's file table.
te_mon_has_peg_article = {
	any_scope_treaty = {
		any_scope_article = {
			OR = {
				has_type = currency_peg
				has_type = imposed_currency_peg
			}
			source_country = root
		}
	}
}

# ...and is it one whose target is $ANCHOR$-valid right now?
te_mon_has_valid_peg_article = {
	any_scope_treaty = {
		any_scope_article = {
			OR = {
				has_type = currency_peg
				has_type = imposed_currency_peg
			}
			source_country = root
			te_mon_peg_article_valid = yes
		}
	}
}

# [6c] ...and is the peg it is living under an IMPOSED one? Read by the spread,
# by the standing cut a chosen discipline earns and an imposed one does not, and
# by te_peg.2's missing Break option. A treaty walk rather than a stored flag:
# the anchored state has one pair and one kind, and which article put it there
# is a property of the treaty, not of the state.
te_mon_is_imposed_pegger = {
	any_scope_treaty = {
		any_scope_article = {
			has_type = imposed_currency_peg
			source_country = root
		}
	}
}

# Does ROOT stand in a valid in-force support article of $TYPE$ as its TARGET (the party that
# receives: the swap line's recipient, the guarantee's ward)?
te_mon_is_article_target = {
	any_scope_treaty = {
		any_scope_article = {
			has_type = $TYPE$
			te_mon_support_article_valid = yes
			target_country = root
		}
	}
}

# ...or as its SOURCE (the provider, the guarantor)?
te_mon_is_article_source = {
	any_scope_treaty = {
		any_scope_article = {
			has_type = $TYPE$
			te_mon_support_article_valid = yes
			source_country = root
		}
	}
}

# [6a] Does ROOT hold the swap line's RECIPIENT role at all — by treaty, or
# because te_debug_monetary.11 pinned one? One name for it, so the discovery's
# role branch and its settle-on-role-end check cannot disagree and settle a line
# that is still standing.
te_mon_has_swap_line_role = {
	OR = {
		te_mon_is_article_target = { TYPE = swap_line }
		AND = {
			has_variable = te_mon_debug_swap_on
			var:te_mon_debug_swap_on = 1
			has_variable = te_mon_debug_swap_provider
			exists = var:te_mon_debug_swap_provider
		}
	}
}

# Guarded reads of what the discovery stored.
te_mon_is_swap_recipient = {
	has_variable = te_mon_swap_recipient
	var:te_mon_swap_recipient = 1
}

te_mon_is_lolr_ward = {
	has_variable = te_mon_lolr_ward
	var:te_mon_lolr_ward = 1
}

# [6c] Is this country's treasury run by somebody else's commission? (§15C.3)
te_mon_is_under_receivership = {
	has_variable = te_mon_receivership
	var:te_mon_receivership = 1
	has_variable = te_mon_receiver
}

# Is the guarantee worth anything this month? Not while the guarantor's
# renege clock runs: "guarantees are only worth the last one honoured".
te_mon_lolr_is_effective = {
	te_mon_is_lolr_ward = yes
	has_variable = te_mon_lolr_guarantor
	exists = var:te_mon_lolr_guarantor
	var:te_mon_lolr_guarantor = {
		NOT = {
			AND = {
				has_variable = te_mon_lolr_suspended_months
				var:te_mon_lolr_suspended_months > 0
			}
		}
	}
}

# [6c] Is some TREATY already taking a share off this country's debt-load
# premium — a guarantee that is worth something, or a receivership? One name for
# it, because a bloc's tier-3 backstop must not pay the same relief twice
# (§15A.3's rule, extended to the receiver). 112 and 114 exclude each other by
# type, so this can only ever be one of them.
te_mon_has_treaty_debt_relief = {
	OR = {
		te_mon_lolr_is_effective = yes
		te_mon_is_under_receivership = yes
	}
}

# Does a kind-1 pegger run te_peg_confidence? (§15A.2) Kinds 2 and 3 have no peg
# to break: a union's valve is the premium, a board's is liberty desire.
te_mon_has_anchor_peg = {
	te_mon_is_anchored = yes
	var:te_mon_anchor_kind = 1
}

# Capital controls, however they came to be on: the dashboard tool, or the
# kind-1 crisis's Defend option forcing them for a year. Read by controls_damp
# and the fatigue counter, so forced controls buy and cost what chosen ones do.
te_mon_capital_controls_in_force = {
	OR = {
		banking_tool_capital_controls_active = yes
		AND = {
			has_variable = te_mon_emergency_controls_months
			var:te_mon_emergency_controls_months > 0
		}
	}
}


# --- Article gates (§15A.2 "Gating") ----------------------------------------
# Written once here so `possible`, the AI filters and the debug harness agree.
# Scope: the would-be SOURCE; $OTHER$ is the other party.

# currency_peg: the pegger has no higher-kind anchor, the target holds a dial
# and is a great or major power or the pegger's market owner.
te_mon_can_peg_to = {
	te_mon_peg_regime_fits = { OTHER = $OTHER$ }
	NOT = { this = $OTHER$ }
	NOT = { te_mon_is_command = yes }
	NOT = {
		AND = {
			te_mon_is_anchored = yes
			var:te_mon_anchor_kind > 1
		}
	}
	$OTHER$ = {
		te_mon_can_be_anchor = yes
		OR = {
			country_rank >= rank_value:major_power
			AND = {
				exists = prev.market
				prev.market.owner = this
			}
		}
	}
}

# swap_line / lender_of_last_resort: the provider has a national bank and
# outranks or out-GDPs the recipient.
te_mon_can_backstop = {
	te_mon_backstop_provider_valid = yes
	NOT = { this = $OTHER$ }
	OR = {
		country_rank > $OTHER$.country_rank
		gdp > $OTHER$.gdp
	}
}


# --- [6c] The coercive gates (§15C.3, ruling C4: IN DEFAULT ONLY) ------------
# Scope: the would-be IMPOSER; $OTHER$ is the country it would impose on. The
# pretext is deliberately narrow — the wider draft (inflation >= 10, or living
# in the imposer's market) was rejected as too loose. A currency in trouble, or
# in somebody's economic space, is not by itself grounds; an insolvent
# government is. It is also what keeps the AI world from filling with these.

# imposed_currency_peg: the imposer can be an anchor and is a major power or
# better; the target is insolvent and could actually take a peg.
te_mon_can_impose_peg_on = {
	NOT = { this = $OTHER$ }
	te_mon_can_be_anchor = yes
	country_rank >= rank_value:major_power
	$OTHER$ = {
		in_default = yes
		te_mon_peg_regime_fits = { OTHER = prev }
		NOT = {
			AND = {
				te_mon_is_anchored = yes
				var:te_mon_anchor_kind > 1
			}
		}
	}
}

# debt_receivership: te_mon_can_backstop with the roles swapped — the receiver
# keeps a bank of its own and outranks or out-GDPs the debtor — and the debtor
# is insolvent. A receiver is a creditor, not a guarantor, but it has to be a
# financial power for the commission to mean anything to the other creditors.
te_mon_can_receive_for = {
	te_mon_can_backstop = { OTHER = $OTHER$ }
	$OTHER$ = {
		in_default = yes
		te_mon_is_command = no
	}
}

# Phase 5c — subject currency boards (§15A.4)

# Is this country a subject WITHOUT monetary autonomy? Vanilla's subject types
# carry `autonomy_level`, which script cannot read, so this is the OR-list of the
# types that have `autonomy_level = 1`.
#
# RE-DERIVE IT ON EVERY VANILLA BUMP (vanilla_patch_runbook.md):
#   grep -n 'autonomy_level\|^subject_type' <game>/common/subject_types/*.txt
# Verified against 1.14.3, 2026-09-21: puppet, vassal, colony, crown_land are 1;
# protectorate, tributary, dominion, personal_union, chartered_company are 2 and
# keep their own monetary arrangement (they may sign a currency_peg with the
# overlord like anyone else). The mod defines no subject types of its own.
te_mon_is_board_subject = {
	is_subject = yes
	OR = {
		is_subject_type = subject_type_puppet
		is_subject_type = subject_type_vassal
		is_subject_type = subject_type_colony
		is_subject_type = subject_type_crown_land
	}
}

# ...and is the board actually in force? Kind 3, stored and checked.
te_mon_is_on_currency_board = {
	te_mon_is_anchored = yes
	var:te_mon_anchor_kind = 3
}
