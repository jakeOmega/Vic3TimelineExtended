# ============================================================================
# AUTOMATION POLICY — triggers (era 13 spec § 7.1)
# ============================================================================
# Laws: common/laws/te_automation_policy_laws.txt. Values:
# common/script_values/te_automation_policy_values.txt.
#
# WHICH METHODS COUNT AS ROBOTIC OR AI-MANAGED. A method counts where robots
# or AI do work people did: in each group, the first robotic or AI-run tier
# and every tier above it, so Human Labor Quotas never moves a building to a
# method that employs fewer people. Each such method marks itself with two
# lines: `building_robotic_or_ai_managed_bool = yes` in its
# building_modifiers' `unscaled` block (the type is in
# common/modifier_type_definitions/te_automation_policy_modifier_types.txt),
# which te_ap_runs_automated_method reads, and
# `disallowing_laws = { law_human_labor_quotas }`. Each group keeps an
# ungated first method, so no building is left without one. A new method
# takes both lines, and its building joins te_ap_automatable_building;
# test_automation_policy.py checks the two lines name the same methods, the
# ladder, every group's fallback and the building list. The 40 methods:
#
#   Factory automation, the robotic tier up (Robotics, era 8, and later):
#     Tooling Workshop, Furniture Manufactory, Motor Industry: Robotic
#       Assembly Lines and the AI-managed tier.
#     Automotive Industry: Robotic Assembly Lines, Intelligent Manufacturing
#       Systems (Supply Chain Management, above robotic assembly in the same
#       group), Self-Optimizing Production.
#     Aerospace Industry: Robotic Assembly Handling, AI-Optimized Operations.
#     Food Industry: Robotic Food Processing.
#   Factory automation, the AI-managed tier (Artificial General Intelligence):
#     Arms Industry and Artillery Foundry, Munition Plant, Chemical Plant and
#     the Synthetics Plants, Explosives Factory, Electronics (the fab), Textile
#     Mill.
#   AI-run infrastructure: the AI-managed Power Plant trio and its Aneutronic
#     Fusion variant (Smart Grids), Autonomous Logistics at the Trade Center
#     (Smart Grids, with a flat crew in place of International Trading's
#     per-level staff),
#     Autonomous Networks (Smart Grids), Fully Automated Content Creation (AI
#     Agents) and AI-Mediated Mind Sharing (Telepathic Communities) in the
#     Software Industry, Automated Labs at the University and the automated
#     Deep-Sea Mine, capped and uncapped (AGI).
#   Robots and self-driving machines in services, transport and farming:
#     Robotic Service Providers (Humanoid Robotics) and Drone Delivery Systems
#     (Autonomous Vehicles) in Urban Centers; Autonomous Trains and Vactrain
#     Freight (Vactrains), the
#     Autonomous Highway and Automated Harvesters and Planters (Autonomous
#     Vehicles).
#
# Left out on purpose:
#   - base methods that set what a building makes and from what, not who does
#     the work: Industrial Robotics and Nanobot Assemblers (Tooling Workshop),
#     Robotic Ship Assembly (Shipyard, whose later Advanced Metamaterials
#     method employs fewer still and would have to go too), Automated Chemical
#     Reactors (the same staff as every fertilizer method);
#   - methods that make robots or chips: the Robotics Industry's base methods
#     (its electronics automation, the AI-managed fab, counts), Neuromorphic
#     Computing;
#   - methods that assist people rather than replace them: AI Assisted
#     Research;
#   - mining machinery, not robots or AI: Ore Sorting and Separation
#     (Robotics), Laser Excavation, Smart Miners;
#   - government staffing, which its own laws set: AI Governance, Automated
#     Casework (Automated Bureaucracy);
#   - ownership methods: Automated Trading, the Manor House's AI ownership.
# ============================================================================

# Building scope. The building runs one of the robotic or AI-managed methods
# above: one of its active methods carries the marker. Its levels count toward
# the automated share. The line sits in `unscaled`, so it holds at any
# staffing, and a building counts once however many of its groups run one.
# Only production methods carry it: a building's modifier: read also carries
# its country's lines (scripting_best_practices.md), so one on a law or a
# technology would mark every building. Whether a method's line reaches the
# read is the self-check automation_marker (te_ap_marker_selfcheck,
# common/scripted_effects/te_automation_policy_effects.txt).
te_ap_runs_automated_method = {
	modifier:building_robotic_or_ai_managed_bool = yes
}

# Building scope. TEMPORARY, for the automation_marker self-check only: the
# building runs one of three sample methods, named directly as
# te_ap_runs_automated_method named all of them before the marker. Robotic
# Assembly Lines in a Tooling Workshop (a per-level staff), Robotic Service
# Providers in an Urban Center, and Autonomous Logistics in a Trade Center (a
# flat crew).
te_ap_marker_sample = {
	OR = {
		AND = {
			is_building_type = building_tooling_workshop
			has_active_production_method = pm_robotic_assembly_lines_building_tooling_workshop
		}
		AND = {
			is_building_type = building_urban_center
			has_active_production_method = pm_robotic_service_providers
		}
		AND = {
			is_building_type = building_trade_center
			has_active_production_method = pm_trade_center_trade_quantity_advanced_4
		}
	}
}

# Country scope. The law whose modifier reads the automated share is in force.

	OR = {
		has_law = law_type:law_robot_tax
		has_law = law_type:law_automation_dividend
	}
}
