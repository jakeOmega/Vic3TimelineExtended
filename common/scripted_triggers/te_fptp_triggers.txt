# First Past the Post triggers (amendment_first_past_the_post,
# common/amendments/te_first_past_the_post.txt).
# Design: docs/superpowers/specs/2026-10-25-first-past-the-post-design.md.

# Country scope. The amendment is on the voting law in force.
te_fptp_in_force = {
	active_law:lawgroup_distribution_of_power ?= {
		has_amendment = amendment_type:amendment_first_past_the_post
	}
}

# Law scope. A law the amendment can attach to (its allowed_laws, restated).
te_fptp_allowed_voting_law = {
	OR = {
		type = law_type:law_landed_voting
		type = law_type:law_wealth_voting
		type = law_type:law_census_voting
		type = law_type:law_universal_suffrage
		type = law_type:law_anarchy
	}
}

# Country scope. A law the amendment can attach to is in force.
te_fptp_on_allowed_voting_law = {
	active_law:lawgroup_distribution_of_power ?= {
		te_fptp_allowed_voting_law = yes
	}
}

# Country scope. A law the amendment can attach to is being enacted.
te_fptp_enacting_allowed_voting_law = {
	OR = {
		is_enacting_law = law_type:law_landed_voting
		is_enacting_law = law_type:law_wealth_voting
		is_enacting_law = law_type:law_census_voting
		is_enacting_law = law_type:law_universal_suffrage
		is_enacting_law = law_type:law_anarchy
	}
}

# Country scope. The countries vanilla's usa_two_party_system join-weight term
# covered (its conditions, restated), on a law the amendment can attach to:
# the seed gives them the amendment once (te_fptp_seed).
te_fptp_seed_eligible = {
	is_in_geographic_region = geographic_region_north_america
	OR = {
		country_has_primary_culture = cu:yankee
		country_has_primary_culture = cu:dixie
	}
	has_law_or_variant = law_type:law_presidential_republic
	te_fptp_on_allowed_voting_law = yes
}

# Party scope, with scope:te_fptp_country saved (te_fptp_refresh_major_parties).
