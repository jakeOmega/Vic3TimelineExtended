# ---- The floor -------------------------------------------------------------------------

# The docket may take up an item that opens a resolution: no vote is in session
# or reserved, and the post-vote recess is over. (Business that opens no
# resolution — an aid appeal, a peacekeeping appeal, a trade dispute — is
# taken up whatever the floor is doing.)
# Scope: any
un_docket_floor_free = {
	NOT = { has_global_variable = un_vote_active }
	un_floor_open_to_ai = yes
}

# The scoped member could be offered a docket item now: a member, not already
# offered the item in hand, and able to table (the recess binds only the AI).
# Scope: the prospective recipient
un_docket_can_be_offered = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	NOT = { has_variable = un_dkt_offered }
	un_may_table_now = yes
	# Article 19 (redesign phase 5): a member that has lost its vote tables
	# nothing.
	NOT = { un_dues_vote_suspended = yes }
}

# ---- Situations --------------------------------------------------------------------------

# A member the emergency lending facility could help (item 14, redesign
# phase 5): a banking panic, or a default, in a contagion wave (it was hit
# by the last wave, or a Great Depression is on somewhere), with no loan
# already running and none offered in the last ten years, while the paying
# members' dues give the UN a budget to lend from.
# Scope: the prospective borrower
un_docket_loan_candidate = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	has_journal_entry = je_banking_cycle
	OR = {
		in_default = yes
		banking_cycle_is_panic = yes
	}
	OR = {
		has_variable = banking_crisis_wave_crashed
		any_country = { has_modifier = great_depression_modifier }
	}
	NOT = { has_variable = un_dkt_loan_raised }
	NOT = { has_modifier = un_loan_repayment_modifier }
	un_loan_amount_value > 0
}

# A war of aggression the docket should take up, seen from the aggressor: it
# began the war (the dossier's test), has fought it un_docket_war_months, holds
# no bound mandate for it, has not been docketed for it, and some member it is
# at war with has a state devastated past un_docket_war_devastation.
# Scope: the prospective aggressor
un_docket_war_candidate = {
	is_at_war = yes
	is_diplomatic_play_initiator = yes
	has_variable = un_dkt_war_months
	var:un_dkt_war_months >= un_docket_war_months
	NOT = { has_variable = un_dkt_war_raised }
	NOT = { var:un_mandate_current ?= { has_tag = un_mandate_bound } }
	any_country = {
		has_war_with = PREV
		is_diplomatic_play_initiator = no
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		any_scope_state = { devastation > un_docket_war_devastation }
	}
}

# A member attacked by scope:condemned_country, with a state devastated past
# the threshold: a victim the war item can name.
# Scope: the prospective victim
un_docket_war_victim = {
	has_war_with = scope:condemned_country
	is_diplomatic_play_initiator = no
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	any_scope_state = { devastation > un_docket_war_devastation }
}

# A collapsing state the docket should take up. Scope: country
un_docket_collapse_candidate = {
	NOT = { is_country_type = decentralized }
	NOT = { has_variable = un_dkt_collapse_raised }
	OR = {
		has_modifier = failed_state_modifier
		has_journal_entry = je_state_collapse
	}
}

# A famine the docket should take up. Scope: country
un_docket_famine_candidate = {
	NOT = { is_country_type = decentralized }
	NOT = { has_variable = un_dkt_famine_raised }
	any_scope_state = { has_famine = yes }
}

# A new nuclear power the docket has not yet seen. Scope: country
un_docket_proliferation_candidate = {
	NOT = { has_variable = un_dkt_nuke_raised }
	has_variable = nuclear_weapons_program_first_nuke_done
	var:nuclear_weapons_program_first_nuke_done > 0
}

# A colonial empire that has collapsed and not yet been docketed. Scope: country
un_docket_colonial_candidate = {
	has_variable = colonial_empire_collapsed_recently
	NOT = { has_variable = un_dkt_colonial_raised }
}

# Humanity has reached another world (the first Moon landing or the first
# colony) and the Assembly has not yet taken it up. Scope: any
un_docket_space_ready = {
	NOT = { has_global_variable = un_dkt_space_raised }
	OR = {
		has_global_variable = sr_global_first_moon_landing
		has_global_variable = sr_colonization_stage
	}
}

# A warming threshold has been crossed that the Assembly has not yet taken up.
# Scope: any
un_docket_warming_ready = {
	has_global_variable = un_dkt_warming_next
	global_var:un_dkt_warming_next < 4
	temperature_anomaly_display >= un_docket_warming_threshold
}

# Two members trading under an embargo, seen from the party that is not yet
# docketed. Scope: country
un_docket_embargo_candidate = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	NOT = { has_variable = un_dkt_embargo_raised }
	any_country = { un_docket_embargo_counterpart = yes }
}

# The scoped country is a member under an embargo with PREV, a member, that is
# a dispute between them. An embargo the Assembly imposed to enforce sanctions
# (either one in the other's un_embargo_by, phase 5) is the Assembly's own
# policy, not a dispute to arbitrate.
# Scope: country; PREV the other member
un_docket_embargo_counterpart = {
	NOT = { this = PREV }
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	has_diplomatic_pact = { who = PREV type = embargo }
	NOT = { is_target_in_variable_list = { name = un_embargo_by target = PREV } }
	NOT = { PREV = { is_target_in_variable_list = { name = un_embargo_by target = PREV } } }
}

# A power that can be asked to send peacekeepers to a collapsing state: a
# major-power member not already deployed or observing somewhere. The docket
# also excludes the host and anyone at war with it.
# Scope: country
un_docket_peacekeeping_power = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	country_rank >= rank_value:major_power
	NOT = { has_modifier = un_peacekeeping_deployed_modifier }
	NOT = { has_modifier = un_observer_mission_modifier }
}

# A power that can be asked for famine aid: a major-power member not already
# giving. The docket also excludes the famine-struck country and anyone at
# war with it.
# Scope: country
un_docket_aid_power = {
	je:je_united_nations ?= { has_modifier = un_member_modifier }
	country_rank >= rank_value:major_power
	NOT = { has_modifier = un_humanitarian_aid_modifier }
	NOT = { has_modifier = un_humanitarian_token_modifier }
}

# ---- Who may be offered what ---------------------------------------------------------------

# An appeal (un_events.2) passed on by the aggrieved party goes to a member
# with a stake: a permanent member, a rival of the accused, or an ally or
# bloc partner of the aggrieved. Never one on the accused's side.
# Scope: the prospective recipient, with scope:condemned_country and
# scope:un_dkt_victim saved
un_docket_appeal_second = {
	un_docket_can_be_offered = yes
	NOT = { this = scope:condemned_country }
	# The aggrieved party may no longer exist (annexed by the war it appealed).
	NOT = { scope:un_dkt_victim ?= this }
	NOT = { is_in_war_together = scope:condemned_country }
	NOT = { has_treaty_alliance_with = { TARGET = scope:condemned_country } }
	NOT = { is_in_same_power_bloc = scope:condemned_country }
	OR = {
		is_un_permanent_member = yes
		has_diplomatic_pact = { who = scope:condemned_country type = rivalry }
		scope:un_dkt_victim ?= {
			OR = {
				has_treaty_alliance_with = { TARGET = PREV }
				is_in_same_power_bloc = PREV
			}
		}
	}
}

# Business with no situation behind it that the docket could take up now.
# Scope: any
un_docket_agenda_ready = {
	NOT = { has_global_variable = un_dkt_agenda_months_left }
	OR = {
		AND = {
			un_docket_topic_open_human_rights = yes
			any_country = { un_docket_may_propose_human_rights = yes }
		}
		AND = {
			un_docket_topic_open_icc = yes
			has_global_variable = un_agency_unhrc
			any_country = { un_docket_may_propose_icc = yes }
		}
		AND = {
			un_docket_topic_open_heritage = yes
			any_country = { un_docket_may_propose_heritage = yes }
		}
		AND = {
			un_docket_topic_open_pandemic = yes
			any_country = { un_docket_may_propose_pandemic = yes }
		}
		AND = {
			un_docket_topic_open_refugee = yes
			any_country = { un_docket_may_propose_refugee = yes }
		}
		AND = {
			un_docket_topic_open_law_of_sea = yes
			any_country = { un_docket_may_propose_law_of_sea = yes }
		}
		AND = {
			un_docket_topic_open_decolonization = yes
			any_country = { un_docket_may_propose_decolonization = yes }
		}
	}
}

# The accused's grievance is of cause $CAUSE$ (1 nuclear, 2 war, 3 covert), as
# the docket wrote it on the accused when it took the item up.
# Scope: the accused (scope:condemned_country)
un_docket_cause_is = {
	has_variable = un_dkt_cause
	var:un_dkt_cause = $CAUSE$
}
