# Experiment (#933): does `should_remove` take a building out of the AI's
# subsidies? It sits beside must_have and nice_to_have among the game
# executable's strings, but no vanilla strategy uses it and no engine doc lists
# the subsidy priorities, so it is unread. The three administrative strategies
# below have no subsidies block in vanilla, so each INJECT adds one with no key
# in common with vanilla's. The default strategy's INJECT
# (edited_default_strategy.txt) sets the same value on a key vanilla's block
# already holds, where which value the engine keeps is unread too. A strategy's
# value overrides the default's (vanilla's 00_default_strategy.txt). The yearly
# self-check te_ai_subsidy_should_remove_selfcheck reads the two groups apart;
# what each result means: docs/testing/ai-economy-playtest.md.
INJECT:ai_strategy_agricultural_expansion = {
	subsidies = {
		building_power_plant = should_remove
	}
}

INJECT:ai_strategy_plantation_economy = {
	subsidies = {
		building_power_plant = should_remove
	}
}

INJECT:ai_strategy_placate_population = {
	subsidies = {
		building_power_plant = should_remove
	}
}
