# =================================================================
# TREATY ARTICLE: debt_receivership             (monetary policy 6c)
# Spec: docs/systems/monetary_policy_design.md §15C.3, ruling C2.
#
# DIRECTED: source = the DEBTOR (concedes), target = the RECEIVER (imposes).
#
# THE COERCIVE COUNTERPART TO 112 IS NOT A GUARANTEE AT GUNPOINT. Forcing a
# country to stand behind another's debts is not a thing history did; forcing a
# weaker country to accept externally supervised finances is — Egypt's Caisse
# de la Dette 1876, the Ottoman Public Debt Administration 1881, the Dominican
# customs receivership 1905. So this is a different relief/extraction shape,
# not 112 with the sign flipped:
#
#   THE DEBTOR gets the guarantee's relief (te_mon_lolr_relief, through the
#   te_mon_receivership role flag) — creditors ARE reassured by a receiver,
#   which is why the instrument existed — and pays for it every month:
#   te_mon_receivership_take, 0.1% of its GDP, straight to the receiver ("the
#   customs receipts"), plus -10 legitimacy for ceding its treasury. Its
#   default under a receivership pays it nothing: te_monetary_on_country_default
#   tests the GUARANTEE, not this, so there is no te_lolr.1 and the ward's call
#   counter never moves while the commission sits. A receiver is a creditor,
#   not a guarantor.
#
#   THE RECEIVER takes the remittance, carries a GDP-scaled share of the relief
#   in te_mon_arr_provider_pts exactly as a guarantor does, and generates the
#   heaviest leverage of the five monetary articles: §15B.1's "the client's
#   growing dependence" is the literal thing here.
#
# THE EXIT RAMP. non_fulfillment withdraws the article once the debtor has held
# scaled_debt under 0.25 and stayed out of default for twelve consecutive
# months (step 1c counts them into te_mon_receivership_clear_months, since a
# treaty condition cannot keep a count of its own). The Dominican 1941 shape:
# a sentence, not a permanent tribute.
#
# NO PRESTIGE LINE (§15C.1, ruling H5): a receiver's reward is leverage.
#
# TECH GATE: international_exchange_standards, not 112's
# intergovernmental_organizations. The history is era 3-4; Bretton Woods would
# put the receivership two eras after the thing it models (ruling C2).
#
# Modifier blocks and hooks: see 110_currency_peg.txt's header.
# =================================================================
debt_receivership = {
	icon = "gfx/interface/icons/law_icons/central_bank_independence.dds"
	kind = directed
	cost = 150

	flags = {
		can_be_enforced
		can_be_renegotiated
		hostile
	}

	# The receiver wants it, so the receiver pays the upkeep — 103's rule. What
	# it gets back is te_mon_receivership_take, which dwarfs it.
	maintenance_paid_by = target_country
	usage_limit = once_per_treaty

	# Two names behind one debt are not two rescues. Declared in 112 too.
	mutual_exclusions = {
		lender_of_last_resort
		debt_receivership
	}

	unlocked_by_technologies = {
		international_exchange_standards
	}

	# The debtor's government looks weak for ceding its treasury.
	source_modifier = {
		country_legitimacy_base_add = -10
		# [6b] The heaviest of the five (§15C.3, ruling C2).
		#
		# PLACEMENT (fixed 2026-09-21): the modifier is generated AGAINST the
		# country whose block carries it (see 110_currency_peg.txt's note for the
		# vanilla evidence). It shipped on the RECEIVER's target_modifier, which
		# handed the debtor leverage over the country running its treasury. The
		# DEBTOR is the source.
		country_treaty_leverage_generation_add = 500
	}

	visible = {
		te_mon_full_system = yes
	}

	# THE PRETEXT (§15C.3, ruling C4: IN DEFAULT ONLY). The draft's
	# scaled_debt >= 0.75 alternative was rejected — a heavily indebted
	# government is not yet somebody else's to run. Split between `possible`
	# (either direction) and `can_ratify` (this one) for the reason 113's
	# header gives: the direction is not fixed in `possible`.
	possible = {
		has_diplomatic_relevance = scope:other_country
		NOT = { is_country_type = decentralized }
		scope:other_country = {
			NOT = { is_country_type = decentralized }
		}
		OR = {
			te_mon_can_receive_for = { OTHER = scope:other_country }
			scope:other_country = {
				te_mon_can_receive_for = { OTHER = root }
			}
		}
	}

	# A receiver in default runs nobody's finances. 112's own check, with the
	# sides the other way round — and its own tooltip key, because "the
	# guarantor" is the wrong noun for a commission.
	requirement_to_maintain = {
		trigger = {
			custom_tooltip = {
				text = te_receivership_receiver_solvent_tt
				scope:target_country = { in_default = no }
			}
		}
	}

	can_ratify = {
		# The pretext with the sides fixed. It already carries
		# te_mon_can_backstop's rank / GDP test with the roles swapped, so
		# there is no separate te_backstop_provider_outranks_tt line here.
		custom_tooltip = {
			text = te_receivership_pretext_tt
			scope:target_country = {
				te_mon_can_receive_for = { OTHER = scope:source_country }
			}
		}
		# One commission, and never beside a guarantee of the same debt.
		custom_tooltip = {
			text = te_lolr_already_guaranteed_tt
			scope:source_country = {
				NOT = {
					any_scope_treaty = {
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							OR = {
								AND = {
									has_type = debt_receivership
									source_country = scope:source_country
								}
								AND = {
									has_type = lender_of_last_resort
									target_country = scope:source_country
								}
							}
						}
					}
				}
			}
		}
	}

	# THE EXIT RAMP (§15C.3). The counter is step 1c's; this reads it. `withdraw`
	# rather than `freeze` for the same reason the friendly three take withdraw
	# (ruling H7): a frozen article's behaviour under any_scope_treaty is §17
	# check 15, unanswered — and a receivership that has done its work should end.
	non_fulfillment = {
		consequences = withdraw
		conditions = {
			monthly = {
				scope:article = {
					source_country = {
						has_variable = te_mon_receivership_clear_months
						var:te_mon_receivership_clear_months >= te_mon_receivership_exit_months
					}
				}
			}
		}
	}

	on_entry_into_force = {
		scope:article_options = {
			source_country = { trigger_event = { id = te_monetary_internal.2 days = 1 } }
			target_country = { trigger_event = { id = te_monetary_internal.2 days = 1 } }
		}
	}

	on_withdrawal = {
		scope:withdrawing_country ?= { trigger_event = { id = te_monetary_internal.2 days = 1 } }
		scope:non_withdrawing_country ?= { trigger_event = { id = te_monetary_internal.2 days = 1 } }
	}

	ai = {
		treaty_categories = { economy }
		# REQUEST ONLY: nobody offers to have its treasury taken over.
		article_ai_usage = { request }

		evaluation_chance = {
			value = 0.01
			if = {
				limit = {
					exists = scope:other_country
					has_attitude = { who = scope:other_country attitude = domineering }
					exists = scope:source_country.market
					scope:source_country.market.owner = this
				}
				add = 0.08
			}
		}

		inherent_accept_score = {
			value = 0

			# --- the DEBTOR'S side ------------------------------------------
			if = {
				limit = { scope:source_country ?= this }
				# -10, not the draft's -40 + 30-if-in-default: the pretext makes
				# insolvency a constant, and a government whose creditors are
				# otherwise sending gunboats takes the commission over them.
				add = {
					desc = "te_ai_receivership_base"
					value = -10
				}
				if = {
					limit = {
						OR = {
							has_attitude = { who = scope:target_country attitude = domineering }
							has_attitude = { who = scope:target_country attitude = belligerent }
							has_attitude = { who = scope:target_country attitude = antagonistic }
						}
					}
					add = {
						desc = "AI_HOSTILE_ATTITUDE"
						value = -20
					}
				}
			}
			# --- the RECEIVER'S side ----------------------------------------
			else = {
				if = {
					limit = {
						exists = scope:source_country.market
						scope:source_country.market.owner = this
					}
					add = {
						desc = "te_ai_peg_shared_market"
						value = 20
					}
				}
				if = {
					limit = { country_rank >= rank_value:great_power }
					add = {
						desc = "te_ai_backstop_influence"
						value = 10
					}
				}
				if = {
					limit = {
						has_attitude = { who = scope:source_country attitude = domineering }
					}
					add = {
						desc = "AI_DOMINEERING_ATTITUDE"
						value = 15
					}
				}
				if = {
					limit = {
						OR = {
							has_attitude = { who = scope:source_country attitude = genial }
							has_attitude = { who = scope:source_country attitude = cooperative }
							has_attitude = { who = scope:source_country attitude = conciliatory }
						}
					}
					add = {
						desc = "AI_FRIENDLY_ATTITUDE"
						value = -20
					}
				}
			}
		}
	}

	# Heavier than the imposed peg: taking a country's treasury is a larger
	# thing than taking its exchange rate.
	wargoal = {
		execution_priority = 70
		contestion_type = control_target_country_capital

		maneuvers = {
			value = 0

			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 2.5
			}

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
					max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy = {
			value = 0

			add = {
				desc = "INFAMY_BASE_VALUE"
				value = 10
			}
			scope:target_country = {
				multiply = {
					desc = "INFAMY_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = 20000000
					add = 1
					max = 2.0
				}
			}
		}
	}
}
