# ============================================================================
# TREATY ARTICLE: Cede Colony (Solar System Colonization, slice 1)
# ============================================================================
# Design: docs/superpowers/specs/2026-10-25-solar-colonization-holders-and-cession-design.md § 11.
# The source gives the target one of its colonies, named by its world's site
# building type (building_sr_site_<key>, never built:
# common/buildings/sr_site_buildings_generated.txt). The colony moves at entry
# into force and for good (sr_cede_colony_apply, generated): no lease, nothing
# on break or withdrawal, as vanilla's state_transfer. Its people, modules,
# specialization and discoveries go with the world. A treaty can carry it once
# per side per colony, so it cedes several, or swaps them. Enforceable, so a
# war's winner can impose it; giftable. Simple and full colonies only
# (sr_colonies_contested_enabled).
# ============================================================================
te_cede_colony = {
	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/te_cede_colony.dds"
	kind = directed
	cost = 0

	flags = {
		can_be_enforced
		giftable
	}

	# As state_transfer: nothing to maintain once the colony has moved.
	maintenance_paid_by = source_country
	usage_limit = once_per_side_with_same_inputs

	required_inputs = {
		building_type
	}

	# root is the prospective source.
	visible = {
		sr_colonies_contested_enabled = yes
		sr_has_space_colony = yes
	}

	possible = {
		sr_has_space_colony = yes
	}

	# root the source, scope:input the candidate: only a world the source holds.
	building_type_valid_trigger = {
		sr_cede_site_input_valid = yes
	}

	# The designer re-walks this every frame with scope:article_options
	# unbound: the call waits for both parties.
	on_entry_into_force = {
		custom_tooltip_no_bullet = te_cede_colony_effects_desc
		scope:article_options = {
			source_country = { save_scope_as = sr_cede_source }
			target_country = { save_scope_as = sr_cede_target }
		}
		if = {
			limit = {
				exists = scope:sr_cede_source
				exists = scope:sr_cede_target
			}
			sr_cede_colony_apply = yes
		}
	}

	ai = {
		treaty_categories = { expansion }
		article_ai_usage = { offer request }

		# Root only (treaty_evaluation_scope_audit). A space power in the race
		# may ask for a colony another holds; a holder that is no space power
		# (one that inherited or was given a colony) may sell one. Acceptance has
		# no such test: any AI weighs a cession by its score below.
		evaluation_chance = {
			value = 0
			if = {
				limit = {
					sr_space_race_participant = yes
					sr_ai_should_participate = yes
					any_country = {
						NOT = { this = root }
						sr_has_space_colony = yes
					}
				}
				add = 0.02
			}
			if = {
				limit = {
					sr_has_space_colony = yes
					NOT = { sr_space_race_participant = yes }
				}
				add = 0.02
			}
		}

		# The giver loses the world's worth and is reluctant besides; the
		# receiver gains its worth, half if it runs no space programme. A Money
		# Transfer the other way is what makes a sale acceptable.
		inherent_accept_score = {
			value = 0
			if = {
				limit = { scope:source_country ?= this }
				add = {
					desc = "te_cede_colony_ai_give"
					value = -10
				}
				subtract = {
					desc = "te_cede_colony_ai_worth_lost"
					value = sr_cede_ai_input_value
				}
			}
			else_if = {
				limit = {
					scope:target_country ?= this
					sr_space_race_participant = yes
				}
				add = {
					desc = "te_cede_colony_ai_worth_gained"
					value = sr_cede_ai_input_value
				}
			}
			else_if = {
				limit = { scope:target_country ?= this }
				add = {
					desc = "te_cede_colony_ai_worth_gained_idle"
					value = sr_cede_ai_input_value
					divide = 2
				}
			}
		}

		proposal_weight = {
			value = 0
			if = {
				limit = {
					scope:source_country ?= { sr_has_space_colony = yes }
				}
				value = 1
			}
		}

		# Pressed in a play (ROOT the holder of the goal): a space power, against
		# a country that holds a colony.
		wargoal_score_multiplier = {
			value = 0
			if = {
				limit = {
					sr_space_race_participant = yes
					scope:source_country ?= { sr_has_space_colony = yes }
				}
				value = 1
			}
		}
	}

	# Pressed in a play: Money Transfer's contestion, as the seabed lease, and
	# flat maneuvers and infamy (design § 16: to tune in play). Against a
	# country that interdicted our colonies in peace within five years, it costs
	# no infamy (fleets design § 12, the casus belli an interdiction gives): ROOT
	# the war goal's holder and scope:target_country the country it is against,
	# as vanilla's Money Transfer reads them (INFAMY_TARGET_COUNTRY_DEDUCTION_FACTOR).
	wargoal = {
		execution_priority = 21

		contestion_type = control_any_target_incorporated_state

		maneuvers = {
			value = 0
			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 5
			}
		}

		infamy = {
			value = 0
			add = {
				desc = "INFAMY_BASE_VALUE"
				value = 10
			}
			if = {
				limit = {
					scope:target_country ?= { sr_interdicted_recently = { TARGET = root } }
				}
				multiply = {
					desc = "INFAMY_SR_INTERDICTOR"
					value = 0
				}
			}
		}
	}
}
