# =================================================================
# TREATY ARTICLE: currency_peg                  (monetary policy 5a)
# Spec: docs/systems/monetary_policy_design.md §15A.2, §0.8.
#
# DIRECTED: source = the PEGGER, target = the ANCHOR.
#
# The pegger becomes ANCHORED, kind 1 (§15A.1): no dial, rate = the anchor's
# + 0.5, the anchor's credibility imported, structural premium -0.5pp — and a
# te_peg_confidence that overvaluation against the anchor drains, ending in
# te_peg.2. The anchor becomes a "reserve currency": -0.1pp of structural
# premium per 5% of world GDP pegged to it, capped at -0.5, for a little
# influence. It takes on no obligation — a bare peg is unilateral in
# substance, which is why it is cheap to grant.
#
# NOTHING MONETARY LIVES IN source_modifier / target_modifier. Every premium
# term is GDP- or share-scaled and goes through the three managed
# te_mon_arrangement_* modifiers, re-applied from the monthly update; whether a
# script_only modifier type even works in a treaty block is an open engine
# check (§17 check 14). The blocks carry flat flavour only.
#
# THE HOOKS DO ONE THING: dispatch the idempotent discovery
# (te_monetary_internal.2) onto both parties, a day later so the treaty's own
# state has settled. ROOT in these hooks is the ARTICLE. The monthly update is
# the source of truth either way — a flagged country re-runs the discovery in
# step 1c — so a hook that misfires costs a month, not a bug.
# =================================================================
currency_peg = {
	icon = "gfx/interface/icons/law_icons/gold_standard.dds"
	kind = directed
	cost = 25

	relations_progress_per_day = 0.5
	relations_improvement_max = 40

	flags = {
		can_be_renegotiated
		giftable
		friendly
	}

	maintenance_paid_by = target_country
	usage_limit = once_per_treaty

	# [6c] ...and the imposed peg (113): one anchored state, whichever article
	# put the country in it. Declared in both files.
	mutual_exclusions = {
		currency_peg
		imposed_currency_peg
	}

	# TECH GATE: INTERNATIONAL EXCHANGE STANDARDS (era 4), not central_banking.
	# A declared peg between two national currencies is the gold-standard era's
	# arrangement: the classical gold standard spread as a set of unilateral
	# convertibility promises, and the *negotiated* peg between currencies —
	# the Latin Monetary Union's successors, the gold-exchange standards of the
	# 1890s — is the later, treaty-borne form of it. Owner decision 2026-09-21;
	# the three articles are staggered (peg here, swap line at macroeconomics,
	# guarantee at intergovernmental_organizations) so the phase-5 toolkit
	# arrives over three eras rather than all at once.
	#
	# REVISED 2026-09-21 (same day): the first draft of this comment justified
	# the gate as "the tech that brings the Gold Standard law in with it".
	# law_gold_standard has since moved to central_banking (era 2 — see
	# common/laws/extra_laws.txt), so that reason is gone; the staggering above
	# is the standing one, and it does not depend on where gold sits. Coherence
	# needs only peg gate >= gold gate, which era 4 still satisfies: a country
	# reaches convertibility first and can be held to it by treaty later.
	unlocked_by_technologies = {
		international_exchange_standards
	}

	# Flat flavour: being the money others hold is prestige.
	#
	# PHASE 6 (§15C.1, ruling H5): this line STAYS AS SHIPPED. The plan proposed
	# folding it into one managed, GDP-scaled term so an anchor's prestige could
	# not scale with pegger headcount; the owner's ruling was to leave it and to
	# read the stacking in-game instead (§17 check 24). The imposed peg (113)
	# carries no prestige line at all: a coerced peg is not a vote of confidence.
	target_modifier = {
		country_prestige_mult = 0.02
	}

	# [6b] A pegger lives in the anchor's monetary orbit (§15C.2, ruling L1),
	# sized against development_assistance's "the client's growing dependence".
	#
	# PLACEMENT (fixed 2026-09-21): country_treaty_leverage_generation_add is
	# generated AGAINST the country whose block carries it, by the dominant
	# counterparty — it is not "leverage this country generates". Vanilla is
	# unambiguous: guarantee_independence puts it on the *guaranteed* party
	# (target), foreign_investment_rights / trade_privilege / host_power_bloc_embassy
	# on the party that *grants* the concession (source). The mod's own
	# development_assistance already had it right, on the client's target_modifier.
	# It shipped here on the ANCHOR's block, which made the PEGGER gain leverage
	# over the anchor — exactly backwards. The pegger is the source, so it goes
	# in source_modifier. Same correction on 111/112 (mirrored the other way:
	# there the weak party is the target) and on 113/114.
	source_modifier = {
		country_treaty_leverage_generation_add = 200
	}

	# The whole phase-5 toolkit is part of the monetary layer, so the
	# simplified setting of banking_system_rule hides it: there is no anchored
	# state to enter when nobody has a dial (te_mon_full_system,
	# common/scripted_triggers/te_monetary_triggers.txt).
	visible = {
		te_mon_full_system = yes
	}

	possible = {
		has_diplomatic_relevance = scope:other_country
		NOT = { has_war_with = scope:other_country }
		NOT = { is_country_type = decentralized }
		scope:other_country = {
			NOT = { is_country_type = decentralized }
		}
	}

	# The anchor must hold a dial of its own. This is also what breaks the
	# article when the anchor loses it — a law change, a command economy, being
	# anchored itself by a higher kind (§15A.1 "Validity").
	requirement_to_maintain = {
		trigger = {
			custom_tooltip = {
				text = te_currency_peg_anchor_needs_dial_tt
				scope:target_country = { te_mon_can_be_anchor = yes }
			}
		}
	}

	# A command economy's rate is a plan figure; it has nothing to peg.
	requirement_to_maintain = {
		trigger = {
			custom_tooltip = {
				text = te_currency_peg_pegger_not_command_tt
				scope:source_country = { te_mon_is_command = no }
			}
		}
	}

	# SAME-DRAFT AND EXISTING-TREATY CHECKS GO HERE, never in `possible`:
	# scope:treaty is unpopulated there (scripting_best_practices.md).
	can_ratify = {
		custom_tooltip = {
			text = te_currency_peg_target_rank_tt
			scope:source_country = {
				te_mon_can_peg_to = { OTHER = scope:target_country }
			}
		}
		# One peg at a time, with anybody — [6c] imposed ones included.
		custom_tooltip = {
			text = te_currency_peg_already_pegged_tt
			scope:source_country = {
				NOT = {
					any_scope_treaty = {
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							OR = {
								has_type = currency_peg
								has_type = imposed_currency_peg
							}
							source_country = scope:source_country
						}
					}
				}
			}
		}
	}

	# [6a] §15C.1 ruling H7: WITHDRAW, not freeze. A peg between belligerents
	# should end, not pause — and §17 check 15 (does a frozen treaty still
	# iterate under any_scope_treaty?) is unanswered, so freezing would leave the
	# discovery anchoring a country to an enemy. Withdrawal runs the ordinary
	# hooks, so the anchored state lapses within a month.
	non_fulfillment = {
		consequences = withdraw
		conditions = {
			weekly = {
				scope:article = {
					OR = {
						root = { has_war_with = scope:other_country }
						root = { has_diplomats_expelled = scope:other_country }
					}
				}
			}
		}
	}

	on_entry_into_force = {
		scope:article_options = {
			source_country = { trigger_event = { id = te_monetary_internal.2 days = 1 } }
			target_country = { trigger_event = { id = te_monetary_internal.2 days = 1 } }
		}
	}

	on_withdrawal = {
		scope:withdrawing_country ?= { trigger_event = { id = te_monetary_internal.2 days = 1 } }
		scope:non_withdrawing_country ?= { trigger_event = { id = te_monetary_internal.2 days = 1 } }
	}

	ai = {
		treaty_categories = { economy }
		article_ai_usage = { offer request }

		# Wide gate, continuous tilt (§15A.2 "AI").
		evaluation_chance = {
			value = 0.02
			if = {
				limit = {
					exists = scope:other_country
					scope:other_country = { country_rank >= rank_value:major_power }
					country_rank < rank_value:major_power
				}
				add = 0.06
			}
			if = {
				limit = {
					has_variable = te_inflation
					var:te_inflation >= 10
				}
				add = 0.1
			}
		}

		inherent_accept_score = {
			value = 0

			# --- the PEGGER'S side ---------------------------------------------
			if = {
				limit = { scope:source_country ?= this }
				add = {
					desc = "te_ai_peg_base"
					value = -15
				}
				# Economic dependence: it already lives in the anchor's market.
				if = {
					limit = {
						exists = market
						market.owner = scope:target_country
						NOT = { this = scope:target_country }
					}
					add = {
						desc = "te_ai_peg_shared_market"
						value = 30
					}
				}
				if = {
					limit = { is_in_same_power_bloc = scope:target_country }
					add = {
						desc = "AI_SAME_POWER_BLOC"
						value = 10
					}
				}
				# Low own credibility: there is more to import.
				if = {
					limit = { te_mon_credibility_c_own < 0.4 }
					add = {
						desc = "te_ai_peg_low_credibility"
						value = 15
					}
				}
				# High own inflation: the peg is a disinflation.
				if = {
					limit = {
						has_variable = te_inflation
						var:te_inflation >= 6
					}
					add = {
						desc = "te_ai_peg_high_inflation"
						value = {
							value = var:te_inflation
							subtract = 6
							multiply = 2
							max = 30
						}
					}
				}
				# A diverging cycle: the imported rate would be the wrong one.
				if = {
					limit = {
						has_variable = te_mon_overvaluation
						var:te_mon_overvaluation > 5
					}
					add = {
						desc = "te_ai_peg_diverging_cycle"
						value = {
							value = var:te_mon_overvaluation
							multiply = -2
							min = -40
						}
					}
				}
				# A great power does not borrow another's money.
				if = {
					limit = { country_rank >= rank_value:great_power }
					add = {
						desc = "te_ai_peg_great_power_pride"
						value = -40
					}
				}
			}
			# --- the ANCHOR'S side: cheap to grant ------------------------------
			else = {
				add = {
					desc = "te_ai_peg_reserve_currency"
					value = 10
				}
			}

			if = {
				limit = { has_diplomatic_pact = { who = scope:other_country type = rivalry } }
				add = {
					desc = "te_ai_arrangement_rivalry"
					value = -1000
				}
			}
			if = {
				limit = {
					OR = {
						has_attitude = { who = scope:other_country attitude = antagonistic }
						has_attitude = { who = scope:other_country attitude = domineering }
						has_attitude = { who = scope:other_country attitude = belligerent }
					}
				}
				add = {
					desc = "AI_HOSTILE_ATTITUDE"
					value = -50
				}
			}
		}
	}
}
