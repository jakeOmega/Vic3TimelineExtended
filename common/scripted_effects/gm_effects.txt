# ============================================================================
# GRAND MONUMENTS — scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-09-27-grand-monument-rework-design.md.
# The state-held records are listed in monument_triggers.txt § "kind, fit and
# status".
#
# Where modifiers are applied: state modifiers only from gm_state_monthly
# (on_monthly_pulse_state, ROOT = the state); national ones only from
# gm_country_refresh with ROOT = the country. A variable backing a multiplier
# is zeroed, never removed.

# ---- Records (§3) ------------------------------------------------------------
# State scope. A dedicated monument the pulse finds without records (a pick
# made in the building panel before sites, or a monument from a save older
# than this system) gets them from its current owner, once. The ceremony
# writes them itself, so this does nothing for a monument it dedicated; a
# finished building is created on its dedication and locked to it (sites spec
# 2026-10-18 §3.1), so a panel pick can no longer reach one.
gm_state_first_sight = {
	if = {
		limit = {
			gm_state_is_dedicated = yes
			NOT = { has_variable = gm_seen }
			# A ceremony asking: the dedication is its to set, and a copy it
			# will replace still reads as dedicated.
			NOT = { has_variable = gm_ceremony_pending }
		}
		set_variable = gm_seen
		# Who raised it, whatever it honours: trophies (gm_trophy_triggers.txt)
		# let a country carry off only what it did not build.
		gm_state_record_builder = yes
		if = {
			limit = { gm_state_is_bound = yes }
			set_variable = { name = gm_raised_by value = owner }
		}
		# Gate the record, not gm_state_leader_fits: adding the gate there would
		# wrongly contest a Leader monument when the same ruler moves from
		# Autocracy to another law. With no honoree recorded, the monument
		# reads "does not fit" and is contested on the same refresh (spec §1) —
		# a panel-picked Leader under a parliamentary republic or a crowned
		# monarchy never earns the commission's regime legitimacy, authority
		# and IG approval it should not have.
		if = {
			limit = {
				gm_state_kind_ruler = yes
				owner = { gm_can_raise_leader_monument = yes }
			}
			gm_state_record_honoree = yes
		}
		if = {
			limit = { gm_state_kind_faith = yes }
			gm_state_record_faith = yes
		}
		if = {
			limit = {
				NOT = { has_variable = gm_skin }
			}
			gm_state_set_default_skin = yes
		}
	}
}

# State scope. The ruler a Leader monument honours, and that ruler's IG as a
# flag (gm_honoree_ig): its supporters, whom a later Tear Down angers (§4.3).
# Recorded now because by the time the monument is contested the ruler has
# changed.
gm_state_record_honoree = {
	if = {
		limit = {
			owner = { exists = ruler }
		}
		# Saved scope, then the variable: vanilla stores a ruler this way
		# (`ruler = { save_scope_as = … }`); cleared first so a stale save
		# from an earlier call cannot be read.
		clear_saved_scope = gm_tmp_ruler
		owner = {
			ruler = { save_scope_as = gm_tmp_ruler }
		}
		set_variable = { name = gm_honoree value = scope:gm_tmp_ruler }
		owner = {
			set_variable = { name = gm_tmp_ig_flag value = flag:none }
			ruler = {
				interest_group ?= {
					gm_ig_to_flag_on_owner = { VAR = gm_tmp_ig_flag }
				}
			}
		}
		set_variable = { name = gm_honoree_ig value = owner.var:gm_tmp_ig_flag }
		owner = { remove_variable = gm_tmp_ig_flag }
	}
}

# Interest-group scope. Writes this IG's type as a flag to variable $VAR$ on
# its owner. A flag, not the IG scope: it names the same group in any country.
gm_ig_to_flag_on_owner = {
	if = {
		limit = { is_interest_group_type = ig_armed_forces }
		owner = { set_variable = { name = $VAR$ value = flag:armed_forces } }
	}
	if = {
		limit = { is_interest_group_type = ig_devout }
		owner = { set_variable = { name = $VAR$ value = flag:devout } }
	}
	if = {
		limit = { is_interest_group_type = ig_industrialists }
		owner = { set_variable = { name = $VAR$ value = flag:industrialists } }
	}
	if = {
		limit = { is_interest_group_type = ig_intelligentsia }
		owner = { set_variable = { name = $VAR$ value = flag:intelligentsia } }
	}
	if = {
		limit = { is_interest_group_type = ig_landowners }
		owner = { set_variable = { name = $VAR$ value = flag:landowners } }
	}
	if = {
		limit = { is_interest_group_type = ig_petty_bourgeoisie }
		owner = { set_variable = { name = $VAR$ value = flag:petty_bourgeoisie } }
	}
	if = {
		limit = { is_interest_group_type = ig_rural_folk }
		owner = { set_variable = { name = $VAR$ value = flag:rural_folk } }
	}
	if = {
		limit = { is_interest_group_type = ig_trade_unions }
		owner = { set_variable = { name = $VAR$ value = flag:trade_unions } }
	}
}

# State scope. The country that first saw it dedicated: its own, for trophies
# (gm_state_built_by_owner). Kept through a gift or a piece taken; cleared
# with the monument (gm_clear_state).
gm_state_record_builder = {
	if = {
		limit = {
			NOT = { has_variable = gm_built_by }
		}
		set_variable = { name = gm_built_by value = owner }
	}
}

# State scope. The faith a Grand Shrine honours.
gm_state_record_faith = {
	clear_saved_scope = gm_tmp_faith_scope
	owner = {
		religion = { save_scope_as = gm_tmp_faith_scope }
	}
	set_variable = { name = gm_faith value = scope:gm_tmp_faith_scope }
}

# ---- Skins (§1.1) ------------------------------------------------------------
# State scope, with the dedication already active. The most specific skin that
# fits, without asking: the state religion's for a Shrine, the first heritage in
# the revival partition's order for a heritage-skinned dedication, else
# generic. Used for old saves, panel picks and as the ceremony follow-up's
# default (monument_events.11). Skins read only the owner's own religion and
# primary cultures.
gm_state_set_default_skin = {
	set_variable = { name = gm_skin value = flag:generic }
	if = {
		limit = { gm_state_kind_faith = yes }
		gm_state_default_faith_skin = yes
		# The mod's own religions have no faith skin: a heritage skin instead.
		if = {
			limit = {
				has_variable = gm_skin
				var:gm_skin = flag:generic
			}
			gm_state_default_heritage_skin = yes
		}
	}
	else_if = {
		limit = { gm_state_takes_heritage_skin = yes }
		gm_state_default_heritage_skin = yes
	}
}

# State scope. Add $AGE$ to the primary's age grandeur (gm_age), creating it.
gm_state_add_age = {
	if = {
		limit = { has_variable = gm_age }
		change_variable = { name = gm_age add = $AGE$ }
	}
	else = {
		set_variable = { name = gm_age value = $AGE$ }
	}
}

# Any scope: a historical monument's records (a landmark, a trophy abroad in
# 1836, a dig). Its $LEVELS$ at the 1836 rate, the grandeur it has today, split
# as if it had been built in $BUILT$ (a year, negative before Christ) and aged
# since at 2% of its starting grandeur a year for at most a century
# (gm_age_cap_years; owner, 2026-10-19): starting grandeur
# $START$ = G / (1 + 0.02 x min(years, 100)) and age $AGE$ = the rest. So the
# Parthenon is a third starting grandeur and two thirds years and ages no
# further, and a monument finished this year is all starting grandeur.
gm_set_historical_grandeur = {
	set_variable = {
		name = $START$
		value = {
			value = $LEVELS$
			multiply = gm_level_grandeur_1836
			divide = {
				value = gm_this_year
				subtract = $BUILT$
				max = gm_age_cap_years
				multiply = 2
				divide = 100
				add = 1
				min = 1
			}
		}
	}
	set_variable = {
		name = $AGE$
		value = {
			value = $LEVELS$
			multiply = gm_level_grandeur_1836
			subtract = var:$START$
			min = 0
		}
	}
}

gm_remove_state_var = {
	if = {
		limit = { has_variable = $VAR$ }
		remove_variable = $VAR$
	}
}

# State scope. Forget a monument that is gone (demolished, torn down, or about
# to be rebuilt). gm_local_steps backs the local modifiers' multiplier: zeroed,
# never removed. KEEP_LANDMARK = yes (Rededicate only) keeps a landmark's look:
# its key, skin, form and picture (phase 4 §3.2). Every other caller passes no:
# Pull Down and demolition end a landmark.
gm_clear_state = {
	gm_remove_local_modifiers = yes
	gm_remove_state_var = { VAR = gm_seen }
	gm_remove_state_var = { VAR = gm_raised_by }
	gm_remove_state_var = { VAR = gm_honoree }
	gm_remove_state_var = { VAR = gm_honoree_ig }
	gm_remove_state_var = { VAR = gm_faith }
	if = {
		limit = {
			NOT = {
				AND = {
					always = $KEEP_LANDMARK$
					gm_state_is_landmark = yes
				}
			}
		}
		gm_remove_state_var = { VAR = gm_landmark }
		gm_remove_state_var = { VAR = gm_skin }
		gm_remove_state_var = { VAR = gm_form }
		gm_remove_state_var = { VAR = gm_picture_code }
		# Its unique dedication (signatures spec §4.4, §5.2): a landmark
		# rededicated keeps it, dormant until it stands on its parent again; any
		# other monument's goes, a special for good.
		gm_remove_state_var = { VAR = gm_signature }
		gm_remove_state_var = { VAR = gm_sig_parent }
	}
	gm_remove_state_var = { VAR = gm_skin_axis }
	gm_remove_state_var = { VAR = gm_contested }
	gm_remove_state_var = { VAR = gm_heritage }
	gm_remove_state_var = { VAR = gm_base_ig }
	gm_remove_state_var = { VAR = gm_supporter_ig }
	gm_remove_state_var = { VAR = gm_new_contest }
	gm_remove_state_var = { VAR = gm_grandeur }
	# Its age, its frozen levels, its starting grandeur and a newcomer's mark
	# (spec 2026-10-16 §1, §3; sites spec 2026-10-18 §3.2). Rededicate keeps
	# half the age (gm_state_rededicate).
	gm_remove_state_var = { VAR = gm_age }
	gm_remove_state_var = { VAR = gm_levels }
	gm_remove_state_var = { VAR = gm_start_grandeur }
	gm_remove_state_var = { VAR = gm_dedication }
	gm_remove_state_var = { VAR = gm_newcomer }
	gm_remove_state_var = { VAR = gm_ceremony_pending }
	gm_remove_state_var = { VAR = gm_ceremony_level }
	# Its name (v2 §3, gm_name_effects.txt): a rebuilt monument is named again.
	gm_state_clear_namesake = yes
	gm_remove_state_var = { VAR = gm_name }
	gm_remove_state_var = { VAR = gm_unveiled_for_com }
	gm_remove_state_var = { VAR = gm_unveil_pending }
	gm_remove_state_var = { VAR = gm_name_choice_open }
	gm_remove_state_var = { VAR = gm_being_named }
	# An opportunity's gift (gm_commission_effects.txt).
	gm_remove_state_var = { VAR = gm_gift }
	gm_remove_state_var = { VAR = gm_restore_pm }
	# Trophies (gm_trophy_effects.txt): its builder, a taking's marks.
	gm_remove_state_var = { VAR = gm_built_by }
	gm_remove_state_var = { VAR = gm_carried_off }
	gm_remove_state_var = { VAR = gm_tr_cooldown }
	gm_remove_state_var = { VAR = gm_tr_offered_to }
	set_variable = { name = gm_local_steps value = 0 }
	set_variable = { name = gm_local_tourism_steps value = 0 }
	gm_state_zero_local_steps = yes
	gm_state_sig_zero_local_steps = yes
	# Its shadow (issue #936, gm_monument_effects.txt): the pass writes the next
	# monument's once it is recorded.
	gm_state_drop_shadow = yes
}

# ---- The local pulse (§2.5): on_monthly_pulse_state, ROOT = the state --------
# Tourism Industry throughput for every monument; the dedication's own local
# effect for every dedicated one (fits, heritage or contested, §2.3). Both on
# the 1/n curve (gm_curve_steps_lin_f5) over the grandeur standing here.
gm_state_monthly = {
	# The construction site's new levels, into its records (sites spec
	# 2026-10-18 §3.2): real execution only, never the walked refresh.
	gm_state_count_site_levels = yes
	gm_remove_local_modifiers = yes
	# A gifted monument's dedication back (opportunities spec §3.1).
	gm_state_restore_dedication = yes
	gm_tr_occupation_read_self_check = yes
	if = {
		limit = { has_building = building_grand_monument }
		# A month older (spec 2026-10-16 §2), before this month's steps read it.
		if = {
			limit = { gm_state_ages = yes }
			gm_state_add_age = { AGE = gm_state_age_step }
		}
		# Last known level, read if the building disappears before the next pulse.
		set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
		# Its dedication, read if the building disappears (sites spec
		# 2026-10-18 §3.7: it is put back on it).
		gm_state_record_dedication = yes
		gm_monument_migration_self_check = yes
		gm_monument_merge_level_self_check = yes
		# Tourism counts everything standing here; the monument policy (v2 §5)
		# scales it and each local effect.
		set_variable = { name = gm_curve_in value = gm_state_grandeur }
		set_variable = {
			name = gm_local_tourism_steps
			value = {
				value = gm_curve_steps_lin_f5
				multiply = gm_policy_factor_tourism
			}
		}
		add_modifier = { name = gm_local_tourism multiplier = var:gm_local_tourism_steps }
		# Upkeep on the building itself (sites spec 2026-10-18 §3.4): the levels
		# its monuments record, at pm_maintenance's 0.1 a level, through one
		# modifier whose multiplier reads ROOT, this state, by name
		# (scripting_best_practices.md). Its variable is overwritten, never
		# removed.
		set_variable = { name = gm_upkeep_mult value = gm_state_upkeep_mult }
		b:building_grand_monument ?= { add_modifier = { name = gm_upkeep multiplier = root.var:gm_upkeep_mult } }
		# Each dedication's own effect counts the grandeur of that dedication
		# standing here, the primary's and those beside it (spec 2026-10-16
		# §4.2); gm_local_steps keeps the primary's for the card.
		set_variable = { name = gm_local_steps value = 0 }
		gm_state_local_steps_one = { KEY = crown CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = republic CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = revolution CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = leader CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = religious CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = civic CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = war_memorial CAP = gm_local_cap_war_memorial }
		gm_state_local_steps_one = { KEY = artistic CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = naturalist CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = scientific CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = industrial CAP = gm_curve_lin_steps }
		gm_state_local_steps_one = { KEY = athletic CAP = gm_local_cap_athletic }
		# And each unique dedication's (signatures spec §2.2), counted the same
		# way, beside its parent's.
		gm_state_sig_local_steps_one = { SIG = antiquity }
		gm_state_sig_local_steps_one = { SIG = archive }
		gm_state_sig_local_steps_one = { SIG = first_emperor }
		# Trophies (gm_trophy_effects.txt): the owner stamped, an occupier's
		# offer or roll. Last: a taking removes the building.
		gm_state_tr_monthly = yes
	}
	else = {
		if = {
			# A landmark too, recorded or not: a rededicated one left undedicated
			# has no gm_seen, and its lock must not pass to the next monument here.
			limit = {
				OR = {
					has_variable = gm_seen
					gm_state_is_landmark = yes
				}
			}
			# The building went while its records stand (sites spec 2026-10-18
			# §3.7: war damage and the like never shrink a monument): put back
			# at level 1 on its dedication, from the hidden event (this pulse
			# reads nothing back).
			if = {
				limit = { gm_state_lost_building = yes }
				trigger_event = { id = monument_events.29 }
			}
			else = {
				# Gone without the Tear Down button: a contested monument counts
				# as torn down (plan decision 6). The building is not downsizeable,
				# so the building panel can no longer demolish one; this stays as
				# the net for a save that did, or any other removal.
				# Not one carried off as a trophy (gm_carried_off).
				if = {
					limit = {
						gm_state_is_contested = yes
						NOT = { has_variable = gm_carried_off }
					}
					gm_state_record_teardown = yes
				}
				gm_clear_state = { KEEP_LANDMARK = no }
			}
		}
		if = {
			limit = { has_variable = gm_local_steps }
			set_variable = { name = gm_local_steps value = 0 }
		}
		if = {
			limit = { has_variable = gm_local_tourism_steps }
			set_variable = { name = gm_local_tourism_steps value = 0 }
		}
		gm_state_zero_local_steps = yes
		gm_state_sig_zero_local_steps = yes
	}
}

# State scope, from gm_state_monthly. Dedication $KEY$'s local effect here: the
# 1/n curve over the grandeur of every monument of that dedication standing
# here, a dedicated primary's (fits, heritage or contested alike, §2.3) and each
# secondary's, at most $CAP$ steps (gm_curve_lin_steps where the curve's own end
# is the only one). Its multiplier variable is zeroed, never removed.
gm_state_local_steps_one = {
	set_variable = {
		name = gm_curve_in
		value = {
			value = 0
			if = {
				limit = {
					gm_state_is_dedicated = yes
					gm_state_has_pm = { PM = pm_monument_$KEY$ }
				}
				add = gm_state_primary_grandeur
			}
			if = {
				limit = { has_variable_list = gm_monuments_here }
				every_in_list = {
					variable = gm_monuments_here

# ---- The national refresh (§2.4) ---------------------------------------------
# Country scope, ROOT = the country: the monthly pulse, monument_events.20,
# a scripted GUI effect or a country event option. Never call it where ROOT is
# a state, building, law or character: the multipliers resolve against ROOT.
gm_country_refresh = {
	gm_init_ledgers = yes
	if = {
		limit = {
			OR = {
				any_scope_state = { has_building = building_grand_monument }
				has_variable = gm_com_open
				gm_tr_country_has_any = yes
				gm_arch_country_has_any = yes
			}
		}
		set_variable = gm_active
	}
	# JE Auto-Activation Requires BOTH (scripting_best_practices.md): the JE's
	# is_shown_when_inactive auto-activates it for new saves, but a new
	# auto-activating JE does not reach a save started before it existed
	# (same doc, following paragraph). Add it explicitly here so saves that
	# predate this entry, and any save where it lapsed, pick it back up.
	if = {
		limit = {
			gm_system_enabled = yes
			NOT = { has_journal_entry = je_grand_monuments }
			OR = {
				any_scope_state = { has_building = building_grand_monument }
				any_scope_state = { has_building = building_grand_monument_site }
				has_variable = gm_com_open
				has_variable = gm_com_offered
				gm_tr_country_has_any = yes
				gm_arch_country_has_any = yes
			}
		}
		add_journal_entry = { type = je_grand_monuments }
	}
	clear_variable_list = gm_states
	every_scope_state = {
		limit = { has_building = building_grand_monument }
		# An older save's monument, once (sites spec 2026-10-18 §3.5): the
		# hidden event, never here.
		if = {
			limit = { gm_state_needs_migration = yes }
			trigger_event = { id = monument_events.32 }
		}
		# A gifted monument's dedication back first (opportunities spec §3.1).
		gm_state_restore_dedication = yes
		# A landmark's unique dedication (signatures spec §4.4).
		gm_state_sig_backfill = yes
		set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
		# A part of this region merged into this state, and the records of a
		# monument that stood there wait to be adopted here (issue #936): the
		# hidden event adopts them first, and nothing reads this building as a
		# copy, a first sight or new levels until it has.
		if = {
			limit = { gm_state_merge_pending = yes }
			trigger_event = { id = monument_events.29 }
		}
		# A revolt's part of a monument's state: the engine's copy of its
		# building, which nothing records and no orphan claims. The monument
		# stays whole with the part that keeps its records (owner, 2026-10-18;
		# sites spec §3.7): the hidden event removes it.
		else_if = {
			limit = { gm_state_stray_building = yes }
			trigger_event = { id = monument_events.29 }
		}
		else = {
			gm_state_first_sight = yes
			# A dedication's unveiling, deferred to here (gm_state_offer_skins);
			# then a monument older than names, or one nobody named, takes the
			# default name (v2 §3.3).
			if = {
				limit = { has_variable = gm_unveil_pending }
				remove_variable = gm_unveil_pending
				gm_state_unveil = yes
			}
			gm_state_name_if_unnamed = yes
			# Not while a gifted monument's method may still read as another's.
			if = {
				limit = {
					NOT = { has_variable = gm_restore_pm }
				}
				gm_state_update_picture = yes
			}
			# A ceremony a finished level's hook missed.
			gm_state_offer_missed_ceremony = yes
			# A monument dedicated beside a grander one gives it first place
			# back once settled (gm_state_settle_newcomer), from the hidden event.
			if = {
				limit = { gm_state_newcomer_settled = yes }
				trigger_event = { id = monument_events.29 }
			}
		}
		owner = {
			add_to_variable_list = { name = gm_states target = prev }
		}
	}
	# A construction site with no finished monument here (sites spec
	# 2026-10-18 §3.1): listed for Awaiting Dedication, and the AI's ceremony
	# once nothing more is queued.
	every_scope_state = {
		limit = {
			has_building = building_grand_monument_site
			NOT = { has_building = building_grand_monument }
		}
		gm_state_offer_missed_ceremony = yes
		owner = {
			add_to_variable_list = { name = gm_states target = prev }
		}
	}
	gm_set_opposition_ig = yes
	gm_check_contests = yes
	# An opportunity completed by this level, dedication or month: before the
	# totals, so its rewards count in this refresh.
	gm_com_check_progress = yes
	# Where Raise It would build, for the panel (usability §1).
	gm_com_update_build_where = yes
	gm_compute_totals = yes
	gm_apply_national_modifiers = yes
	gm_notify_new_contests = yes
}

# Country scope. The monthly pulse: decay, then refresh; gm_active keeps the
# pulse running while a ledger fades after the last monument has gone.
gm_country_monthly = {
	gm_init_ledgers = yes
	gm_decay_ledgers = yes
	# A month off every opportunity boon (gm_commission_effects.txt), and the
	# Homecoming (gm_trophy_effects.txt).
	gm_tick_boons = yes
	gm_tr_country_monthly = yes
	# The open opportunity's month: completed, lapsed or run out before the
	# refresh, which then counts its rewards.
	gm_com_check_progress = yes
	gm_com_monthly = yes
	gm_country_refresh = yes
	every_scope_state = {
		limit = { gm_state_is_contested = yes }
		change_variable = { name = gm_contested add = 1 }
	}
	if = {
		limit = { has_variable_list = gm_monuments_held }
		every_in_list = {
			variable = gm_monuments_held
			limit = { gm_mon_is_contested = yes }
			change_variable = { name = gm_tr_contested add = 1 }
		}
	}
	gm_ai_decide_contests = yes
	gm_policy_ai_choose = yes
	gm_name_sweep = yes
	if = {
		limit = {
			NOT = { any_scope_state = { has_building = building_grand_monument } }
			gm_ledgers_idle = yes
			NOT = { has_variable = gm_com_open }
			NOT = { has_variable = gm_com_offered }
			NOT = { has_variable = gm_naming_active }
		}
		# Ledgers must not freeze un-snapped once the pulse stops running.
		gm_zero_ledgers = yes
		if = {
			limit = { has_variable = gm_active }
			remove_variable = gm_active
		}
	}
	else = {
		set_variable = gm_active
	}
	if = {
		limit = { is_ai = no }
		gm_debug_log = yes
	}
}

# Country scope. Grandeur -> step count and next step, along the curve with

}

# Country scope. Grandeur -> step count and next step along the 1/n curve
# (gm_values.txt) with first step $F$, with no end: step n takes n * $F$, so the
# step being filled, $WIDTH$ wide, completes at $NEXT$. Prestige and cultural
# pull, which count every monument a country has (owner, 2026-10-19). The
# National Effects bar is that step's share filled, (G - N + W) / W.
gm_set_steps_lin_open = {
	set_variable = { name = $OUT$ value = 0 }
	set_variable = { name = $WIDTH$ value = $F$ }
	set_variable = { name = $NEXT$ value = $F$ }
	while = {
		limit = { var:$IN$ >= var:$NEXT$ }
		change_variable = { name = $OUT$ add = 1 }
		change_variable = { name = $WIDTH$ add = $F$ }
		change_variable = { name = $NEXT$ add = var:$WIDTH$ }
	}
	change_variable = {
		name = $OUT$
		add = {
			value = var:$IN$
			subtract = var:$NEXT$
			add = var:$WIDTH$
			divide = var:$WIDTH$
			min = 0
		}
	}
}

# Country scope. One IG's net grandeur (§2.4): its ledger, plus each Fits
# monument it approves of, minus each it opposes, minus each contested
# monument whose new order it is (the standing penalty, §4.2). The result goes
# through the curve on its absolute value, with the sign put back: one cap per
# IG, across all monuments.
gm_compute_ig = {
	set_variable = { name = gm_ig_net_$IG$ value = var:gm_ig_ledger_$IG$ }
	# Trophies held (gm_trophy_values.txt): the Armed Forces' only.
	change_variable = { name = gm_ig_net_$IG$ add = gm_tr_ig_share_$IG$ }
	every_scope_state = {
		limit = { has_building = building_grand_monument }
		if = {
			limit = {
				gm_state_status_fits = yes
				gm_state_approved_by_$IG$ = yes
			}
			owner = {
				change_variable = { name = gm_ig_net_$IG$ add = prev.var:gm_grandeur }
			}
		}
		# A unique dedication's own group (signatures spec §2.3), beside the
		# dedication's.
		if = {
			limit = {
				gm_state_status_fits = yes
				gm_state_sig_approved_by_$IG$ = yes
			}
			owner = {
				change_variable = { name = gm_ig_net_$IG$ add = prev.var:gm_grandeur }
			}
		}
		if = {
			limit = {
				gm_state_status_fits = yes
				gm_state_opposed_by_$IG$ = yes
			}
			owner = {
				change_variable = { name = gm_ig_net_$IG$ subtract = prev.var:gm_grandeur }
			}
		}
		if = {
			limit = {
				gm_state_is_contested = yes
				has_variable = gm_base_ig
				var:gm_base_ig = flag:$IG$
			}
			owner = {
				change_variable = { name = gm_ig_net_$IG$ subtract = prev.var:gm_grandeur }
			}
		}
	}
	# And each monument standing beside another (spec 2026-10-16 §4.2), by its
	# own dedication: its grandeur cached on it, then added on the country.
	if = {
		limit = { has_variable_list = gm_monuments_held }
		save_scope_as = gm_ig_country
		every_in_list = {
			variable = gm_monuments_held
			set_variable = { name = gm_tr_g value = gm_mon_grandeur }
			if = {
				limit = {
					gm_mon_status_fits = yes
					gm_mon_approved_by_$IG$ = yes
				}
				scope:gm_ig_country = {
					change_variable = { name = gm_ig_net_$IG$ add = prev.var:gm_tr_g }
				}
			}
			if = {
				limit = {
					gm_mon_status_fits = yes
					gm_mon_sig_approved_by_$IG$ = yes
				}
				scope:gm_ig_country = {
					change_variable = { name = gm_ig_net_$IG$ add = prev.var:gm_tr_g }
				}
			}
			if = {
				limit = {
					gm_mon_status_fits = yes
					gm_mon_opposed_by_$IG$ = yes
				}
				scope:gm_ig_country = {
					change_variable = { name = gm_ig_net_$IG$ subtract = prev.var:gm_tr_g }
				}
			}
			if = {
				limit = {
					gm_mon_is_contested = yes
					has_variable = gm_tr_base_ig
					var:gm_tr_base_ig = flag:$IG$
				}
				scope:gm_ig_country = {
					change_variable = { name = gm_ig_net_$IG$ subtract = prev.var:gm_tr_g }
				}
			}
		}
	}
	if = {
		limit = { var:gm_ig_net_$IG$ >= 0 }
		set_variable = { name = gm_curve_in value = var:gm_ig_net_$IG$ }
		set_variable = { name = gm_s_ig_$IG$ value = gm_curve_steps_f5 }
	}
	else = {
		set_variable = {
			name = gm_curve_in
			value = {
				value = var:gm_ig_net_$IG$
				multiply = -1
			}
		}
		set_variable = {
			name = gm_s_ig_$IG$
			value = {
				value = gm_curve_steps_f5
				multiply = -1
			}
		}
	}
}

# ---- Contests (§4) -------------------------------------------------------------
# Country scope. From gm_country_refresh: a bound monument whose message no
# longer fits becomes contested; a contested or heritage one whose message
# fits again is restored (a Bourbon restoration gets its statues back). Not
# while this country is a side in a civil war: the winner is judged after it
# (plan decision 2). A civil war's winner first adopts the loser's monuments.
gm_check_contests = {
	if = {
		limit = {
			NOT = { has_variable = te_cw_role }
		}
		every_scope_state = {
			limit = {
				gm_state_is_dedicated = yes
				gm_state_is_bound = yes
				# A copy the ceremony is replacing has no records yet, and
				# would read as not fitting.
				NOT = { has_variable = gm_ceremony_pending }
				# A gifted monument whose rebuilt method may still be the
				# engine's copy of another dedication (opportunities spec §3.1).
				NOT = { has_variable = gm_restore_pm }
				# Recorded: in the refresh's order, first sight or a ceremony
				# always comes first (gm_country_refresh's state loop runs before
				# this), except for a building a merge brought, which waits to be
				# adopted (issue #936, gm_state_merge_pending) and is not contested
				# meanwhile. Moving this check ahead of that loop would skip new
				# monuments for a month.
				has_variable = gm_seen
			}
			if = {
				limit = {
					owner = { has_variable = gm_cw_adopting }
				}
				gm_state_adopt_for_winner = yes
			}
			# A shrine this country did not raise always becomes heritage, whatever
			# the route it arrived by: outright conquest, a secession that drops
			# te_cw_role on the new country (te_civil_war_on_actions.txt), or a
			# civil war of this country's own that ends with a foreign shrine
			# still in hand. Runs before the lift/contest branches so it wins over
			# "contest" for a bound-but-foreign shrine the loop hasn't seen yet.
			if = {
				limit = {
					gm_state_kind_faith = yes
					gm_state_raised_by_owner = no
					gm_state_is_heritage = no
				}
				gm_state_lift_contest = yes
				set_variable = gm_heritage
			}
			else_if = {
				limit = {
					OR = {
						gm_state_is_contested = yes
						gm_state_is_heritage = yes
					}
					gm_state_message_fits = yes
				}
				gm_state_lift_contest = yes
			}
			else_if = {
				limit = {
					gm_state_is_contested = no
					gm_state_is_heritage = no
					gm_state_message_fits = no
				}
				gm_state_contest = yes
			}
		}
		# Each monument standing beside another, on its own (spec 2026-10-16
		# §4.3; gm_monument_effects.txt).
		if = {
			limit = { has_variable_list = gm_monuments_held }
			every_in_list = {
				variable = gm_monuments_held
				gm_mon_check_contest = yes
			}
		}
	}
}

# State scope. Contest this monument (§4.1).
gm_state_contest = {
	set_variable = { name = gm_contested value = 0 }
	set_variable = gm_new_contest
	gm_state_record_contest_igs = yes
}

# State scope. The IG that opposed its message (the new order's base) and the
# IG that approved it (its supporters), as flags, recorded now: the Leader's
# pair is dynamic and would otherwise follow the new government.
gm_state_record_contest_igs = {
	set_variable = { name = gm_base_ig value = flag:none }
	set_variable = { name = gm_supporter_ig value = flag:none }
	if = {
		limit = { gm_state_has_pm = { PM = pm_monument_crown } }
		set_variable = { name = gm_base_ig value = flag:intelligentsia }
		set_variable = { name = gm_supporter_ig value = flag:landowners }
	}
	else_if = {
		limit = { gm_state_has_pm = { PM = pm_monument_republic } }
		set_variable = { name = gm_base_ig value = flag:landowners }
		set_variable = { name = gm_supporter_ig value = flag:intelligentsia }
	}
	else_if = {
		limit = { gm_state_has_pm = { PM = pm_monument_revolution } }
		set_variable = { name = gm_base_ig value = flag:industrialists }
		set_variable = { name = gm_supporter_ig value = flag:trade_unions }
	}
	else_if = {
		limit = { gm_state_kind_faith = yes }
		set_variable = { name = gm_supporter_ig value = flag:devout }
	}
	else_if = {
		limit = { gm_state_kind_ruler = yes }
		if = {
			limit = {
				owner = { has_variable = gm_opposition_ig }
			}
			set_variable = { name = gm_base_ig value = owner.var:gm_opposition_ig }
		}
		if = {
			limit = { has_variable = gm_honoree_ig }
			set_variable = { name = gm_supporter_ig value = var:gm_honoree_ig }
		}
	}
}

# State scope. Lift a contest, or restore a heritage monument to full effect.
gm_state_lift_contest = {
	gm_remove_state_var = { VAR = gm_contested }
	gm_remove_state_var = { VAR = gm_heritage }
	gm_remove_state_var = { VAR = gm_new_contest }
	gm_remove_state_var = { VAR = gm_base_ig }
	gm_remove_state_var = { VAR = gm_supporter_ig }
}

# ---- A change of owner (§1, §4.1) ----------------------------------------------
# State scope, ROOT = the state (on_state_owner_change). A regime or ruler
# monument now owned by a country that did not raise it honours a foreign
# order: contested for the new owner. A shrine becomes heritage, never
# contested (§1). This does not rely on the records surviving the transfer.
# Skipped for a civil war's transfers (the new owner is a side: te_cw_role, or
# a revolutionary country); the nation that raised it taking it back fails
# `gm_state_raised_by_owner = no`, and the monthly check restores it.
gm_state_changed_hands = {
	# A part of this region merged into this state (issue #936): the records of
	# the monuments that stood there are adopted before anything here is judged.
	if = {
		limit = { gm_state_merge_pending = yes }
		gm_state_adopt_orphans = yes
	}
	# A commission's mark and a naming in progress belong to the old owner
	# (v2 §3.3, §3.5); its naming is closed by its own sweep.
	gm_remove_state_var = { VAR = gm_unveiled_for_com }
	gm_remove_state_var = { VAR = gm_being_named }
	# The country it came from, for a trophy taken from it (gm_trophy_effects.txt).
	gm_state_tr_changed_hands = yes
	# The monuments beside it change owner too: the owners' Our Monuments lists
	# (gm_monuments_held) now, not at the next monthly pass, so the panels agree
	# with the standing grandeur, which reads the state's own list at once.
	if = {
		limit = { gm_state_has_secondary = yes }
		every_in_list = {
			variable = gm_monuments_here
			gm_mon_changed_hands = yes
		}
		gm_monument_pass = yes
	}
	if = {
		limit = {
			owner = { has_variable = gm_cw_adopting }
		}
		gm_state_adopt_for_winner = yes
	}
	if = {
		limit = {
			gm_state_is_dedicated = yes
			gm_state_is_bound = yes
			owner = {
				NOT = { has_variable = te_cw_role }
				is_revolutionary = no
			}
			gm_state_raised_by_owner = no
			# Not a building still waiting to be adopted, nor a revolt's stray
			# copy, which the refresh this sends removes.
			gm_state_merge_pending = no
			gm_state_stray_building = no
		}
		set_variable = gm_seen
		if = {
			limit = { gm_state_kind_faith = yes }
			gm_state_lift_contest = yes
			set_variable = gm_heritage
		}
		else = {
			gm_remove_state_var = { VAR = gm_heritage }
			# Unconditional: a monument already contested under the old owner
			# still needs its contest re-recorded for the new one (fresh
			# gm_contested count, fresh IGs, a new gm_new_contest so the
			# conqueror gets the notice) -- gm_state_contest itself resets
			# gm_contested to 0.
			gm_state_contest = yes
		}
	}
	owner = { trigger_event = { id = monument_events.20 } }
}

# ---- Civil wars: the winner continues the nation (#460-#465) -------------------
# Country scope, ROOT = the winner (te_civil_war_on_won, between
# te_civil_war_resolve_sides and te_civil_war_clear). The monuments the loser
# raised become the winner's, now and, for 30 days, in states still changing
# hands (gm_state_changed_hands and gm_check_contests call
# gm_state_adopt_for_winner). The fit test then contests them only if the
# winner's laws differ.
gm_repair_after_civil_war = {
	if = {
		limit = { exists = scope:te_cw_loser }
		set_variable = { name = gm_cw_loser value = scope:te_cw_loser days = 30 }
		set_variable = { name = gm_cw_adopting value = yes days = 30 }
		every_scope_state = {
			limit = { has_building = building_grand_monument }
			gm_state_adopt_for_winner = yes
		}
		# The monuments beside another (spec 2026-10-16 §4.3), from the global
		# list: the winner inherits no lists until the pass writes them.
		if = {
			limit = { has_global_variable_list = gm_monuments }
			every_in_global_list = {
				variable = gm_monuments
				limit = {
					var:gm_tr_origin ?= { owner = root }
				}
				gm_mon_adopt_for_winner = yes
			}
		}
		gm_monument_pass = yes
	}
}

# State scope, owner = a civil war's winner.
gm_state_adopt_for_winner = {
	if = {
		limit = {
			has_variable = gm_raised_by
			owner = { has_variable = gm_cw_loser }
			owner.var:gm_cw_loser ?= { save_temporary_scope_as = gm_tmp_loser }
			var:gm_raised_by = scope:gm_tmp_loser
		}
		set_variable = { name = gm_raised_by value = owner }
	}
}

# ---- The three choices (§4.3) --------------------------------------------------
# State scope. Adds FACTOR x this monument's grandeur to the owner's ledger of

	gm_state_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = industrialists }
	gm_state_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = intelligentsia }
	gm_state_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = landowners }
	gm_state_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = petty_bourgeoisie }
	gm_state_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = rural_folk }
	gm_state_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = trade_unions }
}

gm_state_ledger_ig_one = {
	if = {
		limit = {
			has_variable = $WHO$
			var:$WHO$ = flag:$IG$
		}
		owner = {
			set_variable = {

				value = {

# State scope. What pulling this monument down does (§4.3): its grandeur to
# the teardown ledger (legitimacy for the new order, fading), positive to the
# new order's base, negative to its old supporters. Reads gm_grandeur, the
# level last recorded (current when called from gm_state_tear_down).
gm_state_record_teardown = {
	owner = {
		gm_init_ledgers = yes
		set_variable = {
			name = gm_teardown_ledger
			value = {
				value = var:gm_teardown_ledger
				add = prev.var:gm_grandeur
			}
		}
		# Keep the pulse running so what was just recorded gets decayed, even
		# when this is the last monument and the panel demolished it (a state
		# pulse's else branch, up to a month before the country pulse's
		# gm_ledgers_idle check would otherwise zero it straight back out).
		set_variable = gm_active
	}
	gm_state_ledger_ig = { WHO = gm_base_ig FACTOR = 1 }
	gm_state_ledger_ig = { WHO = gm_supporter_ig FACTOR = -1 }
}

gm_state_tear_down = {
	set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
	gm_state_record_teardown = yes
	owner = {
		remove_list_variable = { name = gm_states target = prev }
	}
	gm_clear_state = { KEEP_LANDMARK = no }
	# Another monument stood beside it: the building is the largest's now
	# (gm_state_promote_secondary, gm_monument_effects.txt). Otherwise it goes.
	if = {
		limit = { gm_state_has_secondary = yes }
		gm_state_promote_secondary = yes
	}
	else = {
		set_variable = { name = gm_rebuilding days = 1 }
		remove_building = building_grand_monument
	}
}

# Rededicate: asked first, the answer pays and halves the records
# (gm_state_apply_rededicate below; sites spec 2026-10-18 §5).
#
# The work runs only under a marker set here: a button's or an option's
# tooltip walks this body every frame it shows, testing each limit and writing
# nothing, so the reads of gm_rebuild_cost and gm_rebuild_level found them
# unset (~600 error lines a frame through the level switch's 200 cases while
# the Rededicate button was hovered, 2026-10-12). Every caller already shows
# gm_rededicate_tt in place of these effects (gm_state_choose_rededicate) or
# needs no tooltip (the AI), so the guard hides nothing
# (scripting_best_practices.md § "An Option's Tooltip Walks Its Effects").
gm_state_rededicate = {
	set_variable = gm_rededicate_click
	if = {
		limit = { has_variable = gm_rededicate_click }
		remove_variable = gm_rededicate_click
		# Asked first (sites spec 2026-10-18 §5, a call for the owner): the
		# ceremony's answer pays and halves (gm_state_apply_rededicate), and
		# "Keep it as it is" changes nothing.
		set_variable = gm_rededicate_asking
		gm_state_start_ceremony = yes
	}
}

# State scope, from gm_state_dedicate in Rededicate's answer (mode rededicate).
# It pays (500 a level, gm_state_rededicate_cost), its supporters lose half its
# grandeur's favour, and it keeps half its levels (rounded up), half its
# starting grandeur and half its years: no rebuild, the records halve. A
# landmark keeps its look and picture; the dedication, the unveiling and the
# naming that follow give it a new meaning and a new name (phase 4 §3.3).
gm_state_apply_rededicate = {
	# Only when it runs: the option's tooltip walks this body writing nothing,
	# and every read of a variable set above it would log an error a frame
	# (gm_state_rededicate's note).
	set_variable = gm_apply_click
	if = {
		limit = { has_variable = gm_apply_click }
		remove_variable = gm_apply_click
		set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
		set_variable = { name = gm_rededicate_cost value = gm_state_rededicate_cost }
		owner = {
			add_treasury = {
				value = prev.var:gm_rededicate_cost
				multiply = -1
			}
			gm_init_ledgers = yes
		}
		gm_state_ledger_ig = { WHO = gm_supporter_ig FACTOR = -0.5 }
		set_variable = {
			name = gm_rededicate_levels
			value = {
				value = gm_state_primary_levels
				divide = 2
				ceiling = yes
			}
		}
		set_variable = {
			name = gm_rededicate_start
			value = {
				value = gm_state_primary_start
				divide = 2
			}
		}
		set_variable = {
			name = gm_rededicate_age
			value = {
				value = gm_state_primary_age
				divide = 2
			}
		}
		gm_clear_state = { KEEP_LANDMARK = yes }
		set_variable = { name = gm_levels value = var:gm_rededicate_levels }
		set_variable = { name = gm_start_grandeur value = var:gm_rededicate_start }
		if = {
			limit = { var:gm_rededicate_age > 0 }
			set_variable = { name = gm_age value = var:gm_rededicate_age }
		}
		remove_variable = gm_rededicate_levels
		remove_variable = gm_rededicate_start
		remove_variable = gm_rededicate_age
		remove_variable = gm_rededicate_cost
	}
}

# State scope. The dedication ceremony, once: a rebuilt monument may also fire
# on_building_built, and two ceremonies queued together would both show.
# gm_ceremony_level records the grandeur it asked at (monument_events.2
# records it again as the event opens, when a rebuilt level is readable), so
# gm_state_offer_missed_ceremony asks once per new level. A new or rebuilt
# monument may stand on its siblings' dedication
# (gm_state_has_copied_dedication); nothing switches it here, the ceremony's
# answer does, once (gm_state_dedicate, gm_state_undedicate).
# What it asks about (sites spec 2026-10-18 §3.1, gm_ceremony_mode): Rededicate's
# question (gm_rededicate_asking, set by gm_state_rededicate) on the finished
# monument here, else the construction site's dedication; with neither, an
# older save's undedicated monument, until its migration makes it a site.
gm_state_start_ceremony = {
	if = {
		limit = {
			NOT = { has_variable = gm_ceremony_pending }
			# A gifted monument is dedicated: a method that reads as
			# Undedicated is the rebuild's, until it is put back. Rededicate
			# asks anyway: on a monument beside another the swap that raised
			# it set the restore, and its answer switches the method itself
			# (gm_state_dedicate drops the restore).
			OR = {
				has_variable = gm_rededicate_asking
				NOT = { has_variable = gm_restore_pm }
			}
		}
		if = {
			limit = { has_variable = gm_rededicate_asking }
			set_variable = { name = gm_ceremony_mode value = flag:rededicate }
			remove_variable = gm_rededicate_asking
			set_variable = { name = gm_ceremony_level value = gm_state_primary_levels }
		}
		else_if = {
			limit = { has_building = building_grand_monument_site }
			set_variable = { name = gm_ceremony_mode value = flag:site }
			set_variable = { name = gm_ceremony_level value = gm_state_site_level }
		}
		else = {
			gm_remove_state_var = { VAR = gm_ceremony_mode }
			set_variable = { name = gm_ceremony_level value = gm_state_primary_levels }
		}
		set_variable = { name = gm_ceremony_pending days = 30 }
		trigger_event = { id = monument_events.2 popup = yes }
	}
	gm_remove_state_var = { VAR = gm_rededicate_asking }
}

# ---- The engine's copy --------------------------------------------------------
# A new building starts on the production methods most of its owner's
# buildings of that type use, not on is_default (grand_monument_pms.txt). The
# lock binds the panel only: activate_production_method moves a monument past
# it, so the ceremony's answer replaces a copy directly.
#
# One switch per answer, and never to the method already active: two in one
# tick (a hook firing on the first) left a monument's dedication listed twice
# in the save (te_debug_monuments probe, 2026-10-08).

# State scope. Back to Undedicated, from a dedication: the ceremony's "leave it
# undedicated" for a copy.
gm_state_undedicate = {
	if = {
		limit = {
			NOT = { gm_state_has_pm = { PM = pm_monument_undedicated } }
		}
		activate_production_method = {
			building_type = building_grand_monument
			production_method = pm_monument_undedicated
		}
	}
	# The answer stands (gm_state_dedicate's note).
	gm_remove_state_var = { VAR = gm_restore_pm }
	# Draped at once: gm_state_update_picture would read back the PM just activated.
	# A landmark keeps its own picture (phase 4 §5.1).
	set_variable = { name = gm_picture_code value = 1 }
	gm_state_picture_landmarks = yes
}

# ---- The AI and the notice (§4.5, §4.6) ----------------------------------------
# Country scope, AI only: a monument contested two months or more is decided,
# by weight. Tear Down is likelier for a small monument or under a
# revolutionary government, Preserve for a tall one, Rededicate only with
# money in the treasury.
gm_ai_decide_contests = {
	if = {
		limit = { is_ai = yes }
		every_scope_state = {
			limit = {
				gm_state_is_contested = yes
				var:gm_contested >= 2
			}
			random_list = {
				10 = {
					modifier = {
						if = {
							limit = { var:gm_grandeur < 5 }
							add = 10
						}
						if = {
							limit = {
								owner = { gm_government_is_revolutionary = yes }
							}
							add = 10
						}
					}
					gm_state_tear_down = yes
				}
				10 = {
					modifier = {
						if = {
							limit = { var:gm_grandeur >= 10 }
							add = 10
						}
						# History mostly kept these (phase 4 §5.3): the
						# Soviets kept the Alexander Column.
						if = {
							limit = { gm_state_is_landmark = yes }
							multiply = 3
						}
					}
					gm_state_preserve = yes
				}
				5 = {
					modifier = {
						if = {
							limit = {
								owner = { scaled_gold_reserves < 0.5 }
							}
							multiply = 0
						}
					}
					gm_state_rededicate = yes
				}
			}
		}
		# The monuments beside another (spec 2026-10-16 §4.3): Pull Down or Keep
		# as Heritage. Rededicating one moves it into the state's first place,
		# which the AI leaves to its own monument.
		if = {
			limit = { has_variable_list = gm_monuments_held }
			every_in_list = {
				variable = gm_monuments_held
				limit = {
					gm_mon_is_contested = yes
					var:gm_tr_contested >= 2
					NOT = { has_variable = gm_mon_order }
				}
				set_variable = { name = gm_tr_g value = gm_mon_grandeur }
				random_list = {
					10 = {
						modifier = {
							if = {
								limit = { var:gm_tr_g < 5 }
								add = 10
							}
							if = {
								limit = {
									var:gm_tr_origin ?= {
										owner = { gm_government_is_revolutionary = yes }
									}
								}
								add = 10
							}
						}
						gm_mon_order_choice = { CHOICE = tear_down }
					}
					10 = {
						modifier = {
							if = {
								limit = { var:gm_tr_g >= 10 }
								add = 10
							}
							if = {
								limit = { has_variable = gm_tr_landmark }
								multiply = 3
							}
						}
						gm_mon_preserve = yes
					}
				}
			}
		}
	}
}

# Country scope. New contests since the last notice: a player gets one notice
# (monument_events.16, which clears the flags as it opens); the AI's flags are
# cleared (it decides monthly, above).
gm_notify_new_contests = {
	if = {
		limit = {
			OR = {
				any_scope_state = { has_variable = gm_new_contest }
				AND = {
					has_variable_list = gm_monuments_held
					any_in_list = {
						variable = gm_monuments_held
						has_variable = gm_tr_new_contest
					}
				}
			}
		}
		if = {
			limit = { is_ai = no }
			trigger_event = { id = monument_events.16 }
		}
		else = {
			every_scope_state = {
				limit = { has_variable = gm_new_contest }
				remove_variable = gm_new_contest
			}
			if = {
				limit = { has_variable_list = gm_monuments_held }
				every_in_list = {
					variable = gm_monuments_held
					limit = { has_variable = gm_tr_new_contest }
					remove_variable = gm_tr_new_contest
				}
			}
		}
	}
}

# ---- The dedication (§3): from the ceremony, monument_events.2 (ROOT = state) ----
# The ceremony's option knows the dedication, so these wrappers never read the
# production method back: whether activate_production_method is visible within
# the same effect is not established.
gm_state_dedicate = {
	# A construction site's answer: the site's records become the state's,
	# a finished monument standing here moving beside first; or Rededicate's
	# (sites spec 2026-10-18 §3.1, §5). Before anything below writes the
	# state's records.
	if = {
		limit = {
			has_variable = gm_ceremony_mode
			var:gm_ceremony_mode = flag:site
		}
		gm_state_finish_site = { KEY = $KEY$ }
	}
	else_if = {
		limit = {
			has_variable = gm_ceremony_mode
			var:gm_ceremony_mode = flag:rededicate
		}
		gm_state_apply_rededicate = yes
	}
	gm_remove_state_var = { VAR = gm_ceremony_mode }
	# A finished building standing here switches once, past the lock, and not
	# at all when it already stands on this one. One just created stands on it
	# (create_building's activate_production_methods), with gm_restore_pm as
	# the fallback.
	if = {
		limit = {
			NOT = { has_variable = gm_site_created }
			NOT = { gm_state_has_pm = { PM = pm_monument_$KEY$ } }
		}
		activate_production_method = {
			building_type = building_grand_monument
			production_method = pm_monument_$KEY$
		}
	}
	gm_remove_state_var = { VAR = gm_ceremony_pending }
	# The ceremony's answer is the method now: a restore recorded before it
	# (a return during the ceremony) must not put another back.
	if = {
		limit = { NOT = { has_variable = gm_site_created } }
		gm_remove_state_var = { VAR = gm_restore_pm }
	}
	gm_remove_state_var = { VAR = gm_site_created }
	set_variable = gm_seen
	# Finished at the levels built (owner, 2026-10-18): the site's, or what an
	# older save's monument stands at. Not read back from a building created in
	# this effect.
	if = {
		limit = { NOT = { has_variable = gm_levels } }
		set_variable = { name = gm_levels value = gm_state_primary_levels }
	}
	# Its starting grandeur (sites spec 2026-10-18 §3.2): its levels, unless
	# Rededicate kept half its own.
	# gm_state_primary_levels, not var:gm_levels: the options' tooltips walk
	# this body writing nothing, so gm_levels set just above reads as unset.
	if = {
		limit = { NOT = { has_variable = gm_start_grandeur } }
		set_variable = { name = gm_start_grandeur value = gm_state_primary_levels }
	}
	gm_state_record_builder = yes
	# The monument a commission met by this dedication names (v2 §2.3).
	owner = { set_variable = { name = gm_com_last_state value = prev } }
}

# ---- Older saves (sites spec 2026-10-18 §3.5) -------------------------------------
# State scope, from monument_events.32 once nothing is under construction here.
# A dedicated monument keeps its size in its records (gm_levels, from the old
# building arithmetic if it had none; its starting grandeur reads as its levels,
# the 1836 rate) and its building goes back to level 1 on its method. An
# undedicated one (left undedicated, or a ceremony that was open) becomes a
# construction site at its level, counted at the 1836 rate; the largest
# monument beside it, if any, takes the finished building.
gm_state_migrate = {
	set_variable = gm_migrated
	if = {
		limit = {
			gm_state_is_dedicated = yes
			has_variable = gm_seen
		}
		set_variable = { name = gm_mig_before value = gm_state_grandeur }
		if = {
			limit = { NOT = { has_variable = gm_levels } }
			set_variable = { name = gm_levels value = gm_state_built_levels }
		}
		# Levels finished above its record (queued before construction sites,
		# when they founded a monument beside it; the AI queued an
		# opportunity's on its tallest): a construction site of that many,
		# counted at the 1836 rate as an undedicated monument's are. Read
		# before the building goes back to level 1.
		set_variable = { name = gm_mig_excess value = gm_state_built_levels }
		change_variable = { name = gm_mig_excess subtract = var:gm_levels }
		gm_state_record_dedication = yes
		if = {
			limit = { gm_state_level > 1 }
			gm_state_mark_restore_pm = yes
			set_variable = { name = gm_rebuilding days = 1 }
			remove_building = building_grand_monument
			create_building = {
				building = building_grand_monument
				level = 1
			}
		}
		if = {
			limit = { var:gm_mig_excess > 0 }
			te_construction_market_build_specified_level = {
				SPEC_TYPE = building_grand_monument_site
				SPEC_LEVEL = var:gm_mig_excess
			}
			set_variable = { name = gm_site_counted value = var:gm_mig_excess }
			set_variable = { name = gm_site_grandeur value = var:gm_mig_excess }
			owner = { trigger_event = { id = monument_events.20 } }
		}
		remove_variable = gm_mig_excess
	}
	else = {
		set_variable = { name = gm_migrate_levels value = gm_state_built_levels }
		gm_remove_state_var = { VAR = gm_ceremony_pending }
		gm_remove_state_var = { VAR = gm_ceremony_level }
		gm_remove_state_var = { VAR = gm_ceremony_mode }
		set_variable = { name = gm_rebuilding days = 1 }
		if = {
			limit = { gm_state_has_secondary = yes }
			gm_state_promote_secondary = yes
		}
		else = {
			gm_clear_state = { KEEP_LANDMARK = no }
		}
		remove_building = building_grand_monument
		# The monument beside it that rose (its records: gm_seen) keeps the
		# finished building, at level 1 on its method (gm_restore_pm).
		if = {
			limit = { has_variable = gm_seen }
			create_building = {
				building = building_grand_monument
				level = 1
			}
		}
		te_construction_market_build_specified_level = {
			SPEC_TYPE = building_grand_monument_site
			SPEC_LEVEL = var:gm_migrate_levels
		}
		set_variable = { name = gm_site_counted value = var:gm_migrate_levels }
		set_variable = { name = gm_site_grandeur value = var:gm_migrate_levels }
		remove_variable = gm_migrate_levels
		owner = { trigger_event = { id = monument_events.20 } }
	}
}

# State scope, from gm_state_monthly while the building stands: once a game, a
# month or more after a dedicated monument's migration, whether it kept its
# grandeur (Review Focus 1 of the plan): within a hundredth below and a
# twentieth above (a month's ageing). Not the day script remade its building.
gm_monument_migration_self_check = {
	if = {
		limit = {
			has_variable = gm_mig_before
			NOT = { has_variable = gm_rebuilding }
			NOT = { has_variable = gm_restore_pm }
		}
		if = {
			limit = { NOT = { has_global_variable = gm_migration_self_checked } }
			set_global_variable = gm_migration_self_checked
			if = {
				limit = {
					gm_state_grandeur >= gm_state_mig_low
					gm_state_grandeur <= gm_state_mig_high
				}
				debug_log = "TE_SELF_CHECK OK monument_migrated (an older save's Grand Monument kept its grandeur through its migration: [SCOPE.ScriptValue('gm_state_grandeur')|1], was [SCOPE.ScriptValue('gm_state_mig_was')|1])"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN monument_migrated (an older save's Grand Monument changed grandeur through its migration: [SCOPE.ScriptValue('gm_state_grandeur')|1], was [SCOPE.ScriptValue('gm_state_mig_was')|1]; see docs/testing/grand-monuments-ageing-playtest.md)"
			}
		}
		remove_variable = gm_mig_before
	}
}

# ---- A construction site's dedication (sites spec 2026-10-18 §3.1) ----------------
# State scope, from gm_state_dedicate in the ceremony's answer (mode site; the
# Dedicate button and the AI ask only with nothing under construction). The
# levels counted to the last are the monument's records; a finished monument
# standing here moves beside it first, and the grander stands first once the
# newcomer is settled (gm_state_settle_newcomer from monument_events.29, a day
# or more later: one method switch a tick). The site goes, and a finished
# building stands at level 1 on the dedication: created here when none stands
# (gm_site_created tells gm_state_dedicate), switched by it when one does.
gm_state_finish_site = {
	# Only when it runs: the option's tooltip walks this body writing nothing,
	# and every read of a variable set above it would log an error a frame
	# (gm_state_rededicate's note).
	set_variable = gm_finish_click
	if = {
		limit = { has_variable = gm_finish_click }
		remove_variable = gm_finish_click
		gm_state_count_site_levels = yes
		if = {
			limit = {
				has_building = building_grand_monument
				has_variable = gm_seen
			}
			set_variable = { name = gm_demote_levels value = gm_state_primary_levels }
			gm_state_demote_primary = yes
			set_variable = gm_newcomer
		}
		set_variable = { name = gm_levels value = var:gm_site_counted }
		set_variable = { name = gm_start_grandeur value = var:gm_site_grandeur }
		set_variable = gm_migrated
		set_variable = { name = gm_rebuilding days = 1 }
		if = {
			limit = { NOT = { has_building = building_grand_monument } }
			remove_building = building_grand_monument_site
			create_building = {
				building = building_grand_monument
				level = 1
				activate_production_methods = { pm_monument_$KEY$ } # REVIEWED 2026-10-18: a create_building parameter (effects.log 1.14.5 § create_building), as the landmarks use it
			}
			set_variable = { name = gm_restore_pm value = flag:$KEY$ }
			set_variable = gm_site_created
			# The self-check monument_created_on_dedication
			# (gm_state_restore_dedication_one).
			set_variable = gm_created_check
		}
		else = {
			remove_building = building_grand_monument_site
		}
		remove_variable = gm_site_counted
		remove_variable = gm_site_grandeur
	}
}

# ---- Vanity backlash (§5) --------------------------------------------------------
# State scope. Levels finished while the country is in default, famine or
# recession: radicals in the state, and $AMOUNT$ more on the vanity ledger (the
# grandeur they add: the same share of the world's wealth draws the same
# outrage in any era, sites spec 2026-10-18 §3.6),
# which takes legitimacy linearly (gm_national_vanity) and fades over about
# two years. The journal entry warns beforehand (gm_hard_times_sgui, Task 9).
# From the construction site's monthly count (gm_state_count_site_levels), so
# the radicals come once a month however many levels finished. $AMOUNT$ is
# read on the owner: a state's variable is passed as prev.var:, and nothing is
# called between the owner block's opening and the read, so prev is the state.
gm_state_vanity_backlash_for = {
	add_radicals_in_state = { value = 0.005 }
	owner = { gm_init_ledgers = yes }
	owner = {
		change_variable = { name = gm_vanity_ledger add = $AMOUNT$ }
		post_notification = gm_vanity_backlash_notice
	}
}

# One level's (the debug event).
gm_state_vanity_backlash = {
	gm_state_vanity_backlash_for = { AMOUNT = 1 }
}

# ---- World wealth (Plan B) ---------------------------------------------------------
# Any scope. The rate written once if none is (an older save, or a game's first
# count before its first January): the order of the global and the state
# monthly pulses is unknown, so the count makes sure itself.
gm_world_rate_ensure = {
	if = {
		limit = { NOT = { has_global_variable = gm_world_rate } }
		set_global_variable = { name = gm_world_rate value = gm_world_rate_now }
	}
}

# ---- The construction site's count (sites spec 2026-10-18 §3.2) ------------------
# State scope, from gm_state_monthly (real execution, never the walked
# refresh) and at the start of every dedication. Levels the site has finished
# above those counted (gm_site_counted) join its starting grandeur so far
# (gm_site_grandeur), each at the world-wealth rate of the month it is counted
# (gm_world_rate_value, Plan B), and draw the vanity backlash in hard times. Fewer levels than counted (downsized, or
# lost) take their share of the grandeur with them. A site gone takes its
# records with it; a dedication reads them first.
gm_state_count_site_levels = {
	gm_world_rate_ensure = yes
	if = {
		limit = { has_building = building_grand_monument_site }
		if = {
			limit = { NOT = { has_variable = gm_site_counted } }
			set_variable = { name = gm_site_counted value = 0 }
			set_variable = { name = gm_site_grandeur value = 0 }
		}
		if = {
			limit = { gm_state_site_level > var:gm_site_counted }
			set_variable = { name = gm_site_new value = gm_state_site_level }
			change_variable = { name = gm_site_new subtract = var:gm_site_counted }
			# Each at the world-wealth rate of the month it is counted (Plan B).
			set_variable = {
				name = gm_site_add
				value = {
					value = var:gm_site_new
					multiply = gm_world_rate_value
				}
			}
			change_variable = { name = gm_site_grandeur add = var:gm_site_add }
			if = {
				limit = {
					owner = { gm_country_hard_times = yes }
					NOT = { has_variable = gm_rebuilding }
				}
				gm_state_vanity_backlash_for = { AMOUNT = prev.var:gm_site_add }
			}
			set_variable = { name = gm_site_counted value = gm_state_site_level }
			remove_variable = gm_site_new
			remove_variable = gm_site_add
		}
		else_if = {
			limit = { gm_state_site_level < var:gm_site_counted }
			set_variable = {
				name = gm_site_grandeur
				value = {
					value = var:gm_site_grandeur
					multiply = gm_state_site_level
					divide = var:gm_site_counted
				}
			}
			set_variable = { name = gm_site_counted value = gm_state_site_level }
		}
	}
	else_if = {
		limit = { has_variable = gm_site_counted }
		remove_variable = gm_site_counted
		remove_variable = gm_site_grandeur
	}
}
