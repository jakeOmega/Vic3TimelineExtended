# Construction-site placement, market games only: a site follows construction
# into each state and leaves when it is done, with a fallback in the capital.
# In direct games the sites are the Construction Sectors, built and expanded by
# their owner, so on_start / on_end do nothing there.

# State scope. Place a level-1 Construction Site. The flag lets the site's
# can_build_government accept this create_building in a market game (see
# common/buildings/te_construction_market_site.txt). A caller that already
# raised it (te_construction_market_country_update_sites) keeps it raised.
te_construction_market_place_site = {
	if = {
		limit = {
			has_global_variable = te_cm_placing_site
		}
		create_building = {
			building = te_construction_market_site
			level = 1
		}
	}
	else = {
		set_global_variable = {
			name = te_cm_placing_site
			value = 1
		}
		create_building = {
			building = te_construction_market_site
			level = 1
		}
		remove_global_variable = te_cm_placing_site
	}
}

te_construction_market_state_on_start = {
	if = {
		limit = {
			te_free_market_construction_on = yes
		}
		te_construction_market_state_on_start_market = yes
	}
}

te_construction_market_state_on_start_market = {
	if = {
		limit = {
			NOT = {
				any_scope_building = {
					is_building_type = te_construction_market_site
				}
			}
		}
		te_construction_market_place_site = yes
	}

	owner = {
		if = {
			limit = {
				capital = {
					AND = {
						any_scope_building = {
							is_building_type = te_construction_market_site
						}
						NOT = {
							any_scope_building = {
								is_under_construction = yes
							}
						}
					}
				}
			}
			capital = {
				remove_building = te_construction_market_site
			}
		}
	}


	# owner = {
	# 	te_construction_market_apply_per_site_modifier = yes
	# }
}

te_construction_market_state_on_end = {
	if = {
		limit = {
			te_free_market_construction_on = yes
		}
		te_construction_market_state_on_end_market = yes
	}
}

te_construction_market_state_on_end_market = {
	if = {
		limit = {
			NOT = {
				any_scope_building = {
					is_under_construction = yes
				}
			}
			any_scope_building = {
				is_building_type = te_construction_market_site
			}
			NOT = {
				is_capital = yes
			}
		}
		remove_building = te_construction_market_site
	}
	owner = {
		if = {
			limit = {
				NOT = {
					any_scope_building = {
						is_building_type = te_construction_market_site
					}
				}
			}
			capital = {
				te_construction_market_place_site = yes
			}

		}
	}
}
