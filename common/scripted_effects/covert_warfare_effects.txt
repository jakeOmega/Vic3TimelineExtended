# ============================================================================
# COVERT WARFARE — Scripted Effects
# ============================================================================
# Every running operation is a script container (1.13.10+):
#
#   tags    iw_op, iw_op_<TYPE>
#   parent  operator country — the engine culls the container if it stops existing
#   vars    iw_target          target country
#           iw_duration        months running (phase: 0-5 prep, 6-11 establishing, 12+ full)
#           iw_target_capital  target's capital state (JE display: the loc chain
#                              Var().GetState.GetCountry.GetName is the proven way to
#                              show a country name; see scripting_best_practices.md)
#           iw_detect          monthly detection % (JE display)
#           iw_tgt_ic          target's intelligence capacity (JE display)
#           iw_tgt_td          target's type-specific covert defense (JE display)
#
# The operator keeps references in its `iw_ops` variable list. Iterate that
# list, never every_container: the global iterator scans every container in
# the game, parent filter or not.
#
# Pacts stay the source of truth for whether an operation exists. The
# accept/break hooks create and destroy containers immediately;
# covert_ops_sync_all reconciles monthly for any path that skips them (target
# annexed, pact removed by script). It also adopts pacts from saves made
# before containers, starting them at month 0; the old slot-numbered
# variables in such saves are inert.
#
# Always remove_list_variable before destroy_container, so iw_ops never holds
# a dead reference. Storage has no per-type limit: the cap lives only in
# covert_ops_type_below_cap.
# ============================================================================

# Create an operation container and add it to the operator's iw_ops list.
# Leaves scope:iw_new_op pointing at the new container.
# Callers must guard any use of scope:iw_new_op with `exists`. When a
# diplomatic action's tooltip or confirmation box renders its accept_effect,
# the engine resolves scope links inside hidden_effect but never creates the
# container, so the scope is unset.
# Parameters: $TYPE$ = short name (e.g. election_interference)
#             $TARGET$ = target country (a scope: reference — it is read inside on_created)
# Scope: operator country
covert_op_create = {
	save_scope_as = iw_operator
	create_container = {
		tags = { iw_op iw_op_$TYPE$ }
		parent = scope:iw_operator
		save_scope_as = iw_new_op
	}
	if = {
		limit = { exists = scope:iw_new_op }
		scope:iw_new_op = {
			set_variable = { name = iw_target value = $TARGET$ }
			set_variable = { name = iw_duration value = 0 }
		}
		add_to_variable_list = { name = iw_ops target = scope:iw_new_op }
	}
}

# Destroy the $TYPE$ operation against $TARGET$, if there is one. The
# no-duplicate gate on every covert action keeps (type, target) unique.
# Parameters: $TYPE$ = short name, $TARGET$ = target country
# Scope: operator country
covert_op_destroy = {
	if = {
		limit = {
			has_variable_list = iw_ops
			any_in_list = {
				variable = iw_ops
				has_tag = iw_op_$TYPE$
				var:iw_target ?= $TARGET$
			}
		}
		random_in_list = {
			variable = iw_ops
			limit = {
				has_tag = iw_op_$TYPE$
				var:iw_target ?= $TARGET$
			}
			save_scope_as = iw_ended_op
		}
		remove_list_variable = { name = iw_ops target = scope:iw_ended_op }
		scope:iw_ended_op = { destroy_container = yes }
	}
}

# Recompute the target and detection data the JE shows for one operation.
# covert_operation_detection_chance reads target_max_ic / target_type_defense
# off the operator, so they are staged there first.
# Parameters: $DEFENSE_MOD$ = type-specific defense modifier
# Scope: operation container; scope:iw_operator = operator country
covert_op_refresh_detection = {
	save_scope_as = iw_op
	var:iw_target = {
		capital = {
			scope:iw_op = { set_variable = { name = iw_target_capital value = PREV } }
		}
		scope:iw_operator = {
			set_variable = { name = target_max_ic value = PREV.intelligence_capacity_total }
			set_variable = { name = target_type_defense value = PREV.modifier:$DEFENSE_MOD$ }
			set_variable = { name = iw_detect_staging value = covert_operation_detection_chance }
		}
	}
	set_variable = { name = iw_detect value = scope:iw_operator.var:iw_detect_staging }
	set_variable = { name = iw_tgt_ic value = scope:iw_operator.var:target_max_ic }
	set_variable = { name = iw_tgt_td value = scope:iw_operator.var:target_type_defense }
	scope:iw_operator = { remove_variable = iw_detect_staging }
}

# Start an operation. Called from the covert action's accept_effect, where
# the pact doesn't exist yet — so this creates the container directly
# instead of reconciling against pacts.
# Parameters: $TYPE$ = short name, $DEFENSE_MOD$ = type-specific defense modifier
# Scope: operator country; scope:target_country must be set
covert_op_start = {
	hidden_effect = {
		# A live duplicate is impossible (iw_no_duplicate_target_tt), but a
		# container whose pact ended through a path with no break hook may still
		# be waiting for the monthly sync. Clear it so the new op starts at month 0.
		covert_op_destroy = { TYPE = $TYPE$ TARGET = scope:target_country }
		covert_op_create = { TYPE = $TYPE$ TARGET = scope:target_country }
		if = {
			limit = { exists = scope:iw_new_op }
			scope:iw_new_op = { covert_op_refresh_detection = { DEFENSE_MOD = $DEFENSE_MOD$ } }
		}
	}
}

# End an operation. Called from the pact's manual_break_effect / auto_break_effect.
# If scope:target_country no longer resolves (target annexed), the monthly
# sync collects the container instead.
# Parameters: $TYPE$ = short name
# Scope: operator country; scope:target_country must be set
covert_op_end = {
	hidden_effect = {
		covert_op_destroy = { TYPE = $TYPE$ TARGET = scope:target_country }
	}
}

# Burn a detected operation: remove its pact and destroy its container.
# Parameters: $TYPE$ = short name, $ACTION$ = diplomatic_action_type key,
#             $TARGET$ = target country
# Scope: operator country
covert_op_burn = {
	remove_diplomatic_pact = { country = $TARGET$ type = $ACTION$ }
	covert_op_destroy = { TYPE = $TYPE$ TARGET = $TARGET$ }
}

# ============================================================================
# MONTHLY RECONCILIATION
# ============================================================================

# Reconcile this type's containers with the operator's pacts, then refresh
# their detection data. Idempotent.
#   1. Mark every container backed by a live pact; a pact with no container
#      gets one at month 0.
#   2. Collect unmarked containers into a temporary list, then remove and
#      destroy them, so iw_ops is never edited while it is being iterated.
# Parameters: $TYPE$ = short name
#             $ACTION$ = diplomatic_action_type key (e.g. covert_election_interference_action)
#             $DEFENSE_MOD$ = type-specific defense modifier
# Scope: operator country
covert_op_sync = {
	save_scope_as = iw_operator
	every_scope_diplomatic_pact = {
		limit = { is_diplomatic_action_type = $ACTION$ first_country = scope:iw_operator }
		every_participant = {
			limit = { NOT = { this = scope:iw_operator } }
			save_scope_as = iw_pact_target
			scope:iw_operator = {
				if = {
					limit = {
						NOT = {
							AND = {
								has_variable_list = iw_ops
								any_in_list = {
									variable = iw_ops
									has_tag = iw_op_$TYPE$
									var:iw_target ?= scope:iw_pact_target
								}
							}
						}
					}
					covert_op_create = { TYPE = $TYPE$ TARGET = scope:iw_pact_target }
				}
				every_in_list = {
					variable = iw_ops
					limit = {
						has_tag = iw_op_$TYPE$
						var:iw_target ?= scope:iw_pact_target
					}
					set_variable = { name = iw_seen value = yes }
				}
			}
		}
	}
	if = {
		limit = {
			has_variable_list = iw_ops
			any_in_list = {
				variable = iw_ops
				has_tag = iw_op_$TYPE$
				NOT = { has_variable = iw_seen }
			}
		}
		every_in_list = {
			variable = iw_ops
			limit = {
				has_tag = iw_op_$TYPE$
				NOT = { has_variable = iw_seen }
			}
			add_to_temporary_list = iw_ended_ops_$TYPE$
		}
		every_in_list = {
			list = iw_ended_ops_$TYPE$
			scope:iw_operator = { remove_list_variable = { name = iw_ops target = PREV } }
			destroy_container = yes
		}
	}
	if = {
		limit = { has_variable_list = iw_ops }
		every_in_list = {
			variable = iw_ops
			limit = { has_tag = iw_op_$TYPE$ }
			remove_variable = iw_seen
			covert_op_refresh_detection = { DEFENSE_MOD = $DEFENSE_MOD$ }
		}
	}
}

# Reconcile all 9 operation types. Called monthly from the JE pulse, before
# durations age and phase effects apply.
# Scope: country
covert_ops_sync_all = {
	covert_op_sync = { TYPE = election_interference ACTION = covert_election_interference_action DEFENSE_MOD = country_covert_defense_ideological_add }
	covert_op_sync = { TYPE = financial_subversion ACTION = covert_financial_subversion_action DEFENSE_MOD = country_covert_defense_economic_add }
	covert_op_sync = { TYPE = infrastructure_sabotage ACTION = covert_infrastructure_sabotage_action DEFENSE_MOD = country_covert_defense_military_add }
	covert_op_sync = { TYPE = comms_disruption ACTION = covert_comms_disruption_action DEFENSE_MOD = country_covert_defense_military_add }
	covert_op_sync = { TYPE = industrial_espionage ACTION = covert_industrial_espionage_action DEFENSE_MOD = country_covert_defense_economic_add }
	covert_op_sync = { TYPE = military_espionage ACTION = covert_military_espionage_action DEFENSE_MOD = country_covert_defense_military_add }
	covert_op_sync = { TYPE = influence_campaign ACTION = covert_influence_campaign_action DEFENSE_MOD = country_covert_defense_ideological_add }
	covert_op_sync = { TYPE = ideological_subversion ACTION = covert_ideological_subversion_action DEFENSE_MOD = country_covert_defense_ideological_add }
	covert_op_sync = { TYPE = destabilization ACTION = covert_destabilization_action DEFENSE_MOD = country_covert_defense_ideological_add }
}

# Age every running operation by one month. A new operation starts at 0, so
# its first pulse reads as month 1.
# Scope: country
covert_ops_age_all_durations = {
	if = {
		limit = { has_variable_list = iw_ops }
		every_in_list = {
			variable = iw_ops
			change_variable = { name = iw_duration add = 1 }
		}
	}
}

# ============================================================================
# PHASE-BASED EFFECT APPLICATION
# ============================================================================
# Phase 1 (months 0-5): Preparatory — no effect
# Phase 2 (months 6-11): Establishing — base multiplier (1x)
# Phase 3 (months 12+): Fully Operational — doubled multiplier (2x)
# Phase triggers (container scope): covert_op_is_established, covert_op_is_fully_operational

# Apply a country-scoped modifier to the TARGET of each $TYPE$ operation,
# at that operation's own phase.
# Parameters: $TYPE$, $MODIFIER$
# Scope: country (ROOT = JE owner / operator)
covert_op_apply_target_effect = {
	every_in_list = {
		variable = iw_ops
		limit = { has_tag = iw_op_$TYPE$ }
		save_scope_as = iw_op
		var:iw_target = {
			remove_modifier = $MODIFIER$
			if = {
				limit = { scope:iw_op = { covert_op_is_fully_operational = yes } }
				add_modifier = { name = $MODIFIER$ multiplier = 2 months = 3 }
			}
			else_if = {
				limit = { scope:iw_op = { covert_op_is_established = yes } }
				add_modifier = { name = $MODIFIER$ months = 3 }
			}
		}
	}
}

# Apply a country-scoped modifier to SELF for a given operation type, at the
# BEST phase among that type's operations.
# Parameters: $TYPE$, $MODIFIER$
# Scope: country (ROOT = JE owner / operator)
covert_op_apply_self_effect = {
	if = {
		limit = { any_in_list = { variable = iw_ops has_tag = iw_op_$TYPE$ } }
		remove_modifier = $MODIFIER$
		if = {
			limit = {
				any_in_list = {
					variable = iw_ops
					has_tag = iw_op_$TYPE$
					covert_op_is_fully_operational = yes
				}
			}
			add_modifier = { name = $MODIFIER$ multiplier = 2 months = 3 }
		}
		else_if = {
			limit = {
				any_in_list = {
					variable = iw_ops
					has_tag = iw_op_$TYPE$
					covert_op_is_established = yes
				}
			}
			add_modifier = { name = $MODIFIER$ months = 3 }
		}
		# Phase 1: all at prep — no modifier
	}
}

# Master effect: apply ALL 9 operation types' phase-based effects.
# Only runs if funding > 0, after covert_ops_sync_all. Called monthly from JE pulse.
# The helpers it calls iterate iw_ops unguarded; the has_variable_list check here covers them.
# Scope: country (ROOT = JE owner)
covert_ops_apply_all_phase_effects = {
	if = {
		limit = { has_variable_list = iw_ops }
		# ---- Election Interference: country modifier on TARGET ----
		covert_op_apply_target_effect = { TYPE = election_interference MODIFIER = covert_election_interference }

		# ---- Financial Subversion: country modifier on TARGET ----
		covert_op_apply_target_effect = { TYPE = financial_subversion MODIFIER = covert_financial_subversion }

		# ---- Infrastructure Sabotage: country morale modifier + state modifier on a random TARGET state ----
		covert_op_apply_target_effect = { TYPE = infrastructure_sabotage MODIFIER = covert_infra_sabotage_morale }
		every_in_list = {
			variable = iw_ops
			limit = {
				has_tag = iw_op_infrastructure_sabotage
				covert_op_is_established = yes
			}
			save_scope_as = iw_op
			var:iw_target = {
				random_scope_state = {
					limit = { any_scope_building = { level >= 1 } }
					remove_modifier = covert_infrastructure_sabotage
					if = {
						limit = { scope:iw_op = { covert_op_is_fully_operational = yes } }
						add_modifier = { name = covert_infrastructure_sabotage multiplier = 2 months = 3 }
					}
					else = {
						add_modifier = { name = covert_infrastructure_sabotage months = 3 }
					}
				}
			}
		}

		# ---- Communications Disruption: country modifier on TARGET ----
		covert_op_apply_target_effect = { TYPE = comms_disruption MODIFIER = covert_comms_disruption }

		# ---- Industrial Espionage: self modifiers (innovation + tech spread) ----
		# Innovation is always applied; tech spread always applies since target must have tech advantage
		covert_op_apply_self_effect = { TYPE = industrial_espionage MODIFIER = covert_espionage_base }
		covert_op_apply_self_effect = { TYPE = industrial_espionage MODIFIER = covert_industrial_espionage }
		# Weak counter-intel marker on the TARGET so covert_warfare.2 can notify them (#158)
		covert_op_apply_target_effect = { TYPE = industrial_espionage MODIFIER = covert_industrial_espionage_detected }

		# ---- Military Espionage: self modifiers (unit stats always, tech spread conditional) ----
		covert_op_apply_self_effect = { TYPE = military_espionage MODIFIER = covert_military_espionage_unit }
		# Weak counter-intel marker on the TARGET so covert_warfare.2 can notify them (#158)
		covert_op_apply_target_effect = { TYPE = military_espionage MODIFIER = covert_military_espionage_detected }
		# Conditional tech spread: only if target has tech advantage (best phase of any operation)
		if = {
			limit = {
				any_in_list = {
					variable = iw_ops
					has_tag = iw_op_military_espionage
					covert_op_is_established = yes
				}
			}
			every_in_list = {
				variable = iw_ops
				limit = { has_tag = iw_op_military_espionage }
				var:iw_target = { save_scope_as = espionage_target }
			}
			if = {
				limit = {
					exists = scope:espionage_target
					espionage_target_tech_advantage > 0
				}
				remove_modifier = covert_military_espionage
				if = {
					limit = {
						any_in_list = {
							variable = iw_ops
							has_tag = iw_op_military_espionage
							covert_op_is_fully_operational = yes
						}
					}
					add_modifier = { name = covert_military_espionage multiplier = 2 months = 3 }
				}
				else = {
					add_modifier = { name = covert_military_espionage months = 3 }
				}
			}
		}

		# ---- Influence Campaign: country modifier on TARGET ----
		covert_op_apply_target_effect = { TYPE = influence_campaign MODIFIER = covert_influence_campaign }

		# ---- Ideological Subversion: country resist modifier + movement modifiers on TARGET ----
		covert_op_apply_target_effect = { TYPE = ideological_subversion MODIFIER = covert_ideological_subversion_resist }
		# Movement pressure from the establishing phase on (same strength once fully operational)
		every_in_list = {
			variable = iw_ops
			limit = {
				has_tag = iw_op_ideological_subversion
				covert_op_is_established = yes
			}
			ROOT = { save_scope_as = cultural_hegemon }
			var:iw_target = {
				ch_apply_hegemon_movement_pressure = { MODIFIER = covert_ideological_subversion FALLBACK_MODIFIER = covert_ideological_subversion MONTHS = 3 DECAYING = no MULT = 1 }
			}
		}

		# ---- Destabilization: country resist modifier + movement modifiers on TARGET ----
		covert_op_apply_target_effect = { TYPE = destabilization MODIFIER = covert_destabilization_resist }
		every_in_list = {
			variable = iw_ops
			limit = {
				has_tag = iw_op_destabilization
				covert_op_is_established = yes
			}
			save_scope_as = iw_op
			var:iw_target = {
				# Separatist movement pressure
				every_political_movement = {
					remove_modifier = covert_destabilization_separatist
					if = {
						limit = { scope:iw_op = { covert_op_is_fully_operational = yes } }
						add_modifier = { name = covert_destabilization_separatist multiplier = 2 months = 3 }
					}
					else = {
						add_modifier = { name = covert_destabilization_separatist months = 3 }
					}
				}
				# General destabilization pressure
				remove_modifier = covert_destabilization_general
				if = {
					limit = { scope:iw_op = { covert_op_is_fully_operational = yes } }
					add_modifier = { name = covert_destabilization_general multiplier = 2 months = 3 }
				}
				else = {
					add_modifier = { name = covert_destabilization_general months = 3 }
				}
			}
		}
	}
}

# ============================================================================
# ELECTION INTERFERENCE — Electoral Confidence Effect
# ============================================================================
# Called from on_election_campaign_end (ROOT = country that just held election).
# Finds the worst-phase election interference operation targeting ROOT.
# Phase 1 (prep, <6 mo): no effect.
# Phase 2 (establishing, 6-11 mo): -0.05 electoral confidence.
# Phase 3 (fully operational, 12+ mo): -0.1 electoral confidence.
# Only the worst (highest phase) among all attackers matters.
# Can only fire once per election (guarded by caller with has_variable check).
covert_op_election_confidence_effect = {
	set_variable = { name = iw_election_worst_phase value = 0 }

	# Every attacker running election interference against ROOT
	every_scope_diplomatic_pact = {
		limit = { is_diplomatic_action_type = covert_election_interference_action second_country = { this = ROOT } }
		every_participant = {
			limit = {
				NOT = { this = ROOT }
				has_variable_list = iw_ops
			}
			# The attacker's operation against ROOT
			every_in_list = {
				variable = iw_ops
				limit = {
					has_tag = iw_op_election_interference
					var:iw_target ?= ROOT
				}
				if = {
					limit = { covert_op_is_fully_operational = yes }
					ROOT = { change_variable = { name = iw_election_worst_phase max = 3 } }
				}
				else_if = {
					limit = { covert_op_is_established = yes }
					ROOT = { change_variable = { name = iw_election_worst_phase max = 2 } }
				}
				else = {
					ROOT = { change_variable = { name = iw_election_worst_phase max = 1 } }
				}
			}
		}
	}

	# Apply electoral confidence penalty based on worst phase
	if = {
		limit = { var:iw_election_worst_phase >= 3 }
		add_electoral_confidence = -0.1
		post_notification = iw_election_interference_confidence
	}
	else_if = {
		limit = { var:iw_election_worst_phase >= 2 }
		add_electoral_confidence = -0.05
		post_notification = iw_election_interference_confidence
	}
	# Phase 1 (prep): no effect

	remove_variable = iw_election_worst_phase
}
