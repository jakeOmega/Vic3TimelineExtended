# ============================================================================
# SOLAR SYSTEM COLONIZATION — launch windows
# ============================================================================
# Global Commons spec § 5. Chemical propulsion reaches each world only in its
# launch window; propellant depots (Asteroid Mining) buy launches outside one
# at the premium the extra propellant costs, a smaller one with Nuclear Thermal
# Propulsion (lunar colonization design § 9); fusion drives end the waiting.
# While no fleet can leave, the programme keeps building: progress past the
# goal is fleets staged in orbit, and they depart one a month while a window
# is open. The windows are generated (scripts/generators/gen_launch_windows.py);
# the rule is common/scripted_triggers/sr_launch_window_triggers.txt and
# common/script_values/sr_launch_window_values.txt.
#
# State: the cache below is global variables (any revolution keeps them, and
# the global monthly pulse rebuilds them anyway); staged progress is the
# country's sr_progress_solar_colonization, which a revolution's winner
# inherits like any milestone progress.
# ============================================================================

# ---- The window cache --------------------------------------------------------
# Any scope. Each body's state this month, so country pulses, the colony
# draw and the panel read one variable each instead of the generated chains:
#   sr_window_is_open_<body>  present while its window is open
#   sr_window_wait_<body>     months to its next window (0 while open)
#   sr_window_cost_<body>     this month's off-window premium (0 while open)
#   sr_window_cost_ntp_<body> the same for nuclear thermal engines (a country
#                             with Nuclear Thermal Propulsion pays it:
#                             sr_window_premium_country_<body>)
#   sr_window_cache_month     the te_history_month_index these were taken for
# Filled from the global monthly pulse (sr_launch_windows_on_actions.txt) and,
# when that has not yet run this month, by the first journal entry that needs
# it (sr_launch_windows_ensure_cache): a country's monthly pulse can come on
# the 1st before the global one.
sr_launch_windows_refresh_cache = {
	set_global_variable = { name = sr_window_cache_month value = te_history_month_index }
	sr_launch_window_cache_body = { BODY = mars }
	sr_launch_window_cache_body = { BODY = ceres }
	sr_launch_window_cache_body = { BODY = vesta }
	sr_launch_window_cache_body = { BODY = psyche }
	sr_launch_window_cache_body = { BODY = pallas }
	sr_launch_window_cache_body = { BODY = hygiea }
	sr_launch_window_cache_body = { BODY = jupiter }
	sr_launch_window_cache_body = { BODY = venus }
	sr_launch_window_cache_body = { BODY = saturn }
	sr_launch_window_cache_body = { BODY = uranus }
	sr_launch_window_cache_body = { BODY = neptune }
	sr_launch_window_cache_body = { BODY = kuiper }
}

sr_launch_window_cache_body = {
	set_global_variable = { name = sr_window_wait_$BODY$ value = sr_window_months_to_next_$BODY$ }
	set_global_variable = { name = sr_window_cost_$BODY$ value = sr_launch_premium_$BODY$ }
	set_global_variable = { name = sr_window_cost_ntp_$BODY$ value = sr_launch_premium_ntp_$BODY$ }
	# A colony fleet's transit leaving this month, by drive (slices 2-3 design
	# § 2; chemical is a constant): read by sr_transit_months_<body>.
	set_global_variable = { name = sr_window_transit_fusion_$BODY$ value = sr_transit_fusion_$BODY$ }
	set_global_variable = { name = sr_window_transit_torch_$BODY$ value = sr_transit_torch_$BODY$ }
	set_global_variable = { name = sr_window_transit_antimatter_$BODY$ value = sr_transit_antimatter_$BODY$ }
	if = {
		limit = { sr_window_open_$BODY$ = yes }
		set_global_variable = { name = sr_window_is_open_$BODY$ value = yes }
	}
	else_if = {
		limit = { has_global_variable = sr_window_is_open_$BODY$ }
		remove_global_variable = sr_window_is_open_$BODY$
	}
}

sr_launch_windows_ensure_cache = {
	if = {
		limit = { NOT = { sr_launch_window_cache_current = yes } }
		sr_launch_windows_refresh_cache = yes
	}
}

# ---- The month's launch --------------------------------------------------------
# Country scope: the solar colonization journal entry's monthly pulse, after
# the month's progress and setback roll, in place of the old bar-fill check.
# At most one colony a month. The fleet costs goal x (1 + premium) for the
# cheapest site that can depart (0 in a window or with fusion drives), and that
# much progress is spent only when a site was really claimed. What is left
# stays staged: it survives a stage change (the stage's goal is read again
# next month) and is lost only when the last world is claimed, as an unfinished
# bar always was (the entry's completion branch removes it).
sr_launch_month = {
	sr_launch_restore_banked_fleets = yes
	sr_launch_windows_ensure_cache = yes
	# A target (the Solar System list's Target button; sr_colony_fleet_effects.txt)
	# that a rival claimed first is dropped: the programme picks for itself again.
	if = {
		limit = {
			has_variable = sr_cf_target
			NOT = { sr_cf_target_valid = yes }
		}
		remove_variable = sr_cf_target
	}
	# Simple and full colonies: a colony fleet travels and claims on arrival
	# (slices 2-3 design § 4).
	if = {
		limit = { sr_colonies_contested_enabled = yes }
		sr_cf_launch_month = yes
	}
	# Races only with a target (the owner's D19): founded at once, at the
	# stage's goal, when its body can be reached this month.
	else_if = {
		limit = { has_variable = sr_cf_target }
		if = {
			limit = { sr_cf_target_ready = yes }
			sr_cf_found_target_now = yes
		}
		else = {
			sr_launch_note_held = yes
		}
	}
	else_if = {
		limit = { sr_launch_fleet_ready = yes }
		# Read before the colony effect, which may advance the stage and raise the goal.
		set_variable = { name = sr_goal_before_colony value = sr_launch_fleet_cost }
		set_local_variable = { name = sr_launch_claimed_before value = sr_total_global_colonies }
		sr_establish_colony_effect = yes
		if = {
			limit = { local_var:sr_launch_claimed_before < sr_total_global_colonies }
			change_variable = { name = sr_progress_solar_colonization subtract = var:sr_goal_before_colony }
		}
		remove_variable = sr_goal_before_colony
	}
	else = {
		# A full bar held by closed windows: the window_gate self-check's
		# evidence that the gate has held a fleet (below). Debug only.
		sr_launch_note_held = yes
	}
}

# ---- Fleets staged before a first colony survive the programme's stop ----------
# Country scope. A programme holding no colony loses its progress when its
# entry fails (its Space Program off Solar Colonization: on_fail's cleanup),
# deactivates (no longer a great or major power: the monthly cleanup) or runs
# `immediate` again (re-activation, or a revolution's winner): the bar
# restarts, as an unfinished bar always did, because progress carried into an
# activation would become the journal bar's baseline. The whole fleets staged
# in orbit are kept apart first (sr_launch_banked_fleets, a count of goals) and
# put back by the programme's next launch month; only the colony being readied,
# the remainder below a goal, is lost. Additive, so a second call after the
# reset adds nothing. Not once every world is claimed: staged progress is lost
# then (spec § 5.2), and the entry can't come back to use it.
sr_launch_bank_staged_fleets = {
	if = {
		limit = {
			has_variable = sr_progress_solar_colonization
			sr_launch_staged_whole_fleets >= 1
			NOT = { has_global_variable = sr_all_colonies_complete }
		}
		if = {
			limit = { NOT = { has_variable = sr_launch_banked_fleets } }
			set_variable = { name = sr_launch_banked_fleets value = 0 }
		}
		change_variable = { name = sr_launch_banked_fleets add = sr_launch_staged_whole_fleets }
	}
}

# From sr_launch_month, the running programme's pulse: the banked fleets back
# into progress, at the current stage's goal (a stage that moved on while the
# programme was stopped takes them as it would staged progress).
sr_launch_restore_banked_fleets = {
	if = {
		limit = { has_variable = sr_launch_banked_fleets }
		if = {
			limit = { NOT = { has_global_variable = sr_all_colonies_complete } }
			if = {
				limit = { NOT = { has_variable = sr_progress_solar_colonization } }
				set_variable = { name = sr_progress_solar_colonization value = 0 }
			}
			set_local_variable = { name = sr_launch_restored value = var:sr_launch_banked_fleets }
			change_local_variable = { name = sr_launch_restored multiply = sr_solar_colonization_goal }
			change_variable = { name = sr_progress_solar_colonization add = local_var:sr_launch_restored }
		}
		remove_variable = sr_launch_banked_fleets
	}
}

# ---- Self-check: window_gate (spec § 8) --------------------------------------------
# Country scope, from each gated site's branch of sr_establish_colony_effect as
# the colony is claimed. A country bound by windows and without propellant
# depots must never found a colony while its body's window is closed. The
# check reads the date's own answer (sr_window_open_<body>, the generated
# chain), not the cache the gate reads, so a stale or wrong cache shows as
# BROKEN. OK is logged once a game, and only once a closed window has held a
# full bar for this country (sr_launch_held_months), so a game where nothing
# ever waited can't pass it. BROKEN is logged once a game too. A country with
# depots that launched outside the window logs the premium it paid, once
# (window_gate_paid: the cache's premium against the date's own).
sr_launch_window_gate_check = {
	if = {
		limit = {
			sr_launch_windows_apply = yes
			NOT = { sr_launch_has_depots = yes }
		}
		if = {
			limit = { NOT = { sr_window_open_$BODY$ = yes } }
			if = {
				limit = { NOT = { has_global_variable = sr_window_gate_broken_logged } }
				set_global_variable = { name = sr_window_gate_broken_logged value = yes }
				sr_window_log_copy = { BODY = $BODY$ }
				debug_log = "TE_SELF_CHECK BROKEN window_gate body=[GetGlobalVariable('sr_window_log_body_now').GetValue|0] (1 Mars, 2 Ceres, 3 Vesta, 4 Psyche, 5 Pallas, 6 Hygiea, 7 Jupiter system, 8 Venus, 9 Saturn system, 10 Uranus system, 11 Neptune system, 12 Kuiper Belt) month=[THIS.ScriptValue('te_history_month_index')|0] months_to_next=[GetGlobalVariable('sr_window_log_wait_date_now').GetValue|0] cached_wait=[GetGlobalVariable('sr_window_log_wait_cached_now').GetValue|0] cache_month=[THIS.ScriptValue('sr_window_disp_cache_month')|0] country=[THIS.GetCountry.GetNameNoFormatting] (a colony was founded while its body's window was closed, for a country with no off-window launches: the cache or the gate is wrong; docs/testing/global-commons-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		else_if = {
			limit = {
				has_variable = sr_launch_held_months
				NOT = { has_global_variable = sr_window_gate_ok_logged }
			}
			set_global_variable = { name = sr_window_gate_ok_logged value = yes }
			sr_window_log_copy = { BODY = $BODY$ }
			debug_log = "TE_SELF_CHECK OK window_gate body=[GetGlobalVariable('sr_window_log_body_now').GetValue|0] (1 Mars, 2 Ceres, 3 Vesta, 4 Psyche, 5 Pallas, 6 Hygiea, 7 Jupiter system, 8 Venus, 9 Saturn system, 10 Uranus system, 11 Neptune system, 12 Kuiper Belt) month=[THIS.ScriptValue('te_history_month_index')|0] held_months=[THIS.Var('sr_launch_held_months').GetValue|0] country=[THIS.GetCountry.GetNameNoFormatting] (closed windows held a full bar, and the colony left in its window) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
	else_if = {
		limit = {
			sr_launch_windows_apply = yes
			NOT = { sr_window_open_$BODY$ = yes }
			NOT = { has_global_variable = sr_window_gate_paid_logged }
		}
		set_global_variable = { name = sr_window_gate_paid_logged value = yes }
		sr_window_log_copy = { BODY = $BODY$ }
		debug_log = "TE_SELF_CHECK OK window_gate_paid body=[GetGlobalVariable('sr_window_log_body_now').GetValue|0] (1 Mars, 2 Ceres, 3 Vesta, 4 Psyche, 5 Pallas, 6 Hygiea, 7 Jupiter system, 8 Venus, 9 Saturn system, 10 Uranus system, 11 Neptune system, 12 Kuiper Belt) month=[THIS.ScriptValue('te_history_month_index')|0] premium_cached=[GetGlobalVariable('sr_window_log_premium_cached_now').GetValue|2] premium_date=[GetGlobalVariable('sr_window_log_premium_date_now').GetValue|2] spent=[THIS.Var('sr_goal_before_colony').GetValue|0] goal=[THIS.ScriptValue('sr_solar_colonization_goal')|0] (an off-window launch: spent should be goal x (1 + premium), and the two premiums equal) country=[THIS.GetCountry.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}

# One body's figures copied into fixed-name globals for a debug line: a
# $BODY$ can't be printed or interpolated inside a debug_log string (the
# character before it is lost and the closing $ stays; scripting_best_practices.md,
# "No $PARAM$ inside the string"), so the lines read these through
# GetGlobalVariable and name the body by its code (sr_window_body_code_<body>,
# legend BODY_CODE_LEGEND in scripts/generators/gen_launch_windows.py). Country
# scope: the premiums are the ones this country's engines pay, nuclear thermal
# with Nuclear Thermal Propulsion, cached and by date. Debug only: nothing else
# reads the sr_window_log_*_now globals.
sr_window_log_copy = {
	set_global_variable = { name = sr_window_log_body_now value = sr_window_body_code_$BODY$ }
	set_global_variable = { name = sr_window_log_wait_date_now value = sr_window_months_to_next_$BODY$ }
	set_global_variable = { name = sr_window_log_wait_cached_now value = sr_window_wait_months_$BODY$ }
	set_global_variable = { name = sr_window_log_offset_now value = sr_launch_offset_$BODY$ }
	set_global_variable = { name = sr_window_log_premium_cached_now value = sr_window_premium_country_$BODY$ }
	if = {
		limit = { sr_launch_has_ntp = yes }
		set_global_variable = { name = sr_window_log_premium_date_now value = sr_launch_premium_ntp_$BODY$ }
	}
	else = {
		set_global_variable = { name = sr_window_log_premium_date_now value = sr_launch_premium_$BODY$ }
	}
	set_global_variable = { name = sr_window_log_cost_now value = sr_launch_cost_$BODY$ }
	set_global_variable = { name = sr_window_log_months_cost_now value = sr_launch_months_cost_$BODY$ }
	set_global_variable = { name = sr_window_log_verdict_now value = sr_window_disp_verdict_$BODY$ }
}

# ---- Setbacks hit only the fleet being readied (spec § 5.2) ------------------------
# A setback, or an in-progress event's loss, never takes more than one goal from
# solar colonization, so fleets staged in orbit are safe: sr_lose_progress_months
# (space_race_effects.txt) caps every loss at the milestone's goal.

# ---- Display: the destinations' lines -----------------------------------------------
# Printed by sr_launch_windows_sgui (common/scripted_guis/sr_launch_window_sguis.txt)
# through ExecuteTooltip, so nothing here ever runs. Country scope (ROOT = the
# panel's owner). One line per window body in play: open now, waiting for its
# window, or (with propellant depots) the premium to launch now and whether the
# approach would.
sr_launch_window_line = {
	if = {
		limit = { sr_window_body_in_play_$BODY$ = yes }
		if = {
			limit = { has_global_variable = sr_window_is_open_$BODY$ }
			custom_tooltip_no_bullet = sr_window_line_open_$BODY$
		}
		else_if = {
			limit = { sr_launch_has_depots = yes }
			custom_tooltip_no_bullet = sr_window_line_paid_$BODY$
			custom_tooltip_no_bullet = sr_window_line_verdict_$BODY$
		}
		else = {
			custom_tooltip_no_bullet = sr_window_line_wait_$BODY$
		}
	}
}
