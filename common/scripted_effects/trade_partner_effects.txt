# Scripted effects for computing Top Trade Partners data for GUI display.
# Uses ordered_market to find the five markets each market trades most with,
# by mutual trade value, and keeps them as a variable list on the market's
# owner country (1 country = 1 market), largest first:
#   te_tp_top_markets   = the partner markets (Top Trade Partners' rows,
#                         a datamodel in gui/market_panel.gui; MKT-1,
#                         2026-10-27)
#   te_tp_import_markets, te_tp_export_markets
#                       = the Import/Export Partners charts' markets
# The rows read each partner's figures live, through the charts' GuiScope
# form: gui_tp_annual_trade (the weekly mutual_trade_value_with_country x 52,
# at base prices) and gui_tp_trade_pct_gdp (that yearly trade as a percentage
# of the owner's yearly GDP), with the partner's owner passed as
# scope:tp_partner_country. The fifteen per-rank variables the five copied
# rows read (tp_<N>_tc, tp_<N>_total, tp_<N>_gdp_pct) and tp_count are gone.
#
# NOTE: Per-good per-partner trade data is not reachable from the SCRIPT layer
# here: market_imports_reliance(target) in parenthesized form only passes the
# target parameter, not the goods parameter, so all goods evaluate to the same
# total reliance (ordered_market_goods then sorts by default order, not per-good).
# (The table used to show that reliance as Imp%/Exp% columns; it now shows the
# trade as a share of GDP instead.)
# It IS reachable from the GUI layer, though: Market.GetImportedAmountFromMarket(
# otherMarket, Goods) / Market.GetExportedAmountToMarket(otherMarket, Goods) give
# per-good per-market-pair amounts. The Top Trade Partners rows use exactly that
# for their goods-breakdown tooltip (see gui/te_trade_partner_tooltips.gui).

# ============================================================================
# PER-MARKET: update the trade partner lists for a single market
# Called in market scope from every_market
# ============================================================================
update_market_trade_partners = {
	# Skip a market with no trade center state or no owner: there is no country
	# to keep its lists on, and scope:tp_market_owner could still hold the owner
	# an earlier market saved, whose lists this one would overwrite (log triage
	# 2026-10-29: trade_center invalid for one market, once a month).
	if = {
		limit = { trade_center ?= { exists = owner } }
		save_scope_as = tp_current_market
		trade_center = {
			owner = {
				save_scope_as = tp_market_owner
			}
		}

		scope:tp_market_owner = {
			if = {
				limit = { has_variable_list = te_tp_top_markets }
				clear_variable_list = te_tp_top_markets
			}
		}

		# The five markets this market trades most with, largest first, appended
		# in ordered_market's order so the stored order is the rank. A market it
		# does not trade with at all is no partner, so it is left out.
		ordered_market = {
			limit = {
				NOT = { this = scope:tp_current_market }
				gui_trade_with_market_owner > 0
			}
			order_by = gui_trade_with_market_owner
			max = 5
			check_range_bounds = no

			save_temporary_scope_as = tp_iter_mkt
			scope:tp_market_owner = {
				add_to_variable_list = { name = te_tp_top_markets target = scope:tp_iter_mkt }
			}
		}

		# ------------------------------------------------------------------------
		# Sorted, partner-only market lists for the Import/Export Partners charts
		# (gui/market_panel.gui). compact_chart cannot hide zero-trade legend rows,
		# so the datamodel must be pre-filtered here. Capped at 15 to bound the
		# monthly every_market cost (chart shows top few + "show more").
		# Mirrors alcah/victoria3-tradepartners' trade_partners_set_*_partners.
		# ------------------------------------------------------------------------
		scope:tp_market_owner = {
			clear_variable_list = te_tp_import_markets
			clear_variable_list = te_tp_export_markets
		}

		# Import partners: markets THIS market imports FROM (they export to us), desc.
		ordered_market = {
			limit = {
				NOT = { this = scope:tp_current_market }
				te_tp_exports_to_current > 0
			}
			order_by = te_tp_exports_to_current
			max = 15
			check_range_bounds = no

			save_temporary_scope_as = tp_iter_mkt
			scope:tp_market_owner = {
				add_to_variable_list = { name = te_tp_import_markets target = scope:tp_iter_mkt }
			}
		}

		# Export partners: markets THIS market exports TO (they import from us), desc.
		ordered_market = {
			limit = {
				NOT = { this = scope:tp_current_market }
				te_tp_imports_from_current > 0
			}
			order_by = te_tp_imports_from_current
			max = 15
			check_range_bounds = no

			save_temporary_scope_as = tp_iter_mkt
			scope:tp_market_owner = {
				add_to_variable_list = { name = te_tp_export_markets target = scope:tp_iter_mkt }
			}
		}
	}
}

# ============================================================================
# MAIN: update trade partners for ALL markets
# Called from on_monthly_pulse_country (player only)
# ROOT = player country
# ============================================================================
update_top_trade_partners = {
	every_market = {
		update_market_trade_partners = yes
	}
}
