# ============================================================================
# RESOLUTION LOBBYING — pledges and campaigns
# ============================================================================
# docs/superpowers/specs/2026-09-27-un-vote-lobbying-design.md. Any member may
# try to move another member's vote on the resolution in session, in either
# direction, two ways:
#
#   a PLEDGE      "Seek a Vote Commitment" (for or against), a
#                 requires_approval diplomatic action: the member may refuse;
#                 if it accepts, its lean moves ±100 and the asker owes it a
#                 vanilla diplomatic obligation, paid up front. Settled when the
#                 member's vote is booked (un_lobby_settle_commitment).
#   a CAMPAIGN    "Lobby For / Against the Resolution", a one-sided pact with
#                 influence upkeep, on an AI member only: +3 lean a month its
#                 way, up to 15; campaigns one way stack to 20 (§ CAMPAIGNS
#                 below).
#
# ---------------------------------------------------------------------------
# PLEDGE STATE
# ---------------------------------------------------------------------------
# On the resolution container:
#   lists   un_res_lobbied         every member approached on this resolution,
#                                  by anyone, accepted or declined (a record)
#           un_res_committed_yes   PENDING pledges to vote for. A member is
#                                  taken off the instant its vote is booked, so
#                                  membership means "still owes a vote"
#           un_res_committed_no    PENDING pledges to vote against, the same way
#           un_res_commit_kept     settled: voted the way it pledged
#           un_res_commit_broken   settled: did not
#   vars    un_res_commit_count    pledges ACCEPTED on the resolution, ever
#                                  (the chamber's Commitments block shows
#                                  itself on it)
#           un_res_commit_pending  unsettled pledges, for the chamber's tally
#                                  line. A variable rather than a list size
#                                  because loc can print a variable and cannot
#                                  print a list size.
# On the member that pledged (country variables, so a civil war's winner
# inherits them and continues the nation's promise):
#           un_pledge_to           the country it pledged to
#           un_pledge_dir          1 for, -1 against
#           un_pledge_res          the resolution
#           un_pledge_askers       list: who has asked it on un_pledge_askers_res
#           un_pledge_askers_res   the resolution those asks were about
# On the asker:
#           un_pledge_count        pledges it has collected on un_pledge_count_res
#           un_pledge_count_res    (un_lobby_commitment_cap = 2 per resolution)
#
# A member pledges once per resolution (the first accepted wins,
# un_pledge_made_by); an asker asks a member once per resolution
# (un_pledge_asked_by). A pledge made before this rework was always to the
# proposer and always for; it carries no un_pledge_to, and settles against the
# proposer as it always did.
#
# OLD SAVES. Nothing here assumes any of the lists or counters exist: every
# read is guarded (common/scripted_triggers/un_lobby_triggers.txt).
# ============================================================================

# Book the ask itself, whatever the answer was. Called from BOTH the accept
# and the decline branch of the pledge actions, and from the chamber's Seek
# Pledge buttons, which is what makes "one ask per asker per member per
# resolution" true for a refusal as well as for a deal — a no is an answer.
#
# A request the recipient never answers at all (it expires unanswered) records
# nothing and may be made again. That is the engine's behaviour: no effect of
# ours runs on an expiry.
# Expects: ROOT = the asker, scope:target_country = the recipient.
un_lobby_record_attempt = {
	save_scope_as = un_lobby_asker
	scope:target_country = { save_scope_as = un_lobby_recipient }
	global_var:un_active_resolution ?= {
		if = {
			limit = { has_tag = un_res_voting }
			save_scope_as = un_lobby_ask_res
			if = {
				limit = { NOT = { un_lobby_in_list = { LIST = un_res_lobbied COUNTRY = scope:un_lobby_recipient } } }
				add_to_variable_list = { name = un_res_lobbied target = scope:un_lobby_recipient }
			}
			scope:un_lobby_recipient = {
				if = {
					limit = { NOT = { var:un_pledge_askers_res ?= scope:un_lobby_ask_res } }
					if = {
						limit = { has_variable_list = un_pledge_askers }
						clear_variable_list = un_pledge_askers
					}
					set_variable = { name = un_pledge_askers_res value = scope:un_lobby_ask_res }
				}
				if = {
					limit = {
						NOT = {
							AND = {
								has_variable_list = un_pledge_askers
								is_target_in_variable_list = { name = un_pledge_askers target = scope:un_lobby_asker }
							}
						}
					}
					add_to_variable_list = { name = un_pledge_askers target = scope:un_lobby_asker }
				}
			}
		}
	}
}

# The recipient said yes: record the pledge and hand over the obligation.
#
# Every precondition is re-checked here against the LIVE resolution rather than
# trusted from the action's `possible`. A human recipient can hold the request
# open for as long as the engine lets it stand, and in that time it may have
# voted, pledged to someone else, the resolution may have closed or lapsed,
# the asker's cap may have filled, or the asker may have been annexed. The
# engine does not re-run `possible` when the answer finally arrives, so this is
# the check that counts.
#
# The pledge and the obligation are created in the SAME `if`, so the recipient
# can never end up pledged without the asker paying, or the asker paying for a
# pledge that was not recorded.
#
# Residual edge, documented rather than fixed: if a SECOND resolution opens
# while an unanswered request from the first is still outstanding, a late
# acceptance binds the recipient on the new resolution —
# global_var:un_active_resolution is read at answer time.
# Parameters: $LIST$ = yes | no (un_res_committed_$LIST$), $DIR$ = 1 | -1
# Expects: ROOT = the asker, scope:target_country = the recipient.
un_lobby_accept_commitment = {
	save_scope_as = un_lobby_lobbyist
	scope:target_country = { save_scope_as = un_lobby_recipient }
	un_lobby_record_attempt = yes
	global_var:un_active_resolution ?= {
		if = {
			limit = {
				un_pledge_deal_available = {
					ASKER = scope:un_lobby_lobbyist
					COUNTRY = scope:un_lobby_recipient
				}
			}
			save_scope_as = un_lobby_pledge_res
			add_to_variable_list = { name = un_res_committed_$LIST$ target = scope:un_lobby_recipient }
			if = {
				limit = { NOT = { has_variable = un_res_commit_count } }
				set_variable = { name = un_res_commit_count value = 0 }
			}
			change_variable = { name = un_res_commit_count add = 1 }
			if = {
				limit = { NOT = { has_variable = un_res_commit_pending } }
				set_variable = { name = un_res_commit_pending value = 0 }
			}
			change_variable = { name = un_res_commit_pending add = 1 }
			scope:un_lobby_recipient = {
				set_variable = { name = un_pledge_to value = scope:un_lobby_lobbyist }
				set_variable = { name = un_pledge_dir value = $DIR$ }
				set_variable = { name = un_pledge_res value = scope:un_lobby_pledge_res }
			}
			scope:un_lobby_lobbyist = {
				if = {
					limit = { NOT = { var:un_pledge_count_res ?= scope:un_lobby_pledge_res } }
					set_variable = { name = un_pledge_count value = 0 }
					set_variable = { name = un_pledge_count_res value = scope:un_lobby_pledge_res }
				}
				change_variable = { name = un_pledge_count add = 1 }
				# The price, paid up front and in full. set_owes_obligation_to is
				# scoped to the country that OWES (vanilla: common/decisions/
				# france_savoy.txt:97, common/history/diplomacy/00_favors.txt:3).
				set_owes_obligation_to = {
					country = scope:un_lobby_recipient
					setting = yes
				}
			}
		}
	}
}

# ============================================================================
# THE SINGLE SETTLEMENT SITE
# ============================================================================
# Called from un_resolution_record_vote and un_resolution_record_veto
# (common/scripted_effects/un_vote_effects.txt) — between them, the only two
# places in the mod that ever book a vote onto a resolution — immediately after
# the vote goes onto its list and inside the same `has_tag = un_res_voting`
# guard, so a ballot cast after the vote closed settles nothing.
#
# IT CANNOT SETTLE THE SAME PLEDGE TWICE. Each branch is gated on "the voter is
# on a PENDING list", and the first statement inside it removes the voter from
# that list. Re-entry on any path finds both conditions false and does
# nothing.
#
# A pledge left pending when the resolution is decided, lapses, or loses its
# recipient is simply never settled: nothing is paid either way, and the
# chamber shows it as unsettled. That is the "void, no penalty" case.
#
# A pledge FOR is kept by a yes. A veto by a member pledged for is a no vote,
# so it breaks the pledge; the veto right itself is never bound. A pledge
# AGAINST is kept by a no, and a veto is a no, so a pledged-against permanent
# member that vetoes keeps its word.
#
# WHICH WAY THEY VOTED is read back off the ballot list the caller wrote one
# line earlier (reading a variable back inside the same effect block is fine;
# only add_modifier / remove_modifier results are deferred).
# Expects: scope:un_res_voter = the country whose vote has just been booked.
# Scope: resolution container (still tagged un_res_voting)
un_lobby_settle_commitment = {
	if = {
		limit = { un_lobby_in_list = { LIST = un_res_committed_yes COUNTRY = scope:un_res_voter } }
		remove_list_variable = { name = un_res_committed_yes target = scope:un_res_voter }
		un_lobby_settle_pledge = { KEPT_LIST = un_res_yes }
	}
	else_if = {
		limit = { un_lobby_in_list = { LIST = un_res_committed_no COUNTRY = scope:un_res_voter } }
		remove_list_variable = { name = un_res_committed_no target = scope:un_res_voter }
		un_lobby_settle_pledge = { KEPT_LIST = un_res_no }
	}
}

# Settle one pledge the voter has just been taken off the pending list for:
# kept if its vote is on $KEPT_LIST$, broken otherwise.
#
# The other party is the country the member pledged to (un_pledge_to), or,
# for a pledge made before this rework (which carries none), the proposer.
# Guarded because an annexed asker leaves a variable that no longer resolves —
# the member's own record still settles, only the bilateral half is skipped.
# Parameters: $KEPT_LIST$ = un_res_yes | un_res_no
# Expects: scope:un_res_voter
# Scope: resolution container
un_lobby_settle_pledge = {
	save_temporary_scope_as = un_lobby_settle_res
	if = {
		limit = { has_variable = un_res_commit_pending }
		change_variable = { name = un_res_commit_pending add = -1 }
	}
	if = {
		limit = {
			scope:un_res_voter = {
				var:un_pledge_res ?= scope:un_lobby_settle_res
				exists = var:un_pledge_to
			}
		}
		scope:un_res_voter = {
			var:un_pledge_to = { save_temporary_scope_as = un_lobby_settle_lobbyist }
		}
	}
	else = {
		var:un_res_proposer ?= { save_temporary_scope_as = un_lobby_settle_lobbyist }
	}
	if = {
		limit = { un_lobby_in_list = { LIST = $KEPT_LIST$ COUNTRY = scope:un_res_voter } }
		add_to_variable_list = { name = un_res_commit_kept target = scope:un_res_voter }
		scope:un_res_voter = {
			un_standing_gain = { AMOUNT = un_standing_commitment_kept_gain REASON = 30 }
			if = {
				limit = { exists = scope:un_lobby_settle_lobbyist }
				change_relations = {
					country = scope:un_lobby_settle_lobbyist
					value = un_standing_commitment_kept_relations
				}
			}
		}
		# The obligation stands. It was the price of the pledge, the pledge
		# was delivered, and the member redeems it whenever it likes through
		# vanilla's own redeem_obligation action.
	}
	else = {
		add_to_variable_list = { name = un_res_commit_broken target = scope:un_res_voter }
		scope:un_res_voter = {
			un_standing_loss = { AMOUNT = un_standing_commitment_broken_loss REASON = 31 }
			if = {
				limit = { exists = scope:un_lobby_settle_lobbyist }
				change_relations = {
					country = scope:un_lobby_settle_lobbyist
					value = un_standing_commitment_broken_relations
				}
			}
		}
		# The asker bought nothing, so it stops paying: the obligation it took
		# on is struck off. `setting = no` is a no-op if the member has already
		# redeemed it — a favour already called in cannot be un-called.
		if = {
			limit = { exists = scope:un_lobby_settle_lobbyist }
			scope:un_lobby_settle_lobbyist = {
				set_owes_obligation_to = {
					country = scope:un_res_voter
					setting = no
				}
			}
		}
	}
}
