# ============================================================================
# RESOLUTION LOBBYING — pledges and campaigns
# ============================================================================
# docs/superpowers/specs/2026-09-27-un-vote-lobbying-design.md. Any member may
# try to move another member's vote on the resolution in session, in either
# direction, two ways:
#
#   a PLEDGE      "Seek a Vote Commitment" (for or against), a
#                 requires_approval diplomatic action: the member may refuse;
#                 if it accepts, its lean moves ±100 and the asker owes it a
#                 vanilla diplomatic obligation, paid up front. Settled when the
#                 member's vote is booked (un_lobby_settle_commitment).
#   a CAMPAIGN    "Lobby For / Against the Resolution", a one-sided pact with
#                 influence upkeep, on an AI member only: +3 lean a month its
#                 way, up to 15; campaigns one way stack to 20 (§ CAMPAIGNS
#                 below).
#
# ---------------------------------------------------------------------------
# PLEDGE STATE
# ---------------------------------------------------------------------------
# On the resolution container:
#   lists   un_res_lobbied         every member approached on this resolution,
#                                  by anyone, accepted or declined (a record)
#           un_res_committed_yes   PENDING pledges to vote for. A member is
#                                  taken off the instant its vote is booked, so
#                                  membership means "still owes a vote"
#           un_res_committed_no    PENDING pledges to vote against, the same way
#           un_res_commit_kept     settled: voted the way it pledged
#           un_res_commit_broken   settled: did not
#   vars    un_res_commit_count    pledges ACCEPTED on the resolution, ever
#                                  (the chamber's Commitments block shows
#                                  itself on it)
#           un_res_commit_pending  unsettled pledges, for the chamber's tally
#                                  line. A variable rather than a list size
#                                  because loc can print a variable and cannot
#                                  print a list size.
# On the member that pledged (country variables, so a civil war's winner
# inherits them and continues the nation's promise):
#           un_pledge_to           the country it pledged to
#           un_pledge_dir          1 for, -1 against
#           un_pledge_res          the resolution
#           un_pledge_askers       list: who has asked it on un_pledge_askers_res
#           un_pledge_askers_res   the resolution those asks were about
# On the asker:
#           un_pledge_count        pledges it has collected on un_pledge_count_res
#           un_pledge_count_res    (un_lobby_commitment_cap = 2 per resolution)
#
# A member pledges once per resolution (the first accepted wins,
# un_pledge_made_by); an asker asks a member once per resolution
# (un_pledge_asked_by). A pledge made before this rework was always to the
# proposer and always for; it carries no un_pledge_to, and settles against the
# proposer as it always did.
#
# OLD SAVES. Nothing here assumes any of the lists or counters exist: every
# read is guarded (common/scripted_triggers/un_lobby_triggers.txt).
# ============================================================================

# Book the ask itself, whatever the answer was. Called from BOTH the accept
# and the decline branch of the pledge actions, and from the chamber's Seek
# Pledge buttons, which is what makes "one ask per asker per member per
# resolution" true for a refusal as well as for a deal — a no is an answer.
#
# A request the recipient never answers at all (it expires unanswered) records
# nothing and may be made again. That is the engine's behaviour: no effect of
# ours runs on an expiry.
# Expects: ROOT = the asker, scope:target_country = the recipient.
un_lobby_record_attempt = {
	save_scope_as = un_lobby_asker
	scope:target_country = { save_scope_as = un_lobby_recipient }
	global_var:un_active_resolution ?= {
		if = {
			limit = { has_tag = un_res_voting }
			save_scope_as = un_lobby_ask_res
			if = {
				limit = { NOT = { un_lobby_in_list = { LIST = un_res_lobbied COUNTRY = scope:un_lobby_recipient } } }
				add_to_variable_list = { name = un_res_lobbied target = scope:un_lobby_recipient }
			}
			scope:un_lobby_recipient = {
				if = {
					limit = { NOT = { var:un_pledge_askers_res ?= scope:un_lobby_ask_res } }
					if = {
						limit = { has_variable_list = un_pledge_askers }
						clear_variable_list = un_pledge_askers
					}
					set_variable = { name = un_pledge_askers_res value = scope:un_lobby_ask_res }
				}
				if = {
					limit = {
						NOT = {
							AND = {
								has_variable_list = un_pledge_askers
								is_target_in_variable_list = { name = un_pledge_askers target = scope:un_lobby_asker }
							}
						}
					}
					add_to_variable_list = { name = un_pledge_askers target = scope:un_lobby_asker }
				}
			}
		}
	}
}

# The recipient said yes: record the pledge and hand over the obligation.
#
# Every precondition is re-checked here against the LIVE resolution rather than
# trusted from the action's `possible`. A human recipient can hold the request
# open for as long as the engine lets it stand, and in that time it may have
# voted, pledged to someone else, the resolution may have closed or lapsed,
# the asker's cap may have filled, or the asker may have been annexed. The
# engine does not re-run `possible` when the answer finally arrives, so this is
# the check that counts.
#
# The pledge and the obligation are created in the SAME `if`, so the recipient
# can never end up pledged without the asker paying, or the asker paying for a
# pledge that was not recorded.
#
# Residual edge, documented rather than fixed: if a SECOND resolution opens
# while an unanswered request from the first is still outstanding, a late
# acceptance binds the recipient on the new resolution —
# global_var:un_active_resolution is read at answer time.
# Parameters: $LIST$ = yes | no (un_res_committed_$LIST$), $DIR$ = 1 | -1
# Expects: ROOT = the asker, scope:target_country = the recipient.
un_lobby_accept_commitment = {
	save_scope_as = un_lobby_lobbyist
	scope:target_country = { save_scope_as = un_lobby_recipient }
	un_lobby_record_attempt = yes
	global_var:un_active_resolution ?= {
		if = {
			limit = {
				un_pledge_deal_available = {
					ASKER = scope:un_lobby_lobbyist
					COUNTRY = scope:un_lobby_recipient
				}
			}
			save_scope_as = un_lobby_pledge_res
			add_to_variable_list = { name = un_res_committed_$LIST$ target = scope:un_lobby_recipient }
			if = {
				limit = { NOT = { has_variable = un_res_commit_count } }
				set_variable = { name = un_res_commit_count value = 0 }
			}
			change_variable = { name = un_res_commit_count add = 1 }
			if = {
				limit = { NOT = { has_variable = un_res_commit_pending } }
				set_variable = { name = un_res_commit_pending value = 0 }
			}
			change_variable = { name = un_res_commit_pending add = 1 }
			scope:un_lobby_recipient = {
				set_variable = { name = un_pledge_to value = scope:un_lobby_lobbyist }
				set_variable = { name = un_pledge_dir value = $DIR$ }
				set_variable = { name = un_pledge_res value = scope:un_lobby_pledge_res }
			}
			scope:un_lobby_lobbyist = {
				if = {
					limit = { NOT = { var:un_pledge_count_res ?= scope:un_lobby_pledge_res } }
					set_variable = { name = un_pledge_count value = 0 }
					set_variable = { name = un_pledge_count_res value = scope:un_lobby_pledge_res }
				}
				change_variable = { name = un_pledge_count add = 1 }
				# The price, paid up front and in full. set_owes_obligation_to is
				# scoped to the country that OWES (vanilla: common/decisions/
				# france_savoy.txt:97, common/history/diplomacy/00_favors.txt:3).
				set_owes_obligation_to = {
					country = scope:un_lobby_recipient
					setting = yes
				}
			}
		}
	}
}

# ============================================================================
# THE SINGLE SETTLEMENT SITE
# ============================================================================
# Called from un_resolution_record_vote and un_resolution_record_veto
# (common/scripted_effects/un_vote_effects.txt) — between them, the only two
# places in the mod that ever book a vote onto a resolution — immediately after
# the vote goes onto its list and inside the same `has_tag = un_res_voting`
# guard, so a ballot cast after the vote closed settles nothing.
#
# IT CANNOT SETTLE THE SAME PLEDGE TWICE. Each branch is gated on "the voter is
# on a PENDING list", and the first statement inside it removes the voter from
# that list. Re-entry on any path finds both conditions false and does
# nothing.
#
# A pledge left pending when the resolution is decided, lapses, or loses its
# recipient is simply never settled: nothing is paid either way, and the
# chamber shows it as unsettled. That is the "void, no penalty" case.
#
# A pledge FOR is kept by a yes. A veto by a member pledged for is a no vote,
# so it breaks the pledge; the veto right itself is never bound. A pledge
# AGAINST is kept by a no, and a veto is a no, so a pledged-against permanent
# member that vetoes keeps its word.
#
# WHICH WAY THEY VOTED is read back off the ballot list the caller wrote one
# line earlier (reading a variable back inside the same effect block is fine;
# only add_modifier / remove_modifier results are deferred).
# Expects: scope:un_res_voter = the country whose vote has just been booked.
# Scope: resolution container (still tagged un_res_voting)
un_lobby_settle_commitment = {
	if = {
		limit = { un_lobby_in_list = { LIST = un_res_committed_yes COUNTRY = scope:un_res_voter } }
		remove_list_variable = { name = un_res_committed_yes target = scope:un_res_voter }
		un_lobby_settle_pledge = { KEPT_LIST = un_res_yes }
	}
	else_if = {
		limit = { un_lobby_in_list = { LIST = un_res_committed_no COUNTRY = scope:un_res_voter } }
		remove_list_variable = { name = un_res_committed_no target = scope:un_res_voter }
		un_lobby_settle_pledge = { KEPT_LIST = un_res_no }
	}
}

# Settle one pledge the voter has just been taken off the pending list for:
# kept if its vote is on $KEPT_LIST$, broken otherwise.
#
# The other party is the country the member pledged to (un_pledge_to), or,
# for a pledge made before this rework (which carries none), the proposer.
# Guarded because an annexed asker leaves a variable that no longer resolves —
# the member's own record still settles, only the bilateral half is skipped.
# Parameters: $KEPT_LIST$ = un_res_yes | un_res_no
# Expects: scope:un_res_voter
# Scope: resolution container
un_lobby_settle_pledge = {
	save_temporary_scope_as = un_lobby_settle_res
	if = {
		limit = { has_variable = un_res_commit_pending }
		change_variable = { name = un_res_commit_pending add = -1 }
	}
	if = {
		limit = {
			scope:un_res_voter = {
				var:un_pledge_res ?= scope:un_lobby_settle_res
				exists = var:un_pledge_to
			}
		}
		scope:un_res_voter = {
			var:un_pledge_to = { save_temporary_scope_as = un_lobby_settle_lobbyist }
		}
	}
	else = {
		var:un_res_proposer ?= { save_temporary_scope_as = un_lobby_settle_lobbyist }
	}
	if = {
		limit = { un_lobby_in_list = { LIST = $KEPT_LIST$ COUNTRY = scope:un_res_voter } }
		add_to_variable_list = { name = un_res_commit_kept target = scope:un_res_voter }
		scope:un_res_voter = {
			un_standing_gain = { AMOUNT = un_standing_commitment_kept_gain REASON = 30 }
			if = {
				limit = { exists = scope:un_lobby_settle_lobbyist }
				change_relations = {
					country = scope:un_lobby_settle_lobbyist
					value = un_standing_commitment_kept_relations
				}
			}
		}
		# The obligation stands. It was the price of the pledge, the pledge
		# was delivered, and the member redeems it whenever it likes through
		# vanilla's own redeem_obligation action.
	}
	else = {
		add_to_variable_list = { name = un_res_commit_broken target = scope:un_res_voter }
		scope:un_res_voter = {
			un_standing_loss = { AMOUNT = un_standing_commitment_broken_loss REASON = 31 }
			if = {
				limit = { exists = scope:un_lobby_settle_lobbyist }
				change_relations = {
					country = scope:un_lobby_settle_lobbyist
					value = un_standing_commitment_broken_relations
				}
			}
		}
		# The asker bought nothing, so it stops paying: the obligation it took
		# on is struck off. `setting = no` is a no-op if the member has already
		# redeemed it — a favour already called in cannot be un-called.
		if = {
			limit = { exists = scope:un_lobby_settle_lobbyist }
			scope:un_lobby_settle_lobbyist = {
				set_owes_obligation_to = {
					country = scope:un_res_voter
					setting = no
				}
			}
		}
	}
}

# ============================================================================
# CAMPAIGNS (spec §3)
# ============================================================================
# A campaign is a one-sided pact, un_lobby_for_action or
# un_lobby_against_action (common/diplomatic_actions/un_lobbying.txt), from any
# member with a vote to an AI member that has not voted, with influence upkeep.
# The pact is the source of truth for whether a campaign runs; a script
# container mirrors each one to count its months:
#
#   tags    un_lobby_campaign, un_lc_for | un_lc_against
#   parent  the lobbyist — the engine culls the container if it stops existing
#   vars    un_lc_lobbyist   the lobbying country
#           un_lc_member     the member lobbied
#           un_lc_months     full months run (the first pulse only arms it:
#                            un_lc_armed)
#           un_lc_shift      its lean shift: 3 a month, up to 15
#                            (un_lobby_campaign_shift_value)
#   listed  on the resolution's un_res_campaigns
#
# Each month (un_lobby_campaigns_monthly, from the global UN pulse, before the
# leans are read): containers whose pact is gone are reaped, pacts without a
# container are adopted, the live ones age, and each member's net shift is
# written as country variables the lean reads (un_lean_lobbying):
#   un_lobby_for_shift, un_lobby_against_shift   each capped at 20
#   un_lobby_campaigns_on                        campaigns on it (the chamber
#                                                orders worked members first)
#   un_lobby_shift_res                           the resolution they belong to
#
# A campaign starts from the pact's accept_effect (the diplomacy panel), from
# the chamber's Lobby buttons and from the AI (un_lobby_ai_monthly), which
# create the pact and then call un_lobby_campaign_start themselves; the start
# is idempotent, so a path that runs both is harmless. It ends when the pact
# breaks (the break hooks call un_lobby_campaign_end, which drops the
# campaign's shift at once), when its member votes (un_lobby_campaigns_end_on,
# from un_resolution_record_vote / _veto) and when the resolution closes
# (un_lobby_campaigns_close, from un_resolution_archive).
#
# Every removal is a `while` over "one matching container left", taking it off
# the list before destroying it, so no list is changed while it is iterated
# and the list never holds a dead reference of our making.
# ============================================================================

# Start a $DIR$ campaign by the scoped lobbyist on scope:target_country, on the
# resolution in session, replacing a container already there for the same
# lobbyist, member and direction. Leaves scope:un_lc_new at the container.
# Callers guard any use of scope:un_lc_new with `exists`: when an action's
# tooltip renders its accept_effect the container is not created.
# Parameters: $DIR$ = for | against
# Scope: the lobbyist; scope:target_country = the member
un_lobby_campaign_start = {
	save_scope_as = un_lc_lobbyist_new
	scope:target_country = { save_scope_as = un_lc_member_new }
	global_var:un_active_resolution ?= {
		if = {
			limit = { has_tag = un_res_voting }
			save_scope_as = un_lc_start_res
			un_lobby_campaign_drop = {
				DIR = $DIR$
				LOBBYIST = scope:un_lc_lobbyist_new
				MEMBER = scope:un_lc_member_new
			}
		}
	}
	if = {
		limit = { scope:un_lc_start_res ?= { has_tag = un_res_voting } }
		create_container = {
			tags = { un_lobby_campaign un_lc_$DIR$ }
			parent = scope:un_lc_lobbyist_new
			save_scope_as = un_lc_new
		}
		if = {
			limit = { exists = scope:un_lc_new }
			scope:un_lc_new = {
				set_variable = { name = un_lc_lobbyist value = scope:un_lc_lobbyist_new }
				set_variable = { name = un_lc_member value = scope:un_lc_member_new }
				set_variable = { name = un_lc_months value = 0 }
				set_variable = { name = un_lc_shift value = 0 }
			}
			scope:un_lc_start_res = {
				add_to_variable_list = { name = un_res_campaigns target = scope:un_lc_new }
			}
		}
	}
}

# Take every $DIR$ campaign by $LOBBYIST$ on $MEMBER$ off the resolution and
# destroy it. Does not touch the pact.
# Parameters: $DIR$, $LOBBYIST$, $MEMBER$ (scopes)
# Scope: resolution container
un_lobby_campaign_drop = {
	while = {
		limit = {
			has_variable_list = un_res_campaigns
			any_in_list = {
				variable = un_res_campaigns
				has_tag = un_lc_$DIR$
				var:un_lc_lobbyist ?= $LOBBYIST$
				var:un_lc_member ?= $MEMBER$
			}
		}
		random_in_list = {
			variable = un_res_campaigns
			limit = {
				has_tag = un_lc_$DIR$
				var:un_lc_lobbyist ?= $LOBBYIST$
				var:un_lc_member ?= $MEMBER$
			}
			save_scope_as = un_lc_drop
		}
		remove_list_variable = { name = un_res_campaigns target = scope:un_lc_drop }
		scope:un_lc_drop = { destroy_container = yes }
	}
}

# The pact broke: manually (cancelled from the diplomacy panel or the chamber,
# or removed by script) or automatically (its requirement failed). Drop the
# campaign and rewrite its member's shift at once, so a cancelled campaign's
# gain is gone even if the member votes before the next monthly pulse (the
# last call can land mid-month). If scope:target_country no longer resolves,
# the monthly reap collects the container instead.
# Parameters: $DIR$ = for | against
# Scope: the lobbyist (the pact's ROOT); scope:target_country = the member
un_lobby_campaign_end = {
	hidden_effect = {
		save_scope_as = un_lc_end_lobbyist
		if = {
			limit = { exists = scope:target_country }
			scope:target_country = { save_scope_as = un_lc_end_member }
			global_var:un_active_resolution ?= {
				un_lobby_campaign_drop = {
					DIR = $DIR$
					LOBBYIST = scope:un_lc_end_lobbyist
					MEMBER = scope:un_lc_end_member
				}
				un_lobby_member_recompute = { MEMBER = scope:un_lc_end_member }
			}
		}
	}
}

# Remove the pact behind the scoped campaign (the container is left to the
# caller). Guarded: a lobbyist or member that stopped existing has no pact.
# Scope: campaign container
un_lobby_campaign_remove_pact = {
	if = {
		limit = {
			exists = var:un_lc_lobbyist
			exists = var:un_lc_member
		}
		var:un_lc_member = { save_temporary_scope_as = un_lc_rm_member }
		if = {
			limit = { has_tag = un_lc_for }
			var:un_lc_lobbyist = {
				remove_diplomatic_pact = { country = scope:un_lc_rm_member type = un_lobby_for_action }
			}
		}
		else = {
			var:un_lc_lobbyist = {
				remove_diplomatic_pact = { country = scope:un_lc_rm_member type = un_lobby_against_action }
			}
		}
	}
}

# $MEMBER$ has voted: every campaign on it ends, pact and all, the same day
# (the pact's own requirement would break it a tick later; this stops the
# influence the day the ballot is cast). Its shift no longer matters.
# Parameters: $MEMBER$ = the member (a scope)
# Scope: resolution container
un_lobby_campaigns_end_on = {
	while = {
		limit = {
			has_variable_list = un_res_campaigns
			any_in_list = {
				variable = un_res_campaigns
				has_tag = un_lobby_campaign
				var:un_lc_member ?= $MEMBER$
			}
		}
		random_in_list = {
			variable = un_res_campaigns
			limit = {
				has_tag = un_lobby_campaign
				var:un_lc_member ?= $MEMBER$
			}
			save_scope_as = un_lc_stop
		}
		remove_list_variable = { name = un_res_campaigns target = scope:un_lc_stop }
		scope:un_lc_stop = {
			un_lobby_campaign_remove_pact = yes
			destroy_container = yes
		}
	}
}

# The resolution closes (decided, lapsed, vetoed): every campaign on it ends.
# Then the list is emptied of anything left (a container the engine culled with
# its lobbyist).
# Scope: resolution container
un_lobby_campaigns_close = {
	while = {
		limit = {
			has_variable_list = un_res_campaigns
			any_in_list = {
				variable = un_res_campaigns
				has_tag = un_lobby_campaign
			}
		}
		random_in_list = {
			variable = un_res_campaigns
			limit = { has_tag = un_lobby_campaign }
			save_scope_as = un_lc_stop
		}
		remove_list_variable = { name = un_res_campaigns target = scope:un_lc_stop }
		scope:un_lc_stop = {
			un_lobby_campaign_remove_pact = yes
			destroy_container = yes
		}
	}
	if = {
		limit = { has_variable_list = un_res_campaigns }
		clear_variable_list = un_res_campaigns
	}
}

# Monthly campaign upkeep, while the resolution votes. From
# un_resolutions_monthly_update, before the leans are read. NOT IDEMPOTENT
# WITHIN A MONTH: campaigns age on every call.
# Scope: resolution container (= scope:un_resolution)
un_lobby_campaigns_monthly = {
	save_temporary_scope_as = un_lc_res_t
	un_lobby_campaigns_reap = yes
	un_lobby_campaigns_adopt = yes
	if = {
		limit = { has_variable_list = un_res_campaigns }
		every_in_list = {
			variable = un_res_campaigns
			limit = { has_tag = un_lobby_campaign }
			if = {
				limit = { has_variable = un_lc_armed }
				change_variable = { name = un_lc_months add = 1 }
			}
			else = {
				set_variable = un_lc_armed
			}
			set_variable = { name = un_lc_shift value = un_lobby_campaign_shift_value }
		}
	}
	un_lobby_campaigns_aggregate = yes
}

# Destroy the campaigns whose pact is gone: broken by a path whose hook did not
# reach the container (its member annexed, the pact removed by the engine), or
# a lobbyist or member that stopped existing. A live pact is found by its
# direction — first_country is the lobbyist, second_country the member — so a
# pact of the same type the other way round never keeps a dead campaign alive.
# Scope: resolution container
un_lobby_campaigns_reap = {
	if = {
		limit = { has_variable_list = un_res_campaigns }
		every_in_list = {
			variable = un_res_campaigns
			limit = { has_tag = un_lobby_campaign }
			if = {
				limit = {
					OR = {
						NOT = { exists = var:un_lc_lobbyist }
						NOT = { exists = var:un_lc_member }
					}
				}
				set_variable = un_lc_dead
			}
			else = {
				var:un_lc_member = { save_temporary_scope_as = un_lc_m }
				if = {
					limit = {
						has_tag = un_lc_for
						var:un_lc_lobbyist = {
							NOT = {
								any_scope_diplomatic_pact = {
									is_diplomatic_action_type = un_lobby_for_action
									first_country = prev # REVIEWED 2026-09-27: prev is var:un_lc_lobbyist on purpose — the pact must run from the lobbyist to the member
									second_country = scope:un_lc_m
								}
							}
						}
					}
					set_variable = un_lc_dead
				}
				else_if = {
					limit = {
						has_tag = un_lc_against
						var:un_lc_lobbyist = {
							NOT = {
								any_scope_diplomatic_pact = {
									is_diplomatic_action_type = un_lobby_against_action
									first_country = prev # REVIEWED 2026-09-27: prev is var:un_lc_lobbyist on purpose — the pact must run from the lobbyist to the member
									second_country = scope:un_lc_m
								}
							}
						}
					}
					set_variable = un_lc_dead
				}
			}
		}
	}
	while = {
		limit = {
			has_variable_list = un_res_campaigns
			any_in_list = {
				variable = un_res_campaigns
				has_tag = un_lobby_campaign
				has_variable = un_lc_dead
			}
		}
		random_in_list = {
			variable = un_res_campaigns
			limit = {
				has_tag = un_lobby_campaign
				has_variable = un_lc_dead
			}
			save_scope_as = un_lc_reaped
		}
		remove_list_variable = { name = un_res_campaigns target = scope:un_lc_reaped }
		scope:un_lc_reaped = { destroy_container = yes }
	}
}

# Give a container to every campaign pact that has none: a pact whose start
# never ran (a path we do not know of). It starts at month 0.
# Scope: resolution container; scope:un_lc_res_t = it
un_lobby_campaigns_adopt = {
	every_country = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		save_temporary_scope_as = un_lca_lobbyist
		every_scope_diplomatic_pact = {
			limit = {
				is_diplomatic_action_type = un_lobby_for_action
				first_country = prev
			}
			second_country = { save_temporary_scope_as = un_lca_member }
			un_lobby_campaign_adopt_one = { DIR = for }
		}
		every_scope_diplomatic_pact = {
			limit = {
				is_diplomatic_action_type = un_lobby_against_action
				first_country = prev
			}
			second_country = { save_temporary_scope_as = un_lca_member }
			un_lobby_campaign_adopt_one = { DIR = against }
		}
	}
}

# Parameters: $DIR$ = for | against
# Scope: any; scope:un_lca_lobbyist, scope:un_lca_member, scope:un_lc_res_t
un_lobby_campaign_adopt_one = {
	if = {
		limit = {
			scope:un_lc_res_t = {
				NOT = {
					AND = {
						has_variable_list = un_res_campaigns
						any_in_list = {
							variable = un_res_campaigns
							has_tag = un_lc_$DIR$
							var:un_lc_lobbyist ?= scope:un_lca_lobbyist
							var:un_lc_member ?= scope:un_lca_member
						}
					}
				}
			}
		}
		scope:un_lca_member = { save_scope_as = target_country }
		scope:un_lca_lobbyist = { un_lobby_campaign_start = { DIR = $DIR$ } }
	}
}

# Rewrite every member's shift from the live campaigns: clear last month's,
# add each campaign's shift to its member, cap each direction at the stack cap.
# Scope: resolution container; scope:un_lc_res_t = it
un_lobby_campaigns_aggregate = {
	every_country = {
		limit = { has_variable = un_lobby_shift_res }
		un_lobby_member_shift_clear = yes
	}
	if = {
		limit = { has_variable_list = un_res_campaigns }
		every_in_list = {
			variable = un_res_campaigns
			limit = {
				has_tag = un_lobby_campaign
				exists = var:un_lc_member
			}
			un_lobby_campaign_add_to_member = yes
		}
	}
	every_country = {
		limit = { var:un_lobby_shift_res ?= scope:un_lc_res_t }
		clamp_variable = { name = un_lobby_for_shift min = 0 max = un_lobby_campaign_stack_cap }
		clamp_variable = { name = un_lobby_against_shift min = 0 max = un_lobby_campaign_stack_cap }
	}
}

# Rewrite one member's shift (a campaign on it has just ended).
# Parameters: $MEMBER$ = the member (a scope)
# Scope: resolution container
un_lobby_member_recompute = {
	save_temporary_scope_as = un_lc_res_t
	$MEMBER$ = { un_lobby_member_shift_clear = yes }
	if = {
		limit = { has_variable_list = un_res_campaigns }
		every_in_list = {
			variable = un_res_campaigns
			limit = {
				has_tag = un_lobby_campaign
				var:un_lc_member ?= $MEMBER$
			}
			un_lobby_campaign_add_to_member = yes
		}
	}
	$MEMBER$ = {
		if = {
			limit = { var:un_lobby_shift_res ?= scope:un_lc_res_t }
			clamp_variable = { name = un_lobby_for_shift min = 0 max = un_lobby_campaign_stack_cap }
			clamp_variable = { name = un_lobby_against_shift min = 0 max = un_lobby_campaign_stack_cap }
		}
	}
}

# Scope: a member country
un_lobby_member_shift_clear = {
	if = {
		limit = { has_variable = un_lobby_shift_res }
		remove_variable = un_lobby_shift_res
		remove_variable = un_lobby_for_shift
		remove_variable = un_lobby_against_shift
		remove_variable = un_lobby_campaigns_on
	}
}

# Add the scoped campaign's shift to its member, writing the member's shift
# variables first if this is the first campaign counted on it. Accumulates
# with set_variable's inline value block: change_variable does not reliably
# resolve a script-value operand.
# Scope: campaign container; scope:un_lc_res_t = its resolution
un_lobby_campaign_add_to_member = {
	save_temporary_scope_as = un_lc_add
	var:un_lc_member ?= {
		if = {
			limit = { NOT = { var:un_lobby_shift_res ?= scope:un_lc_res_t } }
			set_variable = { name = un_lobby_shift_res value = scope:un_lc_res_t }
			set_variable = { name = un_lobby_for_shift value = 0 }
			set_variable = { name = un_lobby_against_shift value = 0 }
			set_variable = { name = un_lobby_campaigns_on value = 0 }
		}
		if = {
			limit = { scope:un_lc_add = { has_tag = un_lc_for } }
			set_variable = { name = un_lobby_for_shift value = { value = var:un_lobby_for_shift add = scope:un_lc_add.var:un_lc_shift } }
		}
		else = {
			set_variable = { name = un_lobby_against_shift value = { value = var:un_lobby_against_shift add = scope:un_lc_add.var:un_lc_shift } }
		}
		change_variable = { name = un_lobby_campaigns_on add = 1 }
	}
}
