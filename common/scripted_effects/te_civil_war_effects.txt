# ============================================================================
# CIVIL WARS — shared bookkeeping for every system's repair
# ============================================================================
# The cross-cutting repair site docs/audits/civil_war_inheritance_audit.md
# proposed, built once (docs/superpowers/specs/2026-09-26-nuclear-custody-design.md
# §3.1). The hooks are in common/on_actions/te_civil_war_on_actions.txt. A
# system that must repair what a revolution's winner loses (#460–#465) adds
# its call to te_civil_war_on_won there, beside nd_custody_on_civil_war_won,
# and reads what te_civil_war_resolve_sides leaves:
#   scope:te_cw_winner   ROOT, the winner
#   scope:te_cw_loser    the loser, when it still resolves (see below)
#   var:te_cw_rebels_won 1 when the revolutionaries won — a new regime
#   var:te_cw_government_won 1 when the continuing government won
#
# Variables:
#   te_cw_origin   on the rebel from the outbreak: the country it rose against;
#                  re-pointed to its continuation when another revolution wins
#   te_cw_role     1 on the original country, 2 on the rebel. Used only by the
#                  readability probe and legacy repairs; concurrent wars can
#                  clear or inherit it, so outcome flags decide which side won
#   global te_cw_ending_origin / te_cw_ending_rebel / te_cw_ending_kind
#                  (1 revolution, 2 secession): the pair on_revolution_end /
#                  on_secession_end reported, consumed by on_civil_war_won,
#                  which vanilla fires straight after on_revolution_end
#
# THE ENGINE QUESTION, ANSWERED 2026-09-26. A dead-but-undeleted loser still
# resolves through a stored pointer at on_civil_war_won, and script reads its
# variables. Five endings in one test game (three rebel wins, two loyalist
# wins) all logged the loser as resolving and dead with te_cw_role present,
# and each rebel win passed the monetary repair's canary, a script comparison
# of the dead loser's rate paid (te_monetary_civil_war_effects.txt).
# te_civil_war_resolve_sides keeps logging each ending (TE_CIVIL_WAR: lines in
# debug.log). The nuclear custody code still reads live ledger containers.
# ============================================================================

# ROOT = the original country, scope:target = the uprising country.
te_civil_war_record_start = {
	scope:target = {
		set_variable = { name = te_cw_origin value = ROOT }
		set_variable = { name = te_cw_role value = 2 }
	}
	set_variable = { name = te_cw_role value = 1 }
	debug_log = "TE_CIVIL_WAR: [THIS.GetCountry.GetNameNoFormatting] faces an uprising"
}

# ROOT = the original country, scope:target = the uprising country. KIND = 1
# for a revolution, 2 for a secession.
te_civil_war_record_ending = {
	set_global_variable = { name = te_cw_ending_origin value = ROOT }
	set_global_variable = { name = te_cw_ending_kind value = $KIND$ }
	if = {
		limit = { exists = scope:target }
		set_global_variable = { name = te_cw_ending_rebel value = scope:target }
	}
	# The kind is read back from the global, not printed from the parameter:
	# a parameter inside a debug_log string comes out mangled.
	if = {
		limit = { global_var:te_cw_ending_kind = 1 }
		debug_log = "TE_CIVIL_WAR: a revolution ended"
	}
	else = {
		debug_log = "TE_CIVIL_WAR: a secession ended"
	}
	if = {
		limit = { is_country_alive = yes }
		debug_log = "TE_CIVIL_WAR: the original country [THIS.GetCountry.GetNameNoFormatting] is alive"
	}
	else = {
		debug_log = "TE_CIVIL_WAR: the original country [THIS.GetCountry.GetNameNoFormatting] is dead"
	}
	if = {
		limit = { scope:target ?= { is_country_alive = yes } }
		debug_log = "TE_CIVIL_WAR: the uprising country is alive"
	}
	else_if = {
		limit = { exists = scope:target }
		debug_log = "TE_CIVIL_WAR: the uprising country is dead but resolves"
	}
	else = {
		debug_log = "TE_CIVIL_WAR: the uprising country does not resolve"
	}
}

# ROOT = the winner (on_civil_war_won). Works out which side won and saves the
# loser if it resolves. The pair identifies the rebel, but its te_cw_origin
# follows the continuing government if the engine still reports a dead original.
# Without a matching pair, the rebel's own te_cw_origin does (on a loyalist winner the merge hands it the rebel's
# te_cw_origin, which then names the winner itself).
te_civil_war_resolve_sides = {
	save_scope_as = te_cw_winner
	set_variable = { name = te_cw_rebels_won value = 0 }
	set_variable = { name = te_cw_government_won value = 0 }
	if = {
		limit = { global_var:te_cw_ending_rebel ?= ROOT }
		set_variable = { name = te_cw_rebels_won value = 1 }
		if = {
			limit = {
				exists = var:te_cw_origin
				NOT = { var:te_cw_origin ?= ROOT }
			}
			var:te_cw_origin ?= { save_scope_as = te_cw_loser }
		}
		else = {
			global_var:te_cw_ending_origin ?= { save_scope_as = te_cw_loser }
		}
		debug_log = "TE_CIVIL_WAR: [THIS.GetCountry.GetNameNoFormatting] won as the uprising (end-hook pair)"
	}
	else_if = {
		limit = {
			OR = {
				global_var:te_cw_ending_origin ?= ROOT
				global_var:te_cw_ending_rebel ?= { var:te_cw_origin ?= ROOT }
			}
		}
		set_variable = { name = te_cw_government_won value = 1 }
		global_var:te_cw_ending_rebel ?= { save_scope_as = te_cw_loser }
		debug_log = "TE_CIVIL_WAR: [THIS.GetCountry.GetNameNoFormatting] won as the original government (end-hook pair)"
	}
	else_if = {
		limit = {
			has_variable = te_cw_origin
			NOT = { var:te_cw_origin ?= ROOT }
		}
		set_variable = { name = te_cw_rebels_won value = 1 }
		var:te_cw_origin ?= { save_scope_as = te_cw_loser }
		debug_log = "TE_CIVIL_WAR: [THIS.GetCountry.GetNameNoFormatting] won as the uprising (no matching end-hook pair; te_cw_origin)"
	}
	else = {
		# Legacy ending without a matching pair: a self-origin is inherited
		# from defeated rebels; role 1 preserves the old government repair.
		if = {
			limit = {
				OR = {
					var:te_cw_origin ?= ROOT
					var:te_cw_role = 1
				}
			}
			set_variable = { name = te_cw_government_won value = 1 }
		}
		debug_log = "TE_CIVIL_WAR: [THIS.GetCountry.GetNameNoFormatting] won; no matching end-hook pair, taken as the original government"
	}
	# A secession's winner is never a new regime of the same state.
	if = {
		limit = {
			has_global_variable = te_cw_ending_kind
			global_var:te_cw_ending_kind = 2
		}
		set_variable = { name = te_cw_rebels_won value = 0 }
	}
	# ---- the engine question: does the loser still resolve, and read? -------
	if = {
		limit = { exists = scope:te_cw_loser }
		if = {
			limit = { scope:te_cw_loser = { is_country_alive = yes } }
			debug_log = "TE_CIVIL_WAR: the loser resolves at on_civil_war_won and is alive"
		}
		else = {
			debug_log = "TE_CIVIL_WAR: the loser resolves at on_civil_war_won and is dead"
		}
		# Read in script: the loc accessor .Var().GetValue printed 0 here even
		# though te_cw_role was set (it is never 0).
		if = {
			limit = { scope:te_cw_loser = { has_variable = te_cw_role } }
			if = {
				limit = { scope:te_cw_loser = { var:te_cw_role = 1 } }
				debug_log = "TE_CIVIL_WAR: the loser's variables read: te_cw_role = 1 (the original government)"
			}
			else_if = {
				limit = { scope:te_cw_loser = { var:te_cw_role = 2 } }
				debug_log = "TE_CIVIL_WAR: the loser's variables read: te_cw_role = 2 (the uprising)"
			}
			else = {
				debug_log = "TE_CIVIL_WAR: the loser's variables read, but te_cw_role is neither 1 nor 2"
			}
		}
		else = {
			debug_log = "TE_CIVIL_WAR: the loser resolves but its te_cw_role does not read"
		}
	}
	else = {
		debug_log = "TE_CIVIL_WAR: the loser does not resolve at on_civil_war_won"
	}
}

# ROOT = the revolutionary winner, after repairs and before its origin is cleared.
# Every live uprising against the defeated government now faces its continuation.
# every_country visits live countries; exclude ROOT, which can carry the same pointer.
te_civil_war_repoint_uprisings = {
	if = {
		limit = {
			var:te_cw_rebels_won = 1
			exists = scope:te_cw_loser
		}
		every_country = {
			limit = {
				NOT = { this = ROOT }
				var:te_cw_origin ?= scope:te_cw_loser
			}
			set_variable = { name = te_cw_origin value = ROOT }
		}
	}
}

# ROOT = the winner, after every system's repair.
te_civil_war_clear = {
	if = {
		limit = { has_variable = te_cw_government_won }
		remove_variable = te_cw_government_won
	}
	if = {
		limit = { has_variable = te_cw_rebels_won }
		remove_variable = te_cw_rebels_won
	}
	if = {
		limit = { has_variable = te_cw_origin }
		remove_variable = te_cw_origin
	}
	if = {
		limit = { has_variable = te_cw_role }
		remove_variable = te_cw_role
	}
	te_civil_war_clear_ending = yes
}

te_civil_war_clear_ending = {
	if = {
		limit = { has_global_variable = te_cw_ending_origin }
		remove_global_variable = te_cw_ending_origin
	}
	if = {
		limit = { has_global_variable = te_cw_ending_rebel }
		remove_global_variable = te_cw_ending_rebel
	}
	if = {
		limit = { has_global_variable = te_cw_ending_kind }
		remove_global_variable = te_cw_ending_kind
	}
}
