# Event-agency package E2 ("misc") — scripted effects.
# Each block names the event(s) that call it.

# ----------------------------------------------------------------------------
# banking_cycle_events.45 — the donor answers an appeal
# ----------------------------------------------------------------------------
# ROOT = the donor, scope:bailout_country = the country that asked
# (banking_cycle_events.68). Records the answer on the asker and sends it
# banking_cycle_events.69, so the treasury transfer (or the refusal's
# relations hit) arrives with an explanation.
#
# Each option calls one of these last, after its own effects.
# te_ea_bailout_answer_aid takes the AMOUNT and DAYS the option passed to
# banking_transfer_emergency_aid (legacy_event_integration_effects.txt) and
# repeats its arithmetic (root.$AMOUNT$ x $DAYS$ / 14) so the notice can name
# the sum; ANSWER is 1 for the rescue package, 2 for the smaller grant.
te_ea_bailout_answer_aid = {
	scope:bailout_country = {
		set_variable = { name = te_ea_bailout_answer value = $ANSWER$ days = 30 }
		set_variable = {
			name = te_ea_bailout_amount
			value = {
				value = root.$AMOUNT$
				multiply = $DAYS$
				divide = 14
			}
			days = 30
		}
		trigger_event = { id = banking_cycle_events.69 }
	}
}

te_ea_bailout_answer_refusal = {
	scope:bailout_country = {
		set_variable = { name = te_ea_bailout_answer value = 3 days = 30 }
		trigger_event = { id = banking_cycle_events.69 }
	}
}
