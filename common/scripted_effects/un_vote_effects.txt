# ============================================================================
# UN RESOLUTIONS — script containers (1.13.10+)
# ============================================================================
# Every General Assembly resolution is a script container:
#
#   tags    un_resolution, un_topic_<topic>
#           un_res_voting           while the vote is open; replaced on close by
#           un_res_passed / un_res_failed / un_res_lapsed (lapsed = un_vote.2
#                                   never fired, e.g. the proposer left the UN)
#           un_res_vetoed           a permanent member vetoed (binding topics)
#           un_res_target_accepted  the target voted in favor of its own censure
#   parent  none — a resolution must outlive a proposer annexed mid-vote
#   vars    un_res_seq        creation order (global counter un_resolution_seq)
#           un_res_proposer   proposing country
#           un_res_target     target country, when the topic names one
#           un_res_vetoer     the last permanent member to veto
#           un_res_promoted   expulsion only: country promoted into the vacated seat
#           un_res_support    yes votes (starts at 1: the proposer)
#           un_res_oppose     no votes (a veto also counts as no)
#           un_res_months     months open (lapse watchdog)
#           un_res_cooldown_months  set on close, counted down monthly and
#                             removed at 0: the topic can't be proposed again
#                             while any closed resolution of it carries it
#                             (military_mandate opts out — see un_buttons.txt).
#                             A month count, not a timed variable: the engine
#                             never expires a timed variable on a container.
#                             Saves from before this carry a frozen timed
#                             un_res_cooldown that nothing reads any more.
#           un_res_mandate_region       military_mandate only: the state region
#                             the resolution would authorize recovering
#           un_res_mandate_beneficiary  military_mandate only: who receives it
#                             (v1: always the proposer)
#   lists   un_res_yes, un_res_no  voters (the proposer is in un_res_yes)
#
# Resolution lobbying adds four more lists (un_res_lobbied,
# un_res_committed_yes, un_res_commit_kept, un_res_commit_broken) and two more
# variables (un_res_commit_count, un_res_commit_pending) to the same container.
# They are documented where they are written, at the top of
# common/scripted_effects/un_lobby_effects.txt, and need nothing from the
# lifecycle here: they are archived with the resolution and destroyed with it.
#
# Globals:
#   un_vote_active         the vote lock: one resolution (or reservation) at a time
#   un_vote_reservation    a proposer event holds the lock before choosing to propose
#   un_active_resolution   the open resolution
#   un_resolution_history  closed resolutions, capped at un_resolution_history_cap
#   un_resolution_seq      counter behind un_res_seq
#
# The vote events receive the resolution as scope:un_resolution through
# trigger_event (saved scopes survive delayed events; vanilla red_scare.12
# fires red_scare.13 three years later and reads its saved culture). They never
# read un_active_resolution, so a resolution stays readable after a newer one
# opens — the lock is the only thing that keeps votes one at a time.
#
# Old saves: a vote in progress when the mod updated is dropped. Its queued
# un_vote.1/.2 events carry no scope:un_resolution and fail their triggers,
# and un_resolutions_monthly_update frees the orphaned lock. The legacy
# un_vote_* globals are left inert (removing them would log "used but never
# set" on every load).
# ============================================================================

# Open a General Assembly vote on $TOPIC$. Stores scope:un_vote_target as the
# resolution's target when it is saved, takes the vote lock, fires un_vote.1
# to every other member in 30 days and schedules un_vote.2 for the proposer
# in 365 days. Leaves scope:un_resolution pointing at the new container.
# Parameters: $TOPIC$ = topic key (condemn, human_rights, sanctions, ...)
# Scope: proposing country
un_resolution_open = {
	save_scope_as = un_res_proposer_scope
	if = {
		limit = { NOT = { has_global_variable = un_resolution_seq } }
		set_global_variable = { name = un_resolution_seq value = 0 }
	}
	change_global_variable = { name = un_resolution_seq add = 1 }
	create_container = {
		tags = { un_resolution un_topic_$TOPIC$ un_res_voting }
		save_scope_as = un_resolution
	}
	if = {
		limit = { exists = scope:un_resolution }
		scope:un_resolution = {
			set_variable = { name = un_res_seq value = global_var:un_resolution_seq }
			set_variable = { name = un_res_proposer value = scope:un_res_proposer_scope }
			if = {
				limit = { exists = scope:un_vote_target }
				set_variable = { name = un_res_target value = scope:un_vote_target }
			}
			set_variable = { name = un_res_support value = 1 }
			set_variable = { name = un_res_oppose value = 0 }
			set_variable = { name = un_res_months value = 0 }
			add_to_variable_list = { name = un_res_yes target = scope:un_res_proposer_scope }
		}
		# The grounds (redesign phase 3): the target's case strength when the
		# resolution was tabled, kept on the resolution for the chamber card.
		# Staged in a global because it is evaluated in the target's scope.
		if = {
			limit = {
				exists = scope:un_resolution
				exists = scope:un_vote_target
			}
			scope:un_vote_target = {
				set_global_variable = { name = un_case_stage value = un_case_strength }
			}
			scope:un_resolution = {
				set_variable = { name = un_res_case value = global_var:un_case_stage }
			}
			remove_global_variable = un_case_stage
		}
	}
	set_global_variable = un_vote_active
	set_global_variable = { name = un_active_resolution value = scope:un_resolution }
	if = {
		limit = { has_global_variable = un_vote_reservation }
		remove_global_variable = un_vote_reservation
	}
	# The ballot (redesign phase 3). Human members get un_vote.1 in thirty
	# days and see their lean in the chamber at once (un_vote.4 writes it).
	# The AI votes in script, by the same lean: un_vote.5 fires to the
	# proposer in thirty days and sends every AI member its un_vote.4, with the
	# monthly pulse as the fallback (un_resolutions_monthly_update).
	every_country = {
		limit = {
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			NOT = { this = scope:un_res_proposer_scope }
			is_ai = no
		}
		trigger_event = { id = un_vote.1 days = 30 }
	}
	un_vote_refresh_human_leans = yes
	trigger_event = { id = un_vote.5 days = 30 }
	trigger_event = { id = un_vote.2 days = 365 }
}

# Hold the vote lock while a proposer event decides whether to propose.
# Option A then calls un_resolution_open; the other options call un_vote_release.
un_vote_reserve = {
	set_global_variable = un_vote_active
	set_global_variable = un_vote_reservation
}

# Give back a reservation taken by un_vote_reserve. Leaves the lock alone if a
# resolution holds it.
un_vote_release = {
	if = {
		limit = { has_global_variable = un_vote_reservation }
		remove_global_variable = un_vote_reservation
	}
	if = {
		limit = {
			has_global_variable = un_vote_active
			NOT = { has_global_variable = un_active_resolution }
		}
		remove_global_variable = un_vote_active
	}
}

# Record the scoped country's vote on scope:un_resolution. A vote cast after
# the resolution closed (a popup answered late) changes nothing.
#
# This is also where a lobbying commitment is settled — see
# un_lobby_settle_commitment (common/scripted_effects/un_lobby_effects.txt),
# which is a no-op for a voter that pledged nothing. It sits inside the
# `un_res_voting` guard on purpose: a ballot that arrives too late to count
# must not settle a promise either.
# Parameters: $LIST$ = un_res_yes | un_res_no
#             $COUNT$ = un_res_support | un_res_oppose
# Scope: voting country
un_resolution_record_vote = {
	save_scope_as = un_res_voter
	# Why it voted this way, counted on the resolution (redesign phase 3,
	# un_dossier_effects.txt). The list name picks the side. Before the vote
	# is booked, while the resolution is still voting.
	if = {
		limit = { scope:un_resolution ?= { has_tag = un_res_voting } }
		un_lean_tally_$LIST$ = yes
	}
	scope:un_resolution ?= {
		if = {
			limit = { has_tag = un_res_voting }
			add_to_variable_list = { name = $LIST$ target = scope:un_res_voter }
			change_variable = { name = $COUNT$ add = 1 }
			un_lobby_settle_commitment = yes
		}
	}
}

# Record a permanent member's veto on scope:un_resolution: a no vote plus the
# un_res_vetoed tag. Every veto is a credibility debit weighted by the vetoer's
# weight in world affairs (un_authority_effects.txt): a superpower killing a
# resolution costs the Assembly far more than a fading one does. At Moribund
# (redesign phase 2) a veto costs nothing, so no entry is booked.
#
# A veto books a NO vote, so it settles a lobbying commitment exactly as an
# ordinary no vote does — as broken. The veto RIGHT is never bound by a pledge
# (the option's own trigger is untouched); exercising it against a resolution
# you promised to carry is simply the most emphatic way of breaking the promise.
# Scope: vetoing country
un_resolution_record_veto = {
	save_scope_as = un_res_voter
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_res_voting }
			NOT = { un_tier_is_moribund = yes }
		}
		un_ledger_actor_entry = { PILLAR = credibility POINTS = -1.5 REASON = 4 }
	}
	if = {
		limit = { scope:un_resolution ?= { has_tag = un_res_voting } }
		un_lean_tally_un_res_no = yes
		# The proposer answers the veto (un_events.11; redesign phase 4).
		un_docket_veto_consequence = yes
	}
	scope:un_resolution ?= {
		if = {
			limit = { has_tag = un_res_voting }
			if = {
				limit = { NOT = { has_tag = un_res_vetoed } }
				add_tag = un_res_vetoed
			}
			set_variable = { name = un_res_vetoer value = scope:un_res_voter }
			add_to_variable_list = { name = un_res_no target = scope:un_res_voter }
			change_variable = { name = un_res_oppose add = 1 }
			un_lobby_settle_commitment = yes
		}
	}
}

# Settle the vote: swap un_res_voting for un_res_passed or un_res_failed.
# Expulsion and charter reform need a 2/3 supermajority of current members
# (un_resolution_needs_supermajority); every other topic a simple majority
# (more yes than no votes).
# Scope: resolution container
un_resolution_decide = {
	remove_tag = un_res_voting
	if = {
		limit = { un_resolution_needs_supermajority = yes }
		if = {
			limit = { un_vote_supermajority_passed >= 0 }
			add_tag = un_res_passed
		}
		else = {
			add_tag = un_res_failed
		}
	}
	else_if = {
		limit = { un_res_margin >= 1 }
		add_tag = un_res_passed
		un_resolution_veto_restraint = yes
	}
	else = {
		add_tag = un_res_failed
	}
}

# Veto restraint (Charter Reform II; redesign phase 5, §4.1): once the charter
# carries both reforms, a veto no longer stops a resolution that a two-thirds
# supermajority of the members with a vote has carried. The resolution loses
# its un_res_vetoed tag, so un_vote.2 applies the full form, and carries
# un_res_veto_overridden for the record. The vetoer has already paid for its
# veto (the credibility entry, un_events.11). The un_res_vetoed tag was set
# when the veto was cast, so it is readable here; the passed tag is not
# re-read, which is why this sits inside the branch that set it.
# Scope: resolution container (a simple-majority topic just carried)
un_resolution_veto_restraint = {
	if = {
		limit = {
			has_tag = un_res_vetoed
			un_charter_level_value >= 2
			un_vote_supermajority_passed >= 0
		}
		remove_tag = un_res_vetoed
		add_tag = un_res_veto_overridden
	}
}

# File a closed resolution: start its topic cooldown, append it to
# un_resolution_history, and release the vote lock if it held it. Past the
# history cap, destroy the oldest entry whose cooldown has run out; entries
# still on cooldown are kept because the proposal gates read them.
# Scope: resolution container (already tagged passed, failed or lapsed)
un_resolution_archive = {
	save_scope_as = un_res_archived
	if = {
		limit = { has_tag = un_topic_expulsion }
		set_variable = { name = un_res_cooldown_months value = un_res_cooldown_months_expulsion }
	}
	else = {
		set_variable = { name = un_res_cooldown_months value = un_res_cooldown_months_default }
	}
	add_to_global_variable_list = { name = un_resolution_history target = scope:un_res_archived }
	if = {
		limit = { global_var:un_active_resolution ?= scope:un_res_archived }
		remove_global_variable = un_active_resolution
		if = {
			limit = { has_global_variable = un_vote_active }
			remove_global_variable = un_vote_active
		}
		# The recess (redesign phase 3, §8.3): for three months only human
		# members may table business (un_floor_open_to_ai).
		set_global_variable = { name = un_recess_months value = un_recess_months_after_vote }
	}
	if = {
		limit = {
			# `any_in_*list = { count >= N }` rather than `*variable_list_size`: in 1.14
			# the engine rejects that trigger's documented `target` key at load
			# ("Failed to read 'target'"), so the limit never passed. N is cap + 1.
			any_in_global_list = { variable = un_resolution_history count >= 26 has_tag = un_resolution } # un_resolution_history_cap + 1
			any_in_global_list = {
				variable = un_resolution_history
				NOT = { has_variable = un_res_cooldown_months }
			}
		}
		ordered_in_global_list = {
			variable = un_resolution_history
			position = 0
			limit = { NOT = { has_variable = un_res_cooldown_months } }
			order_by = un_resolution_age_order
			save_scope_as = un_res_evicted
		}
		remove_list_global_variable = { name = un_resolution_history target = scope:un_res_evicted }
		scope:un_res_evicted = { destroy_container = yes }
	}
}

# Count every closed resolution's topic cooldown down by a month and remove it
# at 0 (var:un_res_cooldown_months — see the header). When the cooldown that
# was holding back a ripe charter reform lifts, the members who may table it
# are told, as they are when it ripens (un_ladder_charter_pressure_update).
# The lifted reform is staged in a global because it is noticed inside the
# list iteration and acted on after it.
# Scope: none (the global monthly pulse). Call it from nowhere else: every
# call is a month.
un_resolution_cooldowns_tick = {
	if = {
		limit = { has_global_variable_list = un_resolution_history }
		every_in_global_list = {
			variable = un_resolution_history
			limit = { has_variable = un_res_cooldown_months }
			change_variable = { name = un_res_cooldown_months subtract = 1 }
			if = {
				limit = { var:un_res_cooldown_months <= 0 }
				remove_variable = un_res_cooldown_months
				if = {
					limit = { has_tag = un_topic_reform }
					set_global_variable = un_reform_cooldown_lifted
				}
			}
		}
	}
	if = {
		limit = { has_global_variable = un_reform_cooldown_lifted }
		remove_global_variable = un_reform_cooldown_lifted
		if = {
			limit = {
				un_charter_reform_ripe = yes
				NOT = { un_resolution_topic_on_cooldown = { TOPIC = reform } }
			}
			un_charter_reform_notify_ripe = yes
		}
	}
}

# Global monthly upkeep (no-country scope; from un_global_authority_on_action).
#   0. Count the closed resolutions' topic cooldowns down
#      (un_resolution_cooldowns_tick). First, so a resolution that lapses in
#      step 1 keeps its full count.
#   1. Age the open resolution. If it is still voting after 14 months its
#      un_vote.2 never fired (the proposer stopped existing or left the UN,
#      which fails the event's trigger), so it lapses and frees the lock.
#      Before containers that case locked the General Assembly for good.
#   2. Free a lock that neither a resolution nor a reservation holds: saves
#      from before containers, or a resolution reference that stopped resolving.
un_resolutions_monthly_update = {
	un_resolution_cooldowns_tick = yes
	if = {
		limit = {
			has_global_variable = un_active_resolution
			NOT = { exists = global_var:un_active_resolution }
		}
		remove_global_variable = un_active_resolution
	}
	global_var:un_active_resolution ?= {
		change_variable = { name = un_res_months add = 1 }
		if = {
			limit = {
				has_tag = un_res_voting
				var:un_res_months >= 14
			}
			remove_tag = un_res_voting
			add_tag = un_res_lapsed
			un_resolution_archive = yes
		}
	}
	# The ballot (redesign phase 3). While a resolution is voting: refresh the
	# human members' lean snapshots, and send the AI its ballots if un_vote.5
	# has not (its recipient, the proposer, is gone, or the resolution predates
	# phase 3). Two months in, so the thirty-day event normally goes first.
	global_var:un_active_resolution ?= {
		if = {
			limit = { has_tag = un_res_voting }
			save_scope_as = un_resolution
			if = {
				limit = {
					NOT = { has_variable = un_res_ai_cast }
					var:un_res_months >= 2
				}
				un_vote_dispatch_ai_ballots = yes
			}
			un_vote_refresh_human_leans = yes
		}
	}
	if = {
		limit = {
			has_global_variable = un_vote_active
			NOT = { has_global_variable = un_active_resolution }
			NOT = { has_global_variable = un_vote_reservation }
		}
		remove_global_variable = un_vote_active
	}
}

# Apply the persistent treaty/declaration modifier for a passed UN resolution.
# Split out from un_vote_apply_resolution_compliance so each per-topic
# add_modifier renders directly in the calling option's tooltip — when
# nested inside the larger compliance effect (which also handles infamy,
# leverage, and contributor pledges), the engine's tooltip rendering
# buried the modifier name and fields, so players couldn't see what they
# were ratifying.
# Called from un_vote.3 options A and B alongside un_vote_apply_resolution_compliance.
# Expects: ROOT = country, scope:un_resolution = the passed resolution.
un_vote_apply_treaty_modifier = {
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_human_rights }
			NOT = { je:je_united_nations ?= { has_modifier = un_human_rights_declaration_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_human_rights_declaration_modifier multiplier = un_convention_multiplier } }
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_npt }
			NOT = { je:je_united_nations ?= { has_modifier = un_nonproliferation_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_nonproliferation_modifier multiplier = un_convention_multiplier } }
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_climate }
			NOT = { je:je_united_nations ?= { has_modifier = un_climate_binding_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_climate_binding_modifier multiplier = un_convention_multiplier } }
	}
	# A vetoed reform is flat-blocked (un_vote.2 applies nothing), so members
	# ratify nothing either.
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_reform }
			NOT = { scope:un_resolution ?= { has_tag = un_res_vetoed } }
			NOT = { has_modifier = un_reform_champion_modifier }
		}
		add_modifier = {
			name = un_reform_champion_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_heritage }
			NOT = { je:je_united_nations ?= { has_modifier = un_heritage_program_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_heritage_program_modifier multiplier = un_convention_multiplier } }
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_decolonization }
			NOT = { has_modifier = un_decolonization_champion_modifier }
		}
		add_modifier = {
			name = un_decolonization_champion_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
	}
	# Voting YES on decolonization is a public commitment — great powers
	# that do so adopt the anti-colonial stance variable consumed by the
	# je_colonial_empire bar (via colonial_gp_condemners_count). Setting
	# the cooldown variable prevents decolonization_events.14 from firing
	# afterward and letting the player re-deliberate the same stance.
	# Only a member that cast a ballot takes the stance: a yes voter (the
	# proposer included), or a no voter that accepts the declaration anyway
	# (un_vote.3 option B). A member that cast none — its vote suspended
	# under Article 19, or joined after the ballot — is bound by the
	# declaration through un_vote.3 option A but committed itself to nothing.
	if = {
		limit = {
			scope:un_resolution ?= {
				has_tag = un_topic_decolonization
				un_lobby_has_voted = { COUNTRY = ROOT }
			}
			country_rank >= rank_value:great_power
		}
		set_variable = gp_anti_colonial_stance
		if = {
			limit = { has_variable = gp_pro_colonial_stance }
			remove_variable = gp_pro_colonial_stance
		}
		set_variable = { name = gp_colonial_policy_cooldown days = 1095 }
		if = {
			limit = { NOT = { has_modifier = gp_anti_colonial_policy_modifier } }
			add_modifier = {
				name = gp_anti_colonial_policy_modifier
				days = 1095
				is_decaying = yes
			}
		}
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_pandemic }
			NOT = { je:je_united_nations ?= { has_modifier = un_pandemic_cooperation_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_pandemic_cooperation_modifier multiplier = un_convention_multiplier } }
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_refugee }
			NOT = { je:je_united_nations ?= { has_modifier = un_refugee_program_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_refugee_program_modifier multiplier = un_convention_multiplier } }
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_space }
			NOT = { je:je_united_nations ?= { has_modifier = un_space_partnership_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_space_partnership_modifier multiplier = un_convention_multiplier } }
	}
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_law_of_sea }
			NOT = { has_modifier = un_law_of_sea_modifier }
		}
		add_modifier = { name = un_law_of_sea_modifier multiplier = un_convention_multiplier }
	}
	# Physical protection of nuclear material: country-scoped, like the law of
	# the sea, so the nuclear code reads it without a je: hop
	# (nd_custody_is_secured, nd_loose_monthly).
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_physical_protection }
			NOT = { has_modifier = un_physical_protection_modifier }
		}
		add_modifier = { name = un_physical_protection_modifier multiplier = un_convention_multiplier }
	}
	# ICC membership: skipped when the resolution was vetoed (graduated form
	# is symbolic censure only — no court jurisdiction is established).
	if = {
		limit = {
			scope:un_resolution ?= { has_tag = un_topic_icc }
			NOT = { scope:un_resolution ?= { has_tag = un_res_vetoed } }
			NOT = { has_modifier = un_icc_member_modifier }
		}
		add_modifier = { name = un_icc_member_modifier multiplier = un_convention_multiplier }
	}
	# The convention regimes (redesign phase 5, §5.3): read our terms now —
	# the member modifiers at the current E, and whatever the regimes make us
	# win or pay (un_regime_effects.txt; the chamber's Our Obligations lists
	# them).
	hidden_effect = {
		un_regime_refresh_country = yes
	}
}

# Catch-up effect: apply every globally-ratified UN convention's binding
# modifier to a country that has just joined the UN. Without this, late
# joiners (decolonization-spawned nations, treaty auto-enrollment) miss
# the gameplay effects of conventions ratified before they joined, even
# though they're now members and bound by them in fiction. Idempotent —
# each branch skips if the modifier is already present, so safe to call
# from every UN-join path (founder charter, voluntary join button,
# treaty-obligation auto-enrollment).
# Expects: ROOT = the country that just joined the UN.
un_apply_ratified_conventions = {

	if = {

# Standing award for actually taking on a share of a peacekeeping deployment
# or an aid programme the Assembly has already voted through. Called from the
# two pledge branches of un_vote_apply_resolution_compliance below.
#
# WHY THIS IS THE DELIVERY SITE, AND WHY IT CANNOT PAY TWICE.
# un_vote.2 (which puts contributors on cooldown) iterates every country that
# *currently carries* the contributor marker, so it sweeps up donors from
# earlier resolutions and, for peacekeeping, every standing programme member
# as well — awarding standing there would pay the same country on every future
# request.
# The pledge is made HERE instead, in un_vote.3, which un_vote.2 fires exactly
# once per member per resolution; an event runs exactly one option, and the two
# options that reach this effect (A and B) are mutually exclusive by their own
# triggers. So this runs at most once per country per resolution, and the
# branch it sits in is the same one that books the contribution count.
#
# The relations gain is the "limited relations with aid recipients / mission
# partners" case, implemented only where an existing site already knows both
# countries: scope:un_vote_proposer is the country that asked for the aid or
# the peacekeepers. One-off, never repeating, and skipped when the contributor
# is the requester itself.
# Expects: ROOT = the contributing country, scope:un_vote_proposer = requester.
un_vote_standing_request_contribution = {
	custom_tooltip = {
		text = un_standing_contribution_tt
		un_standing_gain = { AMOUNT = un_standing_request_contribution_gain REASON = 4 }
	}
	if = {
		limit = {
			exists = scope:un_vote_proposer
			NOT = { THIS = scope:un_vote_proposer }
		}
		change_relations = { country = scope:un_vote_proposer value = un_standing_delivery_relations }
	}
}

# A power-bloc leader's resolution has closed: its bloc gains leverage in (or,
# with a negative $VALUE$, loses leverage in) every other bloc member that sits
# in the Assembly. Called from un_vote.2's carried and failed branches.
#
# The member is named with a saved scope rather than reached with `prev`. The
# first version re-entered the proposer inside the member loop —
# `every_power_bloc_member = { root = { power_bloc ?= { add_leverage = {
# target = prev ... } } } }` — where `prev` is the scope `root = { }` pushed,
# i.e. the proposer itself, not the member. The bloc piled leverage on its own
# leader, once per member.
#
# One summary line ($TT$) instead of the engine's line per member, which ran
# to twenty identical lines in a large bloc.
# Parameters: $VALUE$ = leverage per member, $TT$ = the summary tooltip key
# Scope: the proposer
un_vote_proposer_bloc_leverage = {
	save_temporary_scope_as = un_vote_bloc_leader
	if = {
		limit = {
			is_power_bloc_leader = yes
			power_bloc ?= {
				any_power_bloc_member = {
					NOT = { THIS = scope:un_vote_bloc_leader }
					je:je_united_nations ?= { has_modifier = un_member_modifier }
				}
			}
		}
		custom_tooltip = $TT$
		hidden_effect = {
			power_bloc ?= {
				save_temporary_scope_as = un_vote_leader_bloc
				every_power_bloc_member = {
					limit = {
						NOT = { THIS = scope:un_vote_bloc_leader }
						je:je_united_nations ?= { has_modifier = un_member_modifier }
					}
					save_temporary_scope_as = un_vote_bloc_member
					scope:un_vote_leader_bloc = {
						add_leverage = { target = scope:un_vote_bloc_member value = $VALUE$ }
					}
				}
			}
		}
	}
}

# Apply the residual (non-modifier) effects of complying with a passed UN
# resolution: infamy reduction, power-bloc leverage, peacekeeping/aid contributor
# pledges, and leader leverage on aid/peacekeeping. Per-topic treaty modifiers
# live in un_vote_apply_treaty_modifier (split out so they render visibly in
# option tooltips); call both effects together from un_vote.3 options A and B.
# Expects: ROOT = country, scope:un_vote_proposer = proposer country (if exists),
# scope:un_resolution = the passed resolution.
un_vote_apply_resolution_compliance = {
	# Infamy reduction for supporting humanitarian/aid/peacekeeping votes
	if = {
		limit = {
			OR = {
				scope:un_resolution ?= { has_tag = un_topic_aid_request }
				scope:un_resolution ?= { has_tag = un_topic_peacekeeping_request }
				scope:un_resolution ?= { has_tag = un_topic_human_rights }
				scope:un_resolution ?= { has_tag = un_topic_refugee }
				scope:un_resolution ?= { has_tag = un_topic_pandemic }
			}
			infamy > 0
		}
		change_infamy = -1
	}

	# Power bloc leverage: voting with the proposer's bloc on major resolutions
	if = {
		limit = {
			exists = scope:un_vote_proposer
			scope:un_vote_proposer = { is_power_bloc_leader = yes }
			is_in_same_power_bloc = scope:un_vote_proposer
		}
		scope:un_vote_proposer = {
			power_bloc ?= {
				add_leverage = { target = root value = 5 }
			}
		}
	}

	# ---- Peacekeeping/aid pledges ----
	# A pledge is charged HERE, when it is made. un_vote.2 used to levy the
	# costs, but it runs before un_vote.3 pledges anyone, so a pledger paid
	# nothing until some later request passed; the peacekeeping levy also
	# evaluated gdp in journal-entry scope (reads 'none') and the aid levy was
	# stripped every month by the legacy cleanup. ROOT is the pledger here
	# (un_vote.3), so the root.var: multiplier below resolves correctly.
	# Peacekeeping contributions are skipped when the resolution sent observers
	# only — vetoed, or (redesign phase 5) carried by a UN below Established —
	# because the observer mission doesn't levy contributors.
	# A pledge is made only by a member that cast a ballot: a yes voter, or a
	# no voter that takes part anyway (un_vote.3 option B). A member that
	# cast none (Article 19, or a late joiner) reaches this through option A's
	# acknowledgement and is not signed up for a deployment or its costs.
	if = {
		limit = {
			scope:un_resolution ?= {
				un_resolution_peacekeeping_full = yes
				un_lobby_has_voted = { COUNTRY = ROOT }
			}
			country_rank >= rank_value:major_power
		}
		je:je_united_nations ?= { add_modifier = { name = un_peacekeeping_contributor_modifier } }
		# Joining the deployment enrols the pledger in the paid peacekeeping
		# programme — the same cost un_peacekeeping_mission_button charges.
		# A member already running it keeps its existing cost.
		if = {
			limit = { NOT = { je:je_united_nations ?= { has_modifier = un_peacekeeping_contributor_cost } } }
			set_variable = { name = un_peacekeeping_expense_cached value = un_program_expense_value }
			je:je_united_nations ?= {
				add_modifier = {
					name = un_peacekeeping_contributor_cost
					multiplier = root.var:un_peacekeeping_expense_cached
				}
			}
		}
		if = {
			limit = { NOT = { has_variable = un_peacekeeping_contribution_count } }
			set_variable = { name = un_peacekeeping_contribution_count value = 0 }
		}
		change_variable = { name = un_peacekeeping_contribution_count add = 1 }
		un_vote_standing_request_contribution = yes
		# ...and joins the mission the resolution opened (redesign phase 6).
		un_mission_join_resolution = yes
	}
	if = {
		limit = {
			scope:un_resolution ?= {
				has_tag = un_topic_aid_request
				un_lobby_has_voted = { COUNTRY = ROOT }
			}
			country_rank >= rank_value:major_power
		}
		add_modifier = {
			name = un_humanitarian_aid_modifier
			days = long_modifier_time
			is_decaying = yes
		}
		# The pledge's cost runs as long as the aid modifier it pays for.
		add_modifier = {
			name = un_humanitarian_aid_cost
			multiplier = un_program_expense_value
			days = long_modifier_time
			is_decaying = yes
		}
		if = {
			limit = { NOT = { has_variable = un_aid_contribution_count } }
			set_variable = { name = un_aid_contribution_count value = 0 }
		}
		change_variable = { name = un_aid_contribution_count add = 1 }
		un_vote_standing_request_contribution = yes
		# ...and joins the mission the resolution opened (redesign phase 6).
		un_mission_join_resolution = yes
	}

	# Power bloc leader providing aid/peacekeeping: gain leverage over the
	# recipient. A member that cast no ballot pledged nothing above, so it is
	# not credited with providing anything either.
	if = {
		limit = {
			OR = {
				scope:un_resolution ?= { has_tag = un_topic_aid_request }
				scope:un_resolution ?= { has_tag = un_topic_peacekeeping_request }
			}
			scope:un_resolution ?= { un_lobby_has_voted = { COUNTRY = ROOT } }
			is_power_bloc_leader = yes
			exists = scope:un_vote_proposer
			NOT = { THIS = scope:un_vote_proposer }
		}
		power_bloc ?= {
			add_leverage = { target = scope:un_vote_proposer value = 50 }
		}
	}
}
