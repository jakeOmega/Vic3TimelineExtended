# ============================================================================
# STATE PICTURES - the yearly pass (design: docs/superpowers/specs/2026-11-10-state-pictures-design.md;
# system doc: docs/systems/mod_systems/state_pictures.md).
#
# The banner at the top of the state panel is picked by the weights in
# common/terrain/te_state_pictures_generated.txt, which the engine may evaluate every frame. So a
# weight reads the owner's technology and this pass's stored tags, and nothing dearer:
#   te_sp_size   1 town, 2 city, 3 metropolis, 4 megacity; absent before the state's first walk
#   te_sp_oil    present while oil is a major part of the state's economy (the oil-field pictures)
#   te_sp_coal   present while coal is the larger part of the state's mines (the colliery pictures)
#   te_sp_industry, te_sp_resort   present while heavy industry, or tourism, is a large part of
#                the state's building levels (the heavy-industry city, the resort)
#   te_sp_boom, te_sp_decline   present while the state's people have grown fast, or fallen, for
#                several years (te_sp_growth, a smoothed yearly rate; te_sp_prov, the province
#                count it was last taken at)
#   te_sp_divided   present while a metropolis is very unequal
# A picture can therefore lag a change by up to a year.
# ============================================================================

# State scope, from on_yearly_pulse_state (te_state_picture_on_actions.txt).
te_state_picture_refresh = {
	if = {
		limit = { te_state_picture_city_people < 0 }
		remove_variable = te_sp_size
	}
	# no size yet, or grown past the next floor: the size its people make it
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = te_sp_size }
				var:te_sp_size < te_state_picture_size_up
			}
		}
		set_variable = { name = te_sp_size value = te_state_picture_size_up }
	}
	# shrunk below the share of its floor that keeps it
	else_if = {
		limit = { var:te_sp_size > te_state_picture_size_down }
		set_variable = { name = te_sp_size value = te_state_picture_size_down }
	}
	te_state_picture_refresh_land = yes
	te_state_picture_refresh_city = yes
	te_state_picture_selfcheck = yes
}

# What the state's worked land is known for (owner, 2026-11-10: an oil state is one where oil is a
# major part of the economy, not one with many rigs beside a great city). Script can't read one
# building's share of the state's GDP, so the measure is its share of the state's building levels,
# subsistence buildings left out. One walk fills the four sums. A tag is kept until the figure
# falls a tenth under its floor (te_state_picture_k_keep), so a state at a floor doesn't change
# picture each year.
te_state_picture_refresh_land = {
	te_state_picture_land_sums = yes
	te_state_picture_land_tags = yes
}

# The walk: the state's levels, its Oil Rig levels, its mine levels and its Coal Mine levels, as
# locals for te_state_picture_land_tags; and each tag's floor in levels and the figure that keeps it.
te_state_picture_land_sums = {
	set_local_variable = { name = te_sp_all value = 0 }
	set_local_variable = { name = te_sp_oil_lv value = 0 }
	set_local_variable = { name = te_sp_mine value = 0 }
	set_local_variable = { name = te_sp_coal_lv value = 0 }
	set_local_variable = { name = te_sp_heavy_lv value = 0 }
	set_local_variable = { name = te_sp_tour_lv value = 0 }
	every_scope_building = {
		limit = { is_subsistence_building = no }
		change_local_variable = { name = te_sp_all add = level }
		if = {
			limit = { is_building_type = building_oil_rig }
			change_local_variable = { name = te_sp_oil_lv add = level }
		}
		if = {
			limit = { is_building_group = bg_mining }
			change_local_variable = { name = te_sp_mine add = level }
		}
		if = {
			limit = { is_building_type = building_coal_mine }
			change_local_variable = { name = te_sp_coal_lv add = level }
		}
		if = {
			limit = { is_building_group = bg_heavy_industry }
			change_local_variable = { name = te_sp_heavy_lv add = level }
		}
		if = {
			limit = { is_building_type = building_tourism_industry }
			change_local_variable = { name = te_sp_tour_lv add = level }
		}
	}
}

te_state_picture_land_tags = {
	# oil: a share of the state's levels, and a least number of levels
	set_local_variable = { name = te_sp_floor value = { value = local_var:te_sp_all multiply = te_state_picture_k_oil_share } }
	set_local_variable = { name = te_sp_keep value = { value = local_var:te_sp_floor multiply = te_state_picture_k_keep } }
	if = {
		limit = {
			local_var:te_sp_oil_lv >= te_state_picture_k_oil_levels
			local_var:te_sp_oil_lv >= local_var:te_sp_floor
		}
		set_variable = { name = te_sp_oil value = 1 }
		te_state_picture_land_selfcheck = yes
	}
	else_if = {
		limit = {
			has_variable = te_sp_oil
			OR = {
				local_var:te_sp_oil_lv < te_state_picture_k_oil_levels_keep
				local_var:te_sp_oil_lv < local_var:te_sp_keep
			}
		}
		remove_variable = te_sp_oil
	}
	# coal: a share of the state's mine levels, and a least number of levels
	set_local_variable = { name = te_sp_floor value = { value = local_var:te_sp_mine multiply = te_state_picture_k_coal_share } }
	set_local_variable = { name = te_sp_keep value = { value = local_var:te_sp_floor multiply = te_state_picture_k_keep } }
	if = {
		limit = {
			local_var:te_sp_coal_lv >= te_state_picture_k_coal_levels
			local_var:te_sp_coal_lv >= local_var:te_sp_floor
		}
		set_variable = { name = te_sp_coal value = 1 }
	}
	else_if = {
		limit = {
			has_variable = te_sp_coal
			OR = {
				local_var:te_sp_coal_lv < te_state_picture_k_coal_levels_keep
				local_var:te_sp_coal_lv < local_var:te_sp_keep
			}
		}
		remove_variable = te_sp_coal
	}
}

# What a city is known for, beyond its region and size (design section 7). Called after
# te_state_picture_refresh_land, with its walk's locals still set.
te_state_picture_refresh_city = {
	# heavy industry and tourism: a share of the state's levels and a least number, as oil is
	set_local_variable = { name = te_sp_floor value = { value = local_var:te_sp_all multiply = te_state_picture_k_industry_share } }
	set_local_variable = { name = te_sp_keep value = { value = local_var:te_sp_floor multiply = te_state_picture_k_keep } }
	if = {
		limit = {
			local_var:te_sp_heavy_lv >= te_state_picture_k_industry_levels
			local_var:te_sp_heavy_lv >= local_var:te_sp_floor
		}
		set_variable = { name = te_sp_industry value = 1 }
	}
	else_if = {
		limit = {
			has_variable = te_sp_industry
			OR = {
				local_var:te_sp_heavy_lv < te_state_picture_k_industry_levels_keep
				local_var:te_sp_heavy_lv < local_var:te_sp_keep
			}
		}
		remove_variable = te_sp_industry
	}
	set_local_variable = { name = te_sp_floor value = { value = local_var:te_sp_all multiply = te_state_picture_k_resort_share } }
	set_local_variable = { name = te_sp_keep value = { value = local_var:te_sp_floor multiply = te_state_picture_k_keep } }
	if = {
		limit = {
			local_var:te_sp_tour_lv >= te_state_picture_k_resort_levels
			local_var:te_sp_tour_lv >= local_var:te_sp_floor
		}
		set_variable = { name = te_sp_resort value = 1 }
	}
	else_if = {
		limit = {
			has_variable = te_sp_resort
			OR = {
				local_var:te_sp_tour_lv < te_state_picture_k_resort_levels_keep
				local_var:te_sp_tour_lv < local_var:te_sp_keep
			}
		}
		remove_variable = te_sp_resort
	}

	# growth: the Demographics walk's people against the walk before's, smoothed over several
	# years. A year in which the state gained or lost provinces is left out, as is a ratio the
	# walk can't give.
	if = {
		limit = { te_state_picture_growth_known = yes }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = te_sp_prov }
					var:te_sp_prov = var:te_dg_provinces
				}
			}
			if = {
				limit = { NOT = { has_variable = te_sp_growth } }
				set_variable = { name = te_sp_growth value = 0 }
			}
			set_variable = { name = te_sp_growth value = te_state_picture_growth_smoothed }
		}
		set_variable = { name = te_sp_prov value = var:te_dg_provinces }
	}
	if = {
		limit = {
			has_variable = te_sp_growth
			has_variable = te_sp_size
			var:te_sp_size >= 2
			var:te_sp_growth >= te_state_picture_k_boom
		}
		set_variable = { name = te_sp_boom value = 1 }
	}
	else_if = {
		limit = {
			has_variable = te_sp_boom
			OR = {
				NOT = { has_variable = te_sp_growth }
				NOT = { has_variable = te_sp_size }
				var:te_sp_size < 2
				var:te_sp_growth < te_state_picture_k_boom_keep
			}
		}
		remove_variable = te_sp_boom
	}
	if = {
		limit = {
			has_variable = te_sp_growth
			has_variable = te_sp_size
			var:te_sp_size >= 2
			var:te_sp_growth <= te_state_picture_k_decline
		}
		set_variable = { name = te_sp_decline value = 1 }
	}
	else_if = {
		limit = {
			has_variable = te_sp_decline
			OR = {
				NOT = { has_variable = te_sp_growth }
				NOT = { has_variable = te_sp_size }
				var:te_sp_size < 2
				var:te_sp_growth > te_state_picture_k_decline_keep
			}
		}
		remove_variable = te_sp_decline
	}

	# divided: a metropolis whose Gini (the Demographics walk's te_dg_gini) is very high
	if = {
		limit = {
			has_variable = te_sp_size
			var:te_sp_size >= 3
			has_variable = te_dg_gini
			var:te_dg_gini >= te_state_picture_k_divided
		}
		set_variable = { name = te_sp_divided value = 1 }
	}
	else_if = {
		limit = {
			has_variable = te_sp_divided
			OR = {
				NOT = { has_variable = te_sp_size }
				var:te_sp_size < 3
				NOT = { has_variable = te_dg_gini }
				var:te_dg_gini < te_state_picture_k_divided_keep
			}
		}
		remove_variable = te_sp_divided
	}
}

# Once a game: the first state tagged as oil country prints the sums the tag came from. The walk
# rests on `level` reading a building's levels inside every_scope_building and on the locals
# carrying out of it; OK when the oil levels are at least the floor of levels and no more than
# the state's whole. Only a tagged state prints, so an empty walk can't pass. Called with the
# walk's locals still set. The line also prints the state's mine and Coal Mine levels, which the
# colliery tag rests on: mine_levels counts every building under bg_mining (the mod's mines sit
# in bg_discoverable_mining, a child group), so in a state with coal it should be at least
# coal_levels; a mine_levels of 0 beside coal means the group test doesn't reach child groups.
te_state_picture_land_selfcheck = {
	if = {
		limit = { NOT = { has_global_variable = te_sp_land_selfcheck_done } }
		set_global_variable = { name = te_sp_land_selfcheck_done value = yes }
		set_global_variable = { name = te_sp_chk_oil value = local_var:te_sp_oil_lv }
		set_global_variable = { name = te_sp_chk_all value = local_var:te_sp_all }
		set_global_variable = { name = te_sp_chk_mine value = local_var:te_sp_mine }
		set_global_variable = { name = te_sp_chk_coal value = local_var:te_sp_coal_lv }
		if = {
			limit = {
				local_var:te_sp_oil_lv >= te_state_picture_k_oil_levels
				local_var:te_sp_oil_lv <= local_var:te_sp_all
			}
			debug_log = "TE_SELF_CHECK OK state_picture_oil state=[THIS.GetState.GetNameNoFormatting] owner=[THIS.GetState.GetOwner.GetTagName] oil_levels=[GetGlobalVariable('te_sp_chk_oil').GetValue|0] all_levels=[GetGlobalVariable('te_sp_chk_all').GetValue|0] mine_levels=[GetGlobalVariable('te_sp_chk_mine').GetValue|0] coal_levels=[GetGlobalVariable('te_sp_chk_coal').GetValue|0]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN state_picture_oil state=[THIS.GetState.GetNameNoFormatting] owner=[THIS.GetState.GetOwner.GetTagName] oil_levels=[GetGlobalVariable('te_sp_chk_oil').GetValue|0] all_levels=[GetGlobalVariable('te_sp_chk_all').GetValue|0] mine_levels=[GetGlobalVariable('te_sp_chk_mine').GetValue|0] coal_levels=[GetGlobalVariable('te_sp_chk_coal').GetValue|0] (the walk's sums don't hold together: `level` or the locals don't read as the pass takes them)"
		}
	}
}

# Once a game: the first state stored at metropolis or larger says whether its figures hold together.
# The pass rests on te_dg_urban being a head count on the scale of state_population, so its largest
# city can't hold more people than the state does; BROKEN means it isn't, and the floors in
# te_state_picture_values.txt are in the wrong unit. The picture itself can't be logged
# ([THIS.GetState.GetBestTexture] prints empty), so the play-test list
# (docs/testing/state-pictures-playtest.md) covers the banner.
te_state_picture_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_sp_selfcheck_done }
			has_variable = te_sp_size
			var:te_sp_size >= 3
		}
		set_global_variable = { name = te_sp_selfcheck_done value = yes }
		if = {
			limit = { te_state_picture_city_people <= te_state_picture_state_people }
			debug_log = "TE_SELF_CHECK OK state_picture_size state=[THIS.GetState.GetNameNoFormatting] owner=[THIS.GetState.GetOwner.GetTagName] size=[THIS.ScriptValue('te_state_picture_size_shown')|0] city_people=[THIS.ScriptValue('te_state_picture_city_people')|0] state_people=[THIS.ScriptValue('te_state_picture_state_people')|0]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN state_picture_size state=[THIS.GetState.GetNameNoFormatting] owner=[THIS.GetState.GetOwner.GetTagName] size=[THIS.ScriptValue('te_state_picture_size_shown')|0] city_people=[THIS.ScriptValue('te_state_picture_city_people')|0] state_people=[THIS.ScriptValue('te_state_picture_state_people')|0] (the largest city holds more people than its state: te_dg_urban is not a head count on state_population's scale)"
		}
	}
}
