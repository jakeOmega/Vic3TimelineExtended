# Stable, semicolon-separated observer records (TE_NUCLEAR:), summarised by
# scripts/analysis/nuclear_observer_report.py.
#
# Every record is a literal line. A $PARAM$ inside a debug_log string logs
# garbled (scripting_best_practices.md § "Runtime Debugging with debug_log":
# the character before it is dropped and the closing $ stays), so an action or
# incident name is written out where the caller knows it, and a name that
# depends on a parameter is chosen by branching on the variable it just set.
# Only confirmed accessors: THIS country name, the game date, ScriptValue
# wrappers. The ID and crisis-figure wrappers read global variables that
# nd_log_context fills just before each record: inside debug_log a ScriptValue
# can't see a saved scope, and it evaluates against ROOT, which can be a
# diplomatic play (scripting_best_practices.md § "`debug_log` Loc-String
# Templating Limitations"). The issuer, target, opponent and enemy names print
# through SCOPE.sCountry(<saved scope>), which printed blank in a 2026-10-08
# probe; unconfirmed here.
#
# Event-driven records (actions, incidents, crisis open/acute/end, exchange
# lifecycle) are always on: they are rare. The weekly risk sample and the
# crisis-tick sample are written only while the global nd_observer_logging is
# set (te_debug_deterrence.2's "Toggle observer sampling"): one line per armed
# country per week would fill debug.log (scripting_best_practices.md
# § "Debug instrumentation pattern", item 4).

# Before every record, THIS = the country it is about: copy what the ID and
# crisis-figure wrappers print (nuclear_exchange_values.txt) into globals.
nd_log_context = {
	set_global_variable = { name = nd_log_crisis_id_now value = 0 }
	if = {
		limit = { has_variable = nd_crisis_id }
		set_global_variable = { name = nd_log_crisis_id_now value = var:nd_crisis_id }
	}
	set_global_variable = { name = nd_log_exchange_id_now value = 0 }
	if = {
		limit = { has_variable = nd_exchange_id }
		set_global_variable = { name = nd_log_exchange_id_now value = var:nd_exchange_id }
	}
	set_global_variable = { name = nd_log_crisis_stage_now value = 0 }
	set_global_variable = { name = nd_log_crisis_weeks_now value = 0 }
	set_global_variable = { name = nd_log_crisis_dispute_now value = 0 }
	scope:nd_issuer ?= {
		if = {
			limit = { has_variable = nd_crisis_stage }
			set_global_variable = { name = nd_log_crisis_stage_now value = var:nd_crisis_stage }
		}
		if = {
			limit = { has_variable = nd_crisis_weeks }
			set_global_variable = { name = nd_log_crisis_weeks_now value = var:nd_crisis_weeks }
		}
		if = {
			limit = { has_variable = nd_crisis_dispute }
			set_global_variable = { name = nd_log_crisis_dispute_now value = var:nd_crisis_dispute }
		}
	}
}

# When the AI's strategic strike (nuke_diplo_action) is open to us: once per
# enemy per war, the first week an armed country at war passes the action's
# own gates (a warhead, doctrine, pledges, the war law, assembled forces) and
# the AI's judgement (nd_ai_nuclear_use_justified) against that enemy. The
# 2026-10-06 observer run had no deliberate wartime first strike in 45 years
# despite 22 moves to Warfighting and a taboo down to 4; this says whether the
# path was ever open, or open and passed over. Rare, so always on. The list
# resets once we are at peace. Scope: country (nd_weekly_update)
nd_log_strike_eligibility = {
	if = {
		limit = {
			nd_is_armed = yes
			is_at_war = yes
		}
		save_scope_as = nd_sel_self
		every_country = {
			limit = { has_war_with = scope:nd_sel_self }
			save_scope_as = nd_sel_enemy
			scope:nd_sel_self = {
				if = {
					limit = {
						NOT = { is_target_in_variable_list = { name = nd_sel_logged target = scope:nd_sel_enemy } }
						has_variable = nuclear_weapon_stockpile
						var:nuclear_weapon_stockpile > 0
						nd_doctrine_permits_strike = { ENEMY = scope:nd_sel_enemy }
						nd_pledge_permits_strike = { ENEMY = scope:nd_sel_enemy }
						nd_war_law_permits_strategic_strike = yes
						nd_forces_assembled = yes
						nd_ai_nuclear_use_justified = { ENEMY = scope:nd_sel_enemy }
					}
					add_to_variable_list = { name = nd_sel_logged target = scope:nd_sel_enemy }
					nd_log_context = yes
					debug_log = "TE_NUCLEAR: type=action; action=strike_eligible; date=[TimeKeeper.GetCurrentDate.GetString]; country=[THIS.GetCountry.GetNameNoFormatting]; crisis=[SCOPE.ScriptValue('nd_log_crisis_id')|0]; exchange=[SCOPE.ScriptValue('nd_log_exchange_id')|0]; enemy=[SCOPE.sCountry('nd_sel_enemy').GetNameNoFormatting]"
				}
			}
		}
	}
	else_if = {
		limit = { has_variable_list = nd_sel_logged }
		clear_variable_list = nd_sel_logged
	}
}
