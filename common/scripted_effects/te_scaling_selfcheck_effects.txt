# TEMPORARY self-checks (2026-11-06, #1404): do 1.15's scaling modifiers
# reach the country, and come back with an inherited entry?
#
# Design: docs/superpowers/specs/2026-11-06-scaling-modifiers-design.md § 5.
# Probed: a declared value reaches a modifier: read 2-5 days later, from a
# country-root value. Not probed: the same through the mod's real entries, and
# whether a revolution's winner, which inherits the entry empty, gets the
# declarations back with it.
#
# scaling_live, once a game, from the Strategic Reserve's weekly pulse, before
# the week's income is booked again: a country whose last weekly pulse booked
# at least 1 of sales income, with nothing booking it again since (a panel
# action or un_economy.2 rebooks it mid-week through
# st_res_refresh_hub_flow_effect, and the declaration lags the change), and
# with no added copy left from an older save (the monthly strip takes it off;
# has_modifier sees only that copy, never the declaration).
# modifier:country_te_budget_income_reserve_add (a field only
# st_res_sell_profit_modifier carries) against var:st_res_sell_profit:
#   OK      within 1% (the declaration follows the variable)
#   BROKEN  otherwise. Fallback: put st_res_apply_sell_profit_effect's
#           remove_modifier / add_modifier back on the country.
#
# scaling_inherited, once a game, 30 days after revolutionaries win a civil
# war, if the winner holds Covert Warfare with a funding cost over 1 and the
# rebels ran no Covert Warfare entry of their own during the war (theirs would
# still declare and prove nothing; te_sc_cw_rebel_entry marks one, and the
# winner keeps its own variables over the loser's).
# modifier:country_te_budget_expense_covert_add (a field only
# covert_operation_funding_cost carries) against covert_operation_cost_mult:
#   OK      over 0 (the inherited entry declares it again; an inherited entry
#           arrives with no added modifiers, so a missing declaration reads 0)
#   BROKEN  0. Every inherited entry has lost its declarations: already
#           Covert Warfare's five and Global Warming's six (their monthly pulses
#           used to re-add them; Carbon Removal's and Aerosol Injection's
#           restore effects were deleted), the Strategic Reserve's income if
#           that entry is inherited, and the revolution repairs PRs 3, 7 and 8
#           delete. Fallback: design § 5 (end the winner's inherited entries
#           through an invalid trigger on a flag set at te_civil_war_on_won and
#           add them again, or put the deleted repairs back).
#
# Play-test rows: docs/testing/vanilla-1-15-playtest.md, C7. Delete this file,
# the values in common/script_values/te_scaling_selfcheck_values.txt, the event
# in events/te_scaling_selfcheck_events.txt and their four calls
# (st_res_je_weekly_pulse_effect twice, je_covert_warfare's monthly pulse,
# te_civil_war_on_won) once both read OK.

# Country scope: the Strategic Reserve's weekly pulse, before the week runs.
te_scaling_live_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_sc_scaling_live_done }
			has_variable = te_sc_live_booked
			has_variable = st_res_sell_profit
			var:st_res_sell_profit >= 1
			var:st_res_sell_profit = var:te_sc_live_booked
			NOT = { has_modifier = st_res_sell_profit_modifier }
		}
		set_global_variable = te_sc_scaling_live_done
		set_global_variable = { name = te_sc_live_value value = var:st_res_sell_profit }
		set_global_variable = { name = te_sc_live_modifier value = modifier:country_te_budget_income_reserve_add }
		if = {
			limit = {
				te_sc_live_relative_gap > -0.01
				te_sc_live_relative_gap < 0.01
			}
			debug_log = "TE_SELF_CHECK OK scaling_live (je_strategic_reserve's declared sales income reaches the country: modifier=[GetGlobalVariable('te_sc_live_modifier').GetValue|2] value=[GetGlobalVariable('te_sc_live_value').GetValue|2])"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN scaling_live (je_strategic_reserve's declared sales income: modifier=[GetGlobalVariable('te_sc_live_modifier').GetValue|2] against value=[GetGlobalVariable('te_sc_live_value').GetValue|2], booked a week ago; put the weekly add_modifier back, docs/testing/vanilla-1-15-playtest.md C7)"
		}
	}
}

# Country scope: the end of the Strategic Reserve's weekly pulse, once the
# week's income is booked. The next pulse's check compares against it.
te_scaling_live_selfcheck_book = {
	if = {
		limit = {
			NOT = { has_global_variable = te_sc_scaling_live_done }
			has_variable = st_res_sell_profit
		}
		set_variable = { name = te_sc_live_booked value = var:st_res_sell_profit }
	}
}

# Country scope: Covert Warfare's monthly pulse. Marks rebels who run an
# entry of their own, which scaling_inherited must not count.
te_scaling_inherited_selfcheck_rebel_entry = {
	if = {
		limit = {
			NOT = { has_global_variable = te_sc_scaling_inherited_done }
			is_revolutionary = yes
		}
		set_variable = te_sc_cw_rebel_entry
	}
}

# Country scope: the winner of a civil war (te_civil_war_on_won), while its
# side variables still say who won.
te_scaling_inherited_selfcheck_mark = {
	if = {
		limit = {
			NOT = { has_global_variable = te_sc_scaling_inherited_done }
			var:te_cw_rebels_won ?= 1
		}
		trigger_event = { id = te_scaling_selfcheck.1 days = 30 }
	}
}
