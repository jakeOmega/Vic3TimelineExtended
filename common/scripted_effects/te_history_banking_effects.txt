# ============================================================================
# HISTORY STORE — banking series
# ============================================================================
# The banking system's side of the generic store in te_history_effects.txt:
# three metrics and the markers that explain their turning points. This is the
# whole per-system surface — a future reserve or UN series adds a file exactly
# like this one and nothing else in common/.
#
# Metrics (te_hist_v_<METRIC> on the month's sample container):
#   bank_value      finance_cycle_value      0..100
#   bank_momentum   finance_cycle_momentum   signed, roughly -6..+6
#   bank_bubble     bubble_pressure          0..100
#   mon_policy      te_policy_rate           pp, -3..25 (regime range, §5.1)
#   mon_rate_paid   te_rate_paid_pts         pp, 0.5..60 (§4)
#   mon_inflation   te_inflation             pp, -10..100 (headline, §9.3)
#
# The three monetary series are the oscillation detector the spec asks for
# (monetary_policy_design.md §16.4, §20 risk 5): a target that flaps, a rate
# paid that runs away or an inflation rate that will not settle shows up as a
# sawtooth here long before it shows up as a crisis. Unlike the three cycle
# metrics they are NOT banded — the policy rate is something the player chose,
# the rate paid is something they can read off the budget, and headline
# inflation is displayed exactly by §0's hidden-state rule — so all three charts
# print the exact figure in their bar tooltips.
#
# HEADLINE ONLY. te_inflation_core and te_inflation_noise are hidden state
# (ruling P7) and are never sampled: a stored series is a rendered series.
#
# Markers:
#   on_<button> / off_<button>   a banking policy was enabled/disabled
#   act_ce_invest_pool_*         a one-shot investment-pool action
#   crash                        this country's own banking crash began
#   crash_contagion              a crash arrived through financial contagion
#
# Both helpers re-check the game rule and the journal entry. The sampling call
# already sits inside je_banking_cycle's monthly pulse, but the marker call is
# reached from the policy effects, which the AI can also run from the journal
# entry's scripted buttons — so the gate lives here rather than at each of the
# 62 call sites.
# ============================================================================

# One sample per metric per month, taken from je_banking_cycle's monthly pulse
# after the cycle has advanced, so each sample is that month's settled state.
# Scope: country
te_history_record_banking_samples = {
	if = {
		limit = {
			has_game_rule = banking_system_enabled
			has_journal_entry = je_banking_cycle
		}
		te_history_record_sample = { METRIC = bank_value    VALUE = var:finance_cycle_value }
		te_history_record_sample = { METRIC = bank_momentum VALUE = var:finance_cycle_momentum }
		te_history_record_sample = { METRIC = bank_bubble   VALUE = var:bubble_pressure }

		# The monetary variables are owned by te_monetary_monthly_update, which
		# runs off a different pulse: this journal entry's monthly pulse and the
		# country's may fall in either order in a given month (§17 check 7), and
		# a country restored from a save made before the system existed has run
		# neither yet. Guarded, so a missing variable skips the sample and the
		# chart simply shows no bar for that month — which is exactly what the
		# store's "no te_hist_v_<metric> means no data" rule is for.
		if = {
			limit = { has_variable = te_policy_rate }
			te_history_record_sample = { METRIC = mon_policy VALUE = var:te_policy_rate }
		}
		if = {
			limit = { has_variable = te_rate_paid_pts }
			te_history_record_sample = { METRIC = mon_rate_paid VALUE = var:te_rate_paid_pts }
		}
		if = {
			limit = { has_variable = te_inflation }
			te_history_record_sample = { METRIC = mon_inflation VALUE = var:te_inflation }
		}
	}
}

# Parameters: $MARK$ = marker key (see the header)
# Scope: country
te_history_record_banking_marker = {
	if = {
		limit = {
			has_game_rule = banking_system_enabled
			has_journal_entry = je_banking_cycle
		}
		te_history_record_marker = { MARK = $MARK$ }
	}
}
