# Legislated tax code: the economy snapshot (plan Task 14;
# docs/systems/tax_code_schema.md, "Economy snapshot and estimates").
#
# One dated snapshot per country, against which the review and the offers print
# their revenue estimates: today's Budget receipts (the Budget's own getters, in
# the GUI) scaled by each schedule's index over the index in force at the
# snapshot. The snapshot is what dates an estimate: the panels mark it out of
# date once the month, the code or the bill revision has moved on
# (te_tax_view_snap_current).
#
# CALLERS. The monthly processor (te_tax_process_month, inside the month it
# claims, after the collection write: once per month) and te_tax_bill_open_revision
# (introduction, revision and an accepted offer, each a new revision; after the
# bill is open, so the revision is recorded). Never a GUI: opening a panel reads
# the guarded views only. Its variables are a display cache, not schema tokens:
# every call writes all of them, nothing removes them, and a view reads them
# only through has_variable. A civil war's winner may inherit a loser's snapshot
# for a variable it lacks; the code-version and month tests mark it out of date
# until the next 1st.
#
# Country scope; nothing reads ROOT. Variable writes only: no modifier, no
# native setting, no tooltip.
te_tax_take_snapshot = {
	if = {
		limit = {
			te_tax_code_on = yes
			has_variable = te_tax_schema
		}
		set_variable = { name = te_tax_snap_month value = te_history_month_index }
		# The engine's aggregates (te_tax_display_values.txt). tax_income is the
		# domestic tax aggregate (unclassified): shown under that name only.
		set_variable = { name = te_tax_snap_tax_income value = te_tax_read_tax_income }
		set_variable = { name = te_tax_snap_net_fixed value = te_tax_read_net_fixed_income }
		set_variable = { name = te_tax_snap_net_total value = te_tax_read_net_total_income }
		set_variable = { name = te_tax_snap_gdp value = te_tax_read_gdp }
		# What the snapshot is keyed to: the code version and the bill revision
		# (0 without a bill).
		set_variable = { name = te_tax_snap_code_version value = te_tax_view_code_version }
		set_variable = { name = te_tax_snap_bill_rev value = te_tax_view_bl_rev }
		# The index in force per instrument, the base each estimate scales from.
		set_variable = { name = te_tax_snap_wage value = te_tax_view_en_wage }
		set_variable = { name = te_tax_snap_div value = te_tax_view_en_div }
		set_variable = { name = te_tax_snap_land value = te_tax_view_en_land }
		set_variable = { name = te_tax_snap_head value = te_tax_view_en_head }
		set_variable = { name = te_tax_snap_cons value = te_tax_view_en_cons }
		if = {
			limit = { is_ai = no }
			debug_log = "TE_TAX snapshot tax_income=[SCOPE.ScriptValue('te_tax_view_snap_tax_income')|0] version=[SCOPE.ScriptValue('te_tax_view_code_version')|0] rev=[SCOPE.ScriptValue('te_tax_view_bl_rev')|0] month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
	}
}
