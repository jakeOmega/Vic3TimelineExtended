# ============================================================================
# CULTURE MERGES: the yearly census helpers
# (docs/superpowers/specs/2026-11-07-culture-merges-design.md sections 3-4;
# plan Task 4). The per-merge calls are generated into
# te_culture_merges_generated_effects.txt by scripts/generators/gen_culture_merges.py.
#
# State lives in variables, never in modifiers or variable lists, so a civil war's
# winner carries it on. Every share is taken over the part's pops living in its
# homeland states (owner, 2026-11-07): diaspora pops convert at the switch but do not
# weigh on the gates.
#
# Locals are shared by every effect a call runs, so each helper keeps its own prefix
# and resets what it reads: te_cm_cc_ (te_cm_census_culture), te_cm_cp_
# (te_cm_census_part, te_cm_cp_note_holder), te_cm_sg_ (the SoL gap).
# ============================================================================

# Any scope. $CULTURE$ = a culture id.
# Sets the global te_cm_home_<c>: the total_size of the culture's pops living in its
# homeland states. A culture with no owned homeland state (a merged culture before
# its homelands are set) is counted in the world instead, visiting only the
# countries that hold any of it.
te_cm_census_culture = {
	set_local_variable = { name = te_cm_cc_states value = 0 }
	set_local_variable = { name = te_cm_cc_sum value = 0 }
	cu:$CULTURE$ = {
		every_scope_homeland_state = {
			limit = { exists = owner }
			change_local_variable = { name = te_cm_cc_states add = 1 }
			every_scope_pop = {
				limit = { culture = cu:$CULTURE$ }
				change_local_variable = { name = te_cm_cc_sum add = total_size }
			}
		}
	}
	if = {
		limit = { local_var:te_cm_cc_states = 0 }
		every_country = {
			limit = { culture_percent_country = { target = cu:$CULTURE$ value > 0 } }
			every_scope_pop = {
				limit = { culture = cu:$CULTURE$ }
				change_local_variable = { name = te_cm_cc_sum add = total_size }
			}
		}
	}
	set_global_variable = { name = te_cm_home_$CULTURE$ value = local_var:te_cm_cc_sum }
}

# Country scope: a sphere country that holds some of the part. Adds
# local_var:te_cm_cp_state_pops to the temporary te_cm_tmp_cnt, which
# te_cm_census_part reads for the acceptance test and removes before it returns.
te_cm_cp_note_holder = {
	if = {
		limit = { NOT = { has_variable = te_cm_tmp_cnt } }
		set_variable = { name = te_cm_tmp_cnt value = 0 }
	}
	change_variable = { name = te_cm_tmp_cnt add = local_var:te_cm_cp_state_pops }
}

# Country scope: the candidate country. Measures one part of one merge:
#   $MERGE$ = a merge key, $CULTURE$ = a culture id of one of its slots,
#   $TIER$ = the part's tier in script (1, 2 = tier 1b, 3 = tier 2).
# Sets, on the candidate:
#   te_cm_held_<m>_<c>  the sphere's pops of the part in its homeland states over te_cm_home_<c>
#   te_cm_sol_<m>_<c>   those pops' average standard of living
#   te_cm_ok_<m>_<c>    1 while the part passes its gates, with hysteresis: it opens at
#                       hold >= te_cm_hold_open and acceptance >= te_cm_accept_open, and closes
#                       below te_cm_hold_close or when acceptance falls under te_cm_accept_close.
#                       Tier 2 (the spec's 1b) uses the very high pair instead. Acceptance is
#                       tested in every sphere country holding at least te_cm_meaningful_share
#                       of the sphere's pops of the part.
#   te_cm_years_<m>_<c> tiers 2 and 3: a year more while the gate is on; it never goes down
# The sphere is measured against the candidate, so it is saved as te_cm_starter first
# (te_cm_in_sphere reads it). The caller has run te_cm_census_culture for $CULTURE$.
te_cm_census_part = {
	save_temporary_scope_as = te_cm_starter
	set_local_variable = { name = te_cm_cp_tier value = $TIER$ }
	set_local_variable = { name = te_cm_cp_states value = 0 }
	set_local_variable = { name = te_cm_cp_pops value = 0 }
	set_local_variable = { name = te_cm_cp_solw value = 0 }
	set_local_variable = { name = te_cm_cp_state_pops value = 0 }
	if = {
		limit = { NOT = { has_variable = te_cm_ok_$MERGE$_$CULTURE$ } }
		set_variable = { name = te_cm_ok_$MERGE$_$CULTURE$ value = 0 }
	}

	# ---- The sphere's pops of the part, in the part's homeland states ----
	cu:$CULTURE$ = {
		every_scope_homeland_state = {
			limit = { exists = owner }
			change_local_variable = { name = te_cm_cp_states add = 1 }
			if = {
				limit = { owner = { te_cm_in_sphere = yes } }
				set_local_variable = { name = te_cm_cp_state_pops value = 0 }
				every_scope_pop = {
					limit = { culture = cu:$CULTURE$ }
					change_local_variable = { name = te_cm_cp_state_pops add = total_size }
					change_local_variable = { name = te_cm_cp_solw add = { value = standard_of_living multiply = total_size } }
				}
				if = {
					limit = { local_var:te_cm_cp_state_pops > 0 }
					change_local_variable = { name = te_cm_cp_pops add = local_var:te_cm_cp_state_pops }
					owner = { te_cm_cp_note_holder = yes }
				}
			}
		}
	}
	# A culture with no owned homeland state is counted by its pops in the sphere's countries,
	# the same fallback te_cm_census_culture takes for its total.
	if = {
		limit = { local_var:te_cm_cp_states = 0 }
		every_country = {
			limit = {
				te_cm_in_sphere = yes
				culture_percent_country = { target = cu:$CULTURE$ value > 0 }
			}
			set_local_variable = { name = te_cm_cp_state_pops value = 0 }
			every_scope_pop = {
				limit = { culture = cu:$CULTURE$ }
				change_local_variable = { name = te_cm_cp_state_pops add = total_size }
				change_local_variable = { name = te_cm_cp_solw add = { value = standard_of_living multiply = total_size } }
			}
			if = {
				limit = { local_var:te_cm_cp_state_pops > 0 }
				change_local_variable = { name = te_cm_cp_pops add = local_var:te_cm_cp_state_pops }
				te_cm_cp_note_holder = yes
			}
		}
	}

	# ---- Hold and standard of living ----
	set_variable = { name = te_cm_held_$MERGE$_$CULTURE$ value = 0 }
	if = {
		limit = {
			has_global_variable = te_cm_home_$CULTURE$
			global_var:te_cm_home_$CULTURE$ > 0
		}
		set_variable = {
			name = te_cm_held_$MERGE$_$CULTURE$
			value = {
				value = local_var:te_cm_cp_pops
				divide = global_var:te_cm_home_$CULTURE$
				max = 1
			}
		}
	}
	set_variable = { name = te_cm_sol_$MERGE$_$CULTURE$ value = 0 }
	if = {
		limit = { local_var:te_cm_cp_pops >= 1 }
		set_variable = {
			name = te_cm_sol_$MERGE$_$CULTURE$
			value = {
				value = local_var:te_cm_cp_solw
				divide = local_var:te_cm_cp_pops
			}
		}
	}

	# ---- Acceptance, where the part lives ----
	# Every sphere country holding at least the meaningful share of the sphere's pops of the
	# part must accept it. Counted against all four thresholds, so the open/close and
	# accepted/very-high choices below read bare named values.
	set_local_variable = { name = te_cm_cp_fail_open value = 0 }
	set_local_variable = { name = te_cm_cp_fail_close value = 0 }
	set_local_variable = { name = te_cm_cp_fail_high_open value = 0 }
	set_local_variable = { name = te_cm_cp_fail_high_close value = 0 }
	set_local_variable = {
		name = te_cm_cp_min_cnt
		value = {
			value = local_var:te_cm_cp_pops
			multiply = te_cm_meaningful_share
		}
	}
	every_country = {
		limit = { has_variable = te_cm_tmp_cnt }
		if = {
			limit = { var:te_cm_tmp_cnt >= local_var:te_cm_cp_min_cnt }
			if = {
				limit = { country_average_culture_pop_acceptance = { culture = cu:$CULTURE$ value < te_cm_accept_open } }
				change_local_variable = { name = te_cm_cp_fail_open add = 1 }
			}
			if = {
				limit = { country_average_culture_pop_acceptance = { culture = cu:$CULTURE$ value < te_cm_accept_close } }
				change_local_variable = { name = te_cm_cp_fail_close add = 1 }
			}
			if = {
				limit = { country_average_culture_pop_acceptance = { culture = cu:$CULTURE$ value < te_cm_accept_high_open } }
				change_local_variable = { name = te_cm_cp_fail_high_open add = 1 }
			}
			if = {
				limit = { country_average_culture_pop_acceptance = { culture = cu:$CULTURE$ value < te_cm_accept_high_close } }
				change_local_variable = { name = te_cm_cp_fail_high_close add = 1 }
			}
		}
		remove_variable = te_cm_tmp_cnt
	}
	if = {
		limit = { local_var:te_cm_cp_tier = 2 }
		set_local_variable = { name = te_cm_cp_fail_open value = local_var:te_cm_cp_fail_high_open }
		set_local_variable = { name = te_cm_cp_fail_close value = local_var:te_cm_cp_fail_high_close }
	}

	# ---- The gate, with hysteresis ----
	if = {
		limit = { var:te_cm_ok_$MERGE$_$CULTURE$ = 1 }
		if = {
			limit = {
				OR = {
					var:te_cm_held_$MERGE$_$CULTURE$ < te_cm_hold_close
					local_var:te_cm_cp_fail_close > 0
				}
			}
			set_variable = { name = te_cm_ok_$MERGE$_$CULTURE$ value = 0 }
		}
	}
	else = {
		if = {
			limit = {
				var:te_cm_held_$MERGE$_$CULTURE$ >= te_cm_hold_open
				local_var:te_cm_cp_fail_open = 0
			}
			set_variable = { name = te_cm_ok_$MERGE$_$CULTURE$ value = 1 }
		}
	}

	# ---- Years together (tiers 2 and 3) ----
	if = {
		limit = { local_var:te_cm_cp_tier > 1 }
		if = {
			limit = { NOT = { has_variable = te_cm_years_$MERGE$_$CULTURE$ } }
			set_variable = { name = te_cm_years_$MERGE$_$CULTURE$ value = 0 }
		}
		if = {
			limit = { var:te_cm_ok_$MERGE$_$CULTURE$ = 1 }
			change_variable = { name = te_cm_years_$MERGE$_$CULTURE$ add = 1 }
		}
	}
}

# Country scope: folds one passing part into the SoL gap's running range. $MERGE$,
# $CULTURE$. The caller resets te_cm_sg_min (999), te_cm_sg_max (-1) and te_cm_sg_n (0)
# and reads them back as the gap, largest minus smallest.
te_cm_sol_track = {
	change_local_variable = { name = te_cm_sg_n add = 1 }
	if = {
		limit = { var:te_cm_sol_$MERGE$_$CULTURE$ < local_var:te_cm_sg_min }
		set_local_variable = { name = te_cm_sg_min value = var:te_cm_sol_$MERGE$_$CULTURE$ }
	}
	if = {
		limit = { var:te_cm_sol_$MERGE$_$CULTURE$ > local_var:te_cm_sg_max }
		set_local_variable = { name = te_cm_sg_max value = var:te_cm_sol_$MERGE$_$CULTURE$ }
	}
}
