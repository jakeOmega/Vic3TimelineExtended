# ============================================================================
# CULTURE MERGES: console helpers (events/te_debug_cm_events.txt, `event te_debug_cm.1`)
# ============================================================================
# Option D logs every te_cm_* variable of the merges this country is a candidate or starter for, and of every merge that is
# running or complete, to debug.log as TE_CM_DEBUG lines (grep for them; debug.log rolls over at 512 KB). The per-merge call
# list is generated into te_cmg_debug_dump (scripts/generators/gen_culture_merges.py), because the variable names embed the
# merge key and the culture. These helpers read one merge, or one part of it.
#
# A variable is copied into a fixed slot global (te_cm_dbg_<slot>) and the line reads the slot back, as the self-checks do:
# a variable name with a $X$ inside a [..] accessor is unproven, and an unset one reads as -1 here instead of a blank.

# Any scope. Copies the global $NAME$ into the slot global te_cm_dbg_$SLOT$, -1 when it is not set.
te_cm_dbg_global = {
	set_global_variable = { name = te_cm_dbg_$SLOT$ value = -1 }
	if = {
		limit = { has_global_variable = $NAME$ }
		set_global_variable = { name = te_cm_dbg_$SLOT$ value = global_var:$NAME$ }
	}
}

# Country scope. Copies this country's variable $NAME$ into the slot global te_cm_dbg_$SLOT$, -1 when it is not set.
te_cm_dbg_var = {
	set_global_variable = { name = te_cm_dbg_$SLOT$ value = -1 }
	if = {
		limit = { has_variable = $NAME$ }
		set_global_variable = { name = te_cm_dbg_$SLOT$ value = var:$NAME$ }
	}
}

# Country scope. One merge's own variables and where this country stands in it. $MERGE$ = a merge key.
# state 0 idle, 1 running, 2 complete; starter_exists / we_are_starter 1 or 0; candidate, core_ok and can_start 1 or 0.
# Two forms, so a merge with no core wait never names te_cm_wait_years_<m>, which only a core wait sets: a read of a name
# nothing sets logs "is used but is never set" at every load. te_cmg_debug_dump picks the form.
te_cm_debug_dump_merge = {
	te_cm_debug_dump_merge_slots = { MERGE = $MERGE$ }
	set_global_variable = { name = te_cm_dbg_j value = -1 }
	te_cm_debug_dump_merge_line = { MERGE = $MERGE$ }
}

# Country scope. The same, for a merge with a core part that waits on a vanilla entry: its wait counter too.
te_cm_debug_dump_merge_waiting = {
	te_cm_debug_dump_merge_slots = { MERGE = $MERGE$ }
	te_cm_dbg_var = { SLOT = j NAME = te_cm_wait_years_$MERGE$ }
	te_cm_debug_dump_merge_line = { MERGE = $MERGE$ }
}

# Country scope. Every slot of the merge line but j (the wait counter).
te_cm_debug_dump_merge_slots = {
	te_cm_dbg_global = { SLOT = a NAME = te_cm_state_$MERGE$ }
	set_global_variable = { name = te_cm_dbg_b value = 0 }
	if = {
		limit = { exists = global_var:te_cm_starter_$MERGE$ }
		set_global_variable = { name = te_cm_dbg_b value = 1 }
	}
	set_global_variable = { name = te_cm_dbg_c value = 0 }
	if = {
		limit = { te_cmg_$MERGE$_is_starter = yes }
		set_global_variable = { name = te_cm_dbg_c value = 1 }
	}
	set_global_variable = { name = te_cm_dbg_d value = 0 }
	if = {
		limit = { te_cmg_$MERGE$_is_candidate = yes }
		set_global_variable = { name = te_cm_dbg_d value = 1 }
	}
	set_global_variable = { name = te_cm_dbg_e value = 0 }
	if = {
		limit = { te_cmg_$MERGE$_core_ok = yes }
		set_global_variable = { name = te_cm_dbg_e value = 1 }
	}
	set_global_variable = { name = te_cm_dbg_f value = 0 }
	if = {
		limit = { te_cmg_$MERGE$_can_start = yes }
		set_global_variable = { name = te_cm_dbg_f value = 1 }
	}
	te_cm_dbg_var = { SLOT = g NAME = te_cm_sol_ok_$MERGE$ }
	te_cm_dbg_var = { SLOT = h NAME = te_cm_started_$MERGE$ }
	te_cm_dbg_var = { SLOT = i NAME = te_cm_bar_$MERGE$ }
	te_cm_dbg_var = { SLOT = k NAME = te_cm_converted_$MERGE$ }
}

# Country scope. The merge line, from the slots.
te_cm_debug_dump_merge_line = {
	debug_log = "TE_CM_DEBUG merge $MERGE$ country=[THIS.GetCountry.GetTagName] state=[GetGlobalVariable('te_cm_dbg_a').GetValue|0] starter_exists=[GetGlobalVariable('te_cm_dbg_b').GetValue|0] we_are_starter=[GetGlobalVariable('te_cm_dbg_c').GetValue|0] candidate=[GetGlobalVariable('te_cm_dbg_d').GetValue|0] core_ok=[GetGlobalVariable('te_cm_dbg_e').GetValue|0] can_start=[GetGlobalVariable('te_cm_dbg_f').GetValue|0] sol_ok=[GetGlobalVariable('te_cm_dbg_g').GetValue|0] started=[GetGlobalVariable('te_cm_dbg_h').GetValue|0] bar=[GetGlobalVariable('te_cm_dbg_i').GetValue|0] wait_years=[GetGlobalVariable('te_cm_dbg_j').GetValue|0] converted=[GetGlobalVariable('te_cm_dbg_k').GetValue|0] (-1 = not set)"
}

# Country scope. One slot culture of one merge. $MERGE$ = a merge key, $CULTURE$ = a culture id, $SOL_OK$ = the
# standard-of-living gate the part answers to: its own te_cm_sol_ok_<m>_<c> for an optional part, the merge's
# te_cm_sol_ok_<m> for a core part (only optional parts have their own, so naming one for a core part reads a name
# nothing sets).
# home / home_r: the census's pops in the part's homeland states (the part's, then the result's, once the part has joined);
# joined, in (the index of the merge converting it), dom (the result's share), switched 1 or 0 are global; held, ok, sol,
# sol_ok, years and paused are this country's.
te_cm_debug_dump_part = {
	te_cm_dbg_global = { SLOT = a NAME = te_cm_home_$CULTURE$ }
	te_cm_dbg_global = { SLOT = b NAME = te_cm_home_r_$CULTURE$ }
	te_cm_dbg_global = { SLOT = c NAME = te_cm_joined_$MERGE$_$CULTURE$ }
	te_cm_dbg_global = { SLOT = d NAME = te_cm_dom_$MERGE$_$CULTURE$ }
	te_cm_dbg_global = { SLOT = e NAME = te_cm_in_$CULTURE$ }
	set_global_variable = { name = te_cm_dbg_f value = 0 }
	if = {
		limit = { has_global_variable = te_cm_switched_$CULTURE$ }
		set_global_variable = { name = te_cm_dbg_f value = 1 }
	}
	te_cm_dbg_var = { SLOT = g NAME = te_cm_held_$MERGE$_$CULTURE$ }
	te_cm_dbg_var = { SLOT = h NAME = te_cm_ok_$MERGE$_$CULTURE$ }
	te_cm_dbg_var = { SLOT = i NAME = te_cm_sol_$MERGE$_$CULTURE$ }
	te_cm_dbg_var = { SLOT = j NAME = $SOL_OK$ }
	te_cm_dbg_var = { SLOT = k NAME = te_cm_years_$MERGE$_$CULTURE$ }
	te_cm_dbg_var = { SLOT = l NAME = te_cm_paused_$MERGE$_$CULTURE$ }
	debug_log = "TE_CM_DEBUG part $MERGE$/$CULTURE$ home=[GetGlobalVariable('te_cm_dbg_a').GetValue|0] home_r=[GetGlobalVariable('te_cm_dbg_b').GetValue|0] joined=[GetGlobalVariable('te_cm_dbg_c').GetValue|0] dom=[GetGlobalVariable('te_cm_dbg_d').GetValue|3] in=[GetGlobalVariable('te_cm_dbg_e').GetValue|0] switched=[GetGlobalVariable('te_cm_dbg_f').GetValue|0] held=[GetGlobalVariable('te_cm_dbg_g').GetValue|3] ok=[GetGlobalVariable('te_cm_dbg_h').GetValue|0] sol=[GetGlobalVariable('te_cm_dbg_i').GetValue|2] sol_ok=[GetGlobalVariable('te_cm_dbg_j').GetValue|0] years=[GetGlobalVariable('te_cm_dbg_k').GetValue|0] paused=[GetGlobalVariable('te_cm_dbg_l').GetValue|0] (-1 = not set)"
}
