# THE WORLD DEVELOPMENT FUND (development_fund; no reform needed)
#   One topic, one institution (owner, 2026-10-02). It can be founded under
#   any charter, and once it stands its form follows the charter level
#   (un_charter_level_value) every month, with no new vote:
#     founding charter  voluntary: the contributions only; line a tenth of
#                       the members' average GDP per head
#     Reform I          assessed: 5% of the budget + the contributions; line a
#                       quarter of the average
#     Reform II         full: a quarter of the budget + the contributions; line
#                       half the average
#   (un_dev_fund_budget_share_reform_1/_2, un_charter_values.txt;
#   un_dev_fund_line_share_founding/_reform_1/_reform_2, un_economy_values.txt.)
#   THE VOLUNTARY PART is the Development Programs programme
#   (un_fund_development_button, un_buttons.txt; owner, 2026-10-02): while the
#   Fund stands, every represented member that runs it and pays its dues
#   (un_dev_fund_contributor) pays the Fund's grants with its expense
#   (var:un_development_expense_cached, the multiplier of
#   un_development_contributor_cost). Since 2026-10-05 (owner) it has four
#   sizes, Token to Generous (0.125% to 1% of GDP a year), any member may give
#   whatever its rank, and its benefits (infamy decay, prestige, standing, a
#   lobbying bonus with the recipients) grow by a quarter of Substantial's
#   with each doubling (un_economy_values.txt). The UN's funding pillar counts
#   the money (un_funding_dev_money_ratio_value, un_authority_values.txt).
#   global   un_inst_development_fund    in force (un_development_fund_on_carried)
#            un_dev_fund_members_gdp /   the represented members' GDP and people
#            un_dev_fund_members_pop
#            un_dev_fund_avg             their GDP per head
#            un_dev_fund_line_floor      the poorest member in good standing's
#                                        GDP per head (un_dev_fund_good_standing);
#                                        absent while there is none
#            un_dev_fund_line_base       the level's share of the average
#            un_dev_fund_line            the recipients' line: the base, raised
#                                        to the floor when no member in good
#                                        standing is under it (owner,
#                                        2026-10-03), so one always qualifies
#            un_dev_fund_rich_line       twice it: the richest (the lean)
#            un_dev_fund_eligible_pop    the people of the members below the line
#            un_dev_fund_recipients      how many members they are
#            un_dev_fund_donations       the contributors' programme expense a
#                                        week, together (the voluntary part)
#            un_dev_fund_contributors    how many they are
#   country  un_dev_fund_size            our contribution's size, 1..4 (kept after
#                                        it ends; read through
#                                        un_dev_fund_size_value)
#            un_dev_fund_benefit         its benefit factor: the multiplier of
#                                        un_development_contributor_modifier
#            un_dev_fund_patron          what the recipients remember of our
#                                        money: the multiplier of
#                                        un_development_patron_modifier
#                                        (un_dev_fund_patron_pulse)
#            un_dev_fund_gdp_ph          GDP per head (represented members only)
#            un_dev_fund_grant           this month's weekly grant: the
#                                        multiplier of the grant modifier. Set
#                                        to 0 when the grant stops, never removed
#                                        while the UN stands (the engine re-reads
#                                        a stored multiplier)
#   modifier un_dev_fund_pool_grant_modifier / un_dev_fund_treasury_grant_modifier,
#            one or the other: un_dev_fund_country_refresh (un_economy.1) is
#            the single refresh site, removing and re-adding monthly
#   Fund Development Programs opens only while the Fund stands (owner,
#   2026-10-03), and a contribution running with no Fund lapses at the next
#   monthly update (un_dev_fund_lapse_contributions).
#   The snapshot, the contributions and the grants are written by
#   un_dev_fund_monthly_update (every month, from un_charter_monthly_update,
#   after the budget snapshot in un_authority_monthly_update). Leaving the UN,
#   withholding dues or rising above the line stops the grant at the next
#   month; a contributor that withholds its dues or loses its seat stops
#   counting toward the pot then too.
#
# THE WORLD FOOD RESERVE (food_reserve; no reform)
#   global   un_inst_food_reserve        in force (un_food_reserve_on_carried)
#            un_fr_members_pop /         the members' people, those of them in
#            un_fr_famine_pop            famine-struck states, and the share
#            un_fr_famine_share          (un_food_reserve_hunger_snapshot, from
#                                        un_authority_monthly_update step 3)
#            un_fr_hunger_term_potential what hunger would give delivery
#            un_fr_hunger_term           what it gives: the potential while the
#                                        Reserve stands, else 0
#            un_fr_pledgers / un_fr_pledged_grain
#                                        pledgers and the grain they hold
#            un_fr_holders / un_fr_holders_grain
#                                        members holding grain worth drawing
#            un_fr_draw_potential / un_fr_draw_scale / un_fr_draw_total
#                                        one draw's bookkeeping, removed after it
#   country  un_fr_pledged               we pledge (the pledge buttons)
#            un_fr_last_draw             grain the last draw took from us
#            un_fr_drawn_total           grain the Reserve has taken from us
#   state    un_fr_relief_power          the relief modifier's multiplier, 0..1
#            un_fr_relief_months         pulses of relief left
#   modifier un_food_reserve_relief_state_modifier: un_food_reserve_state_refresh
#            (the state's monthly pulse) is the single refresh site
#
#   THE STRATEGIC RESERVE. A draw lowers the pledger's st_res_grain_stored
#   (never below zero: a draw is at most a quarter of what it holds) and then
#   hands the pledger to un_economy.2, which runs the Strategic Reserve's own
#   clamp and hub refresh with the pledger as ROOT, as the reserve's rate
#   buttons do (st_res_clamp_stockpiles_effect, st_res_refresh_hub_flow_effect).
#   No goods are created: the grain leaves the stockpile the pledger bought,
#   and the relief is a modifier on the state. No Strategic Reserve file is
#   touched.
#
# REASON CODES (allocated 34-36 ledger, 42-44 standing)
#   ledger   34  the World Food Reserve sent grain to <state> (delivery,
#                institutional; subject: the aid mission's state)
#            35, 36  allocated, unused
#   standing 42  the World Food Reserve drew on the grain we pledged

# The World Development Fund carried: the Fund stands from now, in the form
# the charter's level gives it, and its first grants are paid from the next
# monthly update. Recommendatory, so there is no vetoed form.
# Scope: un_vote.2's ROOT (the proposer); expects scope:un_resolution
un_development_fund_on_carried = {
	custom_tooltip = un_vote_development_fund_passed_tt
	un_ledger_entry = { PILLAR = credibility POINTS = 1 REASON = 1 }
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_variable = un_inst_development_fund } }
			set_global_variable = un_inst_development_fund
		}
	}
}

		limit = { un_charter_has_reform = { LEVEL = 1 } }

	else_if = {
		limit = { un_dev_fund_line_floored = yes }
		custom_tooltip_no_bullet = je_un_chamber_preview_development_fund_voluntary_floored
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_preview_development_fund_voluntary
	}
	# Under the founding charter the Fund would pay out only the contributions,
	# and none can run before it stands (un_fund_development_button), so there
	# is no sum to preview: the line says we would share in them.
	if = {
		limit = {
			un_dev_fund_eligible = yes

	}
	else_if = {
		limit = { has_variable = un_dev_fund_gdp_ph }
		custom_tooltip_no_bullet = je_un_chamber_preview_development_fund_not_ours
	}

# Once a month, from un_charter_monthly_update (un_global_authority_on_action,
# the UN founded), after the authority update has snapshotted the budget.
# Scope: none
un_economy_monthly_update = {
	un_dev_fund_monthly_update = yes
	un_food_reserve_monthly_update = yes
}

# The Fund's line, recipients and contributions, snapshotted every month
# whether or not the Fund stands (the chamber's row and the lean read them), at
# the charter's current level; then this month's grants while it stands.
# Scope: none
un_dev_fund_monthly_update = {
	# 0. A Development Programs contribution running with no Fund lapses.
	un_dev_fund_lapse_contributions = yes
	# 1. Every represented member's GDP per head; a stale figure is dropped.
	every_country = {
		if = {
			limit = { un_member_represented = yes }
			set_variable = { name = un_dev_fund_gdp_ph value = un_dev_fund_gdp_per_head }
		}
		else_if = {
			limit = { has_variable = un_dev_fund_gdp_ph }
			remove_variable = un_dev_fund_gdp_ph
		}
	}
	# 2. The members' average, the line (at the charter's current level, never
	#    below the poorest member in good standing: owner, 2026-10-03) and the
	#    rich line. The floor reads step 1's figures and goes before the line.
	set_global_variable = { name = un_dev_fund_members_gdp value = un_dev_fund_members_gdp_value }
	set_global_variable = { name = un_dev_fund_members_pop value = un_dev_fund_members_pop_value }
	set_global_variable = { name = un_dev_fund_avg value = un_dev_fund_avg_value }
	if = {
		limit = {
			any_country = {
				un_dev_fund_good_standing = yes
				has_variable = un_dev_fund_gdp_ph
			}
		}
		ordered_country = {
			limit = {
				un_dev_fund_good_standing = yes
				has_variable = un_dev_fund_gdp_ph
			}
			order_by = un_dev_fund_poverty_order
			position = 0
			set_global_variable = { name = un_dev_fund_line_floor value = var:un_dev_fund_gdp_ph }
		}
	}
	else_if = {
		limit = { has_global_variable = un_dev_fund_line_floor }
		remove_global_variable = un_dev_fund_line_floor
	}
	set_global_variable = { name = un_dev_fund_line_base value = un_dev_fund_line_base_value }
	set_global_variable = { name = un_dev_fund_line value = un_dev_fund_line_value }
	set_global_variable = { name = un_dev_fund_rich_line value = un_dev_fund_rich_line_value }
	# 3. Who the Fund pays, and their people.
	set_global_variable = { name = un_dev_fund_eligible_pop value = un_dev_fund_eligible_pop_value }
	set_global_variable = { name = un_dev_fund_recipients value = un_dev_fund_recipients_value }
	# 4. The voluntary part: the Development Programs contributions of the
	#    represented members that pay their dues, a week, and how many they
	#    are. The pot adds the sum while the Fund stands, so the grants pay out
	#    what the contributors pay.
	set_global_variable = { name = un_dev_fund_donations value = un_dev_fund_donations_value }
	set_global_variable = { name = un_dev_fund_contributors value = un_dev_fund_contributors_value }
	# 5. This month's grants. A recipient's grant is stored here and applied
	#    by un_economy.1, whose ROOT is the recipient: a multiplier read from
	#    this root-less pulse would resolve against nothing and silently apply
	#    1 (scripting_best_practices.md, "multiplier = var:X Resolves Against
	#    ROOT"). A country whose grant has stopped is sent the same event, which
	#    removes its modifier.
	every_country = {
		limit = {
			OR = {
				AND = {
					has_global_variable = un_inst_development_fund
					un_dev_fund_eligible = yes
				}
				AND = {
					has_variable = un_dev_fund_grant
					var:un_dev_fund_grant > 0
				}
				has_modifier = un_dev_fund_pool_grant_modifier
				has_modifier = un_dev_fund_treasury_grant_modifier
			}
		}
		if = {
			limit = {
				has_global_variable = un_inst_development_fund
				un_dev_fund_eligible = yes
			}
			set_variable = { name = un_dev_fund_grant value = un_dev_fund_grant_value }
		}
		else = {
			set_variable = { name = un_dev_fund_grant value = 0 }
		}
		trigger_event = { id = un_economy.1 }
	}
}

# Fund Development Programs opens only while the Fund stands (owner,
# 2026-10-03; un_fund_development_button's possible): before the Assembly
# founds it, the programme's money would reach nobody. A contribution that
# runs with no Fund (one from a save before the gate) ends here, exactly as
# Stop Contributing ends it: both journal-entry modifiers and their mirrors go
# (un_state_off, as leaving the UN does), and the contributor is told why.
# Scope: none
un_dev_fund_lapse_contributions = {
	if = {
		limit = { NOT = { has_global_variable = un_inst_development_fund } }
		every_country = {
			limit = {
				je:je_united_nations ?= {
					OR = {
						has_modifier = un_development_contributor_modifier
						has_modifier = un_development_contributor_cost
					}
				}
			}
			un_state_off = { MODIFIER = un_development_contributor_modifier }
			un_state_off = { MODIFIER = un_development_contributor_cost }
			post_notification = un_dev_fund_contribution_lapsed_notice
		}
	}
}

# ---- The voluntary part: sizes and the recipients' memory ---------------------------

# Raise or Reduce a contribution a size (un_raise_development_button,
# un_reduce_development_button): the tooltip names the size it moves to and
# what it costs and does, then un_dev_fund_resize makes the move.
# Scope: country (the contributor, = ROOT)
un_dev_fund_raise = {
	if = {
		limit = { un_dev_fund_size_stored >= 3 }
		custom_tooltip = un_dev_fund_to_generous_tt
	}
	else_if = {
		limit = { un_dev_fund_size_stored >= 2 }
		custom_tooltip = un_dev_fund_to_substantial_tt
	}
	else = {
		custom_tooltip = un_dev_fund_to_modest_tt
	}
	un_dev_fund_resize = { STEP = 1 }
}

un_dev_fund_reduce = {
	if = {
		limit = { un_dev_fund_size_stored >= 4 }
		custom_tooltip = un_dev_fund_to_substantial_tt
	}
	else_if = {
		limit = { un_dev_fund_size_stored >= 3 }
		custom_tooltip = un_dev_fund_to_modest_tt
	}
	else = {
		custom_tooltip = un_dev_fund_to_token_tt
	}
	un_dev_fund_resize = { STEP = -1 }
}

# Move the contribution $STEP$ sizes, held to 1..4, and re-add both modifiers
# at the new size: the benefits at its factor, the cost at its expense. Both
# multipliers read root.var, so ROOT must be the contributor (a button's
# effect). un_state_off takes each modifier off with its mirror and the add
# records it again: the removes read the modifiers the entry held before the
# click, which a button's own block can see.
# Parameters: $STEP$ = 1 | -1
# Scope: country (the contributor, = ROOT)
un_dev_fund_resize = {
	hidden_effect = {
		set_variable = { name = un_dev_fund_size value = un_dev_fund_size_stored }
		change_variable = { name = un_dev_fund_size add = $STEP$ }
		clamp_variable = { name = un_dev_fund_size min = 1 max = un_dev_fund_size_max }
		set_variable = { name = un_dev_fund_benefit value = un_dev_fund_benefit_value }
		set_variable = { name = un_development_expense_cached value = un_dev_fund_expense_value }
		un_state_off = { MODIFIER = un_development_contributor_modifier }
		un_state_off = { MODIFIER = un_development_contributor_cost }
		je:je_united_nations ?= { add_modifier = {
				name = un_development_contributor_modifier
				multiplier = root.var:un_dev_fund_benefit
			} }
		un_state_record = { MODIFIER = un_development_contributor_modifier }
		je:je_united_nations ?= { add_modifier = {
				name = un_development_contributor_cost
				multiplier = root.var:un_development_expense_cached
			} }
		un_state_record = { MODIFIER = un_development_contributor_cost }
	}
}

# A contribution's month, from je_united_nations' member block (ROOT = the
# member): a contribution from before the sizes becomes Substantial, the size
# it paid for; the benefit factor and the expense follow the size and today's
# GDP. The benefit modifier is re-added only for a contribution that was
# migrated (a save from before the sizes holds it with no multiplier, which is
# Substantial already, so nothing changes but the variable it will read).
# Scope: country (= ROOT)
un_dev_fund_contribution_monthly = {
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_development_contributor_modifier } }
		if = {
			limit = { NOT = { has_variable = un_dev_fund_size } }
			set_variable = { name = un_dev_fund_size value = un_dev_fund_size_legacy }
		}
		set_variable = { name = un_dev_fund_benefit value = un_dev_fund_benefit_value }
		set_variable = { name = un_development_expense_cached value = un_dev_fund_expense_value }
	}
}

# What the Fund's recipients remember of our money, once a month, from
# je_united_nations' pulse (ROOT = the country): var:un_dev_fund_patron moves
# a twelfth of the way toward our benefit factor while our money reaches the
# Fund, toward 0 when it does not; un_development_patron_modifier is
# re-applied at it on the journal entry (the single refresh site; derived,
# so not mirrored: a revolution's winner inherits the variable and gets the
# modifier back here). Runs for a member and a non-member alike, so the
# memory fades after a contribution, or a membership, ends.
# Scope: country (= ROOT)
un_dev_fund_patron_pulse = {
	if = {
		limit = {
			OR = {
				un_dev_fund_contributor = yes
				AND = {
					has_variable = un_dev_fund_patron
					var:un_dev_fund_patron > 0
				}
			}
		}
		set_variable = { name = un_dev_fund_patron value = un_dev_fund_patron_next_value }
		if = {
			limit = {
				NOT = { un_dev_fund_contributor = yes }
				var:un_dev_fund_patron < un_dev_fund_patron_floor
			}
			set_variable = { name = un_dev_fund_patron value = 0 }
		}
		je:je_united_nations ?= {
			if = {
				limit = { has_modifier = un_development_patron_modifier }
				remove_modifier = un_development_patron_modifier
			}
		}
		if = {
			limit = { var:un_dev_fund_patron > 0 }
			je:je_united_nations ?= { add_modifier = {
					name = un_development_patron_modifier
					multiplier = root.var:un_dev_fund_patron
				} }
		}
	}
}

# One country's grant, from its own hidden event (un_economy.1): the single
# refresh site for both grant modifiers. Into the investment pool where the
# pool can take it, into the treasury otherwise.
# Scope: the country (and ROOT)
un_dev_fund_country_refresh = {
	if = {
		limit = { has_modifier = un_dev_fund_pool_grant_modifier }
		remove_modifier = un_dev_fund_pool_grant_modifier
	}
	if = {
		limit = { has_modifier = un_dev_fund_treasury_grant_modifier }
		remove_modifier = un_dev_fund_treasury_grant_modifier
	}
	if = {
		limit = {
			has_global_variable = un_inst_development_fund
			has_variable = un_dev_fund_grant
			var:un_dev_fund_grant > 0
		}
		if = {
			limit = { un_dev_fund_pays_into_pool = yes }
			add_modifier = {
				name = un_dev_fund_pool_grant_modifier
				multiplier = var:un_dev_fund_grant
			}
		}
		else = {
			add_modifier = {
				name = un_dev_fund_treasury_grant_modifier
				multiplier = var:un_dev_fund_grant
			}
		}
	}
}

# The Reserve's month: pledges that no longer qualify lapse, and the chamber's
# figures for the pledgers and for everyone holding grain.
# Scope: none
un_food_reserve_monthly_update = {
	every_country = {
		limit = {
			has_variable = un_fr_pledged
			NOT = { un_food_reserve_can_pledge = yes }
		}
		remove_variable = un_fr_pledged
	}
	set_global_variable = { name = un_fr_pledgers value = 0 }
	set_global_variable = { name = un_fr_pledged_grain value = 0 }
	set_global_variable = { name = un_fr_holders value = 0 }
	set_global_variable = { name = un_fr_holders_grain value = 0 }
	every_country = {
		limit = {
			un_member_represented = yes
			un_food_reserve_can_store_grain = yes
			un_food_reserve_holds_grain = yes
		}
		change_global_variable = { name = un_fr_holders add = 1 }
		change_global_variable = { name = un_fr_holders_grain add = var:st_res_grain_stored }
		if = {

# ---- The World Food Reserve answers an aid mission ------------------------------------

# A new UN aid mission has opened (un_mission_open, un_mission_effects.txt).
# While the Reserve stands, it draws grain from the pledgers holding enough,
# other than the mission's host (not drawn on for its own relief):
# from each the same fraction of its offer (a quarter of its stockpile, at most
# 2,500 units), all of it until the mission's need of 5,000 units is met. The
# mission's state is fed for a year in proportion to what came; each pledger
# drawn on gains standing; the UN books the delivery. With nothing to draw the
# host is told so, and nothing else happens. A Moribund UN cannot call on it.
# Scope: any; expects scope:un_msn_new (the mission), scope:un_msn_state and
# scope:un_msn_host
un_food_reserve_on_aid_mission_opened = {
	hidden_effect = {
		if = {
			limit = {
				has_global_variable = un_inst_food_reserve
				exists = scope:un_msn_state
				un_tier_at_least_contested = yes
			}
			set_global_variable = { name = un_fr_draw_potential value = un_fr_draw_potential_value }
			if = {
				limit = { global_var:un_fr_draw_potential > 0 }
				set_global_variable = { name = un_fr_draw_scale value = un_fr_draw_scale_value }
				set_global_variable = { name = un_fr_draw_total value = 0 }
				# Mirrors un_fr_draw_potential_value's limit: the host is
				# never drawn on for its own relief.
				every_country = {
					limit = {
						un_food_reserve_can_be_drawn = yes
						NOT = { scope:un_msn_host ?= this }
					}
					un_food_reserve_draw_from = yes
				}
				# The relief, for a year from the state's next monthly pulse:
				# its strength is what came against what the mission needs.
				scope:un_msn_state = {
					set_variable = { name = un_fr_relief_power value = un_fr_relief_power_value }
					set_variable = { name = un_fr_relief_months value = un_fr_relief_duration }
				}
				un_ledger_entry_on = { PILLAR = delivery POINTS = un_fr_delivery_points_value REASON = 34 SUBJECT = scope:un_msn_state }
				scope:un_msn_host ?= { post_notification = un_food_reserve_relief_notice }
				remove_global_variable = un_fr_draw_scale
				remove_global_variable = un_fr_draw_total
			}
			else = {
				scope:un_msn_host ?= { post_notification = un_food_reserve_empty_notice }
			}
			remove_global_variable = un_fr_draw_potential
		}
	}
}

# One pledger's part in a draw: the grain leaves its Strategic Reserve, the
# Reserve's own clamp and hub refresh follow from its own hidden event, it is
# told, and its standing rises.
# Scope: the pledger; expects global_var:un_fr_draw_scale and
# global_var:un_fr_draw_total, and scope:un_msn_state for the notice
un_food_reserve_draw_from = {
	set_variable = { name = un_fr_last_draw value = un_fr_draw_value }
	change_variable = { name = st_res_grain_stored subtract = var:un_fr_last_draw }
	# A draw is at most a quarter of the stockpile, so this never bites; it
	# is the floor the Strategic Reserve's bookkeeping assumes.
	if = {
		limit = { var:st_res_grain_stored < 0 }
		set_variable = { name = st_res_grain_stored value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = un_fr_drawn_total } }
		set_variable = { name = un_fr_drawn_total value = 0 }
	}
	change_variable = { name = un_fr_drawn_total add = var:un_fr_last_draw }
	change_global_variable = { name = un_fr_draw_total add = var:un_fr_last_draw }
	un_standing_gain = { AMOUNT = un_fr_pledger_standing_gain REASON = 42 }
	post_notification = un_food_reserve_drawn_notice
	trigger_event = { id = un_economy.2 }
}

# One state's relief, from its own monthly pulse (un_economy_on_actions.txt):
# the single refresh site for un_food_reserve_relief_state_modifier. Applied
# while pulses of relief are left, then removed with its variables.
# Scope: state (and ROOT)
un_food_reserve_state_refresh = {
	if = {
		limit = {
			has_variable = un_fr_relief_power
			has_variable = un_fr_relief_months
			var:un_fr_relief_months > 0
		}
		if = {
			limit = { has_modifier = un_food_reserve_relief_state_modifier }
			remove_modifier = un_food_reserve_relief_state_modifier
		}
		add_modifier = {
			name = un_food_reserve_relief_state_modifier
			multiplier = var:un_fr_relief_power
		}
		change_variable = { name = un_fr_relief_months subtract = 1 }
	}
	else = {
		# The modifier goes before the variable its multiplier reads.
		if = {
			limit = { has_modifier = un_food_reserve_relief_state_modifier }
			remove_modifier = un_food_reserve_relief_state_modifier
		}
		if = {
			limit = { has_variable = un_fr_relief_power }
			remove_variable = un_fr_relief_power
		}
		if = {
			limit = { has_variable = un_fr_relief_months }
			remove_variable = un_fr_relief_months
		}
	}
}
