# ============================================================================
# SOLAR SYSTEM COLONIZATION — colony population: emigration, crews, the bonus
# ============================================================================
# Design: docs/superpowers/specs/2026-10-25-solar-colonization-population-and-transit-design.md
# (slice 2 of the scope, § 10). Simple and full colonies only.
#
# Each world's people are on its global record (sr_site_pop_<key>), so a
# cession or an heir takes them along; growth runs in the world's monthly step
# (sr_site_monthly_<key>). The holder's emigration programme (sr_emig_level:
# 0 Off, 1 Low, 2 Steady, 3 High) takes people from the Space Program's state
# each month and shares them by its priorities (sr_emig_prio_<key>: 0 Closed,
# 1 Low, unset Normal, 3 High) times each colony's free room. The per-world
# parts are generated (sr_emig_distribute, sr_emig_weight_<key>,
# sr_emig_step_prio_from_scope); this file holds the rest.
# ============================================================================

# ---- The month's emigrants (the entry's pulse) --------------------------------
# Country scope. The month's settlers (sr_emig_month_batches: the step's amount,
# never more than the open colonies' free room or 1% of our people) come from
# every state we own, in proportion to its people (colony people design § 6.3;
# the owner, 2026-11-22: "Spread by population"); the Python twin is
# scripts/analysis/sr_settler_draw.py, and test_sr_settler_draw.py holds both to
# one rule. In batches of sr_emig_batch_people (half middle strata, half lower;
# kill_population_in_state takes a constant):
#   1. each state's due is the share x its people + its carry (sr_emig_carry,
#      what it owed and didn't give before), one pass reading only
#      state_population, so no count of its pops;
#   2. largest due first, each state gives its whole batches while the month has
#      batches left (local sr_emig_left), and keeps the rest as its carry; a
#      state the month runs out before keeps its whole due, so it comes earlier
#      next month.
# Over a few months every state gives its share, within a batch, and no month
# sends more than its settlers. The census books each state's take as known
# moves out, by Organized Colonization's profile (te_demog_sr_note_settlers).
# Under Full Colonies none leave while Earth is blockaded or the launch site is
# occupied (slice 6 fleets design § 15).
sr_emig_month = {
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			sr_emig_month_batches >= 1
			sr_has_space_program = yes
			NOT = { sr_launches_barred = yes }
		}
		set_local_variable = { name = sr_emig_left value = sr_emig_month_batches }
		set_local_variable = { name = sr_emig_share value = sr_emig_share }
		set_local_variable = { name = sr_emig_sent value = 0 }
		set_local_variable = { name = sr_emig_fell_before value = 0 }
		set_local_variable = { name = sr_emig_fell_after value = 0 }
		# 1. Each state's due.
		every_scope_state = {
			set_variable = {
				name = sr_emig_due
				value = {
					value = state_population
					multiply = local_var:sr_emig_share
					if = { limit = { has_variable = sr_emig_carry } add = var:sr_emig_carry }
				}
			}
			if = { limit = { has_variable = sr_emig_carry } remove_variable = sr_emig_carry }
		}
		# 2. The take, largest due first; every state, then each keeps what it owes.
		ordered_scope_state = {
			limit = { has_variable = sr_emig_due }
			order_by = sr_emig_due_value
			max = 1000
			check_range_bounds = no
			if = {
				limit = {
					local_var:sr_emig_left >= 1
					sr_emig_due_value >= sr_emig_batch_people
				}
				set_local_variable = {
					name = sr_emig_take
					value = {
						value = sr_emig_due_value
						divide = sr_emig_batch_people
						floor = yes
						max = local_var:sr_emig_left
					}
				}
				set_local_variable = {
					name = sr_emig_took_now
					value = {
						value = local_var:sr_emig_take
						multiply = sr_emig_batch_people
					}
				}
				change_local_variable = { name = sr_emig_left subtract = local_var:sr_emig_take }
				change_local_variable = { name = sr_emig_sent add = local_var:sr_emig_took_now }
				# The emigration_batch self-check's reading (sr_emig_self_check).
				change_local_variable = { name = sr_emig_fell_before add = state_population }
				while = {
					count = local_var:sr_emig_take
					kill_population_in_state = { value = 250 strata = middle }
					kill_population_in_state = { value = 250 strata = lower }
				}
				change_local_variable = { name = sr_emig_fell_after add = state_population }
				te_demog_sr_note_settlers = yes
				change_variable = { name = sr_emig_due subtract = local_var:sr_emig_took_now }
			}
			if = {
				limit = { var:sr_emig_due > 0 }
				set_variable = { name = sr_emig_carry value = var:sr_emig_due }
			}
			remove_variable = sr_emig_due
		}
		# What the distribution shares out, the bill charges and the panel shows.
		if = {
			limit = { local_var:sr_emig_sent > 0 }
			sr_emig_distribute = yes
			set_variable = { name = sr_emig_last_sent value = local_var:sr_emig_sent }
			sr_emig_self_check = yes
		}
		else_if = {
			limit = { has_variable = sr_emig_last_sent }
			remove_variable = sr_emig_last_sent
		}
	}
	else_if = {
		limit = { has_variable = sr_emig_last_sent }
		remove_variable = sr_emig_last_sent
	}
}

# ---- Self-check: emigration_batch (design § 12; colony people design § 6.3) ---------
# Country scope, after the first month's batches: OK when the states taken from
# lost at least 90% of what was sent between them (both read in the same run, so
# a month's births can't hide it); BROKEN otherwise, with the numbers: the strata
# filter removed less, or state_population reads the change only on the next
# tick, as it does after kill_population_percent_in_state (scripting_best_practices
# § "state_population Doesn't Fall Until the Next Tick").
sr_emig_self_check = {
	if = {
		limit = { NOT = { has_global_variable = sr_emig_batch_checked } }
		set_global_variable = { name = sr_emig_batch_checked value = yes }
		set_global_variable = { name = sr_emig_log_before_now value = local_var:sr_emig_fell_before }
		set_global_variable = { name = sr_emig_log_after_now value = local_var:sr_emig_fell_after }
		set_global_variable = { name = sr_emig_log_sent_now value = local_var:sr_emig_sent }
		if = {
			limit = { sr_emig_log_fell >= sr_emig_log_fell_needed }
			debug_log = "TE_SELF_CHECK OK emigration_batch before=[GetGlobalVariable('sr_emig_log_before_now').GetValue|0] after=[GetGlobalVariable('sr_emig_log_after_now').GetValue|0] sent=[GetGlobalVariable('sr_emig_log_sent_now').GetValue|0] country=[THIS.GetCountry.GetNameNoFormatting] (the states taken from lost the batches sent) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN emigration_batch before=[GetGlobalVariable('sr_emig_log_before_now').GetValue|0] after=[GetGlobalVariable('sr_emig_log_after_now').GetValue|0] sent=[GetGlobalVariable('sr_emig_log_sent_now').GetValue|0] country=[THIS.GetCountry.GetNameNoFormatting] (the states taken from lost less than 90% of the settlers sent: the strata filter removed fewer, or state_population reads the change only on the next tick; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
}

# ---- The programme and the priorities --------------------------------------------------
# Country scope. LEVEL a step, 0 Off. The entry declares the programme's upkeep
# from the step (sr_emigration_programme, je_space_race.txt).
sr_emig_set_level = {
	set_variable = { name = sr_emig_level value = $LEVEL$ }
	if = {
		# has_variable first: the Settlers buttons' tooltips walk this writing
		# nothing, so at level 0 the level is unset there (tooltip_walk_variable_audit).
		limit = {
			has_variable = sr_emig_level
			var:sr_emig_level <= 0
		}
		remove_variable = sr_emig_level
	}
}

# Country scope, the Settlers section's - and + (scope:op 0 and 1): - halves
# the amount (500 to Off), + doubles it (colony people design § 6.1).
sr_emig_step_level = {
	if = {
		limit = { scope:op = 0 }
		if = {
			limit = { sr_emig_level_value <= 1 }
			sr_emig_set_level = { LEVEL = 0 }
		}
		else = {
			change_variable = { name = sr_emig_level subtract = 1 }
		}
	}
	else_if = {
		limit = { sr_emig_level_value = 0 }
		sr_emig_set_level = { LEVEL = 1 }
	}
	else_if = {
		limit = { sr_emig_level_value < 16 }
		change_variable = { name = sr_emig_level add = 1 }
	}
}

# Country scope. A colony's priority steps Normal -> High -> Closed -> Low ->
# Normal (the row's button; sr_emig_step_prio_from_scope).
sr_emig_step_prio = {
	if = {
		limit = { NOT = { has_variable = sr_emig_prio_$KEY$ } }
		set_variable = { name = sr_emig_prio_$KEY$ value = 3 }
	}
	else_if = {
		limit = { var:sr_emig_prio_$KEY$ = 3 }
		set_variable = { name = sr_emig_prio_$KEY$ value = 0 }
	}
	else_if = {
		limit = { var:sr_emig_prio_$KEY$ = 0 }
		set_variable = { name = sr_emig_prio_$KEY$ value = 1 }
	}
	else = {
		remove_variable = sr_emig_prio_$KEY$
	}
}

# ---- AI (design § 10.6) ---------------------------------------------------------------------
# Country scope, from the entry's pulse; once a year. Off in default, without
# a Space Program or with no open colony that has room for a batch; Steady
# for a holder or top-three space power (sr_ai_tends_colonies) with a positive
# weekly income; Low otherwise.
# Priorities stay Normal: the weights by free room fill the emptiest first.
sr_emig_ai_review = {
	if = {
		limit = {
			is_ai = yes
			sr_colonies_contested_enabled = yes
			NOT = { has_variable = sr_emig_ai_reviewed }
		}
		set_variable = { name = sr_emig_ai_reviewed value = yes days = 360 }
		if = {
			limit = {
				OR = {
					in_default = yes
					NOT = { sr_has_space_program = yes }
					sr_emig_open_room_total < sr_emig_batch_people
				}
			}
			if = {
				limit = { sr_emig_level_value >= 1 }
				sr_emig_set_level = { LEVEL = 0 }
			}
		}
		else_if = {
			limit = {
				sr_ai_tends_colonies = yes
				net_fixed_income > 0
			}
			if = {
				limit = { NOT = { sr_emig_level_value = 2 } }
				sr_emig_set_level = { LEVEL = 2 }
			}
		}
		else_if = {
			limit = { NOT = { sr_emig_level_value = 1 } }
			sr_emig_set_level = { LEVEL = 1 }
		}
	}
}
