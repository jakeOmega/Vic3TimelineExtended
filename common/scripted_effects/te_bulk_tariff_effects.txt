# ============================================================================
# ALL-GOODS TARIFFS — the Market panel's "All Goods" rows
# ============================================================================
# One import or export level on every tradeable good at once, from the Trade
# tab (gui/te_bulk_tariff_widgets.gui, te_bulk_tariff_sguis.txt). The checks
# the engine's per-good buttons make, and why this restates them, are in
# common/scripted_triggers/te_bulk_tariff_triggers.txt. Country scope.
#
# Cooldown. The per-good buttons wait 3 months after a change, with a week of
# grace to correct it (1.14.5 00_defines.txt:643-646; capability ledger row 15
# in docs/testing/tax-code-capability-ledger.md). The script setter is not
# known to observe that cooldown, so the rows keep their own, per direction and
# of the same length: a use outside the grace week starts both timers, and a
# use inside it changes the levels again without restarting them.
# ============================================================================

# DIR = import | export; KIND = tariff | subvention | none (which treaty can bar
# the level); LEVEL = the engine's level key (max_tariffs ... max_subventions).
#
# While the wait runs (the cooldown set, the grace week over), the tooltip says
# when the row frees up, "again in about 42 days" (MKT-5, 2026-10-27), from the
# day the wait began (te_bulk_tariff_cd_day_<dir>, te_bulk_tariff_values.txt).
# A wait from before that day was recorded keeps the old line.
te_bulk_tariff_apply = {
	custom_tooltip = te_bulk_tariff_effect_$DIR$_$LEVEL$
	if = {
		limit = { has_variable = te_bulk_tariff_grace_$DIR$ }
		custom_tooltip = te_bulk_tariff_effect_grace_$DIR$
	}
	else_if = {
		limit = {
			has_variable = te_bulk_tariff_cd_$DIR$
			has_variable = te_bulk_tariff_cd_day_$DIR$
		}
		custom_tooltip = te_bulk_tariff_effect_wait_$DIR$
	}
	else = {
		custom_tooltip = te_bulk_tariff_effect_cd_$DIR$
	}
	hidden_effect = {
		te_bulk_tariff_all_goods = { DIR = $DIR$ KIND = $KIND$ LEVEL = $LEVEL$ }
		if = {
			limit = { NOT = { has_variable = te_bulk_tariff_grace_$DIR$ } }
			set_variable = { name = te_bulk_tariff_cd_$DIR$ months = 3 }
			set_variable = { name = te_bulk_tariff_grace_$DIR$ days = 7 }
			set_variable = { name = te_bulk_tariff_cd_day_$DIR$ value = game_date }
		}
	}
}

# One good. Dispatched on KIND, so each kind names GOOD, DIR and LEVEL.
te_bulk_tariff_good_tariff = {
	if = {
		limit = { NOT = { te_bulk_tariff_barred = { GOOD = $GOOD$ ARTICLE = no_tariffs } } }
		set_$DIR$_tariff_level = { goods = g:$GOOD$ level = $LEVEL$ }
	}
}

te_bulk_tariff_good_subvention = {
	if = {
		limit = { NOT = { te_bulk_tariff_barred = { GOOD = $GOOD$ ARTICLE = no_subventions } } }
		set_$DIR$_tariff_level = { goods = g:$GOOD$ level = $LEVEL$ }
	}
}

# No tariffs or subventions: what both articles require, so no treaty bars it.
te_bulk_tariff_good_none = {
	set_$DIR$_tariff_level = { goods = g:$GOOD$ level = $LEVEL$ }
}
