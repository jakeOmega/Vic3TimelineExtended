te_construction_market_ai_update_buy = {
	# The AI sets a quantity. A Fixed Budget left by a player who played this
	# country would otherwise override it (te_construction_market_public_setting).
	if = {
		limit = {
			te_construction_market_budget_mode_on = yes
		}
		remove_variable = te_construction_market_budget_mode
	}
	set_variable = {
		name = te_construction_market_public_target
		value = te_construction_market_ai_purchase_target
	}
}
