te_construction_market_ai_update_buy = {
	# The AI sets a quantity. A Fixed Budget left by a player who played this
	# country would otherwise override it (te_construction_market_public_setting).
	if = {
		limit = {
			te_construction_market_budget_mode_on = yes
		}
		remove_variable = te_construction_market_budget_mode
	}
	set_variable = {
		name = te_construction_market_public_target
		value = te_construction_market_ai_purchase_target
	}
}

# TE_SELF_CHECK construction_admin_while_paused (docs/testing/ai-economy-playtest.md,
# "The AI's construction purchase fits its budget"). Once a game, from the
# weekly pulse of an AI country in a market game, after its purchase is set.
#
# The engine pauses an AI's construction at 40% of its debt ceiling and a
# construction it calls critical only at 99%. A country short of bureaucracy
# with a Government Administration level queued goes on buying construction
# through the pause (te_construction_market_ai_purchase_target). Whether the
# engine then builds the level is unread, and this reads it.
#
#   Arm    a country the engine reports paused, short of bureaucracy, with an
#          administration under construction and a Construction Site with a
#          crew (so the purchase is delivered) stores its administration
#          levels and its number of states.
#   Count  each later week in which all of that still holds and the levels
#          are what they were: the week, the purchase now in force, whether
#          the engine reports wasted construction, and the budget's temporary
#          expenses and income (the construction billed, and what the
#          investment pool repaid of it).
#   OK     at a later weekly pass the country is still paused, has the same
#          number of states and more administration levels: a level finished
#          on points bought while paused, so the engine goes on building it.
#          A level cannot finish while paused any other way, since without
#          this purchase a paused country buys nothing. A level that came with
#          a conquered state changes the number of states and decides nothing.
#   BROKEN the purchases counted come to
#          TE_CONSTRUCTION_MARKET_AI_ADMIN_SELFCHECK_POINTS (four levels'
#          worth) and no level has finished: the engine does not spend the
#          points on the administration. spent and wasted_weeks say whether
#          the treasury paid for them all the same. The play-test list has
#          the fallback.
#   A country that leaves the pause, ends its shortage some other way, loses
#   the queued level or changes its number of states is disarmed with no
#   verdict; it or another arms again later. No line in a game means no AI
#   country was ever paused, short of bureaucracy and building an
#   administration long enough to decide.
# The plain START line comes first because the engine drops a templated line
# whole if it cannot render it.
te_construction_market_ai_admin_paused_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_cm_admin_paused_check_done }
		}
		if = {
			limit = {
				has_variable = te_cm_chk_admin_bought
				has_variable = te_cm_chk_admin_levels
				has_variable = te_cm_chk_admin_states
				has_variable = te_cm_chk_admin_weeks
				has_variable = te_cm_chk_admin_wasted
				has_variable = te_cm_chk_admin_spent
				has_variable = te_cm_chk_admin_repaid
			}
			if = {
				limit = {
					is_construction_paused = yes
					te_construction_market_state_count = var:te_cm_chk_admin_states
					te_construction_market_administration_levels > var:te_cm_chk_admin_levels
				}
				set_global_variable = { name = te_cm_admin_paused_check_done value = yes }
				debug_log = "TE_SELF_CHECK START construction_admin_while_paused"
				debug_log = "TE_SELF_CHECK OK construction_admin_while_paused country=[THIS.GetCountry.GetTagName] weeks=[THIS.Var('te_cm_chk_admin_weeks').GetValue|0] bought=[THIS.Var('te_cm_chk_admin_bought').GetValue|1] wasted_weeks=[THIS.Var('te_cm_chk_admin_wasted').GetValue|0] spent=[THIS.Var('te_cm_chk_admin_spent').GetValue|0] repaid=[THIS.Var('te_cm_chk_admin_repaid').GetValue|0] (a Government Administration level finished while the engine reported construction paused: it goes on building one for a country short of bureaucracy)"
				te_construction_market_ai_admin_paused_selfcheck_clear = yes
			}
			else_if = {
				limit = {
					is_construction_paused = yes
					te_construction_market_state_count = var:te_cm_chk_admin_states
					te_construction_market_administration_levels = var:te_cm_chk_admin_levels
					te_construction_market_ai_building_administration = yes
				}
				change_variable = { name = te_cm_chk_admin_weeks add = 1 }
				change_variable = { name = te_cm_chk_admin_bought add = te_construction_market_public_use_target }
				change_variable = { name = te_cm_chk_admin_spent add = te_construction_market_temporary_expenses }
				change_variable = { name = te_cm_chk_admin_repaid add = te_construction_market_temporary_income }
				if = {
					limit = { has_wasted_construction = yes }
					change_variable = { name = te_cm_chk_admin_wasted add = 1 }
				}
				if = {
					limit = {
						var:te_cm_chk_admin_bought >= TE_CONSTRUCTION_MARKET_AI_ADMIN_SELFCHECK_POINTS
					}
					set_global_variable = { name = te_cm_admin_paused_check_done value = yes }
					debug_log = "TE_SELF_CHECK START construction_admin_while_paused"
					debug_log = "TE_SELF_CHECK BROKEN construction_admin_while_paused country=[THIS.GetCountry.GetTagName] weeks=[THIS.Var('te_cm_chk_admin_weeks').GetValue|0] bought=[THIS.Var('te_cm_chk_admin_bought').GetValue|1] wasted_weeks=[THIS.Var('te_cm_chk_admin_wasted').GetValue|0] spent=[THIS.Var('te_cm_chk_admin_spent').GetValue|0] repaid=[THIS.Var('te_cm_chk_admin_repaid').GetValue|0] (no Government Administration level finished on points worth four levels, bought while the engine reported construction paused: it does not build one through the pause)"
					debug_log_scopes = yes
					te_construction_market_ai_admin_paused_selfcheck_clear = yes
				}
			}
			else = {
				te_construction_market_ai_admin_paused_selfcheck_clear = yes
			}
		}
		else_if = {
			limit = {
				is_construction_paused = yes
				te_construction_market_ai_building_administration = yes
				te_construction_market_site_levels >= 1
			}
			set_variable = { name = te_cm_chk_admin_levels value = te_construction_market_administration_levels }
			set_variable = { name = te_cm_chk_admin_states value = te_construction_market_state_count }
			set_variable = { name = te_cm_chk_admin_weeks value = 0 }
			set_variable = { name = te_cm_chk_admin_bought value = 0 }
			set_variable = { name = te_cm_chk_admin_wasted value = 0 }
			set_variable = { name = te_cm_chk_admin_spent value = 0 }
			set_variable = { name = te_cm_chk_admin_repaid value = 0 }
		}
	}
	# Decided elsewhere while this country was armed.
	else_if = {
		limit = { has_variable = te_cm_chk_admin_bought }
		te_construction_market_ai_admin_paused_selfcheck_clear = yes
	}
}

te_construction_market_ai_admin_paused_selfcheck_clear = {
	if = {
		limit = { has_variable = te_cm_chk_admin_levels }
		remove_variable = te_cm_chk_admin_levels
	}
	if = {
		limit = { has_variable = te_cm_chk_admin_states }
		remove_variable = te_cm_chk_admin_states
	}
	if = {
		limit = { has_variable = te_cm_chk_admin_weeks }
		remove_variable = te_cm_chk_admin_weeks
	}
	if = {
		limit = { has_variable = te_cm_chk_admin_bought }
		remove_variable = te_cm_chk_admin_bought
	}
	if = {
		limit = { has_variable = te_cm_chk_admin_wasted }
		remove_variable = te_cm_chk_admin_wasted
	}
	if = {
		limit = { has_variable = te_cm_chk_admin_spent }
		remove_variable = te_cm_chk_admin_spent
	}
	if = {
		limit = { has_variable = te_cm_chk_admin_repaid }
		remove_variable = te_cm_chk_admin_repaid
	}
}
