# ============================================================================
# TABLING A RESOLUTION — SHARED EFFECTS
# ============================================================================
# What each of the seven member-initiated topics actually does when tabled:
# the costs and cooldowns it takes, the target it names, and the
# un_resolution_open call that opens the vote. The sixteen conventions follow at
# the bottom of the file, with their own header.
#
# Extracted word for word out of each scripted button's `effect` in
# common/scripted_buttons/un_buttons.txt, so the button (which is how the AI
# proposes) and the chamber widget's Propose control run the same script.
# The gates live next door in common/scripted_triggers/un_propose_triggers.txt.
#
# TARGET SELECTION. Three topics name a country. Each has a selector effect that
# leaves the pick in a temporary scope, in the shape un_mandate_select_case
# already used: the proposal promotes it to scope:un_vote_target, and the
# chamber's row preview calls the same selector to show who would be named. That
# is what makes the preview unable to drift from the proposal.
#
# Condemnation and sanctions previously picked with `random_country`, which
# cannot be previewed — a row that promised one rival and censured another would
# be worse than no preview at all. For a HUMAN proposer both now pick the
# highest-ranked eligible rival, which is what the expulsion button has always
# done. The AI keeps its random pick from the same pool, so nothing the AI does
# changed.
# ============================================================================

# Scope: the proposing country
un_propose_condemn_select_target = {
	# The AI keeps the random pick it has always made; only a human proposer,
	# who sees the pick previewed in the chamber, needs it to be deterministic.
	if = {
		limit = { is_ai = yes }
		random_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				is_at_war = yes
				is_diplomatic_play_initiator = yes
				un_case_supports = { TOPIC = condemn }
			}
			save_temporary_scope_as = un_propose_condemn_target
		}
	}
	else = {
		ordered_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				is_at_war = yes
				is_diplomatic_play_initiator = yes
				un_case_supports = { TOPIC = condemn }
			}
			order_by = country_rank
			max = 1
			save_temporary_scope_as = un_propose_condemn_target
		}
	}
}

# Scope: the proposing country
un_propose_sanctions_select_target = {
	# The AI keeps the random pick it has always made; only a human proposer,
	# who sees the pick previewed in the chamber, needs it to be deterministic.
	if = {
		limit = { is_ai = yes }
		random_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				NOT = { has_modifier = un_sanctions_target_modifier }
				un_case_supports = { TOPIC = sanctions }
			}
			save_temporary_scope_as = un_propose_sanctions_target
		}
	}
	else = {
		ordered_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				NOT = { has_modifier = un_sanctions_target_modifier }
				un_case_supports = { TOPIC = sanctions }
			}
			order_by = country_rank
			max = 1
			save_temporary_scope_as = un_propose_sanctions_target
		}
	}
}

# Scope: the proposing country
un_propose_expulsion_select_target = {
	ordered_country = {
		limit = {
			NOT = { this = ROOT }
			is_un_permanent_member = yes
			has_modifier = un_veto_isolation_modifier
		}
		# The strongest case for expulsion first (owner, 2026-10-24); every
		# permanent member is a great power, so country_rank tied them all.
		order_by = un_expulsion_case
		max = 1
		save_temporary_scope_as = un_propose_expulsion_target
	}
}

# Authorized military mandate — binding, simple majority, and the only topic
# whose proposal also names a territory. Gates are the fullest of the six.
# Scope: the proposing country
un_propose_mandate_effect = {
	custom_tooltip = {
		text = "UN_PROPOSE_MANDATE_TT"
		un_mandate_select_case = yes
		if = {
			limit = {
				exists = scope:un_mandate_case_target
				exists = scope:un_mandate_case_state
			}
			scope:un_mandate_case_target = { save_scope_as = un_vote_target }
			un_propose_add_sponsor_mark = yes
			un_propose_add_request_cooldown = yes
			set_variable = { name = un_mandate_proposal_cooldown days = 1825 }
			un_resolution_open = { TOPIC = military_mandate }
			# The topic-specific fields ride on the resolution container, so
			# un_vote.1/.2 and the chamber can read them without a
			# country-side copy. un_resolution_open leaves scope:un_resolution
			# pointing at the new container.
			scope:un_resolution ?= {
				set_variable = {
					name = un_res_mandate_region
					value = scope:un_mandate_case_state.state_region
				}
				set_variable = {
					name = un_res_mandate_beneficiary
					value = ROOT
				}
			}
			# A case opened because the holder defied the World Court in our
			# dispute over this region (redesign phase 7, un_justice_triggers.txt):
			# the defiance is the grounds, so the topic's case threshold does not
			# apply to it (un_res_case_threshold, un_chamber_grounds_line).
			if = {
				limit = { scope:un_mandate_case_state = { un_court_defiance_opens_mandate = yes } }
				scope:un_resolution ?= { set_variable = un_res_court_grounds }
			}
		}
	}
	# Outside the case check so the tooltip shows it: un_propose_mandate_possible
	# already asks for a case.
	un_propose_sponsor_punitive_cost = yes
}

# Open a charter-reform resolution for the charter's next stage. The one place
# the reform topic is opened, shared by the button, the chamber and
# un_events.6. The stage rides on the resolution (un_res_charter_stage), so
# un_vote.1/.2 and the chamber read which reform it is without a
# country-side copy, and a resolution tabled for Reform I cannot be adopted as
# Reform II.
# Scope: the proposing country
un_charter_reform_open = {
	un_resolution_open = { TOPIC = reform }
	scope:un_resolution ?= {
		set_variable = { name = un_res_charter_stage value = un_charter_next_stage }
	}
}

# THE SIXTEEN CONVENTIONS
# ============================================================================
# Human rights, ICC, NPT, climate, pandemic, refugee, heritage,
# decolonization, space, the law of the sea, the physical protection of
# nuclear material, cultural diversity, the prohibition of nuclear weapons,
# narcotic drugs, the Antarctic Treaty and the anti-satellite moratorium.
# Each has two effects:
#
#   un_propose_<key>_table   what tabling it does, whoever tables it: the
#                            domestic reaction and the vote itself. The
#                            proposer event's first option calls it when a
#                            member accepts the docket's offer, and
#                            un_propose_<key>_effect calls it too, so the two
#                            paths cannot drift.
#   un_propose_<key>_effect  tabling it on our own motion from the chamber
#                            (ops 7-19 and 29 of un_chamber_propose_sgui): _table,
#                            plus the costs every own-motion proposal takes.
#
# Accepting the docket's offer costs nothing extra: the docket paces the floor
# (at most one item every few months), so taking up what it raised is not
# jumping the queue. Tabling on our own motion is, and pays for it. This is the
# charter reform precedent — un_events.6 is free, un_propose_reform_effect
# takes the same two modifiers as below.
#
# The gates are in un_propose_triggers.txt (un_propose_<key>_available /
# _possible). There are no scripted buttons: the AI reaches these topics only
# through the docket.
# ============================================================================

# The price of tabling business on our own motion, the same modifiers the
# seven topics above take: un_ga_resolution_modifier marks us as a sponsor for
# five years, un_request_cooldown bars our next own-motion proposal, of any
# topic, for ten, and the sponsor's influence cost runs two years. Shown rather
# than folded into a custom tooltip, because the ten-year bar is the real price
# of the choice. The three phase 7 topics that punish a member take the
# punitive sponsor's cost (un_propose_own_motion_costs_punitive).
# Scope: the proposing country
un_propose_own_motion_costs = {
	un_propose_own_motion_marks = yes
	un_propose_sponsor_cost = yes
}

# Scope: the proposing country
un_propose_own_motion_costs_punitive = {
	un_propose_own_motion_marks = yes
	un_propose_sponsor_punitive_cost = yes
}

# Scope: the proposing country
un_propose_own_motion_marks = {
	un_propose_add_sponsor_mark = yes
	un_propose_add_request_cooldown = yes
}

# The sponsor mark (un_ga_resolution_modifier, five years at the base pace) and
# the bar on the next own-motion proposal (un_request_cooldown, ten), each for
# its base times un_propose_pace_mult (un_propose_pace_values.txt). Every site
# that adds either one calls these. The durations are literals because
# add_modifier's days reads a computed script value as 0 ("for 0 days";
# scripting_best_practices.md, "Modifier & Cooldown Duration Units"). The
# multiplier takes ten values, a great power's half times a standing tier, and
# each band below holds one of them; its duration is the base times that value,
# to the nearest day and at least un_propose_min_days. test_un_propose_pace.py
# recomputes both from the tuning, so a retuned multiplier fails it until the
# ladders follow.
# Scope: the proposing country
un_propose_add_sponsor_mark = {
	if = {
		limit = { un_propose_pace_mult < 0.39 }	# x0.35: a great power at Exemplary
		add_modifier = { name = un_ga_resolution_modifier days = 639 is_decaying = yes }
	}
	else_if = {
		limit = { un_propose_pace_mult < 0.46 }	# x0.425: a great power at Respected
		add_modifier = { name = un_ga_resolution_modifier days = 776 is_decaying = yes }
	}
	else_if = {
		limit = { un_propose_pace_mult < 0.56 }	# x0.5: a great power at Neutral
		add_modifier = { name = un_ga_resolution_modifier days = 913 is_decaying = yes }
	}
	else_if = {
		limit = { un_propose_pace_mult < 0.66 }	# x0.625: a great power at Poor
		add_modifier = { name = un_ga_resolution_modifier days = 1141 is_decaying = yes }
	}
	else_if = {
		limit = { un_propose_pace_mult < 0.72 }	# x0.7: Exemplary
		add_modifier = { name = un_ga_resolution_modifier days = 1278 is_decaying = yes }
	}
	else_if = {

	}
	else_if = {
		limit = { un_propose_pace_mult < 0.92 }	# x0.85: Respected

	else_if = {

# ---- International Refugee Resolution (UNHCR) — un_events.18 ----
# The country the displaced are fleeing, which becomes the resolution's target
# so that un_vote.2 can offer its displaced resettlement when the resolution
# carries (un_events.103). A graver case (famine, or a war that has devastated
# a state) is preferred over a merely troubled one. As with condemnation and
# sanctions, a human proposer gets the highest-ranked candidate, so the
# chamber row can preview it; the AI keeps a random pick from the same pool.
# un_events.18's immediate calls this too.
# Scope: the proposing country
un_propose_refugee_select_target = {
	if = {
		limit = { any_country = { un_propose_refugee_grave_source = yes } }
		if = {
			limit = { is_ai = yes }
			random_country = {
				limit = { un_propose_refugee_grave_source = yes }
				save_temporary_scope_as = un_propose_refugee_target
			}
		}
		else = {
			ordered_country = {
				limit = { un_propose_refugee_grave_source = yes }
				order_by = country_rank
				max = 1
				save_temporary_scope_as = un_propose_refugee_target
			}
		}
	}
	else = {
		if = {
			limit = { is_ai = yes }
			random_country = {
				limit = { un_propose_refugee_source = yes }
				save_temporary_scope_as = un_propose_refugee_target
			}
		}
		else = {
			ordered_country = {
				limit = { un_propose_refugee_source = yes }
				order_by = country_rank
				max = 1
				save_temporary_scope_as = un_propose_refugee_target
			}
		}
	}
}

# A refugee resolution accuses nobody: the vote effects treat a target as
# accused only on the accusing topics (un_resolution_accuses_target), so the
# source voting for it is an ordinary yes, not a confession.
# Expects: scope:refugee_source (when there is one)
# Scope: the proposing country
un_propose_refugee_table = {
	social_progressive_reaction_effect = { DAYS = normal_modifier_time }
	hidden_effect = {
		scope:refugee_source ?= { save_scope_as = un_vote_target }
		un_resolution_open = { TOPIC = refugee }
	}
}

# Scope: the proposing country
un_propose_refugee_effect = {
	custom_tooltip = un_vote_initiated_tt
	un_propose_own_motion_costs = yes
	hidden_effect = {
		un_propose_refugee_select_target = yes
		scope:un_propose_refugee_target ?= { save_scope_as = refugee_source }
	}
	un_propose_refugee_table = yes
}
