# ============================================================================
# GRAND MONUMENTS — unique dedications (signatures): scripted effects
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-16-grand-monument-signatures-design.md.
# The records are listed in gm_signature_triggers.txt. A signature adds its
# name (the custom localization, gm_signature_custom_loc.txt) and its effects to
# its parent dedication, which does everything it does today: the production
# method, kind, fit, contests, interest-group stances and opportunity progress
# are the parent's, and nothing here touches them.
#
# Where its effects are applied, as the dedications' are (gm_effects.txt):
# national ones on the journal entry from gm_apply_national_modifiers, ROOT =
# the country; local ones on the state from gm_state_monthly, ROOT = the state.
# A variable backing a multiplier is zeroed, never removed.

# State scope, from the refresh (gm_country_refresh) and the seeding: a landmark
# without a signature gets its own. The net for the landmarks of 1836, a site
# as it is unearthed, a landmark from a save older than signatures, and any copy
# point that misses the records: within a month every landmark has its own.
gm_state_sig_backfill = {
	if = {
		limit = {
			has_variable = gm_landmark
			NOT = { has_variable = gm_signature }
		}
		gm_sig_from_landmark = { LM = gm_landmark SIG = gm_signature PARENT = gm_sig_parent }
	}
}

# Container scope (a monument beside another, a trophy, a site): the same, from
# the monthly passes (gm_monument_monthly, gm_trophy_monthly) and the seeding.
gm_tr_sig_backfill = {
	if = {
		limit = {
			has_variable = gm_tr_landmark
			NOT = { has_variable = gm_tr_signature }
		}
		gm_sig_from_landmark = { LM = gm_tr_landmark SIG = gm_tr_signature PARENT = gm_tr_sig_parent }
	}
}

# ---- The national sums (§2.1, §2.4) -----------------------------------------------
# Country scope, from gm_compute_totals. Each signature's grandeur, from one walk
# over the states and one over each list, not a walk per signature: the
# primaries that count (gm_state_sig_counts), the monuments beside another
# that count, and the trophies this country holds on their parent, at their own
# grandeur (gm_tr_grandeur, already half what was carried off) whatever its
# government. A family's sum counts its one-offs (gm_sig_counts_as_*). Then the

		gm_state_sig_add_to_sum = { SIG = archive }

			gm_mon_sig_add_to_sum = { SIG = archive G = gm_tr_g }
			gm_mon_sig_add_to_sum = { SIG = throne G = gm_tr_g }
			gm_mon_sig_add_to_sum = { SIG = royal_pantheon G = gm_tr_g }
			gm_mon_sig_add_to_sum = { SIG = martial_glory G = gm_tr_g }
			gm_mon_sig_add_to_sum = { SIG = liberty G = gm_tr_g }
			gm_mon_sig_add_to_sum = { SIG = grande_armee G = gm_tr_g }
			gm_mon_sig_add_to_sum = { SIG = mandate_of_heaven G = gm_tr_g }

		}

	if = {
		limit = { has_variable_list = gm_trophies_held }
		save_scope_as = gm_sig_country
		every_in_list = {
			variable = gm_trophies_held
			limit = {

				has_variable = gm_tr_grandeur

	gm_sig_set_steps = { SIG = throne }

	gm_sig_set_steps = { SIG = martial_glory }
	gm_sig_set_steps = { SIG = liberty }

# Country scope. A signature's grandeur and count at 0 before the walks.
gm_sig_zero_sum = {
	set_variable = { name = gm_g_sig_$SIG$ value = 0 }
	set_variable = { name = gm_c_sig_$SIG$ value = 0 }
}

# State scope, a primary that counts: its grandeur (gm_grandeur, written by the
# refresh's state loop) to its owner's sum of $SIG$, if it counts toward it.
gm_state_sig_add_to_sum = {
	if = {
		limit = { gm_sig_counts_as_$SIG$ = { VAR = gm_signature } }
		owner = {
			change_variable = { name = gm_g_sig_$SIG$ add = prev.var:gm_grandeur }
			change_variable = { name = gm_c_sig_$SIG$ add = 1 }
		}
	}
}

# Container scope (scope:gm_sig_country the country summing): its grandeur, the
# variable $G$ on it, to that country's sum of $SIG$, if it counts toward it.
gm_mon_sig_add_to_sum = {
	if = {
		limit = { gm_sig_counts_as_$SIG$ = { VAR = gm_tr_signature } }
		scope:gm_sig_country = {
			change_variable = { name = gm_g_sig_$SIG$ add = prev.var:$G$ }
			change_variable = { name = gm_c_sig_$SIG$ add = 1 }
		}
	}
}

# Country scope. Grandeur -> steps and the next step, then the policy factor.
gm_sig_set_steps = {
	gm_set_steps = { IN = gm_g_sig_$SIG$ OUT = gm_s_sig_$SIG$ NEXT = gm_n_sig_$SIG$ F = 5 }
	gm_policy_scale_steps = { S = sig_$SIG$ FACTOR = gm_policy_factor_standing }
}

# The same on the 1/n curve with no end, for a signature whose effect is
# cultural pull (Ancestral Antiquity), as the monuments' own pull is (owner,
# 2026-10-19: its +3 cap gone, since its monuments are a fixed list that only
# changes hands; none can be built).
gm_sig_set_steps_lin_open = {
	gm_set_steps_lin_open = { IN = gm_g_sig_$SIG$ OUT = gm_s_sig_$SIG$ NEXT = gm_n_sig_$SIG$ WIDTH = gm_w_sig_$SIG$ F = 5 }
	gm_policy_scale_steps = { S = sig_$SIG$ FACTOR = gm_policy_factor_standing }
}

# ---- The local effects (§2.2): on_monthly_pulse_state, ROOT = the state --------------
# From gm_state_monthly, beside gm_state_local_steps_one. Signature $SIG$'s
# local effect here: the 1/n curve over the grandeur of every monument standing here
# that carries it on its parent, the primary's and each secondary's, whatever
# its status (heritage and contested too, as the dedications' local effects).
# Its multiplier variable is zeroed, never removed.
gm_state_sig_local_steps_one = {
	set_variable = {
		name = gm_curve_in
		value = {
			value = 0
			if = {
				limit = {
					gm_state_sig_on_parent = yes
					gm_sig_counts_as_$SIG$ = { VAR = gm_signature }
				}
				add = gm_state_primary_grandeur
			}
			if = {
				limit = { has_variable_list = gm_monuments_here }
				every_in_list = {
					variable = gm_monuments_here
					limit = {
						gm_mon_sig_on_parent = yes
						gm_sig_counts_as_$SIG$ = { VAR = gm_tr_signature }
					}
					add = gm_mon_grandeur
				}
			}
		}
	}
	set_variable = {
		name = gm_sig_local_steps_$SIG$
		value = {
			value = gm_curve_steps_lin_f5
			multiply = gm_policy_factor_local
		}
	}
	if = {
		limit = { var:gm_sig_local_steps_$SIG$ > 0 }
		add_modifier = { name = gm_local_sig_$SIG$ multiplier = var:gm_sig_local_steps_$SIG$ }
	}
}

# State scope. Every signature's local step variable at 0, where one exists.
gm_state_sig_zero_local_steps = {
	gm_state_sig_zero_local_steps_one = { SIG = antiquity }
	gm_state_sig_zero_local_steps_one = { SIG = archive }
	gm_state_sig_zero_local_steps_one = { SIG = first_emperor }
}

gm_state_sig_zero_local_steps_one = {
	if = {
		limit = { has_variable = gm_sig_local_steps_$SIG$ }
		set_variable = { name = gm_sig_local_steps_$SIG$ value = 0 }
	}
}

# ---- Special dedications from opportunities (§5) ----------------------------------
# Country scope, from gm_com_fulfil. The open opportunity's special, from its
# moment, as gm_sig_special (a flag; none for a moment without one). One per
# source: a moment whose dedication is not single (A New Reign, A New Order)
# has none (§5.4). test_grand_monument_registry.py pins it to SIGNATURES.
gm_sig_special_for_source = {
	gm_remove_state_var = { VAR = gm_sig_special }
	if = {
		limit = {
			has_variable = gm_com_source
			var:gm_com_source = flag:victory
		}
		set_variable = { name = gm_sig_special value = flag:to_victory }
	}
	if = {
		limit = {
			has_variable = gm_com_source
			var:gm_com_source = flag:defeat
		}
		set_variable = { name = gm_sig_special value = flag:unknown_soldier }
	}
	if = {
		limit = {
			has_variable = gm_com_source
			var:gm_com_source = flag:great_power
		}
		set_variable = { name = gm_sig_special value = flag:concert_of_nations }
	}
	if = {
		limit = {
			has_variable = gm_com_source
			var:gm_com_source = flag:space
		}
		set_variable = { name = gm_sig_special value = flag:conquerors_of_space }
	}
	if = {
		limit = {
			has_variable = gm_com_source
			var:gm_com_source = flag:tech_athletic
		}
		set_variable = { name = gm_sig_special value = flag:olympic }
	}
}

gm_sig_set_special_parent = {
	if = {
		limit = { var:$SIGVAR$ = flag:to_victory }
		set_variable = { name = $VAR$ value = flag:war_memorial }
	}
	else_if = {
		limit = { var:$SIGVAR$ = flag:unknown_soldier }
		set_variable = { name = $VAR$ value = flag:war_memorial }
	}
	else_if = {
		limit = { var:$SIGVAR$ = flag:concert_of_nations }
		set_variable = { name = $VAR$ value = flag:civic }
	}
	else_if = {
		limit = { var:$SIGVAR$ = flag:conquerors_of_space }
		set_variable = { name = $VAR$ value = flag:civic }
	}
	else_if = {
		limit = { var:$SIGVAR$ = flag:olympic }
		set_variable = { name = $VAR$ value = flag:athletic }
	}
}

# Country scope. One more of special $SIG$ held, for the next monument of its
# dedication; and one less, once a ceremony gives it.
gm_sig_hold = {
	if = {
		limit = { has_variable = gm_sig_held_$SIG$ }
		change_variable = { name = gm_sig_held_$SIG$ add = 1 }
	}
	else = {
		set_variable = { name = gm_sig_held_$SIG$ value = 1 }
	}
}

gm_sig_unhold = {
	if = {
		limit = { has_variable = gm_sig_held_$SIG$ }
		change_variable = { name = gm_sig_held_$SIG$ add = -1 }
		if = {
			limit = { var:gm_sig_held_$SIG$ <= 0 }
			remove_variable = gm_sig_held_$SIG$
		}
	}
}

# Country scope. The special in the country's gm_sig_special held (no monument
# could take it).
gm_sig_hold_special = {
	if = {
		limit = {
			has_variable = gm_sig_special
			var:gm_sig_special = flag:to_victory
		}
		gm_sig_hold = { SIG = to_victory }
	}
	if = {
		limit = {
			has_variable = gm_sig_special
			var:gm_sig_special = flag:unknown_soldier
		}
		gm_sig_hold = { SIG = unknown_soldier }
	}
	if = {
		limit = {
			has_variable = gm_sig_special
			var:gm_sig_special = flag:concert_of_nations
		}
		gm_sig_hold = { SIG = concert_of_nations }
	}
	if = {
		limit = {
			has_variable = gm_sig_special
			var:gm_sig_special = flag:conquerors_of_space
		}
		gm_sig_hold = { SIG = conquerors_of_space }
	}
	if = {
		limit = {
			has_variable = gm_sig_special
			var:gm_sig_special = flag:olympic
		}
		gm_sig_hold = { SIG = olympic }
	}
	set_variable = gm_com_seized_special_held
}

# State scope, from monument_events.26 (the monument gm_com_fulfil marked with
# gm_sig_grant beside gm_gift). The completing monument takes the special, its
# dedication being the special's parent, so nothing is rebuilt; one that
# already carries a signature, or no longer stands on the parent, leaves it
# held by its owner for the next.
gm_state_sig_take_grant = {
	if = {
		limit = { has_variable = gm_sig_grant }
		if = {
			limit = { gm_state_takes_grant = yes }
			set_variable = { name = gm_signature value = var:gm_sig_grant }
			gm_sig_set_special_parent = { VAR = gm_sig_parent SIGVAR = gm_signature }
			owner = { trigger_event = { id = monument_events.20 } }
		}
		else = {
			if = {
				limit = { var:gm_sig_grant = flag:to_victory }
				owner = { gm_sig_hold = { SIG = to_victory } }
			}
			if = {
				limit = { var:gm_sig_grant = flag:unknown_soldier }
				owner = { gm_sig_hold = { SIG = unknown_soldier } }
			}
			if = {
				limit = { var:gm_sig_grant = flag:concert_of_nations }
				owner = { gm_sig_hold = { SIG = concert_of_nations } }
			}
			if = {
				limit = { var:gm_sig_grant = flag:conquerors_of_space }
				owner = { gm_sig_hold = { SIG = conquerors_of_space } }
			}
			if = {
				limit = { var:gm_sig_grant = flag:olympic }
				owner = { gm_sig_hold = { SIG = olympic } }
			}
		}
		remove_variable = gm_sig_grant
	}
}

# Container scope, a monument beside another that completed an opportunity
# (gm_com_fulfil, beside its gift of age; scope:gm_sig_granter its country):
# it takes the special, which writes variables only, as the gift does there.
gm_mon_sig_take_special = {
	set_variable = { name = gm_tr_signature value = scope:gm_sig_granter.var:gm_sig_special }
	gm_sig_set_special_parent = { VAR = gm_tr_sig_parent SIGVAR = gm_tr_signature }
}

# State scope, an option of monument_events.2: the parent's own dedication, then

# ---- Old saves (§5.6) --------------------------------------------------------------
# No scope, from gm_monument_monthly, once a round (gm_sig_seed_round against
# gm_sig_seed_rounds, as gm_arch_seed_round). A save from before specials holds
# none: each country gets one held special for every moment whose boon still
# runs, an opportunity completed in the last ten years. No monument takes it
# then; the next ceremony that can raise its parent offers it.
gm_sig_seed_monthly = {
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = gm_sig_seed_round }
				global_var:gm_sig_seed_round < gm_sig_seed_rounds
			}
		}
		set_global_variable = { name = gm_sig_seed_round value = gm_sig_seed_rounds }
		every_country = {
			limit = { gm_boons_idle = no }
			if = {
				limit = {
					has_variable = gm_boon_victory
					var:gm_boon_victory > 0
				}
				gm_sig_hold = { SIG = to_victory }
			}
			if = {
				limit = {
					has_variable = gm_boon_defeat
					var:gm_boon_defeat > 0
				}
				gm_sig_hold = { SIG = unknown_soldier }
			}
			if = {
				limit = {
					has_variable = gm_boon_great_power
					var:gm_boon_great_power > 0
				}
				gm_sig_hold = { SIG = concert_of_nations }
			}
			if = {
				limit = {
					has_variable = gm_boon_space
					var:gm_boon_space > 0
				}
				gm_sig_hold = { SIG = conquerors_of_space }
			}
			if = {
				limit = {
					has_variable = gm_boon_tech_athletic
					var:gm_boon_tech_athletic > 0
				}
				gm_sig_hold = { SIG = olympic }
			}
		}
	}
}
