# ============================================================================
# Strategic Reserve — Scripted Effects
# ============================================================================
# Scope contract:
# - Country scope owns all persistent Strategic Reserve vars and script values.
# - Building scope is only used to read live hub data and apply transient flow
#   modifiers to the hub itself.
# - Journal entries and scripted buttons should call the wrapper effects in the
#   lower half of this file rather than the low-level scope-bridge helpers.

# --------------------------------------------------------------------------
# st_res_init_good_effect — Per-good half of st_res_init_effect.
# Parameter: GOOD = grain | ammunition | oil | small_arms | artillery |
#            aeroplanes | tanks | fertilizer.
# `st_res_$GOOD$_last_delta` / `st_res_$GOOD$_last_status` are the display
# bookkeeping vars read by the JE inventory widget. They are SAVE-SAFE: a save
# made before the widget existed simply has neither, and this effect seeds them
# with 0 (= no movement recorded / status "Idle") until the next weekly tick
# overwrites them with real data.
#
# The twelve reserve-policy variables are seeded behind ONE guard on
# `st_res_$GOOD$_policy`, because they are only ever written together: a save
# from before policies existed has none of them, gets all twelve at once, and
# can never be left in a half-configured state. The seeded policy is 0 (Manual)
# with the Standard preset's settings, so loading a pre-feature save changes
# nothing a player can see — every good keeps the rate it had and keeps being
# driven by hand until the player picks a policy.
#
# A SECOND guard covers a save from after policies but before price smoothing:
# it seeds only the two smoothing settings, with their identity values (price
# memory 1 = the live price, response ramp 0 = the step response), so a policy
# that was already running keeps behaving exactly as it did until the player
# touches those settings or applies a preset. A new game never takes that
# branch — the preset in the first guard has already written both.
#
# The running price average (`st_res_$GOOD$_price_avg`) is deliberately NOT
# seeded here: st_res_policy_track_price_effect seeds it from the live price on
# the first weekly tick, and a value of 0 would read as "at base price".
# --------------------------------------------------------------------------
st_res_init_good_effect = {
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_stored } }
		set_variable = { name = st_res_$GOOD$_stored value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_rate } }
		set_variable = { name = st_res_$GOOD$_rate value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_last_delta } }
		set_variable = { name = st_res_$GOOD$_last_delta value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_last_status } }
		set_variable = { name = st_res_$GOOD$_last_status value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_policy } }
		set_variable = { name = st_res_$GOOD$_policy value = 0 }
		set_variable = { name = st_res_$GOOD$_engaged value = 0 }
		set_variable = { name = st_res_$GOOD$_policy_status value = 0 }
		set_variable = { name = st_res_$GOOD$_policy_price value = 0 }
		st_res_apply_preset_standard_base = { GOOD = $GOOD$ }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_price_memory } }
		set_variable = { name = st_res_$GOOD$_price_memory value = 1 }
		set_variable = { name = st_res_$GOOD$_ramp value = 0 }
	}
}

# --------------------------------------------------------------------------

# --------------------------------------------------------------------------
st_res_init_effect = {
	st_res_init_good_effect = { GOOD = grain }
	st_res_init_good_effect = { GOOD = ammunition }
	st_res_init_good_effect = { GOOD = oil }

	st_res_init_good_effect = { GOOD = aeroplanes }

	st_res_init_good_effect = { GOOD = fertilizer }

	if = {
		limit = { NOT = { has_variable = st_res_adjust_step_tier } }
		set_variable = { name = st_res_adjust_step_tier value = 1 }
	}

		limit = { NOT = { has_variable = st_res_hub_level_cached } }
		set_variable = { name = st_res_hub_level_cached value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_hub_throughput_cached } }

		limit = { NOT = { has_variable = st_res_hub_workforce_cached } }

	}
}

# Parameter: GOOD.
# --------------------------------------------------------------------------
st_res_reset_good_vars_effect = {
	set_variable = { name = st_res_$GOOD$_stored value = 0 }
	set_variable = { name = st_res_$GOOD$_rate value = 0 }
	set_variable = { name = st_res_$GOOD$_last_delta value = 0 }
	set_variable = { name = st_res_$GOOD$_last_status value = 0 }
	set_variable = { name = st_res_$GOOD$_policy value = 0 }
	set_variable = { name = st_res_$GOOD$_engaged value = 0 }
	set_variable = { name = st_res_$GOOD$_policy_status value = 0 }
	set_variable = { name = st_res_$GOOD$_policy_price value = 0 }
	st_res_apply_preset_standard_base = { GOOD = $GOOD$ }
	# The running price average is removed rather than zeroed, so the next
	# weekly tick re-seeds it from the live price instead of averaging up
	# from "at base price".
	remove_variable = st_res_$GOOD$_price_avg
}

# --------------------------------------------------------------------------
# st_res_refresh_hub_cache_effect — Bridge building -> country scope.
# Reads the live hub's level and throughput modifier from building scope and
# caches them on country vars so the rest of the system can stay country-scoped.
# With no hub every cache reads 0, staffing included: no hub, no staff. It
# read 1 until the entry learned to wait without a hub (je_strategic_reserve
# has no invalid), and every flow gate (st_res_<good>_actual_rate and the
# flow modifiers) tests staffing, so a waiting reserve with a release rate
# still booked sales income.
# --------------------------------------------------------------------------
st_res_refresh_hub_cache_effect = {
	set_variable = { name = st_res_hub_level_cached value = 0 }
	set_variable = { name = st_res_hub_throughput_cached value = 0 }
	set_variable = { name = st_res_hub_workforce_cached value = 0 }

	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_strategic_reserve_hub
			}
		}
		random_scope_building = {
			limit = { is_building_type = building_strategic_reserve_hub }
			owner = {
				set_variable = { name = st_res_hub_level_cached value = prev.level }
				set_variable = { name = st_res_hub_throughput_cached value = prev.modifier:building_throughput_add }
				set_variable = { name = st_res_hub_workforce_cached value = prev.occupancy }
			}
		}
	}
}

# --------------------------------------------------------------------------
# st_res_record_hub_holder_effect — country scope, weekly pulse.
# Stamps the hub's state with the country holding it (state var
# st_res_hub_holder). on_state_owner_change gives script only the state, not
# its previous owner, so st_res_on_state_owner_change reads this stamp to tell
# a capture from a hand-over between the two sides of a civil war
# (st_res_hub_stays_within_nation, st_res_triggers.txt).
# --------------------------------------------------------------------------
st_res_record_hub_holder_effect = {
	save_scope_as = st_res_hub_holder_country
	every_scope_building = {
		limit = { is_building_type = building_strategic_reserve_hub }
		state = {
			set_variable = { name = st_res_hub_holder value = scope:st_res_hub_holder_country }
		}
	}
}

# --------------------------------------------------------------------------
# st_res_relocate_hub_to_capital_effect — country scope, first thing in the
# weekly pulse.
# The hub belongs in the capital: its possible requires it, and the policy
# price signal assumes the capital's market (strategic_reserve_system.md
# §4.6). It ends up elsewhere when the capital moves, or when a hub state
# passes to the other side of a civil war (st_res_hub_stays_within_nation),
# and it cannot be demolished, so the reserve moves it: a level-1 hub is
# created in the capital for free, and every hub outside the capital is
# removed once the capital's stands. The stock lives on country variables and
# does not move. The new building hires its 5000 staff from scratch: until it
# is staffed, flow runs below capacity and the entry reads as deactivated
# (st_res_hub_workforce_cached), so each move costs the reserve some weeks.
#
# Two steps, each safe to repeat, so that no hub is removed before its
# replacement is seen. Whether create_building's result is visible to
# has_building later in the same effect is unread. If it is not, step 2 runs
# a week later, and for that week the country has two hubs, of which the
# st_res effects read either.
#
# create_building checks the unlocking technology since 1.13.9
# (scripting_best_practices.md), hence the logistics test: without it the
# call would fail, and log the failure, every week. Whether it also checks
# possible is unverified, so st_res_hub_relocating lets the hub's
# one-per-country guard pass either way (the te_cm_placing_site pattern).
# --------------------------------------------------------------------------
st_res_relocate_hub_to_capital_effect = {
	# Step 1: no hub in the capital, one elsewhere: build one in the capital.
	if = {
		limit = {
			has_technology_researched = logistics
			any_scope_building = { is_building_type = building_strategic_reserve_hub }
			capital ?= {
				NOT = { has_building = building_strategic_reserve_hub }
			}
		}
		set_variable = { name = st_res_hub_relocating value = 1 }
		capital = {
			create_building = {
				building = building_strategic_reserve_hub
				level = 1
			}
		}
		remove_variable = st_res_hub_relocating
		debug_log = "TE_ST_RES: creating a Strategic Reserve Hub in the capital to replace one outside it"
	}
	# Step 2: the capital has its hub: remove every other one.
	if = {
		limit = {
			capital ?= { has_building = building_strategic_reserve_hub }
		}
		every_scope_state = {
			limit = {
				is_capital = no
				has_building = building_strategic_reserve_hub
			}
			remove_building = building_strategic_reserve_hub
			if = {
				limit = { has_variable = st_res_hub_holder }
				remove_variable = st_res_hub_holder
			}
		}
	}
}

# --------------------------------------------------------------------------
# st_res_startup_good_setup_effect — Country-scoped per-good startup helper.
# Parameter: GOOD = any reserve good (list at st_res_init_good_effect).
# Populates local vars the building-scoped helpers expect before the
# country -> building scope bridge.
# --------------------------------------------------------------------------
st_res_startup_good_setup_effect = {
	# Parameter: GOOD = any reserve good (list at st_res_init_good_effect)
	set_local_variable = { name = st_res_$GOOD$_rate_local value = var:st_res_$GOOD$_actual_rate_cached }
	set_local_variable = { name = st_res_$GOOD$_good_mult_local value = st_res_$GOOD$_good_mult }
	set_local_variable = { name = st_res_$GOOD$_can_store_local value = 0 }
	set_local_variable = { name = st_res_$GOOD$_can_withdraw_local value = 0 }

	# `<=` not `<`: at exactly full capacity the hub still needs to buy decay
	# replacement to maintain the stockpile. weekly_delta nets to ~0 at full so
	# the level stays flat. Using `<` would idle the hub at full and silently
	# leak goods to decay (script reports stable level while market shows nothing).
	if = {
		limit = {
			local_var:st_res_$GOOD$_rate_local > 0
			var:st_res_$GOOD$_stored <= st_res_$GOOD$_capacity
			var:st_res_hub_workforce_cached > 0.99
		}
		set_local_variable = { name = st_res_$GOOD$_can_store_local value = 1 }
	}
	if = {
		limit = {
			local_var:st_res_$GOOD$_rate_local < 0
			var:st_res_$GOOD$_stored > 0
			var:st_res_hub_workforce_cached > 0.99
		}
		set_local_variable = { name = st_res_$GOOD$_can_withdraw_local value = 1 }
	}
}

# --------------------------------------------------------------------------
# st_res_rebuild_good_flow_modifiers_effect — Building-scoped per-good helper.
# Parameter: GOOD = any reserve good (list at st_res_init_good_effect).
# Expects the corresponding st_res_$GOOD$_*_local values to already be populated in
# the calling effect before the country -> building scope bridge.
# --------------------------------------------------------------------------
st_res_rebuild_good_flow_modifiers_effect = {
	if = { limit = { has_modifier = st_res_$GOOD$_store_flow } remove_modifier = st_res_$GOOD$_store_flow }
	if = { limit = { has_modifier = st_res_$GOOD$_withdraw_flow } remove_modifier = st_res_$GOOD$_withdraw_flow }

	if = {
		limit = {
			local_var:st_res_$GOOD$_can_store_local > 0
		}
		add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = -1 }
		add_modifier = { name = st_res_$GOOD$_store_flow multiplier = local_var:st_res_$GOOD$_good_mult_local }
	}
	else_if = {
		limit = {
			local_var:st_res_$GOOD$_can_withdraw_local > 0
		}
		add_modifier = { name = st_res_$GOOD$_store_flow multiplier = -1 }
		add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = local_var:st_res_$GOOD$_good_mult_local }
	}
	else = {
		add_modifier = { name = st_res_$GOOD$_store_flow multiplier = -1 }
		add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = -1 }
	}
}

# --------------------------------------------------------------------------
# st_res_idle_hub_flow_modifiers_effect — building scope (a hub). Sets every
# good's flow to nothing, as st_res_rebuild_good_flow_modifiers_effect's
# `else` does. For a hub whose owner has no running entry to rebuild its
# flow: the rebels of a revolution, who keep the nation's hub
# (st_res_on_state_owner_change) but start no reserve. Without it the hub
# would keep buying or selling on the loser's last settings, with nothing
# booked on either side, for the whole war.
# --------------------------------------------------------------------------
st_res_idle_good_flow_modifiers_effect = {
	if = { limit = { has_modifier = st_res_$GOOD$_store_flow } remove_modifier = st_res_$GOOD$_store_flow }
	if = { limit = { has_modifier = st_res_$GOOD$_withdraw_flow } remove_modifier = st_res_$GOOD$_withdraw_flow }
	add_modifier = { name = st_res_$GOOD$_store_flow multiplier = -1 }
	add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = -1 }
}

# --------------------------------------------------------------------------
# st_res_set_good_status_effect — Derive the display status code for one good.
# Parameter: GOOD.
#
# This is the ONLY site that decides what the JE inventory widget shows as a
# good's status: the widget and its tooltips read `st_res_$GOOD$_last_status`
# and never re-derive the reason from GUI expressions. It is called once, at
# the end of st_res_refresh_hub_flow_effect, which is itself the shared tail of
# both the weekly pulse and every rate/step button press — so one call site
# covers both "what happened this tick" and "what will happen now that you
# clicked", without the label lagging a week behind the player's input.
#
# Status codes (mirrored in st_res_custom_loc.txt and the widget loc keys):
#   0 = Idle              — nothing configured, nothing moving
#   1 = Storing           — configured intake is being applied in full
#   2 = Withdrawing       — configured release is being applied in full
#   3 = Storing, clipped by the hub's weekly flow cap
#   4 = Withdrawing, clipped by the hub's weekly flow cap
#   5 = Blocked, stockpile at capacity
#   6 = Blocked, stockpile empty while a release is configured
#   7 = Blocked, hub understaffed (occupancy below 100%)
#   8 = Blocked, no Strategic Reserve Hub
#
# Precedence note: st_res_clamp_stockpiles_effect drives the configured rate to
# 0 once a stockpile hits a bound, so the "full" branch deliberately accepts
# `rate >= 0` — otherwise a full reserve would report Idle the tick after the
# auto-clamp fires. The "empty" branch keeps `rate < 0` on purpose: an empty,
# unconfigured reserve is genuinely idle, and the row tooltip says it is empty.
# --------------------------------------------------------------------------
st_res_set_good_status_effect = {
	set_local_variable = { name = st_res_status_capacity value = st_res_$GOOD$_capacity }

	if = {
		limit = { var:st_res_hub_level_cached <= 0 }
		set_variable = { name = st_res_$GOOD$_last_status value = 8 }
	}
	else_if = {
		limit = { var:st_res_hub_workforce_cached <= 0.99 }
		set_variable = { name = st_res_$GOOD$_last_status value = 7 }
	}
	else_if = {
		limit = {
			local_var:st_res_status_capacity > 0
			var:st_res_$GOOD$_stored >= st_res_$GOOD$_capacity
			var:st_res_$GOOD$_rate >= 0
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 5 }
	}
	else_if = {
		limit = {
			var:st_res_$GOOD$_stored <= 0
			var:st_res_$GOOD$_rate < 0
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 6 }
	}
	else_if = {
		limit = {
			var:st_res_$GOOD$_rate > 0
			var:st_res_$GOOD$_actual_rate_cached > st_res_weekly_base_rate_cap
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 3 }
	}
	else_if = {
		limit = {
			var:st_res_$GOOD$_rate < 0
			var:st_res_$GOOD$_actual_rate_cached < st_res_neg_weekly_base_rate_cap
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 4 }
	}
	else_if = {
		limit = { var:st_res_$GOOD$_rate > 0 }
		set_variable = { name = st_res_$GOOD$_last_status value = 1 }
	}
	else_if = {
		limit = { var:st_res_$GOOD$_rate < 0 }
		set_variable = { name = st_res_$GOOD$_last_status value = 2 }
	}
	else = {
		set_variable = { name = st_res_$GOOD$_last_status value = 0 }
	}
}

# --------------------------------------------------------------------------
# st_res_refresh_hub_flow_effect — Refresh hub cache, then rebuild the transient
# building modifiers that express the current country-owned reserve settings.
# --------------------------------------------------------------------------
st_res_refresh_hub_flow_effect = {
	st_res_init_effect = yes
	st_res_refresh_hub_cache_effect = yes

	# Cache the computed actual gross flow on country vars so UI/tooltips and
	# button gating can read a stable numeric value without re-evaluating
	# script-values in JE/button scope (avoids 'none').
	set_variable = { name = st_res_grain_actual_rate_cached value = st_res_grain_actual_rate }
	set_variable = { name = st_res_ammunition_actual_rate_cached value = st_res_ammunition_actual_rate }
	set_variable = { name = st_res_oil_actual_rate_cached value = st_res_oil_actual_rate }
	set_variable = { name = st_res_small_arms_actual_rate_cached value = st_res_small_arms_actual_rate }

# --------------------------------------------------------------------------
# st_res_apply_weekly_good_effect — Apply one weekly tick for a single good and
# record what actually moved. Parameter: GOOD.
#
# The stockpile arithmetic is unchanged from the pre-widget implementation
# (add weekly_delta, clamp into [0, capacity]); the only addition is the
# before/after snapshot that writes `st_res_$GOOD$_last_delta`. That variable is
# the ACTUAL net weekly movement, i.e. after the capacity/empty clamp bites, so
# the number shown in the JE inventory widget always matches what bookkeeping
# really did rather than what the configured rate asked for.
# --------------------------------------------------------------------------
st_res_apply_weekly_good_effect = {
	set_local_variable = { name = st_res_weekly_prev value = var:st_res_$GOOD$_stored }

	change_variable = { name = st_res_$GOOD$_stored add = st_res_$GOOD$_weekly_delta }
	clamp_variable = { name = st_res_$GOOD$_stored min = 0 max = st_res_$GOOD$_capacity }

	set_local_variable = { name = st_res_weekly_moved value = var:st_res_$GOOD$_stored }
	change_local_variable = {
		name = st_res_weekly_moved
		add = {
			value = local_var:st_res_weekly_prev
			multiply = -1
		}
	}
	set_variable = { name = st_res_$GOOD$_last_delta value = local_var:st_res_weekly_moved }
}

# ============================================================================
# Reserve policies
# ============================================================================
# A policy lets one good's weekly rate be re-decided automatically from the
# market price instead of by hand. Policies are OFF by default: every good,
# in every save, starts at policy 0 (Manual) with its rate untouched.
#
# Policy ids (st_res_<good>_policy):
#   0 = Manual                   — the player drives the rate; nothing automated
#   1 = Buy When Cheap           — buys below the purchase threshold only
#   2 = Release When Expensive   — sells above the release threshold only
#   3 = Stabilize Prices         — both sides
#   4 = Stockpile                — buys at or below the purchase threshold, at
#                                  full flow (the ramp is not applied to
#                                  purchases), and releases above the release
#                                  threshold only at war; in peacetime only
#                                  above st_res_policy_stockpile_peace_release.
#                                  Selecting it loads the Stockpile preset.
#
# Why a fifth policy rather than a positive purchase threshold on Buy When
# Cheap or Stabilize Prices: a good you want in hand for a war is worth
# buying at about base price, but the ramp sells the reserve short there (a
# price 5 points under a threshold of 0 buys a quarter of the flow on a
# 20-point ramp), and Stabilize Prices would sell the stock into the first
# peacetime spike. Both are rules about how a lane responds, not settings, so
# they need their own code. The purchase threshold's range does go up to
# st_res_policy_buy_thr_ceiling for every policy, so a stockpile can be tuned.
#
# Engagement latch (st_res_<good>_engaged), the hysteresis state:
#   0 = idle | 1 = engaged buying | 2 = engaged releasing
# With a step response an engaged lane keeps going until the price crosses the
# *stop* threshold, which sits st_res_policy_buy_band above the purchase
# threshold, or st_res_policy_sell_band below the release threshold. That is
# what stops a price oscillating around a single trigger from flapping the
# lane on and off week after week.
#
# Two settings shape HOW a lane responds, both per good:
#   st_res_<good>_price_memory — weeks. The policy acts on a running average
#       of the market price, advanced once a week by
#       st_res_policy_track_price_effect: avg += (live - avg) / memory.
#       1 = the live price. Read through st_res_<good>_price_signal.
#   st_res_<good>_ramp — percentage points. 0 = step: full maximum flow the
#       moment a threshold is crossed, with the hysteresis bands. > 0 =
#       linear: the flow rises from 0 at the threshold to the maximum `ramp`
#       points beyond it, and the bands are not applied because a continuous
#       response has no on/off edge to debounce.
# Both exist to stop the reserve's own trading from chasing the price it
# moved — the step response at full flow is the most oscillation-prone limit
# of the family, and it is still available (ramp 0) for anyone who wants it.
#
# Policy status codes (st_res_<good>_policy_status) — the plain-language
# explanation the inventory row shows, rendered through
# st_res_<good>_policy_reason_text in st_res_custom_loc.txt:
#   0 = Manual (no policy)
#   1 = Buying
#   2 = Buying, limited by the weekly budget
#   3 = Releasing
#   4 = Waiting — the price has not crossed a trigger threshold
#   5 = Paused — target stockpile reached
#   6 = Paused — protected stockpile reached
#   7 = Paused — weekly budget spent
#   8 = Paused — hub understaffed
#   9 = Paused — no Strategic Reserve Hub
#  10 = Holding — Stockpile at peace, the price above the release threshold
#       but not above st_res_policy_stockpile_peace_release
#
# st_res_policy_evaluate_good_effect is the ONLY writer of both
# st_res_<good>_policy_status and st_res_<good>_engaged, in the same spirit as
# st_res_set_good_status_effect owning st_res_<good>_last_status. It is called
# from BOTH branches of the weekly pulse and from every policy-panel press, so
# no other effect ever has to guess a status code.
# --------------------------------------------------------------------------

# --------------------------------------------------------------------------
# Presets. THESE FOUR EFFECTS ARE THE TUNING TABLE — the fixed preset numbers
# below exist nowhere else in the mod. Everything else about the
# policies (step sizes, hysteresis bands, stepper bounds) lives in the
# `st_res_policy_*` constants at the bottom of
# common/script_values/st_res_script_values.txt.
#
# Thresholds are percentage points against the good's base price on the
# country's own market. Floor and ceiling are percentages of the good's current
# reserve capacity, so they scale automatically when the player adds Silos.
# Budget is an ESTIMATED weekly spend cap in GBP (see
# st_res_policy_evaluate_good_effect for why it can only be an estimate).
# Price memory is the averaging window in weeks and the ramp is in percentage
# points past each threshold — see st_res_policy_track_price_effect and the
# response-shape block of st_res_policy_evaluate_good_effect. The presets
# smooth harder the more patient they are: a Conservative reserve waits for a
# real bargain and should not be spooked by one week's price.
#
# The stock band is the player's, not the presets': every preset uses the
# whole capacity (floor_pct 0, ceil_pct 100), and the presets differ only in
# how they trade. Until 2026-09-26 they kept 40-60 / 20-80 / 10-95% of
# capacity. A floor the policy can never sell is dead weight: it still pays
# its decay replacement every week (st_res_policy_afford). A ceiling below
# 100 leaves capacity the player built and never uses. Under Conservative,
# the AI's preset, only 40% to 60% of capacity ever moved, and the AI never
# reached the 75% fill at which silos become worth building to it
# (st_res_reserve_filling_up). Players can still set both with the steppers.
#
# Maximum weekly flow and the weekly budget are not fixed numbers: each preset
# sets them as a share of the good's capacity and of weekly GDP
# (st_res_preset_*_flow_pct / _budget_pct, st_res_script_values.txt), and they
# go on following the country until the player sets them by hand
# (st_res_preset_follow_magnitudes_base below).
#
# The four presets keep sell_thr - buy_thr >= st_res_policy_min_gap (15) and
# ceil_pct - floor_pct >= st_res_policy_stock_min_gap (10), which is the same
# non-overlap rule the steppers are gated on.
#

# 2 Standard, 3 Aggressive, 4 Stockpile), the same codes as the follow

# would hardly ever buy. Its only reader is
# st_res_policy_<good>_sgui, which disables the button of the preset in force,
# as it disables the button of the policy in force. A save from before the
# marker has none until the player applies a preset, so no preset button
# starts disabled there.
# --------------------------------------------------------------------------

# --------------------------------------------------------------------------
# Preset magnitudes that follow the country.
#
# st_res_preset_follow_magnitudes_base marks both magnitude settings of one
# good as following preset PRESET (1 Conservative, 2 Standard, 3 Aggressive:
# st_res_<good>_auto_max_flow / _auto_budget) and writes them from today's
# capacity and GDP. st_res_refresh_preset_magnitudes_effect rewrites every
# still-marked setting each weekly pulse, before the policies are evaluated,
# so a preset keeps pace as silos go up and the economy grows. A stepper click
# on either setting removes its marker (st_res_policy_magnitude_step_up_base /
# _down_base), and from then on the player's number stands. Each setting has
# its own marker, so setting the budget by hand leaves the flow following, and
# changing a threshold, the stock band, the price memory or the ramp leaves
# both following.
#
# The writers round to the stepper's grid and clamp to the stepper's range
# (flow: st_res_policy_flow_min to st_res_policy_flow_max, the hub's weekly
# cap; budget: st_res_policy_budget_min to st_res_<good>_policy_budget_max),
# so a following value is always one the steppers could have reached.
# Parameters: GOOD; PRESET; FLOW_PCT and BUDGET_PCT, the preset's script-value
# names.
# --------------------------------------------------------------------------
st_res_preset_follow_magnitudes_base = {
	set_variable = { name = st_res_$GOOD$_auto_max_flow value = $PRESET$ }
	set_variable = { name = st_res_$GOOD$_auto_budget value = $PRESET$ }
	st_res_preset_write_flow_base = { GOOD = $GOOD$ PCT = $FLOW_PCT$ }
	st_res_preset_write_budget_base = { GOOD = $GOOD$ PCT = $BUDGET_PCT$ }
}

			multiply = st_res_policy_flow_step

	}
}

st_res_preset_write_budget_base = {

#
# It writes st_res_<good>_rate and then applies exactly the two clamps
# st_res_clamp_stockpiles_effect applies to that rate, which are the same
# bounds the manual +/- buttons are gated on: the hub's signed weekly flow cap,
# and the remaining storage room / remaining stockpile. Everything downstream —
# the flow cap, hub throughput, staffing, decay, capacity, the per-good tech
# unlocks and the cost of the goods themselves — therefore applies to an

	clamp_variable = { name = st_res_$GOOD$_rate min = st_res_$GOOD$_max_withdrawable max = st_res_$GOOD$_max_storable }
}

# --------------------------------------------------------------------------
# st_res_policy_track_price_effect — advance one good's running price average
# by one week. Parameter: GOOD. Country scope. WEEKLY PULSE ONLY.
#
# The policies act on an exponential moving average of the national market
# price rather than this week's reading:
#
#     avg <- avg + (live - avg) / price_memory
#
# i.e. A * live + (1 - A) * avg with A = 1 / memory, so `price_memory` is the
# averaging time constant in weeks and memory 1 is the live price. The average
# is kept in percentage points against base price, the unit the thresholds
# use, and the widget shows it next to the live price.
#
# It runs for every good under every policy, Manual included, so the average
# is already warm when the player switches a policy on. It is seeded from the
# live price the first time it runs (and again after st_res_reset_vars_effect
# removes it), never from 0 — a cold start at 0 would read as "at base price"
# for weeks.
#
# It is deliberately NOT part of st_res_policy_evaluate_good_effect: the
# evaluator also runs on every policy-panel click, and a click must re-decide
# the lane against the same signal the last weekly tick used, not drag the
# average toward today's price once per click.
# --------------------------------------------------------------------------
st_res_policy_track_price_effect = {
	set_local_variable = {
		name = st_res_policy_live_price
		value = {
			value = st_res_$GOOD$_price_rel
			multiply = 100
		}
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_price_avg } }
		set_variable = { name = st_res_$GOOD$_price_avg value = local_var:st_res_policy_live_price }
	}
	else = {
		change_variable = {
			name = st_res_$GOOD$_price_avg
			add = {
				value = local_var:st_res_policy_live_price
				subtract = var:st_res_$GOOD$_price_avg
				divide = {
					value = var:st_res_$GOOD$_price_memory
					min = 1
				}
			}
		}
	}
}

# --------------------------------------------------------------------------
# st_res_policy_tick_good_effect — one good's weekly policy step: advance the
# price average, then re-decide the lane. Parameter: GOOD. This is what the
# weekly pulse calls, in BOTH of its branches. The policy panel calls the
# evaluator directly (through st_res_policy_apply_good_effect), so a click
# never advances the average.
# --------------------------------------------------------------------------
st_res_policy_tick_good_effect = {
	st_res_policy_track_price_effect = { GOOD = $GOOD$ }
	st_res_policy_evaluate_good_effect = { GOOD = $GOOD$ }
}

# --------------------------------------------------------------------------
# st_res_policy_evaluate_good_effect — one good, one weekly decision.
# Parameter: GOOD. Country scope. Expects st_res_init_effect and
# st_res_refresh_hub_cache_effect to have run.
#
# Cost for a good on Manual or still locked by tech: one variable comparison
# and two variable writes. No iterators anywhere in this file's policy path.
#
# ON THE BUDGET. Reserve purchases are not a money effect — the hub buys its
# input goods through production-method modifiers, so the money leaves the
# treasury inside the normal building-expense system and there is no cost value
# to read back. The weekly budget is therefore an ESTIMATE, computed here as
# (budget / current unit price) - this week's decay replacement, and the
# player-facing localization says so. It is a per-week cap, not a running pot:
# whatever is not spent this week does not carry over. That keeps it stateless,
# save-safe and impossible to desynchronise from the real spend.
# --------------------------------------------------------------------------
st_res_policy_evaluate_good_effect = {
	if = {
		# Manual, or a good this country has not unlocked yet.
		limit = {
			OR = {
				NOT = { has_variable = st_res_$GOOD$_policy }
				var:st_res_$GOOD$_policy = 0
				NOT = { st_res_$GOOD$_unlocked_trigger = yes }
			}
		}
		set_variable = { name = st_res_$GOOD$_policy_status value = 0 }
		set_variable = { name = st_res_$GOOD$_engaged value = 0 }
	}
	else_if = {
		# No hub: nothing to drive. Leave the configured rate alone so it is
		# still there when a hub stands again (the entry waits for one).
		limit = { var:st_res_hub_level_cached <= 0 }
		set_variable = { name = st_res_$GOOD$_policy_status value = 9 }
		set_variable = { name = st_res_$GOOD$_engaged value = 0 }
	}
	else = {
		# --- Inputs -------------------------------------------------------
		# The price signal, in percentage points against base price: the
		# running average kept by st_res_policy_track_price_effect, or the
		# live price until the first weekly tick has seeded it. Reading the
		# stored average rather than the market here is what lets a panel
		# click re-decide the lane without also advancing the average.
		set_local_variable = { name = st_res_policy_price value = st_res_$GOOD$_price_signal }
		set_variable = { name = st_res_$GOOD$_policy_price value = local_var:st_res_policy_price }

		# Protected floor and target ceiling, converted from percent of
		# capacity into units.
		set_local_variable = {
			name = st_res_policy_room
			value = {
				value = st_res_$GOOD$_capacity
				multiply = var:st_res_$GOOD$_ceil_pct
				divide = 100
				subtract = var:st_res_$GOOD$_stored
				min = 0
			}
		}
		set_local_variable = {
			name = st_res_policy_spare
			value = {
				value = var:st_res_$GOOD$_stored
				subtract = {
					value = st_res_$GOOD$_capacity
					multiply = var:st_res_$GOOD$_floor_pct
					divide = 100
				}
				min = 0
			}
		}

		# Estimated units this week's budget can pay for, net of the decay
		# replacement the hub buys before any of the configured rate.
		set_local_variable = {
			name = st_res_policy_afford
			value = {
				value = var:st_res_$GOOD$_budget
				divide = st_res_$GOOD$_unit_price
				subtract = st_res_$GOOD$_weekly_decay
				min = 0
			}
		}

		# --- Hysteresis ---------------------------------------------------
		# With the step response (ramp 0) an engaged lane uses the stop
		# threshold instead of the start one. With a ramp the response is
		# continuous — the flow is already near zero at the threshold — so
		# there is no on/off edge to debounce and the bands are not applied.
		# Stockpile buys at full flow whatever the ramp, so its purchases
		# always take the band.
		set_local_variable = { name = st_res_policy_buy_at value = var:st_res_$GOOD$_buy_thr }
		if = {
			limit = {
				var:st_res_$GOOD$_engaged = 1
				OR = {
					var:st_res_$GOOD$_ramp <= 0
					var:st_res_$GOOD$_policy = 4
				}
			}
			change_local_variable = { name = st_res_policy_buy_at add = st_res_policy_buy_band }
		}
		set_local_variable = { name = st_res_policy_sell_at value = var:st_res_$GOOD$_sell_thr }
		# Stockpile keeps its stock for a war: in peacetime it releases only
		# above st_res_policy_stockpile_peace_release, or the release
		# threshold if that is higher.
		if = {
			limit = {
				var:st_res_$GOOD$_policy = 4
				is_at_war = no
				local_var:st_res_policy_sell_at < st_res_policy_stockpile_peace_release
			}
			set_local_variable = { name = st_res_policy_sell_at value = st_res_policy_stockpile_peace_release }
		}
		if = {
			limit = {
				var:st_res_$GOOD$_engaged = 2
				var:st_res_$GOOD$_ramp <= 0
			}
			change_local_variable = { name = st_res_policy_sell_at subtract = st_res_policy_sell_band }
		}

		# --- Response shape -----------------------------------------------
		# The share of the maximum weekly flow the price position calls for.
		# Step (ramp 0): the full flow whenever the lane fires. Ramp > 0:
		# rises linearly from 0 at the threshold to 1 at `ramp` points beyond
		# it, so the reserve leans harder the further the price strays and
		# eases off as it recovers instead of flipping between full flow and
		# nothing. Both sides are computed here so the buy and release
		# branches below keep their shape; a side that does not fire never
		# reads its fraction.
		set_local_variable = { name = st_res_policy_buy_frac value = 1 }
		set_local_variable = { name = st_res_policy_sell_frac value = 1 }
		if = {
			limit = { var:st_res_$GOOD$_ramp > 0 }
			set_local_variable = {
				name = st_res_policy_buy_frac
				value = {
					value = local_var:st_res_policy_buy_at
					subtract = local_var:st_res_policy_price
					divide = var:st_res_$GOOD$_ramp
					min = 0
					max = 1
				}
			}
			set_local_variable = {
				name = st_res_policy_sell_frac
				value = {
					value = local_var:st_res_policy_price
					subtract = local_var:st_res_policy_sell_at
					divide = var:st_res_$GOOD$_ramp
					min = 0
					max = 1
				}
			}
		}
		# Stockpile is not throttled near its purchase threshold: it buys
		# at full flow toward the target, as near base price as it is set.
		if = {
			limit = { var:st_res_$GOOD$_policy = 4 }
			set_local_variable = { name = st_res_policy_buy_frac value = 1 }
		}

		# --- Decision -----------------------------------------------------
		set_local_variable = { name = st_res_policy_target value = 0 }
		set_local_variable = { name = st_res_policy_next_engaged value = 0 }
		set_local_variable = { name = st_res_policy_next_status value = 4 }

		if = {
			# BUY side — Buy When Cheap and Stabilize Prices below the
			# threshold, Stockpile at or below it.
			limit = {
				OR = {
					AND = {
						OR = {
							var:st_res_$GOOD$_policy = 1
							var:st_res_$GOOD$_policy = 3
						}
						local_var:st_res_policy_price < local_var:st_res_policy_buy_at
					}
					AND = {
						var:st_res_$GOOD$_policy = 4
						local_var:st_res_policy_price <= local_var:st_res_policy_buy_at
					}
				}
			}
			if = {
				limit = { local_var:st_res_policy_room <= 0 }
				set_local_variable = { name = st_res_policy_next_status value = 5 }
			}
			else_if = {
				limit = { local_var:st_res_policy_afford <= 0 }
				set_local_variable = { name = st_res_policy_next_status value = 7 }
			}
			else = {
				# What the policy wants before money is considered: the
				# maximum flow scaled by the response fraction, then capped
				# by the room left under the target and the hub's flow cap.
				set_local_variable = {
					name = st_res_policy_want
					value = {
						value = var:st_res_$GOOD$_max_flow
						multiply = local_var:st_res_policy_buy_frac
						max = local_var:st_res_policy_room
						max = st_res_weekly_base_rate_cap
						min = 0
					}
				}
				# ...and what it can actually afford this week.
				set_local_variable = {
					name = st_res_policy_target
					value = {
						value = local_var:st_res_policy_want
						max = local_var:st_res_policy_afford
					}
				}
				set_local_variable = { name = st_res_policy_next_engaged value = 1 }
				set_local_variable = { name = st_res_policy_next_status value = 1 }
				if = {
					limit = { local_var:st_res_policy_afford < local_var:st_res_policy_want }
					set_local_variable = { name = st_res_policy_next_status value = 2 }
				}
			}
		}
		else_if = {
			# RELEASE side — Release When Expensive, Stabilize Prices and
			# Stockpile (whose sell_at the peacetime rule above raised).
			# Reachable only when the buy side did not fire, and the two bands
			# are held apart by st_res_policy_min_gap, so the sides can never
			# both be true for the same price.
			limit = {
				OR = {
					var:st_res_$GOOD$_policy = 2
					var:st_res_$GOOD$_policy = 3
					var:st_res_$GOOD$_policy = 4
				}
				local_var:st_res_policy_price > local_var:st_res_policy_sell_at
			}
			if = {
				limit = { local_var:st_res_policy_spare <= 0 }
				set_local_variable = { name = st_res_policy_next_status value = 6 }
			}
			else = {
				set_local_variable = {
					name = st_res_policy_target
					value = {
						value = var:st_res_$GOOD$_max_flow
						multiply = local_var:st_res_policy_sell_frac
						max = local_var:st_res_policy_spare
						max = st_res_weekly_base_rate_cap
						min = 0
						multiply = -1
					}
				}
				set_local_variable = { name = st_res_policy_next_engaged value = 2 }
				set_local_variable = { name = st_res_policy_next_status value = 3 }
			}
		}
		# A stockpile at peace with the price past its own release threshold
		# but short of the peacetime one is holding for a war, not waiting:
		# say so, or the row would claim the price "has not crossed a
		# trigger threshold" while it plainly has.
		if = {
			limit = {
				local_var:st_res_policy_next_status = 4
				var:st_res_$GOOD$_policy = 4
				is_at_war = no
				local_var:st_res_policy_price > var:st_res_$GOOD$_sell_thr
			}
			set_local_variable = { name = st_res_policy_next_status value = 10 }
		}

		# An understaffed hub trades nothing whatever the rate says, so the
		# explanation has to lead with that. The target is still written, so
		# the lane resumes on the correct rate the moment staffing recovers.
		if = {
			limit = { var:st_res_hub_workforce_cached <= 0.99 }
			set_local_variable = { name = st_res_policy_next_status value = 8 }
		}

		# --- Commit -------------------------------------------------------
		set_variable = { name = st_res_$GOOD$_engaged value = local_var:st_res_policy_next_engaged }
		set_variable = { name = st_res_$GOOD$_policy_status value = local_var:st_res_policy_next_status }
		st_res_apply_rate_target_base = { GOOD = $GOOD$ }
	}
}

# --------------------------------------------------------------------------
# st_res_policy_apply_good_effect — re-decide ONE good right now and refresh the
# hub. Parameter: GOOD. Every branch of st_res_policy_<good>_sgui ends here, so
# a policy change, a preset or a stepper click shows its consequence on the same
# click instead of a week later — the same reason the rate buttons run the
# refresh tail.
# --------------------------------------------------------------------------
st_res_policy_apply_good_effect = {
	st_res_init_effect = yes
	st_res_refresh_hub_cache_effect = yes
	st_res_policy_evaluate_good_effect = { GOOD = $GOOD$ }
	st_res_refresh_hub_flow_effect = yes
	st_res_clamp_stockpiles_effect = yes
}

# --------------------------------------------------------------------------
# AI
# --------------------------------------------------------------------------
# building_strategic_reserve_hub carries a real ai_value (2000 for a great or
# major power at peace, 1000 at war, plus 10000 for any country above
# st_res_ai_hub_min_gdp — common/buildings/strategic_reserve.txt),
# so AI majors and any rich AI DO build hubs and DO get this journal entry. Before policies existed
# they simply never moved a rate and the reserve sat empty forever.
#
# st_res_ai_seed_good_effect hands an AI country the Conservative preset once,
# on Stabilize Prices for every good; st_res_ai_review_policies_effect (below)
# then moves its six military goods to Stockpile, and back, as its treasury
# allows. From then on the AI runs through st_res_policy_evaluate_good_effect —
# the same evaluator, the same thresholds, the same clamps and the same costs
# as a human on the same preset. There is no AI-only shortcut anywhere.
#
# It is one-shot, marked by st_res_policy_ai_seeded, and gated on is_ai, so it
# never touches a human player's goods and never re-seeds a country it has
# already set up. The one case it does overwrite tuned settings is a country
# that passes from a human to the AI mid-game: the marker was never set, so the
# AI seeds its own presets on the next pulse. That is the intended outcome for
# a country the player has walked away from, but it is a genuine overwrite,
# not something the marker prevents.
#
# Every good is on Stabilize Prices (policy 3) — buy below the threshold,
# release above it. Until 2026-09-26 the six military goods were on Buy When
# Cheap (policy 1), which never sells, so an AI reserve never released a round
# into a war's price spike (mobilized armies burn four times their peacetime
# ammunition: common/mobilization_options/te_wartime_munitions_injections.txt).
# Stabilizing may fill the whole capacity and release down to empty: no
# preset sets a stock band any more (see st_res_apply_preset_conservative_base).
#
# Saves seeded before then carry st_res_policy_ai_seeded but not
# st_res_policy_ai_stabilized. The second block runs once for them: it moves
# any good still on Buy When Cheap to Stabilize Prices, widens every good's
# stock band from the old preset's 40-60% to 0-100%, and sets the flow and
# budget to follow Conservative's shares of capacity and GDP in place of its
# old fixed 100 units and £5 000. It leaves the other settings alone.
# --------------------------------------------------------------------------
st_res_ai_seed_good_effect = {
	st_res_apply_preset_conservative_base = { GOOD = $GOOD$ }
	set_variable = { name = st_res_$GOOD$_policy value = $POLICY$ }
	set_variable = { name = st_res_$GOOD$_engaged value = 0 }
}

st_res_ai_stabilize_good_effect = {
	if = {
		limit = {
			has_variable = st_res_$GOOD$_policy
			var:st_res_$GOOD$_policy = 1
		}
		set_variable = { name = st_res_$GOOD$_policy value = 3 }
		set_variable = { name = st_res_$GOOD$_engaged value = 0 }
	}
	if = {
		limit = { has_variable = st_res_$GOOD$_floor_pct }
		set_variable = { name = st_res_$GOOD$_floor_pct value = 0 }
	}
	if = {
		limit = { has_variable = st_res_$GOOD$_ceil_pct }
		set_variable = { name = st_res_$GOOD$_ceil_pct value = 100 }
	}
	st_res_preset_follow_magnitudes_base = {
		GOOD = $GOOD$
		PRESET = 1
		FLOW_PCT = st_res_preset_conservative_flow_pct
		BUDGET_PCT = st_res_preset_conservative_budget_pct
	}
}

st_res_ai_seed_policies_effect = {
	if = {
		limit = {
			is_ai = yes
			NOT = { has_variable = st_res_policy_ai_seeded }
		}
		set_variable = { name = st_res_policy_ai_seeded value = 1 }
		set_variable = { name = st_res_policy_ai_stabilized value = 1 }
		st_res_ai_seed_good_effect = { GOOD = grain       POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = ammunition  POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = oil         POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = small_arms  POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = artillery   POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = aeroplanes  POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = tanks       POLICY = 3 }
		st_res_ai_seed_good_effect = { GOOD = fertilizer  POLICY = 3 }
	}
	if = {
		limit = {
			is_ai = yes
			NOT = { has_variable = st_res_policy_ai_stabilized }
		}
		set_variable = { name = st_res_policy_ai_stabilized value = 1 }
		st_res_ai_stabilize_good_effect = { GOOD = grain }
		st_res_ai_stabilize_good_effect = { GOOD = ammunition }
		st_res_ai_stabilize_good_effect = { GOOD = oil }
		st_res_ai_stabilize_good_effect = { GOOD = small_arms }
		st_res_ai_stabilize_good_effect = { GOOD = artillery }
		st_res_ai_stabilize_good_effect = { GOOD = aeroplanes }
		st_res_ai_stabilize_good_effect = { GOOD = tanks }
		st_res_ai_stabilize_good_effect = { GOOD = fertilizer }
	}
}

# --------------------------------------------------------------------------
# st_res_ai_review_policies_effect — the AI's military goods, reviewed every
# week. Country scope; weekly pulse only, after st_res_ai_seed_policies_effect.
#
# While the AI can afford to (st_res_ai_wants_stockpile: no default, no loans,
# and a quarter of its gold reserves' limit in hand to start), its six military
# goods run the Stockpile policy on the Stockpile preset: buy at or near base
# price in peace, release into a war's price spike. When it cannot, they go
# back to Stabilize Prices on Conservative, where seeding put them. Grain and
# chemicals stay on Stabilize Prices throughout. The goods are the six of
# st_res_military_goods_stockpile_effect; a locked good is set too, so it is
# ready the week it unlocks (the evaluator treats it as Manual until then).
#
# The stance is stored (st_res_ai_stance: 1 Stockpile, 2 Stabilize; absent
# reads as 2, which is what seeding sets), and the goods are rewritten only
# when it changes, so a settled AI costs a trigger and a comparison a week.
# A save from before the review has no stance, so its AI reserves move to
# Stockpile on their first pulse if they can afford it: that is the
# migration. st_res_reset_vars_effect removes the stance with the seeding
# markers, so a rebuilt hub is seeded and reviewed afresh.
#
# War is the policy's business, not this effect's: a stockpile releases above
# its release threshold at war (st_res_policy_evaluate_good_effect), so
# switching to Release When Expensive or Stabilize Prices on war start would
# only move the release later (Conservative releases above +30%, Stockpile
# above +20%). A war that empties the treasury still ends the stance through
# st_res_ai_wants_stockpile, once the AI borrows.
# --------------------------------------------------------------------------
st_res_ai_review_policies_effect = {
	if = {
		limit = {
			is_ai = yes
			has_variable = st_res_policy_ai_seeded
		}
		if = {
			limit = {
				st_res_ai_wants_stockpile = yes
				NOT = { var:st_res_ai_stance ?= 1 }
			}
			set_variable = { name = st_res_ai_stance value = 1 }
			st_res_set_good_policy_base = { GOOD = ammunition  POLICY = 4 PRESET = stockpile }
			st_res_set_good_policy_base = { GOOD = oil         POLICY = 4 PRESET = stockpile }
			st_res_set_good_policy_base = { GOOD = small_arms  POLICY = 4 PRESET = stockpile }
			st_res_set_good_policy_base = { GOOD = artillery   POLICY = 4 PRESET = stockpile }
			st_res_set_good_policy_base = { GOOD = aeroplanes  POLICY = 4 PRESET = stockpile }
			st_res_set_good_policy_base = { GOOD = tanks       POLICY = 4 PRESET = stockpile }
			debug_log = "TE_ST_RES: AI reserve stockpiles its military goods"
		}
		else_if = {
			limit = {
				st_res_ai_wants_stockpile = no
				var:st_res_ai_stance ?= 1
			}
			set_variable = { name = st_res_ai_stance value = 2 }
			st_res_set_good_policy_base = { GOOD = ammunition  POLICY = 3 PRESET = conservative }
			st_res_set_good_policy_base = { GOOD = oil         POLICY = 3 PRESET = conservative }
			st_res_set_good_policy_base = { GOOD = small_arms  POLICY = 3 PRESET = conservative }
			st_res_set_good_policy_base = { GOOD = artillery   POLICY = 3 PRESET = conservative }
			st_res_set_good_policy_base = { GOOD = aeroplanes  POLICY = 3 PRESET = conservative }
			st_res_set_good_policy_base = { GOOD = tanks       POLICY = 3 PRESET = conservative }
			debug_log = "TE_ST_RES: AI reserve can no longer afford to stockpile; military goods back to Stabilize Prices"
		}
	}
}
