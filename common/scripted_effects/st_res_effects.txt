# ============================================================================
# Strategic Reserve — Scripted Effects
# ============================================================================
# Scope contract:
# - Country scope owns all persistent Strategic Reserve vars and script values.
# - Building scope is only used to read live hub data and apply transient flow
#   modifiers to the hub itself.
# - Journal entries and scripted buttons should call the wrapper effects in the
#   lower half of this file rather than the low-level scope-bridge helpers.

# --------------------------------------------------------------------------
# st_res_init_good_effect — Per-good half of st_res_init_effect.
# Parameter: GOOD = grain | ammunition | oil | small_arms | artillery |
#            aeroplanes | tanks.
# `st_res_$GOOD$_last_delta` / `st_res_$GOOD$_last_status` are the display
# bookkeeping vars read by the JE inventory widget. They are SAVE-SAFE: a save
# made before the widget existed simply has neither, and this effect seeds them
# with 0 (= no movement recorded / status "Idle") until the next weekly tick
# overwrites them with real data.
# --------------------------------------------------------------------------
st_res_init_good_effect = {
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_stored } }
		set_variable = { name = st_res_$GOOD$_stored value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_rate } }
		set_variable = { name = st_res_$GOOD$_rate value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_last_delta } }
		set_variable = { name = st_res_$GOOD$_last_delta value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_$GOOD$_last_status } }
		set_variable = { name = st_res_$GOOD$_last_status value = 0 }
	}
}

# --------------------------------------------------------------------------
# st_res_init_effect — Initialize vars if missing. Called on JE immediate and the
# weekly pulse as a safety net. Existing saves keep their stockpile values and
# start with zero configured flow. The shared step-size tier defaults to 1,
# which means 10-unit adjustments.
# --------------------------------------------------------------------------
st_res_init_effect = {
	st_res_init_good_effect = { GOOD = grain }
	st_res_init_good_effect = { GOOD = ammunition }
	st_res_init_good_effect = { GOOD = oil }
	st_res_init_good_effect = { GOOD = small_arms }
	st_res_init_good_effect = { GOOD = artillery }
	st_res_init_good_effect = { GOOD = aeroplanes }
	st_res_init_good_effect = { GOOD = tanks }

	if = {
		limit = { NOT = { has_variable = st_res_adjust_step_tier } }
		set_variable = { name = st_res_adjust_step_tier value = 1 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_hub_level_cached } }
		set_variable = { name = st_res_hub_level_cached value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_hub_throughput_cached } }
		set_variable = { name = st_res_hub_throughput_cached value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = st_res_hub_workforce_cached } }
		set_variable = { name = st_res_hub_workforce_cached value = 1 }
	}
}

# --------------------------------------------------------------------------
# st_res_refresh_hub_cache_effect — Bridge building -> country scope.
# Reads the live hub's level and throughput modifier from building scope and
# caches them on country vars so the rest of the system can stay country-scoped.
# --------------------------------------------------------------------------
st_res_refresh_hub_cache_effect = {
	set_variable = { name = st_res_hub_level_cached value = 0 }
	set_variable = { name = st_res_hub_throughput_cached value = 0 }
	set_variable = { name = st_res_hub_workforce_cached value = 1 }

	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_strategic_reserve_hub
			}
		}
		random_scope_building = {
			limit = { is_building_type = building_strategic_reserve_hub }
			owner = {
				set_variable = { name = st_res_hub_level_cached value = prev.level }
				set_variable = { name = st_res_hub_throughput_cached value = prev.modifier:building_throughput_add }
				set_variable = { name = st_res_hub_workforce_cached value = prev.occupancy }
			}
		}
	}
}

# --------------------------------------------------------------------------
# st_res_startup_good_setup_effect — Country-scoped per-good startup helper.
# Parameter: GOOD = grain | ammunition | oil.
# Populates local vars the building-scoped helpers expect before the
# country -> building scope bridge.
# --------------------------------------------------------------------------
st_res_startup_good_setup_effect = {
	# Parameter: GOOD = grain | ammunition | oil
	set_local_variable = { name = st_res_$GOOD$_rate_local value = var:st_res_$GOOD$_actual_rate_cached }
	set_local_variable = { name = st_res_$GOOD$_good_mult_local value = st_res_$GOOD$_good_mult }
	set_local_variable = { name = st_res_$GOOD$_can_store_local value = 0 }
	set_local_variable = { name = st_res_$GOOD$_can_withdraw_local value = 0 }

	# `<=` not `<`: at exactly full capacity the hub still needs to buy decay
	# replacement to maintain the stockpile. weekly_delta nets to ~0 at full so
	# the level stays flat. Using `<` would idle the hub at full and silently
	# leak goods to decay (script reports stable level while market shows nothing).
	if = {
		limit = {
			local_var:st_res_$GOOD$_rate_local > 0
			var:st_res_$GOOD$_stored <= st_res_$GOOD$_capacity
			var:st_res_hub_workforce_cached > 0.99
		}
		set_local_variable = { name = st_res_$GOOD$_can_store_local value = 1 }
	}
	if = {
		limit = {
			local_var:st_res_$GOOD$_rate_local < 0
			var:st_res_$GOOD$_stored > 0
			var:st_res_hub_workforce_cached > 0.99
		}
		set_local_variable = { name = st_res_$GOOD$_can_withdraw_local value = 1 }
	}
}

# --------------------------------------------------------------------------
# st_res_rebuild_good_flow_modifiers_effect — Building-scoped per-good helper.
# Parameter: GOOD = grain | ammunition | oil.
# Expects the corresponding st_res_$GOOD$_*_local values to already be populated in
# the calling effect before the country -> building scope bridge.
# --------------------------------------------------------------------------
st_res_rebuild_good_flow_modifiers_effect = {
	if = { limit = { has_modifier = st_res_$GOOD$_store_flow } remove_modifier = st_res_$GOOD$_store_flow }
	if = { limit = { has_modifier = st_res_$GOOD$_withdraw_flow } remove_modifier = st_res_$GOOD$_withdraw_flow }

	if = {
		limit = {
			local_var:st_res_$GOOD$_can_store_local > 0
		}
		add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = -1 }
		add_modifier = { name = st_res_$GOOD$_store_flow multiplier = local_var:st_res_$GOOD$_good_mult_local }
	}
	else_if = {
		limit = {
			local_var:st_res_$GOOD$_can_withdraw_local > 0
		}
		add_modifier = { name = st_res_$GOOD$_store_flow multiplier = -1 }
		add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = local_var:st_res_$GOOD$_good_mult_local }
	}
	else = {
		add_modifier = { name = st_res_$GOOD$_store_flow multiplier = -1 }
		add_modifier = { name = st_res_$GOOD$_withdraw_flow multiplier = -1 }
	}
}

# --------------------------------------------------------------------------
# st_res_set_good_status_effect — Derive the display status code for one good.
# Parameter: GOOD.
#
# This is the ONLY site that decides what the JE inventory widget shows as a
# good's status: the widget and its tooltips read `st_res_$GOOD$_last_status`
# and never re-derive the reason from GUI expressions. It is called once, at
# the end of st_res_refresh_hub_flow_effect, which is itself the shared tail of
# both the weekly pulse and every rate/step button press — so one call site
# covers both "what happened this tick" and "what will happen now that you
# clicked", without the label lagging a week behind the player's input.
#
# Status codes (mirrored in st_res_custom_loc.txt and the widget loc keys):
#   0 = Idle              — nothing configured, nothing moving
#   1 = Storing           — configured intake is being applied in full
#   2 = Withdrawing       — configured release is being applied in full
#   3 = Storing, clipped by the hub's weekly flow cap
#   4 = Withdrawing, clipped by the hub's weekly flow cap
#   5 = Blocked, stockpile at capacity
#   6 = Blocked, stockpile empty while a release is configured
#   7 = Blocked, hub understaffed (occupancy below 100%)
#   8 = Blocked, no Strategic Reserve Hub
#
# Precedence note: st_res_clamp_stockpiles_effect drives the configured rate to
# 0 once a stockpile hits a bound, so the "full" branch deliberately accepts
# `rate >= 0` — otherwise a full reserve would report Idle the tick after the
# auto-clamp fires. The "empty" branch keeps `rate < 0` on purpose: an empty,
# unconfigured reserve is genuinely idle, and the row tooltip says it is empty.
# --------------------------------------------------------------------------
st_res_set_good_status_effect = {
	set_local_variable = { name = st_res_status_capacity value = st_res_$GOOD$_capacity }

	if = {
		limit = { var:st_res_hub_level_cached <= 0 }
		set_variable = { name = st_res_$GOOD$_last_status value = 8 }
	}
	else_if = {
		limit = { var:st_res_hub_workforce_cached <= 0.99 }
		set_variable = { name = st_res_$GOOD$_last_status value = 7 }
	}
	else_if = {
		limit = {
			local_var:st_res_status_capacity > 0
			var:st_res_$GOOD$_stored >= st_res_$GOOD$_capacity
			var:st_res_$GOOD$_rate >= 0
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 5 }
	}
	else_if = {
		limit = {
			var:st_res_$GOOD$_stored <= 0
			var:st_res_$GOOD$_rate < 0
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 6 }
	}
	else_if = {
		limit = {
			var:st_res_$GOOD$_rate > 0
			var:st_res_$GOOD$_actual_rate_cached > st_res_weekly_base_rate_cap
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 3 }
	}
	else_if = {
		limit = {
			var:st_res_$GOOD$_rate < 0
			var:st_res_$GOOD$_actual_rate_cached < st_res_neg_weekly_base_rate_cap
		}
		set_variable = { name = st_res_$GOOD$_last_status value = 4 }
	}
	else_if = {
		limit = { var:st_res_$GOOD$_rate > 0 }
		set_variable = { name = st_res_$GOOD$_last_status value = 1 }
	}
	else_if = {
		limit = { var:st_res_$GOOD$_rate < 0 }
		set_variable = { name = st_res_$GOOD$_last_status value = 2 }
	}
	else = {
		set_variable = { name = st_res_$GOOD$_last_status value = 0 }
	}
}

# --------------------------------------------------------------------------
# st_res_apply_weekly_good_effect — Apply one weekly tick for a single good and
# record what actually moved. Parameter: GOOD.
#
# The stockpile arithmetic is unchanged from the pre-widget implementation
# (add weekly_delta, clamp into [0, capacity]); the only addition is the
# before/after snapshot that writes `st_res_$GOOD$_last_delta`. That variable is
# the ACTUAL net weekly movement, i.e. after the capacity/empty clamp bites, so
# the number shown in the JE inventory widget always matches what bookkeeping
# really did rather than what the configured rate asked for.
# --------------------------------------------------------------------------
st_res_apply_weekly_good_effect = {
	set_local_variable = { name = st_res_weekly_prev value = var:st_res_$GOOD$_stored }

	change_variable = { name = st_res_$GOOD$_stored add = st_res_$GOOD$_weekly_delta }
	clamp_variable = { name = st_res_$GOOD$_stored min = 0 max = st_res_$GOOD$_capacity }

	set_local_variable = { name = st_res_weekly_moved value = var:st_res_$GOOD$_stored }
	change_local_variable = {
		name = st_res_weekly_moved
		add = {
			value = local_var:st_res_weekly_prev
			multiply = -1
		}
	}
	set_variable = { name = st_res_$GOOD$_last_delta value = local_var:st_res_weekly_moved }
}

# --- Per-good rate adjustment bases ----------------------------------------
# Each rate button calls the matching st_res_<dir>_<good>_rate_effect wrapper
# (kept as a named entry point — buttons reference effect names directly).
# All 14 wrappers share the same 4-line body modulo $GOOD$ and the +/- direction;
# each wrapper delegates to one of these two bases.
# Country scope. st_res_init_effect ensures the rate variable exists.

st_res_increase_rate_base = {
	st_res_init_effect = yes
	change_variable = { name = st_res_$GOOD$_rate add = st_res_adjust_step_value }
	st_res_refresh_hub_flow_effect = yes
	st_res_clamp_stockpiles_effect = yes
}

st_res_decrease_rate_base = {
	st_res_init_effect = yes
	change_variable = { name = st_res_$GOOD$_rate subtract = st_res_adjust_step_value }
	st_res_refresh_hub_flow_effect = yes
	st_res_clamp_stockpiles_effect = yes
}

# "Stop" — park one good's lane at 0/week without touching any other good.
# Same shape as the increase/decrease bases so the hub refresh and the stockpile
# clamp run exactly as they do for a stepped adjustment.
st_res_stop_rate_base = {
	st_res_init_effect = yes
	set_variable = { name = st_res_$GOOD$_rate value = 0 }
	st_res_refresh_hub_flow_effect = yes
	st_res_clamp_stockpiles_effect = yes
}
