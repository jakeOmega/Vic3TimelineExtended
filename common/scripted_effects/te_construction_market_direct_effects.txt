# Construction-market subsystem — direct construction, the model games run
# with free_market_construction_rule disabled (te_free_market_construction_off).
# See docs/systems/mod_systems.md, "Free Market Construction off".
#
# The rule's production-method flags do the work (common/game_rules/extra_game_rules.txt):
# the Construction Site is the buildable Construction Sector, its tiers turn
# goods into points, and nothing makes or uses the construction good. Script
# only has two jobs:
#   - convert the Construction Sectors the history places into Construction
#     Sites, once, when the game starts (te_direct_construction_convert_sectors);
#   - restore the law-set private share (te_direct_construction_on_weekly_pulse).

# State scope. Replace this state's Construction Sector with a Construction Site
# of the same level on the matching direct tier. A direct game places no sites
# of its own, so none is there already (if one were, create_building would keep
# the larger of the two levels, per the engine's effects.log). Levels above 100
# are capped at 100, the largest te_construction_market_build_specified_level
# can create.
te_direct_construction_convert_state_sector = {
	set_variable = {
		name = te_dc_convert_level
		value = {
			value = b:building_construction_sector.level
			max = 100
		}
	}
	set_variable = {
		name = te_dc_convert_tier
		value = 1
	}
	te_direct_construction_read_sector_tier = { TIER = 2 PM = pm_iron_frame_buildings }
	te_direct_construction_read_sector_tier = { TIER = 3 PM = pm_steel_frame_buildings }
	te_direct_construction_read_sector_tier = { TIER = 4 PM = pm_arc_welded_buildings }
	te_direct_construction_read_sector_tier = { TIER = 5 PM = pm_reinforced_concrete_buildings }
	te_direct_construction_read_sector_tier = { TIER = 6 PM = pm_advanced_construction }
	te_direct_construction_read_sector_tier = { TIER = 7 PM = pm_nanomaterial_buildings }

	remove_building = building_construction_sector
	if = {
		limit = {
			var:te_dc_convert_level >= 1
		}
		te_construction_market_build_specified_level = {
			SPEC_TYPE = te_construction_market_site
			SPEC_LEVEL = var:te_dc_convert_level
		}
	}

	te_direct_construction_activate_site_tier = { TIER = 1 PM = pm_te_direct_construction_wooden }
	te_direct_construction_activate_site_tier = { TIER = 2 PM = pm_te_direct_construction_iron_frame }
	te_direct_construction_activate_site_tier = { TIER = 3 PM = pm_te_direct_construction_steel_frame }
	te_direct_construction_activate_site_tier = { TIER = 4 PM = pm_te_direct_construction_arc_welded }
	te_direct_construction_activate_site_tier = { TIER = 5 PM = pm_te_direct_construction_reinforced_concrete }
	te_direct_construction_activate_site_tier = { TIER = 6 PM = pm_te_direct_construction_advanced }
	te_direct_construction_activate_site_tier = { TIER = 7 PM = pm_te_direct_construction_nanomaterial }

	remove_variable = te_dc_convert_level
	remove_variable = te_dc_convert_tier
}

# State scope. Helpers of te_direct_construction_convert_state_sector.
te_direct_construction_read_sector_tier = {
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_construction_sector
				has_active_production_method = $PM$
			}
		}
		set_variable = {
			name = te_dc_convert_tier
			value = $TIER$
		}
	}
}

# The site's default method (pm_te_construction_market_base) is disabled in
# direct games, so the engine starts a new site on one of the researched tiers;
# this moves it to the tier the sector had, tier 1 included.
te_direct_construction_activate_site_tier = {
	if = {
		limit = {
			var:te_dc_convert_tier = $TIER$
			any_scope_building = {
				is_building_type = te_construction_market_site
				NOT = { has_active_production_method = $PM$ }
			}
		}
		activate_production_method = {
			building_type = te_construction_market_site
			production_method = $PM$
		}
	}
}

# Global. Convert every Construction Sector in the world, once, then close the
# sector to new construction (building_construction_sector's can_build_* read
# te_direct_construction_converted). Runs from on_game_started_after_lobby, when
# the game rules are final: a rule read during history could still be changed
# in the lobby. No-op in market games.
te_direct_construction_convert_sectors = {
	if = {
		limit = {
			te_free_market_construction_off = yes
			NOT = { has_global_variable = te_direct_construction_converted }
		}
		every_country = {
			every_scope_state = {
				limit = {
					has_building = building_construction_sector
				}
				te_direct_construction_convert_state_sector = yes
			}
		}
		set_global_variable = {
			name = te_direct_construction_converted
			value = 1
		}
	}
}

# Country scope. The weekly pulse body for direct construction: the private
# share the economic-system law sets (te_direct_construction_law_private_share,
# through te_construction_market_private_share_target). Its one refresh site
# after the setup event (te_construction_market_country_setup).
te_direct_construction_on_weekly_pulse = {
	if = {
		limit = {
			NOT = { is_country_type = decentralized }
		}
		te_construction_market_country_update_private_share = yes
	}
}
