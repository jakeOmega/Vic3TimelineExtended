# Construction-market subsystem — direct construction, the model games run
# with free_market_construction_rule disabled (te_free_market_construction_off).
# See docs/systems/mod_systems/construction.md, "Free Market Construction off".
#
# The rule's production-method flags do the work (common/game_rules/extra_game_rules.txt):
# the Construction Site is the buildable Construction Sector, its tiers turn
# goods into points, and nothing makes or uses the construction good.
# Script has five jobs:
#   - give every country Base Construction, so one without sites can still
#     build them (te_direct_construction_ensure_base);
#   - convert the Construction Sectors into Construction Sites when the game
#     starts under the rule (te_direct_construction_convert_sectors), and in
#     either direction when the rule is changed in a running game
#     (te_construction_market_follow_rule, daily);
#   - restore the law-set private share, weekly
#     (te_direct_construction_on_weekly_pulse, as is the next one);
#   - put back the Construction Site levels an AI country removes
#     (te_direct_construction_keep_ai_sites);
#   - log two self-checks (te_direct_construction_base_self_check,
#     te_direct_construction_switch_self_check).

# State scope. This state's Construction Sector, and any Construction Site it
# has, become one Construction Site: te_direct_construction_convert_target_level
# levels (the sector's plus the site's own) on the matching direct tier. Both
# buildings are removed and the site is made new, so the result does not rest on
# what create_building does to a building that is already there, and a market
# site's capacity modifier (te_construction_market_capacity_modifier, sized for
# the market's weekly purchase and never refreshed in a direct game) goes with
# the site that carried it. The state's crew modifier goes too
# (te_construction_market_clear_crew): it cuts the group's jobs by 90%, and a
# direct game's site is a sector in that group. The site's workers are let go
# and hired back a tenth of its jobs a week, as any building hires (measured on
# market sites, 2026-11-13; the group's hiring rate of 1.0 has no effect).
#
# The tier is the sector's, except where a site stands that is real capacity
# (an established direct game's, or one with no sector beside it): then the
# site's own.
#
# A sector whose first level is still under construction is cancelled: nothing
# can use it, and a construction its can_build_government no longer allows
# stays in the queue unfinished (an AI's, queued on a new game's first day,
# stood at 100 points left five months later, 2026-11-13).
#
# The AI site guard's record of the state is set to what was made, so the guard
# holds from the conversion on and not from the next weekly pulse: AI countries
# removed 23 of 378 new sites in the week after a conversion (read live,
# 2026-11-13).
te_direct_construction_convert_state = {
	te_construction_market_clear_crew = yes
	if = {
		limit = {
			te_direct_construction_convert_uncapped_level > te_direct_construction_convert_target_level
		}
		te_direct_construction_count_capped = yes
	}
	set_variable = {
		name = te_dc_convert_level
		value = te_direct_construction_convert_target_level
	}
	set_variable = {
		name = te_dc_target_tier
		value = 1
	}
	if = {
		limit = {
			has_building = te_construction_market_site
			OR = {
				NOT = { has_global_variable = te_dc_from_market }
				NOT = { has_building = building_construction_sector }
			}
		}
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 2 PM = pm_te_direct_construction_iron_frame }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 3 PM = pm_te_direct_construction_steel_frame }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 4 PM = pm_te_direct_construction_arc_welded }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 5 PM = pm_te_direct_construction_reinforced_concrete }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 6 PM = pm_te_direct_construction_advanced }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 7 PM = pm_te_direct_construction_nanomaterial }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 8 PM = pm_te_direct_construction_self_replicating }
	}
	else = {
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 2 PM = pm_iron_frame_buildings }
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 3 PM = pm_steel_frame_buildings }
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 4 PM = pm_arc_welded_buildings }
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 5 PM = pm_reinforced_concrete_buildings }
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 6 PM = pm_advanced_construction }
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 7 PM = pm_nanomaterial_buildings }
		te_direct_construction_read_tier = { BUILDING = building_construction_sector TIER = 8 PM = pm_self_replicating_construction }
	}

	# has_building is false for a building whose first level is still being
	# built; remove_building cancels that construction (read live, 2026-11-13).
	# Both go, standing or not, so the site below is never made over one.
	if = {
		limit = {
			any_scope_building = {
				is_building_type = te_construction_market_site
			}
		}
		remove_building = te_construction_market_site
	}
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_construction_sector
			}
		}
		remove_building = building_construction_sector
	}
	if = {
		limit = {
			var:te_dc_convert_level >= 1
		}
		te_construction_market_build_specified_level = {
			SPEC_TYPE = te_construction_market_site
			SPEC_LEVEL = var:te_dc_convert_level
		}
	}

	te_direct_construction_activate_site_tier = { TIER = 1 PM = pm_te_direct_construction_wooden }
	te_direct_construction_activate_site_tier = { TIER = 2 PM = pm_te_direct_construction_iron_frame }
	te_direct_construction_activate_site_tier = { TIER = 3 PM = pm_te_direct_construction_steel_frame }
	te_direct_construction_activate_site_tier = { TIER = 4 PM = pm_te_direct_construction_arc_welded }
	te_direct_construction_activate_site_tier = { TIER = 5 PM = pm_te_direct_construction_reinforced_concrete }
	te_direct_construction_activate_site_tier = { TIER = 6 PM = pm_te_direct_construction_advanced }
	te_direct_construction_activate_site_tier = { TIER = 7 PM = pm_te_direct_construction_nanomaterial }
	te_direct_construction_activate_site_tier = { TIER = 8 PM = pm_te_direct_construction_self_replicating }

	if = {
		limit = {
			var:te_dc_convert_level >= 1
		}
		set_variable = {
			name = te_dc_site_level
			value = var:te_dc_convert_level
		}
		set_variable = {
			name = te_dc_site_tier
			value = var:te_dc_target_tier
		}
		te_direct_construction_count_made = yes
	}
	else = {
		remove_variable = te_dc_site_level
		remove_variable = te_dc_site_tier
	}
	remove_variable = te_dc_convert_level
	remove_variable = te_dc_target_tier
}

# State scope. The other way, when the market is switched back on in a game
# that ran direct construction: this state's Construction Site (with a sector
# that finished after the conversion, if one stands beside it) becomes a
# Construction Sector of the same level on the same tier, owned by the
# government as the history's sectors are. The site is removed: the market
# places its own, at level 1, where construction is under way
# (te_construction_market_ensure_site and the building hooks).
te_direct_construction_revert_state_site = {
	if = {
		limit = {
			te_direct_construction_revert_uncapped_level > te_direct_construction_revert_target_level
		}
		te_direct_construction_count_capped = yes
	}
	set_variable = {
		name = te_dc_convert_level
		value = te_direct_construction_revert_target_level
	}
	set_variable = {
		name = te_dc_target_tier
		value = 1
	}
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 2 PM = pm_te_direct_construction_iron_frame }
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 3 PM = pm_te_direct_construction_steel_frame }
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 4 PM = pm_te_direct_construction_arc_welded }
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 5 PM = pm_te_direct_construction_reinforced_concrete }
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 6 PM = pm_te_direct_construction_advanced }
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 7 PM = pm_te_direct_construction_nanomaterial }
	te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 8 PM = pm_te_direct_construction_self_replicating }

	remove_building = te_construction_market_site
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_construction_sector
			}
		}
		remove_building = building_construction_sector
	}
	if = {
		limit = {
			var:te_dc_convert_level >= 1
		}
		extended_timeline_build_specified_building_level = {
			SPEC_TYPE = building_construction_sector
			SPEC_LEVEL = var:te_dc_convert_level
			SPEC_OWNER = owner
		}
		te_direct_construction_count_made = yes
	}

	te_direct_construction_activate_sector_tier = { TIER = 1 PM = pm_wooden_buildings }
	te_direct_construction_activate_sector_tier = { TIER = 2 PM = pm_iron_frame_buildings }
	te_direct_construction_activate_sector_tier = { TIER = 3 PM = pm_steel_frame_buildings }
	te_direct_construction_activate_sector_tier = { TIER = 4 PM = pm_arc_welded_buildings }
	te_direct_construction_activate_sector_tier = { TIER = 5 PM = pm_reinforced_concrete_buildings }
	te_direct_construction_activate_sector_tier = { TIER = 6 PM = pm_advanced_construction }
	te_direct_construction_activate_sector_tier = { TIER = 7 PM = pm_nanomaterial_buildings }
	te_direct_construction_activate_sector_tier = { TIER = 8 PM = pm_self_replicating_construction }

	remove_variable = te_dc_site_level
	remove_variable = te_dc_site_tier
	remove_variable = te_dc_convert_level
	remove_variable = te_dc_target_tier
}

# State scope. Tier helpers of the sector conversion and of the AI site guard,
# which pass the tier through the state variable te_dc_target_tier (1-8, the
# order of pmg_base_te_construction_market_site's direct tiers).
#
# Set te_dc_target_tier to TIER if this state's BUILDING runs PM.
te_direct_construction_read_tier = {
	if = {
		limit = {
			any_scope_building = {
				is_building_type = $BUILDING$
				has_active_production_method = $PM$
			}
		}
		set_variable = {
			name = te_dc_target_tier
			value = $TIER$
		}
	}
}

# The site's default method (pm_te_construction_market_base) is disabled in
# direct games, so the engine starts a new site on one of the researched tiers;
# this moves it to tier te_dc_target_tier (the sector's, or the removed site's),
# tier 1 included.
te_direct_construction_activate_site_tier = {
	if = {
		limit = {
			var:te_dc_target_tier = $TIER$
			any_scope_building = {
				is_building_type = te_construction_market_site
				NOT = { has_active_production_method = $PM$ }
			}
		}
		activate_production_method = {
			building_type = te_construction_market_site
			production_method = $PM$
		}
	}
}

# The same for a Construction Sector the market gets back
# (te_direct_construction_revert_state_site).
te_direct_construction_activate_sector_tier = {
	if = {
		limit = {
			var:te_dc_target_tier = $TIER$
			any_scope_building = {
				is_building_type = building_construction_sector
				NOT = { has_active_production_method = $PM$ }
			}
		}
		activate_production_method = {
			building_type = building_construction_sector
			production_method = $PM$
		}
	}
}

# Any scope. One more building a switch should have made, counted while
# te_construction_market_follow_rule has the counter up for the switch's
# self-check (te_direct_construction_switch_self_check compares what stands
# with it). The game-start conversion and the weekly pass count nothing.
te_direct_construction_count_made = {
	if = {
		limit = {
			has_global_variable = te_dc_switch_expected
		}
		change_global_variable = {
			name = te_dc_switch_expected
			add = 1
		}
	}
}

# Any scope. One more state whose levels a switch cut to what the builder can
# make (te_direct_construction_convert_target_level, 200; the way back, 100),
# counted like te_direct_construction_count_made for the self-check's line.
te_direct_construction_count_capped = {
	if = {
		limit = {
			has_global_variable = te_dc_switch_capped
		}
		change_global_variable = {
			name = te_dc_switch_capped
			add = 1
		}
	}
}

# State scope. Forget the AI site guard's record of this state. Both switches
# do it for every state first: a record the market period left behind (a site
# its AI owner removed in the last week, a site kept because its owner has no
# Urbanization) would have the guard build a site from nothing at the next
# switch to direct construction.
te_direct_construction_forget_site_record = {
	if = {
		limit = {
			has_variable = te_dc_site_level
		}
		remove_variable = te_dc_site_level
	}
	if = {
		limit = {
			has_variable = te_dc_site_tier
		}
		remove_variable = te_dc_site_tier
	}
}

# Country scope. Base Construction (te_direct_construction_base_modifier, 10
# points a week) for a country in a direct game: added at the conversion and
# from the weekly pulse, which is what reaches a country made during play and a
# revolution's winner (it inherits no modifiers). Never stacked: only a country
# without it gets it. Not added at country setup, which runs from history, when
# the lobby can still change the rule. te_direct_construction_remove_base takes
# it away: when the market is switched back on, and from the market's weekly
# pulse for any country that still holds it.
#
# Script owns it because the rule cannot: an apply_modifier line in a rule
# setting is not applied by the save's rules (read live, 2026-11-13: a save whose
# rule said Enabled got the Disabled setting's modifier, and one switched to no
# culture merges got the merges setting's); it seems to follow the rules the
# player last applied in a lobby (inferred, #1767).
te_direct_construction_ensure_base = {
	if = {
		limit = {
			NOT = { is_country_type = decentralized }
			NOT = { has_modifier = te_direct_construction_base_modifier }
		}
		add_modifier = {
			name = te_direct_construction_base_modifier
		}
	}
}

# Country scope. A market game has no Base Construction.
te_direct_construction_remove_base = {
	if = {
		limit = {
			has_modifier = te_direct_construction_base_modifier
		}
		remove_modifier = te_direct_construction_base_modifier
	}
}

# Country scope. Construction Cost Scaling rides a different modifier in each
# model (te_cc_apply_scaling, te_construction_cost_effects.txt), and each
# branch backs its own last contribution out of var:prior_construction_scaling.
# When the model changes, both modifiers and the variable go, so the new
# model's branch starts from its first run. A switch in a running game then
# applies it at once (te_direct_construction_switch_cost_scaling); the
# conversion at game start leaves it to the country's first yearly pulse, as
# any new game does.
te_direct_construction_clear_cost_scaling = {
	remove_modifier = construction_cost_scaling
	remove_modifier = construction_cost_scaling_direct
	remove_variable = prior_construction_scaling
}

# Any scope. After a switch in a running game: every country gets the new
# model's scaling now, not at its next yearly pulse. The market's price and its
# goods per point multiply by the live construction_cost_scaling_mult whether or
# not the modifier is in force, so in the months between a rich country's Fixed
# Budget would have bought fewer points than it paid for (a second session's
# review of the PR, checked against te_construction_market_price and
# te_construction_market_goods_per_point). A direct game would have built
# without the scaling for as long.
#
# Through a hidden country event (te_construction_market_recalc_events.4), not
# called here: this runs in one country's pulse event, and add_modifier's
# multiplier resolves against ROOT, so a direct call would give every country
# that one country's scaling until its own yearly pulse
# (docs/guides/scripting_best_practices.md, "add_modifier { multiplier = var:X }
# Resolves Against ROOT").
te_direct_construction_switch_cost_scaling = {
	every_country = {
		# As the weekly and game-start passes: a decentralized country has no
		# events and no economy to scale.
		limit = {
			NOT = { is_country_type = decentralized }
		}
		trigger_event = { id = te_construction_market_recalc_events.4 }
	}
}

# Any scope. Every Construction Sector in the world becomes a Construction Site,
# and the market's own sites are retired with it; then the sector is closed to
# new construction (building_construction_sector's can_build_* read
# te_direct_construction_converted). te_dc_from_market is up while it runs, so
# a market site's own level, which is a converter and not capacity, is not
# carried over (te_direct_construction_convert_target_level). In a state with a
# site and no sector, only what its owner built on top of that level is kept.
te_direct_construction_convert_all = {
	set_global_variable = {
		name = te_dc_from_market
		value = 1
	}
	every_country = {
		every_scope_state = {
			te_direct_construction_forget_site_record = yes
		}
		# Every state, not only those converted below: a state whose market
		# site was downsized away still carries its crew's cut, which would
		# run a sector built there later on a tenth of its workers.
		every_scope_state = {
			te_construction_market_clear_crew = yes
		}
		every_scope_state = {
			limit = {
				any_scope_building = {
					OR = {
						is_building_type = te_construction_market_site
						is_building_type = building_construction_sector
					}
				}
			}
			te_direct_construction_convert_state = yes
		}
		te_direct_construction_clear_cost_scaling = yes
		te_direct_construction_ensure_base = yes
	}
	remove_global_variable = te_dc_from_market
	set_global_variable = {
		name = te_direct_construction_converted
		value = 1
	}
	# The base's self-check is owed again (te_direct_construction_base_self_check).
	if = {
		limit = {
			has_global_variable = te_dc_base_checked
		}
		remove_global_variable = te_dc_base_checked
	}
	if = {
		limit = {
			has_global_variable = te_dc_base_check_passes
		}
		remove_global_variable = te_dc_base_check_passes
	}
}

# Any scope. The way back: every Construction Site becomes a Construction
# Sector, the sector is open to construction again, Base Construction goes, and
# every country has the purchase variables the market reads. No site is placed
# here: each country's weekly pulse ends by placing one in its capital, after
# it has sized the sites (te_construction_market_ensure_site), and the building
# hooks place the rest.
# A site whose owner has no Urbanization is not converted (the sector needs the
# technology, and create_building checks it): it serves as a market site, at
# level 1 as the market's sites are.
te_direct_construction_revert_all = {
	every_country = {
		every_scope_state = {
			limit = {
				has_building = te_construction_market_site
				owner = {
					has_technology_researched = urbanization
				}
			}
			te_direct_construction_revert_state_site = yes
		}
		# A site whose first level is still being built: the market allows
		# nobody to build one, so the construction would never finish.
		every_scope_state = {
			limit = {
				NOT = { has_building = te_construction_market_site }
				any_scope_building = {
					is_building_type = te_construction_market_site
				}
			}
			remove_building = te_construction_market_site
		}
		# A site kept because its owner has no Urbanization serves as a market
		# site, and is made new for it, whatever its level: the market's sites
		# are level 1, each sized for a share of the week's purchase
		# (te_construction_market_site_throughput), and a kept site would hold
		# its direct tier's workforce on the market method's ten jobs, at the
		# engine's own throughput, until the first weekly pass saw it. Until
		# 2026-11-14 only a site above level 1 was made new.
		every_scope_state = {
			limit = {
				has_building = te_construction_market_site
			}
			remove_building = te_construction_market_site
			te_construction_market_place_site = yes
		}
		every_scope_state = {
			te_direct_construction_forget_site_record = yes
		}
		te_direct_construction_clear_cost_scaling = yes
		te_construction_market_init_target = yes
		te_direct_construction_remove_base = yes
	}
	remove_global_variable = te_direct_construction_converted
}

# Any scope. The conversion at game start: from on_game_started_after_lobby,
# when the game rules are final (a rule read during history could still be
# changed in the lobby). A no-op in market games, and once the conversion has
# run.
te_direct_construction_convert_sectors = {
	if = {
		limit = {
			te_free_market_construction_off = yes
			NOT = { has_global_variable = te_direct_construction_converted }
		}
		te_direct_construction_convert_all = yes
	}
}

# Any scope, daily (te_construction_market_pulse_events.1), before the weekly
# bodies. The rule can be changed in a running game: the screen that changes
# country, which is also what a save with no player opens on, has the Game Rules
# button. Until 2026-11-13 the conversion ran only when a game started, so a
# market game switched to Disabled kept its sectors and its market sites, each
# site still carrying the capacity modifier of the market's last week. In a
# three-month-old game switched that way France's sites stood at -100%
# throughput, which left the country the 5 points of Base Construction and
# nothing from any site level it then built (read live, 2026-11-13;
# docs/testing/direct-construction-playtest.md).
#
# te_direct_construction_converted says which model the buildings are in. When
# it and the rule disagree, the buildings follow the rule:
#   - rule off, not converted: te_direct_construction_convert_all. A game that
#     started under the rule was converted at its start, so this is a rule
#     changed since, or a save from before this effect.
#   - rule on, converted: te_direct_construction_revert_all.
# Every fourth week, in a converted game, any Construction Sector that has
# appeared is taken into its state's site, and one still under construction is
# cancelled (an AI queued one on a new game's first day, before the conversion
# closed the sector, 2026-11-13).
te_construction_market_follow_rule = {
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_direct_construction_converted }
			}
			set_global_variable = {
				name = te_dc_switch_expected
				value = 0
			}
			set_global_variable = {
				name = te_dc_switch_capped
				value = 0
			}
			te_direct_construction_convert_all = yes
			te_direct_construction_switch_cost_scaling = yes
			te_direct_construction_switch_self_check = yes
		}
		else_if = {
			limit = {
				te_construction_market_pulse_count_modulo_7 = 0
			}
			# Every fourth week: the walk reads every building in the world,
			# and nothing can queue a sector once the conversion has closed it.
			if = {
				limit = {
					te_construction_market_pulse_count_modulo_28 = 0
				}
				every_country = {
					every_scope_state = {
						limit = {
							any_scope_building = {
								is_building_type = building_construction_sector
							}
						}
						# A finished sector joins the state's site, which is
						# made new at the sum. One still under construction is
						# only cancelled, so a staffed site keeps its workers.
						if = {
							limit = {
								has_building = building_construction_sector
							}
							te_direct_construction_convert_state = yes
						}
						else = {
							remove_building = building_construction_sector
						}
					}
				}
			}
			te_direct_construction_base_self_check = yes
		}
	}
	else_if = {
		limit = {
			has_global_variable = te_direct_construction_converted
		}
		set_global_variable = {
			name = te_dc_switch_expected
			value = 0
		}
		set_global_variable = {
			name = te_dc_switch_capped
			value = 0
		}
		te_direct_construction_revert_all = yes
		te_direct_construction_switch_cost_scaling = yes
		te_direct_construction_switch_self_check = yes
	}
}

# Any scope, in a country's event, weekly in a direct game. Once for each time
# the game becomes one, a week after its first weekly pass, so every country's
# own pulse has had its turn: does every country hold Base Construction, and
# does the engine give its holders the 10 points? Countries are counted with a
# capital and not decentralized. A country made in the last week has not had its
# pulse yet, so the missing ones break the verdict only when they are more than
# a tenth of the holders.
te_direct_construction_base_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = te_dc_base_checked }
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_dc_base_check_passes }
			}
			set_global_variable = {
				name = te_dc_base_check_passes
				value = 0
			}
		}
		change_global_variable = {
			name = te_dc_base_check_passes
			add = 1
		}
		if = {
			limit = {
				global_var:te_dc_base_check_passes >= 2
			}
			set_global_variable = {
				name = te_dc_base_checked
				value = 1
			}
			remove_global_variable = te_dc_base_check_passes
			set_global_variable = {
				name = te_dc_check_countries
				value = 0
			}
			set_global_variable = {
				name = te_dc_check_under_base
				value = 0
			}
			set_global_variable = {
				name = te_dc_check_missing
				value = 0
			}
			every_country = {
				limit = {
					NOT = { is_country_type = decentralized }
					exists = capital
				}
				if = {
					limit = {
						has_modifier = te_direct_construction_base_modifier
					}
					change_global_variable = {
						name = te_dc_check_countries
						add = 1
					}
					if = {
						limit = {
							modifier:country_construction_add < 10
						}
						change_global_variable = {
							name = te_dc_check_under_base
							add = 1
						}
					}
				}
				else = {
					change_global_variable = {
						name = te_dc_check_missing
						add = 1
					}
				}
			}
			if = {
				limit = {
					global_var:te_dc_check_countries >= 1
					global_var:te_dc_check_under_base = 0
					te_dc_check_missing_tenfold <= global_var:te_dc_check_countries
				}
				debug_log = "TE_SELF_CHECK OK direct_base_construction holders=[THIS.ScriptValue('te_dc_check_countries')|0] under_base=[THIS.ScriptValue('te_dc_check_under_base')|0] missing=[THIS.ScriptValue('te_dc_check_missing')|0] (every country that holds Base Construction has its 10 points a week, and at most a few new countries are still without it) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN direct_base_construction holders=[THIS.ScriptValue('te_dc_check_countries')|0] under_base=[THIS.ScriptValue('te_dc_check_under_base')|0] missing=[THIS.ScriptValue('te_dc_check_missing')|0] (under_base: holders of Base Construction with under 10 construction a week, so the engine does not sum the modifier; missing: countries a week of pulses left without it, so te_direct_construction_ensure_base did not reach them; docs/testing/direct-construction-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			remove_global_variable = te_dc_check_countries
			remove_global_variable = te_dc_check_under_base
			remove_global_variable = te_dc_check_missing
		}
	}
}

# Any scope, in a country's event. After the rule was changed in a running game
# and the buildings followed it: one line a switch, with what stands now
# against what the switch should have made (te_dc_switch_expected, counted by
# te_direct_construction_count_made). To direct construction it is BROKEN when
# fewer sites stand than expected, a Construction Sector is left, standing or
# under construction, or a site still carries the market's capacity modifier or
# a state its crew modifier (both counted as stale);
# to the market, when fewer sectors stand than expected, or none. capped is the
# number of states whose levels the switch cut to what the builder can make: it
# does not change the verdict, and tells what the switch cost.
te_direct_construction_switch_self_check = {
	set_global_variable = {
		name = te_dc_check_sites
		value = 0
	}
	set_global_variable = {
		name = te_dc_check_sectors
		value = 0
	}
	set_global_variable = {
		name = te_dc_check_stale
		value = 0
	}
	every_country = {
		every_scope_state = {
			if = {
				# The variable, not has_modifier: the switch has just taken the
				# modifier off in this same effect, where has_modifier still
				# sees it (te_construction_market_remove_crew_cut).
				limit = {
					te_free_market_construction_off = yes
					has_variable = te_cm_site_crew_cut
				}
				change_global_variable = {
					name = te_dc_check_stale
					add = 1
				}
			}
			if = {
				limit = {
					has_building = te_construction_market_site
				}
				change_global_variable = {
					name = te_dc_check_sites
					add = 1
				}
				if = {
					limit = {
						any_scope_building = {
							is_building_type = te_construction_market_site
							has_modifier = te_construction_market_capacity_modifier
						}
					}
					change_global_variable = {
						name = te_dc_check_stale
						add = 1
					}
				}
			}
			if = {
				limit = {
					any_scope_building = {
						is_building_type = building_construction_sector
					}
				}
				change_global_variable = {
					name = te_dc_check_sectors
					add = 1
				}
			}
		}
	}
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		if = {
			limit = {
				global_var:te_dc_check_sectors = 0
				global_var:te_dc_check_stale = 0
				global_var:te_dc_check_sites >= te_dc_check_expected
			}
			debug_log = "TE_SELF_CHECK OK construction_rule_switch to=direct sites=[THIS.ScriptValue('te_dc_check_sites')|0] expected=[THIS.ScriptValue('te_dc_check_expected')|0] capped=[THIS.ScriptValue('te_dc_check_capped')|0] sectors_left=[THIS.ScriptValue('te_dc_check_sectors')|0] market_modifiers_left=[THIS.ScriptValue('te_dc_check_stale')|0] (the rule was switched to Disabled in a running game: every Construction Sector became a Construction Site and the market's sites were retired) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN construction_rule_switch to=direct sites=[THIS.ScriptValue('te_dc_check_sites')|0] expected=[THIS.ScriptValue('te_dc_check_expected')|0] capped=[THIS.ScriptValue('te_dc_check_capped')|0] sectors_left=[THIS.ScriptValue('te_dc_check_sectors')|0] market_modifiers_left=[THIS.ScriptValue('te_dc_check_stale')|0] (sites under expected: create_building refused a site, so a sector's levels are lost; sectors_left or market_modifiers_left above 0: te_direct_construction_convert_state did not replace the buildings; docs/testing/direct-construction-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
	else = {
		if = {
			limit = {
				global_var:te_dc_check_sectors >= 1
				global_var:te_dc_check_sectors >= te_dc_check_expected
			}
			debug_log = "TE_SELF_CHECK OK construction_rule_switch to=market sectors=[THIS.ScriptValue('te_dc_check_sectors')|0] expected=[THIS.ScriptValue('te_dc_check_expected')|0] capped=[THIS.ScriptValue('te_dc_check_capped')|0] sites_left=[THIS.ScriptValue('te_dc_check_sites')|0] (the rule was switched to a market setting in a running game: the Construction Sites became Construction Sectors; a site left belongs to a country without Urbanization) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN construction_rule_switch to=market sectors=[THIS.ScriptValue('te_dc_check_sectors')|0] expected=[THIS.ScriptValue('te_dc_check_expected')|0] capped=[THIS.ScriptValue('te_dc_check_capped')|0] sites_left=[THIS.ScriptValue('te_dc_check_sites')|0] (fewer Construction Sectors stand than the switch to a market setting should have made: te_direct_construction_revert_state_site removed a site and its sector was refused, so that construction is lost; docs/testing/direct-construction-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
	remove_global_variable = te_dc_check_sites
	remove_global_variable = te_dc_check_sectors
	remove_global_variable = te_dc_check_stale
	if = {
		limit = {
			has_global_variable = te_dc_switch_expected
		}
		remove_global_variable = te_dc_switch_expected
	}
	if = {
		limit = {
			has_global_variable = te_dc_switch_capped
		}
		remove_global_variable = te_dc_switch_capped
	}
}

# Country scope. Direct-construction setup (te_construction_market_country_setup):
# the law-set private share from day one. The site guard starts at the first
# weekly pulse, its one refresh site. Base Construction is not added here: this
# runs from history, where a rule read is not final, and a game whose lobby then
# picked a market setting would keep the modifier. The conversion after the
# lobby adds it (te_direct_construction_convert_all).
te_direct_construction_country_setup = {
	if = {
		limit = {
			NOT = { is_country_type = decentralized }
		}
		te_construction_market_country_update_private_share = yes
	}
}

# Country scope. The weekly pulse body for direct construction:
#   - the private share the economic-system law sets
#     (te_direct_construction_law_private_share, through
#     te_construction_market_private_share_target); its one refresh site after
#     setup;
#   - Base Construction, for a country that has none
#     (te_direct_construction_ensure_base);
#   - the AI site guard (te_direct_construction_keep_ai_sites).
te_direct_construction_on_weekly_pulse = {
	if = {
		limit = {
			NOT = { is_country_type = decentralized }
		}
		te_construction_market_country_update_private_share = yes
		te_direct_construction_ensure_base = yes
		te_direct_construction_keep_ai_sites = yes
	}
}

# Country scope, weekly. AI countries keep their Construction Sites.
#
# In play-tests AI countries lost every Construction Site they had. No script
# removes a site in a direct game (the market's placement and removal are gated
# on the rule), so this is the AI downsizing them, which script cannot steer.
# The site has to stay downsizeable for players (a site's wages are paid whether
# or not its points are used), so the guard acts after the fact: each state
# remembers its site's level and tier (te_dc_site_level, te_dc_site_tier;
# recorded for every country, so a player's own downsizing is never undone after
# the country passes to the AI), and an AI country gets back any levels it has
# lost since the last pulse. The AI can still build more levels and sites, and
# still picks each site's tier.
#
# Not for an AI country in default: it may shed sites to cut costs, and the
# record follows it down. Not without Urbanization, which the site's `possible`
# requires in direct games. If an AI keeps shedding levels, each restored level
# rehires within a week (the site's hiring rate is 1.0).
te_direct_construction_keep_ai_sites = {
	if = {
		limit = {
			is_ai = yes
			NOT = { in_default = yes }
			has_technology_researched = urbanization
		}
		every_scope_state = {
			te_direct_construction_restore_state_site = yes
		}
	}
	every_scope_state = {
		te_direct_construction_record_state_site = yes
	}
}

# State scope. Put the site back to its recorded level (capped at 200, the
# largest te_construction_market_build_specified_level can create) if it has
# lost levels.
#
# A site that is still there is raised with create_building, which keeps the
# workers it has: the engine's effects.log says that on an existing building
# create_building sets the larger of the two levels. scripting_best_practices.md
# records that reading as disputed (it may add the levels instead), so if the
# site comes out above the target, it is rebuilt at exactly the target, as a
# removed site is. A rebuilt site starts on a tier the engine picks, so it is
# moved back to the recorded tier.
te_direct_construction_restore_state_site = {
	if = {
		limit = {
			has_variable = te_dc_site_level
			var:te_dc_site_level > te_direct_construction_state_site_level
		}
		set_variable = {
			name = te_dc_restore_level
			value = {
				value = var:te_dc_site_level
				max = 200
			}
		}
		if = {
			limit = {
				has_building = te_construction_market_site
			}
			te_construction_market_build_specified_level = {
				SPEC_TYPE = te_construction_market_site
				SPEC_LEVEL = var:te_dc_restore_level
			}
			if = {
				limit = {
					b:te_construction_market_site.level > var:te_dc_restore_level
				}
				remove_building = te_construction_market_site
				te_construction_market_build_specified_level = {
					SPEC_TYPE = te_construction_market_site
					SPEC_LEVEL = var:te_dc_restore_level
				}
				te_direct_construction_restore_site_tier = yes
			}
		}
		else = {
			te_construction_market_build_specified_level = {
				SPEC_TYPE = te_construction_market_site
				SPEC_LEVEL = var:te_dc_restore_level
			}
			te_direct_construction_restore_site_tier = yes
		}
		remove_variable = te_dc_restore_level
	}
}

# State scope. Move a rebuilt site to its recorded tier.
te_direct_construction_restore_site_tier = {
	if = {
		limit = {
			has_variable = te_dc_site_tier
		}
		set_variable = {
			name = te_dc_target_tier
			value = var:te_dc_site_tier
		}
		te_direct_construction_activate_site_tier = { TIER = 1 PM = pm_te_direct_construction_wooden }
		te_direct_construction_activate_site_tier = { TIER = 2 PM = pm_te_direct_construction_iron_frame }
		te_direct_construction_activate_site_tier = { TIER = 3 PM = pm_te_direct_construction_steel_frame }
		te_direct_construction_activate_site_tier = { TIER = 4 PM = pm_te_direct_construction_arc_welded }
		te_direct_construction_activate_site_tier = { TIER = 5 PM = pm_te_direct_construction_reinforced_concrete }
		te_direct_construction_activate_site_tier = { TIER = 6 PM = pm_te_direct_construction_advanced }
		te_direct_construction_activate_site_tier = { TIER = 7 PM = pm_te_direct_construction_nanomaterial }
		te_direct_construction_activate_site_tier = { TIER = 8 PM = pm_te_direct_construction_self_replicating }
		remove_variable = te_dc_target_tier
	}
}

# State scope. Record this state's site level and tier for the guard; clear the
# record when the state has no site.
te_direct_construction_record_state_site = {
	if = {
		limit = {
			te_direct_construction_state_site_level > 0
		}
		set_variable = {
			name = te_dc_site_level
			value = te_direct_construction_state_site_level
		}
		set_variable = {
			name = te_dc_target_tier
			value = 1
		}
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 2 PM = pm_te_direct_construction_iron_frame }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 3 PM = pm_te_direct_construction_steel_frame }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 4 PM = pm_te_direct_construction_arc_welded }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 5 PM = pm_te_direct_construction_reinforced_concrete }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 6 PM = pm_te_direct_construction_advanced }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 7 PM = pm_te_direct_construction_nanomaterial }
		te_direct_construction_read_tier = { BUILDING = te_construction_market_site TIER = 8 PM = pm_te_direct_construction_self_replicating }
		set_variable = {
			name = te_dc_site_tier
			value = var:te_dc_target_tier
		}
		remove_variable = te_dc_target_tier
	}
	else = {
		remove_variable = te_dc_site_level
		remove_variable = te_dc_site_tier
	}
}
