# ============================================================================
# SOLAR SYSTEM COLONIZATION — fleets: space forces personnel
# ============================================================================
# Design: docs/superpowers/specs/2026-11-24-space-forces-personnel-design.md
# (scope: 2026-11-24-space-forces-personnel-scope.md). Full Colonies only.
#
# Fleets and marines draw on Space Forces Personnel, which Space Forces Bases
# grant (building_space_forces_base, +1,000 a level, workforce-scaled): a point
# of fleet strength at full needs 100 (50 with Artificial Intelligence), a
# marine battalion 1,000, aboard or in a garrison (sr_pers_in_service,
# sr_fleet_values.txt). Hulls leave the ground reserve and marines come aboard
# only as far as the personnel allow (sr_fl_reserve_in_at, generated;
# sr_fl_marines_refill_at, sr_fleet_effects.txt), in the order the toggle
# sets (sr_fl_country_month). A country whose personnel fall short later
# fights at the share it can crew (sr_pers_share, stored here at the end of
# the month; sr_fleet_readiness and the marines' weights read it).
#
# Ships destroyed, marines lost and a seizure's toll kill as many of the
# bases' Spacers (common/pop_types/te_spacers.txt): each loss site adds its
# people to sr_pers_dead_due (sr_pers_note_lost, sr_pers_note_people,
# sr_site_note_garrison_dead), and the month's close kills them in the bases'
# states (sr_pers_kill_step), booked as the census's war dead.
# ============================================================================

# ---- The losses (design § 5.1) ------------------------------------------------------------------
# Country scope, the loser. local_var:sr_pers_lost people at full crewing, set by
# the caller: at the country's crewed share they are the people aboard (an
# uncrewed hull has nobody on it), then noted.
sr_pers_note_lost = {
	if = {
		limit = { local_var:sr_pers_lost > 0 }
		set_local_variable = {
			name = sr_pers_lost
			value = {
				value = local_var:sr_pers_lost
				multiply = sr_pers_share_value
			}
		}
		sr_pers_note_people = yes
	}
}

# Country scope, the loser. local_var:sr_pers_lost people, already at the share:
# owed to the battle's kill. Also added to local_var:sr_pers_noted where a caller
# has set it, so the battle report gives what was booked (sr_bt_cut_slot,
# sr_bt_record).
sr_pers_note_people = {
	if = {
		limit = { local_var:sr_pers_lost > 0 }
		if = {
			limit = { NOT = { has_variable = sr_pers_dead_due } }
			set_variable = { name = sr_pers_dead_due value = 0 }
		}
		change_variable = { name = sr_pers_dead_due add = local_var:sr_pers_lost }
		if = {
			limit = { has_local_variable = sr_pers_noted }
			change_local_variable = { name = sr_pers_noted add = local_var:sr_pers_lost }
		}
	}
}

# Any scope, the world KEY as its garrison falls in an occupation or a
# liberation (before the record is overwritten): the battalions that defended
# it (sr_site_shared_marines_<key>: on the Moon and Mars, its holder's there
# too, all of which the fall removes) die, and their owner keeps the figure for
# its notice (.102, sr_occ_dead).
sr_site_note_garrison_dead = {
	sr_site_note_marines_dead = { KEY = $KEY$ MARINES = sr_site_shared_marines_$KEY$ NOTICE = yes }
}

# Any scope, the world KEY destroyed, before its release: only its own
# battalions die (sr_site_marines_value_<key>). On the Moon and Mars the
# holder's marines at its other colonies there stay in service, since the
# release removes this world's alone (review of #2020). No .102 follows, so
# no notice figure (the destruction has its own, .103).
sr_site_note_destroyed_dead = {
	sr_site_note_marines_dead = { KEY = $KEY$ MARINES = sr_site_marines_value_$KEY$ NOTICE = no }
}

# The two above: MARINES battalions of the world KEY die, at their owner's
# crewed share (sr_site_marines_share_<key>), and their owner (the occupier
# while occupied, else the holder) owes them; with NOTICE = yes it keeps the
# figure for .102.
sr_site_note_marines_dead = {
	set_local_variable = {
		name = sr_pers_lost
		value = {
			value = $MARINES$
			multiply = sr_site_marines_share_$KEY$
			multiply = sr_pers_per_battalion
		}
	}
	if = {
		limit = { local_var:sr_pers_lost > 0 }
		if = {
			limit = { exists = global_var:sr_site_occupier_$KEY$ }
			global_var:sr_site_occupier_$KEY$ = {
				sr_pers_note_people = yes
				if = {
					limit = { always = $NOTICE$ }
					sr_pers_note_occ_dead = yes
				}
			}
		}
		else_if = {
			limit = { exists = global_var:sr_site_holder_$KEY$ }
			global_var:sr_site_holder_$KEY$ = {
				sr_pers_note_people = yes
				if = {
					limit = { always = $NOTICE$ }
					sr_pers_note_occ_dead = yes
				}
			}
		}
	}
}

# Country scope: the people of local_var:sr_pers_lost for the occupation notice
# (.102), for a month; a second loss that month adds to it.
sr_pers_note_occ_dead = {
	set_local_variable = { name = sr_pers_occ_was value = 0 }
	if = {
		limit = { has_variable = sr_occ_dead }
		set_local_variable = { name = sr_pers_occ_was value = var:sr_occ_dead }
	}
	set_variable = {
		name = sr_occ_dead
		value = {
			value = local_var:sr_pers_occ_was
			add = local_var:sr_pers_lost
		}
		days = 30
	}
}

# ---- The month's close (design §§ 4, 5.3) --------------------------------------------------------
# Country scope, last in the fleets' month (sr_fl_country_month): the crewed
# share combat reads until the next month (sr_pers_share), then the dead paid,
# then the self-checks.
sr_pers_close = {
	set_variable = { name = sr_pers_share value = sr_pers_share_calc }
	sr_pers_kill_step = yes
	sr_pers_pop_type_self_check = yes
}

# Country scope. The dead owed (sr_pers_dead_due, people) killed in whole
# hundreds, kill_population_in_state taking a constant (Internal
# Resettlement's pattern): each state with Space Forces Personnel takes its
# share of the hundreds by its share of the capacity, at most its own Spacers,
# and books them as war dead (te_demog_note_war_dead). The fraction under 100
# is carried; hundreds past the month's cap (sr_pers_kill_max_steps) wait for
# the next month; hundreds no state could pay (no Spacers there) are dropped,
# since nobody is left to die. Every hundred killed goes on the year's count.
sr_pers_kill_step = {
	if = {
		limit = {
			has_variable = sr_pers_dead_due
			var:sr_pers_dead_due >= sr_pers_kill_batch
		}
		set_local_variable = {
			name = sr_pk_steps
			value = {
				value = var:sr_pers_dead_due
				divide = sr_pers_kill_batch
				floor = yes
				max = sr_pers_kill_max_steps
			}
		}
		# State by state, each takes its share of the hundreds still to place by
		# its share of the capacity still to share, so the states' hundreds add
		# up to the month's and the last takes what is left. Rounding each
		# state's share of the whole alone dropped a small due among three
		# states or more and doubled it between two (review of #2020).
		set_local_variable = { name = sr_pk_left value = local_var:sr_pk_steps }
		set_local_variable = { name = sr_pk_cap_left value = sr_pers_capacity_bases }
		set_local_variable = { name = sr_pk_killed value = 0 }
		every_scope_state = {
			limit = { modifier:state_sr_space_personnel_add > 0 }
			if = {
				limit = {
					local_var:sr_pk_left > 0
					local_var:sr_pk_cap_left > 0
				}
				set_local_variable = {
					name = sr_pk_here
					value = {
						value = modifier:state_sr_space_personnel_add
						divide = local_var:sr_pk_cap_left
						max = 1
						multiply = local_var:sr_pk_left
						round = yes
						max = sr_pers_state_spacer_steps
					}
				}
				change_local_variable = { name = sr_pk_cap_left subtract = modifier:state_sr_space_personnel_add }
				change_local_variable = { name = sr_pk_left subtract = local_var:sr_pk_here }
				if = {
					limit = { local_var:sr_pk_here > 0 }
					sr_pers_kill_check_arm = yes
					while = {
						count = local_var:sr_pk_here
						kill_population_in_state = { value = 100 pop_type = spacers }
					}
					set_local_variable = {
						name = sr_pk_people
						value = {
							value = local_var:sr_pk_here
							multiply = sr_pers_kill_batch
						}
					}
					change_local_variable = { name = sr_pk_killed add = local_var:sr_pk_people }
					te_demog_note_war_dead = { VALUE = local_var:sr_pk_people }
				}
			}
		}
		# The month's hundreds leave the due, killed or, where a state's Spacers
		# ran out, dropped (nobody is left there to die); the fraction and
		# anything past the cap stay.
		change_variable = {
			name = sr_pers_dead_due
			subtract = {
				value = local_var:sr_pk_steps
				multiply = sr_pers_kill_batch
			}
		}
		if = {
			limit = { local_var:sr_pk_killed > 0 }
			sr_pers_count_year = yes
		}
	}
}

# Country scope: local_var:sr_pk_killed on the year's count (sr_pers_killed_year),
# which starts again twelve months after it began (the panel's hover).
sr_pers_count_year = {
	if = {
		limit = { NOT = { has_variable = sr_pers_killed_year_from } }
		set_variable = { name = sr_pers_killed_year_from value = te_history_month_index }
		set_variable = { name = sr_pers_killed_year value = 0 }
	}
	else_if = {
		limit = { var:sr_pers_killed_year_from < { value = te_history_month_index subtract = 11 } }
		set_variable = { name = sr_pers_killed_year_from value = te_history_month_index }
		set_variable = { name = sr_pers_killed_year value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = sr_pers_killed_year } }
		set_variable = { name = sr_pers_killed_year value = 0 }
	}
	change_variable = { name = sr_pers_killed_year add = local_var:sr_pk_killed }
}

# ---- The order (design § 4) ----------------------------------------------------------------------
# Country scope (sr_pers_priority_sgui): Hulls first (absent, the default) or
# Marines first (sr_pers_marines_first).
sr_pers_toggle_priority = {
	if = {
		limit = { sr_pers_marines_first_on = yes }
		remove_variable = sr_pers_marines_first
	}
	else = {
		set_variable = { name = sr_pers_marines_first value = yes }
	}
}

# Country scope, an AI's month (sr_fl_country_month): marines first at war, so
# it doesn't send fleets without the marines to take colonies; hulls first in
# peace.
sr_pers_ai_priority = {
	if = {
		limit = {
			is_at_war = yes
			NOT = { sr_pers_marines_first_on = yes }
		}
		set_variable = { name = sr_pers_marines_first value = yes }
	}
	else_if = {
		limit = {
			is_at_war = no
			sr_pers_marines_first_on = yes
		}
		remove_variable = sr_pers_marines_first
	}
}

# ---- Old saves (design § 8) ------------------------------------------------------------------------
# Country scope, first in the fleets' month, once a country (sr_pers_migrated):
# a save from before the bases has fleets and marines and no personnel, which
# would wake with a crewed share of 0. With personnel in service and the
# reserve's crews (sr_pers_migrate_need) above the bases' capacity, it creates
# the levels the gap needs, rounded up (at most 100), government owned, in
# Space Command's state, else the capital; and for six months
# (sr_pers_grace_until) the share holds at 1 and the capacity at that need
# (sr_pers_grace_capacity), while the new levels hire, which
# takes about ten weeks at a tenth of the jobs a week. A new game's fleet
# powers pass with nothing in service and only take the mark. A revolution's
# winner inherits it, as it inherits the fleets.
sr_pers_migrate_once = {
	if = {
		limit = { NOT = { has_variable = sr_pers_migrated } }
		set_variable = { name = sr_pers_migrated value = yes }
		if = {
			limit = { sr_pers_migrate_need > sr_pers_capacity_bases }
			set_variable = { name = sr_pers_grace_capacity value = sr_pers_migrate_need }
			set_variable = {
				name = sr_pers_grace_until
				value = {
					value = te_history_month_index
					add = sr_pers_grace_months
				}
			}
			set_local_variable = { name = sr_pers_new_levels value = sr_pers_migrate_levels }
			sr_pers_migrate_build = yes
			debug_log = "TE_SR_PERS migrate country=[THIS.GetCountry.GetTagName] in_service=[THIS.Var('sr_pers_grace_capacity').GetValue|0] grace_until_month=[THIS.Var('sr_pers_grace_until').GetValue|0]"
		}
	}
}

# Country scope, the migrations' base levels: local_var:sr_pers_new_levels of
# them, government owned, in Space Command's state, else the capital.
sr_pers_migrate_build = {
	if = {
		limit = { local_var:sr_pers_new_levels > 0 }
		if = {
			limit = {
				any_scope_state = {
					has_building = building_space_command
				}
			}
			ordered_scope_state = {
				limit = { has_building = building_space_command }
				order_by = sr_sc_keeper_score
				max = 1
				extended_timeline_build_specified_building_level = {
					SPEC_TYPE = building_space_forces_base
					SPEC_LEVEL = local_var:sr_pers_new_levels
					SPEC_OWNER = owner
				}
			}
		}
		else_if = {
			limit = { exists = capital }
			capital = {
				extended_timeline_build_specified_building_level = {
					SPEC_TYPE = building_space_forces_base
					SPEC_LEVEL = local_var:sr_pers_new_levels
					SPEC_OWNER = owner
				}
			}
		}
	}
}

# Global, from the sites' monthly step under Full Colonies, once a save
# (sr_pers_modules_migrated): colony modules take personnel (the owner,
# 2026-11-25; gen_solar_sites.PERSONNEL_TYPES), so a save from before has
# Spaceports, Defenses, Black Sites, Training Grounds and Nuclear Command
# Centers with nobody to staff them, many of them held by countries with no
# fleet, whose month never ran the migration above. Every country holding such
# a module, migrated or not, whose need (sr_pers_migrate_need, modules
# included) is above its bases' capacity gets the levels its need lacks beyond
# those it has built and queued (sr_pers_mod_migrate_levels), and the grace at
# that need. It counts as the migration above (sr_pers_migrated), which would
# otherwise see the new levels' staff as missing while they hire and build them
# again. A new game runs it in its first month, before any colony, so a new
# game's modules wait for bases of their own.
sr_pers_modules_migrate_once = {
	if = {
		limit = { NOT = { has_global_variable = sr_pers_modules_migrated } }
		set_global_variable = { name = sr_pers_modules_migrated value = yes }
		every_country = {
			limit = { sr_pers_module_levels > 0 }
			set_variable = { name = sr_pers_migrated value = yes }
			if = {
				limit = { sr_pers_migrate_need > sr_pers_capacity_bases }
				set_variable = { name = sr_pers_grace_capacity value = sr_pers_migrate_need }
				set_variable = {
					name = sr_pers_grace_until
					value = {
						value = te_history_month_index
						add = sr_pers_grace_months
					}
				}
				set_local_variable = { name = sr_pers_new_levels value = sr_pers_mod_migrate_levels }
				sr_pers_migrate_build = yes
				debug_log = "TE_SR_PERS migrate_modules country=[THIS.GetCountry.GetTagName] need=[THIS.Var('sr_pers_grace_capacity').GetValue|0] grace_until_month=[THIS.Var('sr_pers_grace_until').GetValue|0]"
			}
		}
	}
}

# ---- Self-checks (design § 9; docs/testing/solar-colonies-playtest.md) -----------------------------
# TE_SELF_CHECK pers_pop_type (scope § 9.1, 9.5): the mod's first pop type
# loads, the base hires it, and the capacity sum is the base's own. Armed, on
# the country (a global arm would be cleared every month by a fleet power with
# no base that runs first), the first month it has a Space Forces Base of a
# level or more (sr_pers_has_base_level), whatever its staff, so a pop type that
# never hires still gets a line; read three months later, in such a base's
# state (more than a new base's hiring needs, at a tenth of its jobs a week):
# BROKEN when the state holds no Spacers; re-armed while the base is under half
# staff; else OK when the state's Space Forces Personnel is 0.75 to 1.5 times
# 1,000 x the base's levels x its occupancy (a workforce-scaled modifier follows
# throughput too, which an unincorporated state, economies of scale or a new
# capital move by a tenth or more, and the sum lags the live staff a tick;
# review of #2020), BROKEN outside it. The figures go through global
# variables: a debug line reads no local variable. Once a game.
sr_pers_pop_type_self_check = {
	if = {
		limit = { NOT = { has_global_variable = sr_pers_pop_type_checked } }
		if = {
			limit = {
				has_variable = sr_pers_pop_type_armed
				te_history_month_index >= {
					value = var:sr_pers_pop_type_armed
					add = 3
				}
			}
			if = {
				limit = { NOT = { sr_pers_has_base_level = yes } }
				remove_variable = sr_pers_pop_type_armed
			}
			else = {
				random_scope_state = {
					limit = {
						any_scope_building = {
							is_building_type = building_space_forces_base
							level >= 1
						}
					}
					set_global_variable = { name = sr_pers_log_spacers value = sr_pers_state_spacers }
					set_global_variable = { name = sr_pers_log_cap value = modifier:state_sr_space_personnel_add }
					random_scope_building = {
						limit = { is_building_type = building_space_forces_base }
						set_global_variable = { name = sr_pers_log_occ value = occupancy }
						set_global_variable = {
							name = sr_pers_log_want
							value = {
								value = level
								multiply = occupancy
								multiply = sr_pers_per_level
							}
						}
					}
				}
				if = {
					limit = { global_var:sr_pers_log_spacers <= 0 }
					set_global_variable = { name = sr_pers_pop_type_checked value = yes }
					debug_log = "TE_SELF_CHECK BROKEN pers_pop_type country=[THIS.GetCountry.GetTagName] spacers=0 occupancy=[GetGlobalVariable('sr_pers_log_occ').GetValue|2] personnel=[GetGlobalVariable('sr_pers_log_cap').GetValue|0] (no Spacers in a Space Forces Base's state three months after it was built: the pop type or its social class doesn't load or doesn't hire; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else_if = {
					limit = { global_var:sr_pers_log_occ < 0.5 }
					set_variable = { name = sr_pers_pop_type_armed value = te_history_month_index }
				}
				else_if = {
					limit = {
						global_var:sr_pers_log_cap >= { value = global_var:sr_pers_log_want multiply = 0.75 }
						global_var:sr_pers_log_cap <= { value = global_var:sr_pers_log_want multiply = 1.5 }
					}
					set_global_variable = { name = sr_pers_pop_type_checked value = yes }
					debug_log = "TE_SELF_CHECK OK pers_pop_type country=[THIS.GetCountry.GetTagName] spacers=[GetGlobalVariable('sr_pers_log_spacers').GetValue|0] occupancy=[GetGlobalVariable('sr_pers_log_occ').GetValue|2] personnel=[GetGlobalVariable('sr_pers_log_cap').GetValue|0] want=[GetGlobalVariable('sr_pers_log_want').GetValue|0] (the base hires Spacers and its state's Space Forces Personnel follows its staffed levels) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					set_global_variable = { name = sr_pers_pop_type_checked value = yes }
					debug_log = "TE_SELF_CHECK BROKEN pers_pop_type country=[THIS.GetCountry.GetTagName] spacers=[GetGlobalVariable('sr_pers_log_spacers').GetValue|0] occupancy=[GetGlobalVariable('sr_pers_log_occ').GetValue|2] personnel=[GetGlobalVariable('sr_pers_log_cap').GetValue|0] want=[GetGlobalVariable('sr_pers_log_want').GetValue|0] (the state's Space Forces Personnel is far from 1,000 a staffed level: 0 means the method's state modifier doesn't reach the state; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				if = {
					limit = { has_global_variable = sr_pers_pop_type_checked }
					remove_variable = sr_pers_pop_type_armed
				}
				remove_global_variable = sr_pers_log_spacers
				remove_global_variable = sr_pers_log_cap
				remove_global_variable = sr_pers_log_want
				remove_global_variable = sr_pers_log_occ
			}
		}
		else_if = {
			limit = {
				NOT = { has_variable = sr_pers_pop_type_armed }
				sr_pers_has_base_level = yes
			}
			set_variable = { name = sr_pers_pop_type_armed value = te_history_month_index }
		}
	}
}

# TE_SELF_CHECK pers_kill (scope § 9.2): kill_population_in_state with the mod's
# pop type takes the people. State scope, the first state that kills a hundred
# or more (sr_pers_kill_step, before its kill): its Spacers and the count are
# kept on it, and a day on a hidden event (space_race_events.140, the
# country's) reads them again: soon enough that the base's weekly hiring, which
# refills a small kill within a week, has most likely not run (a week let it,
# review of #2020).
sr_pers_kill_check_arm = {
	if = {
		limit = { NOT = { has_global_variable = sr_pers_kill_armed } }
		set_global_variable = { name = sr_pers_kill_armed value = yes }
		set_variable = { name = sr_pk_check_before value = sr_pers_state_spacers }
		set_variable = {
			name = sr_pk_check_count
			value = {
				value = local_var:sr_pk_here
				multiply = sr_pers_kill_batch
			}
		}
		save_scope_as = sr_pk_check_state
		owner = {
			set_variable = { name = sr_pk_check_state value = scope:sr_pk_check_state }
			trigger_event = { id = space_race_events.140 days = 1 }
		}
	}
}

# Country scope, space_race_events.140's immediate: the read.
sr_pers_kill_self_check = {
	if = {
		limit = {
			has_variable = sr_pk_check_state
			var:sr_pk_check_state ?= {
				has_variable = sr_pk_check_before
				has_variable = sr_pk_check_count
			}
		}
		var:sr_pk_check_state = {
			set_global_variable = { name = sr_pers_log_before value = var:sr_pk_check_before }
			set_global_variable = { name = sr_pers_log_count value = var:sr_pk_check_count }
			set_global_variable = { name = sr_pers_log_after value = sr_pers_state_spacers }
			remove_variable = sr_pk_check_before
			remove_variable = sr_pk_check_count
		}
		remove_variable = sr_pk_check_state
		if = {
			limit = {
				global_var:sr_pers_log_after <= {
					value = global_var:sr_pers_log_count
					multiply = -0.5
					add = global_var:sr_pers_log_before
				}
			}
			debug_log = "TE_SELF_CHECK OK pers_kill country=[THIS.GetCountry.GetTagName] spacers_before=[GetGlobalVariable('sr_pers_log_before').GetValue|0] killed=[GetGlobalVariable('sr_pers_log_count').GetValue|0] spacers_day_on=[GetGlobalVariable('sr_pers_log_after').GetValue|0] (kill_population_in_state with pop_type = spacers took the people) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN pers_kill country=[THIS.GetCountry.GetTagName] spacers_before=[GetGlobalVariable('sr_pers_log_before').GetValue|0] killed=[GetGlobalVariable('sr_pers_log_count').GetValue|0] spacers_day_on=[GetGlobalVariable('sr_pers_log_after').GetValue|0] (the state's Spacers fell by less than half the count a day on: the kill doesn't reach the mod's pop type; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_global_variable = sr_pers_log_before
		remove_global_variable = sr_pers_log_count
		remove_global_variable = sr_pers_log_after
	}
}
