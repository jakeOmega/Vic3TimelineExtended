# Company AI: the AI dissolves a company that has failed for five years.
#
# The engine's AI never disbands a company by choice: in the observer run's
# saves (1972-2029) no surviving country dropped one company to found another,
# and none of the great or major powers lost one. Its slots fill decades before
# the mod's late-era companies unlock, so those never get founded. This stands
# in for the choice a player makes in the companies panel.
#
# Script cannot read a company's profit, so "failing" is prosperity below
# te_company_failing_prosperity, or owned levels below te_company_small_share
# of the country's building levels, at te_company_failing_months consecutive
# monthly checks; a check that passes both clears the count
# (te_company_ai_values.txt has the evidence for the numbers).
# on_monthly_pulse_country is a 30-day timer, so 60 checks are 4.9 years.
#
# Country scope. Every country counts, players included, so the count is a fact
# about the company; only an AI acts on it. A country that changes hands acts
# on the count it inherits.
te_company_failing_monthly_effect = {
	save_temporary_scope_value_as = {
		name = te_company_min_levels
		value = {
			value = building_levels
			multiply = te_company_small_share
		}
	}
	every_company = {
		if = {
			limit = { te_company_fails_this_month = yes }
			if = {
				limit = { has_variable = te_company_months_failing }
				change_variable = { name = te_company_months_failing add = 1 }
			}
			else = {
				set_variable = { name = te_company_months_failing value = 1 }
			}
		}
		else_if = {
			limit = { has_variable = te_company_months_failing }
			remove_variable = te_company_months_failing
		}
	}
	# One company a check at most; a second failing one goes 30 days later.
	if = {
		limit = {
			is_ai = yes
			any_company = { te_company_is_failing = yes }
		}
		random_company = {
			limit = { te_company_is_failing = yes }
			owner ?= {
				remove_company = prev.type
			}
		}
		# The monthly cleanup in extra_on_actions.txt (and on_company_disbanded,
		# if remove_company fires it) would remove the flagship within 30 days;
		# this does it now. Same state-scope call and guard as the monthly one.
		every_scope_state = {
			limit = {
				state_has_building_group_levels = { type = bg_company_buildings value >= 1 }
			}
			remove_disbanded_company_buildings_effect = yes
		}
	}
}

# Company AI: the AI subsidizes its company flagships while it is not in default.
#
# The engine's AI almost never changes a subsidy itself: it keeps the ones a
# building gets by default (ports, railways, power plants) and adds none. On
# the 2073 save it subsidized 0 of its 23 company buildings (the player 20 of
# 20), 0 of 140 shipyards, and 0 of 250 airports and highways outside the laws
# that force them. 12 of the 23 were under half staffed, some losing workers to
# subsidized buildings nearby: a building that is not subsidized stops hiring
# when earnings per worker would fall below
# BUILDING_DEFAULT_MIN_EARNINGS_TO_HIRE_EMPLOYEES. A subsidy only tops a
# building up to SUBSIDIES_MIN_GAIN (0), so a profitable flagship costs nothing.
#
# Country scope, AI only. In default it stops: the AI never drops a subsidy on
# its own (in default it still paid for 60 of 61 railways), so this does. A
# player's subsidies are the player's.
te_company_ai_subsidy_monthly_effect = {
	if = {
		limit = {
			is_ai = yes
			in_default = no
		}
		every_scope_state = {
			limit = {
				state_has_building_group_levels = { type = bg_company_buildings value >= 1 }
			}
			every_scope_building = {
				limit = {
					is_building_group = bg_company_buildings
					is_subsidized = no
				}
				set_subsidized = yes
			}
		}
	}
	else_if = {
		limit = {
			is_ai = yes
			in_default = yes
		}
		every_scope_state = {
			limit = {
				state_has_building_group_levels = { type = bg_company_buildings value >= 1 }
			}
			every_scope_building = {
				limit = {
					is_building_group = bg_company_buildings
					is_subsidized = yes
				}
				set_subsidized = no
			}
		}
	}
}
