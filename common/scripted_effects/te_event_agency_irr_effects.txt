# ============================================================================
# IRREDENTISM: ASKING THE BLOC LEADER TO BLESS A WAR OF REUNIFICATION
# See te_event_agency_irr_triggers.txt for the two paths that send a request.
# ============================================================================

# Country scope (the member asking), with scope:target_country bound to the
# country it would go to war with. Sends irredentism.7 to the member's bloc
# leader and marks the request as awaiting an answer, which blocks a second
# request (irr_war_blessing_requester_ready) until irredentism.7 is answered.
# The 90-day expiry only matters if irredentism.7 never fires (its trigger
# fails because the member left the bloc, say); an answer clears it at once.
irr_send_war_blessing_request = {
	save_scope_as = irr_requester
	set_variable = {
		name = irr_war_blessing_pending
		days = 90
	}
	power_bloc ?= {
		power_bloc_leader ?= {
			trigger_event = { id = irredentism.7 days = 3 }
		}
	}
}

# Country scope (the member that asked). Called by both irredentism.7 options.
irr_clear_war_blessing_pending = {
	if = {
		limit = { has_variable = irr_war_blessing_pending }
		remove_variable = irr_war_blessing_pending
	}
}
