# ============================================================================
# UN MISSIONS — THE ONLY WRITERS (redesign phase 6)
# ============================================================================
# docs/systems/un_redesign_design.md §9; as shipped, §0.6. The figures and the
# table of mission types are at the top of
# common/script_values/un_mission_values.txt; the tests are in
# common/scripted_triggers/un_mission_triggers.txt.
#
# A MISSION is a script container, in the mandate pattern:
#   tags    un_mission, un_msn_<type> (peacekeeping | aid | stabilisation |
#           electoral), un_msn_active, closed as un_msn_succeeded /
#           un_msn_failed / un_msn_lapsed
#   vars    un_msn_seq, un_msn_type (1 / 2 / 3 / 4), un_msn_host (country),
#           un_msn_requested (the host asked the Assembly for it: it lapses if
#           the host leaves the UN),
#           un_msn_state (state), un_msn_months, un_msn_progress (0-100),
#           un_msn_strength (this month's), un_msn_start_year, un_msn_end_year,
#           un_msn_sol_open (the state's underlying SoL at opening: the aid
#           goal's base), un_msn_war_at_open (the host was at war and not the
#           initiator when it opened; cleared by its first month of peace),
#           un_msn_staffed (a contributor has joined at some point),
#           un_msn_slot (0-7, the chamber's display slot while active;
#           un_mission_assign_slot),
#           un_msn_election_held (electoral: an election campaign ended
#           while it ran; it succeeds at the next monthly update)
#   list    un_msn_contributors (countries),
#           un_msn_volunteers (the contributors that joined at will and pay
#           for it: un_mission_volunteer, un_mission_charges),
#           un_msn_withdrawn (members that brought their contingent home and
#           cannot send it back: un_mission_withdraw)
#   global  un_mission_registry (every mission, capped at
#           un_mission_registry_cap; active ones are never evicted),
#           un_mission_seq
#   state   un_msn_here (the state's active mission), un_msn_power (the
#           strength its modifier is multiplied by)
#   country un_msn_service_cached (the weekly cost of the missions it serves
#           in at will: un_mission_service_cost's multiplier),
#           un_msn_electoral_power (the strength of the electoral mission it
#           hosts: un_electoral_observers_modifier's multiplier)
#
# WHO WRITES WHAT
#   un_mission_open / un_mission_join_host   the openers (un_vote.2, un_events.4 A,
#                                             un_events.7 A), and joiners
#                                             (un_vote_apply_resolution_compliance)
#   un_missions_monthly_update                every month, from
#                                             un_global_authority_on_action:
#                                             months, contributors, strength,
#                                             progress, outcome
#   un_mission_state_refresh                  the state's modifier, from the state
#                                             monthly pulse (un_mission_on_actions.txt):
#                                             the single refresh site for the three
#                                             un_mission_*_state_modifier
#   un_missions_dissolve                      a dissolution closes every mission
#   un_mission_volunteer / un_mission_withdraw a member joins or leaves at
#                                             will: the chamber's Missions in
#                                             the Field rows
#                                             (un_mission_slot_volunteer /
#                                             _withdraw) and the AI
#                                             (un_mission_ai_monthly)
#   un_mission_service_refresh                the single refresh site for
#                                             un_mission_service_cost (the JE's
#                                             monthly pulse, and at once on
#                                             joining or leaving)
#   un_mission_open_electoral                 a carried request for electoral
#                                             observers (un_observer_request_on_carried)
#   un_mission_electoral_election_held        on_election_campaign_end
#                                             (un_mission_on_actions.txt): the
#                                             host's election was observed
#   un_mission_electoral_host_refresh         the single refresh site for
#                                             un_electoral_observers_modifier on
#                                             the host (the country monthly pulse,
#                                             un_mission_on_actions.txt)
#
# REASON CODES: ledger 21/22/23 (a peacekeeping / aid / stabilisation mission
# accomplished), 24/25/26 (one failed), each naming the mission's state —
# 19 and 20, the same without the kind or the state, survive only in saves;
# standing 34 (a mission we contributed to accomplished its work), 35 (we
# brought a contingent home before the work was done).
# Phase 7 (§0.12; the missions block is ledger 37-39, standing 45-47): ledger
# 37 an electoral mission accomplished, 38 one failed, each naming its state;
# standing 45 (the host: an election held under the observers), 46 (the host
# did away with its elections while they watched). Ledger 39 and standing 47
# are unused.
# ============================================================================

# ---- Opening and joining ------------------------------------------------------------

# Open a mission of $TYPE$ ($CODE$ 1 peacekeeping, 2 aid, 3 stabilisation,
# 4 electoral) for
# scope:un_msn_host in scope:un_msn_state, or, if that state already holds an
# active mission of the same type, take that one. Leaves scope:un_msn_new
# pointing at the mission (unset if the state holds a mission of another type).
# Scope: any
un_mission_open = {
	if = {
		limit = {
			scope:un_msn_state = {
				var:un_msn_here ?= {
					has_tag = un_msn_active
					has_tag = un_msn_$TYPE$
				}
			}
		}
		scope:un_msn_state = {
			var:un_msn_here = { save_scope_as = un_msn_new }
		}
	}
	else_if = {
		limit = {
			scope:un_msn_state = { NOT = { un_mission_state_occupied = yes } }
		}
		if = {
			limit = { NOT = { has_global_variable = un_mission_seq } }
			set_global_variable = { name = un_mission_seq value = 0 }
		}
		change_global_variable = { name = un_mission_seq add = 1 }
		create_container = {
			tags = { un_mission un_msn_$TYPE$ un_msn_active }
			save_scope_as = un_msn_new
		}
		if = {
			limit = { exists = scope:un_msn_new }
			scope:un_msn_new = {
				set_variable = { name = un_msn_seq value = global_var:un_mission_seq }
				set_variable = { name = un_msn_type value = $CODE$ }
				set_variable = { name = un_msn_host value = scope:un_msn_host }
				set_variable = { name = un_msn_state value = scope:un_msn_state }
				set_variable = { name = un_msn_months value = 0 }
				set_variable = { name = un_msn_progress value = 0 }
				set_variable = { name = un_msn_start_year value = te_history_current_year }
				set_variable = { name = un_msn_strength value = un_mission_strength_value }
				set_variable = { name = un_msn_sol_open value = un_mission_underlying_sol }
				# A mission sent into a war the host did not start is not
				# failed by that war (un_mission_attacked_since_open).
				if = {
					limit = { scope:un_msn_host = { un_mission_host_attacked = yes } }
					set_variable = un_msn_war_at_open
				}
			}
			scope:un_msn_state = {
				set_variable = { name = un_msn_here value = scope:un_msn_new }
			}
			un_mission_register = yes
			scope:un_msn_new = { un_mission_assign_slot = yes }
			scope:un_msn_host = { post_notification = un_mission_opened_notice }
			# The World Food Reserve (redesign phase 7, §0.12): a new aid
			# mission draws grain from the members' pledges
			# (un_economy_effects.txt). A no-op while there is no reserve.
			if = {
				limit = { scope:un_msn_new = { has_tag = un_msn_aid } }
				un_food_reserve_on_aid_mission_opened = yes
			}
		}
	}
}

# Take the host's active mission of $TYPE$ if it has one, or open one in its
# state chosen by $ORDER$ (a state-scope script value; the highest wins) among
# those without a mission. Leaves scope:un_msn_new pointing at the mission
# (unset if every state of the host already holds another kind).
# Parameters: $TYPE$, $CODE$, $ORDER$
# Scope: any; expects scope:un_msn_host
un_mission_join_host = {
	if = {
		limit = {
			has_global_variable_list = un_mission_registry
			any_in_global_list = {
				variable = un_mission_registry
				has_tag = un_msn_active
				has_tag = un_msn_$TYPE$
				var:un_msn_host ?= scope:un_msn_host
			}
		}
		random_in_global_list = {
			variable = un_mission_registry
			limit = {
				has_tag = un_msn_active
				has_tag = un_msn_$TYPE$
				var:un_msn_host ?= scope:un_msn_host
			}
			save_scope_as = un_msn_new
		}
	}
	else = {
		scope:un_msn_host = {
			ordered_scope_state = {
				limit = { NOT = { un_mission_state_occupied = yes } }
				order_by = $ORDER$
				max = 1
				save_scope_as = un_msn_state
			}
		}
		if = {
			limit = { exists = scope:un_msn_state }
			un_mission_open = { TYPE = $TYPE$ CODE = $CODE$ }
		}
	}
}

# The scoped country joins scope:un_msn_new as a contributor (once), and the
# mission's strength is re-read at once. A volunteer that now pledges or
# answers an appeal for the same mission pays through that path from now on,
# so it is no longer billed as a volunteer (un_mission_volunteer adds it to
# un_msn_volunteers after this).
# Scope: the contributing country; expects scope:un_msn_new
un_mission_add_contributor = {
	save_scope_as = un_msn_contributor
	scope:un_msn_new ?= {
		if = {
			limit = { un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = scope:un_msn_contributor } }
			remove_list_variable = { name = un_msn_volunteers target = scope:un_msn_contributor }
		}
		if = {
			limit = {
				has_tag = un_msn_active
				NOT = { un_lobby_in_list = { LIST = un_msn_contributors COUNTRY = scope:un_msn_contributor } }
			}
			add_to_variable_list = { name = un_msn_contributors target = scope:un_msn_contributor }
			set_variable = un_msn_staffed
			set_variable = { name = un_msn_strength value = un_mission_strength_value }
		}
	}
}

# A member pledging to a carried aid or peacekeeping request joins the mission
# the resolution opened (un_res_mission), if it opened one.
# Scope: the pledging country; expects scope:un_resolution
un_mission_join_resolution = {
	scope:un_resolution ?= {
		var:un_res_mission ?= { save_scope_as = un_msn_new }
	}
	if = {
		limit = { scope:un_msn_new ?= { has_tag = un_msn_active } }
		custom_tooltip = un_mission_joined_tt
		hidden_effect = { un_mission_add_contributor = yes }
	}
}

# The host sends the mission home: its active mission of $TYPE$ fails.
# Scope: the host country
un_mission_expel = {
	save_scope_as = un_msn_expeller
	if = {
		limit = { has_global_variable_list = un_mission_registry }
		every_in_global_list = {
			variable = un_mission_registry
			limit = {
				has_tag = un_msn_active
				has_tag = un_msn_$TYPE$
				var:un_msn_host ?= scope:un_msn_expeller
			}
			un_mission_close = { STATUS = failed }
		}
	}
}

# File a new mission in the register; past the cap, destroy the oldest closed
# one. Active missions are never evicted.
# Scope: any; expects scope:un_msn_new
un_mission_register = {
	add_to_global_variable_list = { name = un_mission_registry target = scope:un_msn_new }
	if = {
		limit = {
			# N is un_mission_registry_cap + 1 (the cap is 12).
			any_in_global_list = { variable = un_mission_registry count >= 13 has_tag = un_mission }
			any_in_global_list = {
				variable = un_mission_registry
				NOT = { has_tag = un_msn_active }
			}
		}
		ordered_in_global_list = {
			variable = un_mission_registry
			position = 0
			limit = { NOT = { has_tag = un_msn_active } }
			order_by = un_mission_age_order
			save_scope_as = un_msn_evicted
		}
		remove_list_global_variable = { name = un_mission_registry target = scope:un_msn_evicted }
		scope:un_msn_evicted = { destroy_container = yes }
	}
}

# Give the scoped active mission the lowest free display slot (0-7), if it has
# none. With all eight taken it waits, and un_mission_monthly asks again. The
# chain is the slot table: keep it in step with the three mission sguis and
# the widget's rows (test_un_chamber_mission_slots.py).
# Scope: mission container
un_mission_assign_slot = {
	if = {
		limit = {
			has_tag = un_msn_active
			NOT = { has_variable = un_msn_slot }
		}
		if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 0 } } }
			set_variable = { name = un_msn_slot value = 0 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 1 } } }
			set_variable = { name = un_msn_slot value = 1 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 2 } } }
			set_variable = { name = un_msn_slot value = 2 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 3 } } }
			set_variable = { name = un_msn_slot value = 3 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 4 } } }
			set_variable = { name = un_msn_slot value = 4 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 5 } } }
			set_variable = { name = un_msn_slot value = 5 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 6 } } }
			set_variable = { name = un_msn_slot value = 6 }
		}
		else_if = {
			limit = { NOT = { un_mission_slot_filled = { SLOT = 7 } } }
			set_variable = { name = un_msn_slot value = 7 }
		}
	}
}

# ---- Joining at will ----------------------------------------------------------------
# Any major-power member may send a contingent to a mission in the field, and
# bring it home again, without waiting for the Assembly to ask: the player
# through the chamber's Missions in the Field rows, the AI through
# un_mission_ai_monthly. Both ask un_mission_volunteer_eligible and
# un_mission_accepts_volunteer (un_mission_triggers.txt) and run the two
# writers below. The docket's appeals (un_events.4, un_events.7) and the
# Assembly's resolutions still bring their own contributors, who pay through
# those paths; a volunteer pays un_mission_service_cost instead.

# The scoped country sends a contingent to scope:un_msn_new. It serves as a
# contributor, listed as a volunteer, so it pays for as long as it serves
# (un_mission_charges); the host hears of it.
# Scope: the volunteering country; expects scope:un_msn_new
un_mission_volunteer = {
	save_temporary_scope_as = un_msn_volunteer
	un_mission_volunteer_tooltip = yes
	hidden_effect = {
		un_mission_add_contributor = yes
		scope:un_msn_new ?= {
			if = {
				limit = { NOT = { un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = scope:un_msn_volunteer } } }
				add_to_variable_list = { name = un_msn_volunteers target = scope:un_msn_volunteer }
			}
			var:un_msn_host ?= { post_notification = un_mission_reinforced_notice }
		}
		un_mission_service_refresh = yes
	}
}

# What un_mission_volunteer does, in words: where the contingent goes, what
# it costs, what it adds, and what leaving would cost.
# Scope: the volunteering country; expects scope:un_msn_new
un_mission_volunteer_tooltip = {
	scope:un_msn_new ?= {
		var:un_msn_state ?= { save_temporary_scope_as = un_msn_tt_state }
		var:un_msn_host ?= { save_temporary_scope_as = un_msn_tt_host }
	}
	# An electoral observer mission takes no contingents (the Join control is
	# greyed for it, but its tooltip still runs this): say so and nothing else.
	if = {
		limit = { scope:un_msn_new ?= { has_tag = un_msn_electoral } }
		custom_tooltip = un_mission_volunteer_electoral_tt
	}
	else = {
		if = {
			limit = {
				exists = scope:un_msn_tt_state
				exists = scope:un_msn_tt_host
			}
			if = {
				limit = { scope:un_msn_new ?= { has_tag = un_msn_peacekeeping } }
				custom_tooltip = un_mission_volunteer_peacekeeping_tt
			}
			else_if = {
				limit = { scope:un_msn_new ?= { has_tag = un_msn_aid } }
				custom_tooltip = un_mission_volunteer_aid_tt
			}
			else = {
				custom_tooltip = un_mission_volunteer_stabilisation_tt
			}
			custom_tooltip = un_mission_volunteer_rewards_tt
		}
		if = {
			limit = {
				scope:un_msn_new ?= { un_mission_is_peace_operation = yes }
				un_standing_program_active_peacekeeping = yes
			}
			custom_tooltip = un_mission_volunteer_covered_tt
		}
		else = {
			custom_tooltip = un_mission_volunteer_cost_tt
		}
		# The Standing UN Force counts as a contributor (phase 7).
		if = {
			limit = { scope:un_msn_new ?= { un_mission_strength_heads >= 3 } }
			custom_tooltip = un_mission_volunteer_full_tt
		}
		else = {
			custom_tooltip = un_mission_volunteer_strength_tt
		}
		custom_tooltip = un_mission_volunteer_commitment_tt
	}
}

# The scoped country brings its contingent home from scope:un_msn_new. It
# loses standing and the host's regard, and cannot send it back. A
# peacekeeping or stabilisation mission left with nobody fails at the next
# monthly update (un_mission_monthly), as it would if its last contributor
# left the UN.
# Scope: the withdrawing country; expects scope:un_msn_new
un_mission_withdraw = {
	save_temporary_scope_as = un_msn_leaver
	scope:un_msn_new ?= {
		var:un_msn_state ?= { save_temporary_scope_as = un_msn_tt_state }
		var:un_msn_host ?= { save_temporary_scope_as = un_msn_tt_host }
	}
	if = {
		limit = { exists = scope:un_msn_tt_state }
		custom_tooltip = un_mission_withdraw_tt
	}
	custom_tooltip = {
		text = un_mission_withdraw_standing_tt
		un_standing_loss = { AMOUNT = un_mission_withdraw_standing REASON = 35 }
	}
	if = {
		limit = { exists = scope:un_msn_tt_host }
		custom_tooltip = {
			text = un_mission_withdraw_relations_tt
			scope:un_msn_tt_host = {
				change_relations = { country = scope:un_msn_leaver value = -10 }
			}
		}
	}
	# The last to leave a peacekeeping or stabilisation mission fails it,
	# unless the Standing UN Force serves in it (phase 7).
	if = {
		limit = {
			scope:un_msn_new ?= {
				un_mission_is_peace_operation = yes
				NOT = {
					any_in_list = {
						variable = un_msn_contributors
						exists = this
						NOT = { this = scope:un_msn_leaver }
					}
				}
			}
		}
		if = {
			limit = { scope:un_msn_new ?= { un_mission_force_serves = yes } }
			custom_tooltip = un_mission_withdraw_last_force_tt
		}
		else = {
			custom_tooltip = un_mission_withdraw_last_tt
		}
	}
	custom_tooltip = un_mission_withdraw_final_tt
	hidden_effect = {
		scope:un_msn_new ?= {
			if = {
				limit = { un_mission_served_by = { WHO = scope:un_msn_leaver } }
				remove_list_variable = { name = un_msn_contributors target = scope:un_msn_leaver }
			}
			if = {
				limit = { un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = scope:un_msn_leaver } }
				remove_list_variable = { name = un_msn_volunteers target = scope:un_msn_leaver }
			}
			if = {
				limit = { NOT = { un_mission_withdrawn_by = { WHO = scope:un_msn_leaver } } }
				add_to_variable_list = { name = un_msn_withdrawn target = scope:un_msn_leaver }
			}
			set_variable = { name = un_msn_strength value = un_mission_strength_value }
			var:un_msn_host ?= { post_notification = un_mission_withdrawn_notice }
		}
		un_mission_service_refresh = yes
	}
}

# The mission in display slot $SLOT$, as scope:un_msn_new (unset if the slot
# is empty).
# Scope: any
un_mission_slot_select = {
	if = {
		limit = { un_mission_slot_filled = { SLOT = $SLOT$ } }
		ordered_in_global_list = {
			variable = un_mission_registry
			limit = { un_mission_in_slot = { SLOT = $SLOT$ } }
			order_by = un_mission_recency_order
			max = 1
			save_temporary_scope_as = un_msn_new
		}
	}
}

# The chamber's Join and Withdraw controls, for the mission in slot $SLOT$
# (un_chamber_mission_join_sgui / un_chamber_mission_withdraw_sgui; their
# is_valid asks un_mission_slot_can_volunteer / un_mission_slot_served).
# Scope: country (the player)
un_mission_slot_volunteer = {
	un_mission_slot_select = { SLOT = $SLOT$ }
	if = {
		limit = { exists = scope:un_msn_new }
		un_mission_volunteer = yes
	}
}

un_mission_slot_withdraw = {
	un_mission_slot_select = { SLOT = $SLOT$ }
	if = {
		limit = { exists = scope:un_msn_new }
		un_mission_withdraw = yes
	}
}

# The weekly cost of the missions the scoped country serves in at will. The
# single refresh site for un_mission_service_cost: the journal entry's monthly
# pulse, and un_mission_volunteer / un_mission_withdraw, so a contingent is
# paid for from the week it goes to the week it comes home. The multiplier's
# variable is removed only after the modifier is.
# Scope: country
un_mission_service_refresh = {
	if = {
		limit = {
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			un_mission_service_count > 0
		}
		set_variable = { name = un_msn_service_cached value = un_mission_service_expense_value }
		if = {
			limit = { has_modifier = un_mission_service_cost }
			remove_modifier = un_mission_service_cost
		}
		add_modifier = {
			name = un_mission_service_cost
			multiplier = var:un_msn_service_cached
		}
	}
	else = {
		if = {
			limit = { has_modifier = un_mission_service_cost }
			remove_modifier = un_mission_service_cost
		}
		if = {
			limit = { has_variable = un_msn_service_cached }
			remove_variable = un_msn_service_cached
		}
	}
}

# An AI member's month in the field. A member attacked, in default or deep
# in debt brings one contingent home; otherwise, now and then, one that is
# under its cap sends a contingent where it is wanted most
# (un_mission_ai_join_score). Only contingents sent at will come home here:
# a pledge made to the Assembly stands.
# Scope: country (an AI member; ROOT); from je_united_nations' monthly pulse
un_mission_ai_monthly = {
	if = {
		limit = {
			has_global_variable_list = un_mission_registry
			OR = {
				AND = {
					is_at_war = yes
					is_diplomatic_play_initiator = no
				}
				in_default = yes
				scaled_debt >= 0.75
			}
			any_in_global_list = {
				variable = un_mission_registry
				has_tag = un_msn_active
				un_mission_served_by = { WHO = ROOT }
				un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = ROOT }
			}
		}
		ordered_in_global_list = {
			variable = un_mission_registry
			limit = {
				has_tag = un_msn_active
				un_mission_served_by = { WHO = ROOT }
				un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = ROOT }
			}
			order_by = un_mission_recency_order
			max = 1
			save_scope_as = un_msn_new
		}
		un_mission_withdraw = yes
	}
	else_if = {
		limit = {
			has_global_variable_list = un_mission_registry
			un_mission_volunteer_eligible = yes
			is_at_war = no
			in_default = no
			scaled_debt < 0.5
			un_mission_volunteer_count < un_mission_ai_volunteer_cap
			any_in_global_list = {
				variable = un_mission_registry
				un_mission_accepts_volunteer = { WHO = ROOT }
				un_mission_ai_join_score > un_mission_ai_join_threshold
			}
		}
		random = {
			chance = 10
			ordered_in_global_list = {
				variable = un_mission_registry
				limit = {
					un_mission_accepts_volunteer = { WHO = ROOT }
					un_mission_ai_join_score > un_mission_ai_join_threshold
				}
				order_by = un_mission_ai_join_score
				max = 1
				save_scope_as = un_msn_new
			}
			un_mission_volunteer = yes
		}
	}
}

# ---- Monthly ---------------------------------------------------------------------------

# Global monthly upkeep (no-country scope; from un_global_authority_on_action,
# UN founded). NOT IDEMPOTENT WITHIN A MONTH: the months and progress advance.
un_missions_monthly_update = {
	if = {
		limit = { has_global_variable_list = un_mission_registry }
		every_in_global_list = {
			variable = un_mission_registry
			limit = { has_tag = un_msn_active }
			un_mission_monthly = yes
		}
	}
}

# One active mission's month: the contributors still in it, its strength, its
# progress, and its outcome.
# Scope: mission container
un_mission_monthly = {
	save_scope_as = un_msn_this
	change_variable = { name = un_msn_months add = 1 }
	# A mission from an older save, or one opened while all eight display
	# slots were taken, takes the lowest free one.
	un_mission_assign_slot = yes
	var:un_msn_host ?= { save_scope_as = un_msn_host_now }
	# The war the mission opened into is over.
	if = {
		limit = {
			has_variable = un_msn_war_at_open
			var:un_msn_host ?= { is_at_war = no }
		}
		remove_variable = un_msn_war_at_open
	}
	un_mission_prune_contributors = yes
	set_variable = { name = un_msn_strength value = un_mission_strength_value }
	if = {
		# Electoral observers saw an election through: an election campaign
		# in their host ended while they were there
		# (un_mission_electoral_election_held). Closed here, a month at most
		# after the vote, rather than in on_election_campaign_end itself, so
		# Election Interference's cut at that same campaign end still finds the
		# mission whatever order the two on_actions run in.
		# Only while the mission has not lapsed meanwhile: the host still holds
		# its state and, having asked for it, is still in the UN (the lapse
		# test below, in the same form). Otherwise it lapses there.
		limit = {
			has_tag = un_msn_electoral
			has_variable = un_msn_election_held
			exists = var:un_msn_host
			var:un_msn_state ?= { owner ?= scope:un_msn_host_now }
			NOT = {
				AND = {
					has_variable = un_msn_requested
					scope:un_msn_host_now = {
						NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
					}
				}
			}
		}
		un_mission_close = { STATUS = succeeded }
	}
	else_if = {
		# The host is gone or no longer holds the state, or it asked the
		# Assembly for the mission and has since left the UN: the mission has
		# nothing left to do. (A mission sent to a collapse or a famine the
		# docket took up may serve a country outside the UN.)
		limit = {
			OR = {
				# The variable itself, not the saved scope: a scope saved for
				# an earlier mission in this sweep would still exist.
				NOT = { exists = var:un_msn_host }
				NOT = { var:un_msn_state ?= { owner ?= scope:un_msn_host_now } }
				AND = {
					has_variable = un_msn_requested
					scope:un_msn_host_now = {
						NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
					}
				}
				# A force nobody sent (peacekeeping and stabilisation), unless
				# the Standing UN Force serves in it (phase 7).
				AND = {
					un_mission_is_peace_operation = yes
					NOT = { un_mission_force_serves = yes }
					NOT = { has_variable = un_msn_staffed }
					var:un_msn_months >= un_mission_unstaffed_months
				}
				# Electoral observers with no election to watch (phase 7).
				AND = {
					has_tag = un_msn_electoral
					var:un_msn_months >= un_mission_electoral_lapse_months
				}
			}
		}
		un_mission_close = { STATUS = lapsed }
	}
	else_if = {
		# The host has been attacked since the mission opened, or
		# (peacekeeping and stabilisation) every contributor it had has gone
		# home and the Standing UN Force does not serve in it, or (electoral)
		# the host no longer holds elections.
		limit = {
			OR = {
				un_mission_attacked_since_open = yes
				AND = {
					un_mission_is_peace_operation = yes
					NOT = { un_mission_force_serves = yes }
					has_variable = un_msn_staffed
					NOT = { un_lobby_list_has_entries = { LIST = un_msn_contributors } }
				}
				AND = {
					has_tag = un_msn_electoral
					scope:un_msn_host_now = { NOT = { un_mission_holds_elections = yes } }
				}
			}
		}
		# A host that did away with its elections under the observers' eyes
		# answers for it (phase 7, standing reason 46).
		if = {
			limit = {
				has_tag = un_msn_electoral
				scope:un_msn_host_now = { NOT = { un_mission_holds_elections = yes } }
			}
			scope:un_msn_host_now = {
				un_standing_loss = { AMOUNT = un_mission_electoral_abolished_standing REASON = 46 }
			}
		}
		un_mission_close = { STATUS = failed }
	}
	else = {
		if = {
			limit = { has_tag = un_msn_peacekeeping }
			if = {
				limit = { scope:un_msn_host_now = { is_at_war = no } }
				change_variable = { name = un_msn_progress add = un_mission_pk_progress }
			}
		}
		else_if = {
			limit = { has_tag = un_msn_stabilisation }
			if = {
				limit = { scope:un_msn_host_now = { un_mission_host_collapsing = yes } }
				change_variable = { name = un_msn_progress add = un_mission_stab_progress_collapsing }
			}
			else = {
				change_variable = { name = un_msn_progress add = un_mission_stab_progress_recovered }
			}
		}
		else_if = {
			limit = { has_tag = un_msn_aid }
			set_variable = { name = un_msn_progress value = un_mission_aid_progress_value }
			if = {
				limit = {
					var:un_msn_months >= un_mission_aid_min_months
					un_mission_aid_recovered = yes
				}
				set_variable = { name = un_msn_progress value = 100 }
			}
		}
		else_if = {
			# Electoral: half-way while the host is in an election campaign;
			# success comes from the campaign's end (the first branch above).
			limit = { has_tag = un_msn_electoral }
			if = {
				limit = { scope:un_msn_host_now = { in_election_campaign = yes } }
				set_variable = { name = un_msn_progress value = un_mission_electoral_progress_campaign }
			}
			else = {
				set_variable = { name = un_msn_progress value = 0 }
			}
		}
		# No time limit: a mission that is not done goes on, for as long as its
		# contributors stay and its host is not attacked.
		if = {
			limit = { var:un_msn_progress >= 100 }
			un_mission_close = { STATUS = succeeded }
		}
	}
}

# Drop the contributors that have gone: annexed, or out of the UN. The list is
# rebuilt rather than edited while it is iterated.
# Scope: mission container (saved as scope:un_msn_this)
un_mission_prune_contributors = {
	if = {
		limit = { has_variable_list = un_msn_contributors }
		every_in_list = {
			variable = un_msn_contributors
			limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
			scope:un_msn_this = {
				add_to_variable_list = { name = un_msn_contributors_next target = PREV }
			}
		}
		clear_variable_list = un_msn_contributors
		if = {
			limit = { has_variable_list = un_msn_contributors_next }
			every_in_list = {
				variable = un_msn_contributors_next
				scope:un_msn_this = {
					add_to_variable_list = { name = un_msn_contributors target = PREV }
				}
			}
			clear_variable_list = un_msn_contributors_next
		}
	}
}

# ---- Closing ---------------------------------------------------------------------------

# Close the mission as $STATUS$ (succeeded | failed | lapsed). The state's
# marker goes now; its modifier goes at the state's next monthly pulse
# (un_mission_state_refresh, the single refresh site).
# Scope: mission container
un_mission_close = {
	save_scope_as = un_msn_closing
	remove_tag = un_msn_active
	add_tag = un_msn_$STATUS$
	set_variable = { name = un_msn_end_year value = te_history_current_year }
	if = {
		limit = { has_variable = un_msn_slot }
		remove_variable = un_msn_slot
	}
	var:un_msn_state ?= {
		if = {
			limit = { var:un_msn_here ?= scope:un_msn_closing }
			remove_variable = un_msn_here
		}
	}
	un_mission_outcome_$STATUS$ = yes
}

# Success: the UN delivered. A delivery entry, standing and the host's thanks
# for every contributor, and word to all of them. The notices name the
# mission's kind (un_mission_post_succeeded_notice) and its state, through
# scope:un_msn_state_done.
# Scope: mission container
un_mission_outcome_succeeded = {
	un_mission_ledger_outcome = { PILLAR = delivery POINTS = un_mission_success_delivery_value PEACEKEEPING = 21 AID = 22 STABILISATION = 23 ELECTORAL = 37 }
	var:un_msn_state ?= { save_scope_as = un_msn_state_done }
	var:un_msn_host ?= {
		save_scope_as = un_msn_host_done
		un_mission_post_succeeded_notice = yes
		# Electoral observers have no contributors: the host that invited
		# them, and held its election under their eyes, has the standing
		# (phase 7, reason 45).
		if = {
			limit = { scope:un_msn_closing = { has_tag = un_msn_electoral } }
			un_standing_gain = { AMOUNT = un_mission_electoral_host_standing REASON = 45 }
		}
	}
	if = {
		limit = { has_variable_list = un_msn_contributors }
		every_in_list = {
			variable = un_msn_contributors
			un_standing_gain = { AMOUNT = un_mission_success_standing REASON = 34 }
			if = {
				limit = { exists = scope:un_msn_host_done }
				change_relations = { country = scope:un_msn_host_done value = 15 }
			}
			un_mission_post_succeeded_notice = yes
		}
	}
}

# Failure: the Assembly promised and did not deliver. The notices name the

un_mission_outcome_failed = {

			un_mission_post_failed_notice = yes
		}
	}
}

# Tell the country in scope how the mission being closed ended, in the notice
# for its kind: a notification's text cannot branch on the mission's tags, so
# each kind has its own (common/messages/extra_messages.txt). The kind is read
# from scope:un_msn_closing, which un_mission_close saves.
# Scope: the country to tell
un_mission_post_succeeded_notice = {
	if = {
		limit = { scope:un_msn_closing = { has_tag = un_msn_peacekeeping } }
		post_notification = un_mission_peacekeeping_succeeded_notice
	}
	else_if = {
		limit = { scope:un_msn_closing = { has_tag = un_msn_aid } }
		post_notification = un_mission_aid_succeeded_notice
	}
	else_if = {
		limit = { scope:un_msn_closing = { has_tag = un_msn_electoral } }
		post_notification = un_mission_electoral_succeeded_notice
	}
	else = {
		post_notification = un_mission_stabilisation_succeeded_notice
	}
}

# As un_mission_post_succeeded_notice, for a mission that failed.
# Scope: the country to tell
un_mission_post_failed_notice = {
	if = {
		limit = { scope:un_msn_closing = { has_tag = un_msn_peacekeeping } }
		post_notification = un_mission_peacekeeping_failed_notice
	}
	else_if = {
		limit = { scope:un_msn_closing = { has_tag = un_msn_aid } }
		post_notification = un_mission_aid_failed_notice
	}
	else_if = {
		limit = { scope:un_msn_closing = { has_tag = un_msn_electoral } }
		post_notification = un_mission_electoral_failed_notice
	}
	else = {
		post_notification = un_mission_stabilisation_failed_notice
	}
}

# The ledger entry for an outcome: one reason code per kind of mission, and
# the mission's state as its subject, so the log reads "the peacekeeping
# mission in Attica" and the state can be hovered.
# Parameters: $PILLAR$, $POINTS$, and the code for each kind
# (PEACEKEEPING, AID, STABILISATION, ELECTORAL)
# Scope: mission container
un_mission_ledger_outcome = {
	if = {
		limit = { has_tag = un_msn_peacekeeping }
		un_ledger_entry_on = { PILLAR = $PILLAR$ POINTS = $POINTS$ REASON = $PEACEKEEPING$ SUBJECT = var:un_msn_state }
	}
	else_if = {
		limit = { has_tag = un_msn_aid }
		un_ledger_entry_on = { PILLAR = $PILLAR$ POINTS = $POINTS$ REASON = $AID$ SUBJECT = var:un_msn_state }
	}
	else_if = {
		limit = { has_tag = un_msn_electoral }
		un_ledger_entry_on = { PILLAR = $PILLAR$ POINTS = $POINTS$ REASON = $ELECTORAL$ SUBJECT = var:un_msn_state }
	}
	else = {
		un_ledger_entry_on = { PILLAR = $PILLAR$ POINTS = $POINTS$ REASON = $STABILISATION$ SUBJECT = var:un_msn_state }
	}
}

# Lapsed: the mission had nothing left to do (its host or state is gone, or
# nobody joined a force the Assembly voted for, or electoral observers waited
# five years for an election). No entry.
# Scope: mission container
un_mission_outcome_lapsed = {
	if = {
		limit = { has_tag = un_msn_electoral }
		var:un_msn_host ?= { post_notification = un_mission_electoral_lapsed_notice }
	}
	else = {
		var:un_msn_host ?= { post_notification = un_mission_lapsed_notice }
	}
}

# A dissolution ends every mission, without entries: there is no Assembly
# left to credit or blame.
# Scope: any
un_missions_dissolve = {
	if = {
		limit = { has_global_variable_list = un_mission_registry }
		every_in_global_list = {
			variable = un_mission_registry
			limit = { has_tag = un_msn_active }
			save_scope_as = un_msn_closing
			remove_tag = un_msn_active
			add_tag = un_msn_lapsed
			set_variable = { name = un_msn_end_year value = te_history_current_year }
			if = {
				limit = { has_variable = un_msn_slot }
				remove_variable = un_msn_slot
			}
			var:un_msn_state ?= {
				if = {
					limit = { var:un_msn_here ?= scope:un_msn_closing }
					remove_variable = un_msn_here
				}
			}
		}
	}
}

# ---- The state -----------------------------------------------------------------------------

# The mission's effect on its state: the type's modifier, multiplied by the
# mission's strength. Removed once the state holds no active mission.
# The single refresh site for un_mission_peacekeeping_state_modifier,
# un_mission_aid_state_modifier, un_mission_stabilisation_state_modifier and
# (phase 7) un_mission_electoral_state_modifier.
# The multiplier reads var:un_msn_power on the state, which is removed only
# after the modifiers are.
# Scope: state (from on_monthly_pulse_state, un_mission_on_actions.txt)
un_mission_state_refresh = {
	if = {
		limit = { var:un_msn_here ?= { has_tag = un_msn_active } }
		set_variable = { name = un_msn_power value = var:un_msn_here.var:un_msn_strength }
		un_mission_state_clear_modifiers = yes
		if = {
			limit = { var:un_msn_here = { has_tag = un_msn_peacekeeping } }
			add_modifier = {
				name = un_mission_peacekeeping_state_modifier
				multiplier = var:un_msn_power
			}
		}
		else_if = {
			limit = { var:un_msn_here = { has_tag = un_msn_aid } }
			add_modifier = {
				name = un_mission_aid_state_modifier
				multiplier = var:un_msn_power
			}
		}
		else_if = {
			limit = { var:un_msn_here = { has_tag = un_msn_stabilisation } }
			add_modifier = {
				name = un_mission_stabilisation_state_modifier
				multiplier = var:un_msn_power
			}
		}
		else_if = {
			limit = { var:un_msn_here = { has_tag = un_msn_electoral } }
			add_modifier = {
				name = un_mission_electoral_state_modifier
				multiplier = var:un_msn_power
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_variable = un_msn_here
				has_variable = un_msn_power
			}
		}
		# The modifiers go before the variable their multiplier reads
		# (written out here, not through un_mission_state_clear_modifiers, so
		# modifier_multiplier_var_audit can see the order).
		if = {
			limit = { has_modifier = un_mission_peacekeeping_state_modifier }
			remove_modifier = un_mission_peacekeeping_state_modifier
		}
		if = {
			limit = { has_modifier = un_mission_aid_state_modifier }
			remove_modifier = un_mission_aid_state_modifier
		}
		if = {
			limit = { has_modifier = un_mission_stabilisation_state_modifier }
			remove_modifier = un_mission_stabilisation_state_modifier
		}
		if = {
			limit = { has_modifier = un_mission_electoral_state_modifier }
			remove_modifier = un_mission_electoral_state_modifier
		}
		if = {
			limit = { has_variable = un_msn_here }
			remove_variable = un_msn_here
		}
		if = {
			limit = { has_variable = un_msn_power }
			remove_variable = un_msn_power
		}
	}
}

# Scope: state
un_mission_state_clear_modifiers = {
	if = {
		limit = { has_modifier = un_mission_peacekeeping_state_modifier }
		remove_modifier = un_mission_peacekeeping_state_modifier
	}
	if = {
		limit = { has_modifier = un_mission_aid_state_modifier }
		remove_modifier = un_mission_aid_state_modifier
	}
	if = {
		limit = { has_modifier = un_mission_stabilisation_state_modifier }
		remove_modifier = un_mission_stabilisation_state_modifier
	}
	if = {
		limit = { has_modifier = un_mission_electoral_state_modifier }
		remove_modifier = un_mission_electoral_state_modifier
	}
}

# ---- Electoral observers (redesign phase 7, §0.12 item 11) ---------------------------
# A member that holds elections asks the Assembly for observers at its next
# one (topic observer_request: a request, like aid; recommendatory, a simple
# majority, Charter Reform I). Carried, it opens the fourth kind of mission,
# electoral (type 4), in the requester's capital. The observers are paid from
# the UN's budget: the mission has no contributors, takes no volunteers and
# never lapses unstaffed. It succeeds when an election campaign in its host
# ends while it runs, fails if the host is attacked or stops holding
# elections, and lapses after un_mission_electoral_lapse_months without one.
# While it runs: un_mission_electoral_state_modifier on its state (through
# un_mission_state_refresh), un_electoral_observers_modifier on the host
# (un_mission_electoral_host_refresh), both x strength, and Election
# Interference's cut to the host's electoral confidence is halved
# (covert_op_election_confidence_effect).

# Open the electoral mission for scope:un_msn_host: in its capital, or, if the
# capital already holds another kind of mission, in its most populous state
# without one. Leaves scope:un_msn_new pointing at the mission (unset if every
# state of the host already holds another kind).
# Scope: any; expects scope:un_msn_host
un_mission_open_electoral = {
	scope:un_msn_host = {
		capital ?= {
			if = {
				limit = { NOT = { un_mission_state_occupied = yes } }
				save_scope_as = un_msn_electoral_capital
			}
		}
	}
	if = {
		limit = { exists = scope:un_msn_electoral_capital }
		scope:un_msn_electoral_capital = { save_scope_as = un_msn_state }
		un_mission_open = { TYPE = electoral CODE = 4 }
	}
	else = {
		un_mission_join_host = { TYPE = electoral CODE = 4 ORDER = un_docket_state_pop_order }
	}
}

# An election campaign in the scoped country has ended (on_election_campaign_end,
# un_mission_on_actions.txt). Its active electoral mission saw it through: it
# is marked, and succeeds at the next monthly update (un_mission_monthly).
# Scope: country (the host)
un_mission_electoral_election_held = {
	save_scope_as = un_msn_voting_host
	if = {
		limit = { has_global_variable_list = un_mission_registry }
		every_in_global_list = {
			variable = un_mission_registry
			limit = {
				has_tag = un_msn_active
				has_tag = un_msn_electoral
				var:un_msn_host ?= scope:un_msn_voting_host
			}
			set_variable = un_msn_election_held
			set_variable = { name = un_msn_progress value = 100 }
		}
	}
}

# The observers' effect on the host country: un_electoral_observers_modifier x
# the strength of the electoral mission it hosts. The single refresh site for
# that modifier, from the country monthly pulse (un_mission_on_actions.txt),
# whose ROOT is the host, so the multiplier resolves against it. Re-applied
# every month from the mission register, so a revolution's winner, which
# inherits no modifiers, has it back within a month (§0.9). The multiplier's
# variable is removed only after the modifier is.
# Scope: country (ROOT)
un_mission_electoral_host_refresh = {
	save_scope_as = un_msn_observed_host
	if = {
		limit = {
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			un_mission_hosts = { TYPE = electoral }
		}
		random_in_global_list = {
			variable = un_mission_registry
			limit = {
				has_tag = un_msn_active
				has_tag = un_msn_electoral
				var:un_msn_host ?= scope:un_msn_observed_host
			}
			save_scope_as = un_msn_observing
		}
		set_variable = { name = un_msn_electoral_power value = scope:un_msn_observing.var:un_msn_strength }
		if = {
			limit = { has_modifier = un_electoral_observers_modifier }
			remove_modifier = un_electoral_observers_modifier
		}
		add_modifier = {
			name = un_electoral_observers_modifier
			multiplier = var:un_msn_electoral_power
		}
	}
	else = {
		if = {
			limit = { has_modifier = un_electoral_observers_modifier }
			remove_modifier = un_electoral_observers_modifier
		}
		if = {
			limit = { has_variable = un_msn_electoral_power }
			remove_variable = un_msn_electoral_power
		}
	}
}

# ---- The topic (the hooks un_charter_effects.txt and the chamber call) -----------------

# The state the observers would go to, for the chamber's preview: the capital,
# or the most populous state without a mission. Leaves
# scope:un_observer_request_state (unset if every state holds another kind).
# The same choice un_mission_open_electoral makes when the vote carries.
# Scope: the requesting country
un_observer_request_select_state = {
	if = {
		limit = { capital ?= { NOT = { un_mission_state_occupied = yes } } }
		capital = { save_temporary_scope_as = un_observer_request_state }
	}
	else = {
		ordered_scope_state = {
			limit = { NOT = { un_mission_state_occupied = yes } }
			order_by = un_docket_state_pop_order
			max = 1
			save_temporary_scope_as = un_observer_request_state
		}
	}
}

# What the chamber's row previews while a request can be made: where the
# observers would go.
# Scope: country (the viewer, the would-be requester)
un_observer_request_chamber_preview = {
	un_observer_request_select_state = yes
	if = {
		limit = { exists = scope:un_observer_request_state }
		custom_tooltip_no_bullet = je_un_chamber_propose_preview_observer_request
	}
}

# The outcome: the request carried (it cannot be vetoed). Observers go to the
# requester's capital, unless it no longer holds elections, in which case there
# is nothing to observe, or it has gone to war since it tabled the request:
# observers are not sent into a war (un_observer_request_can_table asks the
# same at tabling), and one they arrived in would never fail them
# (un_msn_war_at_open). Either way nothing is sent. Booked as the aid request
# is: UN credibility +1 under the resolution code.
# Scope: un_vote.2's ROOT (the proposer, the requester); expects
# scope:un_resolution and scope:un_vote_proposer
un_observer_request_on_carried = {
	custom_tooltip = un_vote_observer_request_passed_tt
	un_ledger_entry = { PILLAR = credibility POINTS = 1 REASON = 1 }
	if = {
		limit = { exists = scope:un_vote_proposer }
		if = {
			limit = {
				scope:un_vote_proposer = {
					is_at_war = no
					un_mission_holds_elections = yes
					un_mission_can_join_host = { TYPE = electoral }
				}
			}
			custom_tooltip = un_mission_electoral_opened_tt
			hidden_effect = {
				scope:un_vote_proposer = { save_scope_as = un_msn_host }
				un_mission_open_electoral = yes
				if = {
					limit = { exists = scope:un_msn_new }
					scope:un_resolution ?= { set_variable = { name = un_res_mission value = scope:un_msn_new } }
					scope:un_msn_new = { set_variable = un_msn_requested }
				}
			}
		}
		else = {
			custom_tooltip = un_mission_electoral_fallback_tt
		}
	}
}
