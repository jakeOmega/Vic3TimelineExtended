# ============================================================================
# THE STANDING UN FORCE — THE WRITERS (redesign phase 7, §0.12 item 2)
# ============================================================================
# docs/systems/un_redesign_design.md §0.12. The "Standing Mandate Force" half of
# Charter Reform I: a one-time founding resolution (topic standing_force,
# binding, a simple majority, blocked outright if vetoed) tabled by a major
# power. Its in-force flag is the global un_inst_standing_force (an
# institution, not a convention: un_force_in_force, un_force_triggers.txt).
#
# WHILE IT STANDS
#   missions   a peacekeeping or stabilisation mission counts the Force as one
#              contributor in its strength (un_mission_strength_heads), never
#              lapses for want of volunteers and never fails because its last
#              contributor went home (un_mission_monthly reads
#              un_mission_force_serves; un_mission_*.txt)
#   budget     the Force takes un_force_budget_share of the dues
#              (un_charter_values.txt; the lending facility lends from what is
#              left, and the chamber's budget line shows it)
#   war goals  at Supranational a represented MEMBER's war goal against a
#              country hosting an active peacekeeping or stabilisation mission
#              is struck on adding (un_force_strike_war_goals, called by
#              un_teeth_war_goal_surcharge from on_wargoal_added, after the
#              mandate's own hooks), except a goal of the type its play opened
#              with, which pays the surcharge doubled, as from Strong; a
#              non-member, or a member whose credentials are suspended, pays
#              the doubled surcharge for every goal (owner decision
#              2026-10-02: prohibited for members only; review 2026-10-02:
#              the opening goal is charged, never struck)
#
# WHO WRITES WHAT
#   un_standing_force_on_carried     un_vote.2's carried branch (through
#                                     un_charter_on_carried): sets the flag
#   un_force_on_dissolve              un_dissolve (through un_charter_on_dissolve):
#                                     clears it
#   un_force_strike_war_goals         un_teeth_war_goal_surcharge
#
# REASON CODES (phase 7, missions block: ledger 37-39, standing 45-47)
#   The founding books credibility +1 under the resolution code 1, like every
#   carried decision. The mission codes are the electoral observer mission's
#   (un_mission_effects.txt): ledger 37 accomplished, 38 failed, each naming
#   the mission's state; standing 45 (an election held under the observers),
#   46 (the host did away with its elections while they watched). Ledger 39
#   and standing 47 are unused.
# ============================================================================

# ---- The topic ----------------------------------------------------------------------

# The outcome. Carried without a veto (or over one, under veto restraint): the
# Force is founded. Vetoed: blocked outright; the framework's outcome line
# already says so, and nothing is booked.
# Scope: un_vote.2's ROOT (the proposer); expects scope:un_resolution
un_standing_force_on_carried = {
	if = {
		limit = { scope:un_resolution ?= { NOT = { has_tag = un_res_vetoed } } }
		custom_tooltip = un_vote_standing_force_passed_tt
		un_ledger_entry = { PILLAR = credibility POINTS = 1 REASON = 1 }
		hidden_effect = {
			if = {
				limit = { NOT = { un_force_in_force = yes } }
				set_global_variable = un_inst_standing_force
			}
		}
	}
	else = {
		custom_tooltip = un_vote_standing_force_passed_vetoed_tt
		# No effect: there is no lesser form of a standing force.
	}
}

# What the chamber's row previews while the Force can be founded: what it
# would cost the budget now.
# Scope: country (the viewer)
un_standing_force_chamber_preview = {
	custom_tooltip_no_bullet = je_un_chamber_propose_preview_standing_force
}

# ---- War goals against a mission's host ------------------------------------------

# Strike the goals the initiator holds against the play's target, which hosts
# an active peacekeeping or stabilisation mission, and tell a human initiator
# why. Called only where un_force_war_goal_barred holds, so at least one goes.
# Never a goal of the type the play opened with (un_force_has_strikeable_goal
# says why and lists the types): when that is the goal just added,
# un_force_war_goal_barred fails and un_teeth_war_goal_surcharge charges it the
# surcharge, doubled against the host. Like un_mandate_strike_goal_type,
# remove_war_goal addresses a goal by side and type, so a goal of the same type
# against another country in the same play would go too.
# Scope: diplomatic play; expects scope:actor, the initiator (on_wargoal_added)
un_force_strike_war_goals = {
	target ?= { save_scope_as = un_force_protected }
	if = {
		limit = { exists = scope:un_force_protected }
		# One opening play: struck unless this play is of that type.
		un_force_strike_goal_type = { TYPE = conquer_state PLAY = dp_conquer_state }
		un_force_strike_goal_type = { TYPE = return_state PLAY = dp_return_state }
		un_force_strike_goal_type = { TYPE = take_treaty_port PLAY = dp_take_treaty_port }
		un_force_strike_goal_type = { TYPE = liberate_country PLAY = dp_liberate_country }
		un_force_strike_goal_type = { TYPE = liberate_subject PLAY = dp_liberate_subject }
		un_force_strike_goal_type = { TYPE = secession PLAY = dp_secession }
		un_force_strike_goal_type = { TYPE = restore_union PLAY = dp_restore_union }
		un_force_strike_goal_type = { TYPE = make_protectorate PLAY = dp_make_protectorate }
		un_force_strike_goal_type = { TYPE = make_tributary PLAY = dp_make_tributary }
		un_force_strike_goal_type = { TYPE = make_dominion PLAY = dp_make_dominion }
		un_force_strike_goal_type = { TYPE = release_as_subject PLAY = dp_release_as_subject }
		un_force_strike_goal_type = { TYPE = transfer_subject PLAY = dp_transfer_subject }
		un_force_strike_goal_type = { TYPE = reduce_autonomy PLAY = dp_reduce_autonomy }
		un_force_strike_goal_type = { TYPE = regime_change PLAY = dp_regime_change }
		un_force_strike_goal_type = { TYPE = contain_threat PLAY = dp_contain_threat }
		un_force_strike_goal_type = { TYPE = enforce_treaty_article PLAY = dp_default_treaty_article }
		un_force_strike_goal_type = { TYPE = open_market PLAY = dp_open_market }
		un_force_strike_goal_type = { TYPE = ban_slavery PLAY = dp_ban_slavery }
		un_force_strike_goal_type = { TYPE = colonization_rights PLAY = dp_native_uprising }
		un_force_strike_goal_type = { TYPE = force_nationalization PLAY = dp_force_nationalization }
		un_force_strike_goal_type = { TYPE = foreign_investment_rights PLAY = dp_give_foreign_investment_rights }
		un_force_strike_goal_type = { TYPE = join_power_bloc PLAY = dp_join_power_bloc }
		un_force_strike_goal_type = { TYPE = leave_power_bloc PLAY = dp_leave_power_bloc }
		un_force_strike_goal_type = { TYPE = demand_strait_access PLAY = dp_demand_strait_access }
		un_force_strike_goal_type = { TYPE = demand_toll_exemption PLAY = dp_demand_toll_exemption }
		un_force_strike_goal_type = { TYPE = demand_no_tolls PLAY = dp_demand_no_tolls }
		un_force_strike_goal_type = { TYPE = demand_no_strait_closure PLAY = dp_demand_no_strait_closure }
		# Several opening plays. unification and unification_leadership are
		# never struck (un_force_has_strikeable_goal).
		if = {
			limit = { NOT = { un_force_play_opens_with_annexation = yes } }
			un_force_strike_goal = { TYPE = annex_country }
		}
		if = {
			limit = { NOT = { un_force_play_opens_with_humiliation = yes } }
			un_force_strike_goal = { TYPE = humiliation }
		}
		# No opening play.
		un_force_strike_goal = { TYPE = te_reunify_country }
		un_force_strike_goal = { TYPE = make_personal_union }
		un_force_strike_goal = { TYPE = make_crown_land }
		un_force_strike_goal = { TYPE = make_chartered_company }
		scope:actor = {
			if = {
				limit = { is_ai = no }
				post_notification = un_force_war_goal_struck_notice
			}
		}
	}
}

# Strike the initiator's goal of $TYPE$ unless the play is of type $PLAY$, the
# one that opens with it.
# Parameters: $TYPE$ = war goal type; $PLAY$ = the diplomatic play type whose
# opening goal it is
# Scope: diplomatic play; expects scope:actor
un_force_strike_goal_type = {
	if = {
		limit = { un_force_goal_strikeable = { TYPE = $TYPE$ PLAY = $PLAY$ } }
		remove_war_goal = { who = initiator type = $TYPE$ }
	}
}

# Strike the initiator's goal of $TYPE$, which no play opens with (or, for the
# callers above, not this one).
# Parameters: $TYPE$ = war goal type
# Scope: diplomatic play; expects scope:actor
un_force_strike_goal = {
	if = {
		limit = { un_force_goal_held = { TYPE = $TYPE$ } }
		remove_war_goal = { who = initiator type = $TYPE$ }
	}
}

# ---- Dissolution ------------------------------------------------------------------

# The UN dissolves (un_charter_on_dissolve, after every country has left and
# every mission has ended): the Force goes with it. A refounded UN must found
# it again, under a charter that has been reset.
# Scope: none
un_force_on_dissolve = {
	if = {
		limit = { un_force_in_force = yes }
		remove_global_variable = un_inst_standing_force
	}
}
