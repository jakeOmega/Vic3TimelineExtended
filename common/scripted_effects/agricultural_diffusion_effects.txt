# Agricultural Diffusion Effects
#
# Models global propagation of transformative agricultural techniques:
# the first country to research a diffusion-eligible tech triggers a global
# event applying a permanent state_arable_land_mult modifier to every country
# (including itself), representing how Haber-Bosch / Green Revolution /
# biotech adoption spread beyond their inventors via FAO/CIMMYT/IRRI/USAID.
# The first researcher additionally gets agdiff_first_mover_prestige.
#
# Naming follows the world_first_* / is_world_first_* / first_*_country
# pattern established by nuclear weapons (the on-complete block of
# common/journal_entries/je_nuclear_program.txt).
#
# Diffusion-eligible techs: nitrogen_fixation, modern_chemical_processes,
# green_revolution, gene_splicing, biotechnology.
#
# Wired in via on_acquired_technology in common/on_actions/extra_on_actions.txt.
# Countries the broadcast misses (new tags, and the rebels in a civil war) are
# caught by the backfill below, and a civil war's winner is repaired by
# agdiff_repair_after_civil_war at the end of this file.

# --- Dispatcher (called from agricultural_diffusion_on_action) --------------
# Guard + dispatch in one parameterized block: if $TECH$ is researched and
# no country has yet claimed the world-first, call the per-tech wrapper.
# Replaces five back-to-back inline if-blocks at the on-action call site.
# The wrapper effect name agdiff_on_$TECH$_researched is built via parse-time
# text substitution — same mechanism agdiff_handle_tech_acquired_effect uses
# for $EVENT$.
agdiff_dispatch_if_first = {
	if = {
		limit = {
			has_technology_researched = $TECH$
			NOT = { has_global_variable = world_first_$TECH$ }
		}
		agdiff_on_$TECH$_researched = yes
	}
}

# --- Per-tech entry points --------------------------------------------------
# One named helper per tech so the on_acquired_technology hook can dispatch
# with a `has_technology_researched` guard without needing $TECH$-keyed
# has_global_variable lookups (which the parser handles fine but greps poorly).

agdiff_on_nitrogen_fixation_researched = {
	agdiff_handle_tech_acquired_effect = {
		TECH = nitrogen_fixation
		EVENT = agricultural_diffusion_events.1
	}
}

agdiff_on_modern_chemical_processes_researched = {
	agdiff_handle_tech_acquired_effect = {
		TECH = modern_chemical_processes
		EVENT = agricultural_diffusion_events.2
	}
}

agdiff_on_green_revolution_researched = {
	agdiff_handle_tech_acquired_effect = {
		TECH = green_revolution
		EVENT = agricultural_diffusion_events.3
	}
}

agdiff_on_gene_splicing_researched = {
	agdiff_handle_tech_acquired_effect = {
		TECH = gene_splicing
		EVENT = agricultural_diffusion_events.4
	}
}

agdiff_on_biotechnology_researched = {
	agdiff_handle_tech_acquired_effect = {
		TECH = biotechnology
		EVENT = agricultural_diffusion_events.5
	}
}

# --- Core dispatch ----------------------------------------------------------
# Called by the per-tech helpers above. Only the first researcher trips the
# global-first branch; subsequent researchers get the local tech modifier
# (already attached by the tech entity) plus whatever diffusion modifier they
# previously received when the first researcher fired this effect.
#
# Modifier application lives in the event OPTIONS (agricultural_diffusion_events),
# not here: only option-level add_modifier renders in the event tooltip, so a
# broadcast-then-notify split would read as a no-op to the player. This effect
# only sets the world-first flags (which gate the events' triggered_desc /
# option triggers) and fires the events. The first researcher gets the event
# immediately and applies the diffusion + first-mover-prestige modifiers via its
# .a_first option; every other recognized country gets the event after 14 days
# and applies just the diffusion modifier via its .a option.
agdiff_handle_tech_acquired_effect = {
	if = {
		limit = { NOT = { has_global_variable = world_first_$TECH$ } }
		set_global_variable = { name = world_first_$TECH$ value = yes }
		set_global_variable = { name = first_$TECH$_country value = ROOT }
		set_variable = { name = is_world_first_$TECH$ value = yes }
		# The month the first-mover reward starts, for
		# agdiff_restore_first_mover_prestige: the title above is permanent
		# but the reward is a 20-year decaying modifier, so a revolution's
		# winner needs to know how much of it is left. Recorded here, beside
		# the title, rather than in the event's .a_first option that grants
		# the modifier: that option is the only one open to the first
		# researcher, and the event fires on it at once. One variable for all
		# five techs, like the modifier: a later title refreshes both.
		set_variable = { name = agdiff_first_mover_month value = te_history_month_index }
		# Country-event narrative: same event ID to first researcher and
		# others; the event uses triggered_desc on is_world_first_$TECH$ to
		# branch between first-mover and notification flavor, and applies the
		# diffusion / prestige modifiers in its options (see above).
		trigger_event = { id = $EVENT$ }
		every_country = {
			limit = {
				NOT = { THIS = ROOT }
				is_country_type = recognized
			}
			trigger_event = { id = $EVENT$ days = 14 }
		}
	}
}

# --- Backfill for countries the broadcast missed ----------------------------
# Countries that come into existence AFTER the first researcher (decolonization,
# releases, formables, secession) miss the every_country broadcast in
# agdiff_handle_tech_acquired_effect, and so do the rebels in a revolution: a
# second country object with the same tag, which starts with no modifiers.
# This catch-up effect adds any diffusion modifier whose global-first flag is
# already set but which isn't yet on the country. It reads only global flags
# and the country's own modifiers, so it is safe to run on any country at any
# time, as often as you like.
#
# Wired into (common/on_actions/extra_on_actions.txt, agricultural diffusion
# section):
#   on_country_formed                           ROOT = the new country
#   on_country_released_as_independent          scope:target = released country
#   on_country_released_as_own_subject          scope:target = released country
#   on_country_released_as_overlord_subject     scope:target = released country
#   on_country_released_as_company_subject      scope:target = released country
#   on_revolution_start, on_secession_start     scope:target = uprising country
#   on_civil_war_won                            ROOT = the winner, through
#                                               agdiff_repair_after_civil_war
#
# The rebels get it when the uprising starts, not only when they win: they
# are the object that survives a rebel victory, and a revolution's winner
# inherits none of the loser's modifiers (docs/guides/scripting_best_practices.md
# § "What a Civil War's Winner Inherits"; in the German-revolution saves the
# winner held none of the five). The on_civil_war_won pass is the backstop.
#
# Without this, decolonized 1960+ African nations (the exact target of issue
# #114) would get zero benefit from the Green Revolution diffusion that fired
# years before they existed.
# Backfill one tech: if the world-first flag is set but this country lacks
# the diffusion modifier, apply it. The agdiff_$TECH$ modifier name follows
# the convention established by the five per-tech wrappers above.
agdiff_backfill_one_tech = {
	if = {
		limit = {
			has_global_variable = world_first_$TECH$
			NOT = { has_modifier = agdiff_$TECH$ }
		}
		add_modifier = { name = agdiff_$TECH$ }
	}
}

agdiff_backfill_diffusion_for_country = {
	agdiff_backfill_one_tech = { TECH = nitrogen_fixation }
	agdiff_backfill_one_tech = { TECH = modern_chemical_processes }
	agdiff_backfill_one_tech = { TECH = green_revolution }
	agdiff_backfill_one_tech = { TECH = gene_splicing }
	agdiff_backfill_one_tech = { TECH = biotechnology }
}

# --- After a civil war ------------------------------------------------------
# A revolution's winner inherits the loser's variables but none of its
# modifiers (docs/guides/scripting_best_practices.md § "What a Civil War's
# Winner Inherits"). When the rebels win, the loyalist object dies holding the
# five diffusion modifiers and, if it was a first researcher, the first-mover
# reward. The winner continues the nation, so it gets both back.
#
# Scope: the winner, which must also be ROOT (called bare from
# agdiff_on_civil_war_won; on_civil_war_won runs with ROOT = the winner, after
# the loser's variables have merged in). Every step is a no-op for a winner
# that still holds what it restores, which is what a loyalist victory
# presumably leaves: the original keeps its own modifiers.
agdiff_repair_after_civil_war = {
	agdiff_backfill_diffusion_for_country = yes
	agdiff_restore_first_mover_prestige = yes
	agdiff_repoint_first_country = { TECH = nitrogen_fixation }
	agdiff_repoint_first_country = { TECH = modern_chemical_processes }
	agdiff_repoint_first_country = { TECH = green_revolution }
	agdiff_repoint_first_country = { TECH = gene_splicing }
	agdiff_repoint_first_country = { TECH = biotechnology }
}

# The first-mover reward. is_world_first_<tech> reaches the winner, but it is
# permanent and the reward is a 20-year decaying modifier, so keying on the
# title alone would hand a fresh 20 years to a country whose reward ran out
# decades ago. agdiff_first_mover_month (recorded with the title, and inherited
# like it) says how much is left, and the reward comes back only while some is.
#
# The restored modifier starts at the strength the lost one had decayed to
# (multiplier = the fraction of the 20 years still to run), so the number the
# player was watching carries straight across the win. It then fades over a
# full very_long_modifier_time instead of over what was left, so it fades more
# slowly than the original would have. A duration computed from a script value
# would end it on the original date, but no add_modifier in vanilla or this mod
# takes a non-constant duration, and one the engine ignored would leave the
# modifier permanent.
#
# A title won before agdiff_first_mover_month existed has no record and is not
# restored: there is no way to tell whether its 20 years have run out.
#
# Scope: a country, which must also be ROOT: add_modifier's multiplier is
# resolved against ROOT (docs/guides/scripting_best_practices.md
# § "`add_modifier { multiplier = var:X }` Resolves Against ROOT, Not the
# Scope It Runs In"). `root ?= this` makes any other call site a no-op rather
# than a restore at another country's strength.
agdiff_restore_first_mover_prestige = {
	if = {
		limit = {
			root ?= this
			has_variable = agdiff_first_mover_month
			NOT = { has_modifier = agdiff_first_mover_prestige }
			agdiff_first_mover_months_left > 0
		}
		add_modifier = {
			name = agdiff_first_mover_prestige
			days = very_long_modifier_time
			multiplier = agdiff_first_mover_strength_left
			is_decaying = yes
		}
	}
}

# Re-point global first_$TECH$_country, the originator the diffusion events
# name, at the winner when it points at the loser. Only the five .d loc strings
# in te_events_l_english.yml read it, in the notification every other country
# gets 14 days after the first research, so this matters only when a revolution
# ends inside that window. The pointer moves only when it names a dead country
# with the winner's own definition, i.e. this war's loser; a loyalist winner
# that holds the title is alive, and keeps it.
#
# Untested: whether the dead loser can still be read here (it is still in the
# save just after on_civil_war_won). If it can't, the ?= makes this a no-op.
#
# Scope: the winner, as ROOT.
agdiff_repoint_first_country = {
	if = {
		limit = {
			global_var:first_$TECH$_country ?= {
				is_country_alive = no
				country_definition = root.country_definition
			}
		}
		set_global_variable = { name = first_$TECH$_country value = ROOT }
	}
}
