# Legislated tax code: drafts, bills and passage (docs/systems/tax_code_schema.md,
# "Draft, bill and passage").
#
# One editable draft (te_tax_dr_*) and at most one bill under debate
# (te_tax_bl_*) per country. The draft changes nothing; introducing it copies it
# into the bill and starts debate; interest groups commit to the bill's
# revision through the support refresh; passing it stores an approved package
# in a free slot (te_tax_store_package), which the scheduler commences on its
# due month. Withdrawing a bill or discarding a draft leaves the enacted code
# and the approved packages exactly as they were.
#
# Every command te_tax_cmd_<c> runs only inside
# `if = { limit = { te_tax_can_<c> ... } }` (te_tax_triggers.txt), the same
# trigger a GUI button uses as its is_valid, and says what it does with a
# custom_tooltip, because variable writes show nothing in a tooltip. Country
# scope; nothing here reads ROOT. Commands are effects for clicks and AI
# decisions, never for a GUI's is_shown, is_valid or display values.
#
# The per-instrument, per-good and per-group parts are generated
# (te_tax_generated_bill_effects.txt, gen_tax_code.py).

# Opens a draft: a copy of the bill under debate if there is one (to revise
# it), else every provision untouched and commencement three months on.
te_tax_cmd_draft_new = {
	if = {
		limit = { te_tax_can_draft_new = yes }
		custom_tooltip = te_tax_tt_cmd_draft_new
		if = {
			limit = { te_tax_bill_active = yes }
			te_tax_gen_draft_from_bill = yes
		}
		else = {
			te_tax_gen_draft_init = yes
			set_variable = { name = te_tax_dr_due value = te_tax_due_default }
		}
		set_variable = { name = te_tax_dr_on value = 1 }
	}
}

# Moves one instrument's target in the draft. KEY: wage, div, land, head, cons.
# DIR: 0 one step down, 1 one step up, 2 zero, 3 the highest index, 4 leave the
# provision alone. An untouched provision starts from its baseline (the rate in
# force under existing law in the draft's due month, te_tax_base_dr_<key>).
te_tax_cmd_draft_step = {
	if = {
		limit = { te_tax_can_draft_step = { KEY = $KEY$ DIR = $DIR$ } }
		te_tax_dr_step_$DIR$ = { KEY = $KEY$ }
	}
}

# Records the provision's planned version when the draft first touches it, so
# a later approval of the same provision shows as a changed baseline.
te_tax_dr_touch = {
	if = {
		limit = { var:te_tax_dr_$KEY$ < 0 }
		set_variable = { name = te_tax_dr_$KEY$_pver value = var:te_tax_pver_$KEY$ }
	}
}

te_tax_dr_step_0 = {
	custom_tooltip = te_tax_tt_cmd_draft_step_down
	te_tax_dr_touch = { KEY = $KEY$ }
	set_variable = { name = te_tax_dr_$KEY$ value = te_tax_dr_eff_$KEY$ }
	change_variable = { name = te_tax_dr_$KEY$ add = -1 }
	clamp_variable = { name = te_tax_dr_$KEY$ min = 0 max = te_tax_max_$KEY$ }
}

# Toggles a catalog good in the draft: an untouched good flips from its
# baseline (taxed under existing law in the due month, or not); a touched good
# returns to untouched. The goods' planned version is recorded when the first
# good is touched and cleared when the last is untouched.
te_tax_cmd_draft_good = {
	if = {
		limit = { te_tax_can_draft_good = { GOOD = $GOOD$ } }
		custom_tooltip = te_tax_tt_cmd_draft_good
		if = {
			limit = { var:te_tax_dr_g_$GOOD$ >= 0 }
			set_variable = { name = te_tax_dr_g_$GOOD$ value = -1 }
			if = {
				limit = { NOT = { te_tax_gen_draft_touches_goods = yes } }
				set_variable = { name = te_tax_dr_goods_pver value = -1 }
			}
		}
		else = {
			if = {
				limit = { NOT = { te_tax_gen_draft_touches_goods = yes } }
				set_variable = { name = te_tax_dr_goods_pver value = var:te_tax_pver_goods }
			}
			if = {
				limit = { te_tax_base_dr_g_$GOOD$ > 0 }
				set_variable = { name = te_tax_dr_g_$GOOD$ value = 0 }
			}
			else = {
				set_variable = { name = te_tax_dr_g_$GOOD$ value = 1 }
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Relief (plan Task 11; docs/systems/tax_code_schema.md, "Relief in a bill").
# Agricultural wage relief and regional relief are bands, 0 none, 1 = -25%,
# 2 = -50% (te_tax_max_<key>). Regional relief also names its states, at most
# three, in the country list te_tax_dr_relief_states. The relief group's
# planned version (te_tax_dr_relief_pver) is recorded when the draft first
# changes either relief and cleared when it changes neither again.
# ---------------------------------------------------------------------------

# Moves one relief in the draft. KEY: agrel or regrel. DIR: 0 one band less,
# 1 one band more, 2 none, 3 the deepest band, 4 leave it as existing law has
# it. An untouched relief starts from its baseline (te_tax_base_dr_<key>).
te_tax_cmd_draft_relief = {
	if = {
		limit = { te_tax_can_draft_relief = { KEY = $KEY$ DIR = $DIR$ } }
		te_tax_dr_relief_$DIR$ = { KEY = $KEY$ }
	}
}

# The first change to either relief records the relief group's planned
# version; then the relief itself is taken into the draft at existing law.
te_tax_dr_relief_touch = {
	if = {
		limit = { var:te_tax_dr_agrel < 0 var:te_tax_dr_regrel < 0 }
		set_variable = { name = te_tax_dr_relief_pver value = var:te_tax_pver_relief }
	}
	te_tax_dr_$KEY$_touch = yes
}

te_tax_dr_agrel_touch = {
	if = {
		limit = { var:te_tax_dr_agrel < 0 }
		set_variable = { name = te_tax_dr_agrel value = te_tax_base_dr_agrel }
	}
}

# Regional relief comes into the draft whole: its band and the states existing
# law names in the draft's commencement month, so a draft that changes only the
# band restates the same states. Existing law is the enacted list
# (te_tax_en_relief_states), unless an awaiting package that commences before
# the draft restates the states: the last of them in commencement order
# (te_tax_relief_base_from_<slot>), whose states carry its pending marks.
te_tax_dr_regrel_touch = {
	if = {
		limit = { var:te_tax_dr_regrel < 0 }
		set_variable = { name = te_tax_dr_regrel value = te_tax_base_dr_regrel }
		clear_variable_list = te_tax_dr_relief_states
		if = {
			limit = { var:te_tax_dr_regrel >= 1 }
			save_scope_as = te_tax_country
			if = {
				limit = { te_tax_relief_base_from_a = { DUE = te_tax_dr_due } }
				every_scope_state = {
					limit = { has_variable = te_tax_pending_relief_a var:te_tax_pending_relief_a = 1 }
					owner = { add_to_variable_list = { name = te_tax_dr_relief_states target = prev } }
				}
			}
			else_if = {
				limit = { te_tax_relief_base_from_b = { DUE = te_tax_dr_due } }
				every_scope_state = {
					limit = { has_variable = te_tax_pending_relief_b var:te_tax_pending_relief_b = 1 }
					owner = { add_to_variable_list = { name = te_tax_dr_relief_states target = prev } }
				}
			}
			else_if = {
				limit = { has_variable_list = te_tax_en_relief_states }
				every_in_list = {
					variable = te_tax_en_relief_states
					limit = { owner ?= scope:te_tax_country }
					scope:te_tax_country = { add_to_variable_list = { name = te_tax_dr_relief_states target = PREV } }
				}
			}
		}
	}
}

# After every relief change: a regional relief of none names no state, and a
# draft that changes neither relief drops the relief group's planned version.
te_tax_dr_relief_settle = {
	if = {
		limit = { var:te_tax_dr_regrel < 1 }
		clear_variable_list = te_tax_dr_relief_states
	}
	if = {
		limit = { var:te_tax_dr_agrel < 0 var:te_tax_dr_regrel < 0 }
		set_variable = { name = te_tax_dr_relief_pver value = -1 }
	}
}

# Takes regional relief into the draft as existing law has it (if the draft
# does not change it yet) and lists the country's incorporated states to name
# from (te_tax_dr_relief_candidates), the list the workbench shows. The only
# place the candidates are built: never when a panel opens. Pressed again, it
# refreshes the list after states were gained or lost.
te_tax_cmd_draft_relief_choose = {
	if = {
		limit = { te_tax_can_draft_relief_choose = yes }
		custom_tooltip = te_tax_tt_cmd_draft_relief_choose

# Passes the bill: an approved package in slot a if it is free, else b.
te_tax_cmd_pass = {
	if = {
		limit = { te_tax_can_pass = yes }
		custom_tooltip = te_tax_tt_cmd_pass
		if = {
			limit = { var:te_tax_pa_on = 0 }
			te_tax_pass_into = { SLOT = a OTHER = b }
		}
		else = {
			te_tax_pass_into = { SLOT = b OTHER = a }
		}
	}
}

# Stores the bill in slot SLOT, and only then records it, lets the groups that
# opposed it react and closes it. The whole sequence runs only when the store
# will accept the bill: te_tax_store_package checks exactly
# te_tax_can_store_package, and nothing between this check and the store
# changes what it reads (supersession writes only the OTHER slot, the bump only
# te_tax_pver_*). Checking before the store rather than after keeps the
# per-group disapproval lines in the Pass button's tooltip, which is rendered
# while the slot is still empty.
# Supersession at approval touches only the OTHER slot (provisions it would
# change on or after this bill's due month); the enacted code and its sunsets
# are left for commencement. A draft identical to the bill is discarded with it.
te_tax_pass_into = {
	if = {
		limit = { te_tax_can_store_package = { SLOT = $SLOT$ } }
		te_tax_gen_supersede_$OTHER$ = yes
		te_tax_gen_bump_pver = yes
		te_tax_store_package = { SLOT = $SLOT$ }
		te_tax_history_push = { KIND = 6 SLOT = $SLOT$ }
		te_tax_gen_oppose_approval = yes
		if = {
			limit = {
				te_tax_draft_active = yes
				NOT = { te_tax_gen_draft_differs_from_bill = yes }
			}
			te_tax_draft_close = yes
		}
		te_tax_bill_close = yes
		debug_log = "TE_TAX passed; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
	}
	else = {
		debug_log = "TE_TAX pass_refused the scheduler would not store the bill; nothing changed; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
	}
}

# Moves the bill's commencement to next month once its due month has come
# during debate. Same revision: commitments stay; the support snapshot and the
# minor class follow the new due month. If the new month changes the class
# (the baseline moved, so the steps did), the debate clock restarts, because
# the class decides how long the debate is.
te_tax_cmd_reschedule = {
	if = {
		limit = { te_tax_can_reschedule = yes }
		custom_tooltip = te_tax_tt_cmd_reschedule
		set_variable = { name = te_tax_bl_due value = te_tax_due_earliest }
		if = {
			limit = {
				OR = {
					AND = { var:te_tax_bl_minor = 1 NOT = { te_tax_bill_is_minor = yes } }
					AND = { var:te_tax_bl_minor = 0 te_tax_bill_is_minor = yes }
				}
			}
			set_variable = { name = te_tax_bl_day value = game_date }
		}
		te_tax_bill_set_minor = yes
		te_tax_refresh_support = yes
		debug_log = "TE_TAX rescheduled; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
	}
}

# Commences a package the scheduler held as missed next month instead (SLOT: a
# or b). Its approved expiry months stand: the delay is at most three months
# and every sunset is at least six after the approved month, so each still
# falls at least three months after the new one. Its external versions are
# not re-recorded; a missed package has no conflict to accept, and one that
# meets a change at its new commencement holds as conflicting.
te_tax_cmd_package_reschedule = {
	if = {
		limit = { te_tax_can_package_reschedule = { SLOT = $SLOT$ } }
		custom_tooltip = te_tax_tt_cmd_package_reschedule
		set_variable = { name = te_tax_p$SLOT$_due value = te_tax_due_earliest }
		set_variable = { name = te_tax_p$SLOT$_state value = 1 }
		te_tax_history_push = { KIND = 9 SLOT = $SLOT$ }
		te_tax_recompute_next_month = yes
		debug_log = "TE_TAX package_rescheduled to next month; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
	}
}

# Releases a held package (SLOT: a or b): the slot frees and nothing the
# package would have changed takes effect. A held_conflict package is resolved
# only this way; its terms come back as a new bill.
te_tax_cmd_package_release = {
	if = {
		limit = { te_tax_can_package_release = { SLOT = $SLOT$ } }
		custom_tooltip = te_tax_tt_cmd_package_release
		set_variable = { name = te_tax_p$SLOT$_on value = 0 }
		set_variable = { name = te_tax_p$SLOT$_state value = 0 }
		every_scope_state = {
			limit = { has_variable = te_tax_pending_relief_$SLOT$ }
			set_variable = { name = te_tax_pending_relief_$SLOT$ value = 0 }
		}
		te_tax_history_push = { KIND = 10 SLOT = $SLOT$ }
		te_tax_recompute_next_month = yes
		debug_log = "TE_TAX package_released; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
	}
}

# Writes the support snapshot: each interest group's reasons, score and
# commitment (te_tax_gen_refresh_support). Called on introduction, revision and
# rescheduling, and monthly by the processor (te_tax_process_month) while a
# bill is open. Never from a GUI.
te_tax_refresh_support = {
	if = {
		limit = {
			te_tax_code_on = yes
			te_tax_bill_active = yes
		}
		te_tax_gen_refresh_support = yes
	}
}
