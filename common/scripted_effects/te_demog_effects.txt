#                       15-17, 18-35, 36-59, 60-148): next step's flow shares

#                       a country's: its states' people-weighted mean; the state region's te_dg_pr_lm_r and

#   te_dg_hegemon       phase 3, under Full: the Cultural Hegemony factor the step's desired fertility took (absent at
#                       1; te_demog_fl_effects.txt). The country's phase 3 variables (Family Limitation's te_dg_pr_fl,
#                       te_dg_fl_line, te_dg_fl_nb, te_dg_fl_agr, _lit, _ero; te_dg_desired; te_dg_ch_desired) are

#                       step's other inputs the factors read (te_demog_cause_inputs_store; the last two the
#                       walk's te_dg_sol and te_dg_lit as the step used them)

#   te_dg_old_age       the old-age threshold, the age with 15 years left, women's and men's averaged (generated
#                       te_demog_old_age_pass, phase 2 step 1)
#   te_dg_low_share, te_dg_wbase, te_dg_wf_engine  the walk's, under the cohorts only (te_demog_work_walk_store,

# Flows (Task 9), the people the engine moves and kills (spec §2.5):
#   te_dg_war_in        state: war dead the owner shared out at its monthly pulses
#                       since the state's last step, taken (and removed) by the next step,
#                       so they cover the same months as its population change
#   te_dg_kills_in      state: people the mod's own script killed since the last
#                       step (te_demog_note_kills), taken and removed the same way
#                       (Internal Resettlement's transit dead among them since 2026-10-19)
#   te_dg_rs_in, te_dg_rs_out, te_dg_rs_dead, te_dg_rs_i_*, te_dg_rs_o_*  state: Internal Resettlement's
#                       moves and transit dead since the last step, by programme profile, taken and removed
#                       the same way (te_demog_resettle_effects.txt)
#   te_dg_frontier      state: history's mark on the US frontier, read by its 1836 seeds (the same file)
#   te_dg_net_migration state: last step's net migration, people (signed): the residual's rest after the noise
#                       band (0 inside it) plus Internal Resettlement's known moves (since 2026-10-19)
#   te_dg_rs_net        state: those known moves, arrivals less departures, from a step that had them (removed by a
#                       step or seed without them; te_demog_resettle_effects.txt), so tools can take them out
#   te_dg_mig_rate      state and country: te_dg_net_migration per 1,000 people (issue #1029, the overview's
#                       Net Migration); a country's from its states' sums
#   te_dg_mig_raw       state: the whole residual before the noise band, known moves included; the country's sum of it
#                       (te_demog_country_census) is the census log's gate figure
#   te_dg_inflow_years  state: consecutive steps of net inflow (chain migration)
#   te_dg_war_seen      country: the dead of all its wars at the last monthly pulse
#   te_dg_war_year      country: the year's rise in war dead, until its yearly pulse (read by the
#                       Wealth Concentration war shock; the states take theirs monthly)
#   te_dg_reseed        state: set by a seed made off the state's own pulse (game start, the
#                       console) and by fast mode's first switch; its next pulse (or the
#                       clock's next step) seeds again, so its first step covers a whole step

# Fast mode (te_demog_fast_effects.txt; te_debug_demog.1, console only):
#   te_dg_fast_fb, te_dg_fast_fd  state: the multipliers of te_demog_fast_births and
#                       te_demog_fast_deaths, applied at the clock's last step
#
# The trend arrows (Task 10), on a state and on a country alike:
#   te_dg_stepped       1 when the figures stored now came from a step (whatever its births), 0
#                       when a seed wrote them: a seed writes 0 births and deaths, so its CBR and
#                       CDR are no year to compare with
#   te_dg_trend_ok      1 when te_dg_prev_<name> holds last year's figure from a step and the
#                       figure stored now is a step's as well: false in a seed's census year and
#                       in the first step's after it, true from the second step on. A panel
#                       draws an arrow only while it is 1. The prev copies are made only after
#                       a step, so none is ever a seed's. A state re-seeded (a census gap, an
#                       emptied ring, its people moving more than 25% in a year) starts again;
#                       so does a country whose states are all fresh seeds, but not one that
#                       takes a new state in.
#   te_dg_prev_tfr, te_dg_prev_e0  a state's, copied by te_demog_prepare before it overwrites
#                       them
#
# The country's census (Task 10, te_demog_country_census), at its yearly pulse and at game start
# (under fast mode's clock, a day after each of its steps instead):
#   te_dg_census_year   the year of the country's last census (absent = census pending)
#   te_dg_cbf<b>, te_dg_cbm<b>  people in band b, the sum of its states' te_dg_bf<b>, te_dg_bm<b>
#   te_dg_median, te_dg_young_share, te_dg_working_share, te_dg_old_share, te_dg_dependency,
#   te_dg_sex_balance, te_dg_w1549  from those bands, as a state's (te_demog_store_figures)
#   te_dg_tfr           the states' weighted by their women aged 15-49
#   te_dg_e0, te_dg_e65 the states' weighted by people, and te_dg_e0f, te_dg_e0m (women's and men's) the same way;
#                       te_dg_imr by births (by people while the
#                       census holds no births, as in a seed-only census, whose births are 0)
#   te_dg_cbr, te_dg_cdr  births and deaths per 1,000 people; te_dg_net_migration and
#                       te_dg_mig_raw the sums.
#                       CBR and CDR are not real while te_dg_stepped is 0: a seed writes no births
#                       or deaths, so they read 0. That is apart from te_dg_trend_ok (which
#                       only says whether an arrow may be drawn): a panel shows no CBR and CDR
#                       for te_dg_stepped = 0
#   te_dg_urban_share   urban people over people; te_dg_primacy the largest state's urban people
#                       over all urban people (a state is one city, spec 5.2); te_dg_n_cities the
#                       effective number of cities, (urban)^2 over the sum of squares, both in
#                       units of 100,000 people (values are i64 x 1e-5: 10 million people squared
#                       would overflow, and in millions the square of a state under ~3,200
#                       urban people is under one unit); te_dg_urban_label 0 Primate (primacy
#                       40% or more), 1 Dominant (25%), 2 Balanced (10%), 3 Dispersed.
#                       te_dg_n_cities is a number and te_dg_cities a variable list: kept under
#                       two names in case the engine keeps a variable and a list in one namespace.
#   te_dg_prev_<name>   each of the figures above, as last year's (see the arrows)
#   te_dg_pjf<b>, te_dg_pjm<b>  the player's bands twenty years ahead (te_demog_project)
#   te_dg_dc_*          issue #1029: its states' deaths by cause summed over the states that have them (0 until one
#                       does), with te_dg_dc_total and te_dg_dc_lead as a state's (te_demog_cause_deaths_country_store)
#   te_dg_m_inf, _ext, _chr, _work, _mat, te_dg_acc  the cause multipliers and health-care access, its states'
#                       weighted by people (maternal by births), for the legend's hovers
#   te_dg_inf_ad        #1045: the adults' infection factor, its states' weighted by people x te_dg_m_inf (so with
#                       te_dg_m_inf it gives their adults' infection exactly), for the outline
#   te_dg_cdr_model     under Full, the census's own death rate across its states, weighted by people (the
#                       Added on Top row's measure)
#   te_dg_states_n      how many census states it has (the States section's Show All)
# Variable lists on the country (state scopes, drawn by the panel):
#   te_dg_states        its states with a census, the largest by people first, at most 40
#   te_dg_states_top    the first ten of te_dg_states, the States section's rows until Show All
#   te_dg_cities        its three census states of the largest cities, by state_city_size_rank

# THIS = a state, on its yearly pulse. One pop walk and one building walk fill
# every sum the census, Wealth Concentration and inheritance need (spec §11.3).
te_demog_walks = {
	set_local_variable = { name = te_dg_w_pop value = 0 }
	set_local_variable = { name = te_dg_w_lit value = 0 }
	set_local_variable = { name = te_dg_w_sol value = 0 }
	set_local_variable = { name = te_dg_w_wtfr value = 0 }
	set_local_variable = { name = te_dg_w_eb value = 0 }
	set_local_variable = { name = te_dg_w_ed value = 0 }
	set_local_variable = { name = te_dg_w_eb0 value = 0 }
	set_local_variable = { name = te_dg_w_ed0 value = 0 }
	set_local_variable = { name = te_dg_w_agr value = 0 }
	set_local_variable = { name = te_dg_w_sold value = 0 }
	set_local_variable = { name = te_dg_w_bur value = 0 }
	set_local_variable = { name = te_dg_w_rural value = 0 }
	set_local_variable = { name = te_dg_w_fjob value = 0 }
	set_local_variable = { name = te_dg_w_emp value = 0 }
	te_demog_gini_band_init = yes
	# the replaced sums: what the census absorbs, at each pop's bare curves (phase 2 step 4; generated)
	te_demog_absorbed_walk_init = yes
	# phase 2 step 1: the lower strata, the pops' own ratios and the engine's workforce, under the cohorts only
	te_demog_work_walk_init = yes
	every_scope_pop = {
		change_local_variable = { name = te_dg_w_pop add = total_size }
		change_local_variable = { name = te_dg_w_lit add = te_demog_pop_literate }
		change_local_variable = { name = te_dg_w_sol add = te_demog_pop_sol }
		change_local_variable = { name = te_dg_w_wtfr add = te_demog_pop_wealth_tfr }
		# the pop's curves once, kept bare as well (te_dg_w_eb0/_ed0): fast mode's term reads the
		# state's average multiplier (te_demog_fast_refresh_rates)
		set_local_variable = { name = te_dg_p_eb value = te_demog_pop_engine_births }
		set_local_variable = { name = te_dg_p_ed value = te_demog_pop_engine_deaths }
		change_local_variable = { name = te_dg_w_eb0 add = local_var:te_dg_p_eb }
		change_local_variable = { name = te_dg_w_ed0 add = local_var:te_dg_p_ed }
		set_local_variable = { name = te_dg_p_mb value = te_demog_pop_birth_mult }
		set_local_variable = { name = te_dg_p_md value = te_demog_pop_death_mult }
		# the absorbed terms at its bare curves (literacy, mild starvation, its state's and class's absorbed
		# lines), which the census's terms net out, and the bare curves those terms reach, the pops the engine
		# doesn't floor at 0 (te_demog_rates_refresh)
		te_demog_absorbed_walk_pop = yes
		change_local_variable = { name = te_dg_w_eb add = { value = local_var:te_dg_p_eb multiply = local_var:te_dg_p_mb } }
		change_local_variable = { name = te_dg_w_ed add = { value = local_var:te_dg_p_ed multiply = local_var:te_dg_p_md } }
		# the Gini's wealth band (spec 4.1; generated)
		te_demog_gini_band_add = yes
		if = {
			limit = {
				OR = {
					is_pop_type = peasants
					is_pop_type = farmers
				}
			}
			change_local_variable = { name = te_dg_w_agr add = total_size }
		}
		if = {
			limit = {
				OR = {
					is_pop_type = soldiers
					is_pop_type = officers
				}
			}
			change_local_variable = { name = te_dg_w_sold add = total_size }
		}
		if = {
			limit = { is_pop_type = bureaucrats }
			change_local_variable = { name = te_dg_w_bur add = total_size }
		}
		if = {
			limit = { exists = workplace }
			change_local_variable = { name = te_dg_w_emp add = total_size }
			if = {
				limit = { workplace = { te_demog_rural_workplace = yes } }
				change_local_variable = { name = te_dg_w_rural add = total_size }
			}
			if = {
				limit = { workplace = { te_demog_female_job_workplace = yes } }
				change_local_variable = { name = te_dg_w_fjob add = total_size }
			}
		}
		else_if = {
			limit = { is_pop_type = peasants }
			change_local_variable = { name = te_dg_w_rural add = total_size }
		}
		te_demog_work_walk_pop = yes
	}
	set_variable = { name = te_dg_walk_pop value = local_var:te_dg_w_pop }
	set_variable = { name = te_dg_lit value = { value = local_var:te_dg_w_lit divide = { value = local_var:te_dg_w_pop min = 1 } } }
	set_variable = { name = te_dg_sol value = { value = local_var:te_dg_w_sol divide = { value = local_var:te_dg_w_pop min = 1 } } }
	set_variable = { name = te_dg_wtfr value = { value = local_var:te_dg_w_wtfr divide = { value = local_var:te_dg_w_pop min = 1 } } }
	# the engine's births and deaths over the step (a year, or under fast mode the clock's window of
	# te_demog_step_months whole calendar months): the curves are x 100,000 a month, and each pop's
	# multiplier (its state's modifier plus its own terms, floored at 0) was applied in the walk. A
	# state-level 1 + modifier:state_*_mult left out literacy, starvation and every class, workplace
	# and working-conditions term, so the residual read births 5-9% high and deaths 6% low as
	# emigration everywhere (docs/testing/demographics-growth-probe-results-2026-10-09.md).
	set_variable = {
		name = te_dg_eb
		value = {
			value = local_var:te_dg_w_eb
			multiply = te_demog_step_months
			divide = 100000
		}
	}
	set_variable = {
		name = te_dg_ed
		value = {
			value = local_var:te_dg_w_ed
			multiply = te_demog_step_months
			divide = 100000
		}
	}
	# under Full, the window's events: the mean of the engine's rate at its start and today's (the trapezoid; owner, 2026-10-11)
	te_demog_window_events = yes
	# the Causes of Death section's split of the window's deaths (Full only)
	te_demog_on_top_split = yes
	set_variable = { name = te_dg_agr_share value = { value = local_var:te_dg_w_agr divide = { value = local_var:te_dg_w_pop min = 1 } max = 1 } }
	set_variable = { name = te_dg_soldiers value = local_var:te_dg_w_sold }
	set_variable = { name = te_dg_bureaucrats value = local_var:te_dg_w_bur }
	set_variable = { name = te_dg_urban value = { value = local_var:te_dg_w_pop subtract = local_var:te_dg_w_rural min = 0 } }
	set_variable = { name = te_dg_urban_share value = { value = var:te_dg_urban divide = { value = local_var:te_dg_w_pop min = 1 } } }
	set_variable = { name = te_dg_fjob_share value = { value = local_var:te_dg_w_fjob divide = { value = local_var:te_dg_w_emp min = 1 } } }
	te_demog_gini_band_store = yes
	te_demog_work_walk_store = yes
	# buildings: capital in private, cooperative and state hands; ownership levels
	set_local_variable = { name = te_dg_w_priv value = 0 }
	set_local_variable = { name = te_dg_w_self value = 0 }
	set_local_variable = { name = te_dg_w_ctry value = 0 }
	set_local_variable = { name = te_dg_w_own value = 0 }
	every_scope_building = {
		# level > 0 before the ownership fractions, as vanilla has it: a building

# THIS = a state, straight after its walks (its yearly pulse, or a step of fast mode's
# clock, where te_demog_year is the clock's year): seed or step its census.
te_demog_state_census = {
	if = {
		limit = { te_demog_cohorts_run = yes }
		# No census yet: seed. te_demog_set_scale writes te_dg_year and te_dg_raw
		# together, so the branches below never read a missing variable.
		if = {
			limit = { NOT = { te_demog_has_census = yes } }
			te_demog_seed = yes
		}
		# A census more than a year old (under fast mode's clock, a step), a ring
		# emptied while the state has people (no migrant settles an empty class),
		# or people more than a quarter above or below the last census's
		# (te_dg_pop_last, written with te_dg_year): seed again. A move that size is a state merged or split,
		# not migration (one year's natural change is under 5%), and the step's
		# residual would count it all as migrants. A seed made off this pulse
		# (te_dg_reseed: game start, the console) is made again here, so the first
		# step's population change and natural change cover the same year.
		# The people's move is taken into locals first and the limits compare those
		# with 0. Written as state_population > { value = var:te_dg_pop_last
		# multiply = 1.25 }, the test held for every state with people, even 0.3%
		# up, and its 0.75 twin for none, so every state seeded at every pulse and
		# none ever stepped (observer tests of #830 and #833, 2026-10-09).
		else = {
			set_local_variable = {
				name = te_dg_gap
				value = { value = te_demog_year subtract = var:te_dg_year }
			}
			# the people now less 1.25 and 0.75 times the last census's: above 0 and below 0
			# are a move of more than a quarter up and down
			set_local_variable = {
				name = te_dg_over
				value = { value = state_population subtract = { value = var:te_dg_pop_last multiply = 1.25 } }
			}
			set_local_variable = {
				name = te_dg_under
				value = { value = state_population subtract = { value = var:te_dg_pop_last multiply = 0.75 } }
			}
			# Internal Resettlement's moves since the last census are migration, not a merge: left out of the test
			# (te_demog_rs_known_change, 0 without them; te_demog_resettle_effects.txt)
			change_local_variable = { name = te_dg_over subtract = te_demog_rs_known_change }
			change_local_variable = { name = te_dg_under subtract = te_demog_rs_known_change }
			if = {
				limit = {
					OR = {
						local_var:te_dg_gap > 1
						AND = {
							var:te_dg_raw < 1
							state_population > 0
						}
						local_var:te_dg_over > 0
						local_var:te_dg_under < 0
						has_variable = te_dg_reseed
					}
				}
				te_demog_seed = yes
			}
			# A year on: one step. A second pulse in the same year (gap 0) does nothing.
			else_if = {
				limit = { local_var:te_dg_gap = 1 }
				te_demog_step = yes
			}
		}
	}
}

	# #1045: medicine's second reach on adults' infection, which te_demog_enter_group applies at their ages
	set_local_variable = { name = te_dg_inf_ad value = te_demog_infection_adult }
	set_local_variable = { name = te_dg_m_ext value = te_demog_mult_external }
	set_local_variable = { name = te_dg_m_chr value = te_demog_mult_chronic }

	set_local_variable = { name = te_dg_m_work_f value = { value = te_demog_mult_work multiply = local_var:te_dg_fws_d multiply = 2 } }
	set_local_variable = { name = te_dg_m_work_m value = { value = 1 subtract = local_var:te_dg_fws_d multiply = te_demog_mult_work multiply = 2 } }
	set_variable = { name = te_dg_m_inf value = local_var:te_dg_m_inf }

	set_variable = { name = te_dg_m_chr value = local_var:te_dg_m_chr }
	set_variable = { name = te_dg_m_mat value = local_var:te_dg_mmr }
	set_variable = { name = te_dg_m_work value = te_demog_mult_work }

	# (te_demog_lx_tables, with te_dg_old_age_c), kept before the plain table writes the untreated's figure over it
	set_local_variable = { name = te_dg_wa_oa value = -1 }

# One age group of the life table for both sexes ($N$ years wide; $FERTILE$ adds
# the group to the net reproduction rate's sum). demographics_model.life_table, nrr.
te_demog_lt_group = {
	te_demog_lt_sex = { SEX = f N = $N$ FERTILE = $FERTILE$ }
	te_demog_lt_sex = { SEX = m N = $N$ FERTILE = no }
}

# demographics_model.life_table's loop body for one sex: te_dg_l<SEX> = survivors
# (radix 100,000), te_dg_py<SEX> = person-years so far, te_dg_q<SEX> the group's
# yearly death probability per 100,000 (te_demog_enter_group).
te_demog_lt_sex = {
	set_local_variable = { name = te_dg_q value = { value = local_var:te_dg_q$SEX$ divide = 100000 } }
	set_local_variable = { name = te_dg_p value = { value = 1 subtract = local_var:te_dg_q } }
	set_local_variable = { name = te_dg_pn value = 1 }
	while = {
		count = $N$
		change_local_variable = { name = te_dg_pn multiply = local_var:te_dg_p }
	}
	# survivors summed over the group's single years: l x (1 - p^n) / q, or l x n
	set_local_variable = { name = te_dg_lsum value = { value = local_var:te_dg_l$SEX$ multiply = $N$ } }
	if = {
		limit = { local_var:te_dg_q > 0 }
		set_local_variable = {
			name = te_dg_lsum
			value = {
				value = 1
				subtract = local_var:te_dg_pn
				divide = { value = local_var:te_dg_q min = 0.00001 }
				multiply = local_var:te_dg_l$SEX$
			}
		}
	}
	# person-years count each year at mid-year: the sum x (1 - q / 2)
	change_local_variable = {
		name = te_dg_py$SEX$
		add = { value = local_var:te_dg_q multiply = -0.5 add = 1 multiply = local_var:te_dg_lsum }
	}
	# demographics_model.nrr: survivors to each childbearing age x its share of births
	if = {
		limit = { always = $FERTILE$ }
		change_local_variable = { name = te_dg_nrr_sum add = { value = local_var:te_dg_lsum multiply = local_var:te_dg_asfr } }
	}
	change_local_variable = { name = te_dg_l$SEX$ multiply = local_var:te_dg_pn }
}

# After the last group (demographics_model.life_table's return): e0, e65 (radix
# 100,000), each the mean of the two sexes.
te_demog_lt_finish = {
	set_variable = { name = te_dg_e0f value = { value = local_var:te_dg_pyf divide = 100000 } }
	set_variable = { name = te_dg_e0m value = { value = local_var:te_dg_pym divide = 100000 } }
	set_variable = { name = te_dg_e0 value = { value = var:te_dg_e0f add = var:te_dg_e0m multiply = 0.5 } }
	set_variable = {
		name = te_dg_e65
		value = {
			value = {
				value = local_var:te_dg_pyf
				subtract = local_var:te_dg_py65f
				divide = { value = local_var:te_dg_l65f min = 1 }
			}
			add = {
				value = local_var:te_dg_pym
				subtract = local_var:te_dg_py65m
				divide = { value = local_var:te_dg_l65m min = 1 }
			}
			multiply = 0.5
		}
	}
}

# demographics_model.seed's loop body. Age local_var:te_dg_age (0-149) in slot $S$:
# survivors x growth factor^age. An age fewer than 1 in 100,000 newborns reach
# stays empty (and a re-seed clears it), so the step skips the slot (spec §1).
# The group is the age's own (the seed writes the people of age a, then survives
# the year at a); the step's is the rate age.
te_demog_seed_slot = {
	te_demog_age_boundaries = yes
	if = {
		limit = {
			OR = {
				local_var:te_dg_lsf >= 1
				local_var:te_dg_lsm >= 1
			}
		}
		set_variable = {
			name = te_dg_f$S$
			value = { value = local_var:te_dg_lsf multiply = local_var:te_dg_dpow multiply = te_demog_k_female_births_per_100k divide = 100000 }
		}
		set_variable = {
			name = te_dg_m$S$
			value = { value = local_var:te_dg_lsm multiply = local_var:te_dg_dpow multiply = te_demog_k_male_births_per_100k divide = 100000 }
		}
		# §2.6: the US frontier's men at the 1840 census's ratio (generated; te_demog_resettle_effects.txt)
		if = {
			limit = { local_var:te_dg_fr_on > 0 }
			te_demog_frontier_slot = { S = $S$ }
		}

			}
		}

	}
	change_local_variable = { name = te_dg_dpow multiply = local_var:te_dg_d }
	change_local_variable = { name = te_dg_age add = 1 }
}

			change_local_variable = { name = te_dg_acc_fam add = var:te_dg_m$S$ }

	}
}

# THIS = a state, at the start of its step (after te_demog_prepare, which sets

# reads the rates only while te_dg_mig_in or te_dg_mig_out is above 0). Every branch sets the rest; the step
# reads them unguarded.
te_demog_flows = {
	# war dead the owner shared out monthly since this state's last step (te_demog_share_war_dead)
	set_local_variable = { name = te_dg_war value = 0 }
	if = {
		limit = { has_variable = te_dg_war_in }
		set_local_variable = { name = te_dg_war value = var:te_dg_war_in }
		remove_variable = te_dg_war_in
	}
	# from the pool's men and the women of its ages (phase 2 step 4, §4.2), the people the next sweep takes them from;
	# a census from before step 4 has its 18-40 sums instead, for this one step (the sweep takes the dead at the pool's
	# ages, so these divide by a few more cohorts than they reach)
	set_local_variable = { name = te_dg_war_den_m value = 0 }
	set_local_variable = { name = te_dg_war_den_f value = 0 }
	if = {
		limit = { has_variable = te_dg_fam }
		set_local_variable = { name = te_dg_war_den_m value = var:te_dg_fam }
		set_local_variable = { name = te_dg_war_den_f value = var:te_dg_faw }
	}
	else_if = {
		limit = { has_variable = te_dg_m1840 }
		set_local_variable = { name = te_dg_war_den_m value = var:te_dg_m1840 }
		set_local_variable = { name = te_dg_war_den_f value = var:te_dg_f1840 }
	}
	set_local_variable = {
		name = te_dg_war_fm
		value = { value = local_var:te_dg_war multiply = te_demog_k_war_male_share divide = { value = local_var:te_dg_war_den_m min = 1 } max = 0.5 }
	}
	set_local_variable = {
		name = te_dg_war_ff
		value = {
			value = 1
			subtract = te_demog_k_war_male_share
			multiply = local_var:te_dg_war
			divide = { value = local_var:te_dg_war_den_f min = 1 }
			max = 0.5
		}
	}
	# the mod's own kills since the last step (te_demog_note_kills)
	set_local_variable = { name = te_dg_kills value = 0 }
	if = {
		limit = { has_variable = te_dg_kills_in }
		set_local_variable = { name = te_dg_kills value = var:te_dg_kills_in }
		remove_variable = te_dg_kills_in
	}
	# Internal Resettlement's known moves since the last step (te_demog_resettle_effects.txt): its transit dead are among
	# the kills, but leave by their programme's profile, not from every age
	te_demog_rs_take = yes

		name = te_dg_mig

			value = state_population
			subtract = var:te_dg_pop_last
			subtract = var:te_dg_eb
			add = var:te_dg_ed
			add = local_var:te_dg_war
			add = local_var:te_dg_kills
		}
	}

	# (te_demog_rs_enter_class books them by their programmes' profiles)

		change_local_variable = { name = te_dg_mig add = local_var:te_dg_rs_out }

	}
	# a residual inside the noise band is model error, not migration
	set_local_variable = { name = te_dg_noise value = { value = var:te_dg_people multiply = te_demog_k_residual_noise_share } }
	set_local_variable = { name = te_dg_neg_noise value = { value = local_var:te_dg_noise multiply = -1 } }

		limit = {
			local_var:te_dg_mig < local_var:te_dg_noise
			local_var:te_dg_mig > local_var:te_dg_neg_noise

	}
	set_variable = { name = te_dg_net_migration value = local_var:te_dg_mig }

		limit = { local_var:te_dg_rk_on > 0 }

		change_variable = { name = te_dg_net_migration add = var:te_dg_rs_net }
	}
	else_if = {
		limit = { has_variable = te_dg_rs_net }
		remove_variable = te_dg_rs_net
	}
	set_local_variable = { name = te_dg_mig_in value = { value = local_var:te_dg_mig min = 0 } }
	set_local_variable = { name = te_dg_mig_out value = { value = local_var:te_dg_mig multiply = -1 min = 0 } }
	if = {
		limit = { NOT = { has_variable = te_dg_inflow_years } }

	}

	set_local_variable = { name = te_dg_rights value = { value = local_var:te_dg_fws divide = { value = te_demog_k_full_female_work_share min = 0.01 } } }
	set_local_variable = {
		name = te_dg_family
		value = {
			value = te_demog_k_family_base
			add = te_demog_family_transport
			add = { value = local_var:te_dg_rights multiply = te_demog_k_family_rights_weight }
			add = { value = var:te_dg_inflow_years multiply = te_demog_k_chain_step max = te_demog_k_chain_cap }
			max = { value = 1 subtract = local_var:te_dg_refugee }
			min = 0
		}
	}
	set_local_variable = { name = te_dg_labour value = { value = 1 subtract = local_var:te_dg_family subtract = local_var:te_dg_refugee min = 0 } }
	set_local_variable = {
		name = te_dg_labour_f
		value = {
			value = te_demog_k_labour_female_max
			subtract = te_demog_k_labour_female_min
			multiply = var:te_dg_fjob_share
			multiply = local_var:te_dg_rights
			add = te_demog_k_labour_female_min
			min = te_demog_k_labour_female_min
			max = te_demog_k_labour_female_max
		}
	}
	# the migrant booking (2026-10-12): the labour kind's sex split corrects itself, then the profile, then the cap
	te_demog_labour_balance = yes
	te_demog_set_profile = yes
	te_demog_migrant_rates = yes

	# read last year's count (the model's inp.inflow_years)
	if = {
		limit = { local_var:te_dg_mig > 0 }
		change_variable = { name = te_dg_inflow_years add = 1 }
	}
	else = {
		set_variable = { name = te_dg_inflow_years value = 0 }
	}
}

# THIS = a country, monthly. The dead of every war it is in, summed; the month's
# rise goes to the year's total (the Wealth Concentration war shock reads it at
# the yearly pulse, whatever the rule) and, while the census runs, at once to the
# states (te_demog_share_war_dead), so each state's step takes the dead of the
# same months as its population change. A war that ends drops out of the sum, so
# the change is floored at zero and loses at most that month of any other war's
# dead. The first run starts from the dead already counted, so a save that meets
# the mod mid-war does not read the war's whole past as one month's dead.
te_demog_country_monthly = {

		name = te_dg_rise
		value = {
			value = local_var:te_dg_dead
			subtract = var:te_dg_war_seen
			min = 0
		}
	}
	change_variable = { name = te_dg_war_year add = local_var:te_dg_rise }
	set_variable = { name = te_dg_war_seen value = local_var:te_dg_dead }
	if = {
		limit = {
			te_demog_cohorts_run = yes
			local_var:te_dg_rise > 0
		}
		te_demog_share_war_dead = yes
	}
}

		# the bill stops with the law, the day it goes (the census would take it off only at its yearly pulse); never
		# re-applied here
		if = {
			limit = { has_modifier = te_demog_pension_bill }
			remove_modifier = te_demog_pension_bill
		}
		if = {
			limit = { has_variable = te_dg_pension_bill }
			remove_variable = te_dg_pension_bill
		}
	}
	te_demog_pension_ig_refresh = yes
}

# THIS = a state, once a year. Ages the ring one year (spec §2.1; demographics_model.step):
# the year's rates and flows, then the sweep. te_debug_demog.1 option a runs the two halves
# apart, so its replay line logs the flows the sweep is about to use.
te_demog_step = {
	te_demog_step_begin = yes
	te_demog_step_run = yes
}

# The step's first half: medicine's share of the gap (te_demog_medicine_drift, the Causes of Death
# plan), the practices' drift toward their targets (te_demog_practices_step, Late Marriage and Birth
# Spacing; it reads the drifted access), the year's rates into locals (te_demog_prepare), then the
# flows, which take and remove te_dg_war_in and te_dg_kills_in (te_demog_flows), then the rates
# scaled to what the engine did over the window (te_demog_on_top_scales, phase 2 step 4), then each
# natural cause's deaths at those rates over last count's bands, before the sweep writes this count's
# (te_demog_cause_deaths_estimate, issue #1029).
te_demog_step_begin = {
	# medicine moves its share of the gap, then the practices theirs, then the year's rates use them
	te_demog_medicine_drift = yes
	# phase 3: the neighbouring states' shares, which each practice's step closes part of the gap to (generated)
	te_demog_practices_neighbours = yes
	te_demog_practices_step = yes
	# phase 2 step 4: the ring's sex balance, which the year's fertility and women's rate read (te_dg_sxb)
	te_demog_sex_births_store = yes
	te_demog_prepare = yes
	# phase 3, the late game: eligibility, cover and the age groups' own rates once a carrier or treated people exist
	# (te_demog_lx_effects.txt; local te_dg_lx_on)
	te_demog_lx_begin = yes
	# no known moves unless the flows find Internal Resettlement's counters (te_demog_rs_take; the sweep reads it)
	set_local_variable = { name = te_dg_rk_on value = 0 }
	te_demog_flows = yes
	te_demog_on_top_scales = yes
	te_demog_cause_deaths_estimate = yes
}

# The step's second half: the sweep over the ring with the locals te_demog_step_begin set,
# then the year's totals and the scale. Run it straight after te_demog_step_begin, in the
# same effect (the locals don't outlive it).
te_demog_step_run = {
	# the newborns' split, girls and boys per 100,000 births (phase 2 step 3b: the sex ratio at birth; generated)
	te_demog_newborn_split = yes
	set_local_variable = { name = te_dg_open value = te_demog_open_slot }
	set_local_variable = { name = te_dg_s value = var:te_dg_scale }
	set_local_variable = { name = te_dg_age value = 1 }
	set_local_variable = { name = te_dg_r value = 0 }
	set_local_variable = { name = te_dg_g value = 0 }
	te_demog_enter_group = yes
	te_demog_reset_accumulators = yes
	# the flows' migrant class follows the rate age (class_of(r)), apart from the
	# accumulation classes, which follow the age after the step
	set_local_variable = { name = te_dg_acls value = 0 }
	set_local_variable = { name = te_dg_next_aclass value = 15 }
	te_demog_enter_class = yes
	# Internal Resettlement's known moves, by their programmes' profiles (te_demog_resettle_effects.txt)
	if = {
		limit = { local_var:te_dg_rk_on > 0 }
		te_demog_rs_enter_class = yes
	}
	# the war dead's window (phase 2 step 4): the pool's ages by rate age, the calendar one te_dg_r from
	# te_demog_k_pool_from and the biological one to te_demog_k_pool_to. te_dg_ab is the biological age after the step,
	# a year on from the rate age until phase 3 (which ages the treated more slowly and replaces this 1)
	set_local_variable = { name = te_dg_war_ab value = { value = te_demog_k_pool_to add = 1 } }
	# phase 3: a state in the late game takes the late sweep, its treated parts at their biological ages
	# (te_demog_lx_effects.txt); every other state the plain one, as before
	if = {
		limit = { local_var:te_dg_lx_on > 0 }
		te_demog_lx_sweep_run = yes
	}
	else = {
		te_demog_sweep = yes
	}
	te_demog_step_finish = yes
}

# demographics_model.step's loop over a = 1 .. 150. The people now aged
# local_var:te_dg_age lived through rate age te_dg_r (a - 1) this year: its group's
# rates and its migrant class. Boundaries and rate trackers advance at every
# slot; only an occupied slot is aged.
te_demog_slot = {
	te_demog_age_boundaries = yes
	if = {
		limit = { local_var:te_dg_r = local_var:te_dg_next_group }
		change_local_variable = { name = te_dg_g add = 1 }
		te_demog_enter_group = yes
	}
	if = {
		limit = { local_var:te_dg_r = local_var:te_dg_next_aclass }
		change_local_variable = { name = te_dg_acls add = 1 }
		if = {
			limit = { local_var:te_dg_acls = 1 }
			set_local_variable = { name = te_dg_next_aclass value = 18 }
		}
		else_if = {
			limit = { local_var:te_dg_acls = 2 }
			set_local_variable = { name = te_dg_next_aclass value = 36 }
		}
		else_if = {
			limit = { local_var:te_dg_acls = 3 }
			set_local_variable = { name = te_dg_next_aclass value = 60 }
		}
		else = {
			set_local_variable = { name = te_dg_next_aclass value = 999 }
		}
		te_demog_enter_class = yes
		if = {
			limit = { local_var:te_dg_rk_on > 0 }
			te_demog_rs_enter_class = yes
		}
	}
	# Rustication's ages start or end inside a class: the class's shares are taken again there (generated)
	if = {
		limit = { local_var:te_dg_rk_on > 0 }
		te_demog_rs_youth_hook = yes
	}
	if = {
		limit = { local_var:te_dg_age = 150 }
		te_demog_fold_slot = { S = $S$ }
	}
	else_if = {
		limit = { has_variable = te_dg_f$S$ }
		te_demog_age_slot = { S = $S$ }
	}
	change_local_variable = { name = te_dg_age add = 1 }
	change_local_variable = { name = te_dg_r add = 1 }
}

# demographics_model.step's body for a < 150: one cohort through the year.
te_demog_age_slot = {
	set_local_variable = { name = te_dg_f value = { value = var:te_dg_f$S$ multiply = local_var:te_dg_s } }
	set_local_variable = { name = te_dg_m value = { value = var:te_dg_m$S$ multiply = local_var:te_dg_s } }
	# phase 3: in the late sweep births by part and by clock, and artificial wombs (te_demog_lx_births)
	if = {
		limit = { local_var:te_dg_lx_t > 0 }
		te_demog_lx_births = yes
	}
	else = {
		# births from these women (rate ages 15-49 carry a non-zero te_dg_asfr); generated: planned and unplanned
		te_demog_slot_births = yes
		change_local_variable = { name = te_dg_births add = local_var:te_dg_b }
		change_local_variable = { name = te_dg_mat add = { value = local_var:te_dg_b multiply = local_var:te_dg_mmr divide = 100000 } }
	}
	# phase 3: the mean age at childbirth (te_dg_mab, te_demog_step_finish)
	change_local_variable = { name = te_dg_bage add = { value = local_var:te_dg_b multiply = local_var:te_dg_r } }
	# phase 3: in the late sweep the cohort's treated and untreated parts die apart (te_demog_lx_deaths)
	if = {
		limit = { local_var:te_dg_lx_t > 0 }
		te_demog_lx_deaths = yes
	}
	else = {
		# deaths, and maternal deaths per birth
		set_local_variable = {
			name = te_dg_f2
			value = {
				value = local_var:te_dg_qf
				divide = 100000
				multiply = -1
				add = 1
				multiply = local_var:te_dg_f
				subtract = { value = local_var:te_dg_b multiply = local_var:te_dg_mmr divide = 100000 }
			}
		}
		set_local_variable = {
			name = te_dg_m2
			value = { value = local_var:te_dg_qm divide = 100000 multiply = -1 add = 1 multiply = local_var:te_dg_m }
		}
		change_local_variable = {
			name = te_dg_deaths
			add = { value = local_var:te_dg_f add = local_var:te_dg_m subtract = local_var:te_dg_f2 subtract = local_var:te_dg_m2 }
		}
		# war dead: the pool's ages by rate age (phase 2 step 4, §4.2), from the counts before deaths
		if = {
			limit = {
				local_var:te_dg_r >= te_demog_k_pool_from
				local_var:te_dg_ab <= local_var:te_dg_war_ab
			}
			change_local_variable = { name = te_dg_f2 subtract = { value = local_var:te_dg_f multiply = local_var:te_dg_war_ff } }
			change_local_variable = { name = te_dg_m2 subtract = { value = local_var:te_dg_m multiply = local_var:te_dg_war_fm } }
		}
	}
	# known kills: every age, from the counts before deaths
	change_local_variable = { name = te_dg_f2 subtract = { value = local_var:te_dg_f multiply = local_var:te_dg_kill } }
	change_local_variable = { name = te_dg_m2 subtract = { value = local_var:te_dg_m multiply = local_var:te_dg_kill } }
	# migration: arrivals and departures in proportion to the cohort's share of its class
	change_local_variable = { name = te_dg_f2 add = { value = local_var:te_dg_f multiply = local_var:te_dg_mig_gain_f } }
	change_local_variable = { name = te_dg_m2 add = { value = local_var:te_dg_m multiply = local_var:te_dg_mig_gain_m } }
	if = {
		limit = { local_var:te_dg_mig_frac_f > 0 }
		change_local_variable = {
			name = te_dg_f2
			subtract = { value = local_var:te_dg_f multiply = local_var:te_dg_mig_frac_f max = { value = local_var:te_dg_f2 min = 0 multiply = 0.5 } }
		}
	}
	if = {
		limit = { local_var:te_dg_mig_frac_m > 0 }
		change_local_variable = {
			name = te_dg_m2
			subtract = { value = local_var:te_dg_m multiply = local_var:te_dg_mig_frac_m max = { value = local_var:te_dg_m2 min = 0 multiply = 0.5 } }
		}
	}
	# a cohort under half a person is dropped, so the slot is skipped until it reopens
	if = {
		limit = {
			local_var:te_dg_f2 < 0.5
			local_var:te_dg_m2 < 0.5
		}
		remove_variable = te_dg_f$S$
		remove_variable = te_dg_m$S$
	}
	else = {
		set_variable = { name = te_dg_f$S$ value = { value = local_var:te_dg_f2 min = 0 } }
		set_variable = { name = te_dg_m$S$ value = { value = local_var:te_dg_m2 min = 0 } }
		te_demog_accumulate_slot = { S = $S$ }
	}
}

# demographics_model.step's a == 150 and its pool lines. Age 150: the slot's people
# join the pool, which ages, loses the year's kills and dies at the last group's
# rates; the slot takes this year's births (it is the sweep's last).
te_demog_fold_slot = {
	# phase 3: the cohort's band's treated share and deficit join the pool's first (te_demog_lx_fold_merge)
	if = {
		limit = { local_var:te_dg_lx_t > 0 }
		te_demog_lx_fold_merge = { S = $S$ }
	}
	set_local_variable = { name = te_dg_pool value = { value = var:te_dg_pf add = var:te_dg_pm multiply = local_var:te_dg_s } }
	set_local_variable = { name = te_dg_old value = 0 }
	if = {
		limit = { has_variable = te_dg_f$S$ }
		set_local_variable = { name = te_dg_old value = { value = var:te_dg_f$S$ add = var:te_dg_m$S$ multiply = local_var:te_dg_s } }
		change_variable = { name = te_dg_pf multiply = local_var:te_dg_s }
		change_variable = { name = te_dg_pf add = { value = var:te_dg_f$S$ multiply = local_var:te_dg_s } }
		change_variable = { name = te_dg_pm multiply = local_var:te_dg_s }
		change_variable = { name = te_dg_pm add = { value = var:te_dg_m$S$ multiply = local_var:te_dg_s } }
	}
	else = {
		change_variable = { name = te_dg_pf multiply = local_var:te_dg_s }
		change_variable = { name = te_dg_pm multiply = local_var:te_dg_s }
	}
	set_variable = {
		name = te_dg_pa
		value = {
			value = var:te_dg_pa
			add = 1
			multiply = local_var:te_dg_pool
			add = { value = local_var:te_dg_old multiply = 150 }
			divide = { value = local_var:te_dg_pool add = local_var:te_dg_old min = 0.00001 }
		}
	}
	# known kills take the same share of the pool
	change_variable = { name = te_dg_pf multiply = { value = 1 subtract = local_var:te_dg_kill } }
	change_variable = { name = te_dg_pm multiply = { value = 1 subtract = local_var:te_dg_kill } }
	# phase 3: in the late sweep the pool's treated and untreated parts die apart (te_demog_lx_fold_deaths)
	if = {
		limit = { local_var:te_dg_lx_t > 0 }
		te_demog_lx_fold_deaths = yes
	}
	else = {
		change_local_variable = {
			name = te_dg_deaths
			add = {
				value = var:te_dg_pf
				multiply = local_var:te_dg_qf
				add = { value = var:te_dg_pm multiply = local_var:te_dg_qm }
				divide = 100000
			}
		}
		change_variable = { name = te_dg_pf multiply = { value = local_var:te_dg_qf divide = 100000 multiply = -1 add = 1 } }
		change_variable = { name = te_dg_pm multiply = { value = local_var:te_dg_qm divide = 100000 multiply = -1 add = 1 } }
	}
	set_variable = { name = te_dg_f$S$ value = { value = local_var:te_dg_births multiply = local_var:te_dg_nb_f divide = 100000 } }
	set_variable = { name = te_dg_m$S$ value = { value = local_var:te_dg_births multiply = local_var:te_dg_nb_m divide = 100000 } }
}

# After the sweep (demographics_model.step's tail): newborns into band 0 and
# class 0, the year's totals, then the scale. te_dg_births and te_dg_deaths are
# people this year (the cohorts were scaled to last year's population first).
te_demog_step_finish = {
	te_demog_close_accumulators = yes
	set_local_variable = { name = te_dg_bf_new value = { value = local_var:te_dg_births multiply = local_var:te_dg_nb_f divide = 100000 } }
	set_local_variable = { name = te_dg_bm_new value = { value = local_var:te_dg_births multiply = local_var:te_dg_nb_m divide = 100000 } }
	change_variable = { name = te_dg_bf0 add = local_var:te_dg_bf_new }
	change_variable = { name = te_dg_bm0 add = local_var:te_dg_bm_new }
	change_variable = { name = te_dg_cf0 add = local_var:te_dg_bf_new }
	change_variable = { name = te_dg_cm0 add = local_var:te_dg_bm_new }
	change_local_variable = { name = te_dg_raw add = local_var:te_dg_births }
	# phase 2 step 1's figures, now that te_dg_raw holds the newborns (D4)
	set_local_variable = { name = te_dg_wk_born value = local_var:te_dg_births }
	te_demog_work_finish = yes
	set_variable = { name = te_dg_births value = local_var:te_dg_births }
	set_variable = { name = te_dg_deaths value = local_var:te_dg_deaths }
	# phase 3 (late-game §6.2): the mean age at childbirth, mothers' rate age plus half a year (none with no births)
	if = {
		limit = { local_var:te_dg_births > 0 }
		set_variable = {
			name = te_dg_mab
			value = { value = local_var:te_dg_bage divide = { value = local_var:te_dg_births min = 0.00001 } add = 0.5 }
		}
	}
	# the model's own rates without the on-top scales: what the census's births and deaths modifiers aim at
	# next (te_demog_rates_refresh). Births: / (1 + bz). Deaths: the cause deaths / (1 + dz), the maternal
	# deaths (which follow the births) / (1 + bz).
	set_variable = {
		name = te_dg_cbr_model
		value = {
			value = local_var:te_dg_births
			divide = { value = local_var:te_dg_bz add = 1 min = 0.1 }
			multiply = 1000
			divide = { value = state_population min = 1 }
		}
	}
	set_variable = {
		name = te_dg_cdr_model
		value = {
			value = local_var:te_dg_deaths
			subtract = local_var:te_dg_mat
			divide = { value = local_var:te_dg_dz add = 1 min = 0.1 }
			add = { value = local_var:te_dg_mat divide = { value = local_var:te_dg_bz add = 1 min = 0.1 } }
			multiply = 1000
			divide = { value = state_population min = 1 }
		}
	}
	# issue #1029: the year's deaths by cause, for the Causes of Death section
	te_demog_cause_deaths_store = yes
	# uploaded minds (2026-10-19): which of the deaths uploaded, into the owner's pool (te_demog_um_effects.txt)
	te_demog_um_step = yes
	# a step's figures, whatever its births (te_demog_set_trend)
	set_local_variable = { name = te_dg_from_step value = 1 }
	te_demog_set_scale = yes
	# phase 3: the biological pyramid, at the new scale (te_demog_lx_effects.txt)
	if = {
		limit = { local_var:te_dg_lx_t > 0 }
		te_demog_lx_bio_bands_store = yes
	}
}

# THIS = a state at the end of its step (te_demog_step_finish, with the step's locals): last year's deaths by cause,
# people (issue #1029; demographics_model.cause_deaths), for the Causes of Death section and the country's census,
# which sums them. The four natural causes are te_demog_cause_deaths_estimate's figures (in the late game the late
# sweep's own, te_demog_lx_cause_deaths) scaled to the census's own deaths from them: its deaths less childbirth's
# (te_dg_mat) and, under Full, less what the on-top scale added (te_dg_dz above 0; below 0 the game took deaths away,
# which no slice shows). Childbirth's, the deaths added on top, the war dead (te_dg_war) and the people killed by
# events (te_dg_kills) as counted. Each scaled slice multiplies before it divides, so a small one keeps its digits.
#   te_dg_dc_inf, _ext, _chr, _work  infection and hunger, accidents and violence, chronic illness and old age, work
#   te_dg_dc_mat, te_dg_dc_ontop, te_dg_dc_war, te_dg_dc_kill  childbirth, added on top, war, killed by events
#   te_dg_dc_total  the slices' sum: te_dg_deaths with the war dead and the killed
#   te_dg_dc_fit    the estimate over the census's deaths from the four causes before the on-top split: 1 when the
#                   estimate matches the sweep (the self-check's figure; te_debug_demog_cause_deaths_selfcheck)
te_demog_cause_deaths_store = {
	set_local_variable = { name = te_dg_dc_nat value = { value = local_var:te_dg_deaths subtract = local_var:te_dg_mat min = 0 } }
	set_local_variable = { name = te_dg_dc_top value = 0 }
	if = {
		limit = { local_var:te_dg_dz > 0 }
		set_local_variable = {
			name = te_dg_dc_top
			value = { value = local_var:te_dg_dc_nat multiply = local_var:te_dg_dz divide = { value = local_var:te_dg_dz add = 1 min = 1 } }
		}
	}
	set_local_variable = { name = te_dg_dc_own value = { value = local_var:te_dg_dc_nat subtract = local_var:te_dg_dc_top } }
	set_local_variable = {
		name = te_dg_dc_est
		value = { value = local_var:te_dg_ce_inf add = local_var:te_dg_ce_ext add = local_var:te_dg_ce_chr add = local_var:te_dg_ce_work }
	}
	if = {
		limit = { local_var:te_dg_dc_est > 0 }
		set_variable = { name = te_dg_dc_inf value = { value = local_var:te_dg_ce_inf multiply = local_var:te_dg_dc_own divide = { value = local_var:te_dg_dc_est min = 0.00001 } } }
		set_variable = { name = te_dg_dc_ext value = { value = local_var:te_dg_ce_ext multiply = local_var:te_dg_dc_own divide = { value = local_var:te_dg_dc_est min = 0.00001 } } }
		set_variable = { name = te_dg_dc_chr value = { value = local_var:te_dg_ce_chr multiply = local_var:te_dg_dc_own divide = { value = local_var:te_dg_dc_est min = 0.00001 } } }
		set_variable = { name = te_dg_dc_work value = { value = local_var:te_dg_ce_work multiply = local_var:te_dg_dc_own divide = { value = local_var:te_dg_dc_est min = 0.00001 } } }
	}
	else = {
		set_variable = { name = te_dg_dc_inf value = 0 }
		set_variable = { name = te_dg_dc_ext value = 0 }
		set_variable = { name = te_dg_dc_chr value = 0 }
		set_variable = { name = te_dg_dc_work value = 0 }
	}
	set_variable = { name = te_dg_dc_mat value = local_var:te_dg_mat }
	set_variable = { name = te_dg_dc_ontop value = local_var:te_dg_dc_top }
	set_variable = { name = te_dg_dc_war value = local_var:te_dg_war }
	set_variable = { name = te_dg_dc_kill value = local_var:te_dg_kills }
	te_demog_cause_deaths_total = yes
	set_variable = { name = te_dg_dc_fit value = { value = local_var:te_dg_dc_est divide = { value = local_var:te_dg_dc_nat min = 1 } } }
	# once a game: the estimate against the sweep (te_debug_demog_effects.txt)
	te_debug_demog_cause_deaths_selfcheck = yes
}

# THIS = a state at its seed (issue #1029): a seed counts no deaths (te_dg_deaths 0), so it has none by cause, and no fit.
te_demog_cause_deaths_zero = {
	set_variable = { name = te_dg_dc_inf value = 0 }
	set_variable = { name = te_dg_dc_ext value = 0 }
	set_variable = { name = te_dg_dc_chr value = 0 }
	set_variable = { name = te_dg_dc_work value = 0 }
	set_variable = { name = te_dg_dc_mat value = 0 }
	set_variable = { name = te_dg_dc_ontop value = 0 }
	set_variable = { name = te_dg_dc_war value = 0 }
	set_variable = { name = te_dg_dc_kill value = 0 }
	set_variable = { name = te_dg_dc_total value = 0 }

# THIS = a state or a country, before it stores a new set of figures: sets the local

# THIS = a state or a country, after te_demog_band_figures: the figures from the bands,
# shares of all the people, the dependency ratio and the sex balance.
te_demog_store_figures = {
	set_variable = { name = te_dg_median value = local_var:te_dg_median }
	set_variable = { name = te_dg_young_share value = { value = local_var:te_dg_young divide = { value = local_var:te_dg_total min = 1 } } }
	set_variable = { name = te_dg_working_share value = { value = local_var:te_dg_working divide = { value = local_var:te_dg_total min = 1 } } }

	set_variable = { name = te_dg_dependency value = { value = local_var:te_dg_young add = local_var:te_dg_old divide = { value = local_var:te_dg_working min = 1 } } }

	set_variable = { name = te_dg_stepped value = local_var:te_dg_now }
}

		else_if = {

		# the player's outline twenty years ahead, on the country's average rates: the model's own, the

		te_demog_lx_country_bio_store = yes
		# phase 2 step 4: the country's pool and youth-bulge figures for its tab (te_demog_consequence_effects.txt)
		te_demog_consequences_country = yes

		if = {
			limit = { te_history_country_is_tracked = yes }
			te_demog_history_record = yes
		}
		# the console's census log: one TE_DEMOG_CENSUS line (te_debug_demog_effects.txt)
		if = {
