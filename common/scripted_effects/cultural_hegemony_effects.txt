# ---- MONTHLY: Per-country cache + modifier application ----
# Scope: Country
# Caches all display variables and applies/removes CH modifiers.
ch_monthly_country_update = {
	# Cache variables used by JE triggers (visibility/status conditions)
	set_variable = { name = ch_total value = cultural_pull_total }
	set_variable = { name = ch_art value = cultural_pull_from_art }
	set_variable = { name = ch_sol value = cultural_pull_from_sol }
	set_variable = { name = ch_monuments value = cultural_pull_from_monuments }
	set_variable = { name = ch_megaprojects value = cultural_pull_from_megaprojects }

	# A country that drops below the leaderboard pass's `cultural_pull_raw >= 0.01`
	# limit is never visited again by ch_yearly_global_update, so a stored board
	# position would stay frozen forever. Clear it here instead.
	if = {
		limit = {
			has_variable = ch_rank_self
			cultural_pull_raw < 0.01
		}
		remove_variable = ch_rank_self # REVIEWED 2026-09-18: clears a stale display rank; not a modifier multiplier
	}

	ch_set_display_state = yes

	# Foreign benchmark is now applied in JE on_monthly_pulse
	# (only affects countries that have the journal entry active)
	# Clean up stale foreign benchmark from countries without the JE
	if = {
		limit = {
			has_modifier = cultural_hegemony_foreign_benchmark
			NOT = { has_journal_entry = je_cultural_hegemony }
		}
		remove_modifier = cultural_hegemony_foreign_benchmark
	}
}

# ---- DISPLAY STATE ----
# Scope: Country
# The single site where the journal-entry widget's banded readings are derived.
# Nothing in .gui or localization re-derives any of this, and the share
# thresholds below are the only copy of 2 / 5 / 10 / 15 / 25 in the system.
#
#   ch_tier        0 negligible, 1 minor, 2 moderate, 3 significant,
#                  4 major, 5 hegemon
#   ch_prog_count  how many of the four cultural programmes are running
#
# Called from ch_monthly_country_update (the monthly on-action, which runs for
# every country from game start) AND from the tail of every ch_effect_<button>,
# so the widget's labels react to a click without waiting for the next pulse.
ch_set_display_state = {
	if = {
		limit = { NOT = { has_variable = ch_total } }
		set_variable = { name = ch_tier value = 0 }
	}
	else_if = {
		limit = { var:ch_total >= 25 }
		set_variable = { name = ch_tier value = 5 }
	}
	else_if = {
		limit = { var:ch_total >= 15 }
		set_variable = { name = ch_tier value = 4 }
	}
	else_if = {
		limit = { var:ch_total >= 10 }
		set_variable = { name = ch_tier value = 3 }
	}
	else_if = {
		limit = { var:ch_total >= 5 }
		set_variable = { name = ch_tier value = 2 }
	}
	else_if = {
		limit = { var:ch_total >= 2 }
		set_variable = { name = ch_tier value = 1 }
	}
	else = {
		set_variable = { name = ch_tier value = 0 }
	}

	set_variable = { name = ch_prog_count value = 0 }
	if = {
		limit = { je:je_cultural_hegemony ?= { has_modifier = ch_world_exposition } }
		change_variable = { name = ch_prog_count add = 1 }
	}
	if = {
		limit = { je:je_cultural_hegemony ?= { has_modifier = ch_cultural_institutes_program } }
		change_variable = { name = ch_prog_count add = 1 }
	}
	if = {
		limit = { je:je_cultural_hegemony ?= { has_modifier = ch_global_media_campaign } }
		change_variable = { name = ch_prog_count add = 1 }
	}
	if = {
		limit = { je:je_cultural_hegemony ?= { has_modifier = ch_cultural_protectionism_program } }
		change_variable = { name = ch_prog_count add = 1 }
	}
}

# ============================================================================
# SHARED BUTTON EFFECTS
# ============================================================================
# Each `ch_effect_<button>` holds the body that used to live inline in the
# matching scripted_button's `effect` block. Both the journal-entry button and
# `ch_policy_sgui` call these, so the two surfaces cannot drift.
#
# Three deliberate additions over the original inline bodies:
#   * `ch_set_display_state = yes` in the tail, so a click moves the widget's
#     labels immediately.
#   * on the four programme toggles, one
#     `te_history_record_cultural_hegemony_marker` call, so the history chart can
#     date the change. Both the AI's journal-entry button and the widget reach it
#     from here.
#   * on the two funding steppers, one `custom_tooltip` OUTSIDE the
#     `hidden_effect`: the original body is wholly hidden, so the widget's
#     composed tooltip would otherwise render an empty "what this does" half.
# Everything else is byte-identical to the pre-change button bodies.
# Scope: country (the journal entry's owner).
# ============================================================================

# The remove-then-conditionally-re-add block that the two funding steppers and
# the journal entry's monthly pulse all used to carry verbatim.
# Scope: country
ch_refresh_funding_modifiers = {
	je:je_cultural_hegemony ?= { remove_modifier = ch_cultural_program_funding }
	je:je_cultural_hegemony ?= { remove_modifier = ch_cultural_program_funding_cost }
	if = {
		limit = { ch_program_funding_level_value >= 1 }
		je:je_cultural_hegemony ?= { add_modifier = { name = ch_cultural_program_funding multiplier = ch_program_funding_level_value_for_je } }
		je:je_cultural_hegemony ?= { add_modifier = { name = ch_cultural_program_funding_cost multiplier = ch_program_funding_cost_mult_for_je } }
	}
}

ch_effect_increase_program_funding = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = ch_program_funding_level } }
			set_variable = { name = ch_program_funding_level value = 1 }
		}
		else = {
			change_variable = { name = ch_program_funding_level add = 1 }
			clamp_variable = { name = ch_program_funding_level min = 0 max = ch_program_funding_max_level }
		}

		ch_refresh_funding_modifiers = yes
		ch_set_display_state = yes
	}
	custom_tooltip = ch_policy_funding_step_tt
}

ch_effect_decrease_program_funding = {
	hidden_effect = {
		change_variable = { name = ch_program_funding_level subtract = 1 }
		clamp_variable = { name = ch_program_funding_level min = 0 max = ch_program_funding_max_level }

		ch_refresh_funding_modifiers = yes
		ch_set_display_state = yes
	}
	custom_tooltip = ch_policy_funding_step_tt
}

# POLITICAL MODEL — the one classification
# ============================================================================
# Sorts a country into the political-model family its culture exports. The
# yearly totals, the journal entry's "exported model" reading and the movement
# pressure all read the var:ch_model this writes; nothing else re-derives it.
#
# ch_model codes. 1-12 keep the numbers they had before the three newer
# models, because cultural_hegemony_custom_loc.txt keys off them.
#    1 communist           6 royalist_absolutist      11 liberal
#    2 anarchist           7 royalist_constitutional  12 other
#    3 fascist             8 religious                13 technocratic
#    4 corporatist         9 reactionary              14 republican
#    5 socialist          10 military_junta           15 developmentalist_junta
#
# The ORDER of the checks is part of the classification: a country that fits
# several families is the first one it reaches. A Corporate State that is also
# an Ethnostate is Fascist; a Technocracy under a monarch is Technocratic; a
# junta with State Religion is Reactionary; a constitutional monarchy with
# universal suffrage is Constitutional Monarchist, not Liberal. Every law test
# is has_law_or_variant, so Bakufu and Neo-Absolutism count as Autocracy.
# Neocameralism is deliberately absent: it is not corporatism in Victoria 3's
# sense, and it resolves on the country's other laws.
# Scope: country. Writes var:ch_model.
ch_set_political_model = {
	# Communist / Vanguardist
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_council_republic
				AND = {
					has_law_or_variant = law_type:law_single_party_state
					has_law_or_variant = law_type:law_command_economy
				}
			}
		}
		set_variable = { name = ch_model value = 1 }
	}
	# Anarchist
	else_if = {
		limit = { has_law_or_variant = law_type:law_anarchy }
		set_variable = { name = ch_model value = 2 }
	}
	# Fascist / Ethnonationalist
	else_if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_ethnostate
				AND = {
					has_law_or_variant = law_type:law_national_supremacy
					OR = {
						has_law_or_variant = law_type:law_single_party_state
						has_law_or_variant = law_type:law_autocracy
					}
				}
			}
		}
		set_variable = { name = ch_model value = 3 }
	}
	# Corporatist
	else_if = {
		limit = { has_law_or_variant = law_type:law_corporate_state }
		set_variable = { name = ch_model value = 4 }
	}
	# Socialist / Cooperative
	else_if = {
		limit = { has_law_or_variant = law_type:law_cooperative_ownership }
		set_variable = { name = ch_model value = 5 }
	}
	# Technocratic: rule by experts, or by the mod's algorithmic successor
	else_if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_technocracy
				has_law_or_variant = law_type:law_algorithmic_governance
			}
		}
		set_variable = { name = ch_model value = 13 }
	}
	# Absolutist monarchist
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_monarchy
			has_law_or_variant = law_type:law_autocracy
		}
		set_variable = { name = ch_model value = 6 }
	}
	# Constitutional monarchist
	else_if = {
		limit = { has_law_or_variant = law_type:law_monarchy }
		set_variable = { name = ch_model value = 7 }
	}
	# Theocratic / Religious
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_theocracy
			has_law_or_variant = law_type:law_state_religion
		}
		set_variable = { name = ch_model value = 8 }
	}
	# Reactionary / Authoritarian (autocracy + state religion outside a monarchy)
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_autocracy
			has_law_or_variant = law_type:law_state_religion
		}
		set_variable = { name = ch_model value = 9 }
	}
	# Military junta: developmentalist when it runs a directed economy and has
	# shed the regressive laws, conservative otherwise
	else_if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_single_party_state
			}
			has_law_or_variant = law_type:law_professional_army
			OR = {
				has_law_or_variant = law_type:law_secret_police
				has_law_or_variant = law_type:law_militarized_police
			}
		}
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_interventionism
					has_law_or_variant = law_type:law_command_economy
				}
				NOT = { ch_has_regressive_laws = yes }
			}
			set_variable = { name = ch_model value = 15 }
		}
		else = {
			set_variable = { name = ch_model value = 10 }
		}
	}
	# Liberal / Progressive democratic
	else_if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_universal_suffrage
				has_law_or_variant = law_type:law_direct_democracy
			}
			OR = {
				has_law_or_variant = law_type:law_protected_speech
				has_law_or_variant = law_type:law_right_of_assembly
			}
		}
		set_variable = { name = ch_model value = 11 }
	}
	# Republican: a republic that is not (yet) a liberal democracy
	else_if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_presidential_republic
				has_law_or_variant = law_type:law_parliamentary_republic
				has_law_or_variant = law_type:law_direct_democracy
			}
		}
		set_variable = { name = ch_model value = 14 }
	}
	# Mixed / Other
	else = {
		set_variable = { name = ch_model value = 12 }
	}
}

# ---- Per-model world totals ----
# Globals per model key M (the key names in the code table above):
#   ch_ideology_<M>_count  countries running the model
#   ch_ideology_<M>_raw    their summed cultural_pull_raw
#   ch_ideology_<M>_share  that sum as a percentage of the world's raw pull
# Rebuilt by ch_yearly_global_update. The share is weighted by cultural pull,
# not by head count: a model run by one 80% power outweighs one run by forty
# minor countries.

# Scope: any. Parameter: $MODEL$ (model key)
ch_reset_model_bucket = {
	set_global_variable = { name = ch_ideology_$MODEL$_count value = 0 }
	set_global_variable = { name = ch_ideology_$MODEL$_raw value = 0 }
	set_global_variable = { name = ch_ideology_$MODEL$_share value = 0 }
}

# Scope: country. Parameter: $MODEL$
ch_add_to_model_bucket = {
	change_global_variable = { name = ch_ideology_$MODEL$_count add = 1 }
	change_global_variable = { name = ch_ideology_$MODEL$_raw add = cultural_pull_raw }
}

# Scope: any. Parameter: $MODEL$
# Divides by global_var:ch_cached_global_raw, which the same rebuild refreshed
# before the totals were summed, so the fifteen shares add up to 100. Do not
# sum var:ch_total instead: its denominator is the previous rebuild's.
ch_finish_model_bucket = {
	set_global_variable = { name = ch_ideology_$MODEL$_share value = global_var:ch_ideology_$MODEL$_raw }
	change_global_variable = { name = ch_ideology_$MODEL$_share divide = global_var:ch_cached_global_raw }
	change_global_variable = { name = ch_ideology_$MODEL$_share multiply = 100 }
}

# Scope: country. Classifies it and adds it to its model's totals.
# Also keeps the rebuild's own snapshot of the model: var:ch_model_census (this
# rebuild) and var:ch_model_census_prev (the one before). var:ch_model itself is
# also rewritten between rebuilds whenever the pressure effect re-classifies a
# country it is using as scope:cultural_hegemon, so change detection
# (ch_switched_to_hegemon_model, cultural_hegemony.19/.20) reads the census
# pair instead.
ch_add_country_to_model_totals = {
	if = {
		limit = { has_variable = ch_model_census }
		set_variable = { name = ch_model_census_prev value = var:ch_model_census }

# Applies a movement modifier if the requested movement exists in the target.
# Any matching pressure modifiers passed in CLEAR_MODIFIER_1/2 are removed
# first so repeat calls refresh duration instead of layering duplicates.
# Parameters: $MOVEMENT$ (movement type), $MODIFIER$ (modifier name),
# $CLEAR_MODIFIER_1$/$CLEAR_MODIFIER_2$ (modifier names to refresh),
# $MONTHS$ (duration, a literal number), $DECAYING$ (yes/no),
# $MULT$ (modifier multiplier: 1, or a script value name)
ch_clear_movement_pressure_if_present = {
	if = {
		limit = {
			any_political_movement = {
				is_political_movement_type = $MOVEMENT$
			}
		}
		every_political_movement = {
			limit = { is_political_movement_type = $MOVEMENT$ }
			if = {
				limit = { has_modifier = $CLEAR_MODIFIER_1$ }
				remove_modifier = $CLEAR_MODIFIER_1$
			}
			if = {
				limit = { has_modifier = $CLEAR_MODIFIER_2$ }
				remove_modifier = $CLEAR_MODIFIER_2$
			}
		}
	}
}

ch_apply_movement_pressure_if_present = {
	if = {
		limit = {
			any_political_movement = {
				is_political_movement_type = $MOVEMENT$
			}
		}
		ch_clear_movement_pressure_if_present = {
			MOVEMENT = $MOVEMENT$
			CLEAR_MODIFIER_1 = $CLEAR_MODIFIER_1$
			CLEAR_MODIFIER_2 = $CLEAR_MODIFIER_2$
		}
		every_political_movement = {
			limit = { is_political_movement_type = $MOVEMENT$ }
			add_modifier = { name = $MODIFIER$ months = $MONTHS$ is_decaying = $DECAYING$ multiplier = $MULT$ }
		}
	}
}

# Applies full pressure to the primary movement when it exists; otherwise it
# falls back to aligned movements in priority order using a weaker modifier.
# Parameters: $PRIMARY$ (primary movement type), $FALLBACK_1$/$FALLBACK_2$
# (aligned fallback movement types), $MODIFIER$ (primary modifier name),
# $FALLBACK_MODIFIER$ (fallback modifier name), $MONTHS$ (duration),
# $DECAYING$ (yes/no), $MULT$ (modifier multiplier)
ch_apply_primary_or_fallback_movement_pressure = {
	# Clear stale pressure from the full family before choosing the current
	# target, so a newly created primary movement replaces any old fallback.
	ch_clear_movement_pressure_if_present = {
		MOVEMENT = $PRIMARY$
		CLEAR_MODIFIER_1 = $MODIFIER$
		CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$
	}
	ch_clear_movement_pressure_if_present = {
		MOVEMENT = $FALLBACK_1$
		CLEAR_MODIFIER_1 = $MODIFIER$
		CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$
	}
	ch_clear_movement_pressure_if_present = {
		MOVEMENT = $FALLBACK_2$
		CLEAR_MODIFIER_1 = $MODIFIER$
		CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$
	}

	if = {
		limit = {
			any_political_movement = {
				is_political_movement_type = $PRIMARY$
			}
		}
		ch_apply_movement_pressure_if_present = {
			MOVEMENT = $PRIMARY$
			MODIFIER = $MODIFIER$
			CLEAR_MODIFIER_1 = $MODIFIER$
			CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$
			MONTHS = $MONTHS$
			DECAYING = $DECAYING$
			MULT = $MULT$
		}
	}
	else_if = {
		limit = {
			any_political_movement = {
				is_political_movement_type = $FALLBACK_1$
			}
		}
		ch_apply_movement_pressure_if_present = {
			MOVEMENT = $FALLBACK_1$
			MODIFIER = $FALLBACK_MODIFIER$
			CLEAR_MODIFIER_1 = $MODIFIER$
			CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$
			MONTHS = $MONTHS$
			DECAYING = $DECAYING$
			MULT = $MULT$
		}
	}
	else_if = {
		limit = {
			any_political_movement = {
				is_political_movement_type = $FALLBACK_2$
			}
		}
		ch_apply_movement_pressure_if_present = {
			MOVEMENT = $FALLBACK_2$
			MODIFIER = $FALLBACK_MODIFIER$
			CLEAR_MODIFIER_1 = $MODIFIER$
			CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$
			MONTHS = $MONTHS$
			DECAYING = $DECAYING$
			MULT = $MULT$
		}
	}
}

# Parameters are ch_apply_primary_or_fallback_movement_pressure's.

	ch_apply_movement_pressure_if_present = {

		CLEAR_MODIFIER_1 = $MODIFIER$
		CLEAR_MODIFIER_2 = $FALLBACK_MODIFIER$

# ---- HEGEMON MODEL → MOVEMENT PRESSURE ----
# Scope: Country (the target being influenced)
# Requires scope:cultural_hegemon to be set. Classifies it afresh with
# ch_set_political_model, then boosts the matching political movements here.
# Parameters: $MODIFIER$ (primary modifier name), $FALLBACK_MODIFIER$
# (fallback modifier name), $MONTHS$ (duration, a literal number), $DECAYING$
# (yes/no), $MULT$ (modifier multiplier: 1, or a script value name)
#
# Structure: EXCLUSIVE model branch (one primary political model, fallbacks
# only when the primary movement is absent — they never stack as spillover)
# PLUS NON-EXCLUSIVE social policy checks (independent if blocks — e.g.
# tolerant + feminist + environmentalist can all fire). A Mixed / Other
# hegemon (code 12) pushes only its social policies.
ch_apply_hegemon_movement_pressure = {
	if = {
		limit = { NOT = { exists = scope:cultural_hegemon } }
	}
	else = {
		scope:cultural_hegemon = { ch_set_political_model = yes }

		# ================================================================
		# PART 1: EXCLUSIVE — the hegemon's political model
		# ================================================================

		# 1 Communist / Vanguardist
		if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 1 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_communist
				FALLBACK_1 = movement_socialist
				FALLBACK_2 = movement_labor
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 2 Anarchist
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 2 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_anarchist
				FALLBACK_1 = movement_socialist
				FALLBACK_2 = movement_labor
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 3 Fascist / Ethnonationalist
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 3 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_fascist
				FALLBACK_1 = movement_reactionary
				FALLBACK_2 = movement_corporatist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 4 Corporatist
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 4 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_corporatist
				FALLBACK_1 = movement_reactionary
				FALLBACK_2 = movement_fascist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 5 Socialist / Cooperative
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 5 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_socialist
				FALLBACK_1 = movement_labor
				FALLBACK_2 = movement_communist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 6 Absolutist monarchist
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 6 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_royalist_absolutist
				FALLBACK_1 = movement_royalist_constitutional
				FALLBACK_2 = movement_reactionary
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 7 Constitutional monarchist
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 7 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_royalist_constitutional
				FALLBACK_1 = movement_liberal
				FALLBACK_2 = movement_royalist_absolutist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 8 Theocratic / Religious
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 8 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_religious_majority
				FALLBACK_1 = movement_reactionary
				FALLBACK_2 = movement_royalist_absolutist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 9 Reactionary / Authoritarian
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 9 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_reactionary
				FALLBACK_1 = movement_religious_majority
				FALLBACK_2 = movement_fascist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 10 Conservative military junta: inspires reactionary movements abroad
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 10 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_reactionary
				FALLBACK_1 = movement_fascist
				FALLBACK_2 = movement_corporatist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 11 Liberal / Progressive democratic
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 11 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_liberal
				FALLBACK_1 = movement_civil_rights
				FALLBACK_2 = movement_minority_rights
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 13 Technocratic. Positivists need the positivism global and the
		# target's Psychiatry; transhumanists are the late-game successor.
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 13 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_positivist
				FALLBACK_1 = movement_modernizer
				FALLBACK_2 = movement_transhumanist
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 14 Republican. Radicals only form outside republics, so this lands in
		# monarchies and theocracies.
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 14 } }
			ch_apply_primary_or_fallback_movement_pressure = {
				PRIMARY = movement_radical
				FALLBACK_1 = movement_liberal
				FALLBACK_2 = movement_modernizer
				MODIFIER = $MODIFIER$
				FALLBACK_MODIFIER = $FALLBACK_MODIFIER$
				MONTHS = $MONTHS$
				DECAYING = $DECAYING$
				MULT = $MULT$
			}
		}
		# 15 Developmentalist junta. Modernizers only form where the target
		# still has laws in need of critical modernization.
		else_if = {
			limit = { scope:cultural_hegemon = { var:ch_model = 15 } }

				FALLBACK_1 = movement_land_reform

		if = {
			limit = {
				scope:cultural_hegemon = { has_law_or_variant = law_type:law_womens_suffrage }
			}

		}

		# Tolerant hegemon → boosts civil rights and minority rights movements
		if = {
			limit = {
				scope:cultural_hegemon = { has_law_or_variant = law_type:law_multicultural }
			}
			ch_apply_single_movement_pressure = { MOVEMENT = movement_civil_rights MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
			ch_apply_single_movement_pressure = { MOVEMENT = movement_minority_rights MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
		}

		# Environmentalist hegemon → boosts environmental movements
		if = {
			limit = {
				scope:cultural_hegemon = { has_law_or_variant = law_type:law_ministry_of_the_environment }
			}
			ch_apply_single_movement_pressure = { MOVEMENT = movement_environmental MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
		}

		# Peaceful hegemon → boosts anti-war movements
		if = {
			limit = {
				scope:cultural_hegemon = {
					is_at_war = no
					NOT = { has_law_or_variant = law_type:law_mass_conscription }
				}
			}
			ch_apply_single_movement_pressure = { MOVEMENT = movement_anti_war MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
		}

		# Abolitionist hegemon → boosts anti-slavery movements (only if target has slavery)
		if = {
			limit = {
				scope:cultural_hegemon = { has_law_or_variant = law_type:law_slavery_banned }
				NOT = { has_law_or_variant = law_type:law_slavery_banned }
			}
			ch_apply_single_movement_pressure = { MOVEMENT = movement_anti_slavery MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
		}

		# Pro-labor hegemon → boosts labor movements
		if = {
			limit = {
				scope:cultural_hegemon = {
					OR = {
						has_law_or_variant = law_type:law_worker_protections
						has_law_or_variant = law_type:law_regulatory_bodies
					}
				}
			}
			ch_apply_single_movement_pressure = { MOVEMENT = movement_labor MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
		}

		# Land reform hegemon → boosts land reform movements
		if = {
			limit = {
				scope:cultural_hegemon = {
					OR = {
						# Homesteading is itself a variant (of Peasant Proprietorship),
						# and has_law_or_variant only takes a parent law.
						has_law = law_type:law_homesteading
						has_law_or_variant = law_type:law_collectivized_agriculture
					}
				}
			}
			ch_apply_single_movement_pressure = { MOVEMENT = movement_land_reform MODIFIER = $MODIFIER$ FALLBACK_MODIFIER = $FALLBACK_MODIFIER$ MONTHS = $MONTHS$ DECAYING = $DECAYING$ MULT = $MULT$ }
		}
	}
}

# ---- BASELINE MODEL PRESSURE ----
# Scope: Country. Called for every journal-entry holder from
# ch_yearly_global_update, which saves the rank-1 hegemon as
# scope:cultural_hegemon first.
# Strips the baseline modifiers from every movement, then — for a country that
# trails the hegemon (it carries cultural_hegemony_foreign_benchmark) — re-applies
# them to the hegemon model's movements, scaled by that model's share of world
# culture (ch_model_pressure_mult). Non-decaying, and 13 months long so it never
# lapses between rebuilds; the event-driven pressure (ch_hegemon_ideological_pressure)
# uses different modifier names and stacks on top as a temporary spike.
ch_refresh_baseline_model_pressure = {
	every_political_movement = {
		limit = {
			OR = {
				has_modifier = ch_hegemon_model_pressure
				has_modifier = ch_hegemon_model_pressure_weak
			}
		}
		if = {
			limit = { has_modifier = ch_hegemon_model_pressure }
			remove_modifier = ch_hegemon_model_pressure
		}
		if = {
			limit = { has_modifier = ch_hegemon_model_pressure_weak }
			remove_modifier = ch_hegemon_model_pressure_weak
		}
	}
	if = {
		limit = {
			exists = scope:cultural_hegemon
			NOT = { this = scope:cultural_hegemon }
			has_journal_entry = je_cultural_hegemony
			has_modifier = cultural_hegemony_foreign_benchmark
			# A trigger needs the variable on the LEFT; the threshold is the
			# same script value ch_model_pressure_mult tests.
			has_global_variable = ch_rank_1_ideology_share
			global_var:ch_rank_1_ideology_share >= ch_model_pressure_min_share
		}
		ch_apply_hegemon_movement_pressure = { MODIFIER = ch_hegemon_model_pressure FALLBACK_MODIFIER = ch_hegemon_model_pressure_weak MONTHS = 13 DECAYING = no MULT = ch_model_pressure_mult }
	}
}

# ---- MODEL-CHANGE EVENTS ----
# Scope: any. Called once per rebuild from ch_yearly_global_update, after the
# census and the rank-1 summary. Both events react to a CHANGE between two
# rebuilds, which a yearly random_events pick would miss half the time (the
# census pair is overwritten every rebuild), so they are fired from here.
ch_fire_model_change_events = {
	# The Model Falters: the world's leading cultural power no longer runs the
	# model it ran at the last rebuild (it changed government, or lost the top
	# spot to a country running something else). Every country still running
	# the old model hears of it. Mixed / Other (12) is not a model to lose.
	if = {
		limit = {
			has_global_variable = ch_rank_1_ideology_prev
			has_global_variable = ch_rank_1_ideology
			NOT = { global_var:ch_rank_1_ideology = global_var:ch_rank_1_ideology_prev }
			NOT = { global_var:ch_rank_1_ideology_prev = 12 }
		}
		every_country = {
			limit = {
				has_journal_entry = je_cultural_hegemony
				has_variable = ch_model_census
				var:ch_model_census = global_var:ch_rank_1_ideology_prev
			}
			trigger_event = { id = cultural_hegemony.19 days = 14 }
		}
	}

	# Domino: a country has just switched to the hegemon's model. Its
	# neighbours that run something else hear of it. The neighbour test is
	# state adjacency (vanilla has no country-level neighbour iterator). A
	# dated variable carries the switcher to the event and stops a country
	# with two switching neighbours from getting it twice.
	every_country = {
		limit = { ch_switched_to_hegemon_model = yes }
		save_scope_as = ch_domino_switcher
		every_country = {
			limit = {
				has_journal_entry = je_cultural_hegemony
				NOT = { this = scope:ch_domino_switcher }
				NOT = { this = global_var:ch_rank_1 }
				has_variable = ch_model_census
				NOT = { var:ch_model_census = global_var:ch_rank_1_ideology }
				NOT = { has_variable = ch_domino_source }
				any_scope_state = {
					any_neighbouring_state = { owner = scope:ch_domino_switcher }
				}
			}
			set_variable = { name = ch_domino_source value = scope:ch_domino_switcher days = 90 }
			trigger_event = { id = cultural_hegemony.20 days = 14 }
		}
	}
}
