# Global Warming — shared policy effects.
#
# Each `gw_effect_<button>` holds the body that used to live inline in the
# matching scripted_button's `effect` block in
# common/scripted_buttons/global_warming_buttons.txt. The journal-entry button
# and the widget's scripted GUI both call these.
#
# National policies modify only the acting country's journal entry. Tariffs
# alone belong to the market leader and write native customs settings.

	custom_tooltip = {
		text = "GW_CARBON_TAX_DESC"
		je:je_global_warming = { add_modifier = { name = carbon_tax_modifier } }
	}
}

	}

# Renewable Investment's running cost (renewable_investment_cost_modifier,
# 0.5% of GDP a year). The policy modifier sits on the journal entry; the cost
# sits on the country because its multiplier reads gdp, so the two are kept in
# step here and by the monthly sweep below. Every way the policy goes on (the
# button, the widget, treaty 109) applies the cost through
# gw_renewable_investment_cost_apply; the sweep corrects whatever else added
# or removed the policy modifier (the debug console, a lost entry).
# A same-name add_modifier stacks, so the apply removes the copy it already
# holds first: the monthly sweep runs it for every adopter every month, and
# without the removal each pass would add another full-size weekly cost.
# Scope: country
gw_renewable_investment_cost_apply = {
	remove_modifier_if_exists_effect = { MODIFIER = renewable_investment_cost_modifier }
	add_modifier = {
		name = renewable_investment_cost_modifier
		multiplier = gw_renewable_investment_cost_value
	}
}

# Scope: country. Reads the policy's actual state, so it cannot be called from
# the effect that has just added or removed the policy modifier: an
# add_modifier is not visible in the same effect block.
gw_renewable_investment_cost_sync = {
	if = {
		limit = { gw_policy_renewable_investment_active = yes }
		gw_renewable_investment_cost_apply = yes
	}
	else_if = {
		limit = { has_modifier = renewable_investment_cost_modifier }
		remove_modifier = renewable_investment_cost_modifier
	}
}

# Monthly, from global_warming_events_on_action's global scope: the single
# refresh site for the policy running costs, so the weekly amount follows GDP.
# Runs with the climate rule off too: a policy adopted earlier keeps its
# bonuses, so it keeps its bill. The second limit clause reaches a country
# that has lost its journal entry while still holding the cost.
# Scope: any
gw_refresh_policy_costs_effect = {
	every_country = {
		limit = {
			OR = {
				has_journal_entry = je_global_warming
				has_modifier = renewable_investment_cost_modifier
				has_modifier = aerosol_injection_cost_modifier
			}
		}
		gw_renewable_investment_cost_sync = yes
		gw_aerosol_injection_cost_sync = yes
	}
}

# DISPLAY SNAPSHOTS
# ============================================================================
# The journal entry's figures used to be computed live in loc, every frame the
# panel was open. These three effects move that work onto pulses that were
# already doing it, so the widget only ever reads stored numbers.
#
# CADENCE IS DELIBERATE. Emissions are re-based once a year because that is when
# the simulation actually moves them: `global_warming_update_on_action` runs on
# on_yearly_pulse_state and is the only thing that changes
# global_var:greenhouse_gas_emissions. Snapshotting monthly would multiply the
# market sweep's cost by twelve to show a number that had not changed. The
# widget and the docs say "re-based each January" rather than implying live data.
# ============================================================================

# Country scope, market leader: retain market snapshots for existing market
# overview, map and UN readers. The world accumulates country snapshots instead.
gw_snapshot_market_emissions_effect = {
	hidden_effect = {
		set_variable = {
			name = gw_disp_market_emis
			value = market_greenhouse_gas_emissions_script_value
		}
		set_variable = {
			name = gw_disp_capture
			value = market_carbon_capture_script_value
		}
	}
}

gw_snapshot_country_emissions_effect = {
	hidden_effect = {

			name = gw_disp_country_emis
			value = country_greenhouse_gas_emissions_script_value
		}
	}
}

# The exact annual contribution to the world total, including negative years.
# Existing market totals stay untouched: they cannot be split retrospectively.
# Country totals begin on the first yearly capital pulse after this update,
# follow the country across market changes, and inherit through revolutions.
gw_accumulate_country_emissions_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = gw_disp_country_emis_cum } }
			set_variable = { name = gw_disp_country_emis_cum value = 0 }
		}
		change_variable = { name = gw_disp_country_emis_cum add = var:gw_disp_country_emis }
	}
}

# Re-base the annual world emissions figure each January, as the change in the
# cumulative global variable since last January. O(1), and more truthful than
# re-summing every market: it reports what actually accumulated rather than
# re-estimating the current rate.
#
# `month = 0` is January (te_history_values.txt documents Jan = 0 ... Dec = 11).
# Scope: any (called from the global monthly pulse)
gw_rebase_annual_emissions_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_variable = gw_g_emis_prev } }
			set_global_variable = {
				name = gw_g_emis_prev
				value = 0
			}
		}
		if = {
			limit = {
				month = 0
				has_global_variable = greenhouse_gas_emissions
			}
			set_global_variable = {
				name = gw_g_global_emis
				value = global_var:greenhouse_gas_emissions
			}
			change_global_variable = {
				name = gw_g_global_emis
				subtract = global_var:gw_g_emis_prev
			}
			set_global_variable = {
				name = gw_g_emis_prev
				value = global_var:greenhouse_gas_emissions
			}
		}
	}
}

# Count how many countries run each of the eleven policies, in ONE country
# sweep that fills all eleven counters. Replaces eight separate sweeping script values
# that were rendered live in the reason text.
#
# The same sweep stores the mean standard of living over these countries
# (gw_g_mean_sol), which the AI's climate wills measure wealth against
# (gw_ai_sig_wealth in global_warming_ai_values.txt). It is the only world
# figure those scores need, so it rides this pass instead of adding one. World
# GDP and the aerosol deployers' GDP ride it too, for the aerosol mask's
# target (gw_aerosol_update_mask_effect, which runs after this).
# Scope: any (called from the global monthly pulse)
gw_refresh_global_counts_effect = {
	hidden_effect = {
		set_global_variable = { name = gw_g_n_fossil_fuel_tariffs value = 0 }
		set_global_variable = { name = gw_g_n_carbon_tax value = 0 }
		set_global_variable = { name = gw_g_n_renewable_investment value = 0 }
		set_global_variable = { name = gw_g_n_emission_standards value = 0 }
		set_global_variable = { name = gw_g_n_climate_adaptation value = 0 }
		set_global_variable = { name = gw_g_n_reforestation value = 0 }
		set_global_variable = { name = gw_g_n_carbon_removal value = 0 }
		set_global_variable = { name = gw_g_n_public_transit value = 0 }
		set_global_variable = { name = gw_g_n_fossil_fuel_divestment value = 0 }
		set_global_variable = { name = gw_g_n_green_building_codes value = 0 }
		set_global_variable = { name = gw_g_n_aerosol_injection value = 0 }
		set_global_variable = { name = gw_g_aerosol_gdp value = 0 }
		set_global_variable = { name = gw_g_gdp_sum value = 0 }
		set_global_variable = { name = gw_g_sol_sum value = 0 }
		set_global_variable = { name = gw_g_sol_n value = 0 }
		every_country = {
			limit = { has_journal_entry = je_global_warming }
			if = {
				limit = { je:je_global_warming ?= { has_modifier = fossil_fuel_tariffs_modifier } }
				change_global_variable = { name = gw_g_n_fossil_fuel_tariffs add = 1 }
			}
			change_global_variable = { name = gw_g_sol_sum add = average_sol }
			change_global_variable = { name = gw_g_sol_n add = 1 }
			# World GDP over the same countries: the aerosol mask is the
			# deployers' share of it (global_warming_aerosol_values.txt).
			change_global_variable = { name = gw_g_gdp_sum add = gdp }
			if = {
				limit = { gw_policy_carbon_tax_active = yes }
				change_global_variable = { name = gw_g_n_carbon_tax add = 1 }
			}
			if = {
				limit = { gw_policy_renewable_investment_active = yes }
				change_global_variable = { name = gw_g_n_renewable_investment add = 1 }
			}
			if = {
				limit = { gw_policy_emission_standards_active = yes }
				change_global_variable = { name = gw_g_n_emission_standards add = 1 }
			}
			if = {
				limit = { gw_policy_climate_adaptation_active = yes }
				change_global_variable = { name = gw_g_n_climate_adaptation add = 1 }
			}
			if = {
				limit = { gw_policy_reforestation_active = yes }
				change_global_variable = { name = gw_g_n_reforestation add = 1 }
			}
			if = {
				limit = { gw_policy_carbon_removal_active = yes }
				change_global_variable = { name = gw_g_n_carbon_removal add = 1 }
			}
			if = {
				limit = { gw_policy_public_transit_active = yes }
				change_global_variable = { name = gw_g_n_public_transit add = 1 }
			}
			if = {
				limit = { gw_policy_fossil_fuel_divestment_active = yes }
				change_global_variable = { name = gw_g_n_fossil_fuel_divestment add = 1 }
			}
			if = {
				limit = { gw_policy_green_building_codes_active = yes }
				change_global_variable = { name = gw_g_n_green_building_codes add = 1 }
			}
			if = {
				limit = { gw_policy_aerosol_injection_active = yes }
				change_global_variable = { name = gw_g_n_aerosol_injection add = 1 }
				change_global_variable = { name = gw_g_aerosol_gdp add = gdp }
			}
		}
		if = {
			limit = { global_var:gw_g_sol_n > 0 }
			set_global_variable = {
				name = gw_g_mean_sol
				value = {
					value = global_var:gw_g_sol_sum
					divide = global_var:gw_g_sol_n
				}
			}
		}
	}
}

# Top Emitters (display only)
# ---------------------------------------------------------------------------
# Top Emitters ranks countries by their last yearly country snapshot. Players
# keep the five largest emitters' capitals in a list, largest first, rebuilt
# monthly: the panel draws it with a datamodel (GetList('gw_te_top'), one
# gw_emitter_row per item, datacontext Scope.GetState), the Demographics States
# table's shape (GW-11, 2026-10-27). The capital is only a GUI bridge; the
# numbers belong to its country. Appending in ordered_country's order keeps
# the stored order the rank (scripting_best_practices.md: only a mid-list
# removal reorders a list).
gw_top_emitters_refresh = {
	hidden_effect = {
		save_scope_as = gw_te_viewer
		save_scope_as = gw_te_own_country
		set_variable = { name = gw_emitters_total value = 0 }
		every_country = {
			limit = { gw_is_ranked_emitter = yes }
			scope:gw_te_viewer = {
				change_variable = { name = gw_emitters_total add = 1 }
			}
		}
		if = {
			limit = { has_variable_list = gw_te_top }
			clear_variable_list = gw_te_top
		}
		ordered_country = {
			limit = { gw_is_ranked_emitter = yes }
			order_by = gw_disp_own_emis
			max = 5
			check_range_bounds = no
			capital ?= {
				save_temporary_scope_as = gw_te_capital
				scope:gw_te_viewer = {
					add_to_variable_list = { name = gw_te_top target = scope:gw_te_capital }
				}
			}
		}
		# The fifth country's emissions: "+N more"'s hover lists those below
		# it (gw_top_emitters_sgui). An ordered_ position past the end clamps
		# to the last element instead of skipping, so the count comes first.
		if = {
			limit = { var:gw_emitters_total > 5 }
			ordered_country = {
				limit = { gw_is_ranked_emitter = yes }
				order_by = gw_disp_own_emis
				position = 4
				save_temporary_scope_as = gw_te_fifth
			}
			set_variable = {
				name = gw_emitters_fifth_emis
				value = scope:gw_te_fifth.var:gw_disp_country_emis
			}
		}
		else_if = {
			limit = { has_variable = gw_emitters_fifth_emis }
			remove_variable = gw_emitters_fifth_emis
		}
		# Our country's place: one more than the countries emitting more than it.
		set_variable = { name = gw_emitters_own_rank value = 0 }
		if = {
			limit = {
				exists = scope:gw_te_own_country
				scope:gw_te_own_country = { gw_is_ranked_emitter = yes }
			}
			set_variable = {
				name = gw_emitters_own_emis
				value = scope:gw_te_own_country.var:gw_disp_country_emis
			}
			set_variable = { name = gw_emitters_own_rank value = 1 }
			every_country = {
				limit = {
					gw_is_ranked_emitter = yes
					var:gw_disp_country_emis > scope:gw_te_viewer.var:gw_emitters_own_emis
				}
				scope:gw_te_viewer = {
					change_variable = { name = gw_emitters_own_rank add = 1 }
				}
			}
			remove_variable = gw_emitters_own_emis
		}
		# The five slot variables the rows read before the list (a save from
		# before GW-11 still holds them).
		gw_top_emitters_drop_slots = yes
		gw_top_emitters_selfcheck = yes
	}
}

		remove_variable = gw_emitter_1
	}
	if = {

	if = {

# TE_SELF_CHECK gw_top_emitters_list (docs/testing/global-warming-playtest.md):
# the first time a player's list is built with an emitter in the world, it
# must hold min(5, emitters) capitals whose owners' emissions never rise down
# the list. It rests on ordered_country with max and check_range_bounds = no
# walking the countries largest first, and on add_to_variable_list keeping
# that order, which the rows draw as is. Once a game. Scope: the viewer, in
# gw_top_emitters_refresh's hidden_effect (scope:gw_te_viewer set).
gw_top_emitters_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_selfcheck_gw_top_emitters_done }
			has_variable_list = gw_te_top
			var:gw_emitters_total > 0
		}
		set_global_variable = { name = te_selfcheck_gw_top_emitters_done value = 1 }
		set_variable = { name = gw_sc_rows value = 0 }
		set_variable = { name = gw_sc_rises value = 0 }
		set_variable = { name = gw_sc_expected value = var:gw_emitters_total }
		if = {
			limit = { var:gw_sc_expected > 5 }
			set_variable = { name = gw_sc_expected value = 5 }
		}
		every_in_list = {
			variable = gw_te_top
			save_temporary_scope_as = gw_sc_row
			scope:gw_te_viewer = {
				change_variable = { name = gw_sc_rows add = 1 }
				if = {
					limit = {
						has_variable = gw_sc_last
						scope:gw_sc_row.owner.var:gw_disp_country_emis > var:gw_sc_last
					}
					change_variable = { name = gw_sc_rises add = 1 }
				}
				set_variable = { name = gw_sc_last value = scope:gw_sc_row.owner.var:gw_disp_country_emis }
			}
		}
		if = {
			limit = {
				var:gw_sc_rows = var:gw_sc_expected
				var:gw_sc_rises = 0
			}
			debug_log = "TE_SELF_CHECK OK gw_top_emitters_list rows=[THIS.Var('gw_sc_rows').GetValue|0] expected=[THIS.Var('gw_sc_expected').GetValue|0] rises=[THIS.Var('gw_sc_rises').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN gw_top_emitters_list rows=[THIS.Var('gw_sc_rows').GetValue|0] expected=[THIS.Var('gw_sc_expected').GetValue|0] rises=[THIS.Var('gw_sc_rises').GetValue|0] (the Top Emitters list is short or out of order: an emitter without a capital, ordered_country's max not counting from the top, or the list not keeping its order; see docs/testing/global-warming-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_variable = gw_sc_rows
		remove_variable = gw_sc_rises
		remove_variable = gw_sc_expected
		if = {
			limit = { has_variable = gw_sc_last }
			remove_variable = gw_sc_last
		}
	}
}

# Ownership cleanup only; rate modifiers themselves need no reassertion.
# Run with climate disabled too so the programme's ongoing cost is released.
gw_refresh_fossil_fuel_tariffs_effect = {
	if = {
		limit = {
			je:je_global_warming ?= { has_modifier = fossil_fuel_tariffs_modifier }
			OR = {
				NOT = { te_tax_owns_market = yes }
				NOT = { has_game_rule = global_warming_enabled }
			}
		}
		gw_drop_fossil_fuel_tariffs_effect = yes
	}
}

gw_refresh_tariff_policies_effect = {
	every_country = {
		limit = { has_journal_entry = je_global_warming }
		gw_refresh_fossil_fuel_tariffs_effect = yes
	}
}


# ============================================================================
# STRATOSPHERIC AEROSOL INJECTION (Geoengineering, era 13 spec § 7.5)
# ============================================================================
# A national policy on the journal entry, mirrored in a country variable so a
# civil war's winner keeps it (as Carbon Removal Support does), with a running
# cost on the country (as Renewable Investment's). What it buys is the world's
# aerosol mask (global_warming_aerosol_values.txt), moved monthly by
# gw_aerosol_update_mask_effect below.

# The running cost and the objectors' relations sit outside the first
# custom_tooltip: the cost prints its scaled Expenses line, and the relations
# line says once what the hidden loop does.
gw_effect_aerosol_injection = {
	custom_tooltip = {
		text = "GW_AEROSOL_INJECTION_DESC"
		hidden_effect = { set_variable = { name = gw_aerosol_injection_policy value = yes } }
		je:je_global_warming = { add_modifier = { name = aerosol_injection_modifier } }
	}
	gw_aerosol_injection_cost_apply = yes
	custom_tooltip = {
		text = gw_aerosol_objection_tt
		save_scope_as = gw_aerosol_deployer
		every_country = {
			limit = {
				NOT = { THIS = scope:gw_aerosol_deployer }
				gw_aerosol_objects = yes
			}
			change_relations = { country = scope:gw_aerosol_deployer value = gw_aerosol_objection_at_start }
		}
	}
}

gw_effect_remove_aerosol_injection = {
	custom_tooltip = {
		text = "GW_REMOVE_AEROSOL_INJECTION_DESC"
		hidden_effect = { remove_variable = gw_aerosol_injection_policy }
		je:je_global_warming = { remove_modifier = aerosol_injection_modifier }
		remove_modifier = aerosol_injection_cost_modifier
	}
}

# A revolution inherits the country variable; rebuild the journal-entry
# modifier. The cost follows on the next monthly sweep. From the journal
# entry's monthly pulse.
gw_restore_aerosol_injection_policy_effect = {
	if = {
		limit = {
			has_variable = gw_aerosol_injection_policy
			NOT = { gw_policy_aerosol_injection_active = yes }
		}
		je:je_global_warming = { add_modifier = { name = aerosol_injection_modifier } }
	}
}

# The running cost, 1% of GDP a year, as Renewable Investment's: the removal
# first, since a same-name add_modifier stacks. Scope: country.
gw_aerosol_injection_cost_apply = {
	remove_modifier_if_exists_effect = { MODIFIER = aerosol_injection_cost_modifier }
	add_modifier = {
		name = aerosol_injection_cost_modifier
		multiplier = gw_aerosol_injection_cost_value
	}
}

# From gw_refresh_policy_costs_effect, monthly. Scope: country.
gw_aerosol_injection_cost_sync = {
	if = {
		limit = { gw_policy_aerosol_injection_active = yes }
		gw_aerosol_injection_cost_apply = yes
	}
	else_if = {
		limit = { has_modifier = aerosol_injection_cost_modifier }
		remove_modifier = aerosol_injection_cost_modifier
	}
}

# The aerosol mask's monthly step, the one writer of gw_g_aerosol_mask: the
# target from this month's counts (gw_refresh_global_counts_effect runs just
# before), the mask a step towards it, then termination shock and, each
# January, the objectors' relations. From global_warming_events_on_action,
# under the climate rule. Scope: any (the global monthly pulse).
gw_aerosol_update_mask_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_variable = gw_g_aerosol_mask } }
			set_global_variable = { name = gw_g_aerosol_mask value = 0 }
		}
		set_global_variable = { name = gw_g_aerosol_target value = gw_aerosol_target_value }
		set_global_variable = { name = gw_g_aerosol_mask value = gw_aerosol_mask_next }

		# Termination shock: nobody is flying it any more, and enough of the
		# warming it hid is coming back. Once until someone starts again.
		if = {
			limit = {
				gw_aerosol_deployers < 1
				gw_aerosol_mask >= gw_aerosol_shock_min_mask
				NOT = { has_global_variable = gw_g_aerosol_shock_fired }
			}
			set_global_variable = { name = gw_g_aerosol_shock_fired value = 1 }
			every_country = {
				limit = { NOT = { is_country_type = decentralized } }
				trigger_event = { id = environmentalism_events.24 }
			}
		}
		else_if = {
			limit = {
				gw_aerosol_deployers > 0
				has_global_variable = gw_g_aerosol_shock_fired
			}
			remove_global_variable = gw_g_aerosol_shock_fired
		}

		# The objectors' relations with each country running it, once a year.
		# `month = 0` is January.
		if = {
			limit = {
				month = 0
				gw_aerosol_deployers > 0
			}
			every_country = {
				limit = { gw_policy_aerosol_injection_active = yes }
				save_temporary_scope_as = gw_aerosol_deployer
				every_country = {
					limit = {
						NOT = { THIS = scope:gw_aerosol_deployer }
						gw_aerosol_objects = yes
					}
					change_relations = { country = scope:gw_aerosol_deployer value = gw_aerosol_objection_yearly }
				}
			}
		}
	}
}

# The climate rule is off: no warming, so nothing to mask (the else branch of
# global_warming_events_on_action, which also holds the emissions at 0).
gw_aerosol_clear_mask_effect = {
	if = {
		limit = { has_global_variable = gw_g_aerosol_mask }
		remove_global_variable = gw_g_aerosol_mask
	}
	if = {
		limit = { has_global_variable = gw_g_aerosol_target }
		remove_global_variable = gw_g_aerosol_target
	}
}
