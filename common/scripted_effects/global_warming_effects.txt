# ============================================================================
# DISPLAY SNAPSHOTS
# ============================================================================
# The journal entry's figures used to be computed live in loc, every frame the
# panel was open. These three effects move that work onto pulses that were
# already doing it, so the widget only ever reads stored numbers.
#
# CADENCE IS DELIBERATE. Emissions are re-based once a year because that is when
# the simulation actually moves them: `global_warming_update_on_action` runs on
# on_yearly_pulse_state and is the only thing that changes
# global_var:greenhouse_gas_emissions. Snapshotting monthly would multiply the
# market sweep's cost by twelve to show a number that had not changed. The
# widget and the docs say "re-based each January" rather than implying live data.
# ============================================================================

# Country scope, market leader: retain market snapshots for existing market
# overview, map and UN readers. The world accumulates country snapshots instead.
gw_snapshot_market_emissions_effect = {
	hidden_effect = {
		set_variable = {
			name = gw_disp_market_emis
			value = market_greenhouse_gas_emissions_script_value
		}
		set_variable = {
			name = gw_disp_capture
			value = market_carbon_capture_script_value
		}
	}
}

gw_snapshot_country_emissions_effect = {
	hidden_effect = {
		set_variable = {
			name = gw_disp_country_emis
			value = country_greenhouse_gas_emissions_script_value
		}
	}
}

# The exact annual contribution to the world total, including negative years.
# Existing market totals stay untouched: they cannot be split retrospectively.
# Country totals begin on the first yearly capital pulse after this update,
# follow the country across market changes, and inherit through revolutions.
gw_accumulate_country_emissions_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = gw_disp_country_emis_cum } }
			set_variable = { name = gw_disp_country_emis_cum value = 0 }
		}
		change_variable = { name = gw_disp_country_emis_cum add = var:gw_disp_country_emis }
	}
}

# Re-base the annual world emissions figure each January, as the change in the
# cumulative global variable since last January. O(1), and more truthful than
# re-summing every market: it reports what actually accumulated rather than
# re-estimating the current rate.
#
# `month = 0` is January (te_history_values.txt documents Jan = 0 ... Dec = 11).
# Scope: any (called from the global monthly pulse)
gw_rebase_annual_emissions_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_variable = gw_g_emis_prev } }
			set_global_variable = {
				name = gw_g_emis_prev
				value = 0
			}
		}
		if = {
			limit = {
				month = 0
				has_global_variable = greenhouse_gas_emissions
			}
			set_global_variable = {
				name = gw_g_global_emis
				value = global_var:greenhouse_gas_emissions
			}
			change_global_variable = {
				name = gw_g_global_emis
				subtract = global_var:gw_g_emis_prev
			}
			set_global_variable = {
				name = gw_g_emis_prev
				value = global_var:greenhouse_gas_emissions
			}
		}
	}
}

# Count how many countries run each of the ten policies, in ONE country sweep
# that fills all ten counters. Replaces eight separate sweeping script values
# that were rendered live in the reason text.
#
# The same sweep stores the mean standard of living over these countries
# (gw_g_mean_sol), which the AI's climate wills measure wealth against
# (gw_ai_sig_wealth in global_warming_ai_values.txt). It is the only world
# figure those scores need, so it rides this pass instead of adding one.
# Scope: any (called from the global monthly pulse)
gw_refresh_global_counts_effect = {
	hidden_effect = {
		set_global_variable = { name = gw_g_n_fossil_fuel_tariffs value = 0 }
		set_global_variable = { name = gw_g_n_carbon_tax value = 0 }
		set_global_variable = { name = gw_g_n_renewable_investment value = 0 }
		set_global_variable = { name = gw_g_n_emission_standards value = 0 }
		set_global_variable = { name = gw_g_n_climate_adaptation value = 0 }
		set_global_variable = { name = gw_g_n_reforestation value = 0 }
		set_global_variable = { name = gw_g_n_carbon_removal value = 0 }
		set_global_variable = { name = gw_g_n_public_transit value = 0 }
		set_global_variable = { name = gw_g_n_fossil_fuel_divestment value = 0 }
		set_global_variable = { name = gw_g_n_green_building_codes value = 0 }
		set_global_variable = { name = gw_g_sol_sum value = 0 }
		set_global_variable = { name = gw_g_sol_n value = 0 }
		every_country = {
			limit = { has_journal_entry = je_global_warming }
			if = {
				limit = { je:je_global_warming ?= { has_modifier = fossil_fuel_tariffs_modifier } }
				change_global_variable = { name = gw_g_n_fossil_fuel_tariffs add = 1 }
			}
			change_global_variable = { name = gw_g_sol_sum add = average_sol }
			change_global_variable = { name = gw_g_sol_n add = 1 }
			if = {
				limit = { gw_policy_carbon_tax_active = yes }
				change_global_variable = { name = gw_g_n_carbon_tax add = 1 }
			}
			if = {
				limit = { gw_policy_renewable_investment_active = yes }
				change_global_variable = { name = gw_g_n_renewable_investment add = 1 }
			}
			if = {
				limit = { gw_policy_emission_standards_active = yes }
				change_global_variable = { name = gw_g_n_emission_standards add = 1 }
			}
			if = {
				limit = { gw_policy_climate_adaptation_active = yes }
				change_global_variable = { name = gw_g_n_climate_adaptation add = 1 }
			}
			if = {
				limit = { gw_policy_reforestation_active = yes }
				change_global_variable = { name = gw_g_n_reforestation add = 1 }
			}
			if = {
				limit = { gw_policy_carbon_removal_active = yes }
				change_global_variable = { name = gw_g_n_carbon_removal add = 1 }
			}
			if = {
				limit = { gw_policy_public_transit_active = yes }
				change_global_variable = { name = gw_g_n_public_transit add = 1 }
			}
			if = {
				limit = { gw_policy_fossil_fuel_divestment_active = yes }
				change_global_variable = { name = gw_g_n_fossil_fuel_divestment add = 1 }
			}
			if = {
				limit = { gw_policy_green_building_codes_active = yes }
				change_global_variable = { name = gw_g_n_green_building_codes add = 1 }
			}
		}
		if = {
			limit = { global_var:gw_g_sol_n > 0 }
			set_global_variable = {
				name = gw_g_mean_sol
				value = {
					value = global_var:gw_g_sol_sum
					divide = global_var:gw_g_sol_n
				}
			}
		}
	}
}

# Top Emitters (display only)
# ---------------------------------------------------------------------------
# Top Emitters ranks countries by their last yearly country snapshot. Players
# store five capitals for the GUI and their own place, rebuilt monthly. The
# capital is only a GUI bridge; the numbers belong to its country.
gw_top_emitters_refresh = {
	hidden_effect = {
		save_scope_as = gw_te_viewer
		save_scope_as = gw_te_own_country
		set_variable = { name = gw_emitters_total value = 0 }
		every_country = {
			limit = { gw_is_ranked_emitter = yes }
			scope:gw_te_viewer = {
				change_variable = { name = gw_emitters_total add = 1 }
			}
		}
		set_variable = { name = gw_emitters_own_slot value = 0 }
		gw_top_emitter_slot = { N = 1 POS = 0 }
		gw_top_emitter_slot = { N = 2 POS = 1 }
		gw_top_emitter_slot = { N = 3 POS = 2 }
		gw_top_emitter_slot = { N = 4 POS = 3 }
		gw_top_emitter_slot = { N = 5 POS = 4 }
		# How many rows to show: the filled slots, counted from the first. A
		# slot is empty only past the end of the list (or, in principle, for a
		# country with no capital), and the rows after an empty one hide too.
		set_variable = { name = gw_emitters_shown value = 0 }
		if = {
			limit = { has_variable = gw_emitter_1 }
			set_variable = { name = gw_emitters_shown value = 1 }
			if = {
				limit = { has_variable = gw_emitter_2 }
				set_variable = { name = gw_emitters_shown value = 2 }
				if = {
					limit = { has_variable = gw_emitter_3 }
					set_variable = { name = gw_emitters_shown value = 3 }
					if = {
						limit = { has_variable = gw_emitter_4 }
						set_variable = { name = gw_emitters_shown value = 4 }
						if = {
							limit = { has_variable = gw_emitter_5 }
							set_variable = { name = gw_emitters_shown value = 5 }
						}
					}
				}
			}
		}
		# Our country's place: one more than the countries emitting more than it.
		set_variable = { name = gw_emitters_own_rank value = 0 }
		if = {
			limit = {
				exists = scope:gw_te_own_country
				scope:gw_te_own_country = { gw_is_ranked_emitter = yes }
			}
			set_variable = {
				name = gw_emitters_own_emis
				value = scope:gw_te_own_country.var:gw_disp_country_emis
			}
			set_variable = { name = gw_emitters_own_rank value = 1 }
			every_country = {
				limit = {
					gw_is_ranked_emitter = yes
					var:gw_disp_country_emis > scope:gw_te_viewer.var:gw_emitters_own_emis
				}
				scope:gw_te_viewer = {
					change_variable = { name = gw_emitters_own_rank add = 1 }
				}
			}
			remove_variable = gw_emitters_own_emis
		}
	}
}

# Slot $N$ (1-5) gets the country at position $POS$ ($N$ - 1) by annual
# emissions. An ordered_ position past the end of the list clamps to the last
# element instead of skipping (un_overview_display_effects.txt), so the count
# is checked first. The capital is saved under a name per slot, so a country
# without one can never pass on the previous slot's. Scope: the viewer
# (scope:gw_te_viewer, and scope:gw_te_own_country when it exists, are set).
gw_top_emitter_slot = {
	if = {
		limit = {
			any_country = {
				gw_is_ranked_emitter = yes
				count >= $N$
			}
		}
		ordered_country = {
			limit = { gw_is_ranked_emitter = yes }
			order_by = gw_disp_own_emis
			position = $POS$
			capital ?= { save_scope_as = gw_te_capital_$N$ }
			if = {
				limit = {
					exists = scope:gw_te_own_country
					THIS = scope:gw_te_own_country
				}
				scope:gw_te_viewer = {
					set_variable = { name = gw_emitters_own_slot value = $N$ }
				}
			}
		}
		if = {
			limit = { exists = scope:gw_te_capital_$N$ }
			set_variable = { name = gw_emitter_$N$ value = scope:gw_te_capital_$N$ }
		}
		else_if = {
			limit = { has_variable = gw_emitter_$N$ }
			remove_variable = gw_emitter_$N$
		}
	}
	else_if = {
		limit = { has_variable = gw_emitter_$N$ }
		remove_variable = gw_emitter_$N$
	}
}

# Country scope. A former leader must not retain a tariff policy after its
# customs authority moves. National mitigation choices remain its own.
gw_drop_fossil_fuel_tariffs_effect = {
	je:je_global_warming ?= { remove_modifier = fossil_fuel_tariffs_modifier }
}

# Ownership cleanup only; rate modifiers themselves need no reassertion.
# Run with climate disabled too so the programme's ongoing cost is released.
gw_refresh_fossil_fuel_tariffs_effect = {
	if = {
		limit = {
			je:je_global_warming ?= { has_modifier = fossil_fuel_tariffs_modifier }
			OR = {
				NOT = { te_tax_owns_market = yes }
				NOT = { has_game_rule = global_warming_enabled }
			}
		}
		gw_drop_fossil_fuel_tariffs_effect = yes
	}
}

gw_refresh_tariff_policies_effect = {
	every_country = {
		limit = { has_journal_entry = je_global_warming }
		gw_refresh_fossil_fuel_tariffs_effect = yes
	}
}
