# Global Warming — shared policy effects.
#
# Each `gw_effect_<button>` holds the body that used to live inline in the
# matching scripted_button's `effect` block in
# common/scripted_buttons/global_warming_buttons.txt. The journal-entry button
# and the widget's scripted GUI both call these.
#
# A market-wide repeal skips members bound by an emissions reduction treaty
# (gw_bound_by_emissions_treaty): the treaty locks their policies in, and a
# removal would freeze it.
#
# The `custom_tooltip` wrapper around each market-wide policy is LOAD-BEARING:
# without it the `every_country` iteration enumerates every market member in the
# button/control tooltip (see docs/guides/scripting_best_practices.md,
# "`every_*` Iterators Enumerate in Tooltip Previews").

gw_effect_carbon_tax = {
	custom_tooltip = {
		text = "GW_CARBON_TAX_DESC"
		every_country = {
			limit = {
				market_capital = ROOT.market_capital
				NOT = { je:je_global_warming = { has_modifier = carbon_tax_modifier } }
			}
			je:je_global_warming = { add_modifier = { name = carbon_tax_modifier } }
		}
	}
}

# ============================================================================
# DISPLAY SNAPSHOTS
# ============================================================================
# The journal entry's figures used to be computed live in loc, every frame the
# panel was open. These three effects move that work onto pulses that were
# already doing it, so the widget only ever reads stored numbers.
#
# CADENCE IS DELIBERATE. Emissions are re-based once a year because that is when
# the simulation actually moves them: `global_warming_update_on_action` runs on
# on_yearly_pulse_state and is the only thing that changes
# global_var:greenhouse_gas_emissions. Snapshotting monthly would multiply the
# market sweep's cost by twelve to show a number that had not changed. The
# widget and the docs say "re-based each January" rather than implying live data.
# ============================================================================

# Store this market's annual emissions on its LEADER, whose yearly pulse is
# already evaluating the figure to accumulate the global total.
#
# The order matters: `market_greenhouse_gas_emissions_script_value` is evaluated
# once, into the variable, and the global accumulation then reads the variable.
# Variable writes ARE visible later in the same effect block (unlike
# add_modifier), so this stores and accumulates exactly the same number the
# on_action added before, with no second evaluation of the market-wide sweep.
#
# Scope: country (the market leader)
gw_snapshot_market_emissions_effect = {
	hidden_effect = {
		set_variable = {
			name = gw_disp_market_emis
			value = market_greenhouse_gas_emissions_script_value
		}
		set_variable = {
			name = gw_disp_capture
			value = market_carbon_capture_script_value
		}
	}
}

# The market's CUMULATIVE emissions, for the panel's Top Emitters list: every
# year's figure this country's market added to the world total while this
# country led it. Called right after gw_snapshot_market_emissions_effect and the
# global accumulation (global_warming_update_on_action), so it adds exactly the
# number the world total received that year. Display only; nothing reads it but
# the panel.
#
# THE RULE WHEN LEADERSHIP CHANGES. Like the annual figure, the total is stored
# on the leader, so it follows the country, not the market:
# - a country that stops leading (it joins another market, or a customs union
#   it led dissolves) keeps its total but adds nothing until it leads a market
#   again, when it carries on from where it stopped;
# - a market's new leader starts from its own total (zero if it never led one);
# - when markets merge, the leader of the merged market goes on adding to its
#   own total, and the absorbed leader's total stays with that country.
# The panel labels the figure accordingly ("while this country led its market").
# A variable, so a revolution's winner keeps it (CLAUDE.md's revolution rule).
# The annual figure can be negative (a market that captures more than it
# emits), so the total can fall.
#
# Scope: country (the market leader)
gw_accumulate_market_emissions_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_variable = gw_disp_market_emis_cum } }
			set_variable = {
				name = gw_disp_market_emis_cum
				value = 0
			}
		}
		change_variable = {
			name = gw_disp_market_emis_cum
			add = var:gw_disp_market_emis
		}
	}
}

# Re-base the annual world emissions figure each January, as the change in the
# cumulative global variable since last January. O(1), and more truthful than
# re-summing every market: it reports what actually accumulated rather than
# re-estimating the current rate.
#
# `month = 0` is January (te_history_values.txt documents Jan = 0 ... Dec = 11).
# Scope: any (called from the global monthly pulse)
gw_rebase_annual_emissions_effect = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_variable = gw_g_emis_prev } }
			set_global_variable = {
				name = gw_g_emis_prev
				value = 0
			}
		}
		if = {
			limit = {
				month = 0
				has_global_variable = greenhouse_gas_emissions
			}
			set_global_variable = {
				name = gw_g_global_emis
				value = global_var:greenhouse_gas_emissions
			}
			change_global_variable = {
				name = gw_g_global_emis
				subtract = global_var:gw_g_emis_prev
			}
			set_global_variable = {
				name = gw_g_emis_prev
				value = global_var:greenhouse_gas_emissions
			}
		}
	}
}

# Count how many countries run each of the nine policies, in ONE country sweep
# that fills all nine counters. Replaces eight separate sweeping script values
# that were rendered live in the reason text.
#
# The same sweep stores the mean standard of living over these countries
# (gw_g_mean_sol), which the AI's climate wills measure wealth against
# (gw_ai_sig_wealth in global_warming_ai_values.txt). It is the only world
# figure those scores need, so it rides this pass instead of adding one.
# Scope: any (called from the global monthly pulse)
gw_refresh_global_counts_effect = {
	hidden_effect = {
		set_global_variable = { name = gw_g_n_carbon_tax value = 0 }
		set_global_variable = { name = gw_g_n_renewable_investment value = 0 }
		set_global_variable = { name = gw_g_n_emission_standards value = 0 }
		set_global_variable = { name = gw_g_n_climate_adaptation value = 0 }
		set_global_variable = { name = gw_g_n_reforestation value = 0 }
		set_global_variable = { name = gw_g_n_carbon_removal value = 0 }
		set_global_variable = { name = gw_g_n_public_transit value = 0 }
		set_global_variable = { name = gw_g_n_fossil_fuel_divestment value = 0 }
		set_global_variable = { name = gw_g_n_green_building_codes value = 0 }
		set_global_variable = { name = gw_g_sol_sum value = 0 }
		set_global_variable = { name = gw_g_sol_n value = 0 }
		every_country = {
			limit = { has_journal_entry = je_global_warming }
			change_global_variable = { name = gw_g_sol_sum add = average_sol }
			change_global_variable = { name = gw_g_sol_n add = 1 }
			if = {
				limit = { gw_policy_carbon_tax_active = yes }
				change_global_variable = { name = gw_g_n_carbon_tax add = 1 }
			}
			if = {
				limit = { gw_policy_renewable_investment_active = yes }
				change_global_variable = { name = gw_g_n_renewable_investment add = 1 }
			}
			if = {
				limit = { gw_policy_emission_standards_active = yes }
				change_global_variable = { name = gw_g_n_emission_standards add = 1 }
			}
			if = {
				limit = { gw_policy_climate_adaptation_active = yes }
				change_global_variable = { name = gw_g_n_climate_adaptation add = 1 }
			}
			if = {
				limit = { gw_policy_reforestation_active = yes }
				change_global_variable = { name = gw_g_n_reforestation add = 1 }
			}
			if = {
				limit = { gw_policy_carbon_removal_active = yes }
				change_global_variable = { name = gw_g_n_carbon_removal add = 1 }
			}
			if = {
				limit = { gw_policy_public_transit_active = yes }
				change_global_variable = { name = gw_g_n_public_transit add = 1 }
			}
			if = {
				limit = { gw_policy_fossil_fuel_divestment_active = yes }
				change_global_variable = { name = gw_g_n_fossil_fuel_divestment add = 1 }
			}
			if = {
				limit = { gw_policy_green_building_codes_active = yes }
				change_global_variable = { name = gw_g_n_green_building_codes add = 1 }
			}
		}
		if = {
			limit = { global_var:gw_g_sol_n > 0 }
			set_global_variable = {
				name = gw_g_mean_sol
				value = {
					value = global_var:gw_g_sol_sum
					divide = global_var:gw_g_sol_n
				}
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Top Emitters (display only)
# ---------------------------------------------------------------------------
# The panel's Top Emitters section (global_warming_widget.gui,
# te_gw_sec_emitters) lists the five markets emitting most, each named by its
# leader. There is no datamodel of markets a journal-entry widget can walk (the
# only market lists in any .gui are per-market: Market.AccessParticipants,
# Market.GetExports, ...), so, as the Space Race's Rivals rows do
# (sr_rivals_estimate_refresh), the viewer keeps a snapshot:
#   gw_emitters_total        how many markets emit (gw_is_ranked_emitter)
#   gw_emitter_1 .. _5       the first five leaders' capitals, largest first
#                            (gotcha #11: the GUI reaches a country through a
#                            stored state)
#   gw_emitters_shown        how many of those rows are filled (0-5)
#   gw_emitters_own_slot     1-5 when our own market is one of those rows
#   gw_emitters_own_rank     our market's place among all of them (0: ours
#                            does not emit on balance)
# The figures on each row are read live from the listed leader's own variables
# (gw_disp_own_* in global_warming_values.txt). Monthly and for players only
# (je_global_warming's on_monthly_pulse): the leaders' figures are written on
# their market capitals' yearly state pulses, not on the viewer's, so a
# snapshot taken once a year could predate some of them and list the rows out
# of the order their figures show; monthly, the order is at most a month
# behind. Everything is a variable, so a revolution's winner keeps it, and the
# snapshot is rebuilt every month anyway.
# Scope: country (a player).
gw_top_emitters_refresh = {
	hidden_effect = {
		save_scope_as = gw_te_viewer
		market_capital ?= {
			owner ?= { save_scope_as = gw_te_own_leader }
		}
		set_variable = { name = gw_emitters_total value = 0 }
		every_country = {
			limit = { gw_is_ranked_emitter = yes }
			scope:gw_te_viewer = {
				change_variable = { name = gw_emitters_total add = 1 }
			}
		}
		set_variable = { name = gw_emitters_own_slot value = 0 }
		gw_top_emitter_slot = { N = 1 POS = 0 }
		gw_top_emitter_slot = { N = 2 POS = 1 }
		gw_top_emitter_slot = { N = 3 POS = 2 }
		gw_top_emitter_slot = { N = 4 POS = 3 }
		gw_top_emitter_slot = { N = 5 POS = 4 }
		# How many rows to show: the filled slots, counted from the first. A
		# slot is empty only past the end of the list (or, in principle, for a
		# leader with no capital), and the rows after an empty one hide too.
		set_variable = { name = gw_emitters_shown value = 0 }
		if = {
			limit = { has_variable = gw_emitter_1 }
			set_variable = { name = gw_emitters_shown value = 1 }
			if = {
				limit = { has_variable = gw_emitter_2 }
				set_variable = { name = gw_emitters_shown value = 2 }
				if = {
					limit = { has_variable = gw_emitter_3 }
					set_variable = { name = gw_emitters_shown value = 3 }
					if = {
						limit = { has_variable = gw_emitter_4 }
						set_variable = { name = gw_emitters_shown value = 4 }
						if = {
							limit = { has_variable = gw_emitter_5 }
							set_variable = { name = gw_emitters_shown value = 5 }
						}
					}
				}
			}
		}
		# Our market's place: one more than the markets emitting more than it.
		set_variable = { name = gw_emitters_own_rank value = 0 }
		if = {
			limit = {
				exists = scope:gw_te_own_leader
				scope:gw_te_own_leader = { gw_is_ranked_emitter = yes }
			}
			set_variable = {
				name = gw_emitters_own_emis
				value = scope:gw_te_own_leader.var:gw_disp_market_emis
			}
			set_variable = { name = gw_emitters_own_rank value = 1 }
			every_country = {
				limit = {
					gw_is_ranked_emitter = yes
					var:gw_disp_market_emis > scope:gw_te_viewer.var:gw_emitters_own_emis
				}
				scope:gw_te_viewer = {
					change_variable = { name = gw_emitters_own_rank add = 1 }
				}
			}
			remove_variable = gw_emitters_own_emis
		}
	}
}

# Slot $N$ (1-5) gets the leader at position $POS$ ($N$ - 1) by annual
# emissions. An ordered_ position past the end of the list clamps to the last
# element instead of skipping (un_overview_display_effects.txt), so the count
# is checked first. The capital is saved under a name per slot, so a leader
# without one can never pass on the previous slot's. Scope: the viewer
# (scope:gw_te_viewer, and scope:gw_te_own_leader when it exists, are set).
gw_top_emitter_slot = {
	if = {
		limit = {
			any_country = {
				gw_is_ranked_emitter = yes
				count >= $N$
			}
		}
		ordered_country = {
			limit = { gw_is_ranked_emitter = yes }
			order_by = gw_disp_own_emis
			position = $POS$
			capital ?= { save_scope_as = gw_te_capital_$N$ }
			if = {
				limit = {
					exists = scope:gw_te_own_leader
					THIS = scope:gw_te_own_leader
				}
				scope:gw_te_viewer = {
					set_variable = { name = gw_emitters_own_slot value = $N$ }
				}
			}
		}
		if = {
			limit = { exists = scope:gw_te_capital_$N$ }
			set_variable = { name = gw_emitter_$N$ value = scope:gw_te_capital_$N$ }
		}
		else_if = {
			limit = { has_variable = gw_emitter_$N$ }
			remove_variable = gw_emitter_$N$
		}
	}
	else_if = {
		limit = { has_variable = gw_emitter_$N$ }
		remove_variable = gw_emitter_$N$
	}
}
