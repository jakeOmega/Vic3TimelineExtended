# ============================================================================
# GRAND MONUMENTS — several monuments in one state: scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-10-13-multiple-monuments-per-state-design.md.
#
# A SECONDARY monument is a script container tagged gm_monument, with no parent
# (it outlives every country), in the global list gm_monuments. It is a
# returned trophy that stands apart from the monument already in its state (or
# a co-located landmark, gm_seed_landmark), and keeps its trophy's records
# (gm_tr_*, gm_trophy_effects.txt: skin, form, landmark, name, dedication,
# picture, honoree) so the trophy names, forms, parts and pictures serve it.
#   gm_tr_origin, gm_tr_origin_region   the state it stands in
#   gm_tr_levels                        its grandeur (gm_mon_grandeur)
# The list and the state's holder are read live. The monthly pass
# (gm_monument_monthly) writes each state's gm_monuments_here and
# gm_mon_here_count from the global list, so a state that was merged, or a
# civil war's winner (who inherits no lists), has its rows back.
# No timed variable on a container: it would never expire
# (container_timed_variable_audit).
#
# The building's level is the truth (gm_values.txt, gm_state_grandeur): nothing
# here rebuilds a building on a pulse. A rebuild runs where a monument is added
# or taken, from a hidden event (gm_state_tr_receive and its siblings).

# ---- The monthly pass (from on_monthly_pulse) --------------------------------------
# No scope. Repair each secondary's state, then the lists. Nothing to do in a
# game none has touched: gm_mon_lists_written stands while any list does.
gm_monument_monthly = {
	if = {
		limit = {
			gm_system_enabled = yes
			OR = {
				has_global_variable = gm_mon_lists_written
				has_global_variable_list = gm_monuments
			}
		}
		if = {
			limit = { has_global_variable_list = gm_monuments }
			every_in_global_list = {
				variable = gm_monuments
				gm_tr_repair_places = yes
				# A landmark whose picture is drawn later, or a monument from an
				# older save, catches up within a month (as a trophy does).
				gm_tr_update_picture = yes
			}
		}
		gm_monument_pass = yes
	}
}

# Any scope. Every state's list of secondaries, from the global list. Last
# month's are cleared first; the lists are rebuilt each time.
gm_monument_pass = {
	every_state = {
		limit = { has_variable = gm_mon_here_count }
		clear_variable_list = gm_monuments_here
		remove_variable = gm_mon_here_count
	}
	# The owners' lists, for Our Monuments (gm_monuments_held, a row each).
	every_country = {
		limit = { has_variable = gm_mon_listed }
		clear_variable_list = gm_monuments_held
		remove_variable = gm_mon_listed
	}
	if = {
		limit = { has_global_variable = gm_mon_lists_written }
		remove_global_variable = gm_mon_lists_written
	}
	if = {
		limit = { has_global_variable_list = gm_monuments }
		every_in_global_list = {
			variable = gm_monuments
			gm_mon_list_one = yes
		}
	}
	# A state whose building has gone (whatever removed it) while monuments
	# stand beside it: the largest takes it over.
	every_state = {
		limit = {
			gm_state_has_secondary = yes
			NOT = { has_building = building_grand_monument }
		}
		gm_state_promote_secondary = yes
	}
}

# Container scope: this secondary monument into its state's list.
gm_mon_list_one = {
	save_scope_as = gm_mon_listing
	if = {
		limit = {
			has_variable = gm_tr_origin
			exists = var:gm_tr_origin
		}
		set_global_variable = gm_mon_lists_written
		var:gm_tr_origin = {
			add_to_variable_list = { name = gm_monuments_here target = scope:gm_mon_listing }
			gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
			owner = {
				add_to_variable_list = { name = gm_monuments_held target = scope:gm_mon_listing }
				set_variable = gm_mon_listed
			}
		}
	}
}

# ---- The primary goes (spec §4.2, §4.3) ----------------------------------------------
# State scope, from gm_state_tear_down, gm_state_tr_remove_whole and the pass:
# the primary monument is gone (torn down, carried off whole, or its building
# lost) and another stood beside it. The largest secondary becomes the
# primary: the building is built at the levels of everything that remains, with
# that monument's records and dedication (gm_state_tr_rebuild_from_moving,
# gm_trophy_effects.txt), and its container is destroyed.
# The work runs only under a marker set here: a button's or an option's tooltip
# walks the effects it shows without writing, so the rebuild's read of
# gm_rebuild_level would find it unset through the ladder's 200 cases (see
# gm_state_rededicate).
gm_state_promote_secondary = {
	set_variable = gm_promote_click
	if = {
		limit = { has_variable = gm_promote_click }
		remove_variable = gm_promote_click
		clear_saved_scope = gm_mon_promoted
		clear_saved_scope = gm_tr_moving
		ordered_in_list = {
			variable = gm_monuments_here
			order_by = gm_mon_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_mon_promoted
		}
		if = {
			limit = { exists = scope:gm_mon_promoted }
			scope:gm_mon_promoted = { save_scope_as = gm_tr_moving }
			# Everything that remains, the promoted one included: read before it
			# leaves the list.
			set_variable = {
				name = gm_rebuild_level
				value = {
					value = gm_state_secondary_levels
					max = 200
				}
			}
			remove_list_variable = { name = gm_monuments_here target = scope:gm_mon_promoted }
			gm_tr_count_add = { VAR = gm_mon_here_count ADD = -1 }
			owner = { remove_list_variable = { name = gm_monuments_held target = scope:gm_mon_promoted } }
			remove_list_global_variable = { name = gm_monuments target = scope:gm_mon_promoted }
			gm_state_tr_rebuild_from_moving = yes
			scope:gm_mon_promoted = { destroy_container = yes }
			owner = { trigger_event = { id = monument_events.20 } }
		}
	}
}

# ---- Home again, as a monument of its own (spec §4.1) --------------------------------
# Container scope, the returning trophy (scope:gm_tr_moving; gm_tr_return_home).
# A piece of a monument already standing beside the primary rejoins that one:
# its levels go on the building and on that container.
gm_tr_rejoin_secondary = {
	var:gm_tr_origin = {
		clear_saved_scope = gm_mon_match
		random_in_list = {
			variable = gm_monuments_here
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = scope:gm_tr_moving.var:gm_tr_landmark
			}
			save_scope_as = gm_mon_match
		}
		if = {
			limit = { exists = scope:gm_mon_match }
			gm_state_tr_add_levels = yes
			scope:gm_mon_match = {
				change_variable = { name = gm_tr_levels add = scope:gm_tr_moving.var:gm_tr_levels }
			}
		}
		owner = { trigger_event = { id = monument_events.20 } }
	}
}

# Container scope, the returning trophy (scope:gm_tr_moving). It stands in its
# origin as a monument beside the one there, and keeps everything it was: its
# name, form, landmark, picture and dedication. Its levels go on the building,
# which records the primary's dedication to put back first
# (gm_state_tr_add_levels), then it joins the state's list at once, so no
# reader sees the building taller than its primary and secondaries.
# It is the same container, retagged (add_tag / remove_tag are the status
# marks, scripting_best_practices.md) and moved from gm_trophies to
# gm_monuments: the trophy names, parts and pictures read its gm_tr_* records.
gm_tr_become_monument = {
	var:gm_tr_origin = {
		gm_state_tr_add_levels = yes
		add_to_variable_list = { name = gm_monuments_here target = scope:gm_tr_moving }
		gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
		owner = {
			add_to_variable_list = { name = gm_monuments_held target = scope:gm_tr_moving }
			set_variable = gm_mon_listed
			trigger_event = { id = monument_events.20 }
		}
	}
	remove_list_global_variable = { name = gm_trophies target = scope:gm_tr_moving }
	add_to_global_variable_list = { name = gm_monuments target = scope:gm_tr_moving }
	set_global_variable = gm_mon_lists_written
	remove_tag = gm_trophy
	add_tag = gm_monument
	# Where it stands is where it came from: the pass's repair finds the
	# state again through the region.
	set_variable = { name = gm_tr_site value = var:gm_tr_origin }
	set_variable = { name = gm_tr_site_region value = var:gm_tr_origin_region }
	gm_tr_forget_notes = yes
}

# ---- Carried off (spec §4.2) -------------------------------------------------------
# Container scope, a secondary monument; $WHO$ carries it off from the state it
# stands in (a country that occupies it). Only orders the work, as
# gm_state_tr_order_take does: the state's gm_tr_take_target names it for
# monument_trophy_events.10, which checks again and does it.
gm_mon_tr_order_take = {
	save_scope_as = gm_mon_pick
	var:gm_tr_origin ?= {
		set_variable = { name = gm_tr_take_target value = scope:gm_mon_pick }
		gm_state_tr_order_take = { WHO = $WHO$ ROUTE = occupation }
	}
}

# State scope (scope:gm_tr_state, gm_tr_taker, gm_tr_victim as gm_state_tr_take
# saves them), from it when var:gm_tr_take_target names a secondary monument.
# Whole (an object, or a monument too small to part with a piece): the
# container itself becomes the trophy. Otherwise a piece, by its form's share,
# is a new trophy with its records. Either way the building shrinks by the
# levels taken and everything else standing here stays.
gm_mon_tr_take = {
	clear_saved_scope = gm_mon_target
	clear_saved_scope = gm_tr_site_state
	var:gm_tr_take_target ?= { save_scope_as = gm_mon_target }
	if = {
		limit = {
			exists = scope:gm_mon_target
			scope:gm_tr_taker = { exists = capital }
		}
		scope:gm_tr_taker = {
			capital = { save_scope_as = gm_tr_site_state }
		}
		scope:gm_mon_target = {
			set_variable = { name = gm_tr_take_levels value = gm_mon_tr_levels }
		}
		set_variable = { name = gm_tr_take_levels value = scope:gm_mon_target.var:gm_tr_take_levels }
		scope:gm_mon_target = {
			remove_variable = gm_tr_take_levels
			if = {
				limit = { gm_mon_tr_whole = 1 }
				gm_mon_tr_take_whole = yes
			}
			else = {
				gm_mon_tr_take_piece = yes
			}
		}
	}
}

# Container scope (the secondary; scope:gm_tr_state, gm_tr_taker and
# gm_tr_site_state saved). It stands abroad now: out of the state's list and
# gm_monuments, into gm_trophies, retagged, its site the taker's capital. Its
# records are kept; a piece it already was stays a piece.
gm_mon_tr_take_whole = {
	save_scope_as = gm_tr_new
	scope:gm_tr_state = {
		remove_list_variable = { name = gm_monuments_here target = scope:gm_tr_new }
		gm_tr_count_add = { VAR = gm_mon_here_count ADD = -1 }
		owner = { remove_list_variable = { name = gm_monuments_held target = scope:gm_tr_new } }
	}
	remove_list_global_variable = { name = gm_monuments target = scope:gm_tr_new }
	add_to_global_variable_list = { name = gm_trophies target = scope:gm_tr_new }
	remove_tag = gm_monument
	add_tag = gm_trophy
	set_variable = { name = gm_tr_site value = scope:gm_tr_site_state }
	set_variable = { name = gm_tr_site_region value = scope:gm_tr_site_state.state_region }
	set_variable = { name = gm_tr_year value = gm_this_year }
	set_variable = { name = gm_tr_taken_by value = scope:gm_tr_taker }
	if = {
		limit = { NOT = { has_variable = gm_tr_share } }
		set_variable = { name = gm_tr_share value = flag:whole }
	}
	set_variable = {
		name = gm_tr_grandeur
		value = {
			value = var:gm_tr_levels
			divide = 2
			ceiling = yes
		}
	}
	gm_tr_forget_notes = yes
	scope:gm_tr_state = { gm_state_tr_shrink_by = yes }
}

# Container scope (the secondary, scope:gm_mon_src once saved here). A piece of
# it: a new trophy with its records, the levels taken (the state's
# gm_tr_take_levels) off its own, the building shrunk by them, and the state
# spared for ten years as a piece of the primary spares it.
gm_mon_tr_take_piece = {
	save_scope_as = gm_mon_src
	scope:gm_tr_state = {
		create_container = {
			tags = { gm_trophy }
			save_scope_as = gm_tr_new
			on_created = {
				save_scope_as = gm_tr_new
			}
		}
	}
	if = {
		limit = { exists = scope:gm_tr_new }
		add_to_global_variable_list = { name = gm_trophies target = scope:gm_tr_new }
		scope:gm_tr_new = {
			set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
			set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
			set_variable = { name = gm_tr_site value = scope:gm_tr_site_state }
			set_variable = { name = gm_tr_site_region value = scope:gm_tr_site_state.state_region }
			set_variable = { name = gm_tr_levels value = scope:gm_tr_state.var:gm_tr_take_levels }
			set_variable = {
				name = gm_tr_grandeur
				value = {
					value = scope:gm_tr_state.var:gm_tr_take_levels
					divide = 2
					ceiling = yes
				}
			}
			set_variable = { name = gm_tr_year value = gm_this_year }
			set_variable = { name = gm_tr_taken_by value = scope:gm_tr_taker }
			if = {
				limit = { scope:gm_mon_src = { gm_mon_form_is_piece = yes } }
				set_variable = { name = gm_tr_share value = flag:piece }
			}
			else = {
				set_variable = { name = gm_tr_share value = flag:token }
			}
			gm_mon_copy_record = { KEY = gm_tr_skin }
			gm_mon_copy_record = { KEY = gm_tr_form }
			gm_mon_copy_record = { KEY = gm_tr_landmark }
			gm_mon_copy_record = { KEY = gm_tr_namesake }
			gm_mon_copy_record = { KEY = gm_tr_name_year }
			gm_mon_copy_record = { KEY = gm_tr_name_char }
			gm_mon_copy_record = { KEY = gm_tr_name_occasion }
			gm_mon_copy_record = { KEY = gm_tr_name_enemy_cap }
			gm_mon_copy_record = { KEY = gm_tr_name }
			gm_mon_copy_record = { KEY = gm_tr_honoree }
			gm_mon_copy_record = { KEY = gm_tr_honoree_ig }
			gm_mon_copy_record = { KEY = gm_tr_dedication }
			gm_tr_update_picture = yes
		}
		change_variable = { name = gm_tr_levels subtract = scope:gm_tr_state.var:gm_tr_take_levels }
		scope:gm_tr_state = {
			set_variable = { name = gm_tr_cooldown days = 3650 }
			gm_state_tr_shrink_by = yes
		}
	}
}

# Container scope (the new trophy): the secondary's variable $KEY$, when it has
# one, under the same name.
gm_mon_copy_record = {
	if = {
		limit = { scope:gm_mon_src = { has_variable = $KEY$ } }
		set_variable = { name = $KEY$ value = scope:gm_mon_src.var:$KEY$ }
	}
}

# State scope (var:gm_tr_take_levels the levels carried off). The building is
# built again smaller, keeping the primary's dedication (recorded first, as a
# return records it) and every monument standing here: the primary's own and
# the secondaries' levels still add up to what is left. The ladder cannot take
# levels off a standing building (gm_state_tr_remove_piece).
# Under a marker, as Rededicate is: a tooltip walks this body without writing,
# and the ladder's switch reads gm_rebuild_level.
gm_state_tr_shrink_by = {
	set_variable = gm_shrink_click
	if = {
		limit = { has_variable = gm_shrink_click }
		remove_variable = gm_shrink_click
		# Its method, Undedicated too, and not while a ceremony asks
		# (gm_state_tr_add_levels).
		if = {
			limit = {
				NOT = { has_variable = gm_restore_pm }
				NOT = { has_variable = gm_ceremony_pending }
			}
			gm_state_mark_restore_pm = yes
		}
		set_variable = { name = gm_rebuilding days = 1 }
		set_variable = {
			name = gm_rebuild_level
			value = {
				value = gm_state_level
				subtract = var:gm_tr_take_levels
				min = 1
				max = 200
			}
		}
		remove_building = building_grand_monument
		te_construction_market_build_specified_level = {
			SPEC_TYPE = building_grand_monument
			SPEC_LEVEL = var:gm_rebuild_level
		}
		remove_variable = gm_rebuild_level
		owner = { trigger_event = { id = monument_events.20 } }
	}
}

# ---- The AI's choice of which monument ----------------------------------------------
# State scope, before gm_state_tr_take on the AI's occupation roll. One roll
# falls on one monument of the state: the primary, or (every monument as
# likely) a secondary, named in gm_tr_take_target. A primary that cannot go
# (undedicated, a Grand Shrine: gm_state_tr_takeable) leaves the roll to the
# secondaries, which the caller's gm_state_tr_any_takeable found open.
gm_state_tr_ai_choose_target = {
	gm_remove_state_var = { VAR = gm_tr_take_target }
	if = {
		limit = {
			gm_state_has_secondary = yes
			gm_state_tr_takeable_secondary = yes
		}
		clear_saved_scope = gm_mon_pick
		if = {
			limit = { gm_state_tr_takeable = no }
			random_in_list = { variable = gm_monuments_here save_scope_as = gm_mon_pick }
		}
		else = {
			random = {
				chance = gm_state_tr_ai_secondary_chance
				random_in_list = { variable = gm_monuments_here save_scope_as = gm_mon_pick }
			}
		}
		if = {
			limit = { exists = scope:gm_mon_pick }
			set_variable = { name = gm_tr_take_target value = scope:gm_mon_pick }
		}
	}
}

# ---- A landmark seeded beside another (spec §6) ----------------------------------------
# State scope, the part of a state that already has a monument, from
# gm_seed_landmark (gm_landmark_effects.txt, once per game, at the start). The
# landmark is a secondary monument: a container with the records a returned
# trophy carries (gm_tr_*), its levels added to the state's one building, which
# keeps the primary's dedication (gm_state_tr_add_levels records it first).
gm_seed_landmark_beside = {
	clear_saved_scope = gm_tr_moving
	save_scope_as = gm_mon_seed_state
	create_container = {
		tags = { gm_monument }
		save_scope_as = gm_tr_moving
		on_created = {
			save_scope_as = gm_tr_moving
		}
	}
	if = {
		limit = { exists = scope:gm_tr_moving }
		scope:gm_tr_moving = {
			set_variable = { name = gm_tr_levels value = $LEVEL$ }
			set_variable = { name = gm_tr_dedication value = flag:$DEDICATION$ }
			set_variable = { name = gm_tr_skin value = flag:landmark_$KEY$ }
			set_variable = { name = gm_tr_form value = flag:$FORM$ }
			set_variable = { name = gm_tr_landmark value = flag:$KEY$ }
			set_variable = { name = gm_tr_namesake value = flag:landmark }
			# A whole landmark: its row's title is its name, with no part's.
			set_variable = { name = gm_tr_share value = flag:whole }
			set_variable = { name = gm_tr_origin value = scope:gm_mon_seed_state }
			set_variable = { name = gm_tr_origin_region value = scope:gm_mon_seed_state.state_region }
			set_variable = { name = gm_tr_site value = scope:gm_mon_seed_state }
			set_variable = { name = gm_tr_site_region value = scope:gm_mon_seed_state.state_region }
			gm_tr_update_picture = yes
		}
		# The levels on the building, then the container in the lists.
		gm_state_tr_add_levels = yes
		add_to_global_variable_list = { name = gm_monuments target = scope:gm_tr_moving }
		set_global_variable = gm_mon_lists_written
		add_to_variable_list = { name = gm_monuments_here target = scope:gm_tr_moving }
		gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
		owner = {
			add_to_variable_list = { name = gm_monuments_held target = scope:gm_tr_moving }
			set_variable = gm_mon_listed
			trigger_event = { id = monument_events.20 }
		}
		debug_log = "TE_MONUMENTS: landmark $KEY$ placed beside the monument already standing in its state: level $LEVEL$, form $FORM$"
	}
}

# ---- Self-check (once a game, after the seeding) ---------------------------------------
# No scope, from gm_landmarks_on_game_started_after_lobby. Whether a returning
# trophy is told from a piece of what stands: the rule compares the standing
# monument's landmark with the trophy's across scopes
# (gm_state_stands_for_moving, gm_monument_triggers.txt), which nobody has seen
# the engine do. 1836's data says what each must read as: the Parthenon's
# marbles are a piece of the Parthenon standing in Attica, the Luxor Obelisk
# stands apart from the Obelisks of Karnak. With either premise absent (the
# owner proved wrong, the rule off) it logs nothing, so OK cannot be a false
# pass. The fallback is in docs/testing/grand-monuments-multiple-playtest.md.
gm_monument_self_check = {
	clear_saved_scope = gm_mon_chk_piece
	clear_saved_scope = gm_mon_chk_apart
	if = {
		limit = { has_global_variable_list = gm_trophies }
		random_in_global_list = {
			variable = gm_trophies
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:parthenon
				var:gm_tr_origin ?= {
					has_building = building_grand_monument
					has_variable = gm_landmark
					var:gm_landmark = flag:parthenon
				}
			}
			save_scope_as = gm_mon_chk_piece
		}
		random_in_global_list = {
			variable = gm_trophies
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:luxor_obelisk
				var:gm_tr_origin ?= {
					has_building = building_grand_monument
					has_variable = gm_landmark
					var:gm_landmark = flag:obelisks_of_karnak
				}
			}
			save_scope_as = gm_mon_chk_apart
		}
	}
	if = {
		limit = { exists = scope:gm_mon_chk_piece }
		scope:gm_mon_chk_piece = {
			save_scope_as = gm_tr_moving
			if = {
				limit = { gm_tr_rejoins_primary = yes }
				debug_log = "TE_SELF_CHECK OK monument_piece_rejoins (the Parthenon's marbles read as a piece of the Parthenon in Attica)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN monument_piece_rejoins (the Parthenon's marbles do not read as a piece of the Parthenon standing in Attica, so a returned piece will stand as a monument of its own: take the fallback in docs/testing/grand-monuments-multiple-playtest.md)"
			}
		}
	}
	if = {
		limit = { exists = scope:gm_mon_chk_apart }
		scope:gm_mon_chk_apart = {
			save_scope_as = gm_tr_moving
			if = {
				limit = { gm_tr_stands_apart = yes }
				debug_log = "TE_SELF_CHECK OK monument_artifact_stands_apart (the Luxor Obelisk reads as apart from the Obelisks of Karnak)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN monument_artifact_stands_apart (the Luxor Obelisk does not read as apart from the Obelisks of Karnak, so a returned artifact will be merged into the monument standing: take the fallback in docs/testing/grand-monuments-multiple-playtest.md)"
			}
		}
	}
