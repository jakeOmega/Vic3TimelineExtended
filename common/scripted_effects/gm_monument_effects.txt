# ============================================================================
# GRAND MONUMENTS — several monuments in one state: scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-10-13-multiple-monuments-per-state-design.md.
#
# A SECONDARY monument is a script container tagged gm_monument, with no parent
# (it outlives every country), in the global list gm_monuments. It is a
# returned trophy that stands apart from the monument already in its state (or
# a co-located landmark, gm_seed_landmark), and keeps its trophy's records
# (gm_tr_*, gm_trophy_effects.txt: skin, form, landmark, name, dedication,
# picture, honoree) so the trophy names, forms, parts and pictures serve it.
#   gm_tr_origin, gm_tr_origin_region   the state it stands in
#   gm_tr_levels                        its grandeur (gm_mon_grandeur)
# The list and the state's holder are read live. The monthly pass
# (gm_monument_monthly) writes each state's gm_monuments_here and
# gm_mon_here_count from the global list, so a state that was merged, or a
# civil war's winner (who inherits no lists), has its rows back.
# No timed variable on a container: it would never expire
# (container_timed_variable_audit).
#
# The building's level is the truth (gm_values.txt, gm_state_grandeur): nothing
# here rebuilds a building on a pulse. A rebuild runs where a monument is added
# or taken, from a hidden event (gm_state_tr_receive and its siblings).

# ---- The monthly pass (from on_monthly_pulse) --------------------------------------
# No scope. Repair each secondary's state, then the lists. Nothing to do in a
# game none has touched: gm_mon_lists_written stands while any list does.
gm_monument_monthly = {
	# The world-wealth rate (sites spec 2026-10-18 §3.2, Plan B): each January,
	# and now if there is none; and once, a new game's world GDP beside the
	# constant it is measured against.
	if = {
		limit = { gm_system_enabled = yes }
		gm_world_rate_ensure = yes
		if = {
			limit = { month = 0 }
			set_global_variable = { name = gm_world_rate value = gm_world_rate_now }
		}
		gm_world_base_self_check = yes
	}
	# An older save's finished monuments, marked once before any is migrated
	# (sites spec 2026-10-18 §3.5): gm_state_needs_migration reads the mark, and
	# a finished building with neither it nor gm_migrated is a stray
	# (gm_state_stray_building).
	if = {
		limit = {
			gm_system_enabled = yes
			NOT = { has_global_variable = gm_sites_marked }
		}
		set_global_variable = gm_sites_marked
		every_state = {
			limit = {
				has_building = building_grand_monument
				NOT = { has_variable = gm_migrated }
			}
			set_variable = gm_premigration
		}
	}
	# Old saves' held specials, once a round (signatures spec §5.6).
	if = {
		limit = { gm_system_enabled = yes }
		gm_sig_seed_monthly = yes
	}
	if = {
		limit = {
			gm_system_enabled = yes
			OR = {
				has_global_variable = gm_mon_lists_written
				has_global_variable_list = gm_monuments
			}
		}
		if = {
			limit = { has_global_variable_list = gm_monuments }
			every_in_global_list = {
				variable = gm_monuments
				gm_tr_repair_places = yes
				# A landmark whose picture is drawn later, or a monument from an
				# older save, catches up within a month (as a trophy does).
				gm_tr_update_picture = yes
				# And its unique dedication (signatures spec §4.4).
				gm_tr_sig_backfill = yes
				# Its raiser recorded once (spec 2026-10-16 §4.4), then a month
				# older (§2).
				gm_mon_first_sight = yes
				if = {
					limit = { gm_mon_ages = yes }
					gm_mon_add_age = { AGE = gm_mon_age_step }
				}
				# An order its event never ran (the building was gone that day):
				# its state's event again, which runs it or drops it.
				if = {
					limit = { has_variable = gm_mon_order }
					var:gm_tr_origin ?= {
						trigger_event = { id = monument_events.30 }
					}
				}
			}
		}
		gm_monument_pass = yes
	}
	# Every recorded primary's shadow, in a game with no secondary too (issue
	# #936, § "A merged state" below).
	if = {
		limit = { gm_system_enabled = yes }
		gm_shadow_monthly = yes
	}
}

# Any scope. Every state's list of secondaries, from the global list. Last
# month's are cleared first; the lists are rebuilt each time.
gm_monument_pass = {
	every_state = {
		limit = { has_variable = gm_mon_here_count }
		clear_variable_list = gm_monuments_here
		remove_variable = gm_mon_here_count
	}
	# The owners' lists, for Our Monuments (gm_monuments_held, a row each).
	every_country = {
		limit = { has_variable = gm_mon_listed }
		clear_variable_list = gm_monuments_held
		remove_variable = gm_mon_listed
	}
	if = {
		limit = { has_global_variable = gm_mon_lists_written }
		remove_global_variable = gm_mon_lists_written
	}
	if = {
		limit = { has_global_variable_list = gm_monuments }
		every_in_global_list = {
			variable = gm_monuments
			gm_mon_list_one = yes
		}
	}
	# A state whose building has gone (whatever removed it) while monuments
	# stand beside it: the largest takes it over.
	every_state = {
		limit = {
			gm_state_has_secondary = yes
			NOT = { has_building = building_grand_monument }
		}
		gm_state_promote_secondary = yes
	}
}

# Container scope, a secondary monument. Its records from its state's owner,
# once (spec 2026-10-16 §4.4), as gm_state_first_sight writes a state's: a
# returned trophy, a landmark seeded beside another, a dig placed beside or a
# monument from an older save has no raiser, and a bound one would read as
# "does not fit". A demoted primary carries its own and is seen already.
gm_mon_first_sight = {
	if = {
		limit = {
			NOT = { has_variable = gm_tr_seen }
			has_tag = gm_monument
			has_variable = gm_tr_dedication
			var:gm_tr_origin ?= { exists = owner }
		}
		clear_saved_scope = gm_tmp_mon_owner
		var:gm_tr_origin = {
			owner = { save_scope_as = gm_tmp_mon_owner }
		}
		set_variable = gm_tr_seen
		if = {
			limit = { NOT = { has_variable = gm_tr_built_by } }
			set_variable = { name = gm_tr_built_by value = scope:gm_tmp_mon_owner }
		}
		if = {
			limit = {
				gm_mon_is_bound = yes
				NOT = { has_variable = gm_tr_raised_by }
			}
			set_variable = { name = gm_tr_raised_by value = scope:gm_tmp_mon_owner }
		}
		if = {
			limit = {
				gm_mon_kind_faith = yes
				NOT = { has_variable = gm_tr_faith }
			}
			clear_saved_scope = gm_tmp_mon_faith_s
			scope:gm_tmp_mon_owner = {
				religion = { save_scope_as = gm_tmp_mon_faith_s }
			}
			set_variable = { name = gm_tr_faith value = scope:gm_tmp_mon_faith_s }
		}
	}
}

# Container scope. Add $AGE$ to its age grandeur (gm_tr_age), creating it.
gm_mon_add_age = {
	if = {
		limit = { has_variable = gm_tr_age }
		change_variable = { name = gm_tr_age add = $AGE$ }
	}
	else = {
		set_variable = { name = gm_tr_age value = $AGE$ }
	}
}

# Container scope: this secondary monument into its state's list.
gm_mon_list_one = {
	save_scope_as = gm_mon_listing
	if = {
		limit = {
			has_variable = gm_tr_origin
			exists = var:gm_tr_origin
		}
		set_global_variable = gm_mon_lists_written
		var:gm_tr_origin = {
			add_to_variable_list = { name = gm_monuments_here target = scope:gm_mon_listing }
			gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
			owner = {
				add_to_variable_list = { name = gm_monuments_held target = scope:gm_mon_listing }
				set_variable = gm_mon_listed
			}
		}
	}
}

# ---- The primary goes (spec §4.2, §4.3) ----------------------------------------------
# State scope, from gm_state_tear_down, gm_state_tr_remove_whole and the pass:
# the primary monument is gone (torn down, carried off whole, or its building
# lost) and another stood beside it. The largest secondary becomes the
# primary: its records and dedication are the state's, the building standing
# (or made at level 1, if it was lost) switched to its method
# (gm_state_tr_rebuild_from_moving, gm_trophy_effects.txt; sites spec
# 2026-10-18 §3.3: no rebuild), and its container is destroyed. A monument here
# whose Pull Down is ordered and not yet run goes first
# (gm_state_pull_down_ordered), so Tear Them All Down never raises one it meant
# to pull down. Under a marker: a tooltip walks the effects it shows without
# writing (see gm_state_rededicate).
gm_state_promote_secondary = {
	set_variable = gm_promote_click
	if = {
		limit = { has_variable = gm_promote_click }
		remove_variable = gm_promote_click
		gm_state_pull_down_ordered = yes
		clear_saved_scope = gm_mon_promoted
		clear_saved_scope = gm_tr_moving
		ordered_in_list = {
			variable = gm_monuments_here
			order_by = gm_mon_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_mon_promoted
		}
		if = {
			limit = { exists = scope:gm_mon_promoted }
			scope:gm_mon_promoted = { save_scope_as = gm_tr_moving }
			remove_list_variable = { name = gm_monuments_here target = scope:gm_mon_promoted }
			gm_tr_count_add = { VAR = gm_mon_here_count ADD = -1 }
			owner = { remove_list_variable = { name = gm_monuments_held target = scope:gm_mon_promoted } }
			remove_list_global_variable = { name = gm_monuments target = scope:gm_mon_promoted }
			gm_state_tr_rebuild_from_moving = yes
			scope:gm_mon_promoted = { destroy_container = yes }
			owner = { trigger_event = { id = monument_events.20 } }
		}
	}
}

# State scope, from gm_state_promote_secondary, the building already down: every
# monument here whose Pull Down was ordered and has not run yet (by the filter
# gm_state_run_mon_orders keeps an order by: still contested, ordered by this
# owner) is pulled down now. Its levels need no shrink, since the promotion's
# rebuild reads the levels of what remains. One at a time, not through a
# temporary list: Tear Them All Down pulls down several states in one effect,
# and a temporary list lasts until that effect ends. Each pass takes one off
# the list; gm_pull_n bounds the loop well under the engine's 1000 in case a
# removal ever failed to show.
gm_state_pull_down_ordered = {
	if = {
		limit = { has_variable_list = gm_monuments_here }
		save_scope_as = gm_mon_run_state
		owner = { save_scope_as = gm_mon_run_owner }
		set_variable = { name = gm_tr_take_levels value = 0 }
		set_variable = { name = gm_pull_n value = 0 }
		while = {
			limit = {
				var:gm_pull_n < 16
				any_in_list = {
					variable = gm_monuments_here
					has_variable = gm_mon_order
					var:gm_mon_order = flag:tear_down
					gm_mon_is_contested = yes
					var:gm_mon_order_by ?= { this = scope:gm_mon_run_owner }
				}
			}
			random_in_list = {
				variable = gm_monuments_here
				limit = {
					has_variable = gm_mon_order
					var:gm_mon_order = flag:tear_down
					gm_mon_is_contested = yes
					var:gm_mon_order_by ?= { this = scope:gm_mon_run_owner }
				}
				gm_mon_tear_down_one = yes
			}
			change_variable = { name = gm_pull_n add = 1 }
		}
		remove_variable = gm_pull_n
		remove_variable = gm_tr_take_levels
	}
}

# State scope (var:gm_demote_levels its levels). The primary becomes a
# secondary monument: a gm_monument container with every record the state
# held (CONTAINER_RECORDS in test_grand_monument_registry.py), joined to the
# lists at once; then the state forgets it. The building is untouched: its
# level is still everything standing here.
gm_state_demote_primary = {
	save_scope_as = gm_tr_state
	clear_saved_scope = gm_mon_new
	create_container = {
		tags = { gm_monument }
		save_scope_as = gm_mon_new
		on_created = {
			save_scope_as = gm_mon_new
		}
	}
	if = {
		limit = { exists = scope:gm_mon_new }
		scope:gm_mon_new = {
			set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
			set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
			set_variable = { name = gm_tr_site value = scope:gm_tr_state }
			set_variable = { name = gm_tr_site_region value = scope:gm_tr_state.state_region }
			set_variable = { name = gm_tr_levels value = scope:gm_tr_state.var:gm_demote_levels }
			set_variable = { name = gm_tr_share value = flag:whole }
			gm_tr_record_dedication = yes
			gm_tr_copy_from_state = { FROM = gm_skin TO = gm_tr_skin }
			gm_tr_copy_from_state = { FROM = gm_form TO = gm_tr_form }
			gm_tr_copy_from_state = { FROM = gm_landmark TO = gm_tr_landmark }
			gm_tr_copy_from_state = { FROM = gm_namesake TO = gm_tr_namesake }
			gm_tr_copy_from_state = { FROM = gm_name_year TO = gm_tr_name_year }
			gm_tr_copy_from_state = { FROM = gm_name_char TO = gm_tr_name_char }
			gm_tr_copy_from_state = { FROM = gm_name_occasion TO = gm_tr_name_occasion }
			gm_tr_copy_from_state = { FROM = gm_name_enemy_cap TO = gm_tr_name_enemy_cap }
			gm_tr_copy_from_state = { FROM = gm_name TO = gm_tr_name }
			gm_tr_copy_from_state = { FROM = gm_honoree TO = gm_tr_honoree }
			gm_tr_copy_from_state = { FROM = gm_honoree_ig TO = gm_tr_honoree_ig }
			gm_tr_copy_from_state = { FROM = gm_age TO = gm_tr_age }
			gm_tr_copy_from_state = { FROM = gm_start_grandeur TO = gm_tr_start_grandeur }
			gm_tr_copy_from_state = { FROM = gm_raised_by TO = gm_tr_raised_by }
			gm_tr_copy_from_state = { FROM = gm_faith TO = gm_tr_faith }
			gm_tr_copy_from_state = { FROM = gm_built_by TO = gm_tr_built_by }
			gm_tr_copy_from_state = { FROM = gm_contested TO = gm_tr_contested }
			gm_tr_copy_from_state = { FROM = gm_heritage TO = gm_tr_heritage }
			gm_tr_copy_from_state = { FROM = gm_base_ig TO = gm_tr_base_ig }
			gm_tr_copy_from_state = { FROM = gm_supporter_ig TO = gm_tr_supporter_ig }
			gm_tr_copy_from_state = { FROM = gm_ceremony_level TO = gm_tr_ceremony_level }
			gm_tr_copy_from_state = { FROM = gm_signature TO = gm_tr_signature }
			gm_tr_copy_from_state = { FROM = gm_sig_parent TO = gm_tr_sig_parent }
			set_variable = gm_tr_seen
			gm_tr_update_picture = yes
		}
		add_to_global_variable_list = { name = gm_monuments target = scope:gm_mon_new }
		set_global_variable = gm_mon_lists_written
		add_to_variable_list = { name = gm_monuments_here target = scope:gm_mon_new }
		gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
		owner = {
			add_to_variable_list = { name = gm_monuments_held target = scope:gm_mon_new }
			set_variable = gm_mon_listed
		}
		gm_clear_state = { KEEP_LANDMARK = no }
	}
	remove_variable = gm_demote_levels
}

# ---- A monument beside another: contests (spec 2026-10-16 §4.3) --------------------
# Container scope throughout, a secondary monument: the state's contest effects'
# twins (gm_effects.txt), on its gm_tr_* records.

# The IG that opposed its message (the new order's base) and the IG that
# approved it (its supporters), as flags, recorded now.
gm_mon_record_contest_igs = {
	set_variable = { name = gm_tr_base_ig value = flag:none }
	set_variable = { name = gm_tr_supporter_ig value = flag:none }
	if = {
		limit = { gm_mon_is = { KEY = crown } }
		set_variable = { name = gm_tr_base_ig value = flag:intelligentsia }
		set_variable = { name = gm_tr_supporter_ig value = flag:landowners }
	}
	else_if = {
		limit = { gm_mon_is = { KEY = republic } }
		set_variable = { name = gm_tr_base_ig value = flag:landowners }
		set_variable = { name = gm_tr_supporter_ig value = flag:intelligentsia }
	}
	else_if = {
		limit = { gm_mon_is = { KEY = revolution } }
		set_variable = { name = gm_tr_base_ig value = flag:industrialists }
		set_variable = { name = gm_tr_supporter_ig value = flag:trade_unions }
	}
	else_if = {
		limit = { gm_mon_kind_faith = yes }
		set_variable = { name = gm_tr_supporter_ig value = flag:devout }
	}
	else_if = {
		limit = { gm_mon_kind_ruler = yes }
		clear_saved_scope = gm_tmp_mon_cowner
		var:gm_tr_origin ?= {
			owner = { save_scope_as = gm_tmp_mon_cowner }
		}
		if = {
			limit = {
				exists = scope:gm_tmp_mon_cowner
				scope:gm_tmp_mon_cowner = { has_variable = gm_opposition_ig }
			}
			set_variable = { name = gm_tr_base_ig value = scope:gm_tmp_mon_cowner.var:gm_opposition_ig }
		}
		if = {
			limit = { has_variable = gm_tr_honoree_ig }
			set_variable = { name = gm_tr_supporter_ig value = var:gm_tr_honoree_ig }
		}
	}
}

gm_mon_contest = {
	set_variable = { name = gm_tr_contested value = 0 }
	set_variable = gm_tr_new_contest
	gm_mon_record_contest_igs = yes
}

gm_mon_lift_contest = {
	gm_remove_state_var = { VAR = gm_tr_contested }
	gm_remove_state_var = { VAR = gm_tr_heritage }
	gm_remove_state_var = { VAR = gm_tr_new_contest }
	gm_remove_state_var = { VAR = gm_tr_base_ig }
	gm_remove_state_var = { VAR = gm_tr_supporter_ig }
}

# From gm_check_contests (the owner's refresh), each secondary it holds. Seen
# first, a civil war's winner's adoption, then the state's three branches: a
# shrine its owner did not raise is heritage, a contest whose message stands
# again is lifted, a message that falls is contested.
gm_mon_check_contest = {
	gm_mon_first_sight = yes
	if = {
		limit = {
			var:gm_tr_origin ?= {
				owner = { has_variable = gm_cw_adopting }
			}
		}
		gm_mon_adopt_for_winner = yes
	}
	if = {
		limit = {
			has_tag = gm_monument
			has_variable = gm_tr_seen
			gm_mon_is_bound = yes
		}
		if = {
			limit = {
				gm_mon_kind_faith = yes
				gm_mon_raised_by_owner = no
				gm_mon_is_heritage = no
			}
			gm_mon_lift_contest = yes
			set_variable = gm_tr_heritage
		}
		else_if = {
			limit = {
				OR = {
					gm_mon_is_contested = yes
					gm_mon_is_heritage = yes
				}
				gm_mon_message_fits = yes
			}
			gm_mon_lift_contest = yes
		}
		else_if = {
			limit = {
				gm_mon_is_contested = no
				gm_mon_is_heritage = no
				gm_mon_message_fits = no
			}
			gm_mon_contest = yes
		}
	}
}

# From gm_state_changed_hands, each secondary in the state: a bound one the new
# owner did not raise honours a foreign order (contested), a shrine becomes
# heritage. Not for a civil war's transfers.
gm_mon_changed_hands = {
	if = {
		limit = {
			var:gm_tr_origin ?= {
				owner = { has_variable = gm_cw_adopting }
			}
		}
		gm_mon_adopt_for_winner = yes
	}
	if = {
		limit = {
			has_variable = gm_tr_seen
			gm_mon_is_bound = yes
			var:gm_tr_origin ?= {
				owner = {
					NOT = { has_variable = te_cw_role }
					is_revolutionary = no
				}
			}
			gm_mon_raised_by_owner = no
		}
		if = {
			limit = { gm_mon_kind_faith = yes }
			gm_mon_lift_contest = yes
			set_variable = gm_tr_heritage
		}
		else = {
			gm_remove_state_var = { VAR = gm_tr_heritage }
			gm_mon_contest = yes
		}
	}
}

# A civil war's loser raised it; the winner, its state's owner now, did.
gm_mon_adopt_for_winner = {
	clear_saved_scope = gm_tmp_mon_winner
	var:gm_tr_origin ?= {
		owner = { save_scope_as = gm_tmp_mon_winner }
	}
	if = {
		limit = {
			exists = scope:gm_tmp_mon_winner
			has_variable = gm_tr_raised_by
			scope:gm_tmp_mon_winner = { has_variable = gm_cw_loser }
			scope:gm_tmp_mon_winner.var:gm_cw_loser ?= { save_temporary_scope_as = gm_tmp_mon_loser }
			var:gm_tr_raised_by = scope:gm_tmp_mon_loser
		}
		set_variable = { name = gm_tr_raised_by value = scope:gm_tmp_mon_winner }
	}
}

# FACTOR x this monument's grandeur to its owner's ledger of the IG its
# variable $WHO$ names (gm_tr_base_ig or gm_tr_supporter_ig), as
# gm_state_ledger_ig does. The owner's ledgers must exist (gm_init_ledgers).
gm_mon_ledger_ig = {
	set_variable = { name = gm_tr_g value = gm_mon_grandeur }
	save_scope_as = gm_mon_led
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = armed_forces }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = devout }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = industrialists }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = intelligentsia }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = landowners }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = petty_bourgeoisie }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = rural_folk }
	gm_mon_ledger_ig_one = { WHO = $WHO$ FACTOR = $FACTOR$ IG = trade_unions }
}

gm_mon_ledger_ig_one = {

# State scope, from monument_events.30. One rebuild-type action a run: every
# Pull Down ordered here at once, as one shrink by their levels together,
# else one Rededicate (the secondary swapped in as the state's monument, then
# gm_state_rededicate). A state rebuilt today waits a day; orders left over,
# tomorrow. An order whose state changed hands, or whose monument is no
# longer contested, is dropped.
gm_state_run_mon_orders = {
	if = {
		limit = { has_variable_list = gm_monuments_here }
		save_scope_as = gm_mon_run_state
		owner = { save_scope_as = gm_mon_run_owner }
		every_in_list = {
			variable = gm_monuments_here
			limit = {
				has_variable = gm_mon_order
				OR = {
					gm_mon_is_contested = no
					NOT = { var:gm_mon_order_by ?= { this = scope:gm_mon_run_owner } }
				}
			}
			remove_variable = gm_mon_order
			gm_remove_state_var = { VAR = gm_mon_order_by }
		}
		every_in_list = {
			variable = gm_monuments_here
			limit = {
				has_variable = gm_mon_order
				var:gm_mon_order = flag:tear_down
			}
			add_to_temporary_list = gm_mon_tearing
		}
		every_in_list = {
			list = gm_mon_tearing
			gm_mon_tear_down_one = yes
		}
		if = {
			limit = { NOT = { has_variable = gm_ceremony_pending } }
			clear_saved_scope = gm_mon_raise
			random_in_list = {
				variable = gm_monuments_here
				limit = {
					has_variable = gm_mon_order
					var:gm_mon_order = flag:rededicate
				}
				save_scope_as = gm_mon_raise
			}
			if = {
				limit = { exists = scope:gm_mon_raise }
				scope:gm_mon_raise = {
					remove_variable = gm_mon_order
					gm_remove_state_var = { VAR = gm_mon_order_by }
				}
				if = {
					limit = { gm_state_can_rededicate_beside = yes }
					gm_state_swap_primary = yes
					gm_state_rededicate = yes
				}
			}
		}
		if = {
			limit = {
				has_variable_list = gm_monuments_here
				any_in_list = {
					variable = gm_monuments_here
					has_variable = gm_mon_order
				}
			}
			trigger_event = { id = monument_events.30 days = 1 }
		}
		owner = { trigger_event = { id = monument_events.20 } }
	}
}

# ---- Anniversaries (ageing spec 2026-10-16 §5) -------------------------------------
# Country scope, an anniversary's immediate. The grandest monument of
# dedication $KEY$ that holds anniversaries, a state's own or one beside
# another: scope:monument_state its state, scope:monument_mon the container
# when it is one beside another. What its public option would add to its age
# is cached on the country (gm_ann_gain), for the option's tooltip: event loc
# cannot reach a container (scripting_best_practices.md).
gm_ann_pick = {
	clear_saved_scope = monument_state
	clear_saved_scope = monument_mon
	clear_saved_scope = gm_ann_mon
	set_variable = { name = gm_ann_g value = 0 }
	set_variable = { name = gm_ann_gain value = 0 }
	if = {
		limit = {
			any_scope_state = { gm_state_holds_anniversary = { PM = pm_monument_$KEY$ } }
		}
		ordered_scope_state = {
			limit = { gm_state_holds_anniversary = { PM = pm_monument_$KEY$ } }
			order_by = gm_state_primary_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = monument_state
		}
		scope:monument_state = {
			set_variable = { name = gm_ann_g_here value = gm_state_primary_grandeur }
			set_variable = { name = gm_ann_gain_here value = gm_evt_grandeur }
		}
		set_variable = { name = gm_ann_g value = scope:monument_state.var:gm_ann_g_here }
		set_variable = { name = gm_ann_gain value = scope:monument_state.var:gm_ann_gain_here }
		scope:monument_state = {
			remove_variable = gm_ann_g_here
			remove_variable = gm_ann_gain_here
		}
	}
	if = {
		limit = { has_variable_list = gm_monuments_held }
		ordered_in_list = {
			variable = gm_monuments_held
			limit = { gm_mon_holds_anniversary = { KEY = $KEY$ } }
			order_by = gm_mon_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_ann_mon
		}
	}
	if = {
		limit = { exists = scope:gm_ann_mon }
		scope:gm_ann_mon = {
			set_variable = { name = gm_tr_g value = gm_mon_grandeur }
			set_variable = { name = gm_ann_gain_here value = gm_mon_evt_grandeur }
		}
		if = {
			limit = {
				OR = {
					NOT = { exists = scope:monument_state }
					scope:gm_ann_mon.var:gm_tr_g > var:gm_ann_g
				}
			}
			scope:gm_ann_mon = {
				save_scope_as = monument_mon
				var:gm_tr_origin ?= { save_scope_as = monument_state }
			}
			set_variable = { name = gm_ann_gain value = scope:gm_ann_mon.var:gm_ann_gain_here }
		}
		scope:gm_ann_mon = { remove_variable = gm_ann_gain_here }
	}
	remove_variable = gm_ann_g
}

		if = {
			limit = { has_variable = gm_tr_age }
			change_variable = { name = gm_tr_age subtract = scope:gm_tr_state.var:gm_tr_take_age }
			clamp_variable = { name = gm_tr_age min = 0 }
		}
		if = {
			limit = { has_variable = gm_tr_start_grandeur }
			change_variable = { name = gm_tr_start_grandeur subtract = scope:gm_tr_state.var:gm_tr_take_start }
			clamp_variable = { name = gm_tr_start_grandeur min = 0 }
		}
		scope:gm_tr_state = {
			set_variable = { name = gm_tr_cooldown days = 3650 }
			owner = { trigger_event = { id = monument_events.20 } }
		}
	}
}

# ---- The AI's choice of which monument ----------------------------------------------
# State scope, before gm_state_tr_take on the AI's occupation roll. One roll
# falls on one monument of the state: the primary, or (every monument as
# likely) a secondary, named in gm_tr_take_target. A primary that cannot go
# (undedicated, a Grand Shrine: gm_state_tr_takeable) leaves the roll to the

gm_state_tr_ai_choose_target = {

		limit = {
			gm_state_has_secondary = yes
			gm_state_tr_takeable_secondary = yes
		}
		clear_saved_scope = gm_mon_pick
		if = {
			limit = { gm_state_tr_takeable = no }
			random_in_list = { variable = gm_monuments_here save_scope_as = gm_mon_pick }
		}
		else = {
			random = {
				chance = gm_state_tr_ai_secondary_chance
				random_in_list = { variable = gm_monuments_here save_scope_as = gm_mon_pick }
			}

			limit = { exists = scope:gm_mon_pick }

		}
	}
}

# ---- A merged state (issue #936; sites spec 2026-10-18 §3.7) ------------------------
# A state is a (region, owner) pair: when one country comes to hold two parts
# of a state region the engine merges them, and the part absorbed takes its
# variables with it, where a primary monument's records live (its size, skin,
# form, landmark, name, raiser, age, contest). So each month the pass keeps a
# SHADOW of every recorded primary: a parentless container tagged gm_shadow,
# in the global list gm_shadows, linked from its state (gm_shadow) and back
# (gm_tr_origin, gm_tr_origin_region), with the records a demotion copies
# (CONTAINER_RECORDS in test_grand_monument_registry.py), its levels
# (gm_tr_levels) and its owner (gm_sh_owner). A shadow whose state is gone is
# an ORPHAN, and the part of its region its last owner holds adopts it from
# the records alone (gm_sh_settle_one): onto a finished building there no
# record describes, beside the monument there, or as the monument there,
# raised into a new building, if none stands; with no such part, it is lost.
# The finished building left standing is put back to level 1. That runs
# before anything reads the building: monument_events.1, .29, the refresh and
# gm_state_changed_hands ask gm_state_merge_pending first. A shadow is written
# only by the pass and a raise; the refresh, which tooltips walk, only asks
# the hidden event. No timed variable on a shadow
# (container_timed_variable_audit).

# No scope, from gm_monument_monthly. Orphans first, then stale shadows, then
# every recorded primary's shadow written afresh.
gm_shadow_monthly = {
	if = {
		limit = { has_global_variable_list = gm_shadows }
		# A region with no state left keeps nothing.
		every_in_global_list = {
			variable = gm_shadows
			limit = {
				gm_sh_is_orphan = yes
				var:gm_tr_origin_region = {
					NOT = { any_scope_state = { exists = owner } }
				}
			}
			add_to_temporary_list = gm_sh_nowhere
		}
		every_in_list = {
			list = gm_sh_nowhere
			debug_log = "TE_MONUMENTS: a monument's shadow is dropped: no state is left in its region"
			gm_sh_drop = yes
		}
		# Each orphan's region to a hidden event on one of its states, which
		# adopts every orphan there, with that state as ROOT for its log lines.
		every_in_global_list = {
			variable = gm_shadows
			limit = { gm_sh_is_orphan = yes }
			var:gm_tr_origin_region = { add_to_temporary_list = gm_sh_orphan_regions }
		}
		every_in_list = {
			list = gm_sh_orphan_regions
			random_scope_state = {
				trigger_event = { id = monument_events.31 }
			}
		}
		# A shadow whose state stands with no recorded primary, or links another.
		every_in_global_list = {
			variable = gm_shadows
			limit = {
				exists = var:gm_tr_origin
				save_temporary_scope_as = gm_tmp_sh
				var:gm_tr_origin = {
					OR = {
						NOT = { has_building = building_grand_monument }
						NOT = { has_variable = gm_seen }
						NOT = { var:gm_shadow ?= scope:gm_tmp_sh }
					}
				}
			}
			add_to_temporary_list = gm_sh_stale
		}
		every_in_list = { list = gm_sh_stale gm_sh_drop = yes }
	}
	every_state = {
		limit = {
			has_variable = gm_seen
			has_building = building_grand_monument
		}
		gm_state_write_shadow = yes
	}
}

# State scope, from gm_shadow_monthly and gm_state_raise_orphan. A recorded
# primary's shadow, made once and refreshed in place. Not while its method is
# being put back, a ceremony asks or script made its building today, when its
# dedication may read as the engine's copy: last month's stands meanwhile.
gm_state_write_shadow = {
	if = {
		limit = {
			has_building = building_grand_monument
			has_variable = gm_seen
			gm_state_is_dedicated = yes
			NOT = { has_variable = gm_restore_pm }
			NOT = { has_variable = gm_ceremony_pending }
			NOT = { has_variable = gm_rebuilding }
		}
		save_scope_as = gm_tr_state
		clear_saved_scope = gm_sh_new
		if = {
			limit = {
				has_variable = gm_shadow
				exists = var:gm_shadow
			}
			var:gm_shadow = { save_scope_as = gm_sh_new }
		}
		else = {
			create_container = {
				tags = { gm_shadow }
				save_scope_as = gm_sh_new
				on_created = {
					save_scope_as = gm_sh_new
				}
			}
			if = {
				limit = { exists = scope:gm_sh_new }
				add_to_global_variable_list = { name = gm_shadows target = scope:gm_sh_new }
				set_variable = { name = gm_shadow value = scope:gm_sh_new }
			}
		}
		if = {
			limit = { exists = scope:gm_sh_new }
			set_variable = { name = gm_sh_levels value = gm_state_primary_levels }
			scope:gm_sh_new = {
				set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
				set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
				set_variable = { name = gm_tr_site value = scope:gm_tr_state }
				set_variable = { name = gm_tr_site_region value = scope:gm_tr_state.state_region }
				set_variable = { name = gm_tr_levels value = scope:gm_tr_state.var:gm_sh_levels }
				set_variable = { name = gm_sh_owner value = scope:gm_tr_state.owner }
				set_variable = { name = gm_tr_share value = flag:whole }
				gm_tr_record_dedication = yes
				gm_sh_mirror = { FROM = gm_skin TO = gm_tr_skin }
				gm_sh_mirror = { FROM = gm_form TO = gm_tr_form }
				gm_sh_mirror = { FROM = gm_landmark TO = gm_tr_landmark }
				gm_sh_mirror = { FROM = gm_namesake TO = gm_tr_namesake }
				gm_sh_mirror = { FROM = gm_name_year TO = gm_tr_name_year }
				gm_sh_mirror = { FROM = gm_name_char TO = gm_tr_name_char }
				gm_sh_mirror = { FROM = gm_name_occasion TO = gm_tr_name_occasion }
				gm_sh_mirror = { FROM = gm_name_enemy_cap TO = gm_tr_name_enemy_cap }
				gm_sh_mirror = { FROM = gm_name TO = gm_tr_name }
				gm_sh_mirror = { FROM = gm_honoree TO = gm_tr_honoree }
				gm_sh_mirror = { FROM = gm_honoree_ig TO = gm_tr_honoree_ig }
				gm_sh_mirror = { FROM = gm_age TO = gm_tr_age }
				gm_sh_mirror = { FROM = gm_start_grandeur TO = gm_tr_start_grandeur }
				gm_sh_mirror = { FROM = gm_raised_by TO = gm_tr_raised_by }
				gm_sh_mirror = { FROM = gm_faith TO = gm_tr_faith }
				gm_sh_mirror = { FROM = gm_built_by TO = gm_tr_built_by }
				gm_sh_mirror = { FROM = gm_contested TO = gm_tr_contested }
				gm_sh_mirror = { FROM = gm_heritage TO = gm_tr_heritage }
				gm_sh_mirror = { FROM = gm_base_ig TO = gm_tr_base_ig }
				gm_sh_mirror = { FROM = gm_supporter_ig TO = gm_tr_supporter_ig }
				gm_sh_mirror = { FROM = gm_ceremony_level TO = gm_tr_ceremony_level }
				gm_sh_mirror = { FROM = gm_signature TO = gm_tr_signature }
				gm_sh_mirror = { FROM = gm_sig_parent TO = gm_tr_sig_parent }
				gm_sh_mirror = { FROM = gm_picture_code TO = gm_tr_picture_code }
				set_variable = gm_tr_seen
			}
			remove_variable = gm_sh_levels
		}
	}
}

# Container scope, a shadow (scope:gm_tr_state its state): the state's variable
# $FROM$ as $TO$, or none when the state has none (a contest lifted, a name
# cleared), so a shadow refreshed in place keeps no record its monument lost.
gm_sh_mirror = {
	if = {
		limit = {
			scope:gm_tr_state = { has_variable = $FROM$ }
		}
		set_variable = { name = $TO$ value = scope:gm_tr_state.var:$FROM$ }
	}
	else = {
		gm_remove_state_var = { VAR = $TO$ }
	}
}

# Container scope, a shadow: out of the list, its state's link to it gone,
# destroyed.
gm_sh_drop = {
	save_scope_as = gm_sh_dropping
	if = {
		limit = {
			has_variable = gm_tr_origin
			exists = var:gm_tr_origin
			var:gm_tr_origin = { var:gm_shadow ?= scope:gm_sh_dropping }
		}
		var:gm_tr_origin = { remove_variable = gm_shadow }
	}
	remove_list_global_variable = { name = gm_shadows target = scope:gm_sh_dropping }
	destroy_container = yes
}

# State scope, from gm_clear_state and gm_state_tr_remove_whole: this state's
# primary is forgotten or gone, and its shadow goes at once. A merge later this
# month would otherwise adopt a demoted monument a second time beside its own
# container, or raise one carried off.
gm_state_drop_shadow = {
	if = {
		limit = { has_variable = gm_shadow }
		if = {
			limit = { exists = var:gm_shadow }
			var:gm_shadow = { gm_sh_drop = yes }
		}
		gm_remove_state_var = { VAR = gm_shadow }
	}
}

# State scope, real execution only: monument_events.1, .29 and .31, and
# gm_state_changed_hands. Adopt every orphan of this state's region. With a
# building in the region made or removed today (it may not read back yet), it
# waits a day.
gm_state_adopt_orphans = {
	clear_saved_scope = gm_sh_region
	state_region = { save_scope_as = gm_sh_region }
	if = {
		limit = {
			scope:gm_sh_region = { any_scope_state = { has_variable = gm_rebuilding } }
		}
		trigger_event = { id = monument_events.31 days = 1 }
	}
	else = {
		# The monuments that stood beside it come first: the lists must name
		# them before an orphan is settled.
		if = {
			limit = {
				has_global_variable_list = gm_monuments
				any_in_global_list = {
					variable = gm_monuments
					gm_sh_orphan_of = { REGION = scope:gm_sh_region }
				}
			}
			every_in_global_list = {
				variable = gm_monuments
				limit = { gm_sh_orphan_of = { REGION = scope:gm_sh_region } }
				gm_tr_repair_places = yes
			}
			gm_monument_pass = yes
		}
		if = {
			limit = {
				has_global_variable_list = gm_shadows
				any_in_global_list = {
					variable = gm_shadows
					gm_sh_orphan_of = { REGION = scope:gm_sh_region }
				}
			}
			# The pass may have made a building for one promoted onto a lost one.
			if = {
				limit = {
					scope:gm_sh_region = { any_scope_state = { has_variable = gm_rebuilding } }
				}
				trigger_event = { id = monument_events.31 days = 1 }
			}
			else = {
				gm_sh_settle_region = yes
			}
		}
	}
}

# State scope (scope:gm_sh_region saved), from gm_state_adopt_orphans. Each
# orphan of the region, landmarks first (the issue's rule: a landmark's
# identity wins), then the finished buildings left standing back at level 1.
gm_sh_settle_region = {
	every_in_global_list = {
		variable = gm_shadows
		limit = {
			gm_sh_orphan_of = { REGION = scope:gm_sh_region }
			has_variable = gm_tr_landmark
		}
		add_to_temporary_list = gm_sh_left
	}
	every_in_global_list = {
		variable = gm_shadows
		limit = {
			gm_sh_orphan_of = { REGION = scope:gm_sh_region }
			NOT = { has_variable = gm_tr_landmark }
		}
		add_to_temporary_list = gm_sh_left
	}
	every_in_list = {
		list = gm_sh_left
		gm_sh_settle_one = yes
	}
	# Not an older save's building before monument_events.32 records it: it
	# stands at its full height until then (gm_migrated).
	scope:gm_sh_region = {
		every_scope_state = {
			limit = {
				has_building = building_grand_monument
				has_variable = gm_migrated
				gm_state_level > 1
			}
			gm_state_reset_building_level = yes
		}
	}
}

# Container scope, an orphaned shadow (scope:gm_sh_region its region), still an
# orphan when its turn comes (an earlier one may have settled it). Its ground
# went into the part of its region its last owner holds (gm_sh_owner): there it
# takes a finished building no record describes as its own (the merge brought
# it), stands beside the monument there, or, with no finished building there,
# is raised as that part's monument into a new one. With no such part (it
# changed hands as it merged), the most populous part of the region with a
# finished building takes it beside its monument. Else it is lost.
gm_sh_settle_one = {
	if = {
		limit = { gm_sh_orphan_of = { REGION = scope:gm_sh_region } }
		save_scope_as = gm_sh_pick
		clear_saved_scope = gm_sh_was_owner
		clear_saved_scope = gm_sh_target
		var:gm_sh_owner ?= { save_scope_as = gm_sh_was_owner }
		if = {
			limit = { exists = scope:gm_sh_was_owner }
			scope:gm_sh_region = {
				random_scope_state = {
					limit = { exists = owner owner = scope:gm_sh_was_owner }
					save_scope_as = gm_sh_target
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:gm_sh_target } }
			scope:gm_sh_region = {
				ordered_scope_state = {
					limit = {
						has_building = building_grand_monument
						has_variable = gm_seen
					}
					order_by = state_population
					position = 0
					check_range_bounds = no
					save_scope_as = gm_sh_target
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:gm_sh_target } }
			gm_sh_lost = yes
		}
		else_if = {
			limit = { scope:gm_sh_target = { gm_state_unrecorded = yes } }
			scope:gm_sh_target = { gm_state_raise_orphan = yes }
		}
		else_if = {
			limit = { scope:gm_sh_target = { has_building = building_grand_monument has_variable = gm_seen } }
			gm_sh_adopt_beside = yes
			debug_log = "TE_MONUMENTS: a monument whose state merged into another part of its region stands beside the monument there: [SCOPE.ScriptValue('gm_sh_log_levels')|0] levels recorded"
		}
		else_if = {
			limit = { scope:gm_sh_target = { NOT = { has_building = building_grand_monument } } }
			scope:gm_sh_target = { gm_state_raise_orphan = yes }
		}
		else = {
			gm_sh_lost = yes
		}
	}
}

# Container scope, an orphaned shadow, with scope:gm_sh_target the state that
# holds its region's ground (scope:gm_sh_region its region). It stands there as
# a monument beside the one there, as a returned trophy does
# (gm_tr_become_monument): the same container, retagged and moved from
# gm_shadows to gm_monuments, in the state's and its owner's lists at once, its
# size its records (sites spec 2026-10-18 §3.3).
gm_sh_adopt_beside = {
	save_scope_as = gm_sh_moving
	gm_sh_log_numbers = yes
	remove_list_global_variable = { name = gm_shadows target = scope:gm_sh_moving }
	add_to_global_variable_list = { name = gm_monuments target = scope:gm_sh_moving }
	set_global_variable = gm_mon_lists_written
	remove_tag = gm_shadow
	add_tag = gm_monument
	# A shadow's own note, not a monument's record.
	gm_remove_state_var = { VAR = gm_sh_owner }
	set_variable = { name = gm_tr_origin value = scope:gm_sh_target }
	set_variable = { name = gm_tr_origin_region value = scope:gm_sh_target.state_region }
	set_variable = { name = gm_tr_site value = scope:gm_sh_target }
	set_variable = { name = gm_tr_site_region value = scope:gm_sh_target.state_region }
	gm_tr_update_picture = yes
	scope:gm_sh_target = {
		add_to_variable_list = { name = gm_monuments_here target = scope:gm_sh_moving }
		gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
		owner = {
			add_to_variable_list = { name = gm_monuments_held target = scope:gm_sh_moving }
			set_variable = gm_mon_listed
			trigger_event = { id = monument_events.20 }
		}
	}
	set_global_variable = { name = gm_sh_log_kept_now value = 1 }
	gm_monument_merge_self_check = yes
}

# State scope (scope:gm_sh_target), with scope:gm_sh_pick an orphaned shadow of
# its region: the monument there is that one, come with the merge, onto the
# building no record describes, or into a new one at level 1 where none
# stands. It stands beside first (the lists), then is raised onto the state's
# variables as a swap raises a secondary (gm_state_raise_secondary): its
# records, its dedication put back through gm_restore_pm. Its shadow again at
# once when its method already reads back.
gm_state_raise_orphan = {
	if = {
		limit = { NOT = { has_building = building_grand_monument } }
		set_variable = { name = gm_rebuilding days = 1 }
		create_building = {
			building = building_grand_monument
			level = 1
		}
	}
	scope:gm_sh_pick = { gm_sh_adopt_beside = yes }
	clear_saved_scope = gm_mon_raise
	scope:gm_sh_pick = { save_scope_as = gm_mon_raise }
	gm_state_raise_secondary = yes
	debug_log = "TE_MONUMENTS: a monument whose state merged into another part of its region stands again as the monument of the state that holds its ground: [SCOPE.ScriptValue('gm_sh_log_levels')|0] levels recorded"
	gm_state_restore_dedication = yes
	gm_state_write_shadow = yes
}

# Container scope, an orphaned shadow no part of its region takes. Logged, then
# gone.
gm_sh_lost = {
	gm_sh_log_numbers = yes
	set_global_variable = { name = gm_sh_log_kept_now value = 0 }
	debug_log = "TE_MONUMENTS: a monument whose state merged into another part of its region is lost, no part of the region taking it: [SCOPE.ScriptValue('gm_sh_log_levels')|0] levels recorded, [SCOPE.ScriptValue('gm_sh_log_buildings')|0] finished buildings in its region"
	gm_monument_merge_self_check = yes
	gm_sh_drop = yes
}

# Container scope, a shadow: what the log line and the self-check print.
gm_sh_log_numbers = {
	set_global_variable = { name = gm_sh_log_levels_now value = gm_mon_levels }
	set_global_variable = { name = gm_sh_log_buildings_now value = 0 }
	scope:gm_sh_region = {
		every_scope_state = {
			limit = { has_building = building_grand_monument }
			change_global_variable = { name = gm_sh_log_buildings_now add = 1 }
		}
	}
}

# State scope, a state of a merged region: its finished building stands taller
# than 1 (the merge added two together) and goes back to level 1 on its
# recorded dedication (gm_dedication), not whatever method the merged building
# kept; with none recorded, on its method now (recorded first, as a return
# records it; a raise just recorded the orphan's). What the merge left it at
# is kept for the self-check (gm_monument_merge_level_self_check).
gm_state_reset_building_level = {
	set_variable = { name = gm_merge_check value = gm_state_level }
	if = {
		limit = { has_variable = gm_dedication }
		set_variable = { name = gm_restore_pm value = var:gm_dedication }
	}
	else_if = {
		limit = { NOT = { has_variable = gm_restore_pm } }
		gm_state_mark_restore_pm = yes
	}
	set_variable = { name = gm_rebuilding days = 1 }
	remove_building = building_grand_monument
	create_building = {
		building = building_grand_monument
		level = 1
	}
}

# State scope, from gm_state_monthly: once a game, a day or more after a merge's
# reset (gm_state_reset_building_level) and once the dedication is recorded,
# whether the finished building stands at level 1 on its recorded dedication.
# What a merge leaves a building at (two levels added together, and on which
# method) is unread (sites plan A Task 7); the line prints the level it left.
gm_monument_merge_level_self_check = {
	if = {
		limit = {
			has_variable = gm_merge_check
			has_variable = gm_dedication
			has_building = building_grand_monument
			NOT = { has_variable = gm_rebuilding }
			NOT = { has_variable = gm_restore_pm }
		}
		if = {
			limit = { NOT = { has_global_variable = gm_merge_level_self_checked } }
			set_global_variable = gm_merge_level_self_checked
			if = {
				limit = {
					gm_state_level = 1
					gm_state_stands_on_dedication = yes
				}
				debug_log = "TE_SELF_CHECK OK monument_merge_level1 (a merged region's finished Grand Monument, which the merge left at level [SCOPE.ScriptValue('gm_state_merge_check_was')|0], stands at level 1 on its recorded dedication)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN monument_merge_level1 (a merged region's finished Grand Monument, which the merge left at level [SCOPE.ScriptValue('gm_state_merge_check_was')|0], stands at level [SCOPE.ScriptValue('gm_state_level')|0] and not on its recorded dedication at level 1; see docs/testing/grand-monuments-ageing-playtest.md)"
			}
		}
		remove_variable = gm_merge_check
	}
}

# Container scope, a monument beside another whose state is gone, from
# gm_tr_repair_places: its state again is the part of its region with a
# recorded monument, the most populous; with none, gm_tr_repair_places's rule.
gm_mon_repair_origin = {
	save_scope_as = gm_mon_repairing
	clear_saved_scope = gm_tr_repair_state
	var:gm_tr_origin_region = {
		ordered_scope_state = {
			limit = {
				has_building = building_grand_monument
				has_variable = gm_seen
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			save_scope_as = gm_tr_repair_state
		}
	}
	if = {
		limit = { exists = scope:gm_tr_repair_state }
		set_variable = { name = gm_tr_origin value = scope:gm_tr_repair_state }
		set_variable = { name = gm_tr_site value = scope:gm_tr_repair_state }
		set_variable = { name = gm_tr_site_region value = scope:gm_tr_repair_state.state_region }
	}
}

# ---- Self-check: a merge keeps its monument (issue #936) -------------------------------
# Any scope (the globals gm_sh_log_numbers wrote), the first time a game
# settles an orphan: whether it was kept (stands beside, or raised) or lost.
gm_monument_merge_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = gm_merge_self_checked }
			has_global_variable = gm_sh_log_levels_now
			global_var:gm_sh_log_levels_now >= 1
		}
		set_global_variable = gm_merge_self_checked
		if = {
			limit = { global_var:gm_sh_log_kept_now = 1 }
			debug_log = "TE_SELF_CHECK OK monument_merge_kept (a monument whose state merged into another part of its region stands again from its shadow: [SCOPE.ScriptValue('gm_sh_log_levels')|0] levels recorded, [SCOPE.ScriptValue('gm_sh_log_buildings')|0] finished buildings in the region)"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN monument_merge_kept (a monument whose state merged into another part of its region is lost: no part of its region took it; [SCOPE.ScriptValue('gm_sh_log_levels')|0] levels recorded, [SCOPE.ScriptValue('gm_sh_log_buildings')|0] finished buildings there; see docs/testing/grand-monuments-ageing-playtest.md)"
		}
	}
}

# ---- Self-check: the world-wealth baseline (Plan B) ------------------------------------
# No scope, from gm_monument_monthly, once: on a game's first pass (before
# 1836.3.1), whether world GDP is within a tenth of the constant the rate is
# measured against (gm_world_gdp_1836). An older save never reads it.
gm_world_base_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = gm_world_base_checked }
			game_date < 1836.3.1
		}
		set_global_variable = gm_world_base_checked
		if = {
			limit = {
				global_gdp_script_value >= gm_world_gdp_1836_low
				global_gdp_script_value <= gm_world_gdp_1836_high
			}
			debug_log = "TE_SELF_CHECK OK gm_world_base (world GDP [SCOPE.ScriptValue('global_gdp_script_value')|0] is within a tenth of the 1836 constant [SCOPE.ScriptValue('gm_world_gdp_1836')|0])"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN gm_world_base (world GDP [SCOPE.ScriptValue('global_gdp_script_value')|0] is not within a tenth of the 1836 constant [SCOPE.ScriptValue('gm_world_gdp_1836')|0]: set gm_world_gdp_1836 to it; see docs/testing/grand-monuments-ageing-playtest.md)"
		}
	}
}

# ---- Self-check (once a game, after the seeding) ---------------------------------------
# No scope, from gm_landmarks_on_game_started_after_lobby. Whether a returning
# trophy is told from a piece of what stands: the rule compares the standing
# monument's landmark with the trophy's across scopes
# (gm_state_stands_for_moving, gm_monument_triggers.txt), which nobody has seen
# the engine do. 1836's data says what each must read as: the Parthenon's
# marbles are a piece of the Parthenon standing in Attica, the Luxor Obelisk
# stands apart from the Obelisks of Karnak. With either premise absent (the
# owner proved wrong, the rule off) it logs nothing, so OK cannot be a false
# pass. The fallback is in docs/testing/grand-monuments-multiple-playtest.md.
gm_monument_self_check = {
	clear_saved_scope = gm_mon_chk_piece
	clear_saved_scope = gm_mon_chk_apart
	if = {
		limit = { has_global_variable_list = gm_trophies }
		random_in_global_list = {
			variable = gm_trophies
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:parthenon
				var:gm_tr_origin ?= {
					has_building = building_grand_monument
					has_variable = gm_landmark
					var:gm_landmark = flag:parthenon
				}
			}
			save_scope_as = gm_mon_chk_piece
		}
		random_in_global_list = {
			variable = gm_trophies
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:luxor_obelisk
				var:gm_tr_origin ?= {
					has_building = building_grand_monument
					has_variable = gm_landmark
					var:gm_landmark = flag:obelisks_of_karnak
				}
			}
			save_scope_as = gm_mon_chk_apart
		}
	}
	if = {
		limit = { exists = scope:gm_mon_chk_piece }
		scope:gm_mon_chk_piece = {
			save_scope_as = gm_tr_moving
			if = {
				limit = { gm_tr_rejoins_primary = yes }
				debug_log = "TE_SELF_CHECK OK monument_piece_rejoins (the Parthenon's marbles read as a piece of the Parthenon in Attica)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN monument_piece_rejoins (the Parthenon's marbles do not read as a piece of the Parthenon standing in Attica, so a returned piece will stand as a monument of its own: take the fallback in docs/testing/grand-monuments-multiple-playtest.md)"
			}
		}
	}
	if = {
		limit = { exists = scope:gm_mon_chk_apart }
		scope:gm_mon_chk_apart = {
			save_scope_as = gm_tr_moving
			if = {
				limit = { gm_tr_stands_apart = yes }
				debug_log = "TE_SELF_CHECK OK monument_artifact_stands_apart (the Luxor Obelisk reads as apart from the Obelisks of Karnak)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN monument_artifact_stands_apart (the Luxor Obelisk does not read as apart from the Obelisks of Karnak, so a returned artifact will be merged into the monument standing: take the fallback in docs/testing/grand-monuments-multiple-playtest.md)"
			}
		}
	}
}
