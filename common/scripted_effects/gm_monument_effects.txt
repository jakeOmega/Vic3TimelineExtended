# ============================================================================
# GRAND MONUMENTS — several monuments in one state: scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-10-13-multiple-monuments-per-state-design.md.
#
# A SECONDARY monument is a script container tagged gm_monument, with no parent
# (it outlives every country), in the global list gm_monuments. It is a
# returned trophy that stands apart from the monument already in its state (or
# a co-located landmark, gm_seed_landmark), and keeps its trophy's records
# (gm_tr_*, gm_trophy_effects.txt: skin, form, landmark, name, dedication,
# picture, honoree) so the trophy names, forms, parts and pictures serve it.
#   gm_tr_origin, gm_tr_origin_region   the state it stands in
#   gm_tr_levels                        its grandeur (gm_mon_grandeur)
# The list and the state's holder are read live. The monthly pass
# (gm_monument_monthly) writes each state's gm_monuments_here and
# gm_mon_here_count from the global list, so a state that was merged, or a
# civil war's winner (who inherits no lists), has its rows back.
# No timed variable on a container: it would never expire
# (container_timed_variable_audit).
#
# The building's level is the truth (gm_values.txt, gm_state_grandeur): nothing
# here rebuilds a building on a pulse. A rebuild runs where a monument is added
# or taken, from a hidden event (gm_state_tr_receive and its siblings).

# ---- The monthly pass (from on_monthly_pulse) --------------------------------------
# No scope. Repair each secondary's state, then the lists. Nothing to do in a
# game none has touched: gm_mon_lists_written stands while any list does.
gm_monument_monthly = {
	# Old saves' held specials, once a round (signatures spec §5.6).
	if = {
		limit = { gm_system_enabled = yes }
		gm_sig_seed_monthly = yes
	}
	if = {
		limit = {
			gm_system_enabled = yes
			OR = {
				has_global_variable = gm_mon_lists_written
				has_global_variable_list = gm_monuments
			}
		}
		if = {
			limit = { has_global_variable_list = gm_monuments }
			every_in_global_list = {
				variable = gm_monuments
				gm_tr_repair_places = yes
				# A landmark whose picture is drawn later, or a monument from an
				# older save, catches up within a month (as a trophy does).
				gm_tr_update_picture = yes
				# And its unique dedication (signatures spec §4.4).
				gm_tr_sig_backfill = yes
				# Its raiser recorded once (spec 2026-10-16 §4.4), then a month
				# older (§2).
				gm_mon_first_sight = yes
				if = {
					limit = { gm_mon_ages = yes }
					gm_mon_add_age = { AGE = gm_mon_age_step }
				}
				# An order its event never ran (the building was gone that day):
				# its state's event again, which runs it or drops it.
				if = {
					limit = { has_variable = gm_mon_order }
					var:gm_tr_origin ?= {
						trigger_event = { id = monument_events.30 }
					}
				}
			}
		}
		gm_monument_pass = yes
	}
	# The one-off repair of monuments short of a piece long gone (owner's call
	# 2026-10-17, § "One-off repair" below): before the shadows, which then
	# note what it put back.
	if = {
		limit = { gm_system_enabled = yes }
		gm_short_repair_monthly = yes
	}
	# Every recorded primary's shadow, in a game with no secondary too (issue
	# #936, § "A merged state" below).
	if = {
		limit = { gm_system_enabled = yes }
		gm_shadow_monthly = yes
	}
}

# Any scope. Every state's list of secondaries, from the global list. Last
# month's are cleared first; the lists are rebuilt each time.
gm_monument_pass = {
	every_state = {
		limit = { has_variable = gm_mon_here_count }
		clear_variable_list = gm_monuments_here
		remove_variable = gm_mon_here_count
	}
	# The owners' lists, for Our Monuments (gm_monuments_held, a row each).
	every_country = {
		limit = { has_variable = gm_mon_listed }
		clear_variable_list = gm_monuments_held
		remove_variable = gm_mon_listed
	}
	if = {
		limit = { has_global_variable = gm_mon_lists_written }
		remove_global_variable = gm_mon_lists_written
	}
	if = {
		limit = { has_global_variable_list = gm_monuments }
		every_in_global_list = {
			variable = gm_monuments
			gm_mon_list_one = yes
		}
	}
	# A state whose building has gone (whatever removed it) while monuments
	# stand beside it: the largest takes it over.
	every_state = {
		limit = {
			gm_state_has_secondary = yes
			NOT = { has_building = building_grand_monument }
		}
		gm_state_promote_secondary = yes
	}
}

# Container scope, a secondary monument. Its records from its state's owner,
# once (spec 2026-10-16 §4.4), as gm_state_first_sight writes a state's: a
# returned trophy, a landmark seeded beside another, a dig placed beside or a
# monument from an older save has no raiser, and a bound one would read as
# "does not fit". A demoted primary carries its own and is seen already.
gm_mon_first_sight = {
	if = {
		limit = {
			NOT = { has_variable = gm_tr_seen }
			has_tag = gm_monument
			has_variable = gm_tr_dedication
			var:gm_tr_origin ?= { exists = owner }
		}
		clear_saved_scope = gm_tmp_mon_owner
		var:gm_tr_origin = {
			owner = { save_scope_as = gm_tmp_mon_owner }
		}
		set_variable = gm_tr_seen
		if = {
			limit = { NOT = { has_variable = gm_tr_built_by } }
			set_variable = { name = gm_tr_built_by value = scope:gm_tmp_mon_owner }
		}
		if = {
			limit = {
				gm_mon_is_bound = yes
				NOT = { has_variable = gm_tr_raised_by }
			}
			set_variable = { name = gm_tr_raised_by value = scope:gm_tmp_mon_owner }
		}
		if = {
			limit = {
				gm_mon_kind_faith = yes
				NOT = { has_variable = gm_tr_faith }
			}
			clear_saved_scope = gm_tmp_mon_faith_s
			scope:gm_tmp_mon_owner = {
				religion = { save_scope_as = gm_tmp_mon_faith_s }
			}
			set_variable = { name = gm_tr_faith value = scope:gm_tmp_mon_faith_s }
		}
	}
}

# Container scope. Add $AGE$ to its age grandeur (gm_tr_age), creating it.
gm_mon_add_age = {
	if = {
		limit = { has_variable = gm_tr_age }
		change_variable = { name = gm_tr_age add = $AGE$ }
	}
	else = {
		set_variable = { name = gm_tr_age value = $AGE$ }
	}
}

# Container scope: this secondary monument into its state's list.
gm_mon_list_one = {
	save_scope_as = gm_mon_listing
	if = {
		limit = {
			has_variable = gm_tr_origin
			exists = var:gm_tr_origin
		}
		set_global_variable = gm_mon_lists_written
		var:gm_tr_origin = {
			add_to_variable_list = { name = gm_monuments_here target = scope:gm_mon_listing }
			gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
			owner = {
				add_to_variable_list = { name = gm_monuments_held target = scope:gm_mon_listing }
				set_variable = gm_mon_listed
			}
		}
	}
}

# ---- The primary goes (spec §4.2, §4.3) ----------------------------------------------
# State scope, from gm_state_tear_down, gm_state_tr_remove_whole and the pass:
# the primary monument is gone (torn down, carried off whole, or its building
# lost) and another stood beside it. The largest secondary becomes the
# primary: the building is built at the levels of everything that remains, with
# that monument's records and dedication (gm_state_tr_rebuild_from_moving,
# gm_trophy_effects.txt), and its container is destroyed. A monument here whose
# Pull Down is ordered and not yet run goes first (gm_state_pull_down_ordered),
# so Tear Them All Down never raises one it meant to pull down.
# The work runs only under a marker set here: a button's or an option's tooltip
# walks the effects it shows without writing, so the rebuild's read of
# gm_rebuild_level would find it unset through the ladder's 200 cases (see
# gm_state_rededicate).
gm_state_promote_secondary = {
	set_variable = gm_promote_click
	if = {
		limit = { has_variable = gm_promote_click }
		remove_variable = gm_promote_click
		gm_state_pull_down_ordered = yes
		clear_saved_scope = gm_mon_promoted
		clear_saved_scope = gm_tr_moving
		ordered_in_list = {
			variable = gm_monuments_here
			order_by = gm_mon_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_mon_promoted
		}
		if = {
			limit = { exists = scope:gm_mon_promoted }
			scope:gm_mon_promoted = { save_scope_as = gm_tr_moving }
			# Everything that remains, the promoted one included: read before it
			# leaves the list.
			set_variable = {
				name = gm_rebuild_level
				value = {
					value = gm_state_secondary_levels
					max = 200
				}
			}
			remove_list_variable = { name = gm_monuments_here target = scope:gm_mon_promoted }
			gm_tr_count_add = { VAR = gm_mon_here_count ADD = -1 }
			owner = { remove_list_variable = { name = gm_monuments_held target = scope:gm_mon_promoted } }
			remove_list_global_variable = { name = gm_monuments target = scope:gm_mon_promoted }
			gm_state_tr_rebuild_from_moving = yes
			scope:gm_mon_promoted = { destroy_container = yes }
			owner = { trigger_event = { id = monument_events.20 } }
		}
	}
}

# State scope, from gm_state_promote_secondary, the building already down: every
# monument here whose Pull Down was ordered and has not run yet (by the filter
# gm_state_run_mon_orders keeps an order by: still contested, ordered by this
# owner) is pulled down now. Its levels need no shrink, since the promotion's
# rebuild reads the levels of what remains. One at a time, not through a
# temporary list: Tear Them All Down pulls down several states in one effect,
# and a temporary list lasts until that effect ends. Each pass takes one off
# the list; gm_pull_n bounds the loop well under the engine's 1000 in case a
# removal ever failed to show.
gm_state_pull_down_ordered = {
	if = {
		limit = { has_variable_list = gm_monuments_here }
		save_scope_as = gm_mon_run_state
		owner = { save_scope_as = gm_mon_run_owner }
		set_variable = { name = gm_tr_take_levels value = 0 }
		set_variable = { name = gm_pull_n value = 0 }
		while = {
			limit = {
				var:gm_pull_n < 16
				any_in_list = {
					variable = gm_monuments_here
					has_variable = gm_mon_order
					var:gm_mon_order = flag:tear_down
					gm_mon_is_contested = yes
					var:gm_mon_order_by ?= { this = scope:gm_mon_run_owner }
				}
			}
			random_in_list = {
				variable = gm_monuments_here
				limit = {
					has_variable = gm_mon_order
					var:gm_mon_order = flag:tear_down
					gm_mon_is_contested = yes
					var:gm_mon_order_by ?= { this = scope:gm_mon_run_owner }
				}
				gm_mon_tear_down_one = yes
			}
			change_variable = { name = gm_pull_n add = 1 }
		}
		remove_variable = gm_pull_n
		remove_variable = gm_tr_take_levels
	}
}

# ---- Building beside (spec 2026-10-16 §3.2) ----------------------------------------
# State scope, real execution only (monument_events.1 and .29; never the
# refresh, which tooltips walk). Levels were built on a finished monument:
# it moves into a container beside the levels' new monument, which stands as
# the state's primary, undedicated, waiting for its Dedicate button (owner,
# 2026-10-18), which asks once nothing here is under construction, as the
# AI's ceremony does. Its remaining queued levels join it. gm_newcomer marks it
# for the settle check (gm_state_settle_newcomer). Marked as seen at once
# (gm_state_mark_awaiting): its method was switched in this effect, so a
# reading of the old dedication must not pass for the engine's copy.
gm_state_found_beside = {
	set_variable = { name = gm_demote_levels value = var:gm_levels }
	gm_state_demote_primary = yes
	set_variable = gm_newcomer
	gm_state_undedicate = yes
	gm_state_mark_awaiting = yes
	if = {
		limit = { gm_state_ai_dedicates_now = yes }
		gm_state_start_ceremony = yes
	}
	debug_log = "TE_MONUMENTS: new levels founded a monument beside the one standing; the building stands at [SCOPE.ScriptValue('gm_state_level')|0] levels, [SCOPE.ScriptValue('gm_state_monument_count')|0] monuments"
	owner = { trigger_event = { id = monument_events.20 } }
}

# State scope (var:gm_demote_levels its levels). The primary becomes a
# secondary monument: a gm_monument container with every record the state
# held (CONTAINER_RECORDS in test_grand_monument_registry.py), joined to the
# lists at once; then the state forgets it. The building is untouched: its
# level is still everything standing here.
gm_state_demote_primary = {
	save_scope_as = gm_tr_state
	clear_saved_scope = gm_mon_new
	create_container = {
		tags = { gm_monument }
		save_scope_as = gm_mon_new
		on_created = {
			save_scope_as = gm_mon_new
		}
	}
	if = {
		limit = { exists = scope:gm_mon_new }
		scope:gm_mon_new = {
			set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
			set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
			set_variable = { name = gm_tr_site value = scope:gm_tr_state }
			set_variable = { name = gm_tr_site_region value = scope:gm_tr_state.state_region }
			set_variable = { name = gm_tr_levels value = scope:gm_tr_state.var:gm_demote_levels }
			set_variable = { name = gm_tr_share value = flag:whole }
			gm_tr_record_dedication = yes
			gm_tr_copy_from_state = { FROM = gm_skin TO = gm_tr_skin }
			gm_tr_copy_from_state = { FROM = gm_form TO = gm_tr_form }
			gm_tr_copy_from_state = { FROM = gm_landmark TO = gm_tr_landmark }
			gm_tr_copy_from_state = { FROM = gm_namesake TO = gm_tr_namesake }
			gm_tr_copy_from_state = { FROM = gm_name_year TO = gm_tr_name_year }
			gm_tr_copy_from_state = { FROM = gm_name_char TO = gm_tr_name_char }
			gm_tr_copy_from_state = { FROM = gm_name_occasion TO = gm_tr_name_occasion }
			gm_tr_copy_from_state = { FROM = gm_name_enemy_cap TO = gm_tr_name_enemy_cap }
			gm_tr_copy_from_state = { FROM = gm_name TO = gm_tr_name }
			gm_tr_copy_from_state = { FROM = gm_honoree TO = gm_tr_honoree }
			gm_tr_copy_from_state = { FROM = gm_honoree_ig TO = gm_tr_honoree_ig }
			gm_tr_copy_from_state = { FROM = gm_age TO = gm_tr_age }
			gm_tr_copy_from_state = { FROM = gm_raised_by TO = gm_tr_raised_by }
			gm_tr_copy_from_state = { FROM = gm_faith TO = gm_tr_faith }
			gm_tr_copy_from_state = { FROM = gm_built_by TO = gm_tr_built_by }
			gm_tr_copy_from_state = { FROM = gm_contested TO = gm_tr_contested }
			gm_tr_copy_from_state = { FROM = gm_heritage TO = gm_tr_heritage }
			gm_tr_copy_from_state = { FROM = gm_base_ig TO = gm_tr_base_ig }
			gm_tr_copy_from_state = { FROM = gm_supporter_ig TO = gm_tr_supporter_ig }
			gm_tr_copy_from_state = { FROM = gm_ceremony_level TO = gm_tr_ceremony_level }
			gm_tr_copy_from_state = { FROM = gm_signature TO = gm_tr_signature }
			gm_tr_copy_from_state = { FROM = gm_sig_parent TO = gm_tr_sig_parent }
			set_variable = gm_tr_seen
			gm_tr_update_picture = yes
		}
		add_to_global_variable_list = { name = gm_monuments target = scope:gm_mon_new }
		set_global_variable = gm_mon_lists_written
		add_to_variable_list = { name = gm_monuments_here target = scope:gm_mon_new }
		gm_tr_count_add = { VAR = gm_mon_here_count ADD = 1 }
		owner = {
			add_to_variable_list = { name = gm_monuments_held target = scope:gm_mon_new }
			set_variable = gm_mon_listed
		}
		gm_clear_state = { KEEP_LANDMARK = no }
	}
	remove_variable = gm_demote_levels
}

# ---- A monument beside another: contests (spec 2026-10-16 §4.3) --------------------
# Container scope throughout, a secondary monument: the state's contest effects'
# twins (gm_effects.txt), on its gm_tr_* records.

# The IG that opposed its message (the new order's base) and the IG that
# approved it (its supporters), as flags, recorded now.
gm_mon_record_contest_igs = {
	set_variable = { name = gm_tr_base_ig value = flag:none }
	set_variable = { name = gm_tr_supporter_ig value = flag:none }
	if = {
		limit = { gm_mon_is = { KEY = crown } }
		set_variable = { name = gm_tr_base_ig value = flag:intelligentsia }
		set_variable = { name = gm_tr_supporter_ig value = flag:landowners }
	}
	else_if = {
		limit = { gm_mon_is = { KEY = republic } }
		set_variable = { name = gm_tr_base_ig value = flag:landowners }
		set_variable = { name = gm_tr_supporter_ig value = flag:intelligentsia }
	}
	else_if = {
		limit = { gm_mon_is = { KEY = revolution } }
		set_variable = { name = gm_tr_base_ig value = flag:industrialists }
		set_variable = { name = gm_tr_supporter_ig value = flag:trade_unions }
	}
	else_if = {
		limit = { gm_mon_kind_faith = yes }
		set_variable = { name = gm_tr_supporter_ig value = flag:devout }
	}
	else_if = {
		limit = { gm_mon_kind_ruler = yes }
		clear_saved_scope = gm_tmp_mon_cowner
		var:gm_tr_origin ?= {
			owner = { save_scope_as = gm_tmp_mon_cowner }
		}
		if = {
			limit = {
				exists = scope:gm_tmp_mon_cowner
				scope:gm_tmp_mon_cowner = { has_variable = gm_opposition_ig }
			}
			set_variable = { name = gm_tr_base_ig value = scope:gm_tmp_mon_cowner.var:gm_opposition_ig }
		}
		if = {
			limit = { has_variable = gm_tr_honoree_ig }
			set_variable = { name = gm_tr_supporter_ig value = var:gm_tr_honoree_ig }
		}
	}
}

gm_mon_contest = {
	set_variable = { name = gm_tr_contested value = 0 }
	set_variable = gm_tr_new_contest
	gm_mon_record_contest_igs = yes
}

gm_mon_lift_contest = {
	gm_remove_state_var = { VAR = gm_tr_contested }
	gm_remove_state_var = { VAR = gm_tr_heritage }
	gm_remove_state_var = { VAR = gm_tr_new_contest }
	gm_remove_state_var = { VAR = gm_tr_base_ig }
	gm_remove_state_var = { VAR = gm_tr_supporter_ig }
}

# From gm_check_contests (the owner's refresh), each secondary it holds. Seen
# first, a civil war's winner's adoption, then the state's three branches: a
# shrine its owner did not raise is heritage, a contest whose message stands
# again is lifted, a message that falls is contested.
gm_mon_check_contest = {
	gm_mon_first_sight = yes
	if = {
		limit = {
			var:gm_tr_origin ?= {
				owner = { has_variable = gm_cw_adopting }
			}
		}
		gm_mon_adopt_for_winner = yes
	}
	if = {
		limit = {
			has_tag = gm_monument
			has_variable = gm_tr_seen
			gm_mon_is_bound = yes
		}
		if = {
			limit = {
				gm_mon_kind_faith = yes
				gm_mon_raised_by_owner = no
				gm_mon_is_heritage = no
			}
			gm_mon_lift_contest = yes
			set_variable = gm_tr_heritage
		}
		else_if = {
			limit = {
				OR = {
					gm_mon_is_contested = yes
					gm_mon_is_heritage = yes
				}
				gm_mon_message_fits = yes
			}
			gm_mon_lift_contest = yes
		}
		else_if = {
			limit = {
				gm_mon_is_contested = no
				gm_mon_is_heritage = no
				gm_mon_message_fits = no
			}
			gm_mon_contest = yes
		}
	}
}

# From gm_state_changed_hands, each secondary in the state: a bound one the new
# owner did not raise honours a foreign order (contested), a shrine becomes
# heritage. Not for a civil war's transfers.
gm_mon_changed_hands = {
	if = {
		limit = {
			var:gm_tr_origin ?= {
				owner = { has_variable = gm_cw_adopting }
			}
		}
		gm_mon_adopt_for_winner = yes
	}
	if = {
		limit = {
			has_variable = gm_tr_seen
			gm_mon_is_bound = yes
			var:gm_tr_origin ?= {
				owner = {
					NOT = { has_variable = te_cw_role }
					is_revolutionary = no
				}
			}
			gm_mon_raised_by_owner = no
		}
		if = {
			limit = { gm_mon_kind_faith = yes }
			gm_mon_lift_contest = yes
			set_variable = gm_tr_heritage
		}
		else = {
			gm_remove_state_var = { VAR = gm_tr_heritage }
			gm_mon_contest = yes
		}
	}
}

# A civil war's loser raised it; the winner, its state's owner now, did.
gm_mon_adopt_for_winner = {
	clear_saved_scope = gm_tmp_mon_winner
	var:gm_tr_origin ?= {
		owner = { save_scope_as = gm_tmp_mon_winner }
	}
	if = {
		limit = {
			exists = scope:gm_tmp_mon_winner
			has_variable = gm_tr_raised_by
			scope:gm_tmp_mon_winner = { has_variable = gm_cw_loser }
			scope:gm_tmp_mon_winner.var:gm_cw_loser ?= { save_temporary_scope_as = gm_tmp_mon_loser }
			var:gm_tr_raised_by = scope:gm_tmp_mon_loser
		}
		set_variable = { name = gm_tr_raised_by value = scope:gm_tmp_mon_winner }
	}
}

# FACTOR x this monument's grandeur to its owner's ledger of the IG its
# variable $WHO$ names (gm_tr_base_ig or gm_tr_supporter_ig), as
# gm_state_ledger_ig does. The owner's ledgers must exist (gm_init_ledgers).
gm_mon_ledger_ig = {
	set_variable = { name = gm_tr_g value = gm_mon_grandeur }
	save_scope_as = gm_mon_led

# State scope, from monument_events.30. One rebuild-type action a run: every
# Pull Down ordered here at once, as one shrink by their levels together,
# else one Rededicate (the secondary swapped in as the state's monument, then
# gm_state_rededicate). A state rebuilt today waits a day; orders left over,
# tomorrow. An order whose state changed hands, or whose monument is no
# longer contested, is dropped.
gm_state_run_mon_orders = {
	if = {
		limit = { has_variable = gm_rebuilding }
		trigger_event = { id = monument_events.30 days = 1 }
	}
	else_if = {
		limit = { has_variable_list = gm_monuments_here }
		save_scope_as = gm_mon_run_state
		owner = { save_scope_as = gm_mon_run_owner }
		every_in_list = {
			variable = gm_monuments_here
			limit = {
				has_variable = gm_mon_order
				OR = {
					gm_mon_is_contested = no
					NOT = { var:gm_mon_order_by ?= { this = scope:gm_mon_run_owner } }
				}
			}
			remove_variable = gm_mon_order
			gm_remove_state_var = { VAR = gm_mon_order_by }
		}
		every_in_list = {
			variable = gm_monuments_here
			limit = {
				has_variable = gm_mon_order
				var:gm_mon_order = flag:tear_down
			}
			add_to_temporary_list = gm_mon_tearing
		}
		set_variable = { name = gm_tr_take_levels value = 0 }
		every_in_list = {
			list = gm_mon_tearing
			gm_mon_tear_down_one = yes
		}
		if = {
			limit = { var:gm_tr_take_levels > 0 }
			gm_state_tr_shrink_by = yes
		}
		else = {
			clear_saved_scope = gm_mon_raise
			random_in_list = {
				variable = gm_monuments_here
				limit = {
					has_variable = gm_mon_order
					var:gm_mon_order = flag:rededicate
				}
				save_scope_as = gm_mon_raise
			}
			if = {
				limit = { exists = scope:gm_mon_raise }
				scope:gm_mon_raise = {
					remove_variable = gm_mon_order
					gm_remove_state_var = { VAR = gm_mon_order_by }
				}
				if = {
					limit = { gm_state_can_rededicate_beside = yes }
					gm_state_swap_primary = yes
					gm_state_rededicate = yes
				}
			}
		}
		remove_variable = gm_tr_take_levels
		if = {
			limit = {
				has_variable_list = gm_monuments_here
				any_in_list = {
					variable = gm_monuments_here
					has_variable = gm_mon_order
				}
			}
			trigger_event = { id = monument_events.30 days = 1 }
		}
		owner = { trigger_event = { id = monument_events.20 } }
	}
}

# ---- Anniversaries (ageing spec 2026-10-16 §5) -------------------------------------
# Country scope, an anniversary's immediate. The grandest monument of
# dedication $KEY$ that holds anniversaries, a state's own or one beside
# another: scope:monument_state its state, scope:monument_mon the container
# when it is one beside another. What its public option would add to its age
# is cached on the country (gm_ann_gain), for the option's tooltip: event loc
# cannot reach a container (scripting_best_practices.md).
gm_ann_pick = {
	clear_saved_scope = monument_state
	clear_saved_scope = monument_mon
	clear_saved_scope = gm_ann_mon
	set_variable = { name = gm_ann_g value = 0 }
	set_variable = { name = gm_ann_gain value = 0 }
	if = {
		limit = {
			any_scope_state = { gm_state_holds_anniversary = { PM = pm_monument_$KEY$ } }
		}
		ordered_scope_state = {
			limit = { gm_state_holds_anniversary = { PM = pm_monument_$KEY$ } }
			order_by = gm_state_primary_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = monument_state
		}
		scope:monument_state = {
			set_variable = { name = gm_ann_g_here value = gm_state_primary_grandeur }
			set_variable = { name = gm_ann_gain_here value = gm_evt_grandeur }
		}
		set_variable = { name = gm_ann_g value = scope:monument_state.var:gm_ann_g_here }
		set_variable = { name = gm_ann_gain value = scope:monument_state.var:gm_ann_gain_here }
		scope:monument_state = {
			remove_variable = gm_ann_g_here
			remove_variable = gm_ann_gain_here
		}
	}
	if = {
		limit = { has_variable_list = gm_monuments_held }
		ordered_in_list = {
			variable = gm_monuments_held
			limit = { gm_mon_holds_anniversary = { KEY = $KEY$ } }
			order_by = gm_mon_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_ann_mon
		}
	}
	if = {
		limit = { exists = scope:gm_ann_mon }
		scope:gm_ann_mon = {
			set_variable = { name = gm_tr_g value = gm_mon_grandeur }
			set_variable = { name = gm_ann_gain_here value = gm_mon_evt_grandeur }
		}
		if = {
			limit = {
				OR = {
					NOT = { exists = scope:monument_state }
					scope:gm_ann_mon.var:gm_tr_g > var:gm_ann_g
				}
			}
			scope:gm_ann_mon = {
				save_scope_as = monument_mon
				var:gm_tr_origin ?= { save_scope_as = monument_state }
			}
			set_variable = { name = gm_ann_gain value = scope:gm_ann_mon.var:gm_ann_gain_here }
		}
		scope:gm_ann_mon = { remove_variable = gm_ann_gain_here }
	}
	remove_variable = gm_ann_g
}

			gm_mon_copy_record = { KEY = gm_tr_honoree_ig }
			gm_mon_copy_record = { KEY = gm_tr_dedication }
			# Its unique dedication: a piece counts for its taker at half
			# (signatures spec §2.4, §7).
			gm_mon_copy_record = { KEY = gm_tr_signature }
			gm_mon_copy_record = { KEY = gm_tr_sig_parent }
			gm_tr_update_picture = yes
		}
		change_variable = { name = gm_tr_levels subtract = scope:gm_tr_state.var:gm_tr_take_levels }
		if = {
			limit = { has_variable = gm_tr_age }
			change_variable = { name = gm_tr_age subtract = scope:gm_tr_state.var:gm_tr_take_age }
			clamp_variable = { name = gm_tr_age min = 0 }
		}
		scope:gm_tr_state = {
			set_variable = { name = gm_tr_cooldown days = 3650 }
			gm_state_tr_shrink_by = yes
		}
	}
}

# Container scope (the new trophy): the secondary's variable $KEY$, when it has

}

# State scope (var:gm_tr_take_levels the levels carried off). The building is
# built again smaller, keeping the primary's dedication (recorded first, as a
# return records it) and every monument standing here: the primary's own and
# the secondaries' levels still add up to what is left. The ladder cannot take
# levels off a standing building (gm_state_tr_remove_piece).
# Under a marker, as Rededicate is: a tooltip walks this body without writing,
# and the ladder's switch reads gm_rebuild_level.
gm_state_tr_shrink_by = {
	set_variable = gm_shrink_click
	if = {
		limit = { has_variable = gm_shrink_click }
		remove_variable = gm_shrink_click

		# (gm_state_tr_add_levels).
		if = {
			limit = {
				NOT = { has_variable = gm_restore_pm }
				NOT = { has_variable = gm_ceremony_pending }
			}
			gm_state_mark_restore_pm = yes
		}
		set_variable = { name = gm_rebuilding days = 1 }
		set_variable = {
			name = gm_rebuild_level
			value = {
				value = gm_state_level
				subtract = var:gm_tr_take_levels
				min = 1
				max = 200
			}
		}
		remove_building = building_grand_monument
		te_construction_market_build_specified_level = {
			SPEC_TYPE = building_grand_monument
			SPEC_LEVEL = var:gm_rebuild_level
		}
		set_variable = { name = gm_rebuild_asked value = var:gm_rebuild_level }
		remove_variable = gm_rebuild_level
		owner = { trigger_event = { id = monument_events.20 } }
	}
}

# ---- The AI's choice of which monument ----------------------------------------------
# State scope, before gm_state_tr_take on the AI's occupation roll. One roll
# falls on one monument of the state: the primary, or (every monument as
# likely) a secondary, named in gm_tr_take_target. A primary that cannot go
# (undedicated, a Grand Shrine: gm_state_tr_takeable) leaves the roll to the

gm_state_tr_ai_choose_target = {

		limit = {
			gm_state_has_secondary = yes
			gm_state_tr_takeable_secondary = yes
		}

		if = {
			limit = { gm_state_tr_takeable = no }
			random_in_list = { variable = gm_monuments_here save_scope_as = gm_mon_pick }
		}
		else = {
			random = {
				chance = gm_state_tr_ai_secondary_chance
				random_in_list = { variable = gm_monuments_here save_scope_as = gm_mon_pick }
			}
		}
		if = {
			limit = { exists = scope:gm_mon_pick }
			set_variable = { name = gm_tr_take_target value = scope:gm_mon_pick }
		}
	}
}

# ---- A merged state (issue #936; ageing spec 2026-10-16 §9) ------------------------
# A state is a (region, owner) pair: when one country comes to hold two parts
# of a state region the engine merges them, and the part absorbed takes its
# variables with it, where a primary monument's records live (its skin, form,
# landmark, name, raiser, age, contest). Its building may come along, levels
# and all (unread: docs/testing/grand-monuments-ageing-playtest.md).
# So each month the pass keeps a SHADOW of every recorded primary: a parentless
# container tagged gm_shadow, in the global list gm_shadows, linked from its
# state (gm_shadow) and back (gm_tr_origin, gm_tr_origin_region), with the
# records a demotion copies (CONTAINER_RECORDS in
# test_grand_monument_registry.py) and its levels (gm_tr_levels). A shadow
# whose state is gone is an ORPHAN, and the state that now holds its region's
# ground adopts it (gm_state_adopt_orphans): a building no record describes
# takes it as its own monument, a recorded one whose building holds levels
# its records don't account for takes it beside its own, a monument that
# lacked levels when last seen (a revolt split its state, and the orphan is
# its piece coming home: gm_sh_rejoin, found 2026-10-17) takes back its
# levels, and with none of these it is lost. Each shadow notes what its
# building lacks of its records (gm_sh_short). That runs before anything
# reads the building's levels as new ones:
# monument_events.1, .29, the refresh and gm_state_changed_hands ask
# gm_state_merge_pending first. A shadow is written only by the pass and a
# raise; the refresh, which tooltips walk, only asks the hidden event.
# No timed variable on a shadow (container_timed_variable_audit).

# No scope, from gm_monument_monthly. Orphans first, then stale shadows, then
# every recorded primary's shadow written afresh.
gm_shadow_monthly = {
	if = {
		limit = { has_global_variable_list = gm_shadows }
		# A region with no state left keeps nothing.
		every_in_global_list = {
			variable = gm_shadows
			limit = {
				gm_sh_is_orphan = yes
				var:gm_tr_origin_region = {
					NOT = { any_scope_state = { exists = owner } }
				}
			}
			add_to_temporary_list = gm_sh_nowhere
		}
		every_in_list = {
			list = gm_sh_nowhere
			debug_log = "TE_MONUMENTS: a monument's shadow is dropped: no state is left in its region"
			gm_sh_drop = yes
		}
		# Each orphan's region to a hidden event on one of its states, which
		# adopts every orphan there, with that state as ROOT for its log lines.
		every_in_global_list = {
			variable = gm_shadows
			limit = { gm_sh_is_orphan = yes }
			var:gm_tr_origin_region = { add_to_temporary_list = gm_sh_orphan_regions }
		}
		every_in_list = {
			list = gm_sh_orphan_regions
			random_scope_state = {
				trigger_event = { id = monument_events.31 }
			}
		}
		# A shadow whose state stands with no recorded primary, or links another.
		every_in_global_list = {
			variable = gm_shadows
			limit = {
				exists = var:gm_tr_origin
				save_temporary_scope_as = gm_tmp_sh
				var:gm_tr_origin = {
					OR = {
						NOT = { has_building = building_grand_monument }
						NOT = { has_variable = gm_seen }
						NOT = { var:gm_shadow ?= scope:gm_tmp_sh }
					}
				}
				# Not a monument's shadow its revolt's piece is to take, nor a
				# piece's its monument is to take back: after the merge the
				# engine may still point the merged state's record somewhere
				# (§ "A revolt's split piece").
				NOT = { var:gm_tr_split_piece ?= { has_variable = gm_piece_of } }
				NOT = { var:gm_tr_piece_of ?= { has_variable = gm_split_piece } }
			}
			add_to_temporary_list = gm_sh_stale
		}
		every_in_list = { list = gm_sh_stale gm_sh_drop = yes }
	}
	every_state = {
		limit = {
			has_variable = gm_seen
			has_building = building_grand_monument
		}
		gm_state_write_shadow = yes
	}
}

# State scope, from gm_shadow_monthly and gm_state_raise_orphan. A recorded
# primary's shadow, made once and refreshed in place. Not while its method is
# being put back, a ceremony asks or a rebuild is under way, when its
# dedication may read as the engine's copy: last month's stands meanwhile.
gm_state_write_shadow = {
	if = {
		limit = {
			has_building = building_grand_monument
			has_variable = gm_seen
			gm_state_is_dedicated = yes
			NOT = { has_variable = gm_restore_pm }
			NOT = { has_variable = gm_ceremony_pending }
			NOT = { has_variable = gm_rebuilding }
		}
		save_scope_as = gm_tr_state
		clear_saved_scope = gm_sh_new
		if = {
			limit = {
				has_variable = gm_shadow
				exists = var:gm_shadow
			}
			var:gm_shadow = { save_scope_as = gm_sh_new }
		}
		else = {
			create_container = {
				tags = { gm_shadow }
				save_scope_as = gm_sh_new
				on_created = {
					save_scope_as = gm_sh_new
				}
			}
			if = {
				limit = { exists = scope:gm_sh_new }
				add_to_global_variable_list = { name = gm_shadows target = scope:gm_sh_new }
				set_variable = { name = gm_shadow value = scope:gm_sh_new }
			}
		}
		if = {
			limit = { exists = scope:gm_sh_new }
			set_variable = { name = gm_sh_levels value = gm_state_shadow_levels }
			set_variable = { name = gm_sh_short_now value = gm_state_levels_short }
			scope:gm_sh_new = {
				# What its building lacks of its records: a revolt's split took
				# them, and a piece that comes home finds them missing here. The
				# passes that have seen it short: a split's piece is recognized
				# only while the shortfall is new (gm_state_split_whole).
				if = {
					limit = { scope:gm_tr_state.var:gm_sh_short_now >= 1 }
					set_variable = { name = gm_sh_short value = scope:gm_tr_state.var:gm_sh_short_now }
					if = {
						limit = { has_variable = gm_sh_short_passes }
						change_variable = { name = gm_sh_short_passes add = 1 }
					}
					else = {
						set_variable = { name = gm_sh_short_passes value = 1 }
					}
				}
				else = {
					gm_remove_state_var = { VAR = gm_sh_short }
					gm_remove_state_var = { VAR = gm_sh_short_passes }
				}
				set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
				set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
				set_variable = { name = gm_tr_site value = scope:gm_tr_state }
				set_variable = { name = gm_tr_site_region value = scope:gm_tr_state.state_region }
				set_variable = { name = gm_tr_levels value = scope:gm_tr_state.var:gm_sh_levels }
				set_variable = { name = gm_tr_share value = flag:whole }
				gm_tr_record_dedication = yes
				gm_sh_mirror = { FROM = gm_skin TO = gm_tr_skin }
				gm_sh_mirror = { FROM = gm_form TO = gm_tr_form }
				gm_sh_mirror = { FROM = gm_landmark TO = gm_tr_landmark }
				gm_sh_mirror = { FROM = gm_namesake TO = gm_tr_namesake }
				gm_sh_mirror = { FROM = gm_name_year TO = gm_tr_name_year }
				gm_sh_mirror = { FROM = gm_name_char TO = gm_tr_name_char }
				gm_sh_mirror = { FROM = gm_name_occasion TO = gm_tr_name_occasion }
				gm_sh_mirror = { FROM = gm_name_enemy_cap TO = gm_tr_name_enemy_cap }
				gm_sh_mirror = { FROM = gm_name TO = gm_tr_name }
				gm_sh_mirror = { FROM = gm_honoree TO = gm_tr_honoree }
				gm_sh_mirror = { FROM = gm_honoree_ig TO = gm_tr_honoree_ig }
				gm_sh_mirror = { FROM = gm_age TO = gm_tr_age }
				gm_sh_mirror = { FROM = gm_raised_by TO = gm_tr_raised_by }
				gm_sh_mirror = { FROM = gm_faith TO = gm_tr_faith }
				gm_sh_mirror = { FROM = gm_built_by TO = gm_tr_built_by }
				gm_sh_mirror = { FROM = gm_contested TO = gm_tr_contested }
				gm_sh_mirror = { FROM = gm_heritage TO = gm_tr_heritage }
				gm_sh_mirror = { FROM = gm_base_ig TO = gm_tr_base_ig }
				gm_sh_mirror = { FROM = gm_supporter_ig TO = gm_tr_supporter_ig }
				gm_sh_mirror = { FROM = gm_ceremony_level TO = gm_tr_ceremony_level }
				gm_sh_mirror = { FROM = gm_signature TO = gm_tr_signature }
				gm_sh_mirror = { FROM = gm_sig_parent TO = gm_tr_sig_parent }
				gm_sh_mirror = { FROM = gm_picture_code TO = gm_tr_picture_code }
				# A revolt's piece and its monument know each other (§ "A
				# revolt's split piece"): whichever part's state the merge
				# takes, the other finds it from the shadow left behind.
				gm_sh_mirror = { FROM = gm_piece_of TO = gm_tr_piece_of }
				gm_sh_mirror = { FROM = gm_split_piece TO = gm_tr_split_piece }
				set_variable = gm_tr_seen
			}
			remove_variable = gm_sh_levels
			remove_variable = gm_sh_short_now
		}
	}
}

# Container scope, a shadow (scope:gm_tr_state its state): the state's variable
# $FROM$ as $TO$, or none when the state has none (a contest lifted, a name
# cleared), so a shadow refreshed in place keeps no record its monument lost.
gm_sh_mirror = {
	if = {
		limit = {
			scope:gm_tr_state = { has_variable = $FROM$ }
		}
		set_variable = { name = $TO$ value = scope:gm_tr_state.var:$FROM$ }
	}
	else = {
		gm_remove_state_var = { VAR = $TO$ }
	}
}

# Container scope, a shadow: out of the list, its state's link to it gone,
# destroyed.
gm_sh_drop = {
	save_scope_as = gm_sh_dropping
	if = {
		limit = {
			has_variable = gm_tr_origin
			exists = var:gm_tr_origin
			var:gm_tr_origin = { var:gm_shadow ?= scope:gm_sh_dropping }
		}
		var:gm_tr_origin = { remove_variable = gm_shadow }
	}
	remove_list_global_variable = { name = gm_shadows target = scope:gm_sh_dropping }
	destroy_container = yes
}

# State scope, from gm_clear_state and gm_state_tr_remove_whole: this state's
# primary is forgotten or gone, and its shadow goes at once. A merge later this
# month would otherwise adopt a demoted monument a second time beside its own
# container, or raise one carried off.
gm_state_drop_shadow = {
	if = {
		limit = { has_variable = gm_shadow }
		if = {
			limit = { exists = var:gm_shadow }
			var:gm_shadow = { gm_sh_drop = yes }
		}
		gm_remove_state_var = { VAR = gm_shadow }
	}
}

# State scope, real execution only: monument_events.1, .29 and .31, and
# gm_state_changed_hands. Adopt every orphan of this state's region. With a
# building in the region rebuilt today (it may not read back yet: one rebuild
# a state a day), it waits a day.
gm_state_adopt_orphans = {
	clear_saved_scope = gm_sh_region
	state_region = { save_scope_as = gm_sh_region }
	if = {
		limit = {
			scope:gm_sh_region = { any_scope_state = { has_variable = gm_rebuilding } }
		}
		trigger_event = { id = monument_events.31 days = 1 }
	}
	else = {
		# The monuments that stood beside it come first: their levels are in the
		# building too, and the lists must name them before a level is counted.
		if = {
			limit = {
				has_global_variable_list = gm_monuments
				any_in_global_list = {
					variable = gm_monuments
					gm_sh_orphan_of = { REGION = scope:gm_sh_region }
				}
			}
			every_in_global_list = {
				variable = gm_monuments
				limit = { gm_sh_orphan_of = { REGION = scope:gm_sh_region } }
				gm_tr_repair_places = yes
			}
			gm_monument_pass = yes
		}
		if = {
			limit = {
				has_global_variable_list = gm_shadows
				any_in_global_list = {
					variable = gm_shadows
					gm_sh_orphan_of = { REGION = scope:gm_sh_region }
				}
			}
			# The pass may have promoted one onto a building that was lost: a rebuild.
			if = {
				limit = {
					scope:gm_sh_region = { any_scope_state = { has_variable = gm_rebuilding } }
				}
				trigger_event = { id = monument_events.31 days = 1 }
			}
			else = {
				gm_sh_settle_region = yes
			}
		}
	}
}

# State scope (scope:gm_sh_region saved), from gm_state_adopt_orphans.
gm_sh_settle_region = {
	# A building that no record describes came with the merge: each orphan,
	# landmarks first (the issue's rule: a landmark's identity wins), takes the
	# one in its region that holds the most levels unaccounted for as its own
	# monument, while one is unclaimed.
	every_in_global_list = {
		variable = gm_shadows
		limit = {
			gm_sh_orphan_of = { REGION = scope:gm_sh_region }
			has_variable = gm_tr_landmark
		}
		add_to_temporary_list = gm_sh_raising
	}
	every_in_global_list = {
		variable = gm_shadows
		limit = {
			gm_sh_orphan_of = { REGION = scope:gm_sh_region }
			NOT = { has_variable = gm_tr_landmark }
		}
		add_to_temporary_list = gm_sh_raising
	}
	every_in_list = {
		list = gm_sh_raising
		gm_sh_try_raise = yes
	}
	# Every other orphan stands beside the recorded monument whose building
	# holds the most levels its records don't account for, rejoins the one it
	# was split from, or is lost.
	every_in_global_list = {
		variable = gm_shadows
		limit = { gm_sh_orphan_of = { REGION = scope:gm_sh_region } }
		add_to_temporary_list = gm_sh_left
	}
	every_in_list = {
		list = gm_sh_left
		gm_sh_settle_one = yes
	}
}

# Container scope, an orphaned shadow (scope:gm_sh_region its region): raised
# onto the unclaimed building no record describes that holds the most levels
# unaccounted for, if there is one (not a new monument of 1 level that happens
# to stand in another part of the region, when the merge brought a taller one).
gm_sh_try_raise = {
	# Still an orphan: the refresh a raise before it ran may have settled it.
	if = {
		limit = { gm_sh_orphan_of = { REGION = scope:gm_sh_region } }
		save_scope_as = gm_sh_pick
		clear_saved_scope = gm_sh_target
		scope:gm_sh_region = {
			ordered_scope_state = {
				limit = {
					gm_state_unrecorded = yes
					gm_state_unexplained_levels > 0
				}
				order_by = gm_state_unexplained_levels
				position = 0
				check_range_bounds = no
				save_scope_as = gm_sh_target
			}
		}
		if = {
			limit = { exists = scope:gm_sh_target }
			scope:gm_sh_target = { gm_state_raise_orphan = yes }
		}
	}
}

# Container scope, an orphaned shadow (scope:gm_sh_region its region). Still an
# orphan when its turn comes (an event run by an earlier one may have settled
# it), and not while a piece put back today rebuilds a building here: it is
# settled tomorrow (gm_sh_rejoin sends monument_events.31). A revolt's piece
# and its monument settle with each other first (§ "A revolt's split piece"):
# a piece whose monument stands in its region rejoins it, and a monument whose
# piece stands there is taken by it, whatever else the region's buildings hold.
gm_sh_settle_one = {
	if = {
		limit = {
			gm_sh_orphan_of = { REGION = scope:gm_sh_region }
			NOT = {
				scope:gm_sh_region = { any_scope_state = { has_variable = gm_rebuilding } }
			}
		}
		clear_saved_scope = gm_sh_target
		if = {
			limit = { gm_sh_piece_home = yes }
			var:gm_tr_piece_of = { save_scope_as = gm_sh_target }
			gm_sh_rejoin = yes
		}
		else_if = {
			limit = { gm_sh_whole_of_piece = yes }

		}
		else = {

				ordered_scope_state = {
					limit = { gm_state_takes_orphan_beside = yes }
					order_by = gm_state_unexplained_levels
					position = 0
					check_range_bounds = no
					save_scope_as = gm_sh_target
				}
			}
			if = {
				limit = { exists = scope:gm_sh_target }
				gm_sh_adopt_beside = yes
				debug_log = "TE_MONUMENTS: a monument whose state merged into another part of its region stands beside the monument there: [SCOPE.ScriptValue('gm_sh_log_levels')|0] levels recorded, [SCOPE.ScriptValue('gm_sh_log_found')|0] found unaccounted for"
			}
			else = {
				# No building holds levels nothing records. A monument here that
				# lacked levels when last seen was split by a revolt, and this is
				# its piece coming home (found 2026-10-17, gm_sh_rejoin): a piece
				# recorded as a monument of its own, before the split's piece
				# took its monument's identity (2026-10-17, owner's call).
				scope:gm_sh_region = {

						limit = { gm_state_was_split = yes }
						order_by = gm_state_split_short
						position = 0
						check_range_bounds = no
						save_scope_as = gm_sh_target
					}
				}
				if = {
					limit = { exists = scope:gm_sh_target }
					gm_sh_rejoin = yes
				}
				else = {
					gm_sh_lost = yes
				}
			}
		}
	}
}

# Container scope, an orphaned shadow no building takes, with scope:gm_sh_target
# a monument of its region that lacked levels when last seen
# (gm_state_was_split; scope:gm_sh_region the region). A revolt took part of
# that monument's state: the new part's building held a share of its levels,
# which the refresh recorded as a new monument, and this orphan is that piece,
# whose part has merged back. The monument's records, frozen before the split,
# count it already, so it does not stand beside: its levels go back on the
# building, no more than the building lacks now (none when they came back with
# the merge), and the piece goes. Found 2026-10-17 (Tuscany 1877, the Paris
# Commune 1887: the Arc de Triomphe stood at 11 levels on records of 15).
# Since the split's piece takes its monument's identity (§ "A revolt's split
# piece"), that piece's shadow (gm_tr_piece_of) comes here too, whatever else
# the region holds, and so does a stand-in for one that never had a shadow
# (gm_state_split_piece_home): its years come back with it.
# It is no orphan before the rebuild, which runs monument_events.1 and the
# refresh from inside this effect; the rest of the region waits a day.
gm_sh_rejoin = {
	save_scope_as = gm_sh_moving
	gm_sh_log_numbers = yes
	scope:gm_sh_target = {
		set_variable = { name = gm_sh_lacked value = gm_state_split_short }
		set_variable = { name = gm_sh_lacking value = gm_state_levels_short }
	}
	set_global_variable = { name = gm_sh_log_piece_now value = gm_mon_levels }
	set_global_variable = { name = gm_sh_log_lacked_now value = scope:gm_sh_target.var:gm_sh_lacked }
	set_global_variable = { name = gm_sh_log_lacking_now value = scope:gm_sh_target.var:gm_sh_lacking }
	# A piece that carried its monument's identity: its years come back with
	# it, and the monument forgets the split.
	if = {
		limit = { var:gm_tr_piece_of ?= scope:gm_sh_target }
		scope:gm_sh_target = {
			if = {
				limit = { scope:gm_sh_moving = { has_variable = gm_tr_age } }
				gm_state_add_age = { AGE = scope:gm_sh_moving.var:gm_tr_age }
			}
			gm_state_forget_split_notes = yes
		}
	}
	# What goes back: the piece's levels, no more than its monument lacks now.
	set_variable = {
		name = gm_tr_levels
		value = {
			value = gm_mon_levels
			max = scope:gm_sh_target.var:gm_sh_lacking
		}
	}
	set_global_variable = { name = gm_sh_log_put_back_now value = var:gm_tr_levels }
	# Out of the list and its region forgotten: no orphan for an event below.
	remove_list_global_variable = { name = gm_shadows target = scope:gm_sh_moving }
	remove_tag = gm_shadow
	remove_variable = gm_tr_origin_region
	debug_log = "TE_MONUMENTS: a monument whose state merged into another part of its region was a piece a revolt split from the monument there, and rejoins it: [SCOPE.ScriptValue('gm_sh_log_piece')|0] levels recorded; the monument lacked [SCOPE.ScriptValue('gm_sh_log_lacked')|0] when last seen and [SCOPE.ScriptValue('gm_sh_log_lacking')|0] now, [SCOPE.ScriptValue('gm_sh_log_put_back')|0] put back"
	gm_monument_split_self_check = yes
	scope:gm_sh_target = {
		# The monument's note keeps only what is still missing once this piece
		# is settled, so a second piece claims no more (the next shadow, which
		# waits out the rebuild, measures it again).
		set_variable = {
			name = gm_sh_lacking
			value = {
				value = var:gm_sh_lacking
				subtract = scope:gm_sh_moving.var:gm_tr_levels
				min = 0
			}
		}
		var:gm_shadow ?= {
			if = {
				limit = { scope:gm_sh_target.var:gm_sh_lacking >= 1 }
				set_variable = { name = gm_sh_short value = scope:gm_sh_target.var:gm_sh_lacking }
			}
			else = {
				gm_remove_state_var = { VAR = gm_sh_short }
				gm_remove_state_var = { VAR = gm_sh_short_passes }
			}
		}
		remove_variable = gm_sh_lacked
		remove_variable = gm_sh_lacking
	}
	if = {
		limit = { var:gm_tr_levels >= 1 }
		clear_saved_scope = gm_tr_moving
		save_scope_as = gm_tr_moving
		scope:gm_sh_target = {
			gm_state_tr_add_levels = yes
			trigger_event = { id = monument_events.31 days = 1 }
		}
	}
	destroy_container = yes
}

# ---- A revolt's split piece (owner's call 2026-10-17) -------------------------------
# A revolt that takes part of a monument's state gives the new part's building
# a share of the levels (Tuscany 1877, Dahomey 1874; the Paris Commune of 1886,
# a country a vanilla event creates). That piece is no new monument: at first
# sight (gm_state_split_pending, monument_events.29) it takes the monument's
# identity, as a returning trophy takes its records
# (gm_state_tr_take_records_from_moving, from the monument's shadow): its name,
# look, picture, landmark, dedication, raiser and contest, frozen at its own
# levels, with those levels' share of the monument's years, which leave the
# monument as a trophy's do (gm_state_tr_age). No ceremony asks. The monument's
# records still count the piece's levels (gm_state_levels_short), so the two
# settle when one owner holds both parts again and the engine merges them:
#   the monument's state survives: the piece rejoins it (gm_sh_rejoin, from the
#     piece's orphaned shadow, or gm_state_split_piece_home): what the building
#     lacks of the records goes back by one rebuild, with the piece's years;
#   the piece's state survives: the piece is the monument itself
#     (gm_sh_piece_takes_whole, from the monument's orphaned shadow, or
#     gm_state_piece_takes_whole): the monument's records and years, frozen at
#     the records, what the building lacks of them put back.
# Which side won does not matter. A revolution's winner continues the nation
# and annexes the loser, so the parts merge, and gm_repair_after_civil_war
# makes the monument the winner's. A secession's piece stays a piece of the
# monument abroad until the parts merge. Links: gm_piece_of on the piece's
# state, gm_split_piece on the monument's, with what the split took noted there
# (gm_split_levels, gm_split_age, gm_split_short), mirrored on the shadows
# (gm_tr_piece_of, gm_tr_split_piece); gm_clear_state breaks them
# (gm_state_forget_split), and the other part then stands on its own.

# State scope, the piece (monument_events.29, real execution only). The
# monument it was split from (gm_state_split_whole) must lack at least the
# piece's levels, as every split read so far did, the piece's exactly; else
# the building is no split's piece and goes to the ceremony as before
# (gm_split_rejected, a refresh tomorrow).
gm_state_take_split_identity = {
	save_scope_as = gm_split_piece_state
	save_scope_as = gm_tmp_split_piece
	owner = { save_scope_as = gm_tmp_split_owner }
	clear_saved_scope = gm_split_whole
	state_region = {
		ordered_scope_state = {
			limit = { gm_state_split_whole = yes }
			order_by = gm_state_levels_short
			position = 0
			check_range_bounds = no
			save_scope_as = gm_split_whole
		}
	}
	set_variable = { name = gm_split_p value = gm_state_primary_levels }
	if = {
		limit = { exists = scope:gm_split_whole }
		scope:gm_split_whole = {
			set_variable = { name = gm_split_short value = gm_state_levels_short }
		}
	}
	if = {
		limit = {
			exists = scope:gm_split_whole
			var:gm_split_p >= 1
			scope:gm_split_whole.var:gm_split_short >= var:gm_split_p
		}
		clear_saved_scope = gm_tr_moving
		scope:gm_split_whole = {
			var:gm_shadow = { save_scope_as = gm_tr_moving }
			# What the split took, for the piece's return.
			set_variable = { name = gm_split_piece value = scope:gm_split_piece_state }
			set_variable = { name = gm_split_levels value = scope:gm_split_piece_state.var:gm_split_p }
			set_variable = {
				name = gm_split_age
				value = {
					value = 0
					if = {
						limit = { has_variable = gm_age }
						value = var:gm_age
						multiply = scope:gm_split_piece_state.var:gm_split_p
						divide = var:gm_levels
						max = var:gm_age
					}
				}
			}
			if = {
				limit = { var:gm_split_age > 0 }
				change_variable = { name = gm_age subtract = var:gm_split_age }
				clamp_variable = { name = gm_age min = 0 }
			}
		}
		gm_clear_state = { KEEP_LANDMARK = no }
		gm_state_tr_take_records_from_moving = yes
		gm_tr_copy_to_state = { FROM = gm_tr_picture_code TO = gm_picture_code }
		# Its own levels and its share of the years, not the monument's.
		set_variable = { name = gm_levels value = var:gm_split_p }
		set_variable = { name = gm_ceremony_level value = var:gm_split_p }
		gm_remove_state_var = { VAR = gm_age }
		if = {
			limit = { scope:gm_split_whole.var:gm_split_age > 0 }
			set_variable = { name = gm_age value = scope:gm_split_whole.var:gm_split_age }
		}
		set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
		set_variable = { name = gm_piece_of value = scope:gm_split_whole }
		# Its dedication now: the building may stand on any method.
		gm_state_restore_dedication = yes
		set_global_variable = { name = gm_split_log_piece_now value = var:gm_split_p }
		set_global_variable = { name = gm_split_log_short_now value = scope:gm_split_whole.var:gm_split_short }
		debug_log = "TE_MONUMENTS: a revolt took part of a monument's state, and the new part's piece of it takes the monument's identity, with no ceremony: [SCOPE.ScriptValue('gm_split_log_piece')|0] levels on the piece, the monument lacking [SCOPE.ScriptValue('gm_split_log_short')|0]"
		# The first piece of a game is read back (gm_monument_split_identity_self_check).
		if = {
			limit = { NOT = { has_global_variable = gm_split_identity_checked } }
			set_variable = { name = gm_split_check_p value = var:gm_split_p }
			set_variable = { name = gm_split_check_s value = scope:gm_split_whole.var:gm_split_short }
			set_variable = { name = gm_split_check_n value = 0 }
		}
	}
	else = {
		if = {
			limit = { exists = scope:gm_split_whole }
			set_global_variable = { name = gm_split_log_piece_now value = var:gm_split_p }
			set_global_variable = { name = gm_split_log_short_now value = scope:gm_split_whole.var:gm_split_short }
			gm_monument_split_gate_self_check = yes
			scope:gm_split_whole = { remove_variable = gm_split_short }
		}
		set_variable = gm_split_rejected
		owner = { trigger_event = { id = monument_events.20 days = 1 } }
	}
	remove_variable = gm_split_p
}

# State scope, from gm_clear_state: a primary forgotten breaks its split's
# links both ways, and the other part stands on its own. A piece that is no
# longer one (pulled down, carried off, rededicated) takes its levels out of
# its monument's records for good, as a trophy's piece does

gm_state_forget_split = {

			if = {

						has_variable = gm_levels

					clamp_variable = { name = gm_levels min = 1 }
				}

			}
		}
		remove_variable = gm_piece_of
	}
	if = {
		limit = { has_variable = gm_split_piece }
		save_scope_as = gm_split_forgetting
		var:gm_split_piece ?= {
			if = {
				limit = { var:gm_piece_of ?= scope:gm_split_forgetting }
				remove_variable = gm_piece_of
			}
		}
		gm_state_forget_split_notes = yes
	}
}

# Container scope, the shadow of a monument whose revolt's piece stands in its
# region (scope:gm_sh_target the piece's state, scope:gm_sh_region the region):
# the monument's state merged into the piece's. The piece is the monument
# itself: frozen at the monument's records, with its years added to the
# piece's share, and what the building lacks of the records goes back by one
# rebuild (gm_state_tr_add_levels, as gm_sh_rejoin). It is no shadow before
# the rebuild, which runs monument_events.1 and the refresh from inside this
# effect; the rest of the region waits a day.
gm_sh_piece_takes_whole = {
	save_scope_as = gm_sh_moving
	set_global_variable = { name = gm_split_log_records_now value = var:gm_tr_levels }
	scope:gm_sh_target = {
		set_global_variable = { name = gm_split_log_piece_now value = var:gm_levels }
		remove_variable = gm_piece_of
		set_variable = { name = gm_levels value = scope:gm_sh_moving.var:gm_tr_levels }
		if = {
			limit = { scope:gm_sh_moving = { has_variable = gm_tr_age } }
			gm_state_add_age = { AGE = scope:gm_sh_moving.var:gm_tr_age }
		}
		set_variable = { name = gm_sh_lacking value = gm_state_levels_short }
	}
	# What goes back: what the building lacks of the records.
	set_variable = { name = gm_tr_levels value = scope:gm_sh_target.var:gm_sh_lacking }
	set_global_variable = { name = gm_sh_log_put_back_now value = var:gm_tr_levels }
	scope:gm_sh_target = { remove_variable = gm_sh_lacking }
	# Out of the list and its region forgotten: no shadow for an event below.
	remove_list_global_variable = { name = gm_shadows target = scope:gm_sh_moving }
	remove_tag = gm_shadow
	gm_remove_state_var = { VAR = gm_tr_origin_region }
	debug_log = "TE_MONUMENTS: a revolt's piece of a monument stands as the monument itself, the monument's state merged into the piece's: [SCOPE.ScriptValue('gm_split_log_piece')|0] levels on the piece, [SCOPE.ScriptValue('gm_split_log_records')|0] on the monument's records, [SCOPE.ScriptValue('gm_sh_log_put_back')|0] put back"
	if = {
		limit = { var:gm_tr_levels >= 1 }
		clear_saved_scope = gm_tr_moving
		save_scope_as = gm_tr_moving
		scope:gm_sh_target = {
			gm_state_tr_add_levels = yes
			trigger_event = { id = monument_events.31 days = 1 }
		}
	}
	destroy_container = yes
}

# ---- One-off repair: a monument short of a piece long gone (owner's call 2026-10-17) ----
# OLD-SAVE REPAIR (#936): remove after 2026-12-17 or at vanilla 1.15, whichever comes first.
#
# Before 2026-10-17 a revolt's split piece was recorded as a monument of its
# own, and when its part merged back before gm_sh_rejoin existed, its orphan
# was lost: the monument's records, frozen before the split, still count the
# piece's levels, and its building stands that much lower. In the 1898 crash
# save (crash_2026-10-17_china_opportunity.v3) one monument is: the Arc de
# Triomphe, 11 levels on records of 15, the Paris Commune's piece of 4 lost in
# 1887. Each such monument gets its levels back by one rebuild
# (gm_state_tr_add_levels, as gm_sh_rejoin), once a save. From the monthly
# pass (gm_monument_monthly), never the refresh, which tooltips walk. A
# shortfall with another explanation is left alone
# (gm_state_short_unexplained, gm_monument_triggers.txt); one with something
# under way (gm_state_short_ready) waits a month, for up to a year
# (gm_short_repair_passes). Loop safety: the rebuild runs monument_events.1
# and the refresh inside the effect, and neither calls this; each state is
# marked repaired (gm_short_repaired) before its rebuild, which is its only
# one. The verdict comes a pass later (gm_monument_short_repair_self_check).
gm_short_repair_monthly = {
	gm_monument_short_repair_self_check = yes
	if = {
		limit = { NOT = { has_global_variable = gm_short_repair_done } }
		if = {
			limit = { has_global_variable = gm_short_repair_passes }
			change_global_variable = { name = gm_short_repair_passes add = 1 }
		}
		else = {
			set_global_variable = { name = gm_short_repair_passes value = 1 }
			set_global_variable = { name = gm_short_repair_n value = 0 }
			set_global_variable = { name = gm_short_repair_levels value = 0 }
		}
		every_state = {
			limit = {
				gm_state_short_unexplained = yes
				gm_state_short_ready = yes
			}
			add_to_temporary_list = gm_short_repairing
		}
		every_in_list = {
			list = gm_short_repairing
			gm_state_short_repair = yes
		}
		# Done once nothing waits, or after a year of waiting.
		if = {
			limit = {
				OR = {
					global_var:gm_short_repair_passes >= 12
					NOT = { any_state = { gm_state_short_unexplained = yes } }
				}
			}
			set_global_variable = gm_short_repair_done
			set_global_variable = { name = gm_short_repair_left value = 0 }
			every_state = {
				limit = { gm_state_short_unexplained = yes }
				change_global_variable = { name = gm_short_repair_left add = 1 }
			}
			debug_log = "TE_MONUMENTS: the one-off repair of monuments short of a piece long gone is done: [SCOPE.ScriptValue('gm_short_log_n')|0] repaired, [SCOPE.ScriptValue('gm_short_log_levels')|0] levels put back, [SCOPE.ScriptValue('gm_short_log_left')|0] still short and waiting"
		}
	}
}

# State scope, from gm_short_repair_monthly: a monument short of a piece long
# gone. Marked repaired first; then the levels its building lacks go back on
# it by one rebuild, through a stand-in that carries them as a piece would
# (gm_state_tr_add_levels reads scope:gm_tr_moving), destroyed after under a
# name of its own: the rebuild's monument_events.1 and refresh run inside this
# effect, and what they run may save another container as gm_tr_moving.
gm_state_short_repair = {
	set_variable = gm_short_repaired
	save_scope_as = gm_short_state
	set_variable = { name = gm_short_put_back value = gm_state_levels_short }
	set_global_variable = { name = gm_short_log_put_back_now value = var:gm_short_put_back }
	set_global_variable = { name = gm_short_log_stood_now value = gm_state_level }
	change_global_variable = { name = gm_short_repair_n add = 1 }
	change_global_variable = { name = gm_short_repair_levels add = var:gm_short_put_back }
	if = {
		limit = { NOT = { has_global_variable = gm_short_repair_checked } }
		set_global_variable = gm_short_repair_check_due
	}
	debug_log = "TE_MONUMENTS: the one-off repair puts back the levels of a piece long gone on the monument in [THIS.GetState.GetNameNoFormatting]: [SCOPE.ScriptValue('gm_short_log_put_back')|0] levels on the [SCOPE.ScriptValue('gm_short_log_stood')|0] standing"
	clear_saved_scope = gm_short_standin
	create_container = {
		tags = { gm_repair }
		save_scope_as = gm_short_standin
		on_created = {
			save_scope_as = gm_short_standin
		}
	}
	if = {
		limit = { exists = scope:gm_short_standin }
		scope:gm_short_standin = {
			set_variable = { name = gm_tr_levels value = scope:gm_short_state.var:gm_short_put_back }
			save_scope_as = gm_tr_moving
		}
		gm_state_tr_add_levels = yes
		scope:gm_short_standin = { destroy_container = yes }
	}
	remove_variable = gm_short_put_back
}
