# ============================================================================
# The Creditors' Consortium power bloc identity: effects
# ============================================================================
# Design: docs/superpowers/specs/2026-11-16-power-bloc-identities-design.md § 5.
# System: docs/systems/mod_systems/power_blocs.md § "Creditors' Consortium".
# Triggers and values: te_creditors_consortium_triggers.txt / _values.txt.
#
# THE FLOW OF A RESCUE
#   on_country_default -> te_monetary_on_country_default (te_monetary_on_actions.txt)
#     a treaty guarantor first (te_lolr.1); else, for a member whose consortium
#     guarantee stands (te_cred_guarantee_is_effective), te_cred_call_consortium:
#   te_cc.1  the leader: rescue on one term, and as many more as its
#            Conditionality allows, or refuse
#   te_cc.2  the member: accept the terms (paid, te_cred_apply_rescue) or refuse
#   te_cc.3  the leader hears the answer
#   te_cc.4  the member hears a refusal
#   te_cc.9  three days on, hidden: did the treaties enter into force? (§ 11
#            facts 1 and 3; TE_SELF_CHECK cc_receivership_treaty / cc_monopoly_treaty)
#
# Scopes saved by te_cred_call_consortium and read by every event of the chain:
#   scope:te_cc_debtor    the member in default
#   scope:te_cc_creditor  its consortium's leader
# ============================================================================


# --- The call -------------------------------------------------------------------

# Scope: the member in default (ROOT of on_country_default). Called by
# te_monetary_on_country_default after it has counted the call on the member's
# record (te_mon_lolr_calls, the cooldown), exactly as for a treaty guarantee.
te_cred_call_consortium = {
	save_scope_as = te_cc_debtor
	power_bloc ?= {
		power_bloc_leader = {
			save_scope_as = te_cc_creditor
			trigger_event = { id = te_cc.1 popup = yes }
		}
	}
}


# --- te_cc.1: the leader's offer --------------------------------------------------

# Scope: the leader. Writes the chosen terms on the member and asks it.
# $ADJ$, $CONC$, $PRIV$: yes or no, structural adjustment, concessions,
# privatization. Adjustment and privatization are written on the member as 1
# or 0; concessions as the number of the rescue's terms spent on them, what
# the other two leave of te_cred_terms_to_pick (te_cred_conc_slots_0 / _1 /
# _2), with the monopolies those buy beside it (te_cred_offer_monopolies), so
# the count the leader offered is the count demanded. They last 120 days
# there: an answer that never comes (the event lapses) leaves nothing behind.
te_cred_offer_rescue = {
	te_cred_offer_rescue_tooltip = { ADJ = $ADJ$ CONC = $CONC$ PRIV = $PRIV$ }
	hidden_effect = {
		scope:te_cc_debtor = {
			set_variable = { name = te_cred_offer_adjustment value = 0 days = 120 }
			set_variable = { name = te_cred_offer_concessions value = 0 days = 120 }
			set_variable = { name = te_cred_offer_privatization value = 0 days = 120 }
			if = {
				limit = { always = $ADJ$ }
				set_variable = { name = te_cred_offer_adjustment value = 1 days = 120 }
			}
			set_variable = { name = te_cred_offer_monopolies value = 0 days = 120 }
			if = {
				limit = { always = $CONC$ }
				if = {
					limit = {
						always = $ADJ$
						always = $PRIV$
					}
					set_variable = { name = te_cred_offer_concessions value = scope:te_cc_creditor.te_cred_conc_slots_2 days = 120 }
					set_variable = { name = te_cred_offer_monopolies value = scope:te_cc_creditor.te_cred_offer_monopolies_2 days = 120 }
				}
				else_if = {
					limit = {
						OR = {
							always = $ADJ$
							always = $PRIV$
						}
					}
					set_variable = { name = te_cred_offer_concessions value = scope:te_cc_creditor.te_cred_conc_slots_1 days = 120 }
					set_variable = { name = te_cred_offer_monopolies value = scope:te_cc_creditor.te_cred_offer_monopolies_1 days = 120 }
				}
				else = {
					set_variable = { name = te_cred_offer_concessions value = scope:te_cc_creditor.te_cred_conc_slots_0 days = 120 }
					set_variable = { name = te_cred_offer_monopolies value = scope:te_cc_creditor.te_cred_offer_monopolies_0 days = 120 }
				}
			}
			if = {
				limit = { always = $PRIV$ }
				set_variable = { name = te_cred_offer_privatization value = 1 days = 120 }
			}
			set_variable = { name = te_cred_offer_cost value = te_mon_lolr_honour_cost days = 120 }
			trigger_event = { id = te_cc.2 popup = yes }
		}
	}
}

# The option's tooltip: what the member will be asked to take. Scope: the
# leader. Shown in te_cc.1, before anything is written.
te_cred_offer_rescue_tooltip = {
	custom_tooltip = {
		text = te_cred_offer_payment_tt
	}
	if = {
		limit = { te_cred_receivership_possible = yes }
		custom_tooltip = { text = te_cred_offer_receivership_tt }
	}
	else = {
		custom_tooltip = { text = te_cred_offer_no_receivership_tt }
	}
	if = {
		limit = { always = $ADJ$ }
		custom_tooltip = { text = te_cred_offer_adjustment_tt }
	}
	# Concessions take the terms the other two leave, and each buys the tier's
	# monopolies: one line for each count of other terms, since a text reads a
	# script value and takes no argument. Without Conditionality, or where no
	# monopoly can be demanded, the concessions are the investment rights alone.
	if = {
		limit = { always = $CONC$ }
		if = {
			limit = {
				always = $ADJ$
				always = $PRIV$
			}
			if = {
				limit = { te_cred_offer_monopolies_2 > 0 }
				custom_tooltip = { text = te_cred_offer_concessions_2_tt }
			}
			else = {
				custom_tooltip = { text = te_cred_offer_concessions_plain_tt }
			}
		}
		else_if = {
			limit = {
				OR = {
					always = $ADJ$
					always = $PRIV$
				}
			}
			if = {
				limit = { te_cred_offer_monopolies_1 > 0 }
				custom_tooltip = { text = te_cred_offer_concessions_1_tt }
			}
			else = {
				custom_tooltip = { text = te_cred_offer_concessions_plain_tt }
			}
		}
		else = {
			if = {
				limit = { te_cred_offer_monopolies_0 > 0 }
				custom_tooltip = { text = te_cred_offer_concessions_0_tt }
			}
			else = {
				custom_tooltip = { text = te_cred_offer_concessions_plain_tt }
			}
		}
	}
	if = {
		limit = { always = $PRIV$ }
		custom_tooltip = { text = te_cred_offer_privatization_tt }
	}
}

# Scope: the leader. Refuse: te_lolr.1's Renege without a treaty to withdraw
# from, so no infamy. The guarantee clock is the one te_lolr.1 starts
# (te_mon_lolr_suspended_months, five years), so every member loses the
# standing guarantee (te_cred_backer_has_reneged), and every treaty guarantee
# the leader gives lapses with it. The bloc loses a fifth of its cohesion at
# once.
te_cred_refuse_rescue = {
	custom_tooltip = te_cred_refuse_suspends_tt
	hidden_effect = {
		set_variable = { name = te_mon_lolr_suspended_months value = te_mon_lolr_suspension_months }
	}
	if = {
		limit = { has_modifier = te_mon_lolr_reneged }
		hidden_effect = { remove_modifier = te_mon_lolr_reneged }
	}
	add_modifier = {
		name = te_mon_lolr_reneged
		days = 1825
	}
	power_bloc ?= {
		custom_tooltip = {
			text = te_cred_refuse_cohesion_tt
			add_cohesion_percent = -0.2
		}
	}
	scope:te_cc_debtor = {
		change_relations = {
			country = root
			value = -40
		}
		hidden_effect = { trigger_event = { id = te_cc.4 } }
	}
	hidden_effect = {
		debug_log = "TE_CC refused debtor=[SCOPE.sCountry('te_cc_debtor').GetNameNoFormatting] creditor=[ROOT.GetCountry.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}


# --- te_cc.2: the member's answer ----------------------------------------------------

# Scope: the member, accepting. Pays the rescue, then makes each term.
te_cred_apply_rescue = {
	# The payment: the guarantee's honour cost, fixed when the leader offered
	# (te_cred_offer_cost, set by te_cc.1's option, so it is there while this
	# option's tooltip is read), so the sum the leader agreed to is the sum
	# paid. te_cred_paid keeps it for te_cc.3's text.
	scope:te_cc_creditor = {
		add_treasury = {
			value = scope:te_cc_debtor.var:te_cred_offer_cost
			multiply = -1
		}
	}
	add_treasury = var:te_cred_offer_cost
	hidden_effect = {
		set_variable = { name = te_cred_paid value = var:te_cred_offer_cost days = 365 }
		# Holds the adjustment term until the discovery has read the new
		# receivership (internal.2 a day on, te_cc.9 three days on).
		set_variable = { name = te_cred_rescue_pending value = 1 days = 10 }
	}
	change_relations = {
		country = scope:te_cc_creditor
		value = 25
	}

	# The receivership, where the leader has its technology.
	if = {
		limit = { scope:te_cc_creditor = { te_cred_receivership_possible = yes } }
		custom_tooltip = te_cred_accept_receivership_tt
		hidden_effect = { te_cred_make_receivership = yes }
	}

	# Structural adjustment: the leader may impose its economic laws while the
	# receivership stands (te_cred_can_impose_adjustment). Removed by the monthly
	# pulse once the receivership has gone (te_cred_member_monthly).
	if = {
		limit = { var:te_cred_offer_adjustment = 1 }
		custom_tooltip = te_cred_accept_adjustment_tt
		hidden_effect = {
			set_variable = { name = te_cred_adjustment_by value = scope:te_cc_creditor }
		}
	}

	# Concessions: Foreign Investment Rights for the leader, then the
	# monopolies the leader offered for (te_cred_offer_monopolies: its tier's
	# figure for each term it spent on concessions, none without the group).
	if = {
		limit = { var:te_cred_offer_concessions >= 1 }
		if = {
			limit = { var:te_cred_offer_monopolies >= 1 }
			custom_tooltip = te_cred_accept_concessions_tt
		}
		else = {
			custom_tooltip = te_cred_accept_concessions_plain_tt
		}
		hidden_effect = {
			te_cred_grant_investment_rights = yes
			te_cred_demand_monopolies = yes
		}
	}

	# Privatization: article 105, with the member as its source.
	if = {
		limit = { var:te_cred_offer_privatization = 1 }
		custom_tooltip = te_cred_accept_privatization_tt
		hidden_effect = { te_cred_make_privatization = yes }
	}

	hidden_effect = {
		remove_variable = te_cred_offer_adjustment
		remove_variable = te_cred_offer_concessions
		remove_variable = te_cred_offer_monopolies
		remove_variable = te_cred_offer_privatization
		remove_variable = te_cred_offer_cost
		trigger_event = { id = te_cc.9 days = 3 }
		trigger_event = { id = te_monetary_internal.2 days = 1 }
		# The answer is the DEBTOR's (te_cc.3 reads it through scope:te_cc_debtor),
		# so two members answering the same leader at once can't overwrite each other.
		set_variable = { name = te_cc_answer value = 1 days = 120 }
		scope:te_cc_creditor = {
			trigger_event = { id = te_monetary_internal.2 days = 1 }
			trigger_event = { id = te_cc.3 }
		}
		debug_log = "TE_CC rescued debtor=[ROOT.GetCountry.GetNameNoFormatting] creditor=[SCOPE.sCountry('te_cc_creditor').GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}

# Scope: the member, refusing. It stays in default, loses the standing
# guarantee for ten years whichever bloc it is in (the cooling-off rule: the
# variable is on the country, so leaving and rejoining restores nothing), and
# may leave the bloc whatever the leader's leverage (the identity's can_leave).
te_cred_refuse_terms = {
	custom_tooltip = te_cred_terms_refused_tt
	hidden_effect = {
		set_variable = { name = te_cred_terms_refused days = 3650 }
		remove_variable = te_cred_offer_adjustment
		remove_variable = te_cred_offer_concessions
		remove_variable = te_cred_offer_monopolies
		remove_variable = te_cred_offer_privatization
		remove_variable = te_cred_offer_cost
		trigger_event = { id = te_monetary_internal.2 days = 1 }
	}
	change_relations = {
		country = scope:te_cc_creditor
		value = -20
	}
	hidden_effect = {
		set_variable = { name = te_cc_answer value = 2 days = 120 }
		scope:te_cc_creditor = {
			trigger_event = { id = te_monetary_internal.2 days = 1 }
			trigger_event = { id = te_cc.3 }
		}
		debug_log = "TE_CC terms_refused debtor=[ROOT.GetCountry.GetNameNoFormatting] creditor=[SCOPE.sCountry('te_cc_creditor').GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}


# --- The treaties ----------------------------------------------------------------------
# Each term is its own treaty, created in force (is_draft = no), each behind a
# can_create_treaty that runs the article's own gates, so a term the engine
# refuses is skipped rather than half made. Five-year binding periods, the
# engine's default (TREATY_DEFAULT_BINDING_DURATION_YEARS).

# Scope: the member. The receivership: article 114 with the member as its
# source and the leader as its target, as the article reads them. Where
# can_create_treaty refuses it (the article's can_ratify asks that the leader
# outrank the member or out-earn it, te_mon_can_receive_for), the receivership
# is set directly (te_cred_set_direct_receivership). Where the treaty is made,
# te_cc.9 checks three days on that it entered into force, and sets it directly
# if not (§ 11 fact 1).
te_cred_make_receivership = {
	if = {
		limit = {
			can_create_treaty = {
				first_country = root
				second_country = scope:te_cc_creditor
				is_draft = no
				binding_period = { years = 5 }
				articles_to_create = {
					{
						article = debt_receivership
						source_country = root
						target_country = scope:te_cc_creditor
					}
				}
			}
		}
		# Keep identical to the can_create_treaty above.
		create_treaty = {
			first_country = root
			second_country = scope:te_cc_creditor
			is_draft = no
			binding_period = { years = 5 }
			articles_to_create = {
				{
					article = debt_receivership
					source_country = root
					target_country = scope:te_cc_creditor
				}
			}
		}
		set_variable = { name = te_cred_rcv_check value = scope:te_cc_creditor days = 30 }
	}
	else = {
		te_cred_set_direct_receivership = yes
	}
}

# Scope: the member. The treaty-less receivership (§ 11 fact 1's fallback, the
# treaty-transfer helper's shape): the role variables the monetary layer's
# discovery derives from article 114, set from the consortium's own record.
# The discovery honors it (te_cred_discover_direct_receivership) under the
# article's own conditions, the exit ramp included.
te_cred_set_direct_receivership = {
	set_variable = { name = te_cred_direct_receivership value = 1 }
	set_variable = { name = te_cred_receiver value = scope:te_cc_creditor }
	# The article's binding period: te_decision_cred_end_receivership opens after it.
	set_variable = { name = te_cred_direct_binding value = 1 days = 1825 }
	set_variable = { name = te_mon_receiver value = scope:te_cc_creditor }
	set_variable = { name = te_mon_receivership_clear_months value = 0 }
	trigger_event = { id = te_monetary_internal.2 days = 1 }
	scope:te_cc_creditor = { trigger_event = { id = te_monetary_internal.2 days = 1 } }
}

# Scope: the member. Foreign Investment Rights, member -> leader, unless the
# leader holds them already.
te_cred_grant_investment_rights = {
	if = {
		limit = {
			NOT = {
				any_scope_treaty = {
					any_scope_article = {
						has_type = foreign_investment_rights
						source_country = root
						target_country = scope:te_cc_creditor
					}
				}
			}
			can_create_treaty = {
				first_country = root
				second_country = scope:te_cc_creditor
				is_draft = no
				binding_period = { years = 5 }
				articles_to_create = {
					{
						article = foreign_investment_rights
						source_country = root
						target_country = scope:te_cc_creditor
					}
				}
			}
		}
		# Keep identical to the can_create_treaty above.
		create_treaty = {
			first_country = root
			second_country = scope:te_cc_creditor
			is_draft = no
			binding_period = { years = 5 }
			articles_to_create = {
				{
					article = foreign_investment_rights
					source_country = root
					target_country = scope:te_cc_creditor
				}
			}
		}
	}
}

# Scope: the member. Monopolies for the leader's companies, as many as the
# leader offered for (te_cred_offer_monopolies: its Conditionality tier's
# figure for each term it spent on concessions; none without the group, where
# this makes nothing). Each monopoly is its own acquire_monopoly_for_company
# treaty, which vanilla's can_ratify allows with Foreign Investment Rights in
# force, and refuses for a building type the member has already granted.
#
# WHICH ONES (owner, 2026-11-17): the member's largest industries first. A
# pair is one of the leader's companies and one building type of its type
# (the generated ladder, te_cred_monopoly_ladder_generated); its score is the
# member's levels of that building type. Each round walks every pair twice:
# once to find the best one left (te_cred_mono_best, te_cred_mono_best_seq,
# the pair's number in the walk), once more to make it. "Left" is what comes
# after the last pair taken, by score and then by number
# (te_cred_mono_last_score, te_cred_mono_last_seq), so no pair is tried twice
# and the loop ends when a round finds none, whatever the engine does with a
# treaty made a moment ago. Equal scores go to the company walked first: the
# leader's most prosperous (te_cred_walk_companies).
te_cred_demand_monopolies = {
	set_variable = { name = te_cred_mono_left value = var:te_cred_offer_monopolies }
	set_variable = { name = te_cred_mono_made value = 0 }
	# For te_cc.9's self-check: an article from before this rescue can't pass it.
	if = {
		limit = {
			any_scope_treaty = {
				any_scope_article = {
					has_type = acquire_monopoly_for_company
					source_country = root
				}
			}
		}
		set_variable = { name = te_cred_mono_prior value = 1 days = 30 }
	}
	# Nothing to pick where the leader offered for none.
	if = {
		limit = {
			has_variable = te_cred_mono_left
			var:te_cred_mono_left > 0
		}
		set_variable = { name = te_cred_mono_last_score value = 1000000 }
		set_variable = { name = te_cred_mono_last_seq value = 0 }
		set_variable = { name = te_cred_mono_best value = 0 }
		set_variable = { name = te_cred_mono_best_seq value = 0 }
		te_cred_check_company_order = yes
		set_variable = te_cred_mono_more
		while = {
			limit = {
				has_variable = te_cred_mono_more
				has_variable = te_cred_mono_left
				var:te_cred_mono_left > 0
			}
			remove_variable = te_cred_mono_more
			# The best pair left.
			set_variable = { name = te_cred_mono_best value = 0 }
			set_variable = { name = te_cred_mono_best_seq value = 0 }
			set_variable = { name = te_cred_mono_seq value = 0 }
			te_cred_walk_companies = yes
			if = {
				limit = {
					has_variable = te_cred_mono_best
					var:te_cred_mono_best > 0
				}
				# ...and the same walk again, making the pair whose number it is.
				set_variable = { name = te_cred_mono_last_score value = var:te_cred_mono_best }
				set_variable = { name = te_cred_mono_last_seq value = var:te_cred_mono_best_seq }
				set_variable = { name = te_cred_mono_seq value = 0 }
				set_variable = te_cred_mono_take
				te_cred_walk_companies = yes
				remove_variable = te_cred_mono_take
				set_variable = te_cred_mono_more
			}
		}
		remove_variable = te_cred_mono_best
		remove_variable = te_cred_mono_best_seq
		remove_variable = te_cred_mono_last_score
		remove_variable = te_cred_mono_last_seq
	}
	remove_variable = te_cred_mono_left
	if = {
		limit = { has_variable = te_cred_mono_more }
		remove_variable = te_cred_mono_more
	}
	if = {
		limit = { has_variable = te_cred_mono_seq }
		remove_variable = te_cred_mono_seq
	}
	if = {
		limit = { has_variable = te_cred_mono_score }
		remove_variable = te_cred_mono_score
	}
	if = {
		limit = { has_variable = te_cred_mono_unordered }
		remove_variable = te_cred_mono_unordered
	}
}

# Scope: the member. One walk of the leader's companies, each trying every
# building type of its type (te_cred_try_monopoly). The most prosperous
# company first (ordered_company by prosperity), so two companies of one
# industry settle a tie the same way every time. Prosperity and not size: it
# is the one figure of a company's that script reads as a value (its levels
# are comparison triggers only; scripting_best_practices.md § "Company AI
# Fields"). Vanilla never uses ordered_company, so te_cred_check_company_order
# checks it before the first walk and sets te_cred_mono_unordered where it
# fails: the walk is then every_company's, in the engine's order.
te_cred_walk_companies = {
	if = {
		limit = { has_variable = te_cred_mono_unordered }
		scope:te_cc_creditor = {
			every_company = {
				save_scope_as = te_cred_mono_company
				scope:te_cc_debtor = { te_cred_monopoly_ladder_generated = yes }
			}
		}
	}
	else = {
		scope:te_cc_creditor = {
			ordered_company = {
				order_by = prosperity
				max = 1000
				check_range_bounds = no
				save_scope_as = te_cred_mono_company
				scope:te_cc_debtor = { te_cred_monopoly_ladder_generated = yes }
			}
		}
	}
}

# Scope: the member. Does ordered_company walk every company of the leader's,
# the most prosperous first? Counts them both ways and keeps the prosperity of
# the first and the last it walks. Fewer walked than every_company finds: the ordered
# walk is not used in this rescue (te_cred_mono_unordered). The self-check is
# logged once a game, and only where the first and the last differ, so one
# company, or a value that reads 0 for all of them, can't pass it.
te_cred_check_company_order = {
	set_variable = { name = te_cred_mono_companies value = 0 }
	set_variable = { name = te_cred_mono_walked value = 0 }
	set_variable = { name = te_cred_mono_pr_first value = 0 }
	set_variable = { name = te_cred_mono_pr_last value = 0 }
	scope:te_cc_creditor = {
		every_company = {
			scope:te_cc_debtor = { change_variable = { name = te_cred_mono_companies add = 1 } }
		}
		ordered_company = {
			order_by = prosperity
			max = 1000
			check_range_bounds = no
			save_scope_as = te_cred_mono_company
			scope:te_cc_debtor = {
				if = {
					limit = {
						has_variable = te_cred_mono_walked
						var:te_cred_mono_walked < 1
					}

# Scope: the member, scope:te_cred_mono_company a company of scope:te_cc_creditor.
# One rung of the ladder: $BT$ a building type of that company's type, and one
# pair of te_cred_demand_monopolies's walk. It numbers itself
# (te_cred_mono_seq), then either scores (the member's levels of $BT$, kept as
# the best where it beats the best so far and comes after the last pair taken)
# or, on the second walk (te_cred_mono_take), makes the treaty if its number
# is the best one's.
te_cred_try_monopoly = {
	if = {
		limit = { has_variable = te_cred_mono_seq }
		change_variable = { name = te_cred_mono_seq add = 1 }
		if = {
			limit = {
				has_variable = te_cred_mono_take
				has_variable = te_cred_mono_best_seq
				var:te_cred_mono_seq >= var:te_cred_mono_best_seq
				var:te_cred_mono_seq <= var:te_cred_mono_best_seq
				# No company_has_building_type_monopoly here: it names no country, so
				# it would skip a company that holds the monopoly anywhere. A second
				# monopoly on the same building type here is refused by the article
				# itself (can_ratify's duplicate check), which can_create_treaty runs.
				can_create_treaty = {
					first_country = root
					second_country = scope:te_cc_creditor
					is_draft = no
					binding_period = { years = 5 }
					articles_to_create = {
						{
							article = acquire_monopoly_for_company
							source_country = root
							target_country = scope:te_cc_creditor
							inputs = {
								{ company = scope:te_cred_mono_company } # REVIEWED 2026-11-16: a create_treaty input (effects.log § create_treaty), not an effect
								{ building_type = bt:$BT$ }
							}
						}
					}
				}
			}
			# Keep identical to the can_create_treaty above.
			create_treaty = {
				first_country = root
				second_country = scope:te_cc_creditor
				is_draft = no
				binding_period = { years = 5 }
				articles_to_create = {
					{
						article = acquire_monopoly_for_company
						source_country = root

						inputs = {
							{ company = scope:te_cred_mono_company } # REVIEWED 2026-11-16: a create_treaty input (effects.log § create_treaty), not an effect
							{ building_type = bt:$BT$ }
						}
					}
				}
			}
			change_variable = { name = te_cred_mono_left add = -1 }
			change_variable = { name = te_cred_mono_made add = 1 }
		}
		else_if = {
			limit = { NOT = { has_variable = te_cred_mono_take } }
			set_variable = {
				name = te_cred_mono_score
				value = {
					value = 0
					every_scope_building = {
						limit = { is_building_type = $BT$ }
						add = level
					}
				}
			}
			if = {
				limit = {
					has_variable = te_cred_mono_score
					has_variable = te_cred_mono_best
					has_variable = te_cred_mono_last_score
					has_variable = te_cred_mono_last_seq
					var:te_cred_mono_score > var:te_cred_mono_best
					OR = {
						var:te_cred_mono_score < var:te_cred_mono_last_score
						AND = {
							var:te_cred_mono_score <= var:te_cred_mono_last_score
							var:te_cred_mono_seq > var:te_cred_mono_last_seq
						}
					}
					# Keep identical to the can_create_treaty above.
					can_create_treaty = {
						first_country = root
						second_country = scope:te_cc_creditor
						is_draft = no
						binding_period = { years = 5 }
						articles_to_create = {
							{
								article = acquire_monopoly_for_company
								source_country = root
								target_country = scope:te_cc_creditor
								inputs = {
									{ company = scope:te_cred_mono_company } # REVIEWED 2026-11-16: a create_treaty input (effects.log § create_treaty), not an effect
									{ building_type = bt:$BT$ }
								}
							}
						}
					}
				}
				set_variable = { name = te_cred_mono_best value = var:te_cred_mono_score }
				set_variable = { name = te_cred_mono_best_seq value = var:te_cred_mono_seq }
			}
		}
	}
}

# --- The monetary layer's hooks ----------------------------------------------------------
# Called from te_monetary_discover_arrangements (te_monetary_arrangement_effects.txt).
# Scope: country, ROOT = this, the discovery's own contract.

# After the article receivership is read: the treaty-less one, under the same
# conditions as the article's requirement_to_maintain (te_cred_has_direct_receivership)
# and the same exit ramp (twelve months and more, by Conditionality tier, out of
# default and under a quarter of the ceiling). A treaty receivership found by
# the walk wins, and ends the direct one: two commissions over one treasury
# are not two.
#
# Off the Full banking rule the receivership PAUSES: te_cred_has_direct_receivership
# fails, nothing is paid or relieved, and the record waits; back on Full, the
# next discovery finds it again. The article receivership is not like that: its
# requirement_to_maintain removes it from its treaty.
te_cred_discover_direct_receivership = {
	if = {
		limit = { has_variable = te_cred_direct_receivership }
		if = {
			limit = { var:te_mon_receivership = 1 }
			te_cred_end_direct_receivership = yes
		}
		else_if = {
			limit = {
				has_variable = te_mon_receivership_clear_months
				var:te_mon_receivership_clear_months >= te_mon_receivership_exit_months
			}
			te_cred_end_direct_receivership = yes
		}
		else_if = {
			limit = { te_cred_has_direct_receivership = yes }
			set_variable = { name = te_mon_receivership value = 1 }
			set_variable = { name = te_mon_receiver value = var:te_cred_receiver }
		}
		else_if = {
			limit = { te_mon_full_system = yes }
			# Its conditions failed under the Full rule: the debtor left the bloc
			# or became a command economy, or the receiver stopped leading it or
			# backing anyone (in default, no national bank). Over, as the
			# article's requirement_to_maintain would end it.
			te_cred_end_direct_receivership = yes
		}
	}
}

te_cred_end_direct_receivership = {
	remove_variable = te_cred_direct_receivership
	if = {
		limit = { has_variable = te_cred_receiver }
		remove_variable = te_cred_receiver
	}
	if = {
		limit = { has_variable = te_cred_direct_binding }
		remove_variable = te_cred_direct_binding
	}
}

# Scope: the leader.
te_cred_leader_monthly = {
	# A member the monetary layer does not read monthly yet gets its discovery
	# now, and the leader with it. The layer reads a country every month only
	# while its te_mon_arr_scan is 1, which its own discovery sets (a consortium
	# member is on the list); a member that joined, or left and came back, has 0
	# or none from its last discovery outside, and would wait for the yearly
	# scan while the leader pays for its guarantee from the first month. Keyed on
	# the flag itself, so a rejoining member is caught however soon it returns.
	if = {
		limit = { te_mon_full_system = yes }
		power_bloc = {
			every_power_bloc_member = {
				limit = {
					NOT = { this = root }
					NOT = { var:te_mon_arr_scan ?= 1 }
				}
				trigger_event = { id = te_monetary_internal.2 days = 1 }
				root = { set_variable = { name = te_cred_new_member value = 1 days = 5 } }
			}
		}
		if = {
			limit = { has_variable = te_cred_new_member }
			trigger_event = { id = te_monetary_internal.2 days = 1 }
			remove_variable = te_cred_new_member
		}
	}
	te_cred_dl_pulse = yes
}

# Scope: any country, monthly.

	# receivership rather than ending it.
	if = {

				}

			# the new receivership, treaty or not (te_cc.9 is due three days on).
			NOT = { has_variable = te_cred_rescue_pending }
		}
		remove_variable = te_cred_adjustment_by
	}
	te_cred_dl_sync_receiver_modifier = yes
}


# --- Development Lending (§ 5.4) -------------------------------------------------------------
# Scope: the leader. Members above the bloc's GDP per head pay a share of a
# month of their investment pool's income (te_cred_dl_payment); members below
# it, the leader excepted, share what was paid by population. Money moves pool
# to pool (add_investment_pool, paired), so none is made or lost, and none is
# paid while there is nobody to receive it. Runs under any banking rule: the
# pools are vanilla's, not the monetary layer's.
te_cred_dl_pulse = {
	if = {
		limit = { te_cred_dl_tier_1_plus = yes }
		set_variable = { name = te_cred_dl_bloc_gdppc value = te_cred_dl_bloc_gdppc_now }
		# Self-check, once a game: the bloc's GDP per head is above 0. The
		# leader is a member with GDP and people of its own, so 0 means the
		# members' GDP read gave nothing (the load-time refusal of 2026-11-19,
		# te_creditors_consortium_values.txt) and nobody ever receives.
		if = {
			limit = { NOT = { has_global_variable = te_cc_sc_dl_gdppc_logged } }
			set_global_variable = te_cc_sc_dl_gdppc_logged
			if = {
				limit = { var:te_cred_dl_bloc_gdppc > 0 }
				debug_log = "TE_SELF_CHECK OK cc_dl_bloc_gdppc (bloc GDP per head [ROOT.GetCountry.MakeScope.Var('te_cred_dl_bloc_gdppc').GetValue|2]: the members' GDP is read) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN cc_dl_bloc_gdppc (bloc GDP per head [ROOT.GetCountry.MakeScope.Var('te_cred_dl_bloc_gdppc').GetValue|2]: the members' GDP read gave nothing; see docs/testing/power-bloc-identities-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		set_variable = { name = te_cred_dl_pool value = 0 }
		set_variable = { name = te_cred_dl_pop_m value = 0 }
		# 1. The receivers' population.
		power_bloc = {
			every_power_bloc_member = {
				set_variable = { name = te_cred_dl_paid value = 0 }
				set_variable = { name = te_cred_dl_received value = 0 }
				if = {
					limit = { te_cred_dl_is_receiver = yes }
					save_scope_as = te_cred_dl_m
					root = { change_variable = { name = te_cred_dl_pop_m add = scope:te_cred_dl_m.te_cred_dl_pop_weight } }
				}
			}
		}
		if = {
			limit = { var:te_cred_dl_pop_m > 0 }
			# 2. The donors pay in.
			power_bloc = {
				every_power_bloc_member = {
					if = {
						limit = { te_cred_dl_is_donor = yes }
						set_variable = { name = te_cred_dl_paid value = te_cred_dl_payment }
						if = {
							limit = { var:te_cred_dl_paid > 0 }
							add_investment_pool = {
								value = var:te_cred_dl_paid
								multiply = -1
							}
							save_scope_as = te_cred_dl_m
							root = { change_variable = { name = te_cred_dl_pool add = scope:te_cred_dl_m.var:te_cred_dl_paid } }
						}
					}
				}
			}
			# 3. The receivers are paid out.
			if = {
				limit = { var:te_cred_dl_pool > 0 }
				power_bloc = {
					every_power_bloc_member = {
						if = {
							limit = { te_cred_dl_is_receiver = yes }
							set_variable = { name = te_cred_dl_received value = te_cred_dl_receipt }
							add_investment_pool = var:te_cred_dl_received
						}
					}
				}
			}
		}
	}
	else_if = {
		limit = { has_variable = te_cred_dl_bloc_gdppc }
		remove_variable = te_cred_dl_bloc_gdppc
	}
}

			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN cc_adjustment_imposition ([ROOT.GetCountry.GetNameNoFormatting] may impose none of the [ROOT.GetCountry.MakeScope.Var('te_cred_sc_tested').GetValue|0] economic laws it holds and [SCOPE.sCountry('te_cc_sc_target').GetNameNoFormatting], in its receivership, lacks: the INJECTed can_impose is not read; see docs/testing/power-bloc-identities-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		remove_variable = te_cred_sc_tested
		remove_variable = te_cred_sc_rights
	}
}

# One law of the ladder above. Scope: the leader.
te_cred_sc_imposition_rung = {
	if = {
		limit = {
			has_law = law_type:$LAW$
			scope:te_cc_sc_target = { NOT = { has_law = law_type:$LAW$ } }
		}
		change_variable = { name = te_cred_sc_tested add = 1 }
		if = {
			limit = {
				has_law_imposition_rights = {
					target_country = scope:te_cc_sc_target
					# law_type, as vanilla's 00_impose_law.txt writes it; the
					# engine docs' `type =` is refused at load (2026-11-19).
					law_type = law_type:$LAW$
				}
			}
			change_variable = { name = te_cred_sc_rights add = 1 }
		}
	}
}
