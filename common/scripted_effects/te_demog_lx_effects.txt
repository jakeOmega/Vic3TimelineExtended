# Inert before the first ageing carrier: a state with no carrier and no stored eligibility takes the plain sweep
# (te_demog_step_run), so every earlier number is the census's as before (test_demographics_golden.py). Under Display
# only it moves the census's figures; outside Full nothing reaches the engine.
#
# State variables (only once a state's late game has started: an owner with a carrier, or treated people):
#   te_dg_s<j>, te_dg_d<j>  per five-year birth band j (ring slot / 5), adults only: the share of its people who have
#                       been in longevity care, and their mean deficit (calendar less biological age). A band with no
#                       adult has neither, so its next cohorts start afresh at 25
#   te_dg_lx_ps, te_dg_lx_pd  the pool's (150 and over), likewise
#   te_dg_elig          eligibility E, the share within reach of longevity care, drifting toward its target (§4.2) at
#                       the medical diffusion rate both ways
#   te_dg_cover         cover c, the share of the ever-treated still in care: falls with E in proportion, returns first
#                       when E rises (bodies out of care keep their age, call 9)
#   te_dg_lx_sbar       the adults' ever-treated share after the last step (the joining share's denominator)
#   te_dg_age_rate      r, the ageing rate of those in care the step used
#   te_dg_lx_pub        the share on the public route (the health bill's longevity-care term, §5.2)
#   te_dg_lx_l100, te_dg_lx_l100t, te_dg_lx_l100u  Living to 100: the state's, the treated's and the untreated's
#   te_dg_old_age_c     the untreated's old-age threshold (Linked to Common Longevity follows it; te_dg_old_age is the
#                       state's blend after a late step)
#   te_dg_lx_rho        the adults' treated deaths per head over the untreated's
#   te_dg_lx_age, te_dg_lx_bio  the mean calendar and biological ages after the step
# Country: te_dg_lx_w, the market threshold (§4.2; from the country's first carrier, falling yearly), and the people-
# weighted te_dg_elig, te_dg_lx_pub, te_dg_lx_l100, _l100t, _l100u, te_dg_old_age_c, te_dg_lx_age, te_dg_lx_bio.
# ============================================================================

# THIS = a state, in te_demog_step_begin after te_demog_prepare (demographics_model.lx_begin): whether this step is a
# late one (local te_dg_lx_on), and if so the year's ageing rate, pull, eligibility, cover and joining share, and the age
# groups' rates at the model's own multipliers (before the step's on-top scales).
te_demog_lx_begin = {
	set_local_variable = { name = te_dg_lx_on value = 0 }
	# uploaded minds (te_demog_um_effects.txt): whether this step books uploads, and keeps the sweep's sums for them
	set_local_variable = { name = te_dg_um_on value = 0 }
	set_local_variable = { name = te_dg_um_sums value = 0 }
	set_local_variable = { name = te_dg_lx_lv value = te_demog_lx_ecto_levels }
	if = {
		limit = {
			OR = {
				has_variable = te_dg_elig
				te_demog_lx_carrier_on > 0
				local_var:te_dg_lx_lv > 0
			}
		}
		set_local_variable = { name = te_dg_lx_on value = 1 }
		# artificial wombs (§3.5; demographics_model.ecto_share): their capacity over last year's births, at most the cap
		set_local_variable = { name = te_dg_lx_ecto value = 0 }
		if = {
			limit = {
				local_var:te_dg_lx_lv > 0
				has_variable = te_dg_births
				var:te_dg_births > 0
			}
			set_local_variable = {
				name = te_dg_lx_ecto
				value = {
					value = local_var:te_dg_lx_lv
					multiply = te_demog_k_lx_ecto_per_level
					divide = { value = var:te_dg_births min = 0.00001 }
					max = te_demog_k_lx_ecto_max
				}
			}
		}
		set_local_variable = { name = te_dg_lx_rr value = te_demog_lx_rate }
		set_local_variable = { name = te_dg_lx_pull value = te_demog_lx_pull }
		# eligibility's target and the public route (generated)
		te_demog_lx_access = yes
		set_local_variable = { name = te_dg_lx_e0 value = 0 }
		if = {
			limit = { has_variable = te_dg_elig }
			set_local_variable = { name = te_dg_lx_e0 value = var:te_dg_elig }
		}
		set_local_variable = {
			name = te_dg_lx_e
			value = {
				value = local_var:te_dg_lx_target
				subtract = local_var:te_dg_lx_e0
				multiply = te_demog_medical_diffusion
				add = local_var:te_dg_lx_e0
			}
		}
		set_local_variable = { name = te_dg_lx_gap value = { value = local_var:te_dg_lx_target subtract = local_var:te_dg_lx_e } }
		if = {
			limit = {
				local_var:te_dg_lx_gap < te_demog_k_lx_snap
				local_var:te_dg_lx_gap > te_demog_k_lx_snap_neg
			}
			set_local_variable = { name = te_dg_lx_e value = local_var:te_dg_lx_target }
		}
		# cover and the joining share: when E falls, cover falls in proportion; when it rises, cover returns first, then
		# the same share of every band's untreated joins with no deficit (§1.2's step 4)
		set_local_variable = { name = te_dg_lx_c0 value = 1 }
		if = {
			limit = { has_variable = te_dg_cover }
			set_local_variable = { name = te_dg_lx_c0 value = var:te_dg_cover }
		}
		set_local_variable = { name = te_dg_lx_sbar value = 0 }
		if = {
			limit = { has_variable = te_dg_lx_sbar }
			set_local_variable = { name = te_dg_lx_sbar value = var:te_dg_lx_sbar }
		}
		set_local_variable = { name = te_dg_lx_phi value = 0 }
		if = {
			limit = { local_var:te_dg_lx_e0 <= 0 }
			if = {
				limit = {
					local_var:te_dg_lx_sbar > 0
					local_var:te_dg_lx_sbar >= local_var:te_dg_lx_e
				}
				set_local_variable = { name = te_dg_lx_c value = { value = local_var:te_dg_lx_e divide = { value = local_var:te_dg_lx_sbar min = 0.00001 } } }
			}
			else = {
				set_local_variable = { name = te_dg_lx_c value = 1 }
				set_local_variable = {
					name = te_dg_lx_phi
					value = {
						value = local_var:te_dg_lx_e
						subtract = local_var:te_dg_lx_sbar
						divide = { value = 1 subtract = local_var:te_dg_lx_sbar min = 0.00001 }
					}
				}
			}
		}
		else_if = {
			limit = { local_var:te_dg_lx_e <= local_var:te_dg_lx_e0 }
			set_local_variable = {
				name = te_dg_lx_c
				value = { value = local_var:te_dg_lx_c0 multiply = local_var:te_dg_lx_e divide = { value = local_var:te_dg_lx_e0 min = 0.00001 } }
			}
		}
		else = {
			set_local_variable = {
				name = te_dg_lx_c2
				value = { value = local_var:te_dg_lx_c0 multiply = local_var:te_dg_lx_e divide = { value = local_var:te_dg_lx_e0 min = 0.00001 } }
			}
			if = {
				limit = { local_var:te_dg_lx_c2 <= 1 }
				set_local_variable = { name = te_dg_lx_c value = local_var:te_dg_lx_c2 }
			}
			else = {
				set_local_variable = { name = te_dg_lx_c value = 1 }
				set_local_variable = {
					name = te_dg_lx_phi
					value = {
						value = local_var:te_dg_lx_sbar
						multiply = local_var:te_dg_lx_c2
						max = 1
						subtract = local_var:te_dg_lx_sbar
						divide = { value = 1 subtract = local_var:te_dg_lx_sbar min = 0.00001 }
					}
				}
			}
		}
		set_local_variable = { name = te_dg_lx_c value = { value = local_var:te_dg_lx_c min = 0 max = 1 } }
		set_local_variable = { name = te_dg_lx_phi value = { value = local_var:te_dg_lx_phi min = 0 max = 1 } }
		set_variable = { name = te_dg_elig value = local_var:te_dg_lx_e }
		set_variable = { name = te_dg_cover value = local_var:te_dg_lx_c }
		set_variable = { name = te_dg_age_rate value = local_var:te_dg_lx_rr }
		set_variable = { name = te_dg_lx_pub value = local_var:te_dg_lx_pub }
		# the age groups' rates at the model's own multipliers (generated), before te_demog_on_top_scales scales them
		te_demog_lx_group_rates_unscaled = yes
		# uploaded minds: under Full with the owner's capacity, the age groups' infection and chronic rates too
		te_demog_um_begin = yes
	}
}

# THIS = a state, in te_demog_step_run in place of the plain sweep while te_dg_lx_on (te_demog_lx_begin): the age groups'
# rates with the step's scales (which re-enters group 0 for the sweep), the sweep's late sums, the late sweep, then the
# bands' and the tables' figures (te_demog_lx_finish) and the year's deaths by cause (te_demog_lx_cause_deaths).

	set_local_variable = { name = te_dg_g value = 0 }
	te_demog_enter_group = yes
	set_local_variable = { name = te_dg_lx_t value = 1 }

	set_local_variable = { name = te_dg_lx_bs value = 0 }

	set_local_variable = { name = te_dg_lx_bda value = 0 }
	set_local_variable = { name = te_dg_lx_bt value = 0 }
	set_local_variable = { name = te_dg_lx_ba value = 0 }
	set_local_variable = { name = te_dg_lx_bj value = 0 }
	set_local_variable = { name = te_dg_lx_bta value = 0 }
	set_local_variable = { name = te_dg_lx_sk value = 0 }
	set_local_variable = { name = te_dg_lx_sk2 value = 0 }
	set_local_variable = { name = te_dg_lx_ab_t value = 0 }
	set_local_variable = { name = te_dg_lx_guf value = 0 }
	set_local_variable = { name = te_dg_lx_gum value = 0 }
	set_local_variable = { name = te_dg_lx_gtf value = 0 }
	set_local_variable = { name = te_dg_lx_gtm value = 0 }
	set_local_variable = { name = te_dg_lx_gtqf value = 0 }

# The late sweep's band calls (generated, te_demog_lx_sweep): $WHEN$ open or close, in pass 1 for a band all below the open
# slot (its top slot $TOP$ below it), in pass 2 for the rest. Each band is opened and closed once a step.
te_demog_lx_pass1_band = {
	if = {
		limit = { local_var:te_dg_open > $TOP$ }
		te_demog_lx_band_$WHEN$ = { J = $J$ }
	}
}
te_demog_lx_pass2_band = {
	if = {
		limit = { NOT = { local_var:te_dg_open > $TOP$ } }
		te_demog_lx_band_$WHEN$ = { J = $J$ }
	}
}

# Band $J$ before its slots: its share and deficit into locals, the deficit it will have after this year's ageing (the
# slots' biological ages after the step read it), and its sums at 0.
te_demog_lx_band_open = {
	set_local_variable = { name = te_dg_lx_bs value = 0 }
	set_local_variable = { name = te_dg_lx_bd value = 0 }
	if = {
		limit = { has_variable = te_dg_s$J$ }
		set_local_variable = { name = te_dg_lx_bs value = var:te_dg_s$J$ }
	}
	if = {
		limit = { has_variable = te_dg_d$J$ }
		set_local_variable = { name = te_dg_lx_bd value = var:te_dg_d$J$ }
	}
	set_local_variable = {
		name = te_dg_lx_bda
		value = { value = 1 subtract = local_var:te_dg_lx_rr multiply = local_var:te_dg_lx_c add = local_var:te_dg_lx_bd }
	}
	set_local_variable = { name = te_dg_lx_bt value = 0 }
	set_local_variable = { name = te_dg_lx_ba value = 0 }
	set_local_variable = { name = te_dg_lx_bj value = 0 }
	set_local_variable = { name = te_dg_lx_bta value = 0 }
}

# Band $J$ after its slots (demographics_model.lx_close_bands): selection (the treated survivors over all survivors),
# ageing and the pull, the year's joining share of its untreated, then the cohort that turned 25 at E. A band with no
# adult left drops both variables.
te_demog_lx_band_close = {
	set_local_variable = { name = te_dg_lx_bn value = { value = local_var:te_dg_lx_ba add = local_var:te_dg_lx_bj } }
	if = {
		limit = { local_var:te_dg_lx_bn > 0 }
		set_local_variable = { name = te_dg_lx_aged value = local_var:te_dg_lx_bda }
		if = {
			limit = { local_var:te_dg_lx_bt > 0 }
			change_local_variable = {
				name = te_dg_lx_aged
				add = {
					value = local_var:te_dg_lx_bta
					divide = { value = local_var:te_dg_lx_bt min = 0.00001 }
					subtract = local_var:te_dg_lx_aged
					subtract = te_demog_k_lx_pull_to
					min = 0
					multiply = local_var:te_dg_lx_pull
					multiply = local_var:te_dg_lx_c
				}
			}
		}
		set_local_variable = { name = te_dg_lx_ssel value = 0 }
		if = {
			limit = { local_var:te_dg_lx_ba > 0 }
			set_local_variable = { name = te_dg_lx_ssel value = { value = local_var:te_dg_lx_bt divide = { value = local_var:te_dg_lx_ba min = 0.00001 } } }
		}
		set_local_variable = {
			name = te_dg_lx_s1
			value = { value = 1 subtract = local_var:te_dg_lx_ssel multiply = local_var:te_dg_lx_phi add = local_var:te_dg_lx_ssel }
		}
		set_local_variable = { name = te_dg_lx_d1 value = 0 }
		if = {
			limit = { local_var:te_dg_lx_s1 > 0 }
			set_local_variable = {
				name = te_dg_lx_d1
				value = { value = local_var:te_dg_lx_aged multiply = local_var:te_dg_lx_ssel divide = { value = local_var:te_dg_lx_s1 min = 0.00001 } }
			}
		}
		set_local_variable = {
			name = te_dg_lx_tn
			value = {
				value = local_var:te_dg_lx_s1
				multiply = local_var:te_dg_lx_ba
				add = { value = local_var:te_dg_lx_e multiply = local_var:te_dg_lx_bj }
			}
		}
		if = {
			limit = { local_var:te_dg_lx_tn > 0 }
			set_variable = { name = te_dg_s$J$ value = { value = local_var:te_dg_lx_tn divide = { value = local_var:te_dg_lx_bn min = 0.00001 } } }
			set_variable = {
				name = te_dg_d$J$
				value = {
					value = local_var:te_dg_lx_d1
					multiply = local_var:te_dg_lx_s1
					multiply = local_var:te_dg_lx_ba
					divide = { value = local_var:te_dg_lx_tn min = 0.00001 }
				}
			}
		}
		else = {
			te_demog_lx_drop_band = { J = $J$ }
		}
		change_local_variable = { name = te_dg_lx_tsum add = local_var:te_dg_lx_tn }
		change_local_variable = { name = te_dg_lx_asum add = local_var:te_dg_lx_bn }
	}
	else = {
		te_demog_lx_drop_band = { J = $J$ }
	}
}

te_demog_lx_drop_band = {
	if = {
		limit = { has_variable = te_dg_s$J$ }
		remove_variable = te_dg_s$J$
	}
	if = {
		limit = { has_variable = te_dg_d$J$ }
		remove_variable = te_dg_d$J$
	}
}

# The late sweep's slots (generated, te_demog_lx_sweep): the plain sweep's guards, then te_demog_slot after the slot's late
# locals (te_demog_lx_slot_pre).
te_demog_lx_slot_p1 = {
	if = {
		limit = { local_var:te_dg_open > $S$ }
		te_demog_lx_slot_pre = yes
		te_demog_slot = { S = $S$ }
	}
}
te_demog_lx_slot_p2 = {
	if = {
		limit = { NOT = { local_var:te_dg_open > $S$ } }
		te_demog_lx_slot_pre = yes
		te_demog_slot = { S = $S$ }
	}
}

# Before te_demog_slot: the age group closing as this slot's rate age leaves it keeps its sums for the life tables (te_demog_slot
# enters the next), and the slot's treated share is its band's for an adult rate age (25 and over), else 0.
te_demog_lx_slot_pre = {
	if = {
		limit = { local_var:te_dg_r = local_var:te_dg_next_group }
		te_demog_lx_group_flush = yes
	}
	set_local_variable = { name = te_dg_lx_sk value = 0 }
	if = {
		limit = { local_var:te_dg_r >= te_demog_k_lx_join_age }
		set_local_variable = { name = te_dg_lx_sk value = local_var:te_dg_lx_bs }
	}
}

# THIS = a state, in te_demog_age_slot while te_dg_lx_t (demographics_model.lx_births): the cohort's births with its two parts
# (locals te_dg_f, te_dg_m; the slot's share te_dg_lx_sk and its band's deficit te_dg_lx_bd). Planned births follow the
# calendar, as families intend them, x the fecundity of the body that bears them; unplanned births the body (§3.1-§3.2;
# both schedules are today's shape, te_dg_asfr, at the calendar age). Artificial wombs carry the share te_dg_lx_ecto of the
# planned births with no fecundity loss and no maternal death, from the cohort's women and men alike (§3.5). Sets te_dg_b
# and the women's own births by part, te_dg_lx_bmu and te_dg_lx_bmt, which carry the maternal deaths
# (te_demog_lx_deaths); adds to the step's births and maternal deaths, and to the births to women of 50 and over.
# Each term multiplies the people and the schedule (per 100,000) first and divides by 100000 last: the engine keeps five
# decimals, and TFR / 100000 taken first (0.00001 for a TFR of 1 to 2) times a share under 1 came out 0, so no late state
# under a TFR of 2 had a birth (scripting_best_practices.md § values in units of 1e-5).
te_demog_lx_births = {
	set_local_variable = { name = te_dg_lx_ft value = { value = local_var:te_dg_f multiply = local_var:te_dg_lx_sk } }
	set_local_variable = { name = te_dg_lx_fu value = { value = local_var:te_dg_f subtract = local_var:te_dg_lx_ft } }
	set_local_variable = { name = te_dg_lx_int value = 0 }
	set_local_variable = { name = te_dg_lx_plu value = 0 }
	set_local_variable = { name = te_dg_lx_unu value = 0 }
	set_local_variable = { name = te_dg_lx_plt value = 0 }
	set_local_variable = { name = te_dg_lx_unt value = 0 }
	if = {
		limit = { local_var:te_dg_asfr > 0 }
		set_local_variable = { name = te_dg_lx_fa value = local_var:te_dg_r }
		te_demog_lx_fert_at = yes
		set_local_variable = { name = te_dg_lx_int0 value = local_var:te_dg_lx_int }
		set_local_variable = {
			name = te_dg_lx_plu
			value = {
				value = local_var:te_dg_lx_fu
				multiply = local_var:te_dg_asfr
				multiply = local_var:te_dg_sh_pl
				multiply = { value = 1 subtract = local_var:te_dg_lx_ecto }
				multiply = local_var:te_dg_tfr
				divide = 100000
			}
		}
		set_local_variable = {
			name = te_dg_lx_unu
			value = {
				value = local_var:te_dg_lx_fu
				multiply = local_var:te_dg_asfr
				multiply = local_var:te_dg_sh_un
				multiply = local_var:te_dg_tfr
				divide = 100000
			}
		}
		set_local_variable = { name = te_dg_lx_int value = local_var:te_dg_lx_int0 }
	}
	if = {
		limit = { local_var:te_dg_lx_sk > 0 }
		set_local_variable = { name = te_dg_lx_fa value = { value = local_var:te_dg_r subtract = local_var:te_dg_lx_bd } }
		set_local_variable = { name = te_dg_lx_intr value = local_var:te_dg_lx_int }
		te_demog_lx_fert_at = yes
		set_local_variable = {
			name = te_dg_lx_plt
			value = {
				value = local_var:te_dg_lx_ft
				multiply = local_var:te_dg_lx_intr
				multiply = local_var:te_dg_lx_fec
				multiply = local_var:te_dg_sh_pl
				multiply = { value = 1 subtract = local_var:te_dg_lx_ecto }
				multiply = local_var:te_dg_tfr
				divide = 100000
			}
		}
		set_local_variable = {
			name = te_dg_lx_unt
			value = {
				value = local_var:te_dg_lx_ft
				multiply = local_var:te_dg_lx_asfr
				multiply = local_var:te_dg_sh_un
				multiply = local_var:te_dg_tfr
				divide = 100000
			}
		}
		set_local_variable = { name = te_dg_lx_int value = local_var:te_dg_lx_intr }
	}

# THIS = a state after the late sweep (demographics_model.lx_finish): the last age group's sums, the adults' ever-treated
# share, the three life tables and rho (generated), the mean ages (the pool at its mean age, the newborns at 0).
te_demog_lx_finish = {
	te_demog_lx_group_flush = yes
	# births to women of 50 and over, over all births (children per woman's hover; §6.2)
	if = {
		limit = {
			local_var:te_dg_lx_b50 > 0
			local_var:te_dg_births > 0
		}
		set_variable = { name = te_dg_lx_b50 value = { value = local_var:te_dg_lx_b50 divide = { value = local_var:te_dg_births min = 0.00001 } } }
	}
	else_if = {
		limit = { has_variable = te_dg_lx_b50 }
		remove_variable = te_dg_lx_b50
	}
	set_variable = { name = te_dg_lx_sbar value = 0 }
	if = {
		limit = { local_var:te_dg_lx_asum > 0 }
		set_variable = { name = te_dg_lx_sbar value = { value = local_var:te_dg_lx_tsum divide = { value = local_var:te_dg_lx_asum min = 0.00001 } } }
	}
	te_demog_lx_tables = yes
	# rich pops die less (late-game §4.3, call 14): the census's wealth slope, which te_demog_rates_refresh applies under Full
	te_demog_lx_wealth_slope = yes
	set_local_variable = { name = te_dg_lx_pp value = { value = var:te_dg_pf add = var:te_dg_pm } }
	change_local_variable = { name = te_dg_lx_mp add = { value = local_var:te_dg_lx_pp add = local_var:te_dg_births } }
	change_local_variable = { name = te_dg_lx_mcal add = { value = local_var:te_dg_lx_pp multiply = var:te_dg_pa } }
	if = {
		limit = {
			has_variable = te_dg_lx_ps
			has_variable = te_dg_lx_pd
		}
		change_local_variable = {
			name = te_dg_lx_mdef
			add = { value = local_var:te_dg_lx_pp multiply = var:te_dg_lx_ps multiply = var:te_dg_lx_pd }
		}
	}
	set_variable = { name = te_dg_lx_age value = { value = local_var:te_dg_lx_mcal divide = { value = local_var:te_dg_lx_mp min = 0.00001 } } }
	set_variable = {
		name = te_dg_lx_bio
		value = {
			value = local_var:te_dg_lx_mcal
			subtract = local_var:te_dg_lx_mdef
			divide = { value = local_var:te_dg_lx_mp min = 0.00001 }
		}
	}
}

# THIS = a state, in its seed after te_demog_prepare (demographics_model.seed): once ageing is slowed (an owner with a carrier)

		}
	}

# THIS = a state, in its seed's slot of an adult while te_dg_lx_seed_on (demographics_model.seed): the year's survival at

te_demog_lx_seed_decrement = {

		name = te_dg_lsf
		multiply = {
			value = local_var:te_dg_asfr
			multiply = local_var:te_dg_tfr
			multiply = local_var:te_dg_mmr
			divide = 100000
			add = {
				value = 1
				subtract = local_var:te_dg_lx_e
				multiply = local_var:te_dg_qf
				add = { value = local_var:te_dg_lx_e multiply = local_var:te_dg_lx_qf }
			}
			divide = 100000
			multiply = -1
			add = 1
		}
	}
	change_local_variable = {
		name = te_dg_lsm
		multiply = {
			value = 1
			subtract = local_var:te_dg_lx_e
			multiply = local_var:te_dg_qm
			add = { value = local_var:te_dg_lx_e multiply = local_var:te_dg_lx_qm }
			divide = 100000
			multiply = -1
			add = 1
		}
	}
}

# THIS = ROOT = a state, in te_demog_rates_refresh's deaths under Full (demographics_model.lx_wealth_term; late-game §4.3,
# call 14): the wealth slope the census applies, local te_dg_rate_slope (per wealth level, on the census's rate; 0
# outside the late game), with the richest pop (the walk's te_dg_w_wtop) held at the treated's rate over the state's
# (te_dg_lx_rho over te_dg_lx_mbar), so no pop's deaths go below zero; and te_dg_rate_slope_net, what it moves on the
# bare curves' scale, which M takes out (te_dg_rate_tb x the slope x the bare deaths' mean wealth, te_dg_w_edw over
# te_dg_w_ed0), so the state's total stays the census's.
te_demog_lx_rate_slope = {
	set_local_variable = { name = te_dg_rate_slope value = 0 }
	set_local_variable = { name = te_dg_rate_slope_net value = 0 }
	if = {
		limit = {
			has_variable = te_dg_lx_slope
			local_var:te_dg_w_ed0 > 0
		}
		set_local_variable = { name = te_dg_rate_slope value = var:te_dg_lx_slope }
		set_local_variable = { name = te_dg_rate_wbar value = { value = local_var:te_dg_w_edw divide = { value = local_var:te_dg_w_ed0 min = 1 } } }
		if = {
			limit = { local_var:te_dg_w_wtop > local_var:te_dg_rate_wbar }
			set_local_variable = {
				name = te_dg_rate_slope
				value = {
					value = var:te_dg_lx_rho
					divide = { value = var:te_dg_lx_mbar min = 0.00001 }
					subtract = 1
					divide = { value = local_var:te_dg_w_wtop subtract = local_var:te_dg_rate_wbar min = 0.00001 }
					min = var:te_dg_lx_slope
				}
			}
		}
		set_local_variable = {
			name = te_dg_rate_slope_net
			value = { value = local_var:te_dg_rate_slope multiply = local_var:te_dg_rate_tb multiply = local_var:te_dg_rate_wbar }
		}
	}
}

# THIS = a country, on its yearly pulse whatever the rule (te_demog_country_yearly): the country's longevity-gap modifier
# comes off once the rule is no longer Full. Its refresh site (te_demog_lx_country_store) runs only inside the census, which
# Disabled never reaches, so a game switched from Full to Disabled would otherwise keep the unions' and farmers' approval
# down for good. The state's modifier needs no such step: its refresh runs on every state pulse whatever the rule.
te_demog_lx_politics_clear = {
	if = {
		limit = {
			NOT = { te_demog_effects_run = yes }
			has_modifier = te_demog_lx_gap_politics
		}
		remove_modifier = te_demog_lx_gap_politics
	}
}

# THIS = ROOT = a state, after the census's other terms (te_demog_state_yearly, fast mode's clock step): the longevity gap's
# state modifier (late-game §4.4, call 15), the one refresh site of te_demog_lx_gap_labor. Applied only under Full with
# te_demog_k_lx_politics on (on since 2026-10-16, owner), at the state's gap (Living to 100, treated less untreated).
te_demog_lx_politics_refresh = {
	if = {
		limit = { has_modifier = te_demog_lx_gap_labor }
		remove_modifier = te_demog_lx_gap_labor
	}
	if = {
		limit = {
			te_demog_effects_run = yes
			te_demog_k_lx_politics > 0
			te_demog_lx_gap > 0.01
		}
		add_modifier = { name = te_demog_lx_gap_labor multiplier = te_demog_lx_gap }
	}
}
