# ============================================================================
# GLOBAL COMMONS — SEABED LICENCES: mechanics
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-16-global-commons-design.md § 2.
# The script follows the Python model scripts/analysis/seabed_model.py step
# for step; each effect names the model function it stands for
# (test_seabed_allocation.py runs the model over random worlds;
# test_seabed_script.py holds the names, constants and call order to it).
#
# STATE, all in variables, so a revolution's winner keeps it (CLAUDE.md, Top
# gotchas) and the two modifiers are rebuilt from it:
#   state    te_seabed_basin              1 Pacific .. 5 Southern (generated)
#            te_seabed_lic_<basin>        licences recorded on its Deep-Sea
#                                         Mine levels (built or queued)
#            te_seabed_holder             its owner when last recorded
#            te_seabed_unb_snap1/2        licences on its unbuilt levels at
#                                         the last two first-sharings of a
#                                         year (the holding fee's grace)
#   country  te_seabed_unbuilt_<basin>    entitled and unused
#            te_seabed_held_<basin>       records in its states (yearly)
#            te_seabed_ent_<basin>        entitlement (yearly)
#            te_seabed_w_<basin>          claim weight (yearly)
#            te_seabed_home_<basin>       flag: borders the basin
#            te_seabed_policy(_prev)      1 Survey, 2 Sponsor, 3 Unilateral
#                                         (absent = 1; never seeded, so a
#                                         revolution's winner inherits it)
#            te_seabed_floor_<basin>      Sponsor's floor: the share a sharing
#                                         may not go below (the year's first
#                                         sharing sets it; absent = none)
#            te_seabed_fprot_<basin>      the claims part of that floor
#            te_seabed_fz_<basin>         what the sharing freezes it at:
#                                         max(held, min(floor, held +
#                                         protected claims)), scaled to fit
#            te_seabed_relinq_<basin>     flag: relinquished, weight 0 there
#                                         until Claim Again
#            te_seabed_sh_<basin>         held as the sharing counts it: a
#                                         lease's licences go to its lessor
#            te_seabed_lease_src_<basin>  its lease there: the lessor (a
#                                         country), with te_seabed_lease_n_,
#                                         _built_ and _unused_<basin> (its
#                                         total, licences built on it, unused
#                                         claims, part of unbuilt); _past_
#                                         <basin> built on ended leases from
#                                         the same lessor, still its charge
#            te_seabed_subj_unused_<basin> claims its subjects passed it,
#                                         unused (part of unbuilt); on a
#                                         subject te_seabed_pass_unused_ and
#                                         _pass_built_<basin>, its own share
#                                         of them and the licences built on
#                                         them, charged to te_seabed_pass_to;
#                                         te_seabed_shleft_<basin> the
#                                         sharing's holdings not yet shifted
#            te_seabed_fee_cached         the holding fee a week, the
#                                         modifier's multiplier
#            te_seabed_policy_changed     month of the last policy change
#                                         (the five-year lock; absent = free)
#            te_seabed_policy_cost_cached the influence the policy holds,
#                                         its modifier's multiplier
#            te_seabed_queued_<basin>,    licences on its queued levels, and
#            te_seabed_lic_now_<basin>    all its licences, by basin (monthly)
#            te_seabed_need_navy_<basin>  the "+1 licence" readout (yearly)
#            te_seabed_coast_<basin>/_all coastal states, unincorporated ones
#                                         at half (yearly); _coast_states_
#                                         <basin> all of them, counted whole
#            te_seabed_dr_<basin>         a frozen share's reserved-half part
#            te_seabed_rkcap_<b>_<n>      players: ranked rivals' capitals
#   global   te_seabed_rk_<basin>_<n>     the six largest shares by basin
#            te_seabed_last_sharing_month the panel's "as of"
#            te_seabed_country            flag: has been recorded
#            te_seabed_shared             flag: took part in a sharing
#            te_seabed_claimed_<basin>    flag: has had a share there (the
#                                         new-claimant notice)
#   global   te_seabed_regime             1 Open, 2 Regulated, 3 Scramble
#                                         (the UN hook writes it)
#            te_seabed_records_<basin>    the world's records (yearly)
#            te_seabed_world_w_<basin>    the world's claim weight (yearly)
#            te_seabed_first_claimant     the country the rush event names
#            te_seabed_rush_announced,    flags: the rush event sent; the
#            te_seabed_full_<basin>,      basin told it is fully licensed;
#            te_seabed_claimed_seeded     the claimed marks seeded
#
# ONE DEPARTURE FROM THE SPEC'S WORDING: a level is recorded as soon as it is
# queued (its building's level_after_queued_constructions), not when it is
# finished. The model's header says why: it stops a sharing that shrinks an
# entitlement from stranding levels under construction without a licence.
#
# A LEVEL ITS OWNER QUEUED PAST ITS ROOM WAITS UNRECORDED (review H1). The
# gate's under-construction branch lets one more level in at room 0. Recorded
# against no share it would reopen the branch and grow the basin a level a
# month, so a recording by the state's own stamped owner leaves its shortfall
# unrecorded ("pending"): room stays below 0 and both branches shut until a
# share pays for it. Only § 2.4's three cases go against no share: no stamp (an
# old save's first recording, a split's new state, a resumed rule), and a
# holder gone when its state changes hands. Model: seabed_model.py's header.
#
# PULSES (te_global_commons_seabed_on_actions.txt):
#   monthly country   te_seabed_monthly_country: record (te_seabed_record_country),
#                     the holding fee (te_seabed_fee_refresh)
#   monthly state     te_seabed_refresh_long_haul: Long-Haul Seabed Operations
#   owner change      te_seabed_on_state_owner_change
#   treaty article    a Lease <Basin> Seabed Claims article's entry
#                     (122_seabed_lease.txt): te_seabed_lease_start, then the
#                     hidden te_seabed.1 a day later (te_seabed_events.txt)
#   monthly global    te_seabed_global_monthly: the rush event, the yearly
#                     sharing when due (te_seabed_yearly_pass), the
#                     follow-state self-check
#   yearly country    te_seabed_ai_choose_policy (AI countries only)
# ============================================================================

# ============================================================================
# Initialisation
# ============================================================================

# Country scope. The country's variables, at its first recording; never the
# policy (absent reads as Survey, so a revolution's winner inherits the
# nation's own choice rather than a seeded 1).
te_seabed_init_country = {
	if = {
		limit = { NOT = { has_variable = te_seabed_country } }
		set_variable = te_seabed_country
		te_seabed_init_country_basin = { B = pacific }
		te_seabed_init_country_basin = { B = atlantic }
		te_seabed_init_country_basin = { B = indian }
		te_seabed_init_country_basin = { B = arctic }
		te_seabed_init_country_basin = { B = southern }
	}
}

te_seabed_init_country_basin = {
	if = {
		limit = { NOT = { has_variable = te_seabed_unbuilt_$B$ } }
		set_variable = { name = te_seabed_unbuilt_$B$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = te_seabed_held_$B$ } }
		set_variable = { name = te_seabed_held_$B$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = te_seabed_ent_$B$ } }
		set_variable = { name = te_seabed_ent_$B$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = te_seabed_w_$B$ } }
		set_variable = { name = te_seabed_w_$B$ value = 0 }
	}
}

# State scope. The state's records, at zero, the first time it is recorded.
te_seabed_init_state = {
	if = {
		limit = { NOT = { has_variable = te_seabed_lic_pacific } }
		set_variable = { name = te_seabed_lic_pacific value = 0 }
		set_variable = { name = te_seabed_lic_atlantic value = 0 }
		set_variable = { name = te_seabed_lic_indian value = 0 }
		set_variable = { name = te_seabed_lic_arctic value = 0 }
		set_variable = { name = te_seabed_lic_southern value = 0 }
	}
}

# State scope. The state's basin, when it has none (§ 2.1: conquest and splits
# make new state scopes). te_seabed_set_state_basin is generated
# (te_seabed_basins_generated.txt); a landlocked state stays without one.
te_seabed_ensure_state_basin = {
	if = {
		limit = { NOT = { has_variable = te_seabed_basin } }
		te_seabed_set_state_basin = yes
	}
}

# ============================================================================
# Recording (model: record_country and its three state steps)
# ============================================================================

# Country scope. Model: record_country. From the monthly country pulse, and
# for every participant at the start of the yearly sharing. Returns first,
# over every coastal state; then new levels and swaps; then the holder stamp
# on every coastal state (so a level built since is charged to this country
# if the state changes hands before the next recording); then what it took
# comes off its own claims before its leased ones (te_seabed_lease_consume).
te_seabed_record_country = {
	te_seabed_init_country = yes
	save_scope_as = te_seabed_payer
	set_local_variable = { name = te_sb_monthly value = 1 }
	every_scope_state = {
		limit = { is_coastal = yes }
		te_seabed_ensure_state_basin = yes
	}
	every_scope_state = {
		limit = {
			is_coastal = yes
			has_variable = te_seabed_basin
			te_seabed_state_has_data = yes
		}
		te_seabed_init_state = yes
		te_seabed_state_return_excess = yes
	}
	set_local_variable = { name = te_sb_q_pacific value = 0 }
	set_local_variable = { name = te_sb_q_atlantic value = 0 }
	set_local_variable = { name = te_sb_q_indian value = 0 }
	set_local_variable = { name = te_sb_q_arctic value = 0 }
	set_local_variable = { name = te_sb_q_southern value = 0 }
	set_local_variable = { name = te_sb_l_pacific value = 0 }
	set_local_variable = { name = te_sb_l_atlantic value = 0 }
	set_local_variable = { name = te_sb_l_indian value = 0 }
	set_local_variable = { name = te_sb_l_arctic value = 0 }
	set_local_variable = { name = te_sb_l_southern value = 0 }
	every_scope_state = {
		limit = {
			is_coastal = yes
			has_variable = te_seabed_basin
			has_variable = te_seabed_lic_pacific
		}
		# Model: record_country's `own`. Stamped by this country at its last
		# recording: a shortfall is a level it queued past its room, and waits.
		set_local_variable = { name = te_sb_pend value = 0 }
		if = {
			limit = {
				var:te_seabed_holder ?= { this = scope:te_seabed_payer }
			}
			set_local_variable = { name = te_sb_pend value = 1 }
		}
		te_seabed_state_record_new = yes
		te_seabed_state_swap_local = yes
		te_seabed_follow_watch = yes
		te_seabed_queued_state = yes
	}
	set_variable = { name = te_seabed_queued_pacific value = local_var:te_sb_q_pacific }
	set_variable = { name = te_seabed_queued_atlantic value = local_var:te_sb_q_atlantic }
	set_variable = { name = te_seabed_queued_indian value = local_var:te_sb_q_indian }
	set_variable = { name = te_seabed_queued_arctic value = local_var:te_sb_q_arctic }
	set_variable = { name = te_seabed_queued_southern value = local_var:te_sb_q_southern }
	set_variable = { name = te_seabed_lic_now_pacific value = local_var:te_sb_l_pacific }
	set_variable = { name = te_seabed_lic_now_atlantic value = local_var:te_sb_l_atlantic }
	set_variable = { name = te_seabed_lic_now_indian value = local_var:te_sb_l_indian }
	set_variable = { name = te_seabed_lic_now_arctic value = local_var:te_sb_l_arctic }
	set_variable = { name = te_seabed_lic_now_southern value = local_var:te_sb_l_southern }
	every_scope_state = {
		limit = { is_coastal = yes }
		set_variable = { name = te_seabed_holder value = scope:te_seabed_payer }
	}
	te_seabed_lease_consume = yes
}

# State scope (te_seabed_record_country's third walk). Model: queued_by_basin.
# The licences on the state's queued levels count against its distant
# licences first, in basin order, then its own basin's (a level takes a
# distant licence only once the local share is gone, so distant ones are
# normally the newest); the per-basin sums are the country's
# te_seabed_queued_<basin>. Also sums its licences per basin
# (te_seabed_lic_now_<basin>), so the panel reads variables only.
te_seabed_queued_state = {
	set_local_variable = { name = te_sb_qn value = te_seabed_state_unbuilt_lic }
	te_seabed_queued_take_distant = { B = pacific N = 1 }
	te_seabed_queued_take_distant = { B = atlantic N = 2 }
	te_seabed_queued_take_distant = { B = indian N = 3 }
	te_seabed_queued_take_distant = { B = arctic N = 4 }
	te_seabed_queued_take_distant = { B = southern N = 5 }
	te_seabed_queued_take_own = { B = pacific N = 1 }
	te_seabed_queued_take_own = { B = atlantic N = 2 }
	te_seabed_queued_take_own = { B = indian N = 3 }
	te_seabed_queued_take_own = { B = arctic N = 4 }
	te_seabed_queued_take_own = { B = southern N = 5 }
	change_local_variable = { name = te_sb_l_pacific add = var:te_seabed_lic_pacific }
	change_local_variable = { name = te_sb_l_atlantic add = var:te_seabed_lic_atlantic }
	change_local_variable = { name = te_sb_l_indian add = var:te_seabed_lic_indian }
	change_local_variable = { name = te_sb_l_arctic add = var:te_seabed_lic_arctic }
	change_local_variable = { name = te_sb_l_southern add = var:te_seabed_lic_southern }
}

te_seabed_queued_take_distant = {
	if = {
		limit = {
			local_var:te_sb_qn > 0
			NOT = { var:te_seabed_basin = $N$ }
			var:te_seabed_lic_$B$ > 0
		}
		te_seabed_queued_take = { B = $B$ }
	}
}

te_seabed_queued_take_own = {
	if = {
		limit = {
			local_var:te_sb_qn > 0
			var:te_seabed_basin = $N$
			var:te_seabed_lic_$B$ > 0
		}
		te_seabed_queued_take = { B = $B$ }
	}
}

te_seabed_queued_take = {
	set_local_variable = {
		name = te_sb_qk
		value = {
			value = var:te_seabed_lic_$B$
			max = local_var:te_sb_qn
		}
	}
	change_local_variable = { name = te_sb_q_$B$ add = local_var:te_sb_qk }
	change_local_variable = { name = te_sb_qn subtract = local_var:te_sb_qk }
}

# ---- Returns (model: state_return_excess) ----

# State scope. Licences above the state's levels (built or queued) go back to
# its owner's shares: distant basins first, in basin order, then its own.
te_seabed_state_return_excess = {
	if = {
		limit = { te_seabed_state_excess > 0 }
		set_local_variable = { name = te_sb_x value = te_seabed_state_excess }
		if = {
			limit = { var:te_seabed_basin = 1 }
			te_seabed_return_into = { OWN = pacific A = atlantic B = indian C = arctic D = southern }
		}
		else_if = {
			limit = { var:te_seabed_basin = 2 }
			te_seabed_return_into = { OWN = atlantic A = pacific B = indian C = arctic D = southern }
		}
		else_if = {
			limit = { var:te_seabed_basin = 3 }
			te_seabed_return_into = { OWN = indian A = pacific B = atlantic C = arctic D = southern }
		}
		else_if = {
			limit = { var:te_seabed_basin = 4 }
			te_seabed_return_into = { OWN = arctic A = pacific B = atlantic C = indian D = southern }
		}
		else_if = {
			limit = { var:te_seabed_basin = 5 }
			te_seabed_return_into = { OWN = southern A = pacific B = atlantic C = indian D = arctic }
		}
	}
}

# State scope. $A$..$D$ are the four other basins, in basin order.
te_seabed_return_into = {
	te_seabed_return_from = { B = $A$ }
	te_seabed_return_from = { B = $B$ }
	te_seabed_return_from = { B = $C$ }
	te_seabed_return_from = { B = $D$ }
	te_seabed_return_from = { B = $OWN$ }
}

# State scope: give back up to local_var:te_sb_x licences of basin $B$.
te_seabed_return_from = {
	if = {
		limit = {
			local_var:te_sb_x > 0
			var:te_seabed_lic_$B$ > 0
		}
		set_local_variable = {
			name = te_sb_k
			value = {
				value = local_var:te_sb_x
				max = var:te_seabed_lic_$B$
			}
		}
		change_variable = { name = te_seabed_lic_$B$ subtract = local_var:te_sb_k }
		owner = {
			change_variable = { name = te_seabed_unbuilt_$B$ add = local_var:te_sb_k }
		}
		change_local_variable = { name = te_sb_x subtract = local_var:te_sb_k }
	}
}

# ---- New levels (model: state_record_new) ----

# State scope. Each level (built or queued) without a record takes a licence
# from scope:te_seabed_payer's unbuilt share: the state's own basin first,
# then the basin with the most. What the payer can't cover waits unrecorded
# when the caller set local te_sb_pend = 1 (the payer is the state's own
# stamped owner, or a holder that exists); otherwise it is recorded in the
# state's own basin against no share (grandfathered, § 2.4) and counted in
# te_seabed_noshare_<basin>. Model: state_record_new's `pend`.
# The payer is the owner in te_seabed_record_country, the stamped holder in
# te_seabed_on_state_owner_change, and absent when the holder is gone.
te_seabed_state_record_new = {
	if = {
		limit = { te_seabed_state_unrecorded > 0 }
		set_local_variable = { name = te_sb_n value = te_seabed_state_unrecorded }
		if = {
			limit = { var:te_seabed_basin = 1 }
			te_seabed_record_into = { OWN = pacific }
		}
		else_if = {
			limit = { var:te_seabed_basin = 2 }
			te_seabed_record_into = { OWN = atlantic }
		}
		else_if = {
			limit = { var:te_seabed_basin = 3 }
			te_seabed_record_into = { OWN = indian }
		}
		else_if = {
			limit = { var:te_seabed_basin = 4 }
			te_seabed_record_into = { OWN = arctic }
		}
		else_if = {
			limit = { var:te_seabed_basin = 5 }
			te_seabed_record_into = { OWN = southern }
		}
	}
}

te_seabed_record_into = {
	if = {
		limit = {
			exists = scope:te_seabed_payer
			# a stamped holder that was never recorded has no shares to read
			scope:te_seabed_payer = { has_variable = te_seabed_unbuilt_pacific }
		}
		te_seabed_take = { B = $OWN$ }
		# The basin with the most, until the levels are covered or the shares
		# are spent: at most four rounds.
		while = {
			limit = {
				local_var:te_sb_n > 0
				scope:te_seabed_payer = { te_seabed_unbuilt_total > 0 }
			}
			te_seabed_take_most = yes
		}
	}
	if = {
		limit = {
			local_var:te_sb_n > 0
			NOT = { local_var:te_sb_pend = 1 }
		}
		change_variable = { name = te_seabed_lic_$OWN$ add = local_var:te_sb_n }
		if = {
			limit = { NOT = { has_global_variable = te_seabed_noshare_$OWN$ } }
			set_global_variable = { name = te_seabed_noshare_$OWN$ value = 0 }
		}
		change_global_variable = { name = te_seabed_noshare_$OWN$ add = local_var:te_sb_n }
		# Model: stamped_no_share. Never legitimate (seabed_cap's BROKEN): a
		# monthly recording grandfathering a state that carries a stamp.
		if = {
			limit = {
				local_var:te_sb_monthly = 1
				has_variable = te_seabed_holder
			}
			if = {
				limit = { NOT = { has_global_variable = te_seabed_cap_stamped_noshare } }
				set_global_variable = { name = te_seabed_cap_stamped_noshare value = 0 }
			}
			change_global_variable = { name = te_seabed_cap_stamped_noshare add = local_var:te_sb_n }
		}
		set_local_variable = { name = te_sb_n value = 0 }
	}
}

# State scope: take up to local_var:te_sb_n licences of basin $B$ from the payer.
te_seabed_take = {
	if = {
		limit = {
			local_var:te_sb_n > 0
			scope:te_seabed_payer = { var:te_seabed_unbuilt_$B$ > 0 }
		}
		scope:te_seabed_payer = {
			set_local_variable = { name = te_sb_u value = var:te_seabed_unbuilt_$B$ }
		}
		set_local_variable = {
			name = te_sb_k
			value = {
				value = local_var:te_sb_n
				max = local_var:te_sb_u
			}
		}
		change_variable = { name = te_seabed_lic_$B$ add = local_var:te_sb_k }
		scope:te_seabed_payer = {
			change_variable = { name = te_seabed_unbuilt_$B$ subtract = local_var:te_sb_k }
		}
		change_local_variable = { name = te_sb_n subtract = local_var:te_sb_k }
	}
}

# State scope. Model: _argmax_unbuilt, the first in basin order on a tie.
te_seabed_take_most = {
	if = {
		limit = {
			scope:te_seabed_payer = {
				var:te_seabed_unbuilt_pacific > 0
				var:te_seabed_unbuilt_pacific >= te_seabed_unbuilt_max
			}
		}
		te_seabed_take = { B = pacific }
	}
	else_if = {
		limit = {
			scope:te_seabed_payer = {
				var:te_seabed_unbuilt_atlantic > 0
				var:te_seabed_unbuilt_atlantic >= te_seabed_unbuilt_max
			}
		}
		te_seabed_take = { B = atlantic }
	}
	else_if = {
		limit = {
			scope:te_seabed_payer = {
				var:te_seabed_unbuilt_indian > 0
				var:te_seabed_unbuilt_indian >= te_seabed_unbuilt_max
			}
		}
		te_seabed_take = { B = indian }
	}
	else_if = {
		limit = {
			scope:te_seabed_payer = {
				var:te_seabed_unbuilt_arctic > 0
				var:te_seabed_unbuilt_arctic >= te_seabed_unbuilt_max
			}
		}
		te_seabed_take = { B = arctic }
	}
	else_if = {
		limit = {
			scope:te_seabed_payer = { var:te_seabed_unbuilt_southern > 0 }
		}
		te_seabed_take = { B = southern }
	}
	else = {
		# nothing left to take: end the caller's loop
		set_local_variable = { name = te_sb_n value = 0 }
	}
}

# State scope, ROOT = the state under its new owner (on_state_owner_change
# gives no scope for the old one). Levels not yet recorded are recorded
# against the stamped holder's shares; a holder that no longer holds a state
# (annexed whole) pays nothing and they go against no share. Then the records
# stay on the state: the new owner holds them, and its own shares are
# untouched. The stamp names the new owner.
te_seabed_on_state_owner_change = {
	# First, before anything here writes: does the state still carry its
	# records? (self-check seabed_licence_follows_state)
	te_seabed_follow_check_hook = yes
	te_seabed_ensure_state_basin = yes
	if = {
		limit = {
			has_variable = te_seabed_basin
			te_seabed_state_has_data = yes
		}
		te_seabed_init_state = yes
		if = {
			limit = { exists = scope:te_seabed_payer }
			clear_saved_scope = te_seabed_payer
		}
		var:te_seabed_holder ?= {
			if = {
				limit = { any_scope_state = { always = yes } }
				save_scope_as = te_seabed_payer
			}
		}
		# A holder that exists but can't pay leaves the level waiting, now for
		# the new owner's share; no stamp, or a holder gone, grandfathers it.
		set_local_variable = { name = te_sb_monthly value = 0 }
		set_local_variable = { name = te_sb_pend value = 0 }
		if = {
			limit = { exists = scope:te_seabed_payer }
			set_local_variable = { name = te_sb_pend value = 1 }
		}
		te_seabed_state_record_new = yes
		owner = { te_seabed_init_country = yes }
	}
	owner = { save_scope_as = te_seabed_new_owner }
	set_variable = { name = te_seabed_holder value = scope:te_seabed_new_owner }
}

# No scope (on_monthly_pulse). The rush event, once a game; the sharing when
# due (te_seabed_pass_due); the second half of the follow-state self-check.
te_seabed_global_monthly = {
	# The rule switched off after the system ran, then on again: levels built
	# while it was off take a licence like an old save's (§ 2.9), so the
	# stamps go and the next recording grandfathers them instead of leaving
	# them waiting as levels queued past room.
	if = {
		limit = {
			NOT = { te_global_commons_enabled = yes }
			has_global_variable = te_seabed_pass_year
			NOT = { has_global_variable = te_seabed_paused }
		}
		set_global_variable = te_seabed_paused
	}
	else_if = {
		limit = {
			te_global_commons_enabled = yes
			has_global_variable = te_seabed_paused
		}
		remove_global_variable = te_seabed_paused
		every_country = {
			limit = { has_variable = te_seabed_country }
			every_scope_state = {
				limit = { has_variable = te_seabed_holder }
				remove_variable = te_seabed_holder
			}
		}
	}
	# The rush, announced (seabed claims spec § 7; an event, owner 2026-10-17),
	# once a game: the first month a country can claim, before its first
	# sharing runs (which sets te_seabed_pass_year this same month), to every
	# player (the event is news, with nothing for an AI to choose). A game
	# already past its first sharing, an old save, sets the flag without it.
	# The first claimant is also kept in a global variable, which te_seabed.3
	# reads back. With the rule on, the research's headline stands down
	# (common/on_actions/headlines.txt).
	if = {
		limit = {
			te_global_commons_enabled = yes
			NOT = { has_global_variable = te_seabed_rush_announced }
			any_country = { te_seabed_is_claimant = yes }
		}
		set_global_variable = te_seabed_rush_announced
		if = {
			limit = { NOT = { has_global_variable = te_seabed_pass_year } }
			random_country = {
				limit = { te_seabed_is_claimant = yes }
				save_scope_as = te_seabed_first_claimant
			}
			set_global_variable = { name = te_seabed_first_claimant value = scope:te_seabed_first_claimant }
			every_country = {
				limit = { is_ai = no }
				trigger_event = { id = te_seabed.3 popup = yes }
			}
		}
	}
	if = {
		limit = {
			te_global_commons_enabled = yes
			te_seabed_pass_due = yes
		}
		te_seabed_yearly_pass = yes
	}
	te_seabed_follow_check_month = yes
}

# No scope. Model: yearly_pass. The order is the spec's: state basins, the
# regime (the UN hook), then each participant's recording, held licences and
# the leases' shift, Sponsor's floors at the year's first sharing, weights,
# then the sharing of each pool, each after its floors are fitted, then the
# leases' settling.
te_seabed_yearly_pass = {
	if = {
		limit = { te_seabed_first_sharing_of_year = yes }
		set_global_variable = te_seabed_first_of_year
	}
	set_global_variable = { name = te_seabed_pass_year value = te_history_current_year }
	if = {
		limit = { has_global_variable = te_seabed_reshare }
		remove_global_variable = te_seabed_reshare
	}
	# 1. State basins on coastal states without one.
	every_country = {
		limit = { te_seabed_takes_part = yes }
		every_scope_state = {
			limit = { is_coastal = yes }
			te_seabed_ensure_state_basin = yes
		}
	}
	# 2. The regime.
	te_seabed_un_update_regime = yes
	# 3. Every participant records, so every level is held.
	every_country = {
		limit = { te_seabed_takes_part = yes }
		set_variable = te_seabed_part
		te_seabed_record_country = yes
	}
	te_seabed_cap_check = yes
	# 3b. The holding fee's snapshots, at the year's first sharing.
	if = {
		limit = { has_global_variable = te_seabed_first_of_year }
		every_country = {
			limit = { has_variable = te_seabed_part }
			every_scope_state = {
				limit = {
					is_coastal = yes
					has_variable = te_seabed_lic_pacific
				}
				te_seabed_take_snapshots = yes
			}
		}
	}
	# 4. Held licences, then the leases' shift: the sharing counts a lease's
	# licences as its lessor's (te_seabed_sh_<basin>).
	every_country = {
		limit = { has_variable = te_seabed_part }
		te_seabed_compute_held = yes
	}
	te_seabed_lease_shift_held = yes
	# 5. The floors (the year's first sharing only; a mid-year one honours
	# January's) and weights.
	every_country = {
		limit = { has_variable = te_seabed_part }
		if = {
			limit = { has_global_variable = te_seabed_first_of_year }
			te_seabed_set_floors = yes
		}
		if = {
			limit = { te_seabed_is_claimant = yes }
			te_seabed_compute_weights = yes
		}
		else = {
			te_seabed_zero_weights = yes
		}
	}
	# 6. Each pool, its floors fitted first.
	te_seabed_fit_floors = { B = pacific }
	te_seabed_share_basin = { B = pacific }
	te_seabed_fit_floors = { B = atlantic }
	te_seabed_share_basin = { B = atlantic }
	te_seabed_fit_floors = { B = indian }
	te_seabed_share_basin = { B = indian }
	te_seabed_fit_floors = { B = arctic }
	te_seabed_share_basin = { B = arctic }
	te_seabed_fit_floors = { B = southern }
	te_seabed_share_basin = { B = southern }
	# 7. A country that takes no part loses any share it had (it lost its
	# coast, or its records): its unbuilt share would otherwise stand beside
	# the pool just shared.
	every_country = {
		limit = {
			has_variable = te_seabed_shared
			NOT = { has_variable = te_seabed_part }
		}
		te_seabed_clear_shares = yes
	}
	every_country = {
		limit = { has_variable = te_seabed_part }
		set_variable = te_seabed_shared
		remove_variable = te_seabed_part
	}
	# 7b. Leases: those no longer standing go, the rest are topped up; then
	# subjects under Administer Seabed Claims pass their claims again.
	te_seabed_settle_leases = yes
	te_seabed_pass_subject_claims = yes
	# 7c. The rush's newcomers: a country this sharing first gave a share in a
	# basin, told to the countries already there (seabed claims spec § 7).
	te_seabed_claimant_notices = yes
	# 8. A policy that holds influence for a country that can no longer claim.
	every_country = {
		limit = {
			has_variable = te_seabed_policy
			NOT = { te_seabed_policy_survey = yes }
			NOT = { te_seabed_is_claimant = yes }
		}
		te_seabed_set_policy = { POLICY = 1 }
		# A forced reset is no choice: it leaves the country free to choose.
		remove_variable = te_seabed_policy_changed
	}
	# 9. Unilateral claimants sharing a basin: once a year, on the first
	# sharing of the year, not on a new claimant's mid-year one (review M2).
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = te_seabed_relations_year }
				AND = {
					has_global_variable = te_seabed_relations_year
					global_var:te_seabed_relations_year < te_history_current_year
				}
			}
		}
		set_global_variable = { name = te_seabed_relations_year value = te_history_current_year }
		te_seabed_unilateral_relations = yes
	}
	# 10. The world's figures for the panels and the next year's cap check.
	te_seabed_world_totals = { B = pacific }
	te_seabed_world_totals = { B = atlantic }
	te_seabed_world_totals = { B = indian }
	te_seabed_world_totals = { B = arctic }
	te_seabed_world_totals = { B = southern }
	# 11. The panel's rivals and its "as of".
	te_seabed_rank_basin = { B = pacific }
	te_seabed_rank_basin = { B = atlantic }
	te_seabed_rank_basin = { B = indian }
	te_seabed_rank_basin = { B = arctic }
	te_seabed_rank_basin = { B = southern }
	set_global_variable = { name = te_seabed_last_sharing_month value = te_history_month_index }
	te_seabed_floor_check = yes
	te_seabed_power_projection_check = yes
	if = {
		limit = { has_global_variable = te_seabed_first_of_year }
		remove_global_variable = te_seabed_first_of_year
	}
}

# Country scope. Model: compute_weights (§ 2.3; seabed claims spec § 5.1):
# every coastal state counts, an unincorporated one at te_seabed_uninc_coast
# in the coast term and the navy's spread; its ports count in full.
te_seabed_compute_weights = {
	set_local_variable = { name = te_sb_wn_all value = 0 }
	te_seabed_weight_reset = { B = pacific }
	te_seabed_weight_reset = { B = atlantic }
	te_seabed_weight_reset = { B = indian }
	te_seabed_weight_reset = { B = arctic }
	te_seabed_weight_reset = { B = southern }
	every_scope_state = {
		limit = {
			is_coastal = yes
			has_variable = te_seabed_basin
		}
		set_local_variable = { name = te_sb_wc value = 1 }
		if = {
			limit = { is_incorporated = no }
			set_local_variable = { name = te_sb_wc value = te_seabed_uninc_coast }
		}
		change_local_variable = { name = te_sb_wn_all add = local_var:te_sb_wc }
		te_seabed_weight_tally = { B = pacific N = 1 }
		te_seabed_weight_tally = { B = atlantic N = 2 }
		te_seabed_weight_tally = { B = indian N = 3 }
		te_seabed_weight_tally = { B = arctic N = 4 }
		te_seabed_weight_tally = { B = southern N = 5 }
	}
	set_variable = { name = te_seabed_coast_all value = local_var:te_sb_wn_all }
	set_local_variable = { name = te_sb_navy value = te_seabed_navy_effective }
	set_local_variable = { name = te_sb_factor value = te_seabed_policy_factor }
	te_seabed_weight_set = { B = pacific }
	te_seabed_weight_set = { B = atlantic }
	te_seabed_weight_set = { B = indian }
	te_seabed_weight_set = { B = arctic }
	te_seabed_weight_set = { B = southern }
	if = {
		limit = {
			has_variable = te_antarctic_station
			NOT = { has_variable = te_seabed_relinq_southern }
		}
		change_variable = { name = te_seabed_w_southern add = te_seabed_station_weight }
	}
	if = {
		limit = { te_seabed_southern_barred = yes }
		set_variable = { name = te_seabed_w_southern value = 0 }
	}
}

te_seabed_weight_reset = {
	set_local_variable = { name = te_sb_wn_$B$ value = 0 }

}

# State scope (inside te_seabed_compute_weights' walk): the state's coast
# count (local te_sb_wc, half if unincorporated), its port levels in full, and
# the coastal states the Free figure's hover lists (review I4: a level in an
# unincorporated one works at full throughput too).
te_seabed_weight_tally = {
	if = {
		limit = { var:te_seabed_basin = $N$ }
		change_local_variable = { name = te_sb_wn_$B$ add = local_var:te_sb_wc }
		change_local_variable = { name = te_sb_wp_$B$ add = te_seabed_state_port_levels }
		change_local_variable = { name = te_sb_ws_$B$ add = 1 }
	}
}

# Country scope. Home: (2n + ports + navy × n ÷ N) × factor; distant:
# 0.05 × navy × factor; 0 where the country relinquished its claims (the home
# flag stays, so the coast still counts as home).
te_seabed_weight_set = {
	set_variable = { name = te_seabed_coast_$B$ value = local_var:te_sb_wn_$B$ }
	set_variable = { name = te_seabed_coast_states_$B$ value = local_var:te_sb_ws_$B$ }
	set_variable = { name = te_seabed_ports_$B$ value = local_var:te_sb_wp_$B$ }
	if = {
		limit = { local_var:te_sb_wn_$B$ > 0 }
		set_variable = te_seabed_home_$B$
		set_variable = {
			name = te_seabed_w_$B$
			value = {
				value = local_var:te_sb_navy
				multiply = local_var:te_sb_wn_$B$
				divide = local_var:te_sb_wn_all
				add = local_var:te_sb_wp_$B$
				add = {
					value = local_var:te_sb_wn_$B$
					multiply = te_seabed_coast_weight
				}
				multiply = local_var:te_sb_factor
			}
		}
	}
	else = {
		if = {
			limit = { has_variable = te_seabed_home_$B$ }
			remove_variable = te_seabed_home_$B$
		}
		set_variable = {
			name = te_seabed_w_$B$
			value = {
				value = local_var:te_sb_navy
				multiply = te_seabed_distant_navy_share
				multiply = local_var:te_sb_factor
			}
		}
	}
	if = {
		limit = { has_variable = te_seabed_relinq_$B$ }
		set_variable = { name = te_seabed_w_$B$ value = 0 }
	}
}

# ---- The water-filling (model: share_basin) ----

# No scope. Water-fill pool $B$ (§ 2.4 step 2; § 3.2). Participants are the
# countries marked te_seabed_part with weight or holdings here (te_seabed_sh_$B$,
# after the leases' shift). Under the
# Regulated regime the pool is a reserved half (eligible parties only) and an
# open half (everyone), filled together: a country's tentative entitlement is
# its share of each half it claims in, and it is frozen when its holdings
# reach the sum. A frozen country's holdings come off the halves in proportion
# to its two tentative shares (the model's header says why). Repeats until no
# country is newly frozen: at most one pass per participant (a real while,
# as decolonization.txt's flood-fill). Then whole licences, rounded down, and
# the leftovers by largest remainder, so the pool is fully shared.
te_seabed_share_basin = {
	every_country = {
		limit = {
			has_variable = te_seabed_part
			OR = {
				var:te_seabed_w_$B$ > 0
				var:te_seabed_sh_$B$ > 0
			}
		}
		set_variable = te_seabed_act_$B$
		if = {
			limit = {
				te_seabed_regime_regulated = yes
				te_seabed_reserved_half_eligible = yes
			}
			set_variable = te_seabed_rsv_$B$
		}
		else_if = {
			limit = { has_variable = te_seabed_rsv_$B$ }
			remove_variable = te_seabed_rsv_$B$
		}
	}
	if = {
		limit = { te_seabed_regime_regulated = yes }
		set_global_variable = { name = te_seabed_rem_res_$B$ value = te_seabed_pool_reserved_$B$ }
	}
	else = {
		set_global_variable = { name = te_seabed_rem_res_$B$ value = 0 }
	}
	set_global_variable = {
		name = te_seabed_rem_open_$B$
		value = {
			value = te_seabed_pool_$B$
			subtract = global_var:te_seabed_rem_res_$B$
		}
	}
	set_global_variable = { name = te_seabed_frozen_$B$ value = 0 }
	set_global_variable = { name = te_seabed_passes_$B$ value = 0 }
	set_global_variable = { name = te_seabed_newly_$B$ value = 1 }
	while = {
		limit = { global_var:te_seabed_newly_$B$ > 0 }
		change_global_variable = { name = te_seabed_passes_$B$ add = 1 }
		te_seabed_share_fold_unclaimed = { B = $B$ }
		te_seabed_share_tentative = { B = $B$ }
		set_global_variable = { name = te_seabed_newly_$B$ value = 0 }
		every_country = {
			limit = {
				has_variable = te_seabed_act_$B$
				var:te_seabed_fz_$B$ >= var:te_seabed_e_$B$
			}
			te_seabed_share_freeze = { B = $B$ }
		}
		te_seabed_share_rebalance = { B = $B$ }
	}
	te_seabed_share_round = { B = $B$ }
}

# Slot $K$ (position $P$) only when the basin has at least $K$ such states: a
# position past the list's end clamps to its last state, which then filled
# every slot after it, and logged "Given position was bigger than the list,
# capping at list size" on each hover, check_range_bounds = no or not
# (scripting_best_practices.md, ordered_scope_state clamping; 2026-10-17).
te_seabed_build_in_slot = {
	if = {
		limit = {
			any_scope_state = {
				is_coastal = yes
				has_variable = te_seabed_basin
				var:te_seabed_basin = $N$
				count >= $K$
			}
		}

			limit = {
				is_coastal = yes
				has_variable = te_seabed_basin
				var:te_seabed_basin = $N$
			}
			order_by = te_seabed_state_port_levels
			position = $P$
			check_range_bounds = no
			save_temporary_scope_as = te_sb_build_$K$
		}

		limit = { exists = scope:te_sb_build_$K$ }
		custom_tooltip_no_bullet = te_seabed_build_in_state_$K$
	}
}

# ---- Sponsor Contractors' floors (seabed claims spec § 2.2) ----

# Country scope (the yearly pass, the year's first sharing, after held).
# Model: set_floors. On Sponsor Contractors, outside a Scramble, a claimant's
# floor in each basin is its holdings plus te_seabed_floor_keep of its claims
# going into the sharing, rounded down, and te_seabed_fprot_<basin> that
# protected part alone; its Southern floor is its holdings where the Antarctic
# Treaty bars it. Otherwise no floor.
te_seabed_set_floors = {
	if = {
		limit = {
			te_seabed_policy_sponsor = yes
			NOT = { te_seabed_regime_scramble = yes }
			te_seabed_is_claimant = yes
		}
		te_seabed_set_floor = { B = pacific }
		te_seabed_set_floor = { B = atlantic }
		te_seabed_set_floor = { B = indian }
		te_seabed_set_floor = { B = arctic }
		te_seabed_set_floor = { B = southern }
		if = {
			limit = { te_seabed_southern_barred = yes }
			set_variable = { name = te_seabed_floor_southern value = var:te_seabed_sh_southern }
			set_variable = { name = te_seabed_fprot_southern value = 0 }
		}
	}
	else = {
		te_seabed_clear_floor = { B = pacific }
		te_seabed_clear_floor = { B = atlantic }
		te_seabed_clear_floor = { B = indian }
		te_seabed_clear_floor = { B = arctic }
		te_seabed_clear_floor = { B = southern }
	}
}

# Own claims only: no floor protects leased or subject-passed claims (spec
# § 2.2). Holdings as the sharing counts them, after the leases' shift.
te_seabed_set_floor = {
	set_variable = {
		name = te_seabed_fprot_$B$
		value = {
			value = te_seabed_own_claims_$B$
			multiply = te_seabed_floor_keep
			floor = yes
		}
	}
	set_variable = {
		name = te_seabed_floor_$B$
		value = {
			value = var:te_seabed_fprot_$B$
			add = var:te_seabed_sh_$B$
		}
	}
}

te_seabed_clear_floor = {
	set_variable = { name = te_seabed_floor_$B$ value = 0 }
	set_variable = { name = te_seabed_fprot_$B$ value = 0 }
}

# Country scope (the basin's Relinquish button). Model: relinquish (seabed
# claims spec § 2.5). The claims in basin $B$ go at once, the floor with them,
# and the weight there is 0 at every sharing until Claim Again. Licences on
# its levels stay, those queued since the last recording included (it
# records first). Never sets te_seabed_reshare: a mid-year sharing honours
# January's floors, so a Sponsor country could otherwise relinquish one claim
# and take its rivals' unprotected claims months early. The freed claims wait
# for the next sharing something else brings about.
te_seabed_relinquish = {
	# Levels queued since the monthly recording take their licences first
	# (review I1: they'd be left without one once the claims are gone).
	te_seabed_record_country = yes
	set_variable = { name = te_seabed_unbuilt_$B$ value = 0 }
	set_variable = { name = te_seabed_floor_$B$ value = 0 }
	set_variable = { name = te_seabed_fprot_$B$ value = 0 }
	set_variable = te_seabed_relinq_$B$
	# Leased and subject-passed claims go too, without counting as built.

		limit = { has_variable = te_seabed_lease_unused_$B$ }
		set_variable = { name = te_seabed_lease_unused_$B$ value = 0 }

		limit = { has_variable = te_seabed_subj_unused_$B$ }
		set_variable = { name = te_seabed_subj_unused_$B$ value = 0 }
	}
	save_scope_as = te_sb_cons
	every_country = {
		limit = {
			has_variable = te_seabed_pass_unused_$B$
			var:te_seabed_pass_to ?= { this = scope:te_sb_cons }
		}
		set_variable = { name = te_seabed_pass_unused_$B$ value = 0 }
	}
}

# A Lease <Basin> Seabed Claims article (122_seabed_lease.txt) leases N of
# the lessor's own claims in a basin to the lessee, N being the lease's total:
# the licences built on it plus its unused claims. The lessee keeps it in
# te_seabed_lease_src_<basin> (the lessor), _n_, _built_ and _unused_<basin>;
# the unused claims are part of its te_seabed_unbuilt_<basin>. The treaty is
# the record: each sharing reads the in-force article again
# (te_seabed_lease_shift_basin), so a lease ends there whatever ended its
# article, annexation included. An ended lease's licences go on charging its
# lessor (te_seabed_lease_past_<basin>) while both take part, so ending and
# re-signing one never moves their cost onto a third party (review C2); the
# record then admits a new lease from the same lessor only.

# No scope: an article's on_entry_into_force, the parties saved as
# scope:te_sb_lessor and scope:te_sb_lessee. The hook only marks the lessee
# and dispatches te_seabed.1 a day later, which reads the article from the
# lessee's treaties (scripting_best_practices.md: a treaty hook dispatches a
# country event, a day late when it walks treaties).
te_seabed_lease_start = {
	hidden_effect = {
		scope:te_sb_lessee = {
			set_variable = te_seabed_lease_due_$B$
			trigger_event = { id = te_seabed.1 days = 1 }
		}
	}
}

# Country scope, the lessee. Model: start_lease. A lease due in basin $B$
# starts from its in-force article: with no record, any lessor's; with one,
# only the recorded lessor's (te_seabed_can_lease: an ended lease's licences go
# on charging that lessor), and only if that lease isn't running already. The
# lessor and N (the article's quantity, at least 1) are stored and up to N of
# the lessor's own claims there move at once, so the lessee can build before
# the next sharing; an ended lease's past licences carry on. The due flag goes
# either way: the sharing's walk takes up what this misses (review I1). Local
# te_sb_le: 1 taken up, 2 nothing could be read; te_seabed_lease_take_up logs
# it.
te_seabed_lease_enter = {
	if = {
		limit = { has_variable = te_seabed_lease_due_$B$ }
		remove_variable = te_seabed_lease_due_$B$
		save_scope_as = te_sb_lessee
		if = {
			limit = { exists = scope:te_sb_found }
			clear_saved_scope = te_sb_found
		}
		if = {
			limit = { exists = scope:te_sb_lessor }
			clear_saved_scope = te_sb_lessor
		}
		var:te_seabed_lease_src_$B$ ?= { save_scope_as = te_sb_lessor }
		if = {
			limit = { exists = scope:te_sb_lessor }
			every_scope_treaty = {
				every_scope_article = {
					limit = {
						has_type = te_seabed_lease_$B$
						source_country = scope:te_sb_lessor
						target_country = scope:te_sb_lessee
						exists = input_quantity
					}
					save_scope_as = te_sb_found
				}
			}
		}
		else = {
			every_scope_treaty = {
				every_scope_article = {
					limit = {
						has_type = te_seabed_lease_$B$
						target_country = scope:te_sb_lessee
						exists = input_quantity
					}
					save_scope_as = te_sb_found
				}
			}
			if = {
				limit = { exists = scope:te_sb_found }
				scope:te_sb_found = {
					source_country = { save_scope_as = te_sb_lessor }
				}
			}
		}
		if = {
			limit = {
				exists = scope:te_sb_found
				scope:te_sb_found = { input_quantity > 0 }
			}
			set_local_variable = { name = te_sb_le value = 1 }
			# No lease running here: no record, or one an ended lease left at 0
			# (te_seabed_settle_lease keeps it while licences built on it count). trigger_if, not a two-child NOT: a NOT read the var with
			# has_variable false (2026-10-18 log), and as NOR it would turn the
			# kept 0 away (scripting_best_practices.md, multi-child NOT).
			if = {
				limit = {
					trigger_if = {
						limit = { has_variable = te_seabed_lease_n_$B$ }
						NOT = { var:te_seabed_lease_n_$B$ > 0 }
					}
				}
				te_seabed_init_country = yes
				scope:te_sb_lessor = { te_seabed_init_country = yes }
				set_variable = { name = te_seabed_lease_src_$B$ value = scope:te_sb_lessor }
				set_variable = { name = te_seabed_lease_n_$B$ value = scope:te_sb_found.input_quantity }
				set_variable = { name = te_seabed_lease_built_$B$ value = 0 }
				if = {
					limit = { NOT = { has_variable = te_seabed_lease_past_$B$ } }
					set_variable = { name = te_seabed_lease_past_$B$ value = 0 }
				}
				set_local_variable = {
					name = te_sb_lk
					value = {
						value = var:te_seabed_lease_n_$B$
						max = scope:te_sb_lessor.te_seabed_own_claims_$B$
						min = 0
					}
				}
				scope:te_sb_lessor = {
					change_variable = { name = te_seabed_unbuilt_$B$ subtract = local_var:te_sb_lk }
					if = {
						limit = { NOT = { has_variable = te_seabed_leased_out_$B$ } }
						set_variable = { name = te_seabed_leased_out_$B$ value = 0 }
					}
					change_variable = { name = te_seabed_leased_out_$B$ add = local_var:te_sb_lk }
				}
				change_variable = { name = te_seabed_unbuilt_$B$ add = local_var:te_sb_lk }
				set_variable = { name = te_seabed_lease_unused_$B$ value = local_var:te_sb_lk }
			}
		}
		else = {
			set_local_variable = { name = te_sb_le value = 2 }
		}
	}
}

# Country scope (the end of te_seabed_record_country). Model: lease_consume.
# What the recording took comes off the country's own claims first, then off
# those its subjects passed it, then off its leased ones, which become
# licences built on the lease.
te_seabed_lease_consume = {

	te_seabed_lease_consume_basin = { B = pacific }
	te_seabed_lease_consume_basin = { B = atlantic }
	te_seabed_lease_consume_basin = { B = indian }
	te_seabed_lease_consume_basin = { B = arctic }
	te_seabed_lease_consume_basin = { B = southern }
}

te_seabed_lease_consume_basin = {
	if = {
		limit = {
			has_variable = te_seabed_lease_unused_$B$
			var:te_seabed_lease_unused_$B$ > var:te_seabed_unbuilt_$B$
		}
		set_local_variable = {
			name = te_sb_lc
			value = {
				value = var:te_seabed_unbuilt_$B$
				min = 0
			}
		}
		set_variable = {
			name = te_seabed_lease_built_$B$
			value = {
				value = var:te_seabed_lease_built_$B$
				add = var:te_seabed_lease_unused_$B$
				subtract = local_var:te_sb_lc
			}
		}
		set_variable = { name = te_seabed_lease_unused_$B$ value = local_var:te_sb_lc }
	}
	# Then claims its subjects passed it: what the recording took beyond its
	# own claims and before its leased ones, booked as built on their claims on
	# each country whose counters charge this one (te_seabed_pass_to; review
	# C2: a former subject's passed claims are still here until the sharing).
	if = {
		limit = {
			has_variable = te_seabed_subj_unused_$B$
			var:te_seabed_subj_unused_$B$ > 0
		}
		set_local_variable = {
			name = te_sb_sl
			value = {
				value = var:te_seabed_unbuilt_$B$
				min = 0
			}
		}
		if = {
			limit = { has_variable = te_seabed_lease_unused_$B$ }
			change_local_variable = { name = te_sb_sl subtract = var:te_seabed_lease_unused_$B$ }
		}
		if = {
			limit = { var:te_seabed_subj_unused_$B$ > local_var:te_sb_sl }
			set_local_variable = {
				name = te_sb_su
				value = {
					value = var:te_seabed_subj_unused_$B$
					subtract = local_var:te_sb_sl
				}
			}
			set_variable = {
				name = te_seabed_subj_unused_$B$
				value = {
					value = local_var:te_sb_sl
					min = 0
				}
			}
			every_country = {
				limit = {
					has_variable = te_seabed_pass_unused_$B$
					var:te_seabed_pass_unused_$B$ > 0
					var:te_seabed_pass_to ?= { this = scope:te_sb_cons }
				}
				if = {
					limit = { local_var:te_sb_su > 0 }
					set_local_variable = {
						name = te_sb_sk
						value = {
							value = local_var:te_sb_su
							max = var:te_seabed_pass_unused_$B$
						}
					}
					change_variable = { name = te_seabed_pass_unused_$B$ subtract = local_var:te_sb_sk }
					if = {
						limit = { NOT = { has_variable = te_seabed_pass_built_$B$ } }
						set_variable = { name = te_seabed_pass_built_$B$ value = 0 }
					}
					change_variable = { name = te_seabed_pass_built_$B$ add = local_var:te_sb_sk }
					change_local_variable = { name = te_sb_su subtract = local_var:te_sb_sk }
				}
			}
		}
	}
}

# No scope, the yearly pass after every participant's held. Model: share_held.
# Every participant's te_seabed_sh_<basin> starts at its held; then each
# standing lease's licences, at most the lessee's holdings, count as its
# lessor's instead, standing or ended (review C2); then each subject
# record's, at most what is left (te_seabed_shleft_<basin>), as the
# subject's.
te_seabed_lease_shift_held = {

		te_seabed_lease_shift_basin = { B = indian }

			te_seabed_pact_shift_basin = { B = southern }

# Country scope, a participant. Model: lease_stands, lease_charged,
# share_held. The treaty is the record: its lease in basin $B$ stands while an
# in-force article from its recorded lessor names it and both claim
# (te_seabed_lease_stands; flag te_seabed_lease_ok_$B$; N read again, as a
# renegotiation may change it). With no record, an in-force article naming it
# is taken up here (review I1: a start event that never ran, an enforced
# lease), its claims arriving with the top-up. The licences built on the
# lease and on ended ones from the same lessor, at most its holdings, count as
# the lessor's while the lessor takes part too (flag te_seabed_lease_chg_$B$;
# review C2).
te_seabed_lease_shift_basin = {
	if = {
		limit = {
			OR = {
				has_variable = te_seabed_lease_src_$B$
				any_scope_treaty = {
					any_scope_article = { has_type = te_seabed_lease_$B$ }
				}
			}
		}
		if = {
			limit = { has_variable = te_seabed_lease_ok_$B$ }
			remove_variable = te_seabed_lease_ok_$B$
		}
		if = {
			limit = { has_variable = te_seabed_lease_chg_$B$ }
			remove_variable = te_seabed_lease_chg_$B$
		}
		save_scope_as = te_sb_lessee
		if = {
			limit = { exists = scope:te_sb_lessor }
			clear_saved_scope = te_sb_lessor
		}
		if = {
			limit = { exists = scope:te_sb_found }
			clear_saved_scope = te_sb_found
		}
		var:te_seabed_lease_src_$B$ ?= { save_scope_as = te_sb_lessor }
		if = {
			limit = { exists = scope:te_sb_lessor }
			every_scope_treaty = {
				every_scope_article = {
					limit = {
						has_type = te_seabed_lease_$B$
						source_country = scope:te_sb_lessor
						target_country = scope:te_sb_lessee
						exists = input_quantity
					}
					save_scope_as = te_sb_found
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:te_sb_lessor }
				any_scope_treaty = {
					any_scope_article = { has_type = te_seabed_lease_$B$ }
				}
			}
			every_scope_treaty = {
				every_scope_article = {
					limit = {
						has_type = te_seabed_lease_$B$
						target_country = scope:te_sb_lessee
						exists = input_quantity
					}
					save_scope_as = te_sb_found
				}
			}
			if = {
				limit = { exists = scope:te_sb_found }
				scope:te_sb_found = {
					source_country = { save_scope_as = te_sb_lessor }
				}
				scope:te_sb_lessor = { te_seabed_init_country = yes }
				set_variable = { name = te_seabed_lease_src_$B$ value = scope:te_sb_lessor }
				set_variable = { name = te_seabed_lease_n_$B$ value = scope:te_sb_found.input_quantity }
				set_variable = { name = te_seabed_lease_built_$B$ value = 0 }
				set_variable = { name = te_seabed_lease_unused_$B$ value = 0 }
				set_variable = { name = te_seabed_lease_past_$B$ value = 0 }
			}
		}
		if = {
			limit = {
				exists = scope:te_sb_found
				te_seabed_lease_stands = { B = $B$ }
			}
			set_variable = te_seabed_lease_ok_$B$
			set_variable = { name = te_seabed_lease_n_$B$ value = scope:te_sb_found.input_quantity }
		}
		if = {
			limit = {
				exists = scope:te_sb_lessor
				scope:te_sb_lessor = { has_variable = te_seabed_part }
			}
			set_variable = te_seabed_lease_chg_$B$
			set_local_variable = {
				name = te_sb_ls
				value = {
					value = 0
					if = {
						limit = { has_variable = te_seabed_lease_built_$B$ }
						add = var:te_seabed_lease_built_$B$
					}
					if = {
						limit = { has_variable = te_seabed_lease_past_$B$ }
						add = var:te_seabed_lease_past_$B$
					}
					max = var:te_seabed_held_$B$
					min = 0
				}
			}
			change_variable = { name = te_seabed_sh_$B$ subtract = local_var:te_sb_ls }
			change_variable = { name = te_seabed_shleft_$B$ subtract = local_var:te_sb_ls }
			scope:te_sb_lessor = {
				change_variable = { name = te_seabed_sh_$B$ add = local_var:te_sb_ls }
				change_variable = { name = te_seabed_charged_$B$ add = local_var:te_sb_ls }
			}
		}
	}
}

# No scope, the yearly pass after the sharing and the stale shares. Model:

# Country scope. Model: settle_leases. A record the shift step found
# uncharged (te_seabed_lease_chg_$B$ unset: the lessor, or this country, no
# longer takes part) is cleared: its licences are the lessee's own from now
# on. Otherwise at most the lessee's holdings count as built on the lease and
# on ended ones. An ended lease (te_seabed_lease_ok_$B$ unset) folds its
# licences into te_seabed_lease_past_$B$, which goes on charging the lessor
# (review C2), and the record clears once that is nothing. A standing one,
# unless the lessee relinquished the basin, is topped back up to N less its
# licences from the lessor's own claims, as far as they go.
te_seabed_settle_lease = {
	if = {
		limit = { has_variable = te_seabed_lease_src_$B$ }
		if = {
			limit = { NOT = { has_variable = te_seabed_lease_chg_$B$ } }
			te_seabed_lease_clear = { B = $B$ }
		}
		else = {
			remove_variable = te_seabed_lease_chg_$B$
			if = {
				limit = { NOT = { has_variable = te_seabed_lease_past_$B$ } }
				set_variable = { name = te_seabed_lease_past_$B$ value = 0 }
			}
			set_variable = {
				name = te_seabed_lease_built_$B$
				value = {
					value = var:te_seabed_lease_built_$B$
					max = var:te_seabed_held_$B$
				}
			}
			set_variable = {
				name = te_seabed_lease_past_$B$
				value = {
					value = var:te_seabed_lease_past_$B$
					max = {
						value = var:te_seabed_held_$B$
						subtract = var:te_seabed_lease_built_$B$
					}
					min = 0
				}
			}
			if = {
				limit = { NOT = { has_variable = te_seabed_lease_ok_$B$ } }
				set_variable = {
					name = te_seabed_lease_past_$B$
					value = {
						value = var:te_seabed_lease_past_$B$
						add = var:te_seabed_lease_built_$B$
					}
				}
				set_variable = { name = te_seabed_lease_n_$B$ value = 0 }
				set_variable = { name = te_seabed_lease_built_$B$ value = 0 }
				set_variable = { name = te_seabed_lease_unused_$B$ value = 0 }
				if = {
					limit = { NOT = { var:te_seabed_lease_past_$B$ > 0 } }
					te_seabed_lease_clear = { B = $B$ }
				}
			}
			else = {
				remove_variable = te_seabed_lease_ok_$B$
				if = {
					limit = { NOT = { has_variable = te_seabed_relinq_$B$ } }
					var:te_seabed_lease_src_$B$ = { save_scope_as = te_sb_lessor }
					set_local_variable = {
						name = te_sb_lt
						value = {
							value = var:te_seabed_lease_n_$B$
							subtract = var:te_seabed_lease_built_$B$
							subtract = var:te_seabed_lease_unused_$B$
							max = scope:te_sb_lessor.te_seabed_own_claims_$B$
							min = 0
						}
					}
					scope:te_sb_lessor = {
						change_variable = { name = te_seabed_unbuilt_$B$ subtract = local_var:te_sb_lt }
					}
					change_variable = { name = te_seabed_unbuilt_$B$ add = local_var:te_sb_lt }
					change_variable = { name = te_seabed_lease_unused_$B$ add = local_var:te_sb_lt }
				}
				set_local_variable = { name = te_sb_lu value = var:te_seabed_lease_unused_$B$ }
				var:te_seabed_lease_src_$B$ = {
					change_variable = { name = te_seabed_leased_out_$B$ add = local_var:te_sb_lu }
				}
			}
		}
	}
}

# Country scope: the lease record in basin $B$ goes.
te_seabed_lease_clear = {
	remove_variable = te_seabed_lease_src_$B$
	if = {
		limit = { has_variable = te_seabed_lease_n_$B$ }
		remove_variable = te_seabed_lease_n_$B$
	}
	if = {
		limit = { has_variable = te_seabed_lease_built_$B$ }
		remove_variable = te_seabed_lease_built_$B$
	}
	if = {
		limit = { has_variable = te_seabed_lease_unused_$B$ }
		remove_variable = te_seabed_lease_unused_$B$
	}
	if = {
		limit = { has_variable = te_seabed_lease_past_$B$ }
		remove_variable = te_seabed_lease_past_$B$
	}
	if = {
		limit = { has_variable = te_seabed_lease_ok_$B$ }
		remove_variable = te_seabed_lease_ok_$B$
	}
}

# ============================================================================
# Subjects: Administer Seabed Claims (model: pact_stands, pass_claims,
# start_pact, pass_subject_claims; seabed claims spec § 5.2)
# ============================================================================
# An overlord's pact with a direct subject (te_seabed_administer_claims.txt)
# is a lease with no total that keeps the same two counters, on the subject:
# te_seabed_pass_unused_<basin> (its claims at the overlord, unused) and
# te_seabed_pass_built_<basin> (the licences the overlord built on them). The
# overlord's te_seabed_subj_unused_<basin> is their sum, part of its unbuilt.
# At the sharing the subject's pass_built counts as its holdings, not the
# overlord's (te_seabed_pact_shift_basin), so its weight pays for them once:
# without that the pair would take a third party's share year by year. The
# counters name the overlord they charge (te_seabed_pass_to) and go on
# charging it after the pact ends, while both take part, so ending and taking
# the pact again moves nothing onto a third party (review C2); only a pact with
# another overlord starts them afresh. The pact itself is the record of
# whether claims pass (te_seabed_pact_stands); when it ends the passed claims
# return through the next sharing.

# Country scope, the overlord, from the action's accept_effect with the
# subject saved as scope:te_sb_subject. Model: start_pact. The subject's
# counters charge this overlord (te_seabed_pact_charge_to: another's record
# is dropped, this one's carries on, review C2) and its own claims pass at
# once.
te_seabed_pact_start = {
	save_scope_as = te_sb_over
	te_seabed_init_country = yes
	scope:te_sb_subject = {
		te_seabed_init_country = yes
		te_seabed_pact_charge_to = yes
		te_seabed_pass_basin = { B = pacific }
		te_seabed_pass_basin = { B = atlantic }
		te_seabed_pass_basin = { B = indian }
		te_seabed_pass_basin = { B = arctic }
		te_seabed_pass_basin = { B = southern }
		# Self-check: the pact read from the subject's side a day later.
		hidden_effect = {
			trigger_event = { id = te_seabed.2 days = 1 }
		}
	}
}

# Country scope, a subject, its overlord saved as scope:te_sb_over. Model:
# _charge_to. Its counters charge scope:te_sb_over (te_seabed_pass_to): a
# record charging another overlord is dropped, its licences that overlord's
# own; this one's carries on.
te_seabed_pact_charge_to = {
	if = {
		limit = {
			NOT = {
				var:te_seabed_pass_to ?= { this = scope:te_sb_over }
			}
		}
		set_variable = { name = te_seabed_pass_built_pacific value = 0 }
		set_variable = { name = te_seabed_pass_built_atlantic value = 0 }
		set_variable = { name = te_seabed_pass_built_indian value = 0 }
		set_variable = { name = te_seabed_pass_built_arctic value = 0 }
		set_variable = { name = te_seabed_pass_built_southern value = 0 }
		set_variable = { name = te_seabed_pass_unused_pacific value = 0 }
		set_variable = { name = te_seabed_pass_unused_atlantic value = 0 }
		set_variable = { name = te_seabed_pass_unused_indian value = 0 }
		set_variable = { name = te_seabed_pass_unused_arctic value = 0 }
		set_variable = { name = te_seabed_pass_unused_southern value = 0 }
		set_variable = { name = te_seabed_pass_to value = scope:te_sb_over }
	}
}

# Country scope, a subject, its overlord saved as scope:te_sb_over. Model:
# pass_claims, one basin. Its own claims in basin $B$ pass to the overlord,
# where the pact's terms hold, unless the overlord relinquished the basin.
te_seabed_pass_basin = {
	if = {
		limit = {
			te_seabed_pact_ok = { B = $B$ }
			scope:te_sb_over = {
				NOT = { has_variable = te_seabed_relinq_$B$ }
			}
			te_seabed_own_claims_$B$ > 0
		}
		set_local_variable = { name = te_sb_pk value = te_seabed_own_claims_$B$ }
		change_variable = { name = te_seabed_unbuilt_$B$ subtract = local_var:te_sb_pk }
		if = {
			limit = { NOT = { has_variable = te_seabed_pass_unused_$B$ } }
			set_variable = { name = te_seabed_pass_unused_$B$ value = 0 }
		}
		change_variable = { name = te_seabed_pass_unused_$B$ add = local_var:te_sb_pk }
		scope:te_sb_over = {
			if = {
				limit = { NOT = { has_variable = te_seabed_subj_unused_$B$ } }
				set_variable = { name = te_seabed_subj_unused_$B$ value = 0 }
			}
			change_variable = { name = te_seabed_unbuilt_$B$ add = local_var:te_sb_pk }
			change_variable = { name = te_seabed_subj_unused_$B$ add = local_var:te_sb_pk }
		}
	}
}

# Country scope, a subject or former one whose overlord, the one its counters
# charge, is saved as scope:te_sb_over (te_seabed_lease_shift_held's third
# walk). Model: share_held's pact part. What that overlord built on this
# country's passed claims, at most what its lease left of its holdings, counts
# as this country's, pact or no pact (review C2).
te_seabed_pact_shift_basin = {
	if = {
		limit = {
			has_variable = te_seabed_pass_built_$B$
			var:te_seabed_pass_built_$B$ > 0
			scope:te_sb_over = { has_variable = te_seabed_shleft_$B$ }
		}
		set_local_variable = {
			name = te_sb_ps
			value = {
				value = var:te_seabed_pass_built_$B$
				max = scope:te_sb_over.var:te_seabed_shleft_$B$
				min = 0
			}
		}
		change_variable = { name = te_seabed_sh_$B$ add = local_var:te_sb_ps }
		change_variable = { name = te_seabed_charged_$B$ add = local_var:te_sb_ps }
		scope:te_sb_over = {
			change_variable = { name = te_seabed_sh_$B$ subtract = local_var:te_sb_ps }
			change_variable = { name = te_seabed_shleft_$B$ subtract = local_var:te_sb_ps }
		}
	}
}

# No scope, the yearly pass after te_seabed_settle_leases. Model:
# pass_subject_claims. A record the shift step found uncharged
# (te_seabed_pass_chg unset: the country or its te_seabed_pass_to overlord no
# longer takes part) is cleared, its licences that overlord's own; otherwise
# at most the overlord's holdings count as built on the country's claims, pact
# or no pact (review C2). Then claims pass where a pact stands.
te_seabed_pass_subject_claims = {
	every_country = {
		limit = { has_variable = te_seabed_country }
		save_scope_as = te_sb_subject
		if = {
			limit = { has_variable = te_seabed_pass_to }
			if = {
				limit = { NOT = { has_variable = te_seabed_pass_chg } }
				te_seabed_pact_clear = yes
			}
			else = {
				remove_variable = te_seabed_pass_chg
				if = {
					limit = { exists = scope:te_sb_paid }
					clear_saved_scope = te_sb_paid
				}
				var:te_seabed_pass_to = { save_scope_as = te_sb_paid }
				te_seabed_pact_clamp = { B = pacific }
				te_seabed_pact_clamp = { B = atlantic }
				te_seabed_pact_clamp = { B = indian }
				te_seabed_pact_clamp = { B = arctic }
				te_seabed_pact_clamp = { B = southern }
			}
		}
		if = {
			limit = { exists = scope:te_sb_over }
			clear_saved_scope = te_sb_over
		}
		overlord ?= { save_scope_as = te_sb_over }
		te_seabed_pact_settle = { B = pacific }
		te_seabed_pact_settle = { B = atlantic }
		te_seabed_pact_settle = { B = indian }
		te_seabed_pact_settle = { B = arctic }
		te_seabed_pact_settle = { B = southern }
	}
}

# Country scope, its record's overlord saved as scope:te_sb_paid: at most that
# overlord's holdings count as built on its claims.
te_seabed_pact_clamp = {
	if = {
		limit = { has_variable = te_seabed_pass_built_$B$ }
		set_variable = {
			name = te_seabed_pass_built_$B$
			value = {
				value = var:te_seabed_pass_built_$B$
				max = scope:te_sb_paid.var:te_seabed_held_$B$
			}
		}
	}
}

# Country scope, a recorded country, its overlord (if any) saved as
# scope:te_sb_over: where its pact stands, its counters charge that overlord
# and its claims pass again.
te_seabed_pact_settle = {
	if = {
		limit = { te_seabed_pact_stands = { B = $B$ } }
		te_seabed_pact_charge_to = yes
		te_seabed_pass_basin = { B = $B$ }
	}
}

# No scope, before te_seabed_share_basin. Model: fit_floors. Each
# participant's te_seabed_fz_$B$ is what the sharing freezes it at:
# max(held, min(floor, held + protected claims)), held being te_seabed_sh_$B$
# (after the leases' shift), the stored floor counting for
# nothing where te_seabed_floor_void_$B$ holds (a mid-year sharing honours
# January's figures without recomputing them; a licence lost since adds no
# protection, review I3). Every participant's is written first, 0 outside the
# basin's set: te_seabed_share_round gives a participant that isn't active its
# te_seabed_fz_$B$ as entitlement (review C1; the model's ent.get(cid, 0)).
# Where the protected parts (floor - held) exceed what the basin can share
# beyond its holdings (the pool less the holdings, at least 0), each is scaled
# down by one factor, rounding down; holdings are never touched.
te_seabed_fit_floors = {
	set_global_variable = { name = te_seabed_fz_held_$B$ value = 0 }
	set_global_variable = { name = te_seabed_fz_prot_$B$ value = 0 }
	every_country = {
		limit = { has_variable = te_seabed_part }
		set_variable = { name = te_seabed_fz_$B$ value = 0 }
	}
	every_country = {
		limit = {
			has_variable = te_seabed_part
			OR = {
				var:te_seabed_w_$B$ > 0
				var:te_seabed_sh_$B$ > 0
			}
		}
		if = {
			limit = {
				has_variable = te_seabed_floor_$B$
				has_variable = te_seabed_fprot_$B$
				NOT = { te_seabed_floor_void_$B$ = yes }
				var:te_seabed_floor_$B$ > var:te_seabed_sh_$B$
			}
			set_variable = {
				name = te_seabed_fz_$B$
				value = {
					value = var:te_seabed_sh_$B$
					add = var:te_seabed_fprot_$B$
					max = var:te_seabed_floor_$B$
					subtract = var:te_seabed_sh_$B$
					min = 0
				}
			}
		}
		change_global_variable = { name = te_seabed_fz_held_$B$ add = var:te_seabed_sh_$B$ }
		change_global_variable = { name = te_seabed_fz_prot_$B$ add = var:te_seabed_fz_$B$ }
	}
	set_local_variable = {
		name = te_sb_fz_room
		value = {
			value = te_seabed_pool_$B$
			subtract = global_var:te_seabed_fz_held_$B$
			min = 0
		}
	}
	if = {
		limit = { global_var:te_seabed_fz_prot_$B$ > 0 }
		set_global_variable = te_seabed_floor_applied
	}
	every_country = {
		limit = {
			has_variable = te_seabed_part
			OR = {
				var:te_seabed_w_$B$ > 0
				var:te_seabed_sh_$B$ > 0
			}
		}
		if = {
			limit = { global_var:te_seabed_fz_prot_$B$ > local_var:te_sb_fz_room }
			set_variable = {
				name = te_seabed_fz_$B$
				value = {
					value = var:te_seabed_fz_$B$
					multiply = local_var:te_sb_fz_room
					divide = global_var:te_seabed_fz_prot_$B$
					floor = yes
				}
			}
		}
		change_variable = { name = te_seabed_fz_$B$ add = var:te_seabed_sh_$B$ }
	}
}

# No scope, the yearly pass's last step. Self-check (seabed claims spec § 9):
# after a sharing in which a floor protected something, every basin's
# entitlements stay within its pool, or its holdings if more (the model's
# test_floors_scale_down_to_fit). OK once a game, BROKEN whenever it fails.
te_seabed_floor_check = {
	if = {
		limit = { has_global_variable = te_seabed_floor_applied }
		remove_global_variable = te_seabed_floor_applied
		set_global_variable = { name = te_seabed_floor_over value = 0 }
		te_seabed_floor_check_basin = { B = pacific }
		te_seabed_floor_check_basin = { B = atlantic }
		te_seabed_floor_check_basin = { B = indian }
		te_seabed_floor_check_basin = { B = arctic }
		te_seabed_floor_check_basin = { B = southern }
		if = {
			limit = { global_var:te_seabed_floor_over = 0 }
			if = {
				limit = { NOT = { has_global_variable = te_seabed_floor_ok_logged } }
				set_global_variable = te_seabed_floor_ok_logged
				debug_log = "TE_SELF_CHECK OK seabed_floor_fits protected pacific=[GetGlobalVariable('te_seabed_fz_prot_pacific').GetValue] atlantic=[GetGlobalVariable('te_seabed_fz_prot_atlantic').GetValue] indian=[GetGlobalVariable('te_seabed_fz_prot_indian').GetValue] arctic=[GetGlobalVariable('te_seabed_fz_prot_arctic').GetValue] southern=[GetGlobalVariable('te_seabed_fz_prot_southern').GetValue] (Sponsor floors applied, every basin's entitlements within its pool or its holdings)"
			}
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN seabed_floor_fits over=[GetGlobalVariable('te_seabed_floor_over').GetValue] unbuilt pacific=[GetGlobalVariable('te_seabed_world_unbuilt_pacific').GetValue] atlantic=[GetGlobalVariable('te_seabed_world_unbuilt_atlantic').GetValue] indian=[GetGlobalVariable('te_seabed_world_unbuilt_indian').GetValue] arctic=[GetGlobalVariable('te_seabed_world_unbuilt_arctic').GetValue] southern=[GetGlobalVariable('te_seabed_world_unbuilt_southern').GetValue] (a basin's entitlements above its pool and its holdings after the floors: te_seabed_fit_floors' scaling; docs/testing/global-commons-playtest.md)"
		}
	}
}

# No scope, at the end of the sharing: the seabed_power_projection self-check
# (seabed claims spec § 6.1), once a game, at the first sharing with a
# claimant that has a navy (navy_size > 0, so a country with no fleet can't
# make OK pass). One line for each such claimant, its naval power projection
# read as a value, its navy_size and their ratio (the play-test sets
# te_seabed_power_projection_scale from the median), then the verdict: OK when
# the read gives any of them power projection above 0, BROKEN when all read 0
# (the read fails and the navy term is 0 for everyone: fall back to
# navy_size). A fleet with no crews can read 0 honestly, so BROKEN needs all.
# The marker is _v2 because the first run's INFO lines printed nothing: a
# debug_log needs [THIS.GetCountry.GetNameNoFormatting], and a bare
# [THIS.GetNameNoFormatting] drops the whole line (2026-10-17).
te_seabed_power_projection_check = {
	if = {
		limit = {
			NOT = { has_global_variable = te_seabed_npp_check_done_v2 }
			any_country = {
				te_seabed_is_claimant = yes
				navy_size > 0
			}
		}
		set_global_variable = te_seabed_npp_check_done_v2
		set_global_variable = { name = te_seabed_npp_chk_n value = 0 }
		set_global_variable = { name = te_seabed_npp_chk_ok value = 0 }
		every_country = {
			limit = {
				te_seabed_is_claimant = yes
				navy_size > 0
			}
			change_global_variable = { name = te_seabed_npp_chk_n add = 1 }
			if = {
				limit = { te_seabed_npp_disp > 0 }
				change_global_variable = { name = te_seabed_npp_chk_ok add = 1 }
			}
			debug_log = "TE_SELF_CHECK INFO seabed_power_projection country=[THIS.GetCountry.GetNameNoFormatting] npp=[THIS.ScriptValue('te_seabed_npp_disp')|1] navy_size=[THIS.ScriptValue('te_seabed_navy_size_disp')|0] ratio=[THIS.ScriptValue('te_seabed_npp_ratio_disp')|2]"
		}
		if = {
			limit = { global_var:te_seabed_npp_chk_ok > 0 }
			debug_log = "TE_SELF_CHECK OK seabed_power_projection claimants=[GetGlobalVariable('te_seabed_npp_chk_n').GetValue] reading=[GetGlobalVariable('te_seabed_npp_chk_ok').GetValue] (naval_power_projection reads as a value for the claimants with a navy; the INFO lines above give each one's ratio to navy_size: set te_seabed_power_projection_scale to their median)"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN seabed_power_projection claimants=[GetGlobalVariable('te_seabed_npp_chk_n').GetValue] reading=0 (every claimant with a navy reads 0 naval power projection as a value, so the claim weight's navy term is 0: revert te_seabed_navy_effective to navy_size, docs/testing/global-commons-playtest.md)"
		}
	}
}

# Entitlements = records + unbuilt (te_seabed_world_totals); over when the
# unbuilt exceed what the pool leaves beyond the records.
te_seabed_floor_check_basin = {
	set_local_variable = {
		name = te_sb_fl_room
		value = {
			value = te_seabed_pool_$B$
			subtract = global_var:te_seabed_records_$B$
			min = 0
		}
	}
	if = {
		limit = { global_var:te_seabed_world_unbuilt_$B$ > local_var:te_sb_fl_room }
		change_global_variable = { name = te_seabed_floor_over add = 1 }
	}
}

# No scope. Model: share_basin's fold_unclaimed. A reserved half no active
# claimant is eligible for joins the open half (review L1), so each pool is
# always fully shared.
te_seabed_share_fold_unclaimed = {
	if = {
		limit = {
			global_var:te_seabed_rem_res_$B$ > 0
			NOT = {
				any_country = {
					has_variable = te_seabed_act_$B$
					has_variable = te_seabed_rsv_$B$
					var:te_seabed_w_$B$ > 0
				}
			}
		}
		change_global_variable = { name = te_seabed_rem_open_$B$ add = global_var:te_seabed_rem_res_$B$ }
		set_global_variable = { name = te_seabed_rem_res_$B$ value = 0 }
	}
}

# No scope. Each active country's tentative entitlement, te_seabed_e_$B$, of
# which te_seabed_er_$B$ from the reserved half.
te_seabed_share_tentative = {
	set_global_variable = { name = te_seabed_wsum_res_$B$ value = 0 }
	set_global_variable = { name = te_seabed_wsum_all_$B$ value = 0 }
	every_country = {
		limit = { has_variable = te_seabed_act_$B$ }
		change_global_variable = { name = te_seabed_wsum_all_$B$ add = var:te_seabed_w_$B$ }
		if = {
			limit = { has_variable = te_seabed_rsv_$B$ }
			change_global_variable = { name = te_seabed_wsum_res_$B$ add = var:te_seabed_w_$B$ }
		}
	}
	every_country = {
		limit = { has_variable = te_seabed_act_$B$ }
		if = {
			limit = {
				has_variable = te_seabed_rsv_$B$
				global_var:te_seabed_wsum_res_$B$ > 0
			}
			set_variable = {
				name = te_seabed_er_$B$
				value = {
					value = global_var:te_seabed_rem_res_$B$
					multiply = var:te_seabed_w_$B$
					divide = global_var:te_seabed_wsum_res_$B$
				}
			}
		}
		else = {
			set_variable = { name = te_seabed_er_$B$ value = 0 }
		}
		if = {
			limit = { global_var:te_seabed_wsum_all_$B$ > 0 }
			set_variable = {
				name = te_seabed_e_$B$
				value = {
					value = global_var:te_seabed_rem_open_$B$
					multiply = var:te_seabed_w_$B$
					divide = global_var:te_seabed_wsum_all_$B$
					add = var:te_seabed_er_$B$
				}
			}
		}
		else = {
			set_variable = { name = te_seabed_e_$B$ value = var:te_seabed_er_$B$ }
		}
	}
}

# Country scope: frozen at what it holds, or its floor if more
# (te_seabed_fz_$B$). That comes off the halves in proportion to its tentative
# shares.
te_seabed_share_freeze = {
	remove_variable = te_seabed_act_$B$
	change_global_variable = { name = te_seabed_newly_$B$ add = 1 }
	change_global_variable = { name = te_seabed_frozen_$B$ add = var:te_seabed_fz_$B$ }
	if = {
		limit = { var:te_seabed_e_$B$ > 0 }
		set_local_variable = {
			name = te_sb_dr
			value = {
				value = var:te_seabed_fz_$B$
				multiply = var:te_seabed_er_$B$
				divide = var:te_seabed_e_$B$
			}
		}
	}
	else = {
		set_local_variable = { name = te_sb_dr value = 0 }
	}
	set_variable = { name = te_seabed_dr_$B$ value = local_var:te_sb_dr }
	change_global_variable = { name = te_seabed_rem_res_$B$ subtract = local_var:te_sb_dr }
	change_global_variable = { name = te_seabed_rem_open_$B$ subtract = var:te_seabed_fz_$B$ }
	change_global_variable = { name = te_seabed_rem_open_$B$ add = local_var:te_sb_dr }
}

# No scope. A half driven below zero by grandfathered holdings takes the
# deficit from the other, so the basin never shares out more than its pool.
te_seabed_share_rebalance = {
	if = {
		limit = { global_var:te_seabed_rem_res_$B$ < 0 }
		change_global_variable = { name = te_seabed_rem_open_$B$ add = global_var:te_seabed_rem_res_$B$ }
		set_global_variable = { name = te_seabed_rem_res_$B$ value = 0 }
	}
	if = {
		limit = { global_var:te_seabed_rem_open_$B$ < 0 }
		change_global_variable = { name = te_seabed_rem_res_$B$ add = global_var:te_seabed_rem_open_$B$ }
		set_global_variable = { name = te_seabed_rem_open_$B$ value = 0 }
		if = {
			limit = { global_var:te_seabed_rem_res_$B$ < 0 }
			set_global_variable = { name = te_seabed_rem_res_$B$ value = 0 }
		}
	}
}

# No scope. Entitlements: the active countries' tentative shares rounded down,
# the leftovers (the pool less the frozen amounts less the floors) one each
# by largest remainder; a frozen country's is what it was frozen at (its
# holdings or its floor); and unbuilt = entitlement - held (te_seabed_sh_$B$)
# for every participant.
te_seabed_share_round = {
	te_seabed_share_fold_unclaimed = { B = $B$ }
	te_seabed_share_tentative = { B = $B$ }
	set_global_variable = {
		name = te_seabed_left_$B$
		value = {
			value = te_seabed_pool_$B$
			subtract = global_var:te_seabed_frozen_$B$
			min = 0
		}
	}
	every_country = {
		limit = { has_variable = te_seabed_act_$B$ }
		set_variable = {
			name = te_seabed_ent_$B$
			value = {
				value = var:te_seabed_e_$B$
				floor = yes
			}
		}
		set_variable = {
			name = te_seabed_frac_$B$
			value = {
				value = var:te_seabed_e_$B$
				subtract = var:te_seabed_ent_$B$
			}
		}
		change_global_variable = { name = te_seabed_left_$B$ subtract = var:te_seabed_ent_$B$ }
	}
	while = {
		limit = {
			global_var:te_seabed_left_$B$ > 0
			any_country = {
				has_variable = te_seabed_act_$B$
				NOT = { has_variable = te_seabed_bonus_$B$ }
			}
		}
		ordered_country = {
			limit = {
				has_variable = te_seabed_act_$B$
				NOT = { has_variable = te_seabed_bonus_$B$ }
			}
			order_by = te_seabed_frac_value_$B$
			position = 0
			change_variable = { name = te_seabed_ent_$B$ add = 1 }
			set_variable = te_seabed_bonus_$B$
			change_global_variable = { name = te_seabed_left_$B$ subtract = 1 }
		}
	}
	# The engine's truncation can leave the floors one over: take it back from
	# the smallest remainders that stay at or above what they're frozen at.
	while = {
		limit = {
			global_var:te_seabed_left_$B$ < 0
			any_country = {
				has_variable = te_seabed_act_$B$
				NOT = { has_variable = te_seabed_cut_$B$ }
				var:te_seabed_ent_$B$ > var:te_seabed_fz_$B$
			}
		}
		ordered_country = {
			limit = {
				has_variable = te_seabed_act_$B$
				NOT = { has_variable = te_seabed_cut_$B$ }
				var:te_seabed_ent_$B$ > var:te_seabed_fz_$B$
			}
			order_by = te_seabed_frac_neg_value_$B$
			position = 0
			change_variable = { name = te_seabed_ent_$B$ subtract = 1 }
			set_variable = te_seabed_cut_$B$
			change_global_variable = { name = te_seabed_left_$B$ add = 1 }
		}
	}
	te_seabed_readout = { B = $B$ }
	every_country = {
		limit = { has_variable = te_seabed_part }
		if = {
			limit = { NOT = { has_variable = te_seabed_act_$B$ } }
			set_variable = { name = te_seabed_ent_$B$ value = var:te_seabed_fz_$B$ }
		}
		set_variable = {
			name = te_seabed_unbuilt_$B$
			value = {
				value = var:te_seabed_ent_$B$
				subtract = var:te_seabed_sh_$B$
				min = 0
			}
		}
		# The sharing re-divides leased claims too; te_seabed_settle_leases
		# tops each standing lease back up from its lessor's claims.
		if = {
			limit = { has_variable = te_seabed_lease_unused_$B$ }
			set_variable = { name = te_seabed_lease_unused_$B$ value = 0 }
		}
		if = {
			limit = { has_variable = te_seabed_subj_unused_$B$ }
			set_variable = { name = te_seabed_subj_unused_$B$ value = 0 }
		}
		if = {
			limit = { has_variable = te_seabed_pass_unused_$B$ }
			set_variable = { name = te_seabed_pass_unused_$B$ value = 0 }
		}
		if = {
			limit = { has_variable = te_seabed_act_$B$ }
			remove_variable = te_seabed_act_$B$
		}
		if = {
			limit = { has_variable = te_seabed_bonus_$B$ }
			remove_variable = te_seabed_bonus_$B$
		}
		if = {
			limit = { has_variable = te_seabed_cut_$B$ }
			remove_variable = te_seabed_cut_$B$
		}
	}
}

# No scope, in te_seabed_share_round after the entitlements and before the
# active flags go. Model: need_weight, need_navy (seabed claims spec § 3.5).
# Each participant with weight here stores te_seabed_need_navy_$B$: about how
# much more navy (the claim-weight term te_seabed_navy_effective counts) would
# win it one more licence at this sharing, rounded up; -1 when no half has a
# licence left for it ("full"), -2 when it has no weight here. A sufficient
# figure: the sharing's rounding can hand the licence over for less. With R
# left to the active set, W its weight and e = R·w/W our share of a half, the
# deficit d = t - e needs Δw = d·W / (R - e - d) from that half alone; an
# eligible party takes the smaller of its two halves. A frozen country's target
# is what it was frozen at + 1, its holdings back in the halves they came off.
te_seabed_readout = {
	every_country = {
		limit = { has_variable = te_seabed_part }
		if = {
			limit = { var:te_seabed_w_$B$ > 0 }
			set_local_variable = { name = te_sb_ro_rr value = global_var:te_seabed_rem_res_$B$ }
			set_local_variable = { name = te_sb_ro_ro value = global_var:te_seabed_rem_open_$B$ }
			set_local_variable = { name = te_sb_ro_wr value = global_var:te_seabed_wsum_res_$B$ }
			set_local_variable = { name = te_sb_ro_wa value = global_var:te_seabed_wsum_all_$B$ }
			if = {
				limit = { has_variable = te_seabed_act_$B$ }
				set_local_variable = {
					name = te_sb_ro_t
					value = {
						value = var:te_seabed_ent_$B$
						add = 1
					}
				}
			}
			else = {
				set_local_variable = {
					name = te_sb_ro_t
					value = {
						value = var:te_seabed_fz_$B$
						add = 1
					}
				}
				set_local_variable = { name = te_sb_ro_dr value = 0 }
				if = {
					limit = { has_variable = te_seabed_dr_$B$ }
					set_local_variable = { name = te_sb_ro_dr value = var:te_seabed_dr_$B$ }
				}
				change_local_variable = { name = te_sb_ro_rr add = local_var:te_sb_ro_dr }
				change_local_variable = { name = te_sb_ro_ro add = var:te_seabed_fz_$B$ }
				change_local_variable = { name = te_sb_ro_ro subtract = local_var:te_sb_ro_dr }
				change_local_variable = { name = te_sb_ro_wa add = var:te_seabed_w_$B$ }
				if = {
					limit = { has_variable = te_seabed_rsv_$B$ }
					change_local_variable = { name = te_sb_ro_wr add = var:te_seabed_w_$B$ }
				}
			}
			set_local_variable = {
				name = te_sb_ro_eo
				value = {
					value = local_var:te_sb_ro_ro
					multiply = var:te_seabed_w_$B$
					divide = local_var:te_sb_ro_wa
				}
			}
			set_local_variable = { name = te_sb_ro_er value = 0 }
			set_local_variable = { name = te_sb_ro_res value = 0 }
			if = {
				limit = {
					has_variable = te_seabed_rsv_$B$
					local_var:te_sb_ro_rr > 0
					local_var:te_sb_ro_wr > 0
				}
				set_local_variable = { name = te_sb_ro_res value = 1 }
				set_local_variable = {
					name = te_sb_ro_er
					value = {
						value = local_var:te_sb_ro_rr
						multiply = var:te_seabed_w_$B$
						divide = local_var:te_sb_ro_wr
					}
				}
			}
			set_local_variable = {
				name = te_sb_ro_d
				value = {
					value = local_var:te_sb_ro_t
					subtract = local_var:te_sb_ro_eo
					subtract = local_var:te_sb_ro_er
					min = 0
				}
			}
			# The open half.
			set_local_variable = { name = te_sb_ro_x value = -1 }
			set_local_variable = {
				name = te_sb_ro_den
				value = {
					value = local_var:te_sb_ro_ro
					subtract = local_var:te_sb_ro_eo
					subtract = local_var:te_sb_ro_d
				}
			}
			if = {
				limit = { local_var:te_sb_ro_den > 0.01 }
				set_local_variable = {
					name = te_sb_ro_x
					value = {
						value = local_var:te_sb_ro_d
						multiply = local_var:te_sb_ro_wa
						divide = local_var:te_sb_ro_den
					}
				}
			}
			# The reserved half, for an eligible party: the smaller of the two.
			if = {
				limit = { local_var:te_sb_ro_res = 1 }
				set_local_variable = {
					name = te_sb_ro_den
					value = {
						value = local_var:te_sb_ro_rr
						subtract = local_var:te_sb_ro_er
						subtract = local_var:te_sb_ro_d
					}
				}
				if = {
					limit = { local_var:te_sb_ro_den > 0.01 }
					set_local_variable = {
						name = te_sb_ro_x2
						value = {
							value = local_var:te_sb_ro_d
							multiply = local_var:te_sb_ro_wr
							divide = local_var:te_sb_ro_den
						}
					}
					if = {
						limit = {
							OR = {
								local_var:te_sb_ro_x < 0
								local_var:te_sb_ro_x2 < local_var:te_sb_ro_x
							}
						}
						set_local_variable = { name = te_sb_ro_x value = local_var:te_sb_ro_x2 }
					}
				}
			}
			if = {
				limit = { local_var:te_sb_ro_x < 0 }
				set_variable = { name = te_seabed_need_navy_$B$ value = -1 }
			}
			else = {
				# Weight to navy: at home the navy counts by our coastal states
				# here over all of them; far away at te_seabed_distant_navy_share.
				set_local_variable = { name = te_sb_ro_mult value = 1 }
				if = {
					limit = { te_seabed_navy_assertive = yes }
					set_local_variable = { name = te_sb_ro_mult value = te_seabed_assertive_navy_mult }
				}
				if = {
					limit = {
						has_variable = te_seabed_coast_$B$
						var:te_seabed_coast_$B$ > 0
					}
					set_local_variable = {
						name = te_sb_ro_per
						value = {
							value = te_seabed_policy_factor
							multiply = local_var:te_sb_ro_mult
							multiply = var:te_seabed_coast_$B$
							divide = var:te_seabed_coast_all
						}
					}
				}
				else = {
					set_local_variable = {
						name = te_sb_ro_per
						value = {
							value = te_seabed_policy_factor
							multiply = local_var:te_sb_ro_mult
							multiply = te_seabed_distant_navy_share
						}
					}
				}
				set_variable = {
					name = te_seabed_need_navy_$B$
					value = {
						value = local_var:te_sb_ro_x
						divide = local_var:te_sb_ro_per
						ceiling = yes
					}
				}
			}
		}
		else = {
			set_variable = { name = te_seabed_need_navy_$B$ value = -2 }
		}
	}
}

# No scope: the world's records and weight in basin $B$, after the sharing,
# and the ceiling the next year's cap check holds the records to (model test:
# GrandfatheredTest): the pool, or the grandfathered records if more.
te_seabed_world_totals = {
	set_global_variable = { name = te_seabed_records_$B$ value = 0 }
	set_global_variable = { name = te_seabed_world_w_$B$ value = 0 }
	set_global_variable = { name = te_seabed_world_unbuilt_$B$ value = 0 }
	every_country = {
		limit = { has_variable = te_seabed_shared }
		change_global_variable = { name = te_seabed_records_$B$ add = var:te_seabed_held_$B$ }
		change_global_variable = { name = te_seabed_world_w_$B$ add = var:te_seabed_w_$B$ }
		change_global_variable = { name = te_seabed_world_unbuilt_$B$ add = var:te_seabed_unbuilt_$B$ }
	}
	set_global_variable = {
		name = te_seabed_ceiling_$B$
		value = {
			value = global_var:te_seabed_records_$B$
			add = global_var:te_seabed_world_unbuilt_$B$
			min = te_seabed_pool_$B$
		}
	}
	set_global_variable = { name = te_seabed_noshare_$B$ value = 0 }
	set_global_variable = { name = te_seabed_cap_stamped_noshare value = 0 }
	# The basin fully licensed (seabed claims spec § 7): told once each time the
	# world's licences reach the pool, and re-armed only when they fall below
	# it, to every player with a share there or coast on it. A full basin's
	# shares equal their holdings, so a share alone would miss a coastal
	# claimant with no mines there yet (review I3).
	if = {
		limit = { global_var:te_seabed_records_$B$ >= te_seabed_pool_$B$ }
		if = {
			limit = { NOT = { has_global_variable = te_seabed_full_$B$ } }
			set_global_variable = te_seabed_full_$B$
			every_country = {
				limit = {
					is_ai = no
					has_variable = te_seabed_shared
					OR = {
						AND = {
							has_variable = te_seabed_ent_$B$
							var:te_seabed_ent_$B$ > 0
						}
						AND = {
							has_variable = te_seabed_coast_$B$
							var:te_seabed_coast_$B$ > 0
						}
					}
				}
				post_notification = te_seabed_basin_full_notice_$B$
			}
		}
	}
	else_if = {
		limit = { has_global_variable = te_seabed_full_$B$ }
		remove_global_variable = te_seabed_full_$B$
	}
}

# No scope, after the sharing's leases and pacts (seabed claims spec § 7): in
# each basin, a country this sharing first gives a share (te_seabed_ent above
# 0 and no te_seabed_claimed mark yet) is told to every other player with a
# share there since an earlier sharing and coast on the basin (review I2: a
# distant share in every basin had a great power hear of each newcomer about
# four times; the message group lets a player mute the rest); then every
# country with a share is marked, for good, so one that loses its share and
# wins it back isn't new.

te_seabed_claimant_notices = {

		te_seabed_claimant_notice = { B = indian }
		te_seabed_claimant_notice = { B = arctic }
		te_seabed_claimant_notice = { B = southern }
	}
	te_seabed_mark_claimed = { B = pacific }
	te_seabed_mark_claimed = { B = atlantic }
	te_seabed_mark_claimed = { B = indian }
	te_seabed_mark_claimed = { B = arctic }
	te_seabed_mark_claimed = { B = southern }
	set_global_variable = te_seabed_claimed_seeded
}

# No scope. Each newcomer in basin $B$ in turn is saved as te_seabed_newcomer,
# the scope its notice names; two newcomers of one sharing aren't told of
# each other.
te_seabed_claimant_notice = {
	every_country = {
		limit = {
			has_variable = te_seabed_ent_$B$
			var:te_seabed_ent_$B$ > 0
			NOT = { has_variable = te_seabed_claimed_$B$ }

		save_scope_as = te_seabed_newcomer
		every_country = {
			limit = {
				NOT = { this = scope:te_seabed_newcomer }
				is_ai = no
				has_variable = te_seabed_claimed_$B$
				has_variable = te_seabed_ent_$B$

				has_variable = te_seabed_coast_$B$
				var:te_seabed_coast_$B$ > 0
			}
			post_notification = te_seabed_new_claimant_notice_$B$
		}
	}
}

# No scope: every country with a share in basin $B$ now has had one.

	}
}

}

# Country scope: the influence modifier the policy holds, rebuilt from the
# variables (so a revolution's winner, which inherits variables and no
# modifiers, pays from its next monthly pulse); the two modifiers' one refresh
# site. Unit modifiers with the cost as multiplier (seabed claims spec § 2.3),
# cached so the monthly check can tell when it moves (the cache stays on
# Survey, as the royalty's does: a multiplier's variable is never removed
# where its modifier is applied). A multiplier reads ROOT:
# every caller that adds one has ROOT = this country (its monthly pulse, the
# policy buttons, the AI's yearly pulse); the yearly pass's forced reset to
# Survey only removes.
te_seabed_refresh_policy_cost = {
	remove_modifier = te_seabed_policy_sponsor_cost
	remove_modifier = te_seabed_policy_unilateral_cost
	if = {
		limit = {
			te_global_commons_enabled = yes
			te_seabed_policy_sponsor = yes
		}
		set_variable = { name = te_seabed_policy_cost_cached value = te_seabed_sponsor_cost_value }
		add_modifier = {
			name = te_seabed_policy_sponsor_cost
			multiplier = var:te_seabed_policy_cost_cached
		}
	}
	else_if = {
		limit = {
			te_global_commons_enabled = yes
			te_seabed_policy_unilateral = yes
		}
		set_variable = { name = te_seabed_policy_cost_cached value = te_seabed_unilateral_cost_value }
		add_modifier = {
			name = te_seabed_policy_unilateral_cost
			multiplier = var:te_seabed_policy_cost_cached
		}
	}
}

# Country scope (on_monthly_pulse_country). Model: record_country. A claimant
# with no share yet asks for a sharing next month (te_seabed_reshare). With
# the rule off, the policy's modifier comes off.
te_seabed_monthly_country = {
	if = {
		limit = {
			te_global_commons_enabled = yes
			te_seabed_records_monthly = yes
		}
		te_seabed_record_country = yes
		te_seabed_policy_cost_check = yes
		te_seabed_fee_refresh = yes
		if = {
			limit = {
				te_seabed_is_claimant = yes
				NOT = { has_variable = te_seabed_shared }
				NOT = { has_global_variable = te_seabed_reshare }
			}
			set_global_variable = te_seabed_reshare
		}
	}
	else_if = {
		limit = { has_variable = te_seabed_country }
		te_seabed_policy_cost_check = yes
		te_seabed_fee_refresh = yes
	}
}

# ---- The holding fee (seabed claims spec § 2.1) ----

# State scope (the yearly pass, the year's first sharing, after recording).
# Model: take_snapshots. The state keeps its last two counts of licences on
# unbuilt levels.
te_seabed_take_snapshots = {
	if = {
		limit = { has_variable = te_seabed_unb_snap1 }
		set_variable = { name = te_seabed_unb_snap2 value = var:te_seabed_unb_snap1 }
	}
	else = {
		set_variable = { name = te_seabed_unb_snap2 value = 0 }
	}
	set_variable = { name = te_seabed_unb_snap1 value = te_seabed_state_unbuilt_lic }
}

# Country scope (te_seabed_monthly_country): te_seabed_holding_fee_modifier's
# one refresh site, re-priced from the fee base; off with the rule off or
# nothing to pay. The cached figure is a variable, so a revolution's winner
# pays from its next pulse. While the country pays royalties' way (a Law of
# the Sea party under the Regulated regime) te_seabed_royalties_total_value
# sends the fee with them.
te_seabed_fee_refresh = {
	if = {
		limit = { te_global_commons_enabled = yes }
		set_local_variable = { name = te_sb_fee value = te_seabed_fee_weekly_value }
	}
	else = {
		set_local_variable = { name = te_sb_fee value = 0 }
	}
	if = {
		limit = { local_var:te_sb_fee > 0 }
		set_variable = { name = te_seabed_fee_cached value = local_var:te_sb_fee }
		if = {
			limit = { has_modifier = te_seabed_holding_fee_modifier }
			remove_modifier = te_seabed_holding_fee_modifier
		}
		add_modifier = {
			name = te_seabed_holding_fee_modifier
			multiplier = var:te_seabed_fee_cached
		}
	}
	else = {
		if = {
			limit = { has_modifier = te_seabed_holding_fee_modifier }
			remove_modifier = te_seabed_holding_fee_modifier
		}
		if = {
			limit = { has_variable = te_seabed_fee_cached }
			remove_variable = te_seabed_fee_cached
		}
	}
}

# Country scope: rebuild the policy's modifier only when it doesn't match the
# variable (a revolution's winner, the rule switched, an old save) or its cost
# has moved with the claims, so a standing policy isn't removed and re-added
# every month for nothing.
te_seabed_policy_cost_check = {
	if = {
		limit = {
			OR = {
				AND = {
					te_global_commons_enabled = yes
					NOT = { te_seabed_policy_survey = yes }
					OR = {
						NOT = { has_variable = te_seabed_policy_cost_cached }
						NOT = { var:te_seabed_policy_cost_cached = te_seabed_policy_cost_value }
					}
				}
				AND = {
					te_global_commons_enabled = yes
					te_seabed_policy_sponsor = yes
					NOT = { has_modifier = te_seabed_policy_sponsor_cost }
				}
				AND = {
					te_global_commons_enabled = yes
					te_seabed_policy_unilateral = yes
					NOT = { has_modifier = te_seabed_policy_unilateral_cost }
				}
				AND = {
					has_modifier = te_seabed_policy_sponsor_cost
					OR = {
						NOT = { te_global_commons_enabled = yes }
						NOT = { te_seabed_policy_sponsor = yes }
					}
				}
				AND = {
					has_modifier = te_seabed_policy_unilateral_cost
					OR = {
						NOT = { te_global_commons_enabled = yes }
						NOT = { te_seabed_policy_unilateral = yes }
					}
				}
			}
		}
		te_seabed_refresh_policy_cost = yes
	}
}

# Country scope (on_yearly_pulse_country, ROOT = the country), AI countries
# only: hold the policy with the highest score (te_global_commons_seabed_ai_values.txt).
# The engine's builder obeys the gate, so building needs no AI code. Ties keep
# the lower policy; the current policy carries a stay bonus there.
te_seabed_ai_choose_policy = {
	if = {
		limit = {
			te_global_commons_enabled = yes
			is_ai = yes
			te_seabed_is_claimant = yes
			has_variable = te_seabed_shared
			# A revolution's side writes no policy: if it wins, its own value
			# would replace the nation's (review L3).
			NOT = { is_revolutionary = yes }
		}
		set_local_variable = { name = te_sb_ai_1 value = te_seabed_ai_score_survey }
		set_local_variable = { name = te_sb_ai_2 value = te_seabed_ai_score_sponsor }
		set_local_variable = { name = te_sb_ai_3 value = te_seabed_ai_score_unilateral }
		# Only to a policy its buttons would allow: the five-year lock and its
		# carve-outs bind the AI too (seabed claims spec § 2.4).
		if = {
			limit = {
				local_var:te_sb_ai_3 > local_var:te_sb_ai_2
				local_var:te_sb_ai_3 > local_var:te_sb_ai_1
			}
			if = {
				limit = { te_seabed_can_choose_unilateral = yes }
				te_seabed_set_policy = { POLICY = 3 }
			}
		}
		else_if = {
			limit = { local_var:te_sb_ai_2 > local_var:te_sb_ai_1 }
			if = {
				limit = { te_seabed_can_choose_sponsor = yes }
				te_seabed_set_policy = { POLICY = 2 }
			}
		}
		else_if = {
			limit = { te_seabed_can_choose_survey = yes }
			te_seabed_set_policy = { POLICY = 1 }
		}
	}
}

# No scope, in the yearly pass after every participant has recorded: seabed_cap.
# BROKEN when (a) a basin's records grew past last year's ceiling (its pool, or
# the grandfathered records if more) plus what was recorded against no share
# since; (b) a monthly recording grandfathered a stamped state since the last
# pass (stamped_noshare: the H1 leak's signature, which the no-share counter in
# (a) would otherwise absorb, review H1); or (c) a participant leaves a level
# unlicensed while it still holds unbuilt share (escaped). Levels waiting for a
# licence because their owner queued past its room are counted (pending), not
# BROKEN. Waits for a previous pass (the ceilings); OK is logged once a game and
# only with records in the world, BROKEN every year it fails.
te_seabed_cap_check = {
	if = {
		limit = { has_global_variable = te_seabed_ceiling_pacific }
		set_global_variable = { name = te_seabed_cap_pending value = 0 }
		set_global_variable = { name = te_seabed_cap_escaped value = 0 }
		set_global_variable = { name = te_seabed_cap_records value = 0 }
		if = {
			limit = { NOT = { has_global_variable = te_seabed_cap_stamped_noshare } }
			set_global_variable = { name = te_seabed_cap_stamped_noshare value = 0 }
		}
		te_seabed_cap_check_basin = { B = pacific }
		te_seabed_cap_check_basin = { B = atlantic }
		te_seabed_cap_check_basin = { B = indian }
		te_seabed_cap_check_basin = { B = arctic }
		te_seabed_cap_check_basin = { B = southern }
		every_country = {
			limit = { has_variable = te_seabed_part }
			set_local_variable = { name = te_sb_cap_pend value = te_seabed_unrecorded_total }
			change_global_variable = { name = te_seabed_cap_pending add = local_var:te_sb_cap_pend }
			if = {
				limit = {
					local_var:te_sb_cap_pend > 0
					te_seabed_unbuilt_total > 0
				}
				change_global_variable = { name = te_seabed_cap_escaped add = 1 }
			}
		}
		if = {
			limit = {
				global_var:te_seabed_cap_escaped = 0
				global_var:te_seabed_cap_stamped_noshare = 0
				global_var:te_seabed_cap_slack_pacific >= 0
				global_var:te_seabed_cap_slack_atlantic >= 0
				global_var:te_seabed_cap_slack_indian >= 0
				global_var:te_seabed_cap_slack_arctic >= 0
				global_var:te_seabed_cap_slack_southern >= 0
			}
			if = {
				limit = {
					NOT = { has_global_variable = te_seabed_cap_ok_logged }
					global_var:te_seabed_cap_records > 0
				}
				set_global_variable = te_seabed_cap_ok_logged
				debug_log = "TE_SELF_CHECK OK seabed_cap records=[GetGlobalVariable('te_seabed_cap_records').GetValue] pending=[GetGlobalVariable('te_seabed_cap_pending').GetValue] slack pacific=[GetGlobalVariable('te_seabed_cap_slack_pacific').GetValue] atlantic=[GetGlobalVariable('te_seabed_cap_slack_atlantic').GetValue] indian=[GetGlobalVariable('te_seabed_cap_slack_indian').GetValue] arctic=[GetGlobalVariable('te_seabed_cap_slack_arctic').GetValue] southern=[GetGlobalVariable('te_seabed_cap_slack_southern').GetValue] (every basin within its pool or its grandfathered records, nothing grandfathered in a stamped state, every level licensed but those waiting past room)"
			}
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN seabed_cap stamped_noshare=[GetGlobalVariable('te_seabed_cap_stamped_noshare').GetValue] escaped=[GetGlobalVariable('te_seabed_cap_escaped').GetValue] pending=[GetGlobalVariable('te_seabed_cap_pending').GetValue] slack pacific=[GetGlobalVariable('te_seabed_cap_slack_pacific').GetValue] atlantic=[GetGlobalVariable('te_seabed_cap_slack_atlantic').GetValue] indian=[GetGlobalVariable('te_seabed_cap_slack_indian').GetValue] arctic=[GetGlobalVariable('te_seabed_cap_slack_arctic').GetValue] southern=[GetGlobalVariable('te_seabed_cap_slack_southern').GetValue] (stamped_noshare above 0: a monthly recording grandfathered a stamped state, the review's H1 leak; escaped above 0: a country left a level unlicensed while holding share; a negative slack: a basin's records grew past its ceiling. docs/testing/global-commons-playtest.md)"
		}
	}
}

te_seabed_cap_check_basin = {
	set_global_variable = { name = te_seabed_cap_now_$B$ value = 0 }
	every_country = {
		limit = { has_variable = te_seabed_part }
		every_scope_state = {
			limit = { has_variable = te_seabed_lic_$B$ }
			change_global_variable = { name = te_seabed_cap_now_$B$ add = var:te_seabed_lic_$B$ }
		}
	}
	change_global_variable = { name = te_seabed_cap_records add = global_var:te_seabed_cap_now_$B$ }
	if = {
		limit = { NOT = { has_global_variable = te_seabed_noshare_$B$ } }
		set_global_variable = { name = te_seabed_noshare_$B$ value = 0 }
	}
	set_global_variable = {
		name = te_seabed_cap_slack_$B$
		value = {
			value = global_var:te_seabed_ceiling_$B$
			add = global_var:te_seabed_noshare_$B$
			subtract = global_var:te_seabed_cap_now_$B$
		}
	}
}

# State scope (te_seabed_record_country): keep a list of states holding
# licences, independent of their own variables, until the follow-state check
# has run, so the check can tell a state that lost its records from one that
# never had any.
te_seabed_follow_watch = {
	if = {
		limit = {
			NOT = { has_global_variable = te_seabed_follow_chk_done }
			NOT = { has_variable = te_seabed_watched }
			te_seabed_state_records > 0
		}
		set_variable = te_seabed_watched
		add_to_global_variable_list = { name = te_seabed_follow_watch target = this }
	}
}

# State scope, ROOT = the state under its new owner, first thing in
# te_seabed_on_state_owner_change: seabed_licence_follows_state, first half.
# The first watched state to change owner: its records and holder stamp must
# read back here. BROKEN at once if not; otherwise the second half
# (te_seabed_follow_check_month) reads them again a month later. Unverified
# engine fact: a state keeps its variables across a change of owner (§ 2.4).
te_seabed_follow_check_hook = {
	if = {
		limit = {
			NOT = { has_global_variable = te_seabed_follow_chk_done }
			NOT = { has_global_variable = te_seabed_follow_chk_month }
			has_global_variable_list = te_seabed_follow_watch
			is_target_in_global_variable_list = { name = te_seabed_follow_watch target = this }
		}
		if = {
			limit = {
				te_seabed_state_records > 0
				has_variable = te_seabed_holder
			}
			set_global_variable = { name = te_seabed_follow_chk_state value = this }
			set_global_variable = { name = te_seabed_follow_chk_month value = te_history_month_index }
			debug_log = "TE_SELF_CHECK START seabed_licence_follows_state state=[THIS.GetState.GetNameNoFormatting] records=[THIS.ScriptValue('te_seabed_state_records')|0] (its records and holder read back inside on_state_owner_change; the verdict follows in a month) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			set_global_variable = te_seabed_follow_chk_done
			clear_global_variable_list = te_seabed_follow_watch
			debug_log = "TE_SELF_CHECK BROKEN seabed_licence_follows_state state=[THIS.GetState.GetNameNoFormatting] records=[THIS.ScriptValue('te_seabed_state_records')|0] (a state that held seabed licences reached on_state_owner_change without its te_seabed_lic_* or te_seabed_holder: take the fallback, per-region global records, docs/testing/global-commons-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
}

# No scope (te_seabed_global_monthly): the second half, a month or more after
# the first. A state gone since (merged away) re-arms the check.
te_seabed_follow_check_month = {
	if = {
		limit = {
			has_global_variable = te_seabed_follow_chk_month
			NOT = { has_global_variable = te_seabed_follow_chk_done }
			te_history_month_index > global_var:te_seabed_follow_chk_month
		}
		if = {
			limit = {
				global_var:te_seabed_follow_chk_state ?= { always = yes }
			}
			global_var:te_seabed_follow_chk_state = {
				if = {
					limit = {
						has_variable = te_seabed_lic_pacific
						has_variable = te_seabed_holder
					}
					debug_log = "TE_SELF_CHECK OK seabed_licence_follows_state state=[THIS.GetState.GetNameNoFormatting] records=[THIS.ScriptValue('te_seabed_state_records')|0] (its records and holder stamp read back inside on_state_owner_change and a month later) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN seabed_licence_follows_state state=[THIS.GetState.GetNameNoFormatting] records=[THIS.ScriptValue('te_seabed_state_records')|0] (its records read back inside on_state_owner_change but not a month later: take the fallback, per-region global records, docs/testing/global-commons-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
			set_global_variable = te_seabed_follow_chk_done
			clear_global_variable_list = te_seabed_follow_watch
		}
		else = {
			remove_global_variable = te_seabed_follow_chk_month
		}
	}
}
