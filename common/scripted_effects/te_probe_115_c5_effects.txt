# ============================================================================
# te_probe_115: the vanilla 1.15 migration's engine probes, C5's reads (2026-11-05)
# ============================================================================
# What it answers: C5 of docs/audits/vanilla_1_15_prep.md § 2, the root of a scripted progress bar's script-value
# min_value (common/scripted_progress_bars/te_probe_115_bars.txt: 40 for the owner, 20 for the entry, 5 for neither).
# Step 2 reads the bar as it stands, then adds 1; step 3 reads it again. Unclamped, it reads 0 and then 1; clamped to
# its minimum, that minimum (or the minimum plus 1, if it was clamped at activation). Called from te_probe_115_weekly_effect
# (common/scripted_effects/te_probe_115_effects.txt). THIS = the probe country.
#
# Delete once answered (see docs/testing/vanilla-1-15-playtest.md).

# Step 2: before any progress is added. Was the start value clamped at activation?
te_probe_115_c5_first_read_effect = {
	je:te_probe_115 ?= {
		if = {
			limit = { scripted_bar_progress = { name = te_probe_115_c5_bar value > 39.5 } }
			debug_log = "TE_PROBE_115 c5_at_activation 40 (the start value was clamped at activation to the owner's minimum)"
		}
		else_if = {
			limit = {
				scripted_bar_progress = { name = te_probe_115_c5_bar value > 19.5 }
				scripted_bar_progress = { name = te_probe_115_c5_bar value < 20.5 }
			}
			debug_log = "TE_PROBE_115 c5_at_activation 20 (the start value was clamped at activation to the entry's minimum)"
		}
		else_if = {
			limit = {
				scripted_bar_progress = { name = te_probe_115_c5_bar value > 4.5 }
				scripted_bar_progress = { name = te_probe_115_c5_bar value < 5.5 }
			}
			debug_log = "TE_PROBE_115 c5_at_activation 5 (the start value was clamped at activation to the minimum for neither: the root is not told here, since the owner's variable may not have been set yet; read c5_root)"
		}
		else_if = {
			limit = {
				scripted_bar_progress = { name = te_probe_115_c5_bar value > -0.5 }
				scripted_bar_progress = { name = te_probe_115_c5_bar value < 0.5 }
			}
			debug_log = "TE_PROBE_115 c5_at_activation 0 (the start value was not clamped at activation)"
		}
		else = {
			debug_log = "TE_PROBE_115 c5_at_activation other (the bar reads none of 0, 5, 20 or 40)"
		}
		add_progress = { name = te_probe_115_c5_bar value = 1 }
	}
}

# Step 3: a week or more after add_progress. Where was it clamped?
te_probe_115_c5_root_read_effect = {
	je:te_probe_115 ?= {
		if = {
			limit = { scripted_bar_progress = { name = te_probe_115_c5_bar value > 39.5 } }
			debug_log = "TE_PROBE_115 c5_root OWNER (a script-value min_value runs with the owning country as root and reads its variables)"
		}
		else_if = {
			limit = {
				scripted_bar_progress = { name = te_probe_115_c5_bar value > 19.5 }
				scripted_bar_progress = { name = te_probe_115_c5_bar value < 21.5 }
			}
			debug_log = "TE_PROBE_115 c5_root ENTRY (a script-value min_value runs with the journal entry as root)"
		}
		else_if = {
			limit = {
				scripted_bar_progress = { name = te_probe_115_c5_bar value > 4.5 }
				scripted_bar_progress = { name = te_probe_115_c5_bar value < 6.5 }
			}
			debug_log = "TE_PROBE_115 c5_root NONE (a script-value min_value ran, but its root is neither the owner nor the entry)"
		}
		else_if = {
			limit = {
				scripted_bar_progress = { name = te_probe_115_c5_bar value > 0.5 }
				scripted_bar_progress = { name = te_probe_115_c5_bar value < 1.5 }
			}
			debug_log = "TE_PROBE_115 c5_root UNCLAMPED (the bar was never clamped to its script-value minimum)"
		}
		else = {
			debug_log = "TE_PROBE_115 c5_root OTHER (the bar reads none of 1, 5, 20 or 40)"
		}
	}
}
