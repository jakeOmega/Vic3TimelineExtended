# ============================================================================
# UN TEST HELPERS — console only
# ============================================================================
# Nothing in the mod calls these. They exist so the UN chamber, mandates,
# standing and lobbying can be exercised in minutes instead of decades. Vic3's
# console has no `effect` command, so they are reached through a console-fired
# event (events/te_debug_un_events.txt):
#
#   event te_debug_un.1
#
# They skip votes, costs and eligibility on purpose, so never call them from
# gameplay script.
# ============================================================================

# Found the UN if nobody has, and seat the player plus every great and major
# power as members (the first five great powers, player included, as permanent
# members). Safe to run twice: run it again if the chamber still says
# "non-member", which means the journal entry had not activated yet.
# Scope: country
te_debug_un_setup = {
	if = {
		limit = { NOT = { has_global_variable = un_founded } }
		set_global_variable = un_founded
		set_global_variable = { name = un_hq_country value = ROOT }
	}
	if = {
		limit = { NOT = { has_global_variable = un_authority } }
		set_global_variable = { name = un_authority value = 80 }
	}
	te_debug_un_seat_member = yes
	every_country = {
		limit = {
			NOT = { this = ROOT }
			country_rank >= rank_value:major_power
			is_subject = no
		}
		te_debug_un_seat_member = yes
	}
}

# Scope: country
te_debug_un_seat_member = {
	if = {
		limit = { NOT = { has_journal_entry = je_united_nations } }
		add_journal_entry = { type = je_united_nations }
	}
	je:je_united_nations ?= {
		if = {
			limit = { NOT = { has_modifier = un_member_modifier } }
			add_modifier = { name = un_member_modifier }
		}
	}
	if = {
		limit = {
			country_rank >= rank_value:great_power
			un_permanent_seat_count < 5
			je:je_united_nations ?= { NOT = { has_modifier = un_permanent_member_modifier } }
		}
		je:je_united_nations ?= {
			add_modifier = { name = un_security_council_modifier }
			add_modifier = { name = un_permanent_member_modifier }
		}
	}
}

# Hand the player a live mandate without a vote: claims $REGION$ for the player
# (the war goal's own `valid` needs the claim), then creates the same container
# un_mandate_create builds. $TARGET$ must own a state in $REGION$ and the player
# must be a UN member, or the monthly sweep voids the mandate.
# Parameters: $TARGET$ = c:TAG, $REGION$ = s:STATE_X
# Scope: country
te_debug_un_grant_mandate = {
	save_scope_as = te_debug_mnd_actor
	$REGION$ = { add_claim = scope:te_debug_mnd_actor }
	if = {
		limit = { NOT = { has_global_variable = un_mandate_seq } }
		set_global_variable = { name = un_mandate_seq value = 0 }
	}
	change_global_variable = { name = un_mandate_seq add = 1 }
	create_container = {
		tags = { un_mandate un_mandate_active }
		save_scope_as = te_debug_mnd
	}
	if = {
		limit = { exists = scope:te_debug_mnd }
		scope:te_debug_mnd = {
			set_variable = { name = un_mnd_seq value = global_var:un_mandate_seq }
			set_variable = { name = un_mnd_actor value = scope:te_debug_mnd_actor }
			set_variable = { name = un_mnd_beneficiary value = scope:te_debug_mnd_actor }
			set_variable = { name = un_mnd_target value = $TARGET$ }
			set_variable = { name = un_mnd_region value = $REGION$ }
			set_variable = { name = un_mnd_months value = 0 }
			set_variable = { name = un_mnd_closed_months value = 0 }
		}
		add_to_global_variable_list = { name = un_mandate_registry target = scope:te_debug_mnd }
		set_variable = { name = un_mandate_current value = scope:te_debug_mnd }
	}
}

# Set the player's UN Standing directly, to look at a tier without earning it.
# The tier modifier swaps on the next monthly pulse. Deliberately bypasses
# un_standing_gain / un_standing_loss — test helper only.
# Parameters: $VALUE$ = 0-100
# Scope: country
te_debug_un_set_standing = {
	set_variable = { name = un_standing value = $VALUE$ } # REVIEWED 2026-09-18: console-only test helper; the gameplay writers are un_standing_gain / un_standing_loss
}

# Open a resolution on $TOPIC$ with $PROPOSER$ as its sponsor, so the player can
# vote on it from the chamber card on day one instead of waiting for a proposer
# event to fire and then thirty days for the un_vote.1 popup. $PROPOSER$ must be
# a UN member other than the player (run te_debug_un_setup first); the player is
# then an ordinary voter with Vote in Favour / Vote Against live immediately,
# and Veto too if the player holds a permanent seat and $TOPIC$ is binding.
#
# $TARGET$ is ALWAYS required — a scripted effect has no default parameters, and
# a second definition supplying one would be a duplicated top-level key the
# engine silently drops. Only the topics that name a country read it (condemn,
# sanctions, expulsion, military_mandate, and the two request topics, where the
# requester is its own target); for every other topic pass the proposer again,
# e.g. `TARGET = c:FRA`. un_resolution_open stores it as un_res_target either
# way and every consumer of that variable is guarded by `has_variable`.
#
# Costs, cooldowns, the standing floor and the authority gate are all skipped:
# this reaches straight past un_propose_*_possible into un_resolution_open.
# Parameters: $PROPOSER$ = a country scope or a tag (`scope:x`, `c:FRA`)
#             $TOPIC$    = condemn | sanctions | climate | ...
#             $TARGET$   = a country scope or a tag
# Scope: country (the player; only used to reach the proposer)
# Called from te_debug_un.1's options — there is no console `effect` command.
te_debug_un_open_resolution = {
	te_debug_un_clear_vote = yes
	$PROPOSER$ = {
		$TARGET$ = { save_scope_as = un_vote_target }
		un_resolution_open = { TOPIC = $TOPIC$ }
	}
}

# Throw away whatever is on the floor: destroy the open resolution and give back
# the lock, so another te_debug_un_open_resolution can run straight away. The
# queued un_vote.1 / un_vote.2 events for the discarded resolution fail their
# own triggers once its container is gone.
# Scope: any
te_debug_un_clear_vote = {
	global_var:un_active_resolution ?= { destroy_container = yes }
	if = {
		limit = { has_global_variable = un_active_resolution }
		remove_global_variable = un_active_resolution
	}
	if = {
		limit = { has_global_variable = un_vote_reservation }
		remove_global_variable = un_vote_reservation
	}
	if = {
		limit = { has_global_variable = un_vote_active }
		remove_global_variable = un_vote_active
	}
}

# ---- Redesign phase 2: the ladder, the crisis and the floor ----

# Doom the UN: authority 1, and the credibility and order ledgers driven to
# their floors so the target falls below the collapse line. The next monthly
# update opens the crisis and starts the collapse clock; the UN dissolves a
# few months later (authority falls at least a quarter point a month).
# Scope: any
te_debug_un_doom = {
	if = {
		limit = { has_global_variable = un_founded }
		set_global_variable = { name = un_authority value = 1 }
		set_global_variable = { name = un_ledger_credibility value = -15 }
		set_global_variable = { name = un_ledger_order value = -20 }
	}
}

# Put the charter's next reform on the table now: authority at the ceiling and
# the 24-month clock already run. The chamber's Charter Reform row goes live at
# once (un_charter_reform_ripe reads the counter directly).
# Scope: any
te_debug_un_ripen_charter = {
	if = {
		limit = { un_charter_can_be_reformed = yes }
		set_global_variable = { name = un_authority value = un_charter_cap }
		set_global_variable = { name = un_charter_pressure_months value = un_charter_reform_months_needed }
	}
}

# Skip the refounding cooldown, so a founding conference can be convened now.
# Scope: any
te_debug_un_skip_refounding_cooldown = {
	if = {
		limit = { has_global_variable = un_refounding_cooldown }
		remove_global_variable = un_refounding_cooldown
	}
	if = {
		limit = { has_global_variable = un_founding_conference }
		set_global_variable = { name = un_conference_months value = un_conference_months_needed }
	}
}

# ---- Redesign phase 4: the docket ----

# Open the docket now: clear its cadence, the Assembly-business cadence and
# the recess, and scan at once (un_docket_effects.txt). A vote in session
# still holds the floor, so only business that opens no resolution can be
# taken up while one is voting.
# Scope: any
te_debug_un_open_docket = {
	if = {
		limit = { has_global_variable = un_docket_months_left }
		remove_global_variable = un_docket_months_left
	}
	if = {
		limit = { has_global_variable = un_dkt_agenda_months_left }
		remove_global_variable = un_dkt_agenda_months_left
	}
	if = {
		limit = { has_global_variable = un_recess_months }
		remove_global_variable = un_recess_months
	}
	un_docket_seed = yes
	un_docket_scan = yes
}

# Stage a grievance against us: the strongest other member is caught running a
# severe covert operation on us, with a covert record heavy enough for a
# condemnation, and the docket opens at once.
# Scope: the country that ran the console event
te_debug_un_stage_grievance = {
	save_scope_as = te_debug_un_wronged
	ordered_country = {
		limit = {
			NOT = { this = scope:te_debug_un_wronged }
			je:je_united_nations ?= { has_modifier = un_member_modifier }
		}
		order_by = un_docket_share_order
		max = 1
		un_dossier_record = { RECORD = covert POINTS = 40 }
		un_docket_note_grievance = { KIND = covert VICTIM = scope:te_debug_un_wronged }
	}
	te_debug_un_open_docket = yes
}
