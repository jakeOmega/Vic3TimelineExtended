# Legislated tax code: the collection writer (docs/systems/tax_code_schema.md).
#
# te_tax_sync_collection is the only effect that changes a rule-on country's
# native fiscal state: the rate amendments on law_te_tax_code, the
# consumption-taxed goods, the relief modifiers, the native tax level and, under
# the customs option for a market owner, the tariff levels (plan Task 15). It
# reads the enacted code from the country's variables and changes only what
# differs. Its per-instrument and per-good parts are generated
# (te_tax_generated_effects.txt, gen_tax_code.py).
#
# Call it at most once per effect execution: amendment, goods and modifier
# changes are not visible later in the same execution, so a second call there
# would act on stale state. A call in a later execution changes nothing.
#
# At most one sync a day (final review, A-I1). Two events the engine runs in
# the same daily tick are separate executions, but whether the first one's
# add_amendment is visible to the second is unverified; if it is not, the
# second would add the matching rate amendment again, and nothing removes or
# detects a duplicate of the enacted type. Three callers can land on one day:
# the watchdog and the processor on the 1st (te_tax_watchdog_month,
# te_tax_process_month), a te_tax.4 raised the day before (a migration, a
# re-assert, a copied code) landing on the 1st, and several te_tax.4 from one
# law cascade. So every sync stamps the day it ran (te_tax_sync_day), and a
# second call that day writes nothing and hands the sync to te_tax.4 the next
# day (te_tax_defer_sync), at most one deferral a day. A package's collections
# then change a day after its commencement on the 1st, when another sync ran
# first that day. The two day variables are a guard, not schema tokens: read
# only through has_variable, never copied at an outbreak.
#
# Country scope, with ROOT = THIS = the country: call it from a country event
# or a country hook, never from the global monthly pulse, which has no ROOT.
# Nothing here reads ROOT; the country and the sponsor are saved scopes.
# Nothing below re-reads what it has just written.
#
# Before it writes, it counts native drift (te_tax_detect_drift): what changed
# the native state outside the code since the last full sync, such as the AI
# moving the tax level or toggling a taxed good, or a vanilla taxation law
# activated over the carrier. The count reads and never writes native state.

te_tax_sync_collection = {
	if = {
		limit = {
			te_tax_code_on = yes
			has_law = law_type:law_te_tax_code
			has_variable = te_tax_schema
		}
		# A sync already ran today: write nothing, sync tomorrow (see above).
		if = {
			limit = { te_tax_synced_today = yes }
			te_tax_defer_sync = yes
		}
		else = {
			# Backfills tokens a newer schema added; never overwrites one.
			te_tax_init_country = yes
			save_scope_as = te_tax_country
			# Counters only, before anything below changes the native state.
			te_tax_detect_drift = yes
			# Customs (the customs option, a market owner; plan Task 15): before the
			# amendment block, so a level adopted here moves te_tax_code_version
			# before the sync mark below records it.
			te_tax_sync_customs = yes
			te_tax_pick_sponsor = yes
			if = {
				# te_tax_pick_sponsor saved one of THIS country's groups exactly when it has one.
				limit = { any_interest_group = { always = yes } }
				te_tax_gen_sync_wage = yes
				te_tax_gen_sync_div = yes
				te_tax_gen_sync_land = yes
				te_tax_gen_sync_head = yes
				te_tax_gen_sync_cons = yes
				# A full sync: from here on, drift is measured against this code.
				set_variable = { name = te_tax_sync_version value = var:te_tax_code_version }
			}
			else = {
				debug_log = "TE_TAX sync: no interest group to sponsor an amendment; rate amendments left as they were"
			}
			te_tax_gen_sync_goods = yes
			te_tax_sync_relief = yes
			# The carrier and its amendments carry the same rates at every native
			# level, so the level moves only vanilla's political tax_modifier_<level>
			# statics; pinning it keeps identical codes from differing by them.
			if = { limit = { NOT = { tax_level = medium } } set_tax_level = medium }
			# The day of this sync: a second call today defers (te_tax_synced_today).
			set_variable = { name = te_tax_sync_day value = te_tax_today }
		}
	}
	else_if = {
		limit = {
			te_tax_code_on = yes
			has_law = law_type:law_te_tax_code
			NOT = { has_variable = te_tax_schema }
		}
		debug_log = "TE_TAX sync skipped: [THIS.GetCountry.GetNameNoFormatting] holds the tax code law but has no te_tax_schema"
	}
}

# Country scope; te_tax_sync_collection calls it instead of syncing when a sync
# already ran today (te_tax_synced_today). Raises te_tax.4 for tomorrow, which
# syncs then, unless a deferral raised today is already pending: several
# same-day callers make one deferred sync, not a chain of them. Writes only the
# day of the deferral (te_tax_sync_defer_day) and one debug line.
te_tax_defer_sync = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = te_tax_sync_defer_day }
				var:te_tax_sync_defer_day < te_tax_today
			}
		}
		set_variable = { name = te_tax_sync_defer_day value = te_tax_today }
		trigger_event = { id = te_tax.4 days = 1 }
	}
	debug_log = "TE_TAX sync_deferred a sync already ran today; the code is synced tomorrow (te_tax.4); month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
}

# Country scope; te_tax_sync_collection calls it once, after te_tax_init_country
# and before any part of the sync. Counts the native state that differs from the
# enacted code, and logs one line when anything does:
#   te_tax_drift_level +1 if the native tax level is not medium (the AI's
#     tax-level changes, vanilla's negotiation option 7);
#   te_tax_drift_goods +1 per good taxed or untaxed against the enacted list;
#   te_tax_drift_amend +1 per instrument whose carrier amendments differ from
#     its enacted index (a vanilla law activated over the carrier drops them);
#   te_tax_drift_customs +1 per tradeable good and direction whose market level
#     differs from the code's, while the code holds the country's customs
#     (te_tax_customs_authority: the AI's tariff changes, a treaty's entry).
# Only while te_tax_sync_version equals te_tax_code_version: the code has not
# changed since the last full sync. A sunset or a commencement changes the code
# before the processor's sync, and the migration before te_tax.4's, so the
# native state then differs by law, not by drift; whatever else drifted that
# month is put back by the same sync uncounted. Reads native state and writes
# only the three counters, so the sync sees what it would have seen without it.
# One debug line per drift for a player's country, the first only for an AI's.
te_tax_detect_drift = {
	if = {
		limit = {
			var:te_tax_sync_version = var:te_tax_code_version
			OR = {
				NOT = { tax_level = medium }
				te_tax_gen_goods_drift = yes
				te_tax_gen_amend_drift_wage = yes
				te_tax_gen_amend_drift_div = yes
				te_tax_gen_amend_drift_land = yes
				te_tax_gen_amend_drift_head = yes
				te_tax_gen_amend_drift_cons = yes
				AND = { te_tax_customs_authority = yes te_tax_gen_customs_drift = yes }
			}
		}
		# A player's country logs every time. An AI country logs its first drift
		# only: the AI may move the level every month the code pins it back, and
		# its counters keep the rest.
		if = {
			limit = {
				OR = {
					is_ai = no
					AND = {
						var:te_tax_drift_level = 0
						var:te_tax_drift_goods = 0
						var:te_tax_drift_amend = 0
						# An AI market owner may move tariffs every month (customs option).
						NAND = { has_variable = te_tax_drift_customs var:te_tax_drift_customs > 0 }
					}
				}
			}
			te_tax_count_drift = yes
			debug_log = "TE_TAX drift native state differed from the code before the sync; totals level=[SCOPE.ScriptValue('te_tax_view_drift_level')|0] goods=[SCOPE.ScriptValue('te_tax_view_drift_goods')|0] amend=[SCOPE.ScriptValue('te_tax_view_drift_amend')|0] customs=[SCOPE.ScriptValue('te_tax_view_drift_customs')|0]; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
		else = {
			te_tax_count_drift = yes
		}
	}
}

# Country scope; te_tax_detect_drift's counts, once it has found drift.
te_tax_count_drift = {
	if = { limit = { NOT = { tax_level = medium } } change_variable = { name = te_tax_drift_level add = 1 } }
	te_tax_gen_count_goods_drift = yes
	if = { limit = { te_tax_gen_amend_drift_wage = yes } change_variable = { name = te_tax_drift_amend add = 1 } }
	if = { limit = { te_tax_gen_amend_drift_div = yes } change_variable = { name = te_tax_drift_amend add = 1 } }
	if = { limit = { te_tax_gen_amend_drift_land = yes } change_variable = { name = te_tax_drift_amend add = 1 } }
	if = { limit = { te_tax_gen_amend_drift_head = yes } change_variable = { name = te_tax_drift_amend add = 1 } }
	if = { limit = { te_tax_gen_amend_drift_cons = yes } change_variable = { name = te_tax_drift_amend add = 1 } }
	if = { limit = { te_tax_customs_authority = yes } te_tax_gen_count_customs_drift = yes }
}

# Country scope. Saves scope:te_tax_sponsor, the interest group recorded as
# sponsoring the amendments the sync adds: a group in government, else the
# group with the most clout. add_amendment needs one; the choice changes no
# rate. Saved, not read from ROOT, so the sync can run from any country event.
# Branches on THIS country's groups, never on whether the scope already
# exists, so a sponsor saved for another country in the same execution
# cannot leak in.
te_tax_pick_sponsor = {
	if = {
		limit = { any_interest_group = { is_in_government = yes } }
		random_interest_group = {
			limit = { is_in_government = yes }
			save_scope_as = te_tax_sponsor
		}
	}
	else = {
		ordered_interest_group = {
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = te_tax_sponsor
		}
	}
}

# Country scope; needs scope:te_tax_country. Relief bands as static modifiers:
# agricultural wage relief on the country (te_tax_en_agrel), regional relief
# on every owned state flagged te_tax_relief_state = 1 (te_tax_en_regrel).
# The flags are derived first from the enacted list of named states
# (te_tax_rebuild_relief_marks, te_tax_civil_war_effects.txt), which also drops
# from the list a state the code no longer holds. Variables are visible at
# once, so the modifiers below read the rebuilt flags. A band is added only
# when missing and removed only when present, so an unchanged code adds and
# removes nothing.
te_tax_sync_relief = {
	te_tax_rebuild_relief_marks = yes
	if = {
		limit = { var:te_tax_en_agrel = 1 }
		if = { limit = { has_modifier = te_tax_relief_ag_2 } remove_modifier = te_tax_relief_ag_2 }
		if = { limit = { NOT = { has_modifier = te_tax_relief_ag_1 } } add_modifier = { name = te_tax_relief_ag_1 } }
	}
	else_if = {
		limit = { var:te_tax_en_agrel = 2 }
		if = { limit = { has_modifier = te_tax_relief_ag_1 } remove_modifier = te_tax_relief_ag_1 }
		if = { limit = { NOT = { has_modifier = te_tax_relief_ag_2 } } add_modifier = { name = te_tax_relief_ag_2 } }
	}
	else = {
		if = { limit = { has_modifier = te_tax_relief_ag_1 } remove_modifier = te_tax_relief_ag_1 }
		if = { limit = { has_modifier = te_tax_relief_ag_2 } remove_modifier = te_tax_relief_ag_2 }
	}
	every_scope_state = {
		if = {
			limit = {
				has_variable = te_tax_relief_state
				var:te_tax_relief_state = 1
				scope:te_tax_country.var:te_tax_en_regrel = 1
			}
			if = { limit = { has_modifier = te_tax_relief_region_2 } remove_modifier = te_tax_relief_region_2 }
			if = { limit = { NOT = { has_modifier = te_tax_relief_region_1 } } add_modifier = { name = te_tax_relief_region_1 } }
		}
		else_if = {
			limit = {
				has_variable = te_tax_relief_state
				var:te_tax_relief_state = 1
				scope:te_tax_country.var:te_tax_en_regrel = 2
			}
			if = { limit = { has_modifier = te_tax_relief_region_1 } remove_modifier = te_tax_relief_region_1 }
			if = { limit = { NOT = { has_modifier = te_tax_relief_region_2 } } add_modifier = { name = te_tax_relief_region_2 } }
		}
		else = {
			if = { limit = { has_modifier = te_tax_relief_region_1 } remove_modifier = te_tax_relief_region_1 }
			if = { limit = { has_modifier = te_tax_relief_region_2 } remove_modifier = te_tax_relief_region_2 }
		}
	}
}

# Country scope; needs scope:te_tax_country. The customs schedule's sync (plan
# Task 15; docs/systems/tax_code_schema.md, "Customs schedule"): only while the
# code holds the country's customs (te_tax_customs_authority: the customs
# option, the country owns its market and has taken its customs up), so a
# customs-union member, a subject in its overlord's market and every country
# under the plain rule option are left alone. Per tradeable good and direction
# (te_tax_gen_sync_customs, generated): a level the market lacks is set again,
# and a failed re-assert is counted (te_tax_cretry_<d>_<good>) on the month's
# first customs sync only (te_tax_customs_month: te_tax.4 and the watchdog may
# sync again in the month), so the count is of consecutive monthly syncs.
# Once te_tax_customs_adopt_after (4) monthly re-asserts in a row have not
# taken (a treaty forbids the level, or the 3-month tariff cooldown kept
# refusing it; controller ruling), the market's level is adopted into the code.
# Adoptions happen only on that first sync, so at most once a month they move
# the customs external version once (a change made outside legislation: a bill
# or a package touching customs then holds), te_tax_code_version once, and write
# one history entry (kind 15) carrying how many levels were adopted in its _inst;
# the drift counter keeps the per-good detail. The per-good counter and blocked
# mark (te_tax_cretry_<d>_<good>, te_tax_cblock_<d>_<good>) and the month guard
# (te_tax_customs_month) are transient: not schema tokens and not copied at an
# outbreak, so the rebels count afresh; te_tax_gen_customs_clear_retries drops
# the counters when the market is lost, at a release and at reunification.
te_tax_sync_customs = {
	if = {
		limit = { te_tax_customs_authority = yes }
		set_local_variable = { name = te_tax_cu_counts value = 0 }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = te_tax_customs_month }
					var:te_tax_customs_month < te_history_month_index
				}
			}
			set_local_variable = { name = te_tax_cu_counts value = 1 }
			set_variable = { name = te_tax_customs_month value = te_history_month_index }
		}
		set_local_variable = { name = te_tax_cu_adopted value = 0 }
		te_tax_gen_sync_customs = yes
		if = {
			limit = { local_var:te_tax_cu_adopted > 0 }
			change_variable = { name = te_tax_xver_customs add = 1 }
			change_variable = { name = te_tax_code_version add = 1 }
			te_tax_gen_history_write = { KIND = 15 SLOT = none INST = local_var:te_tax_cu_adopted }
			# Every time for a player's country, the first time only for an AI's
			# (te_tax_detect_drift's rule): the AI may keep moving its tariffs.
			if = {
				limit = { OR = { is_ai = no NOT = { has_variable = te_tax_customs_logged } } }
				debug_log = "TE_TAX customs_adopted tariff levels the code re-asserted for te_tax_customs_adopt_after months in a row did not take (a treaty or the cooldown); the market's levels were adopted into the code (count: the newest history entry); month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
				if = { limit = { is_ai = yes } set_variable = { name = te_tax_customs_logged value = 1 } }
			}
		}
	}
}
