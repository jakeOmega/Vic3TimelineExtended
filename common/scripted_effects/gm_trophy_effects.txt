# ============================================================================
# GRAND MONUMENTS — trophies: scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-10-12-grand-monument-trophies-design.md.
#
# A country can carry off a Grand Monument (or a piece of one) from a state it
# occupies in a war, from its own unincorporated state if it did not raise the
# monument, or by a treaty (Surrender a Monument). What it carries off is a
# TROPHY: a script container tagged gm_trophy, with no parent (it outlives
# every country), listed in the global list gm_trophies:
#   gm_tr_origin_region, gm_tr_origin   the state region and state it came from
#   gm_tr_site_region, gm_tr_site       where it stands: the taker's capital
#   gm_tr_levels                        the levels taken, rebuilt on return
#   gm_tr_grandeur                      half of them, rounded up: what it
#                                       counts for abroad
#   gm_tr_share                         flag: whole, piece or token
#   gm_tr_dedication, gm_tr_skin, gm_tr_form, gm_tr_landmark
#   gm_tr_picture_code                  its monument's picture
#                                       (gm_tr_update_picture)
#   gm_tr_namesake, gm_tr_name_year, gm_tr_name_char, gm_tr_name_occasion,
#   gm_tr_name_enemy_cap, gm_tr_name    its name (gm_trophy_custom_loc.txt)
#   gm_tr_honoree, gm_tr_honoree_ig     To the Leader's records
#   gm_tr_year                          the year it was taken
#   gm_tr_home_notified, gm_tr_reach_by, gm_tr_looter   the pass's notes
# No timed variable on a trophy: it would never expire
# (container_timed_variable_audit).
#
# The holder is the site's owner and the claimant the origin's owner, both
# read live: annexation, a civil war or the capture of the site moves a trophy
# with no special case. The monthly pass (gm_trophy_monthly) writes each
# country's lists and totals from the trophies; the national refresh reads the
# totals (gm_sum_standing, gm_compute_ig).
#
# No taking or return runs where a tooltip walks the effect: each reads back
# what it writes, and a walk logs every such read. A scripted GUI, an event
# option or a treaty only marks the work (gm_state_tr_order_take,
# gm_tr_order_take_site, gm_tr_order_return) and a hidden event does it
# (monument_trophy_events.10-.13), as the opportunities' gift does.

# ---- Ordering the work ---------------------------------------------------------------
# State scope. $WHO$ is to carry this monument off by $ROUTE$ (occupation, own
# or treaty): monument_trophy_events.10 checks again and does it.
gm_state_tr_order_take = {
	set_variable = { name = gm_tr_take_by value = $WHO$ }
	set_variable = { name = gm_tr_take_route value = flag:$ROUTE$ }
	trigger_event = { id = monument_trophy_events.10 }
}

# State scope. The choice, with the tooltip that says what it costs, for the
# event's option and the Carry It Off buttons. The loc reads the state as THIS.
gm_state_tr_choose_take = {
	custom_tooltip = {
		text = gm_tr_take_$ROUTE$_tt
		gm_state_tr_order_take = { WHO = $WHO$ ROUTE = $ROUTE$ }
	}
}

# Container scope. $WHO$ is to carry this trophy off its occupied site
# (monument_trophy_events.11).
gm_tr_order_take_site = {
	save_scope_as = gm_tr_pick
	$WHO$ = { trigger_event = { id = monument_trophy_events.11 } }
}

# Container scope. Its holder sends it home (monument_trophy_events.12).
gm_tr_order_return = {
	save_scope_as = gm_tr_pick
	$WHO$ = { trigger_event = { id = monument_trophy_events.12 } }
}

# ---- Taking a monument (spec §1-§3) ---------------------------------------------------
# State scope, from monument_trophy_events.10, the AI's monthly roll and the
# debug event. $WHO$ carries it off by $ROUTE$. The caller has checked it may.
# The taker needs a capital, where the trophy will stand.
gm_state_tr_take = {
	clear_saved_scope = gm_tr_new
	clear_saved_scope = gm_tr_taker
	clear_saved_scope = gm_tr_victim
	save_scope_as = gm_tr_state
	$WHO$ = { save_scope_as = gm_tr_taker }
	owner = { save_scope_as = gm_tr_victim }
	if = {
		limit = {
			scope:gm_tr_taker = { exists = capital }
		}
		# One monument of the state: a secondary monument named by
		# gm_mon_tr_order_take or the AI's choice (gm_monument_effects.txt), else
		# the primary, as before.
		if = {
			limit = { has_variable = gm_tr_take_target }
			gm_mon_tr_take = yes
		}
		else = {
			set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
			set_variable = { name = gm_tr_take_levels value = gm_state_tr_levels }
			set_variable = { name = gm_tr_take_age value = gm_state_tr_age }
			set_variable = { name = gm_tr_take_start value = gm_state_tr_start }
			set_variable = { name = gm_tr_take_value value = gm_state_tr_value }
			set_variable = { name = gm_tr_take_whole value = gm_state_tr_whole }
			set_variable = { name = gm_tr_take_left value = gm_state_tr_levels_left }
			gm_state_tr_make_trophy = yes
			if = {
				limit = { exists = scope:gm_tr_new }
				if = {
					limit = { var:gm_tr_take_whole = 1 }
					gm_state_tr_remove_whole = yes
				}
				else = {
					gm_state_tr_remove_piece = yes
				}
			}
		}
		if = {
			limit = { exists = scope:gm_tr_new }
			gm_state_tr_consequences_$ROUTE$ = yes
		}
		gm_remove_state_var = { VAR = gm_tr_take_target }
		gm_remove_state_var = { VAR = gm_tr_take_levels }
		gm_remove_state_var = { VAR = gm_tr_take_age }
		gm_remove_state_var = { VAR = gm_tr_take_start }
		gm_remove_state_var = { VAR = gm_tr_take_value }
		gm_remove_state_var = { VAR = gm_tr_take_whole }
		gm_remove_state_var = { VAR = gm_tr_take_left }
	}
}

# State scope (scope:gm_tr_state), from gm_state_tr_take: the trophy, with
# this monument's records. Its site is the taker's capital.
gm_state_tr_make_trophy = {
	create_container = {
		tags = { gm_trophy }
		save_scope_as = gm_tr_new
		on_created = {
			save_scope_as = gm_tr_new
		}
	}
	if = {
		limit = { exists = scope:gm_tr_new }
		add_to_global_variable_list = { name = gm_trophies target = scope:gm_tr_new }
		scope:gm_tr_taker = {
			capital = { save_scope_as = gm_tr_site_state }
		}
		scope:gm_tr_new = {
			set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
			set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
			set_variable = { name = gm_tr_site value = scope:gm_tr_site_state }
			set_variable = { name = gm_tr_site_region value = scope:gm_tr_site_state.state_region }
			set_variable = { name = gm_tr_levels value = scope:gm_tr_state.var:gm_tr_take_levels }
			set_variable = { name = gm_tr_age value = scope:gm_tr_state.var:gm_tr_take_age }
			set_variable = { name = gm_tr_start_grandeur value = scope:gm_tr_state.var:gm_tr_take_start }
			set_variable = { name = gm_tr_grandeur value = scope:gm_tr_state.var:gm_tr_take_value }
			set_variable = { name = gm_tr_year value = gm_this_year }
			set_variable = { name = gm_tr_taken_by value = scope:gm_tr_taker }
			if = {
				limit = {
					scope:gm_tr_state = { var:gm_tr_take_whole = 1 }
				}
				set_variable = { name = gm_tr_share value = flag:whole }
			}
			else_if = {
				limit = {
					scope:gm_tr_state = { gm_state_tr_form_is_piece = yes }
				}
				set_variable = { name = gm_tr_share value = flag:piece }
			}
			else = {
				set_variable = { name = gm_tr_share value = flag:token }
			}
			gm_tr_record_dedication = yes
			gm_tr_copy_from_state = { FROM = gm_skin TO = gm_tr_skin }
			gm_tr_copy_from_state = { FROM = gm_form TO = gm_tr_form }
			gm_tr_copy_from_state = { FROM = gm_landmark TO = gm_tr_landmark }
			gm_tr_copy_from_state = { FROM = gm_namesake TO = gm_tr_namesake }
			gm_tr_copy_from_state = { FROM = gm_name_year TO = gm_tr_name_year }
			gm_tr_copy_from_state = { FROM = gm_name_char TO = gm_tr_name_char }
			gm_tr_copy_from_state = { FROM = gm_name_occasion TO = gm_tr_name_occasion }
			gm_tr_copy_from_state = { FROM = gm_name_enemy_cap TO = gm_tr_name_enemy_cap }
			gm_tr_copy_from_state = { FROM = gm_name TO = gm_tr_name }
			gm_tr_copy_from_state = { FROM = gm_honoree TO = gm_tr_honoree }
			gm_tr_copy_from_state = { FROM = gm_honoree_ig TO = gm_tr_honoree_ig }
			# Its unique dedication, a piece's too: it counts for whoever holds it,
			# at the trophy's grandeur (signatures spec §2.4, §7).
			gm_tr_copy_from_state = { FROM = gm_signature TO = gm_tr_signature }
			gm_tr_copy_from_state = { FROM = gm_sig_parent TO = gm_tr_sig_parent }
			gm_tr_update_picture = yes
		}
	}
}

# Container scope: the state's variable $FROM$, when it has one, as $TO$.
gm_tr_copy_from_state = {
	if = {
		limit = {
			scope:gm_tr_state = { has_variable = $FROM$ }
		}
		set_variable = { name = $TO$ value = scope:gm_tr_state.var:$FROM$ }
	}
}

# Container scope. The trophy's picture: the monument it was taken from, as
# its records describe it (gm_tr_skin, gm_tr_form, gm_tr_landmark), by the
# monuments' own table (gm_picture_effects.txt). The table reads a state's
# names, so the records are lent under them and taken back at once. A piece
# shows its monument. Set when the trophy is made or seeded and again by the
# monthly pass, so a landmark whose picture is drawn later, or a trophy from
# an older save, catches up within a month; gm_tr_disp_picture_code reads the
# draped picture until then. Then the trophy's own picture, when its landmark
# has one (gm_tr_picture_trophies; added 2026-10-16).
gm_tr_update_picture = {
	gm_tr_lend_look = { FROM = gm_tr_skin TO = gm_skin }
	gm_tr_lend_look = { FROM = gm_tr_form TO = gm_form }
	gm_tr_lend_look = { FROM = gm_tr_landmark TO = gm_landmark }
	set_variable = { name = gm_picture_code value = 1 }
	if = {
		limit = { has_variable = gm_form }
		gm_picture_from_look = yes
	}
	gm_state_picture_landmarks = yes
	set_variable = { name = gm_tr_picture_code value = var:gm_picture_code }
	gm_remove_state_var = { VAR = gm_skin }
	gm_remove_state_var = { VAR = gm_form }
	gm_remove_state_var = { VAR = gm_landmark }
	gm_remove_state_var = { VAR = gm_picture_code }
	gm_tr_picture_trophies = yes
}

# Container scope: the trophy's variable $FROM$, when it has one, as $TO$, for
# gm_tr_update_picture.
gm_tr_lend_look = {
	if = {
		limit = { has_variable = $FROM$ }
		set_variable = { name = $TO$ value = var:$FROM$ }
	}
}

# State scope. Carried off whole. Another monument stood beside it: the
# building is the largest's now (gm_state_promote_secondary,
# gm_monument_effects.txt), which takes its records over the taken one's.
# Otherwise the building goes; its records stay until the state's next
# monthly pulse clears them (gm_state_monthly), so the notices raised now can
# still name it, and gm_carried_off keeps that pulse from counting a contested
# monument as torn down, or putting its building back.
gm_state_tr_remove_whole = {
	owner = {
		remove_list_variable = { name = gm_states target = scope:gm_tr_state }
	}
	set_variable = gm_carried_off
	# Its shadow too, now: its records stay a month, but a merge must not raise
	# a monument carried off (issue #936).
	gm_state_drop_shadow = yes
	if = {
		limit = { gm_state_has_secondary = yes }
		gm_state_promote_secondary = yes
	}
	else = {
		set_variable = { name = gm_rebuilding days = 1 }
		remove_building = building_grand_monument
	}
	owner = { trigger_event = { id = monument_events.20 } }
}

# State scope. A piece carried off: what stands keeps the rest of the records
# (sites spec 2026-10-18 §3.3: no rebuild), and the state is spared for ten
# years.
gm_state_tr_remove_piece = {
	set_variable = { name = gm_levels value = var:gm_tr_take_left }
	# The age and starting grandeur that stay with what stands (spec
	# 2026-10-16 §5; sites spec 2026-10-18 §5).
	if = {
		limit = { has_variable = gm_age }
		change_variable = { name = gm_age subtract = var:gm_tr_take_age }
		clamp_variable = { name = gm_age min = 0 }
	}
	if = {
		limit = { has_variable = gm_start_grandeur }
		change_variable = { name = gm_start_grandeur subtract = var:gm_tr_take_start }
		clamp_variable = { name = gm_start_grandeur min = 0 }
	}
	set_variable = { name = gm_grandeur value = gm_state_primary_grandeur }
	set_variable = { name = gm_tr_cooldown days = 3650 }
	owner = { trigger_event = { id = monument_events.20 } }
}

# ---- Consequences (spec §3), state scope (scope:gm_tr_state) ---------------------------
# In a war: infamy, the owner's anger, and its people's resolve.
gm_state_tr_consequences_occupation = {
	scope:gm_tr_new = { set_variable = { name = gm_tr_tmp_g value = gm_mon_grandeur } }
	scope:gm_tr_taker = {
		set_variable = { name = gm_tr_tmp_grandeur value = scope:gm_tr_new.var:gm_tr_tmp_g }
		change_infamy = gm_tr_infamy
		remove_variable = gm_tr_tmp_grandeur
		scope:gm_tr_new = { remove_variable = gm_tr_tmp_g }
		change_relations = { country = scope:gm_tr_victim value = gm_tr_relations_taken }
		every_scope_war = {
			limit = { is_war_participant = scope:gm_tr_victim }
			add_war_war_support = { target = scope:gm_tr_victim value = gm_tr_war_support }
		}
	}
	gm_tr_notify_taking = yes
}

# From our own unincorporated state: its people resent it, and the country we
# took the state from, if it still stands, is angered.
gm_state_tr_consequences_own = {
	add_radicals_in_state = { value = 0.1 }
	if = {
		limit = {
			has_variable = gm_taken_from
			var:gm_taken_from ?= {
				is_country_alive = yes
				NOT = { this = scope:gm_tr_taker }
			}
		}
		var:gm_taken_from = { save_scope_as = gm_tr_former_owner }
		scope:gm_tr_taker = {
			change_relations = { country = scope:gm_tr_former_owner value = gm_tr_relations_taken }
		}
	}
	gm_tr_notify_taking = yes
}

# Surrendered by a treaty.
gm_state_tr_consequences_treaty = {
	scope:gm_tr_taker = {
		change_relations = { country = scope:gm_tr_victim value = gm_tr_relations_taken }
	}
	gm_tr_notify_taking = yes
}

# Any scope. Toasts for a player on either side of a taking.
gm_tr_notify_taking = {
	scope:gm_tr_victim = {
		if = {
			limit = {
				is_ai = no
				NOT = { this = scope:gm_tr_taker }
			}
			post_notification = gm_tr_lost_notice
		}
	}
	scope:gm_tr_taker = {
		if = {
			limit = { is_ai = no }
			post_notification = gm_tr_taken_notice
		}
	}
}

# ---- Taking a trophy from its site (spec §2.1, §5) -------------------------------------
# Container scope, from monument_trophy_events.11 and the AI's monthly roll.
# $WHO$ carries off a trophy on display in a state it occupies. One whose
# origin it owns goes straight home; any other becomes its own trophy,
# standing in its capital, at the price of a taking in war.
gm_tr_take_from_site = {
	clear_saved_scope = gm_tr_taker
	clear_saved_scope = gm_tr_victim
	save_scope_as = gm_tr_moving
	$WHO$ = { save_scope_as = gm_tr_taker }
	var:gm_tr_site ?= {
		owner = { save_scope_as = gm_tr_victim }
	}
	if = {
		limit = { gm_tr_claimed_by = { WHO = scope:gm_tr_taker } }
		gm_tr_return_home = { RELATIONS = no }
	}
	else_if = {
		limit = {
			exists = scope:gm_tr_victim
			scope:gm_tr_taker = { exists = capital }
		}
		scope:gm_tr_moving = { set_variable = { name = gm_tr_tmp_g value = gm_mon_grandeur } }
		scope:gm_tr_taker = {
			capital = { save_scope_as = gm_tr_site_state }
			set_variable = { name = gm_tr_tmp_grandeur value = scope:gm_tr_moving.var:gm_tr_tmp_g }
			change_infamy = gm_tr_infamy
			remove_variable = gm_tr_tmp_grandeur
			scope:gm_tr_moving = { remove_variable = gm_tr_tmp_g }
			change_relations = { country = scope:gm_tr_victim value = gm_tr_relations_taken }
			every_scope_war = {
				limit = { is_war_participant = scope:gm_tr_victim }
				add_war_war_support = { target = scope:gm_tr_victim value = gm_tr_war_support }
			}
		}
		set_variable = { name = gm_tr_site value = scope:gm_tr_site_state }
		set_variable = { name = gm_tr_site_region value = scope:gm_tr_site_state.state_region }
		set_variable = { name = gm_tr_taken_by value = scope:gm_tr_taker }
		gm_tr_forget_notes = yes
		gm_tr_notify_taking = yes
	}
}

# Container scope. The pass's notes belong to the old holder and the old site.
gm_tr_forget_notes = {
	gm_remove_state_var = { VAR = gm_tr_home_notified }
	gm_remove_state_var = { VAR = gm_tr_reach_by }
	gm_remove_state_var = { VAR = gm_tr_looter }
}

# ---- Home again (spec §5) ---------------------------------------------------------------
# Container scope. Rebuilt in its origin state for that state's owner, who
# gets the Homecoming; the trophy is destroyed. RELATIONS = yes when its holder
# hands it back of its own accord or by treaty: relations +40 with the
# receiver if that is another country. The caller runs gm_tr_finish (or the
# monthly pass) afterwards: the lists still name this trophy until then.
gm_tr_return_home = {
	clear_saved_scope = gm_tr_receiver
	clear_saved_scope = gm_tr_giver
	save_scope_as = gm_tr_moving
	var:gm_tr_origin ?= {
		owner = { save_scope_as = gm_tr_receiver }
	}
	var:gm_tr_site ?= {
		owner = { save_scope_as = gm_tr_giver }
	}
	# No return waits any more (sites spec 2026-10-18 §3.3: records, no
	# rebuild); one an older save queued (gm_tr_return_queued) comes home from
	# the monthly pass. A building raised in its origin today (gm_rebuilding)
	# counts as standing (gm_tr_stands_apart), so a second trophy home the same
	# day stands beside the first.
	if = {
		limit = { exists = scope:gm_tr_receiver }
		gm_remove_state_var = { VAR = gm_tr_return_queued }
		gm_remove_state_var = { VAR = gm_tr_return_relations }
		# A piece of a secondary rejoins it; anything else with a monument
		# standing in its origin becomes a monument of its own (spec
		# 2026-10-13 §4.1); a piece of the primary, or with none standing, is
		# the Quadriga's way back and the rebuild.
		if = {
			limit = { gm_tr_rejoins_secondary = yes }
			gm_tr_rejoin_secondary = yes
		}
		else_if = {
			limit = { gm_tr_stands_apart = yes }
			gm_tr_become_monument = yes
		}
		else = {
			var:gm_tr_origin = { gm_state_tr_receive = yes }
		}
		# The Homecoming is for getting back what another took: a country
		# that carried it off itself (from its own colony, or before it
		# conquered the origin) only moves it.
		if = {
			limit = {
				NOT = { var:gm_tr_taken_by ?= { this = scope:gm_tr_receiver } }
			}
			scope:gm_tr_receiver = {
				set_variable = { name = gm_boon_homecoming value = gm_boon_months }
				set_variable = gm_active
				if = {
					limit = { is_ai = no }
					post_notification = gm_tr_homecoming_notice
				}
			}
		}
		if = {
			limit = {
				always = $RELATIONS$
				exists = scope:gm_tr_giver
				NOT = { scope:gm_tr_giver = scope:gm_tr_receiver }
			}
			scope:gm_tr_giver = {
				change_relations = { country = scope:gm_tr_receiver value = gm_tr_relations_returned }
			}
		}
		# A trophy that became a monument is the same container: it stays.
		if = {
			limit = { NOT = { has_tag = gm_monument } }
			remove_list_global_variable = { name = gm_trophies target = scope:gm_tr_moving }
			destroy_container = yes
		}
	}
}

# State scope (the origin state, scope:gm_tr_moving the trophy). A monument
# standing here takes the levels into its records, keeping its own and its
# dedication (the Quadriga goes back on the Gate). With none, a new one stands
# with the trophy's: its look, name, landmark and dedication, recorded as seen
# so it is never read as the engine's copy, and raised by its new owner.
gm_state_tr_receive = {
	if = {
		limit = { has_building = building_grand_monument }
		# The piece is part of the monument again (sites spec 2026-10-18 §3.3).
		if = {
			limit = { NOT = { has_variable = gm_levels } }
			set_variable = { name = gm_levels value = gm_state_primary_levels }
		}
		change_variable = { name = gm_levels add = scope:gm_tr_moving.var:gm_tr_levels }
		# Its age comes back with its levels (spec 2026-10-16 §5); a trophy
		# from an older save has none. Its starting grandeur too.

			gm_state_add_age = { AGE = scope:gm_tr_moving.var:gm_tr_age }
		}
		gm_state_add_start_from_moving = yes
	}
	else = {
		gm_state_tr_rebuild_from_moving = yes
	}
	owner = { trigger_event = { id = monument_events.20 } }
}

# State scope (scope:gm_tr_moving the trophy or monument to stand as this
# state's primary monument). Its records become the state's (sites spec

# stands keeps standing, its method put back to the moving monument's
# dedication (gm_restore_pm, from gm_state_tr_take_records_from_moving); with

gm_state_tr_rebuild_from_moving = {

	set_variable = gm_migrated
}

	# redraws one only once the dedication is back (gm_restore_pm gone), so
	# until then the card showed the undedicated drape (Troy, 2026-10-18).
	gm_tr_copy_to_state = { FROM = gm_tr_picture_code TO = gm_picture_code }
	gm_tr_copy_to_state = { FROM = gm_tr_namesake TO = gm_namesake }
	gm_tr_copy_to_state = { FROM = gm_tr_name_year TO = gm_name_year }

	gm_tr_copy_to_state = { FROM = gm_tr_name_occasion TO = gm_name_occasion }
	gm_tr_copy_to_state = { FROM = gm_tr_name_enemy_cap TO = gm_name_enemy_cap }
	gm_tr_copy_to_state = { FROM = gm_tr_name TO = gm_name }
	gm_tr_copy_to_state = { FROM = gm_tr_honoree TO = gm_honoree }
	gm_tr_copy_to_state = { FROM = gm_tr_honoree_ig TO = gm_honoree_ig }
	gm_tr_copy_to_state = { FROM = gm_tr_dedication TO = gm_restore_pm }

	gm_tr_copy_to_state = { FROM = gm_tr_faith TO = gm_faith }
	gm_tr_copy_to_state = { FROM = gm_tr_contested TO = gm_contested }
	gm_tr_copy_to_state = { FROM = gm_tr_heritage TO = gm_heritage }
	gm_tr_copy_to_state = { FROM = gm_tr_base_ig TO = gm_base_ig }

	# never an older save's monument to migrate (sites spec 2026-10-18 §3.5).

		set_variable = { name = gm_raised_by value = scope:gm_tr_moving.var:gm_tr_raised_by }
	}

	}

# ---- The state's monthly part (from gm_state_monthly, ROOT = the state) ------------------
# The owner stamped (on_state_owner_change names no previous owner); the
# builder recorded for a monument from an older save; then an occupier's
# chance: a player's notice once in each occupation, the AI's monthly roll; and
# the AI owner's own carrying home from an unincorporated state.
gm_state_tr_monthly = {
	set_variable = { name = gm_holder value = owner }
	if = {
		limit = { has_variable = gm_seen }
		gm_state_record_builder = yes
	}
	# The primary or a monument beside it (gm_monument_triggers.txt). A player's
	# offer (.1) names the primary, so it comes only while the primary can go;
	# a secondary has its own Carry It Off on the state card.
	if = {
		limit = { gm_state_tr_any_takeable = yes }
		clear_saved_scope = gm_tr_occupier
		gm_state_tr_find_occupier = yes
		if = {
			limit = {
				exists = scope:gm_tr_occupier
				gm_state_tr_occupied_by = { WHO = scope:gm_tr_occupier }
			}
			if = {
				limit = {
					scope:gm_tr_occupier = { is_ai = yes }
				}
				scope:gm_tr_occupier = { save_scope_as = gm_tr_taker }
				random = {
					chance = gm_tr_ai_loot_chance
					gm_state_tr_ai_choose_target = yes
					gm_state_tr_take = { WHO = scope:gm_tr_occupier ROUTE = occupation }
				}
			}
			else_if = {
				limit = {
					gm_state_tr_takeable = yes
					NOT = {
						has_variable = gm_tr_offered_to
						var:gm_tr_offered_to ?= { this = scope:gm_tr_occupier }
					}
				}
				set_variable = { name = gm_tr_offered_to value = scope:gm_tr_occupier }
				save_scope_as = monument_state
				owner = { save_scope_as = monument_owner }
				scope:gm_tr_occupier = { trigger_event = { id = monument_trophy_events.1 } }
			}
		}
		else = {
			gm_remove_state_var = { VAR = gm_tr_offered_to }
		}
	}
	if = {
		limit = {
			gm_state_tr_takeable = yes
			owner = {
				is_ai = yes
				save_temporary_scope_as = gm_tmp_tr_owner
			}
			gm_state_tr_own_takeable_by = { WHO = scope:gm_tmp_tr_owner }
			OR = {
				owner = { ig:ig_armed_forces ?= { is_in_government = yes } }
				gm_state_grandeur >= 10
			}
		}
		owner = { save_scope_as = gm_tr_own_taker }
		random = {
			chance = gm_tr_ai_own_chance
			gm_state_tr_take = { WHO = scope:gm_tr_own_taker ROUTE = own }
		}
	}
}

# ---- Self-check (once a game) -------------------------------------------------------
# State scope, from gm_state_monthly. The first monument state whose owner is
# at peace: its owner holds every one of its provinces, so the occupation read
# (gm_state_tr_holds_half, a script value over the provinces) must count at
# least one and find the owner holding half. Runs in a save already under way.
gm_tr_occupation_read_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = gm_occupation_read_checked }
			has_building = building_grand_monument
			owner = { is_at_war = no }
		}
		set_global_variable = gm_occupation_read_checked
		if = {
			limit = {
				gm_state_province_count >= 1
				gm_state_tr_holds_half = { WHO = owner }
			}
			debug_log = "TE_SELF_CHECK OK gm_occupation_read (a state at peace counts [SCOPE.ScriptValue('gm_state_province_count')|0] provinces and its owner holds half of them, so an occupier can be found)"
		}
		else_if = {
			limit = { gm_state_province_count >= 1 }
			debug_log = "TE_SELF_CHECK BROKEN gm_occupation_read (a state at peace counts [SCOPE.ScriptValue('gm_state_province_count')|0] provinces but its owner holds under half: the province count can't see the saved country; see docs/testing/grand-monuments-multiple-playtest.md)"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN gm_occupation_read (a state counts no provinces: a script value can't iterate a state's provinces either, so no state reads as occupied; see docs/testing/grand-monuments-multiple-playtest.md)"
		}
	}
}

# ---- The monthly pass (spec §4), from on_monthly_pulse --------------------------------------
# No scope. Repair each trophy's places, then the AI's decisions (collected
# first: a return destroys a trophy, a taking moves one), a player's notes,
# and the lists.
gm_trophy_monthly = {
	if = {
		limit = {
			gm_system_enabled = yes
			has_global_variable_list = gm_trophies
		}
		every_in_global_list = {
			variable = gm_trophies
			gm_tr_repair_places = yes
			gm_tr_update_picture = yes
			# A trophy of 1836, or one from a save older than signatures, takes
			# its landmark's unique dedication (signatures spec §4.4).
			gm_tr_sig_backfill = yes
		}
		# Returns that waited for a rebuild (gm_tr_return_home): one to each
		# state now; another bound for the same state waits again.
		every_in_global_list = {
			variable = gm_trophies
			limit = { has_variable = gm_tr_return_queued }
			add_to_temporary_list = gm_tr_queued_home
		}
		every_in_list = {
			list = gm_tr_queued_home
			if = {
				limit = { has_variable = gm_tr_return_relations }
				gm_tr_return_home = { RELATIONS = yes }
			}
			else = {
				gm_tr_return_home = { RELATIONS = no }
			}
		}
		every_in_global_list = {
			variable = gm_trophies
			limit = { NOT = { has_variable = gm_tr_return_queued } }
			gm_tr_monthly_one = yes
		}
		every_in_list = {
			list = gm_tr_ai_home
			gm_tr_return_home = { RELATIONS = yes }
		}
		every_in_list = {
			list = gm_tr_ai_loot
			clear_saved_scope = gm_tr_loot_by
			var:gm_tr_looter ?= { save_scope_as = gm_tr_loot_by }
			remove_variable = gm_tr_looter
			if = {
				limit = { exists = scope:gm_tr_loot_by }
				gm_tr_take_from_site = { WHO = scope:gm_tr_loot_by }
			}
		}
	}
	gm_trophy_pass = yes
}

# Container scope. A state that has stopped existing (a region's parts merged)
# is found again in its region: the most populous state there. A monument
# standing beside another goes back to the state whose building holds its

gm_tr_repair_places = {

			NOT = { exists = var:gm_tr_origin }
			has_variable = gm_tr_origin_region
		}
		gm_mon_repair_origin = yes
	}

			NOT = { exists = var:gm_tr_origin }
			has_variable = gm_tr_origin_region
		}
		clear_saved_scope = gm_tr_repair_state
		var:gm_tr_origin_region = {
			ordered_scope_state = {
				order_by = state_population
				position = 0
				check_range_bounds = no
				save_scope_as = gm_tr_repair_state
			}
		}
		if = {
			limit = { exists = scope:gm_tr_repair_state }
			set_variable = { name = gm_tr_origin value = scope:gm_tr_repair_state }
		}
	}
	if = {
		limit = {
			NOT = { exists = var:gm_tr_site }
			has_variable = gm_tr_site_region
		}
		clear_saved_scope = gm_tr_repair_state
		var:gm_tr_site_region = {
			ordered_scope_state = {
				order_by = state_population
				position = 0
				check_range_bounds = no
				save_scope_as = gm_tr_repair_state
			}
		}
		if = {
			limit = { exists = scope:gm_tr_repair_state }
			set_variable = { name = gm_tr_site value = scope:gm_tr_repair_state }
		}
	}
}

# Container scope, once a month. An AI holder brings home what it owns the
# origin of, unless it carried it off itself (gm_tr_held_by_its_taker), and
# returns to an ally or bloc partner at gm_tr_ai_return_chance.
# An occupier of the site: the AI takes its own home, and carries off any
# other at gm_tr_ai_loot_chance_trophy; a player is told once per
# occupation. A player holder is told once that it can bring one home.
gm_tr_monthly_one = {
	clear_saved_scope = gm_tr_ai_holder
	clear_saved_scope = gm_tr_ai_claimant
	var:gm_tr_site ?= {
		owner = { save_scope_as = gm_tr_ai_holder }
	}
	var:gm_tr_origin ?= {
		owner = { save_scope_as = gm_tr_ai_claimant }
	}
	if = {
		limit = {
			exists = scope:gm_tr_ai_holder
			scope:gm_tr_ai_holder = { is_ai = yes }
		}
		# Its own back, unless it carried it off itself: an AI that took a
		# monument from its own colony would otherwise bring it straight back.
		if = {
			limit = { gm_tr_holder_owns_origin = yes }
			if = {
				limit = { gm_tr_held_by_its_taker = no }
				add_to_temporary_list = gm_tr_ai_home
			}
		}
		else_if = {
			limit = {
				exists = scope:gm_tr_ai_claimant
				scope:gm_tr_ai_holder = { gm_tr_ai_returns_to = { WHO = scope:gm_tr_ai_claimant } }
			}
			random = {
				chance = gm_tr_ai_return_chance
				add_to_temporary_list = gm_tr_ai_home
			}
		}
	}
	else_if = {
		limit = {
			exists = scope:gm_tr_ai_holder
			gm_tr_holder_owns_origin = yes
			gm_tr_held_by_its_taker = no
			NOT = { has_variable = gm_tr_home_notified }
		}
		set_variable = gm_tr_home_notified
		scope:gm_tr_ai_holder = { post_notification = gm_tr_home_reach_notice }
	}
	if = {
		limit = {
			has_variable = gm_tr_home_notified
			gm_tr_holder_owns_origin = no
		}
		remove_variable = gm_tr_home_notified
	}
	# An occupier of the site.
	if = {
		limit = {
			has_variable = gm_tr_site
			exists = var:gm_tr_site
			NOT = { is_in_list = gm_tr_ai_home }
		}
		clear_saved_scope = gm_tr_occupier
		var:gm_tr_site = { gm_state_tr_find_occupier = yes }
		if = {
			limit = {
				exists = scope:gm_tr_occupier
				gm_tr_site_occupied_by = { WHO = scope:gm_tr_occupier }
			}
			if = {
				limit = {
					scope:gm_tr_occupier = { is_ai = yes }
				}
				scope:gm_tr_occupier = { save_scope_as = gm_tr_taker }
				if = {
					limit = { gm_tr_claimed_by = { WHO = scope:gm_tr_occupier } }
					set_variable = { name = gm_tr_looter value = scope:gm_tr_occupier }
					add_to_temporary_list = gm_tr_ai_loot
				}
				else = {
					random = {
						chance = gm_tr_ai_loot_chance_trophy
						set_variable = { name = gm_tr_looter value = scope:gm_tr_occupier }
						add_to_temporary_list = gm_tr_ai_loot
					}
				}
			}
			else_if = {
				limit = {
					NOT = {
						has_variable = gm_tr_reach_by
						var:gm_tr_reach_by ?= { this = scope:gm_tr_occupier }
					}
				}
				set_variable = { name = gm_tr_reach_by value = scope:gm_tr_occupier }
				scope:gm_tr_occupier = { post_notification = gm_tr_site_reach_notice }
			}
		}
		else = {
			gm_remove_state_var = { VAR = gm_tr_reach_by }
		}
	}
}

# Any scope. Every country's and state's trophy lists and totals, from the
# trophies (spec §4). Last month's are cleared first; lists are rebuilt each
# time, so a civil war's winner (who inherits no lists) has its rows back.
gm_trophy_pass = {
	# Nothing to do in a game no trophy has touched: gm_tr_lists_written
	# stands while any list does.
	if = {
		limit = {
			OR = {
				has_global_variable = gm_tr_lists_written
				has_global_variable_list = gm_trophies
			}
		}
		every_country = {
			limit = { has_variable = gm_tr_listed }
			clear_variable_list = gm_trophies_held
			clear_variable_list = gm_trophies_abroad
			set_variable = { name = gm_tr_held_count value = 0 }
			set_variable = { name = gm_tr_held_grandeur value = 0 }
			set_variable = { name = gm_tr_abroad_count value = 0 }
			remove_variable = gm_tr_listed
		}
		every_state = {
			limit = { has_variable = gm_tr_here_count }
			clear_variable_list = gm_trophies_here
			remove_variable = gm_tr_here_count
			gm_remove_state_var = { VAR = gm_tr_here_grandeur }
		}
		if = {
			limit = { has_global_variable = gm_tr_lists_written }
			remove_global_variable = gm_tr_lists_written
		}
		if = {
			limit = { has_global_variable_list = gm_trophies }
			every_in_global_list = {
				variable = gm_trophies
				gm_tr_list_one = yes
			}
		}
	}
}

# Container scope: this trophy into its site's, holder's and claimant's lists.
gm_tr_list_one = {
	save_scope_as = gm_tr_listing
	if = {
		limit = {
			has_variable = gm_tr_site
			exists = var:gm_tr_site
		}
		var:gm_tr_site = {
			add_to_variable_list = { name = gm_trophies_here target = scope:gm_tr_listing }
			gm_tr_count_add = { VAR = gm_tr_here_count ADD = 1 }
			gm_tr_count_add = { VAR = gm_tr_here_grandeur ADD = scope:gm_tr_listing.var:gm_tr_grandeur }
			owner = {
				gm_tr_country_listed = yes
				add_to_variable_list = { name = gm_trophies_held target = scope:gm_tr_listing }
				change_variable = { name = gm_tr_held_count add = 1 }
				change_variable = { name = gm_tr_held_grandeur add = scope:gm_tr_listing.var:gm_tr_grandeur }
			}
		}
	}
	if = {
		limit = {
			has_variable = gm_tr_origin
			exists = var:gm_tr_origin
			gm_tr_holder_owns_origin = no
		}
		var:gm_tr_origin = {
			owner = {
				gm_tr_country_listed = yes
				add_to_variable_list = { name = gm_trophies_abroad target = scope:gm_tr_listing }
				change_variable = { name = gm_tr_abroad_count add = 1 }
			}
		}
	}
}

# Country scope: listed this month, with its counts started at 0, and its
# national pulse running (the entry stays while it holds or misses a trophy).
gm_tr_country_listed = {
	if = {
		limit = {
			NOT = { has_global_variable = gm_tr_lists_written }
		}
		set_global_variable = gm_tr_lists_written
	}
	if = {
		limit = {
			NOT = { has_variable = gm_tr_listed }
		}
		set_variable = gm_tr_listed
		set_variable = { name = gm_tr_held_count value = 0 }
		set_variable = { name = gm_tr_held_grandeur value = 0 }
		set_variable = { name = gm_tr_abroad_count value = 0 }
	}
	set_variable = gm_active
}

# Any scope: add $ADD$ to the variable $VAR$, starting it at 0.
gm_tr_count_add = {
	if = {
		limit = { has_variable = $VAR$ }
		change_variable = { name = $VAR$ add = $ADD$ }
	}
	else = {
		set_variable = { name = $VAR$ value = $ADD$ }
	}
}

# The trophy of the landmark $LANDMARK$, taken in $YEAR$ from $ORIGIN_TAG$'s
# part of $ORIGIN$ and standing in $SITE_TAG$'s part of $SITE$, made as a taking
# makes one (gm_state_tr_make_trophy). Places nothing, and says so in
# debug.log, when either part is not the named owner's, when one country owns
# both (the monthly pass would bring it home at once), or when a trophy of this
# landmark already exists (the debug event run twice).
gm_seed_trophy = {
	clear_saved_scope = gm_tr_state
	clear_saved_scope = gm_tr_site_state
	clear_saved_scope = gm_tr_seed_holder
	clear_saved_scope = gm_tr_new
	s:$ORIGIN$.region_state:$ORIGIN_TAG$ ?= { save_scope_as = gm_tr_state }
	s:$SITE$.region_state:$SITE_TAG$ ?= {
		save_scope_as = gm_tr_site_state
		owner = { save_scope_as = gm_tr_seed_holder }
	}
	if = {
		limit = {
			exists = scope:gm_tr_state
			exists = scope:gm_tr_site_state
			scope:gm_tr_state = {
				NOT = { owner = scope:gm_tr_seed_holder }
			}
			NOT = {
				AND = {
					has_global_variable_list = gm_trophies
					any_in_global_list = {
						variable = gm_trophies
						has_variable = gm_tr_landmark
						var:gm_tr_landmark = flag:$LANDMARK$
					}
				}
			}
		}
		scope:gm_tr_site_state = {
			create_container = {
				tags = { gm_trophy }
				save_scope_as = gm_tr_new
				on_created = {
					save_scope_as = gm_tr_new
				}
			}
		}
		if = {
			limit = { exists = scope:gm_tr_new }
			add_to_global_variable_list = { name = gm_trophies target = scope:gm_tr_new }
			scope:gm_tr_new = {
				set_variable = { name = gm_tr_origin value = scope:gm_tr_state }
				set_variable = { name = gm_tr_origin_region value = scope:gm_tr_state.state_region }
				set_variable = { name = gm_tr_site value = scope:gm_tr_site_state }
				set_variable = { name = gm_tr_site_region value = scope:gm_tr_site_state.state_region }
				set_variable = { name = gm_tr_levels value = $LEVELS$ }
				# The 1836 rate, as the landmarks are seeded at
				# (gm_seed_landmark); abroad, half of it, rounded up
				# (gm_state_tr_value).
				set_variable = {
					name = gm_tr_start_grandeur
					value = {
						value = $LEVELS$
						multiply = gm_level_grandeur_1836
					}
				}
				set_variable = {
					name = gm_tr_grandeur
					value = {
						value = $LEVELS$
						multiply = gm_level_grandeur_1836
						divide = 2
						ceiling = yes
					}
				}
				set_variable = { name = gm_tr_year value = $YEAR$ }
				set_variable = { name = gm_tr_taken_by value = scope:gm_tr_seed_holder }
				set_variable = { name = gm_tr_share value = flag:$SHARE$ }
				set_variable = { name = gm_tr_dedication value = flag:$DEDICATION$ }
				set_variable = { name = gm_tr_skin value = flag:landmark_$LANDMARK$ }
				set_variable = { name = gm_tr_form value = flag:$FORM$ }
				set_variable = { name = gm_tr_landmark value = flag:$LANDMARK$ }
				set_variable = { name = gm_tr_namesake value = flag:landmark }
				gm_tr_sig_backfill = yes
				gm_tr_update_picture = yes
			}
			debug_log = "TE_MONUMENTS: trophy $LANDMARK$ placed: from $ORIGIN_TAG$'s part of $ORIGIN$, standing in $SITE_TAG$'s part of $SITE$, $LEVELS$ levels"
		}
	}
	else = {
		debug_log = "TE_MONUMENTS: trophy $LANDMARK$ not placed: $ORIGIN$ not held by $ORIGIN_TAG$, $SITE$ not held by $SITE_TAG$, one country holds both, or it is already abroad"
	}
}
