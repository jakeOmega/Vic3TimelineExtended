# ============================================================================
# UN CONVENTION REGIMES, SOVEREIGNTY AND INTELLIGENCE — THE WRITERS (phase 5)
# ============================================================================
# docs/systems/un_redesign_design.md §5.3, §7.1 and §7.4; as shipped, §0.5.
# The table of who wins and who pays is at the top of
# common/script_values/un_regime_values.txt; the tests are in
# common/scripted_triggers/un_regime_triggers.txt.
#
# WHAT A COUNTRY'S "TERMS" ARE. For a member:
#   - the member modifier of every convention it has ratified, multiplied by
#     the UN's enforcement E (declared by je_united_nations while the
#     convention's mirror is set, so it follows E within days; nothing here
#     touches it);
#   - the winner or loser modifier each convention it has RATIFIED gives it
#     (§5.3). A member that refused to ratify (un_vote.3's "Refuse to ratify
#     it", PR #422) stays outside the convention: neither its member modifier
#     nor its terms. The decolonisation declaration is a decision, not a
#     convention, and binds every member;
#   - from the Strong tier, shared intelligence (§7.4) and the reaction of its
#     interest groups (§7.1): nationalists resent the UN, humanitarians and
#     pacifists welcome it.
# For a colony of a member, the decolonisation declaration's liberty desire —
# whether or not the colony is itself a member (§0.8: an overseas protectorate
# of another heritage can be one, and then gets both). For anyone else,
# nothing.
#
# WHO WRITES WHAT
#   un_regime_*_modifier_on (country)      un_regime_refresh_country, the single
#                                          refresh site (un_regime_term_on);
#                                          un_regime_clear_country clears them
#                                          all. je_united_nations declares each
#                                          un_regime_*_modifier from its variable
#                                          at its multiplier (#1404)
#   un_sovereignty_*_ig_modifier (IG)      un_regime_refresh_country;
#                                          un_regime_clear_country removes them
#   var:un_regime_stamp (country)          un_regime_refresh_country
#   un_regime_epoch / _months / _tier      un_regime_monthly_update, un_regime_bump
#   un_space_top_score (global)            un_regime_bump
#   un_regime_decolonization (global)      un_vote.2, the declaration carried
#   un_sov_leave_demand_cd (country)       un_regime_sovereignty_leave_check
#   un_icc_indicted_cd (country)           un_regime_icc_note_crime
#
# CALL SITES
#   un_regime_monthly_update       un_global_authority_on_action (UN founded)
#   un_regime_country_pulse        je_united_nations on_monthly_pulse (everyone)
#   un_regime_refresh_country      the pulse above; un_vote_apply_treaty_modifier
#                                  (ratifying); un_apply_ratified_conventions
#                                  (joining); un_events.36 option A (ratifying
#                                  what passed while our voice was suspended)
#   un_regime_clear_country        un_membership_end_effect (leaving, dissolving)
#   un_regime_icc_note_crime       the first-strike and tactical nuclear effects
#                                  (extra_effects.txt), covert_warfare.1
# ============================================================================

# ---- Monthly ------------------------------------------------------------------

# Global monthly upkeep (no-country scope, UN founded). Every country re-reads
# its terms when the tier changes and once a year.
un_regime_monthly_update = {
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = un_regime_epoch }
				NOT = { has_global_variable = un_regime_tier }
				NOT = { global_var:un_regime_tier = un_disp_tier }
				global_var:un_regime_months >= un_regime_refresh_months
			}
		}
		un_regime_bump = yes
	}
	else = {
		change_global_variable = { name = un_regime_months add = 1 }
	}
}

# Tell every country to re-read its terms at its next monthly pulse, and take
# the space race's standings for the outer-space regime.
# Scope: any
un_regime_bump = {
	if = {
		limit = { NOT = { has_global_variable = un_regime_epoch } }
		set_global_variable = { name = un_regime_epoch value = 0 }
	}
	change_global_variable = { name = un_regime_epoch add = 1 }
	set_global_variable = { name = un_regime_months value = 0 }
	set_global_variable = { name = un_regime_tier value = un_disp_tier }
	set_global_variable = { name = un_space_top_score value = 0 }
	ordered_country = {
		order_by = un_space_milestones
		max = 1
		set_global_variable = { name = un_space_top_score value = un_space_milestones }
	}
}

# Scope: country (every holder of the journal entry, monthly)
un_regime_country_pulse = {
	if = {
		limit = {
			has_global_variable = un_regime_epoch
			OR = {
				NOT = { has_variable = un_regime_stamp }
				NOT = { var:un_regime_stamp = global_var:un_regime_epoch }
			}
		}
		un_regime_refresh_country = yes
		# Once a year, at Strong and above: whether our resentful interest
		# groups demand we leave (un_events.34).
		un_regime_sovereignty_leave_check = yes
	}
}

# ---- Reading a country's terms ------------------------------------------------------

# Clear and re-apply everything the regimes do to this country.
#
# Every convention is read from its mirror (<modifier>_on), which is the
# ratification: the entry declares the member modifier from it. Ratifying,
# joining and un_events.36 set the mirror in the very block that calls this,
# and a variable is visible there at once. Membership is read from the
# modifier OR its mirror (un_state_record): joining adds the modifier in that
# same block, where has_modifier cannot see it yet; the stamp below then keeps
# the monthly pulse from reading our terms again until the next
# un_regime_bump, up to un_regime_refresh_months later. On the pulse itself
# un_state_reconcile has just made the mirror match the modifier, so the
# mirror changes nothing there.
# Scope: country
un_regime_refresh_country = {
	un_regime_clear_country = yes
	if = {
		limit = { has_global_variable = un_regime_epoch }
		set_variable = { name = un_regime_stamp value = global_var:un_regime_epoch }
	}
	if = {
		limit = {
			has_global_variable = un_founded
			OR = {
				je:je_united_nations ?= { has_modifier = un_member_modifier }
				has_variable = un_member_modifier_on
			}
		}
		# Climate (UNEP): the high emitters pay, the low emitters are helped.
		# Not under the disabled global warming rule, which stops the
		# emissions and temperature the terms are read from.
		if = {
			limit = {
				has_game_rule = global_warming_enabled
				has_global_variable = un_agency_unep
				has_variable = un_climate_binding_modifier_on
			}
			if = {
				limit = { un_regime_climate_emitter = yes }
				un_regime_term_on = { TERM = climate_emitter MULT = un_enforcement }
			}
			else_if = {
				limit = {
					un_regime_climate_low_emitter = yes
					un_regime_adaptation_multiplier > 0
				}
				un_regime_term_on = { TERM = climate_adaptation MULT = un_regime_adaptation_multiplier }
			}
		}
		# Non-proliferation (IAEA): inspections for the threshold states, a
		# guarantee for those without the bomb.
		if = {
			limit = {
				has_global_variable = un_agency_iaea
				has_variable = un_nonproliferation_modifier_on
			}
			if = {
				limit = { un_regime_npt_threshold = yes }
				un_regime_term_on = { TERM = npt_inspection MULT = un_enforcement }
			}
			else_if = {
				limit = { un_regime_npt_guaranteed = yes }
				un_regime_term_on = { TERM = npt_guarantee MULT = un_enforcement }
			}
		}
		# The law of the sea (ITLOS): the naval powers are held to it.
		if = {
			limit = {
				has_global_variable = un_agency_itlos
				has_variable = un_law_of_sea_modifier_on
				un_regime_naval_power = yes
			}
			un_regime_term_on = { TERM = naval_curb MULT = un_enforcement }
		}
		# Human rights (UNHRC): the violators pay.
		if = {
			limit = {
				has_global_variable = un_agency_unhrc
				has_variable = un_human_rights_declaration_modifier_on
				un_regime_human_rights_violator = yes
			}
			un_regime_term_on = { TERM = rights_violator MULT = un_enforcement }
		}
		# Decolonisation: the colonial powers pay.
		if = {
			limit = {
				has_global_variable = un_regime_decolonization
				un_regime_colonial_power = yes
			}
			un_regime_term_on = { TERM = colonial_pressure MULT = un_enforcement }
		}
		# Refugees (UNHCR): the hosts take the strain, the sources the relief.
		if = {
			limit = {
				has_global_variable = un_agency_unhcr
				has_variable = un_refugee_program_modifier_on
			}
			if = {
				limit = { un_regime_refugee_host = yes }
				un_regime_term_on = { TERM = refugee_host MULT = un_enforcement }
			}
			else_if = {
				limit = { un_regime_refugee_source = yes }
				un_regime_term_on = { TERM = refugee_source MULT = un_enforcement }
			}
		}
		# Outer space (UNOOSA): the leader shares, the laggards gain, and a
		# battlestation is barred.
		if = {
			limit = {
				has_global_variable = un_agency_unoosa
				has_variable = un_space_partnership_modifier_on
			}
			if = {
				limit = { un_regime_space_leader = yes }
				un_regime_term_on = { TERM = space_leader MULT = un_enforcement }
			}
			else_if = {
				limit = { un_regime_space_laggard = yes }
				un_regime_term_on = { TERM = space_laggard MULT = un_enforcement }
			}
			if = {
				limit = { un_regime_space_weaponized = yes }
				un_regime_term_on = { TERM = space_weapon MULT = un_enforcement }
			}
		}
		# Cultural heritage (UNESCO): the wonder states gain and are bound.
		if = {
			limit = {
				has_global_variable = un_agency_unesco
				has_variable = un_heritage_program_modifier_on
				un_regime_heritage_site = yes
			}
			un_regime_term_on = { TERM = heritage_site MULT = un_enforcement }
		}
		# Cultural diversity (CCD, redesign phase 7): the world's cultural
		# hegemon, if a party, projects less cultural pull. The parties'
		# own terms are in the member modifier (un_cultural_diversity_modifier).
		if = {
			limit = {
				has_global_variable = un_agency_ccd
				has_variable = un_cultural_diversity_modifier_on
				un_regime_cultural_hegemon = yes
			}
			un_regime_term_on = { TERM = cultural_hegemon MULT = un_enforcement }
		}
		# The Antarctic Treaty (ATS, the Global Commons §4): a party that runs
		# an Antarctic research station shares its science. Its other terms are
		# read as checks (te_seabed_southern_barred) and its Policy grant is
		# applied at x1 (te_antarctic_policy_refresh).
		if = {
			limit = {
				has_global_variable = un_agency_ats
				has_variable = un_antarctic_treaty_modifier_on
				te_antarctic_has_station = yes
			}
			un_regime_term_on = { TERM = antarctic_station MULT = un_enforcement }
		}
		# The anti-satellite moratorium (UNODA, the Global Commons §6.5) has
		# no regime modifier: its terms are the strike's doubled costs
		# (te_asat_moratorium_party) and the cleaners' Delivery entries
		# (te_asat_moratorium_delivery_monthly).
		# The Convention on the Settlement of Outer Space (COPUOS, solar
		# colonization fleets § 13) has no regime modifier either: its terms
		# are the fleets' doubled prices of force (sr_colony_convention_party).
		# The prohibition of nuclear weapons (TPNW, redesign phase 7) has no
		# regime modifier: its terms are the ceiling the nuclear entry keeps
		# (nd_taboo_refresh_ban_ceiling, nuclear_taboo_effects.txt) and the
		# taboo it raises (nd_taboo_part_un_value).
		# Intelligence (§7.4): from the Strong tier, members share it.
		if = {
			limit = { un_regime_shared_intelligence = yes }
			un_regime_term_on = { TERM = shared_intelligence MULT = un_regime_strong_steps }
		}
		# Sovereignty (§7.1): from the Strong tier, our interest groups react.
		if = {
			limit = { un_tier_at_least_strong = yes }
			every_interest_group = {
				limit = { un_regime_ig_resents_un = yes }
				add_modifier = {
					name = un_sovereignty_resentment_ig_modifier
					multiplier = un_regime_strong_steps
				}
			}
			every_interest_group = {
				limit = {
					un_regime_ig_welcomes_un = yes
					NOT = { un_regime_ig_resents_un = yes }
				}
				add_modifier = {
					name = un_sovereignty_welcome_ig_modifier
					multiplier = un_regime_strong_steps
				}
			}
		}
	}
	# Decolonisation reaches the colonies of members through their overlord's
	# membership, whether or not the colony holds a seat of its own (§0.8
	# ruling 13).
	if = {
		limit = {
			has_global_variable = un_founded
			has_global_variable = un_regime_decolonization
			un_regime_member_colony = yes
		}
		un_regime_term_on = { TERM = colony_liberty MULT = un_enforcement }
	}
}

# End every regime term of this country (their presence variables) and
# remove the sovereignty reaction from its interest groups.
# The convention member modifiers are not touched here: the entry declares
# them from the mirrors that ratifying sets and leaving clears
# (un_conventions_end).
# Scope: country
un_regime_clear_country = {
	un_regime_term_off = { TERM = climate_emitter }
	un_regime_term_off = { TERM = climate_adaptation }
	un_regime_term_off = { TERM = npt_inspection }
	un_regime_term_off = { TERM = npt_guarantee }
	un_regime_term_off = { TERM = naval_curb }
	un_regime_term_off = { TERM = rights_violator }
	un_regime_term_off = { TERM = colonial_pressure }
	un_regime_term_off = { TERM = colony_liberty }
	un_regime_term_off = { TERM = refugee_host }
	un_regime_term_off = { TERM = refugee_source }
	un_regime_term_off = { TERM = space_leader }
	un_regime_term_off = { TERM = space_laggard }
	un_regime_term_off = { TERM = space_weapon }
	un_regime_term_off = { TERM = heritage_site }
	un_regime_term_off = { TERM = cultural_hegemon }
	un_regime_term_off = { TERM = antarctic_station }
	un_regime_term_off = { TERM = shared_intelligence }
	every_interest_group = {
		limit = {
			OR = {
				has_modifier = un_sovereignty_resentment_ig_modifier
				has_modifier = un_sovereignty_welcome_ig_modifier
			}
		}
		un_regime_clear_modifier = { MODIFIER = un_sovereignty_resentment_ig_modifier }
		un_regime_clear_modifier = { MODIFIER = un_sovereignty_welcome_ig_modifier }
	}
}

# One of our terms: je_united_nations declares un_regime_$TERM$_modifier at
# its multiplier while the presence variable is set (#1404). The refresh
# still decides the terms once an epoch (un_regime_country_pulse: the tier
# changed, or a year passed), on purpose: a term that came and went with
# every emissions or rank flicker would be noise. The multiplier is read
# live. The line previews the modifier where a tooltip shows it (un_events.36
# option A); $MULT$ names the value the declaration reads.
# Parameters: $TERM$ = the term (un_regime_<term>_modifier), $MULT$ = its multiplier
# Scope: country
un_regime_term_on = {
	show_as_tooltip = {
		je:je_united_nations ?= { add_modifier = { name = un_regime_$TERM$_modifier multiplier = $MULT$ } }
	}
	set_variable = un_regime_$TERM$_modifier_on
}

# Parameters: $TERM$
# Scope: country
un_regime_term_off = {
	legacy_scaling_strip_united_nations = yes		# PENDING REMOVAL after 2026-12-31 (#1404): an older save's copy goes first, so the strip can't set this mirror again
	if = {
		limit = { has_variable = un_regime_$TERM$_modifier_on }
		remove_variable = un_regime_$TERM$_modifier_on
	}
}

# Parameters: $MODIFIER$
# Scope: interest group (the sovereignty reaction, which no entry can declare)
un_regime_clear_modifier = {
	if = {
		limit = { has_modifier = $MODIFIER$ }
		remove_modifier = $MODIFIER$
	}
}

# ---- Sovereignty: the demand to leave (§7.1) ---------------------------------------------

# Once a year, at Strong and above: when the interest groups that resent the
# UN are strong enough (un_regime_sovereignty_demand), they demand that the
# country leave (un_events.34). At most once a decade.
# Scope: country
un_regime_sovereignty_leave_check = {
	if = {
		limit = {
			has_global_variable = un_founded
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			un_tier_at_least_strong = yes
			NOT = { has_variable = un_sov_leave_demand_cd }
			un_regime_sovereignty_demand = yes
		}
		set_variable = { name = un_sov_leave_demand_cd days = 3650 }
		trigger_event = { id = un_events.34 days = 30 }
	}
}

# ---- The ICC: indictments (§5.3) ------------------------------------------------------------

# Note a crime the International Criminal Court may indict our ruler for: a
# nuclear strike, or an exposed operation to overthrow a government. The court
# has jurisdiction over the members that ratified the Rome Statute
# (un_icc_member_modifier) from the Established tier, and over everyone at
# Supranational. It indicts at most once a decade; the indictment arrives a
# month later (un_events.33).
# Parameters: $CRIME$ = 1 nuclear use, 2 an exposed regime-change operation
# Scope: the offender
un_regime_icc_note_crime = {
	if = {
		limit = {
			has_global_variable = un_founded
			has_global_variable = un_agency_icc
			un_tier_at_least_established = yes
			exists = ruler
			NOT = { has_variable = un_icc_indicted_cd }
			OR = {
				has_variable = un_icc_member_modifier_on
				un_tier_is_supranational = yes
			}
		}
		set_variable = { name = un_icc_indicted_cd days = 3650 }
		set_variable = { name = un_icc_crime value = $CRIME$ }
		trigger_event = { id = un_events.33 days = 30 }
	}
}
