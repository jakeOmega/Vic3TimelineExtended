# ============================================================================
# SPACE RACE TEST HELPERS — console only
# ============================================================================
# Nothing in the mod calls these. They exist so the milestone panel
# (gui/journal_entry_widgets/space_race_widget.gui) can be seen in every state
# it has, in minutes rather than decades. Vic3's console has no `effect` command,
# so they are reached through a console-fired event
# (events/te_debug_space_race_events.txt):
#
#   event te_debug_space_race.1
#
# They skip costs, risk and eligibility on purpose. Never call them from
# gameplay script.
# ============================================================================

# Drive every running milestone to $FRACTION$ of its goal, so the panel's
# estimate, the native bar and (at 1) the completion path can all be seen at
# once. Scope: country.
te_debug_sr_set_progress_base = {
	if = {
		limit = { has_variable = sr_active_$MILESTONE$ }
		set_variable = { name = sr_progress_$MILESTONE$ value = sr_$MILESTONE$_goal }
		change_variable = { name = sr_progress_$MILESTONE$ multiply = $FRACTION$ }
	}
}

te_debug_sr_set_progress = {
	te_debug_sr_set_progress_base = { MILESTONE = suborbital         FRACTION = 0.9 }
	te_debug_sr_set_progress_base = { MILESTONE = orbital            FRACTION = 0.9 }
	te_debug_sr_set_progress_base = { MILESTONE = moon_landing       FRACTION = 0.9 }
	te_debug_sr_set_progress_base = { MILESTONE = probe              FRACTION = 0.9 }

# Raise the failure flag on every running milestone, and mark it as the
# milestone the setback helpers act on (they read scope:sr_setback_<milestone>,
# which the failure dispatchers save in real play). Scope: country.
te_debug_sr_flag_setback_base = {
	if = {
		limit = { has_variable = sr_active_$MILESTONE$ }
		set_variable = { name = sr_failed_$MILESTONE$ value = yes }
		sr_count_setback_base = { MILESTONE = $MILESTONE$ }
		save_scope_as = sr_setback_$MILESTONE$
	}
}

# One real setback on every running milestone, down the same path the monthly
# pulse takes: flag, progress loss, safety review on the journal entry, cooldown.
# No event fires (option j does that). Ends on sr_recalculate_cost so the panel's status flips to
# "shielded" immediately instead of next month. Scope: country.
te_debug_sr_force_setback = {
	te_debug_sr_flag_setback_base = { MILESTONE = suborbital         }
	te_debug_sr_flag_setback_base = { MILESTONE = orbital            }
	te_debug_sr_flag_setback_base = { MILESTONE = moon_landing       }
	te_debug_sr_flag_setback_base = { MILESTONE = probe              }
	te_debug_sr_flag_setback_base = { MILESTONE = moon_base          }
	te_debug_sr_flag_setback_base = { MILESTONE = mars_landing       }
	te_debug_sr_flag_setback_base = { MILESTONE = interstellar_probe }
	te_debug_sr_flag_setback_base = { MILESTONE = solar_colonization }
	sr_apply_safe_setback_progress_loss = yes
	sr_temporary_safety_review_effect = yes
	set_variable = { name = sr_failure_cooldown value = 6 }
	sr_recalculate_cost = yes
}

# One real setback EVENT for the most advanced running milestone, down the
# monthly pulse's own path: flag, setback counter, then the themed dispatcher for
# its approach (ambitious pool, otherwise the safe pool). Checks that the event's
# theme and its options follow the milestone that failed. Exactly one milestone
# (else_if): the dispatcher's marker is a saved scope, and a second dispatch in
# the same script run would hand the second event both markers — in real play
# every journal entry's pulse is its own run. Scope: country.
te_debug_sr_fire_setback_base = {
	set_variable = { name = sr_failed_$MILESTONE$ value = yes }
	sr_count_setback_base = { MILESTONE = $MILESTONE$ }
	if = {
		limit = { has_variable = sr_ambitious_$MILESTONE$ }
		sr_fire_ambitious_failure_event = { MILESTONE = $MILESTONE$ }
	}
	else = {
		sr_fire_safe_setback_event = { MILESTONE = $MILESTONE$ }
	}
}

te_debug_sr_fire_setback = {
	if = {
		limit = { has_variable = sr_active_solar_colonization }
		te_debug_sr_fire_setback_base = { MILESTONE = solar_colonization }
	}
	else_if = {
		limit = { has_variable = sr_active_interstellar_probe }
		te_debug_sr_fire_setback_base = { MILESTONE = interstellar_probe }
	}
	else_if = {
		limit = { has_variable = sr_active_mars_landing }
		te_debug_sr_fire_setback_base = { MILESTONE = mars_landing }
	}
	else_if = {
		limit = { has_variable = sr_active_moon_base }
		te_debug_sr_fire_setback_base = { MILESTONE = moon_base }
	}
	else_if = {
		limit = { has_variable = sr_active_probe }
		te_debug_sr_fire_setback_base = { MILESTONE = probe }
	}
	else_if = {
		limit = { has_variable = sr_active_moon_landing }
		te_debug_sr_fire_setback_base = { MILESTONE = moon_landing }
	}
	else_if = {
		limit = { has_variable = sr_active_orbital }
		te_debug_sr_fire_setback_base = { MILESTONE = orbital }
	}
	else_if = {
		limit = { has_variable = sr_active_suborbital }
		te_debug_sr_fire_setback_base = { MILESTONE = suborbital }
	}
}

# The two most prestigious other powers each "complete" Suborbital Flight, down
# the real notification path, inside one 14-day window. Every recognized country
# (us included) should get TWO popups, each naming its own achiever — the case a
# single shared receiver variable used to get wrong. Touches no milestone state:
# only the notification. Scope: country (the player).
te_debug_sr_mock_notices = {
	ordered_country = {
		limit = {
			NOT = { THIS = ROOT }
			is_country_type = recognized
		}
		order_by = prestige
		position = 0
		sr_notify_milestone_achievement = { MILESTONE = suborbital }
	}
	ordered_country = {
		limit = {
			NOT = { THIS = ROOT }
			is_country_type = recognized
		}
		order_by = prestige
		position = 1
		sr_notify_milestone_achievement = { MILESTONE = suborbital }
	}
}

# Drop the cooldown, so the panel goes back to showing a real setback risk.
# Scope: country.
te_debug_sr_clear_cooldown = {
	set_variable = { name = sr_failure_cooldown value = 0 }
	sr_recalculate_cost = yes
}

# Copy one of our running milestones onto another country so it shows up in
# "Who else is racing". Scope: the rival. ROOT is the player.
te_debug_sr_mirror_base = {
	if = {
		limit = {
			ROOT = { has_variable = sr_active_$MILESTONE$ }
			NOT = { has_variable = sr_active_$MILESTONE$ }
		}
		set_variable = { name = sr_active_$MILESTONE$ value = yes }
		set_variable = { name = sr_progress_$MILESTONE$ value = 0 }
		set_variable = { name = sr_funding_$MILESTONE$ value = 1 }
		set_variable = { name = sr_safe_$MILESTONE$ value = yes }
	}
}

# Put a programme in the top two other prestige powers — the rivals list only
# counts countries that would actually tick, which is the same
# sr_ai_should_participate test the monthly pulse uses.
#
# These flags are NOT backed by a journal entry, so
# sr_cleanup_inactive_space_race_milestones wipes them on the next monthly tick.
# Pause the game, run this, and open the panel. Scope: country (the player).
te_debug_sr_seed_rivals = {
	every_country = {
		limit = {
			NOT = { THIS = ROOT }
			is_country_type = recognized
			global_country_ranking <= 3
		}
		te_debug_sr_mirror_base = { MILESTONE = suborbital         }
		te_debug_sr_mirror_base = { MILESTONE = orbital            }
		te_debug_sr_mirror_base = { MILESTONE = moon_landing       }
		te_debug_sr_mirror_base = { MILESTONE = probe              }
		te_debug_sr_mirror_base = { MILESTONE = moon_base          }
		te_debug_sr_mirror_base = { MILESTONE = mars_landing       }
		te_debug_sr_mirror_base = { MILESTONE = interstellar_probe }
		te_debug_sr_mirror_base = { MILESTONE = solar_colonization }
	}
}

# Solar colonization is absent from both lists on purpose: it has no "first"
# reward and never writes a sr_global_first_ flag, so setting one would invent
# state no gameplay path produces.
# Hand "the first" for one milestone to nobody in particular: the global flag is
# what the panel and the reward split read, and setting it is exactly what
# another power finishing first would do. Only touches milestones we have not
# already completed, so our own sr_was_first_<m> is never contradicted.
# Scope: country.
te_debug_sr_claim_first_base = {
	if = {
		limit = {
			NOT = { has_variable = sr_completed_$MILESTONE$ }
			NOT = { has_global_variable = sr_global_first_$MILESTONE$ }
		}
		set_global_variable = { name = sr_global_first_$MILESTONE$ value = yes }
	}
}

# Undo the two global-state options so a test can be repeated. Scope: country.
te_debug_sr_release_first_base = {
	if = {
		limit = {
			has_global_variable = sr_global_first_$MILESTONE$
			NOT = { has_variable = sr_was_first_$MILESTONE$ }
		}
		remove_global_variable = sr_global_first_$MILESTONE$
	}
}
