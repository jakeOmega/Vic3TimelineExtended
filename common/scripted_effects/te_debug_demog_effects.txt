# Console only, apart from the census log's two hooks in te_demog_effects.txt, which
# run only while global_var:te_demog_census_log is set (te_demog_census_log_on).
# Every line goes to debug.log; the lines and what reads them:
#   TE_DEMOG_REPLAY head/before/after  option a: one step of the capital's census.
#                       demographics_harness.py replay steps the model from the
#                       before lines and compares the after lines; format_replay
#                       defines the format, and test_demographics_harness.py checks
#                       the head line below against it
#   TE_DEMOG_CENSUS:    option b's log: one line per country of a million people or
#                       more at each census (31 December), `key=value` pairs split by
#                       `;` (demographics_observer_report.py reads them)
#   TE_DEMOG_PULSE      option b's log: the player's capital at each yearly state pulse,
#                       its census year before the pulse reads it (one step per year?)
#   TE_DEMOG_BENCH      option d: the cohort step in every state of the world, three
#                       passes, timed by debug.log's one-second timestamps
#   TE_DEMOG_WARDEAD    option e: the player's dead in each of its wars, its monthly
#                       tally, and each state's dead still to be taken
#   TE_DEMOG_SKEW       options f and g: the capital's pops' workforce and dependents
#   TE_DEMOG_MODS       option l: the census's modifier types as the capital and an
#                       unincorporated state read them (live), and the cause multipliers
#                       as the last step left them (-1 before a census's first step); the
#                       practices' shares and targets (lm=, bs=, lm_target=, bs_target=; -1
#                       with no census) and the absorbed births sum (absorbed_b=)
#   TE_DEMOG_SHOCK:     option m (probe P1): every devastated or polluted state's
#                       mortality and birth reads, devastation, turmoil and pollution,
#                       each line followed by the state's scope
#   TE_DEMOG_RATES:     option n (probe P3): the capital's census terms, the terms in
#                       force, its last step's on-top scales and the reads, and the split of
#                       its window's deaths (the Causes of Death section)
#   TE_DEMOG_PROBE      option o (the late-marriage plan, Task 9): the capital's absorbed
#                       births sum before a scaled script-only probe and a day after
#   TE_DEMOG_PROBE_OWNER option q (the late-marriage plan, Stop B's I1): a state given to a
#                       neighbour, before, in on_state_owner_change and a day after: does
#                       the hook's ROOT still hold the old state's variables?
#
# A debug_log line reads only variables and script values of THIS (locals don't reach
# it, nor do saved scopes), so the numbers a line needs are copied into te_dg_dbg_*
# variables of the scope that logs it and removed after the line: a line renders when
# it runs. People counts print as millions_thousands_units groups
# (te_debug_demog_values.txt); a signed count is printed from its absolute value
# with the sign written by the line, so each such line has two variants.
# ============================================================================

# re-seed afterwards (option c). Each run also drifts the census's medicine and the practices one

# ---- b: the census log -----------------------------------------------------------
# THIS = ROOT = a country, from the end of te_demog_country_census, which has just
# stored the figures this line prints; reads its local te_dg_c_people. A country under
# a million people writes nothing. crisis is the capital's te_demog_crisis, beside the
# two bareword reads it sums as they are: devastation and turmoil, each a 0-1 share.
# A country with no capital prints -1 for all three. mig_raw (te_debug_demog_mig_raw) is
# the migration residual before each state's noise band, per 1,000 people:
# demographics_observer_report.py --closed-borders reads it. lag counts the country's
# states whose census year is not the census's (0 when every state stepped first: under
# fast mode's clock, the check that the step events ran before the census events), and
# date is the calendar date, which the report matches to saves (under the clock the
# census year is the clock's).
te_debug_demog_census_line = {
	if = {
		limit = { local_var:te_dg_c_people >= 1000000 }
		set_variable = { name = te_dg_dbg_people value = local_var:te_dg_c_people }
		set_local_variable = { name = te_dg_dbg_lag value = 0 }
		every_scope_state = {
			limit = {
				te_demog_has_census = yes
				NOT = { var:te_dg_year = te_demog_year }
			}
			change_local_variable = { name = te_dg_dbg_lag add = 1 }
		}
		set_variable = { name = te_dg_dbg_lag value = local_var:te_dg_dbg_lag }
		# phase 2 step 4: the census's terms in force and the on-top scales, weighted by people over the
		# census states, the share of people in clamped states, and war (the gate's peacetime runs)
		set_local_variable = { name = te_dg_dbg_w value = 0 }
		set_local_variable = { name = te_dg_dbg_mb value = 0 }
		set_local_variable = { name = te_dg_dbg_md value = 0 }
		set_local_variable = { name = te_dg_dbg_bz value = 0 }
		set_local_variable = { name = te_dg_dbg_dz value = 0 }
		set_local_variable = { name = te_dg_dbg_cl value = 0 }
		every_scope_state = {
			limit = { te_demog_has_census = yes }
			change_local_variable = { name = te_dg_dbg_w add = var:te_dg_people }
			change_local_variable = { name = te_dg_dbg_mb add = { value = te_demog_rate_births_applied multiply = var:te_dg_people } }
			change_local_variable = { name = te_dg_dbg_md add = { value = te_demog_rate_deaths_applied multiply = var:te_dg_people } }
			if = {
				limit = { has_variable = te_dg_bz }
				change_local_variable = { name = te_dg_dbg_bz add = { value = var:te_dg_bz multiply = var:te_dg_people } }
				change_local_variable = { name = te_dg_dbg_dz add = { value = var:te_dg_dz multiply = var:te_dg_people } }
			}
			if = {
				limit = {
					has_variable = te_dg_rate_clamped
					var:te_dg_rate_clamped > 0
				}
				change_local_variable = { name = te_dg_dbg_cl add = var:te_dg_people }
			}
		}
		set_variable = { name = te_dg_dbg_mb value = { value = local_var:te_dg_dbg_mb divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_md value = { value = local_var:te_dg_dbg_md divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_bz value = { value = local_var:te_dg_dbg_bz divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_dz value = { value = local_var:te_dg_dbg_dz divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_cl value = { value = local_var:te_dg_dbg_cl divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_war value = 0 }
		if = {
			limit = { is_at_war = yes }
			set_variable = { name = te_dg_dbg_war value = 1 }
		}
		set_variable = { name = te_dg_dbg_mig value = var:te_dg_net_migration }
		set_local_variable = { name = te_dg_dbg_crisis value = -1 }
		set_local_variable = { name = te_dg_dbg_dev value = -1 }
		set_local_variable = { name = te_dg_dbg_turmoil value = -1 }
		if = {
			limit = { exists = capital }
			capital = {
				set_local_variable = { name = te_dg_dbg_crisis value = te_demog_crisis }
				set_local_variable = { name = te_dg_dbg_dev value = devastation }
				set_local_variable = { name = te_dg_dbg_turmoil value = turmoil }
			}
		}
		set_variable = { name = te_dg_dbg_crisis value = local_var:te_dg_dbg_crisis }
		set_variable = { name = te_dg_dbg_dev value = local_var:te_dg_dbg_dev }
		set_variable = { name = te_dg_dbg_turmoil value = local_var:te_dg_dbg_turmoil }
		if = {
			limit = { var:te_dg_dbg_mig < 0 }
			debug_log = "TE_DEMOG_CENSUS: tag=[THIS.GetCountry.GetTagName]; year=[THIS.Var('te_dg_census_year').GetValue|0]; people=[THIS.ScriptValue('te_debug_demog_people_g1')|0]_[THIS.ScriptValue('te_debug_demog_people_g2')|0]_[THIS.ScriptValue('te_debug_demog_people_g3')|0]; median=[THIS.Var('te_dg_median').GetValue|2]; tfr=[THIS.Var('te_dg_tfr').GetValue|3]; e0=[THIS.Var('te_dg_e0').GetValue|2]; e65=[THIS.Var('te_dg_e65').GetValue|2]; imr=[THIS.Var('te_dg_imr').GetValue|2]; cbr=[THIS.Var('te_dg_cbr').GetValue|2]; cdr=[THIS.Var('te_dg_cdr').GetValue|2]; mig=-[THIS.ScriptValue('te_debug_demog_mig_g1')|0]_[THIS.ScriptValue('te_debug_demog_mig_g2')|0]_[THIS.ScriptValue('te_debug_demog_mig_g3')|0]; mig_raw=[THIS.ScriptValue('te_debug_demog_mig_raw')|3]; young=[THIS.Var('te_dg_young_share').GetValue|4]; working=[THIS.Var('te_dg_working_share').GetValue|4]; old=[THIS.Var('te_dg_old_share').GetValue|4]; sex_balance=[THIS.Var('te_dg_sex_balance').GetValue|2]; gini=[THIS.ScriptValue('te_debug_demog_gini')|3]; wc=[THIS.ScriptValue('te_debug_demog_wc')|2]; primacy=[THIS.Var('te_dg_primacy').GetValue|3]; crisis=[THIS.Var('te_dg_dbg_crisis').GetValue|3]; devastation=[THIS.Var('te_dg_dbg_dev').GetValue|3]; turmoil=[THIS.Var('te_dg_dbg_turmoil').GetValue|3]; lag=[THIS.Var('te_dg_dbg_lag').GetValue|0]; m_b=[THIS.Var('te_dg_dbg_mb').GetValue|3]; m_d=[THIS.Var('te_dg_dbg_md').GetValue|3]; bz=[THIS.Var('te_dg_dbg_bz').GetValue|3]; dz=[THIS.Var('te_dg_dbg_dz').GetValue|3]; clamped=[THIS.Var('te_dg_dbg_cl').GetValue|3]; war=[THIS.Var('te_dg_dbg_war').GetValue|0]; date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_DEMOG_CENSUS: tag=[THIS.GetCountry.GetTagName]; year=[THIS.Var('te_dg_census_year').GetValue|0]; people=[THIS.ScriptValue('te_debug_demog_people_g1')|0]_[THIS.ScriptValue('te_debug_demog_people_g2')|0]_[THIS.ScriptValue('te_debug_demog_people_g3')|0]; median=[THIS.Var('te_dg_median').GetValue|2]; tfr=[THIS.Var('te_dg_tfr').GetValue|3]; e0=[THIS.Var('te_dg_e0').GetValue|2]; e65=[THIS.Var('te_dg_e65').GetValue|2]; imr=[THIS.Var('te_dg_imr').GetValue|2]; cbr=[THIS.Var('te_dg_cbr').GetValue|2]; cdr=[THIS.Var('te_dg_cdr').GetValue|2]; mig=[THIS.ScriptValue('te_debug_demog_mig_g1')|0]_[THIS.ScriptValue('te_debug_demog_mig_g2')|0]_[THIS.ScriptValue('te_debug_demog_mig_g3')|0]; mig_raw=[THIS.ScriptValue('te_debug_demog_mig_raw')|3]; young=[THIS.Var('te_dg_young_share').GetValue|4]; working=[THIS.Var('te_dg_working_share').GetValue|4]; old=[THIS.Var('te_dg_old_share').GetValue|4]; sex_balance=[THIS.Var('te_dg_sex_balance').GetValue|2]; gini=[THIS.ScriptValue('te_debug_demog_gini')|3]; wc=[THIS.ScriptValue('te_debug_demog_wc')|2]; primacy=[THIS.Var('te_dg_primacy').GetValue|3]; crisis=[THIS.Var('te_dg_dbg_crisis').GetValue|3]; devastation=[THIS.Var('te_dg_dbg_dev').GetValue|3]; turmoil=[THIS.Var('te_dg_dbg_turmoil').GetValue|3]; lag=[THIS.Var('te_dg_dbg_lag').GetValue|0]; m_b=[THIS.Var('te_dg_dbg_mb').GetValue|3]; m_d=[THIS.Var('te_dg_dbg_md').GetValue|3]; bz=[THIS.Var('te_dg_dbg_bz').GetValue|3]; dz=[THIS.Var('te_dg_dbg_dz').GetValue|3]; clamped=[THIS.Var('te_dg_dbg_cl').GetValue|3]; war=[THIS.Var('te_dg_dbg_war').GetValue|0]; date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_variable = te_dg_dbg_people
		remove_variable = te_dg_dbg_mig
		remove_variable = te_dg_dbg_crisis
		remove_variable = te_dg_dbg_dev
		remove_variable = te_dg_dbg_turmoil
		remove_variable = te_dg_dbg_lag
		remove_variable = te_dg_dbg_mb
		remove_variable = te_dg_dbg_md
		remove_variable = te_dg_dbg_bz
		remove_variable = te_dg_dbg_dz
		remove_variable = te_dg_dbg_cl
		remove_variable = te_dg_dbg_war
	}

# THIS = a state at the top of its yearly pulse (te_demog_state_yearly) or of a step of
# fast mode's clock (te_demog_state_clock_step), before the seed-or-step branches read
# its census year. Only the player's capital writes: one line a year means one step a
# year (a second line in a calendar year does nothing, a skipped year re-seeds). An
# observer game has no player, so it writes none.
te_debug_demog_pulse_line = {
	if = {
		limit = {
			is_capital = yes
			owner = { is_ai = no }
		}
		debug_log = "TE_DEMOG_PULSE state=capital year=[THIS.ScriptValue('te_demog_year')|0] te_dg_year=[THIS.ScriptValue('te_debug_demog_census_year')|0] date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}

# ---- d: benchmark ----------------------------------------------------------------
# THIS = ROOT = the player. The cohort step in every state of the world with a census,
# three passes; debug.log's timestamps (whole seconds) around each pass time it (spec
# §14 Q9's late-game re-run). Each state's census year goes back a year first, so every
# pass is a second step in the same year and leaves the census wrong (this year's newborns
# folded into the over-150 pool); after the timed passes, untimed, every state with a census
# is seeded again from its population now, so the save stays playable.
te_debug_demog_benchmark = {
	set_local_variable = { name = te_dg_dbg_n value = 0 }
	every_state = {
		limit = { te_demog_has_census = yes }
		change_local_variable = { name = te_dg_dbg_n add = 1 }
	}
	set_variable = { name = te_dg_dbg_states value = local_var:te_dg_dbg_n }
	# full=1: the walk sums what the census absorbs (Full); 0 under Display only. The rule can be switched
	# mid-game, so the same save times both (owner, 2026-10-11).
	set_variable = { name = te_dg_dbg_full value = 0 }
	if = {
		limit = { te_demog_effects_run = yes }
		set_variable = { name = te_dg_dbg_full value = 1 }
	}
	debug_log = "TE_DEMOG_BENCH start states=[THIS.Var('te_dg_dbg_states').GetValue|0] full=[THIS.Var('te_dg_dbg_full').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
	remove_variable = te_dg_dbg_full
	debug_log = "TE_DEMOG_BENCH walk=1 begin"
	te_debug_demog_bench_walk = yes
	debug_log = "TE_DEMOG_BENCH walk=1 end"
	debug_log = "TE_DEMOG_BENCH walk=2 begin"
	te_debug_demog_bench_walk = yes
	debug_log = "TE_DEMOG_BENCH walk=2 end"
	debug_log = "TE_DEMOG_BENCH walk=3 begin"
	te_debug_demog_bench_walk = yes
	debug_log = "TE_DEMOG_BENCH walk=3 end"
	debug_log = "TE_DEMOG_BENCH pass=1 begin"
	te_debug_demog_bench_pass = yes
	debug_log = "TE_DEMOG_BENCH pass=1 end"
	debug_log = "TE_DEMOG_BENCH pass=2 begin"

				}
				any_neighbouring_state = {
					exists = owner
					NOT = { owner = ROOT }
				}
			}
		}
		random_scope_state = {
			limit = {
				NOT = { is_capital = yes }
				state_region = {
					NOT = {
						any_scope_state = { NOT = { owner = ROOT } }
					}
				}
				any_neighbouring_state = {
					exists = owner
					NOT = { owner = ROOT }
				}
			}
			random_neighbouring_state = {
				limit = {
					exists = owner
					NOT = { owner = ROOT }
				}
				owner = { save_scope_as = te_probe_owner_to }
			}
			set_variable = te_probe_owner_var
			state_region = { set_variable = te_probe_owner_region }
			set_global_variable = { name = te_probe_owner_to value = scope:te_probe_owner_to }
			set_global_variable = te_probe_owner_on
			debug_log = "TE_DEMOG_PROBE_OWNER before name=[THIS.GetState.GetNameNoFormatting] id=[THIS.GetState.GetID] owner=[THIS.GetState.GetOwner.GetTag] var=[THIS.ScriptValue('te_debug_demog_probe_owner_var')|0] year=[THIS.ScriptValue('te_debug_demog_probe_owner_year')|0] lm=[THIS.ScriptValue('te_debug_demog_probe_owner_lm')|3] date=[TimeKeeper.GetCurrentDate.GetString]"
			debug_log_scopes = yes
			set_state_owner = scope:te_probe_owner_to
		}
		trigger_event = { id = te_debug_demog.4 days = 1 }
	}
	else = {
		debug_log = "TE_DEMOG_PROBE_OWNER none: no state of ours is alone in its region with a foreign neighbour"
	}
}

# ---- Call 3's carrier kinds and the walk's workforce read: once-a-game self-checks (phase 2 step 1; owner, 2026-10-12)
# Instead of an in-game gate before the merge (docs/testing/demographics-playtest.md, "Self-checks"). Two of call 3's
# carrier kinds are unverified: whether the engine sums a script-only type from an amendment, and from an IG trait,
# into a state's read (docs/superpowers/specs/2026-10-12-demographics-phase2-consequences-design.md §1.7). Each check
# arms once a game, the first month a country holds such a carrier, and a week later (a day may come before the engine
# re-sums a state's modifiers) its capital logs the census-only participation input that carrier holds beside its
# moved line, against the line's size (te_debug_demog.6, .7; TE_SELF_CHECK). The moved line is on the same block, so
# one verdict covers both. Both read participation inputs, which a release comments out, so they arm wherever those
# inputs exist: every development build under any rule, Disabled included, since the facts decide what a Disabled game
# keeps (te_debug_demog_selfchecks_run; a release sets it to always = no and would otherwise log a false BROKEN). A
# third reads the walk's te_dg_wf_engine once, where the census runs: a pop's `workforce` read as a value (the plan's
# D6). A fourth, workforce_lines_net (te_debug_demog.8), checks once a game that the capital's working-adult field less
# the pair is about 0: each vanilla carrier's INJECTed cancel nets out against its own line.
# THIS = a country, on its monthly pulse (te_demog_country_monthly_on_action), under every rule.
te_debug_demog_wf_selfchecks = {
	if = {
		limit = {
			te_debug_demog_selfchecks_run = yes
			exists = capital
		}
		if = {

			set_global_variable = { name = te_demog_selfcheck_net_done value = yes }
			capital = {
				trigger_event = { id = te_debug_demog.8 days = 7 }
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_demog_selfcheck_amend_done }
				te_debug_demog_wf_amendment_held = yes
			}
			set_global_variable = { name = te_demog_selfcheck_amend_done value = yes }
			capital = {
				trigger_event = { id = te_debug_demog.6 days = 7 }
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_demog_selfcheck_ig_done }
				te_debug_demog_wf_ig_trait_plain = yes
			}
			set_global_variable = { name = te_demog_selfcheck_ig_done value = yes }
			capital = {
				trigger_event = { id = te_debug_demog.7 days = 7 }
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_demog_selfcheck_wf_read_done }
				capital = { has_variable = te_dg_wf_engine }
			}
			set_global_variable = { name = te_demog_selfcheck_wf_read_done value = yes }
			capital = {
				if = {
					limit = { var:te_dg_wf_engine > 0.01 }
					debug_log = "TE_SELF_CHECK OK pop_workforce_read wf_engine=[THIS.ScriptValue('te_debug_demog_wf_engine_read')|3] date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN pop_workforce_read wf_engine=[THIS.ScriptValue('te_debug_demog_wf_engine_read')|3] (0: a pop's workforce doesn't read as a value in the walk; find another read before step 2's stop rule, docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				debug_log_scopes = yes
			}
		}
	}
}

# ---- p: call 3's carriers (phase 2 step 1; spec §1.7, plan D12) ------------------------------------------------------
# THIS = ROOT = the player. The carrier kinds no test can read, set up in one go: Community Service (an amendment, on
# Punishment-Focused Criminal Justice, its only allowed law), Woman Soldiers at x0.5 for 60 days (a country modifier)
# and Trade Unions pushed past loyal (its Solidarity or Bourse du Travail trait; halved while the group is marginal,
# doubled while it is powerful). Logs the capital now, then a day and a month later (te_debug_demog.3). The amendment and
# the trait also arm the self-checks workforce_amendment_type and workforce_ig_trait_type (te_debug_demog_wf_selfchecks)
# under Display only or Full. A runtime activate_law can fail with no log line (scripting_best_practices.md): a
# TE_DEMOG_CALL3 line says so, and then the amendment can't attach.
te_debug_demog_call3_setup = {
	capital = { te_debug_demog_mods_line = yes }
	if = {
		limit = { NOT = { has_law = law_type:law_punishment_focused_criminal_justice } }
		activate_law = law_type:law_punishment_focused_criminal_justice
	}
	if = {
		limit = { NOT = { has_law = law_type:law_punishment_focused_criminal_justice } }
		debug_log = "TE_DEMOG_CALL3 Punishment-Focused Criminal Justice didn't take, so Community Service can't attach: enact it, then run p again date=[TimeKeeper.GetCurrentDate.GetString]"
	}
	active_law:lawgroup_criminal_justice ?= {
		add_amendment = {
			type = amendment_community_service
			sponsor = PREV.ig:ig_intelligentsia
			cooldown = 60
		}
	}
	add_modifier = { name = modifier_woman_soldiers multiplier = 0.5 days = 60 }
	ig:ig_trade_unions ?= {
		add_modifier = { name = te_demog_debug_ig_loyal days = 60 }
	}
	trigger_event = { id = te_debug_demog.3 days = 1 }
	trigger_event = { id = te_debug_demog.3 days = 31 }
}

# ---- Phase 2 step 2's workforce self-checks (docs/testing/demographics-playtest.md, "Self-checks") ------------------
# THIS = ROOT = a state, at the top of te_demog_workforce_refresh (its census step), before the refresh moves anything:
# the walk has just read the engine's workforce (te_dg_wf_engine), and the state's working-adult read is the one its pops
# held over the past year. Each check writes one TE_SELF_CHECK line a game, from a capital. They read only the engine's
# own fields and the walk, so they run whatever te_debug_demog_selfchecks_run says, wherever the census's cohorts run.
#   workforce_ratio_scale (the design §1.8's stop rule): before any W (Display only, or the capital's first step under
#     Full), is the engine's figure the set ratio read as workforce over people (WORKFORCE_SCALE's assumption; about 0.25
#     in 1836) or as workforce over dependents (about 0.20)? OK names the first, BROKEN the second.
#   workforce_follows_target (§2.1's open fact, the pull's speed): once W has run five census years in a capital and
#     moved more than the tolerance since it landed, is the engine's figure within 0.02 of the ratio the census set?
te_debug_demog_wf_term_checks = {
	if = {
		limit = {
			NOT = { has_global_variable = te_demog_selfcheck_wf_scale_done }
			te_demog_cohorts_run = yes
			is_capital = yes
			has_variable = te_dg_wf_engine
			has_variable = te_dg_wbase
			NOT = { has_variable = te_dg_mw }
			state_population > 10000
			var:te_dg_wf_engine > 0.01
		}
		set_global_variable = { name = te_demog_selfcheck_wf_scale_done value = yes }
		if = {
			limit = { te_debug_demog_wf_people_gap < te_debug_demog_wf_dependents_gap }
			debug_log = "TE_SELF_CHECK OK workforce_ratio_scale wf_engine=[THIS.ScriptValue('te_debug_demog_wf_engine_read')|3] per_people=[THIS.ScriptValue('te_debug_demog_wf_per_people')|3] per_dependents=[THIS.ScriptValue('te_debug_demog_wf_per_dependents')|3] (near per_people: the engine's ratio is workforce over people, as WORKFORCE_SCALE assumes) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN workforce_ratio_scale wf_engine=[THIS.ScriptValue('te_debug_demog_wf_engine_read')|3] per_people=[THIS.ScriptValue('te_debug_demog_wf_per_people')|3] per_dependents=[THIS.ScriptValue('te_debug_demog_wf_per_dependents')|3] (near per_dependents: the ratio is workforce over dependents; re-derive WORKFORCE_SCALE and W on that scale, the design §1.8) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
	}
	if = {
		limit = {
			NOT = { has_global_variable = te_demog_selfcheck_wft_done }
			has_variable = te_dg_wft_y0
			has_variable = te_dg_wft_w0
			has_variable = te_dg_mw
			has_variable = te_dg_wf_engine
			te_debug_demog_wft_years >= 5
			te_debug_demog_wft_moved > te_debug_demog_wft_tolerance
		}
		set_global_variable = { name = te_demog_selfcheck_wft_done value = yes }
		if = {
			limit = { te_debug_demog_wf_people_gap <= te_debug_demog_wft_tolerance }
			debug_log = "TE_SELF_CHECK OK workforce_follows_target engine=[THIS.ScriptValue('te_debug_demog_wf_engine_read')|3] set=[THIS.ScriptValue('te_debug_demog_wf_per_people')|3] census_target=[THIS.ScriptValue('te_demog_wf_engine_target')|3] w=[THIS.ScriptValue('te_debug_demog_wft_w')|3] w_target=[THIS.ScriptValue('te_debug_demog_wft_w_target')|3] since=[THIS.ScriptValue('te_debug_demog_wft_y0_read')|0] (the engine keeps within 0.02 of the ratio the census sets) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN workforce_follows_target engine=[THIS.ScriptValue('te_debug_demog_wf_engine_read')|3] set=[THIS.ScriptValue('te_debug_demog_wf_per_people')|3] census_target=[THIS.ScriptValue('te_demog_wf_engine_target')|3] w=[THIS.ScriptValue('te_debug_demog_wft_w')|3] w_target=[THIS.ScriptValue('te_debug_demog_wft_w_target')|3] since=[THIS.ScriptValue('te_debug_demog_wft_y0_read')|0] (the engine lags the ratio the census sets by more than 0.02: the pull is slower than the ramp, docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
	}
	# the landing marks go once the check has run, in any capital
	if = {
		limit = {
			has_global_variable = te_demog_selfcheck_wft_done
			has_variable = te_dg_wft_y0
		}
		remove_variable = te_dg_wft_y0
		remove_variable = te_dg_wft_w0
	}
}
