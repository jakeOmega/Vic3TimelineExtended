# Console only, apart from the census log's two hooks in te_demog_effects.txt, which
# run only while global_var:te_demog_census_log is set (te_demog_census_log_on).
# Every line goes to debug.log; the lines and what reads them:
#   TE_DEMOG_REPLAY head/before/after  option a: one step of the capital's census.
#                       demographics_harness.py replay steps the model from the
#                       before lines and compares the after lines; format_replay
#                       defines the format, and test_demographics_harness.py checks
#                       the head line below against it
#   TE_DEMOG_CENSUS:    option b's log: one line per country of a million people or
#                       more at each census (31 December), `key=value` pairs split by
#                       `;` (demographics_observer_report.py reads them)
#   TE_DEMOG_PULSE      option b's log: the player's capital at each yearly state pulse,
#                       its census year before the pulse reads it (one step per year?)
#   TE_DEMOG_BENCH      option d: the cohort step in every state of the world, three
#                       passes, timed by debug.log's one-second timestamps
#   TE_DEMOG_WARDEAD    option e: the player's dead in each of its wars, its monthly
#                       tally, and each state's dead still to be taken
#   TE_DEMOG_SKEW       options f and g: the capital's pops' workforce and dependents
#   TE_DEMOG_MODS       option l: the census's modifier types as the capital and an
#                       unincorporated state read them (live), and the cause multipliers
#                       as the last step left them (-1 before a census's first step); the
#                       practices' shares and targets (lm=, bs=, lm_target=, bs_target=; -1
#                       with no census) and the absorbed births sum (absorbed_b=); phase 3: the owner's Family
#                       Limitation line in the means (fl_line=) and the Cultural Hegemony factor the step took
#                       (hegemon=)
#   TE_DEMOG_SHOCK:     option m (probe P1): every devastated or polluted state's
#                       mortality and birth reads, devastation, turmoil and pollution,
#                       each line followed by the state's scope
#   TE_DEMOG_RATES:     option n (probe P3): the capital's census terms, the terms in
#                       force, its last step's on-top scales and the reads, and the split of
#                       its window's deaths (the Causes of Death section)
#   TE_DEMOG_PROBE      option o (the late-marriage plan, Task 9): the capital's absorbed
#                       births sum before a scaled script-only probe and a day after
#   TE_DEMOG_PROBE_OWNER option q (the late-marriage plan, Stop B's I1): a state given to a
#                       neighbour, before, in on_state_owner_change and a day after: does
#                       the hook's ROOT still hold the old state's variables?
#
# A debug_log line reads only variables and script values of THIS (locals don't reach
# it, nor do saved scopes), so the numbers a line needs are copied into te_dg_dbg_*
# variables of the scope that logs it and removed after the line: a line renders when
# it runs. People counts print as millions_thousands_units groups
# (te_debug_demog_values.txt); a signed count is printed from its absolute value
# with the sign written by the line, so each such line has two variants.
# ============================================================================

# ---- b: the census log -----------------------------------------------------------
# THIS = ROOT = a country, from the end of te_demog_country_census, which has just
# stored the figures this line prints; reads its local te_dg_c_people. A country under
# a million people writes nothing. crisis is the capital's te_demog_crisis, beside the
# two bareword reads it sums as they are: devastation and turmoil, each a 0-1 share.
# A country with no capital prints -1 for all three. mig_raw (te_debug_demog_mig_raw) is
# the migration residual before each state's noise band, per 1,000 people:
# demographics_observer_report.py --closed-borders reads it. lag counts the country's
# states whose census year is not the census's (0 when every state stepped first: under
# fast mode's clock, the check that the step events ran before the census events), and
# date is the calendar date, which the report matches to saves (under the clock the
# census year is the clock's).
te_debug_demog_census_line = {
	if = {
		limit = { local_var:te_dg_c_people >= 1000000 }
		set_variable = { name = te_dg_dbg_people value = local_var:te_dg_c_people }
		set_local_variable = { name = te_dg_dbg_lag value = 0 }
		every_scope_state = {
			limit = {
				te_demog_has_census = yes
				NOT = { var:te_dg_year = te_demog_year }
			}
			change_local_variable = { name = te_dg_dbg_lag add = 1 }
		}
		set_variable = { name = te_dg_dbg_lag value = local_var:te_dg_dbg_lag }
		# phase 2 step 4: the census's terms in force and the on-top scales, weighted by people over the
		# census states, the share of people in clamped states, and war (the gate's peacetime runs)
		set_local_variable = { name = te_dg_dbg_w value = 0 }
		set_local_variable = { name = te_dg_dbg_mb value = 0 }
		set_local_variable = { name = te_dg_dbg_md value = 0 }
		set_local_variable = { name = te_dg_dbg_bz value = 0 }
		set_local_variable = { name = te_dg_dbg_dz value = 0 }
		set_local_variable = { name = te_dg_dbg_cl value = 0 }
		every_scope_state = {
			limit = { te_demog_has_census = yes }
			change_local_variable = { name = te_dg_dbg_w add = var:te_dg_people }
			change_local_variable = { name = te_dg_dbg_mb add = { value = te_demog_rate_births_applied multiply = var:te_dg_people } }
			change_local_variable = { name = te_dg_dbg_md add = { value = te_demog_rate_deaths_applied multiply = var:te_dg_people } }
			if = {
				limit = { has_variable = te_dg_bz }
				change_local_variable = { name = te_dg_dbg_bz add = { value = var:te_dg_bz multiply = var:te_dg_people } }
				change_local_variable = { name = te_dg_dbg_dz add = { value = var:te_dg_dz multiply = var:te_dg_people } }
			}
			if = {
				limit = {
					has_variable = te_dg_rate_clamped
					var:te_dg_rate_clamped > 0
				}
				change_local_variable = { name = te_dg_dbg_cl add = var:te_dg_people }
			}
		}
		set_variable = { name = te_dg_dbg_mb value = { value = local_var:te_dg_dbg_mb divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_md value = { value = local_var:te_dg_dbg_md divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_bz value = { value = local_var:te_dg_dbg_bz divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_dz value = { value = local_var:te_dg_dbg_dz divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_cl value = { value = local_var:te_dg_dbg_cl divide = { value = local_var:te_dg_dbg_w min = 1 } } }
		set_variable = { name = te_dg_dbg_war value = 0 }
		if = {
			limit = { is_at_war = yes }
			set_variable = { name = te_dg_dbg_war value = 1 }
		}
		set_variable = { name = te_dg_dbg_mig value = var:te_dg_net_migration }
		set_local_variable = { name = te_dg_dbg_crisis value = -1 }
		set_local_variable = { name = te_dg_dbg_dev value = -1 }
		set_local_variable = { name = te_dg_dbg_turmoil value = -1 }
		if = {
			limit = { exists = capital }
			capital = {
				set_local_variable = { name = te_dg_dbg_crisis value = te_demog_crisis }
				set_local_variable = { name = te_dg_dbg_dev value = devastation }
				set_local_variable = { name = te_dg_dbg_turmoil value = turmoil }
			}
		}
		set_variable = { name = te_dg_dbg_crisis value = local_var:te_dg_dbg_crisis }
		set_variable = { name = te_dg_dbg_dev value = local_var:te_dg_dbg_dev }
		set_variable = { name = te_dg_dbg_turmoil value = local_var:te_dg_dbg_turmoil }
		if = {
			limit = { var:te_dg_dbg_mig < 0 }
			debug_log = "TE_DEMOG_CENSUS: tag=[THIS.GetCountry.GetTagName]; year=[THIS.Var('te_dg_census_year').GetValue|0]; people=[THIS.ScriptValue('te_debug_demog_people_g1')|0]_[THIS.ScriptValue('te_debug_demog_people_g2')|0]_[THIS.ScriptValue('te_debug_demog_people_g3')|0]; median=[THIS.Var('te_dg_median').GetValue|2]; tfr=[THIS.Var('te_dg_tfr').GetValue|3]; e0=[THIS.Var('te_dg_e0').GetValue|2]; e65=[THIS.Var('te_dg_e65').GetValue|2]; imr=[THIS.Var('te_dg_imr').GetValue|2]; cbr=[THIS.Var('te_dg_cbr').GetValue|2]; cdr=[THIS.Var('te_dg_cdr').GetValue|2]; mig=-[THIS.ScriptValue('te_debug_demog_mig_g1')|0]_[THIS.ScriptValue('te_debug_demog_mig_g2')|0]_[THIS.ScriptValue('te_debug_demog_mig_g3')|0]; mig_raw=[THIS.ScriptValue('te_debug_demog_mig_raw')|3]; young=[THIS.Var('te_dg_young_share').GetValue|4]; working=[THIS.Var('te_dg_working_share').GetValue|4]; old=[THIS.Var('te_dg_old_share').GetValue|4]; sex_balance=[THIS.Var('te_dg_sex_balance').GetValue|2]; gini=[THIS.ScriptValue('te_debug_demog_gini')|3]; wc=[THIS.ScriptValue('te_debug_demog_wc')|2]; primacy=[THIS.Var('te_dg_primacy').GetValue|3]; crisis=[THIS.Var('te_dg_dbg_crisis').GetValue|3]; devastation=[THIS.Var('te_dg_dbg_dev').GetValue|3]; turmoil=[THIS.Var('te_dg_dbg_turmoil').GetValue|3]; lag=[THIS.Var('te_dg_dbg_lag').GetValue|0]; m_b=[THIS.Var('te_dg_dbg_mb').GetValue|3]; m_d=[THIS.Var('te_dg_dbg_md').GetValue|3]; bz=[THIS.Var('te_dg_dbg_bz').GetValue|3]; dz=[THIS.Var('te_dg_dbg_dz').GetValue|3]; clamped=[THIS.Var('te_dg_dbg_cl').GetValue|3]; war=[THIS.Var('te_dg_dbg_war').GetValue|0]; date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_DEMOG_CENSUS: tag=[THIS.GetCountry.GetTagName]; year=[THIS.Var('te_dg_census_year').GetValue|0]; people=[THIS.ScriptValue('te_debug_demog_people_g1')|0]_[THIS.ScriptValue('te_debug_demog_people_g2')|0]_[THIS.ScriptValue('te_debug_demog_people_g3')|0]; median=[THIS.Var('te_dg_median').GetValue|2]; tfr=[THIS.Var('te_dg_tfr').GetValue|3]; e0=[THIS.Var('te_dg_e0').GetValue|2]; e65=[THIS.Var('te_dg_e65').GetValue|2]; imr=[THIS.Var('te_dg_imr').GetValue|2]; cbr=[THIS.Var('te_dg_cbr').GetValue|2]; cdr=[THIS.Var('te_dg_cdr').GetValue|2]; mig=[THIS.ScriptValue('te_debug_demog_mig_g1')|0]_[THIS.ScriptValue('te_debug_demog_mig_g2')|0]_[THIS.ScriptValue('te_debug_demog_mig_g3')|0]; mig_raw=[THIS.ScriptValue('te_debug_demog_mig_raw')|3]; young=[THIS.Var('te_dg_young_share').GetValue|4]; working=[THIS.Var('te_dg_working_share').GetValue|4]; old=[THIS.Var('te_dg_old_share').GetValue|4]; sex_balance=[THIS.Var('te_dg_sex_balance').GetValue|2]; gini=[THIS.ScriptValue('te_debug_demog_gini')|3]; wc=[THIS.ScriptValue('te_debug_demog_wc')|2]; primacy=[THIS.Var('te_dg_primacy').GetValue|3]; crisis=[THIS.Var('te_dg_dbg_crisis').GetValue|3]; devastation=[THIS.Var('te_dg_dbg_dev').GetValue|3]; turmoil=[THIS.Var('te_dg_dbg_turmoil').GetValue|3]; lag=[THIS.Var('te_dg_dbg_lag').GetValue|0]; m_b=[THIS.Var('te_dg_dbg_mb').GetValue|3]; m_d=[THIS.Var('te_dg_dbg_md').GetValue|3]; bz=[THIS.Var('te_dg_dbg_bz').GetValue|3]; dz=[THIS.Var('te_dg_dbg_dz').GetValue|3]; clamped=[THIS.Var('te_dg_dbg_cl').GetValue|3]; war=[THIS.Var('te_dg_dbg_war').GetValue|0]; date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_variable = te_dg_dbg_people
		remove_variable = te_dg_dbg_mig
		remove_variable = te_dg_dbg_crisis
		remove_variable = te_dg_dbg_dev

		remove_variable = te_dg_dbg_lag
		remove_variable = te_dg_dbg_mb
		remove_variable = te_dg_dbg_md

# THIS = a state at the top of its yearly pulse (te_demog_state_yearly) or of a step of
# fast mode's clock (te_demog_state_clock_step), before the seed-or-step branches read
# its census year. Only the player's capital writes: one line a year means one step a
# year (a second line in a calendar year does nothing, a skipped year re-seeds). An
# observer game has no player, so it writes none.
te_debug_demog_pulse_line = {
	if = {
		limit = {
			is_capital = yes
			owner = { is_ai = no }
		}
		debug_log = "TE_DEMOG_PULSE state=capital year=[THIS.ScriptValue('te_demog_year')|0] te_dg_year=[THIS.ScriptValue('te_debug_demog_census_year')|0] date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}

# ---- c_world: re-seed every late-game census -------------------------------------
# THIS = ROOT = the player. Every state of the world whose late game has started (te_dg_elig) starts its census
# again at its own equilibrium, as option c does for ours: the walks first, then a seed off the state's pulse (so
# it seeds again at its next one and steps a year after that), then our census at once. Other countries take their
# figures at their own census on 31 December: te_demog_country_census writes its bands and lists to ROOT
# (te_demog_add_bands_to_root, root = { }), so it runs for ROOT alone. The repair for the late sweep's lost births
# (fixed 2026-10-16): until then a late state under a TFR of 2 recorded none, so its ring holds too few young
# people, and once the census steers births again it would carry that gap forward for decades. A seed leaves the
# births and deaths terms as they were, so a frozen term stays until the state's first step.
te_debug_demog_reseed_late = {
	set_local_variable = { name = te_dg_dbg_n value = 0 }
	every_state = {
		limit = {
			te_demog_has_census = yes
			has_variable = te_dg_elig
		}
		te_demog_walks = yes
		te_demog_seed_off_pulse = yes
		change_local_variable = { name = te_dg_dbg_n add = 1 }
	}
	te_demog_country_census = yes
	set_variable = { name = te_dg_dbg_states value = local_var:te_dg_dbg_n }
	debug_log = "TE_DEMOG_RESEED_LATE states=[THIS.Var('te_dg_dbg_states').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
	remove_variable = te_dg_dbg_states
}

# ---- d: benchmark ----------------------------------------------------------------

# three passes; debug.log's timestamps (whole seconds) around each pass time it (spec
# §14 Q9's late-game re-run). Each state's census year goes back a year first, so every
# pass is a second step in the same year and leaves the census wrong (this year's newborns
# folded into the over-150 pool); after the timed passes, untimed, every state with a census

# ---- Phase 2 step 4: conscription, the youth bulge and sex balance (the consequences design §4) --------------------
# Two once-a-game self-checks (owner, 2026-10-12; docs/testing/demographics-playtest.md, "Self-checks"), each called
# from te_demog_consequences_refresh (te_demog_consequence_effects.txt), ROOT the state:
#   conscription_term_read  the first state to take a fresh conscription pair (its term at least te_demog_k_conscription_cut
#                           either way, and only while the effect is on, as it is since the migrant booking; held back, it
#                           arms never): its
#                           state_conscription_rate_mult before the add, then a week later (te_debug_demog.30; a modifier
#                           sum can lag a day) whether it moved by the term. It tests the term's path into the state's
#                           read; whether the read moves the state's potential conscripts is the play-test's item (no
#                           script reads a state's potential conscripts: army_size_including_conscripts counts raised
#                           ones). Another change to the field inside the week (a military technology, a law) reads as
#                           BROKEN with its own size: the line prints both.
#   unemployment_rate_read  the first census state whose walk counted unemployment above 0.02 (te_dg_unemp_walk: the
#                           workforce of pops with no workplace over all the workforce): the engine's
#                           state_unemployment_rate against that count x the walk's workforce share (te_dg_wf_engine),
#                           the idle workforce over all the state's people. The 2026-10-17 observer run read 0.048
#                           against a count of 0.195 in January 1836, when the workforce is about a quarter of the
#                           people. OK within a quarter below or a third above: the engine's figure is per person, so
#                           the conversion the census stores (te_dg_unemp, te_state_unemployment_share; the youth
#                           bulge, the wave, the pension AI and the panel read it since 2026-10-17) is the share of the
#                           workforce. BROKEN: it is another figure again, and the conversion is wrong for every reader;
#                           take unemployment_share_read's fallback (docs/testing/demographics-playtest.md). The same
#                           comparison as unemployment_share_read (te_unemployment_effects.txt), made from the census's
#                           walk: one engine fact, read twice.
te_debug_demog_conscription_selfcheck_start = {
	if = {
		limit = {
			NOT = { has_global_variable = te_demog_selfcheck_consc_done }
			NOT = { has_modifier = te_demog_census_conscription_up }
			NOT = { has_modifier = te_demog_census_conscription_down }
			# a pair the refresh will add: the effect on (held back, it adds none, and the check would read a false
			# BROKEN) and a term past the cut-off

			te_demog_conscription_size >= te_demog_k_conscription_cut
		}

		set_variable = { name = te_dg_chk_consc_before value = modifier:state_conscription_rate_mult }

	}

# ---- The Causes of Death estimate (issue #1029): a once-a-game self-check -----------------------------------------------
# The section's slices rest on an estimate (te_demog_cause_deaths_estimate: last count's bands at the step's rates) scaled
# to the sweep's own deaths, so they always add up; what the estimate claims is that the bands at the step's start are the
# people the sweep takes its dead from. Once a game, the first state to step with a thousand deaths or more from the four
# natural causes and no one in longevity care (whose treated die at their biological ages, where the estimate is only
# approximate) logs the estimate over those deaths (fit, 1 when it matches; 0.97 for a poor farm state, 1.00 for a rich
# one in demographics_model) and the slices' sum against the census's deaths with the war dead and the killed. BROKEN:
# the estimate is off by more than a tenth, or the slices don't add up; count the deaths by group in the sweep instead
# (the issue's method b), docs/testing/demographics-playtest.md. THIS = a state, in te_demog_cause_deaths_store, with
# its locals.
te_debug_demog_cause_deaths_selfcheck = {
	if = {
		limit = {
			te_debug_demog_selfchecks_run = yes

		}
		set_global_variable = { name = te_demog_selfcheck_cause_done value = yes }
		set_local_variable = {
			name = te_dg_dc_chk_gap
			value = {
				value = var:te_dg_dc_total
				subtract = local_var:te_dg_deaths
				subtract = local_var:te_dg_war
				subtract = local_var:te_dg_kills
			}
		}
		# the tolerance either side as locals: a comparison with an inline value block held for every state (the note in
		# te_demog_state_census)
		set_local_variable = { name = te_dg_dc_chk_tol value = { value = var:te_dg_dc_total multiply = 0.001 min = 1 } }
		set_local_variable = { name = te_dg_dc_chk_neg value = { value = local_var:te_dg_dc_chk_tol multiply = -1 } }
		if = {
			limit = {
				var:te_dg_dc_fit >= 0.9
				var:te_dg_dc_fit <= 1.1

		}

	}

# ---- Phase 3's self-checks (te_demog_fl_effects.txt; owner, 2026-10-12: a once-a-game line in place of an in-game gate)
# docs/testing/demographics-playtest.md, "Self-checks". Three engine facts the spreading machinery rests on:
# - family_limitation_scaled: a script-only state type (state_fertility_means_add) on a country static modifier, scaled
#   by its multiplier, reaches the state's read. The type-on-a-country half is already seen in saves (France's states
#   read history's 0.6 in te_dg_means, 2026-10-12), the multiplier half only on a state modifier (primogeniture_scaled).
#   Once a game under Full, the first country whose line goes on fresh (it carried none, so the read before holds none)
#   with a line from te_demog_k_fl_line_min to 0.3 (multiplier 0.5 or less: an unscaled 0.6 reads at least 0.3 apart);
#   the verdict a week later from the capital (te_debug_demog.90).
# - fl_neighbours_read: every_neighbouring_state from country scope returns the states across the borders. Once a game
#   under Full, at the first Family Limitation step of a country with a foreign neighbour by the state-scope test.
# - neighbouring_state_reads: every_neighbouring_state from state scope returns a land neighbour (the practices' spread
#   rests on it), and whether it crosses a strait (Ulster to the Scottish Lowlands: the spec's open question). Once a game
#   wherever the census runs, at the first country census on 31 December.

# THIS = ROOT = a country under Full, in te_demog_fl_apply before the line is swapped (te_dg_fl_line is the new line).
te_debug_demog_fl_selfcheck_start = {
	if = {
		limit = {
			NOT = { has_global_variable = te_demog_fl_selfcheck_done }
			NOT = { has_modifier = te_demog_family_limitation }
			var:te_dg_fl_line >= te_demog_k_fl_line_min
			var:te_dg_fl_line <= 0.3
			exists = capital
		}
		set_global_variable = { name = te_demog_fl_selfcheck_done value = yes }
		set_local_variable = { name = te_dg_fl_chk_line value = var:te_dg_fl_line }
		capital = {
			set_variable = { name = te_dg_fl_chk_before value = modifier:state_fertility_means_add }
			set_variable = { name = te_dg_fl_chk_line value = local_var:te_dg_fl_chk_line }
			set_variable = { name = te_dg_fl_chk_owner value = ROOT }
			debug_log = "TE_SELF_CHECK START family_limitation_scaled before=[THIS.Var('te_dg_fl_chk_before').GetValue|4] line=[THIS.Var('te_dg_fl_chk_line').GetValue|4] (the verdict follows in a week; none means the capital changed hands first) date=[TimeKeeper.GetCurrentDate.GetString]"
			debug_log_scopes = yes
			trigger_event = { id = te_debug_demog.90 days = 7 }
		}
	}
}

# THIS = ROOT = a country under Full, in te_demog_fl_step after its walk over its neighbouring states (locals
# te_dg_fl_nf: the states other countries own, te_dg_fl_own: its own the iterator returned too).
te_debug_demog_fl_neighbours_check = {
	if = {
		limit = {
			NOT = { has_global_variable = te_demog_fl_nb_selfcheck_done }
			any_scope_state = {
				any_neighbouring_state = {
					exists = owner
					NOT = { owner = ROOT }
				}
			}
		}
		set_global_variable = { name = te_demog_fl_nb_selfcheck_done value = yes }
		set_variable = { name = te_dg_fl_chk_nf value = local_var:te_dg_fl_nf }
		set_variable = { name = te_dg_fl_chk_own value = local_var:te_dg_fl_own }
		if = {
			limit = { local_var:te_dg_fl_nf > 0 }
			debug_log = "TE_SELF_CHECK OK fl_neighbours_read foreign=[THIS.Var('te_dg_fl_chk_nf').GetValue|0] own=[THIS.Var('te_dg_fl_chk_own').GetValue|0] (own above 0: the iterator returns the country's own states too, which the step leaves out) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN fl_neighbours_read foreign=0 own=[THIS.Var('te_dg_fl_chk_own').GetValue|0] (every_neighbouring_state from country scope returns no foreign state, so Family Limitation never spreads: walk the country's own states' every_neighbouring_state instead, docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
		remove_variable = te_dg_fl_chk_nf
		remove_variable = te_dg_fl_chk_own
	}
}

# THIS = a country at its 31 December census, wherever the census runs (te_demog_country_yearly).
te_debug_demog_neighbours_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_demog_nb_selfcheck_done }
			s:STATE_ULSTER = { any_scope_state = { exists = owner } }
		}
		set_global_variable = { name = te_demog_nb_selfcheck_done value = yes }
		set_local_variable = { name = te_dg_nb_chk_land value = 0 }
		set_local_variable = { name = te_dg_nb_chk_strait value = 0 }
		s:STATE_ULSTER = {
			if = {
				limit = { any_scope_state = { any_neighbouring_state = { state_region = s:STATE_LEINSTER } } }
				set_local_variable = { name = te_dg_nb_chk_land value = 1 }
			}
			if = {
				limit = { any_scope_state = { any_neighbouring_state = { state_region = s:STATE_LOWLANDS } } }
				set_local_variable = { name = te_dg_nb_chk_strait value = 1 }
			}
		}
		# kept on the country for the line (a debug_log line reads THIS.Var, not a local)
		set_variable = { name = te_dg_nb_chk_strait value = local_var:te_dg_nb_chk_strait }
		if = {
			limit = { local_var:te_dg_nb_chk_land = 1 }
			debug_log = "TE_SELF_CHECK OK neighbouring_state_reads land=1 strait=[THIS.Var('te_dg_nb_chk_strait').GetValue|0] (strait=1: the practices spread across straits too, so the harness's replay takes demographics_adjacency's straits; strait=0: islands spread only through their own conditions) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN neighbouring_state_reads land=0 strait=[THIS.Var('te_dg_nb_chk_strait').GetValue|0] (every_neighbouring_state from a state doesn't return Ulster's land neighbour Leinster, so Late Marriage and Birth Spacing never spread: set PRACTICE_SPREAD to 0 and regenerate, docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
		remove_variable = te_dg_nb_chk_strait
	}
}
