# One national policy, a country flag (a civil war's winner keeps it). Each is
# the best at one thing, and its cost is the others given up (owner):
#   Open to the Public   monument tourism x1.5
#   State Ceremonial     legitimacy and the Leader's authority count a quarter
#                        of every monument's grandeur, before the curve
#   Cultural Promotion   monument cultural pull x1.5 (prestige x1.5 with the
#                        Cultural Hegemony rule off); the default
#   Public Education     education access in every state, from every monument
# The factors are script values (gm_values.txt § "The monument policy") read
# by gm_compute_totals and gm_state_monthly. No policy changes upkeep (there is
# none) or anniversaries.

# Country scope. Multiply step count gm_s_$S$ by the policy factor $FACTOR$ (a
# script value): the per-step values move by the factor, as §5 asks.
gm_policy_scale_steps = {
	set_variable = {
		name = gm_s_$S$
		value = {
			value = var:gm_s_$S$
			multiply = $FACTOR$
		}
	}
}

# Country scope. Set policy $P$. Every change after the first starts a
# 60-month wait; the first choice is free, so a first pick can be corrected.
gm_policy_set = {
	set_variable = { name = gm_policy value = flag:$P$ }
	if = {
		limit = { has_variable = gm_policy_chosen }
		set_variable = { name = gm_policy_cooldown value = yes months = 60 }
		set_variable = { name = gm_policy_changed_month value = te_history_month_index }
	}
	else = {
		set_variable = gm_policy_chosen
	}
}

# Country scope, from gm_country_refresh before anything reads the policy: an
# older save's Mothballed or Standard, both gone, reads as the default (Cultural
# Promotion). The wait and the first-choice mark are kept, so a wait running
# keeps running and no new one starts (spec § 6).
gm_policy_migrate = {
	if = {
		limit = {
			has_variable = gm_policy
			OR = {
				var:gm_policy = flag:mothballed
				var:gm_policy = flag:standard
			}
		}
		remove_variable = gm_policy
	}
}

# Country scope, AI only, each January while it owns a monument and may
# change: Mothballed in hard times or in debt; State Ceremonial with legitimacy
# under 40 and regime grandeur to multiply; Open to the Public when a monument
# stands beside a Tourism Industry; else Standard. Changes only when the pick
# differs, so a settled AI never starts a wait.
gm_policy_ai_choose = {
	if = {
		limit = {
			is_ai = yes
			month = 0
			gm_policy_can_change = yes
			any_scope_state = { has_building = building_grand_monument }
		}
		set_variable = { name = gm_policy_pick value = flag:culture }
		if = {
			limit = {
				OR = {
					gm_country_hard_times = yes
					scaled_debt > 0
				}
			}
			set_variable = { name = gm_policy_pick value = flag:education }
		}
		else_if = {
			limit = {
				government_legitimacy < 40
				has_variable = gm_g_regime
				var:gm_g_regime > 0
			}
			set_variable = { name = gm_policy_pick value = flag:ceremonial }
		}
		else_if = {
			limit = {
				any_scope_state = {
					has_building = building_grand_monument
					has_building = building_tourism_industry
				}
			}
			set_variable = { name = gm_policy_pick value = flag:open }
		}
		gm_policy_ai_apply = { P = open STANDARD = no }
		gm_policy_ai_apply = { P = ceremonial STANDARD = no }
		gm_policy_ai_apply = { P = culture STANDARD = yes }
		gm_policy_ai_apply = { P = education STANDARD = no }
		remove_variable = gm_policy_pick
	}
}

gm_policy_ai_apply = {
	if = {
		limit = {
			has_variable = gm_policy_pick
			var:gm_policy_pick = flag:$P$
			NOT = { gm_policy_is = { P = $P$ STANDARD = $STANDARD$ } }
		}
		gm_policy_set = { P = $P$ }
	}
}
