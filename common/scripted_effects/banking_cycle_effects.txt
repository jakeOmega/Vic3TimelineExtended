# Banking cycle random event selection
# Called from je_banking_cycle monthly pulse.
# Handles cooldown countdown and weighted random event selection.

banking_cycle_random_event_effect = {
	# Cooldown countdown
	if = {
		limit = { has_variable = banking_event_cooldown }
		change_variable = { name = banking_event_cooldown add = -1 }
		if = {
			limit = { var:banking_event_cooldown <= 0 }
			remove_variable = banking_event_cooldown
		}
	}
	if = {
		limit = {
			NOT = { has_variable = banking_event_cooldown }
			var:crash <= 0  # don't fire events in the same month as a crash
		}
		random_list = {
			260 = { } # no event this month

			# ---- EARLY ERA (no extra tech gate beyond stock_exchange) ----

			# Event 1: Speculative Mania (expansion+, bubble building)
			0 = {
				modifier = {
					if = { limit = { var:bubble_pressure >= 15 banking_cycle_is_expansion = yes } add = 5 }

					if = { limit = { var:bubble_pressure >= 15 banking_cycle_is_frenzy = yes } add = 5 }
					if = { limit = { banking_cycle_is_frenzy = yes } add = 3 }
				}
				if = {
					limit = { has_law = law_type:law_command_economy }
					trigger_event = { id = banking_cycle_events.101 }
				}
				else_if = {
					limit = { has_law = law_type:law_cooperative_ownership }
					trigger_event = { id = banking_cycle_events.151 }
				}
				else = {
					trigger_event = { id = banking_cycle_events.1 }
				}
				set_variable = { name = banking_event_cooldown value = 18 }
			}
			# Event 2: Foreign Capital Inflow (stable+, not at war)
			0 = {
				trigger = { banking_is_market_economy = yes }
				modifier = {
					if = {
						limit = {
							is_at_war = no
							OR = {
								banking_cycle_is_stable = yes
								banking_cycle_is_expansion = yes
								banking_cycle_is_boom = yes
							}
						}
						add = 5
					}
				}
				trigger_event = { id = banking_cycle_events.2 }
				set_variable = { name = banking_event_cooldown value = 18 }
			}
			# Event 3: Credit Crunch Fears (stagnation/downturn)
			0 = {
				trigger = { banking_is_market_economy = yes }
				modifier = {
					if = { limit = { banking_cycle_is_stagnation = yes } add = 6 }
					if = { limit = { banking_cycle_is_downturn = yes } add = 9 }
				}
				trigger_event = { id = banking_cycle_events.3 }
				set_variable = { name = banking_event_cooldown value = 18 }
			}
			# Event 4: Country Bank Run (RARE, HIGH IMPACT - downturn/panic)
			# The bubble bonus sits inside the recession: "depositors queuing in
			# the streets" cannot fire while the banner reads Boom or Frenzy.
			0 = {
				modifier = {
					if = { limit = { banking_cycle_is_downturn = yes } add = 1 }
					if = { limit = { banking_cycle_is_panic = yes } add = 2 }
					if = {
						limit = {
							banking_cycle_is_recession = yes
							var:bubble_pressure >= 50
						}
						add = 1
					}
				}
				if = {
					limit = { has_law = law_type:law_command_economy }
					trigger_event = { id = banking_cycle_events.104 }
				}

					limit = { has_law = law_type:law_cooperative_ownership }
					trigger_event = { id = banking_cycle_events.154 }
				}
				else = {
					trigger_event = { id = banking_cycle_events.4 }
				}
				set_variable = { name = banking_event_cooldown value = 18 }
			}

# Composite helper for the conforming softening options of
# minor_events_timelineextended.6 (event 6). Posts the origin tier first to set
# baseline cycle/momentum/bubble (must run before per-option additions because
# origin uses set_variable, then per-option change_variable adds on top), then
# applies the per-option softening tooltip, GDP-scaled treasury cost (6 months,
# decaying), year-long intervention modifier, and upper-strata radicals.
#
# Country scope. Trigger and ai_chance stay in each option block — they vary
# unrelatedly per option (law/tech gating + AI weighting heuristics).
#
# 10 of 12 options of event 6 use this helper. Two outliers stay inline:
#   - Option A (.6.a "weather the storm"): no softening, no intervention modifier,
#     different radicals shape (medium upper + small middle).
#   - Option .6.capital_controls: no momentum softening (only finance_cycle_value
#     changes), so a $MOMENTUM_ADD$=0 call would emit an ugly "+0" tooltip line.
#
# Parameters:
#   $TT_KEY$       — custom_tooltip text loc key (CRASH_SOFTEN_<X>_DESC)
#   $CYCLE_ADD$    — finance_cycle_value softening add (3..12)
#   $MOMENTUM_ADD$ — finance_cycle_momentum softening add (1..3)
#   $COST_SCALE$   — banking_event_expense_<small|medium|large|huge>
#   $INTERVENTION$ — banking_crash_intervention_<X> modifier name
#   $RADICALS$     — radicals size (very_small|small|medium); resolves to
#                    <RADICALS>_radicals script value at the call site.
apply_banking_crash_softening_option_effect = {
	apply_banking_crash_origin_effects = yes
	custom_tooltip = {
		text = $TT_KEY$
		change_variable = { name = finance_cycle_value    add = $CYCLE_ADD$    }
		change_variable = { name = finance_cycle_momentum add = $MOMENTUM_ADD$ }
	}
	add_modifier = {
		name = banking_crash_response_cost
		multiplier = $COST_SCALE$
		days = 180
		is_decaying = yes
	}
	add_modifier = {
		name = $INTERVENTION$
		days = 365
	}
	add_radicals = {
		value = $RADICALS$_radicals
		strata = upper
	}
}

# Auto-disable crash interventions when total intervention points have gone negative
# (e.g. because the player stacked emergency response modifiers from a crash event on top
# of ongoing JE intervention buttons, or because their financial-regulation law was downgraded).
#
# Removes ONE crash-tagged modifier per monthly pulse, walking a fixed priority cascade:
# tech-family interventions first (these are layered on top of the per-law base, so removing
# them preserves the player's "core" emergency response), then per-law base options from
# cheapest IP cost to most expensive (so the strongest emergency tool is shed last).
#
# Only one removal per pulse: the engine doesn't refresh modifier:X reads inside the same
# effect block (CLAUDE.md "modifier-after-effect read ordering"), so a tight loop would
# fire on stale data. Spacing removals by month also gives the player visible feedback.
banking_crash_check_overdrawn_interventions = {
	if = {
		limit = { modifier:country_banking_intervention_max_add < 0 }
		# --- Family interventions (shed first) ---
		if = {
			limit = { has_modifier = banking_crash_intervention_capital_controls }
			remove_modifier = banking_crash_intervention_capital_controls
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_swap_lines }
			remove_modifier = banking_crash_intervention_swap_lines
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_discount_window }
			remove_modifier = banking_crash_intervention_discount_window
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_keynesian_stimulus }
			remove_modifier = banking_crash_intervention_keynesian_stimulus
		}
		# --- Per-law base options (one of these may be active; cheapest IP cost first) ---
		else_if = {
			limit = { has_modifier = banking_crash_intervention_suspend_convertibility }
			remove_modifier = banking_crash_intervention_suspend_convertibility
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_mutual_clearing }
			remove_modifier = banking_crash_intervention_mutual_clearing
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_correspondent_liquidity }
			remove_modifier = banking_crash_intervention_correspondent_liquidity
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_dev_bank_lending }
			remove_modifier = banking_crash_intervention_dev_bank_lending
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_open_market }
			remove_modifier = banking_crash_intervention_open_market
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_deposit_guarantee }
			remove_modifier = banking_crash_intervention_deposit_guarantee
		}
		else_if = {
			limit = { has_modifier = banking_crash_intervention_nationalize }
			remove_modifier = banking_crash_intervention_nationalize
		}
	}
}

# Run contagion crash check in event .7 immediate block.
# Probability based on TARGET's own bubble pressure, elevated by origin crash severity.
# Countries with low bubble pressure are mostly immune; high bubble = very vulnerable.
# Sets contagion_crash = 1 (crash) or 0 (scare).
# If crash: calculates crash_severity from target's bubble + origin boost + random multiplier.
banking_contagion_crash_check = {
	random_list = {
		0 = {
			modifier = {
				# Target's own bubble pressure is the primary crash factor
				add = var:bubble_pressure
				# Boost from origin crash severity — even responsible countries
				# face elevated risk from a severe foreign collapse
				if = {
					limit = { scope:banking_crash_origin_country = { var:crash_severity >= 80 } }
					add = 30
				}
				else_if = {
					limit = { scope:banking_crash_origin_country = { var:crash_severity >= 60 } }
					add = 20
				}
				else_if = {
					limit = { scope:banking_crash_origin_country = { var:crash_severity >= 40 } }
					add = 10
				}
				else = {
					add = 5
				}
				if = {
					limit = { has_law = law_type:law_command_economy }
					multiply = 0.20
				}
				else_if = {
					limit = { has_law = law_type:law_cooperative_ownership }
					multiply = 0.55
				}
				multiply = banking_crash_chance_multiplier_value
				min = 0
			}
			set_variable = { name = contagion_crash value = 1 }
		}
		100 = {
			set_variable = { name = contagion_crash value = 0 }
		}
	}

	if = {
		limit = { var:contagion_crash > 0 }
		# History-chart marker on this month's bar (te_history_effects.txt)
		te_history_record_banking_marker = { MARK = crash_contagion }
		# Crash severity = target's own bubble pressure (scaled as for an origin
		# crash — banking_crash_severity_scale_value) + boost from origin severity
		set_variable = {
			name = crash_severity
			value = {
				value = var:bubble_pressure
				multiply = banking_crash_severity_scale_value
			}
		}
		if = {
			limit = { scope:banking_crash_origin_country = { var:crash_severity >= 80 } }
			change_variable = { name = crash_severity add = 20 }
		}
		else_if = {
			limit = { scope:banking_crash_origin_country = { var:crash_severity >= 60 } }
			change_variable = { name = crash_severity add = 15 }
		}
		else_if = {
			limit = { scope:banking_crash_origin_country = { var:crash_severity >= 40 } }
			change_variable = { name = crash_severity add = 10 }
		}
		else = {
			change_variable = { name = crash_severity add = 5 }
		}
		# Random severity multiplier (same distribution as origin crashes)
		random_list = {
			10 = { change_variable = { name = crash_severity multiply = 0.5 } }
			20 = { change_variable = { name = crash_severity multiply = 0.75 } }
			40 = { }
			20 = { change_variable = { name = crash_severity multiply = 1.5 } }
			10 = { change_variable = { name = crash_severity multiply = 2 } }
		}
		if = {
			limit = { has_law = law_type:law_command_economy }
			change_variable = { name = crash_severity multiply = 0.45 }
		}
		else_if = {
			limit = { has_law = law_type:law_cooperative_ownership }
			change_variable = { name = crash_severity multiply = 0.75 }
		}
		# [Monetary policy 5a, §15A.2] A lender of last resort whose word is still
		# good: an imported panic starts a fifth less deep, because depositors know
		# who stands behind the banks. Off while the guarantor's renege clock runs.
		if = {
			limit = { te_mon_has_effective_backstop = yes }
			change_variable = { name = crash_severity multiply = 0.8 }
		}
		# Clamp severity to 100
		if = {
			limit = { var:crash_severity > 100 }
			set_variable = { name = crash_severity value = 100 }
		}
	}
}

# Apply contagion effects to a country hit by financial contagion (event .7).
# Checks contagion_crash (set by banking_contagion_crash_check in immediate).
# If crash: applies full crash effects (same as origin crash).
# If scare: applies minor cycle/momentum reduction (markets wobble but hold).
apply_banking_contagion_effects = {
	if = {
		limit = { var:contagion_crash > 0 }
		# Full crash — reuse origin crash effects (cycle reset, bubble clear)
		apply_banking_crash_origin_effects = yes
	}
	else = {
		# Scare — spillovers are muted in more state-directed systems.
		if = {
			limit = { has_law = law_type:law_command_economy }
			custom_tooltip = {
				text = CONTAGION_SCARE_CYCLE_CMD_DESC
				change_variable = { name = finance_cycle_value add = -1 }
			}
			custom_tooltip = {
				text = CONTAGION_SCARE_MOMENTUM_CMD_DESC
				change_variable = { name = finance_cycle_momentum add = -0.5 }
			}
		}
		else_if = {
			limit = { has_law = law_type:law_cooperative_ownership }
			custom_tooltip = {
				text = CONTAGION_SCARE_CYCLE_COOP_DESC
				change_variable = { name = finance_cycle_value add = -2 }
			}
			custom_tooltip = {
				text = CONTAGION_SCARE_MOMENTUM_COOP_DESC
				change_variable = { name = finance_cycle_momentum add = -0.5 }
			}
		}
		else = {
			custom_tooltip = {
				text = CONTAGION_SCARE_CYCLE_DESC
				change_variable = { name = finance_cycle_value add = -3 }
			}
			custom_tooltip = {
				text = CONTAGION_SCARE_MOMENTUM_DESC
				change_variable = { name = finance_cycle_momentum add = -1 }
			}
		}
		# Clamp
		if = {
			limit = { var:finance_cycle_value < 5 }
			set_variable = { name = finance_cycle_value value = 5 }
		}
		if = {
			limit = { var:finance_cycle_momentum < -5 }
			set_variable = { name = finance_cycle_momentum value = -5 }
		}
	}
}

# ---- Crisis Wave Tracking (Great Depression mechanic) ----

# Initialize a new crisis wave when an origin crash fires.
# Clears any previous wave tracking and starts fresh.
banking_crisis_wave_init = {
	# Clear previous wave flags from all countries
	every_country = {
		limit = {
			OR = {
				has_variable = banking_crisis_wave_crashed
				has_variable = banking_crisis_wave_counted
			}
		}
		if = { limit = { has_variable = banking_crisis_wave_crashed } remove_variable = banking_crisis_wave_crashed }
		if = { limit = { has_variable = banking_crisis_wave_counted } remove_variable = banking_crisis_wave_counted }
	}
	# Initialize global wave counter
	set_global_variable = { name = great_depression_wave_count value = 1 }
	set_global_variable = { name = great_depression_wave_gdp value = this_country_gdp_value }
	# Mark this country as part of the wave
	set_variable = { name = banking_crisis_wave_crashed value = 1 }
}

# Increment wave tracking when a contagion crash triggers in a target country.
# Called from event .7 after block when contagion_crash > 0.
banking_crisis_wave_increment = {
	if = {
		limit = { NOT = { has_variable = banking_crisis_wave_counted } }
		set_variable = { name = banking_crisis_wave_counted value = 1 }
		if = {
			limit = { has_global_variable = great_depression_wave_count }
			change_global_variable = { name = great_depression_wave_count add = 1 }
			change_global_variable = { name = great_depression_wave_gdp add = this_country_gdp_value }
		}
	}
}

# Check if the crisis wave has reached Great Depression thresholds.
# Requirements:
#   - 10+ countries crashed in the wave
#   - Wave GDP >= 20% of global GDP
#   - No country currently has a Great Depression modifier (acts as cooldown)
# If triggered, fires the Great Depression event for all countries with banking systems.
banking_crisis_wave_check_great_depression = {
	if = {
		limit = {
			has_global_variable = great_depression_wave_count
			global_var:great_depression_wave_count >= 10
			global_var:great_depression_wave_gdp >= great_depression_gdp_threshold
			# Cooldown: no new Great Depression while any country still has the modifier
			NOT = {
				any_country = {
					OR = {
						has_modifier = great_depression_modifier
						has_modifier = great_depression_bystander_modifier
						has_modifier = great_depression_isolated_modifier
					}
				}
			}
		}
		# Fire Great Depression event for all countries with banking journals
		every_country = {
			limit = {
				has_journal_entry = je_banking_cycle
			}
			trigger_event = { id = minor_events_timelineextended.8 days = 3 }
		}
	}
}

# 1. Fiscal policy modifier: re-apply every month to reflect current deficit/surplus.
banking_cycle_update_fiscal_policy = {
	je:je_banking_cycle = { remove_modifier = financial_cycle_government_fiscal_policy_effect }
	je:je_banking_cycle = { add_modifier = {
			name = financial_cycle_government_fiscal_policy_effect
			multiplier = financial_cycle_government_fiscal_policy_effect_size
		} }

# 2. Advance all cycle variables for the month.
#    - Momentum decays towards zero (×0.9)
#    - Modifier-driven changes to momentum, bubble, and value
#    - Investment pool contribution
#    - Random mean-reversion nudge (biased towards equilibrium)
#    - Apply momentum to value
#    - Clamp all variables to valid ranges
banking_cycle_advance_variables = {
	# Momentum decays towards zero
	change_variable = { name = finance_cycle_momentum multiply = 0.9 }

	# Modifier-driven monthly changes
	change_variable = {
		name = finance_cycle_momentum
		add = "modifier:country_finance_momentum_monthly_add"
	}
	change_variable = {
		name = bubble_pressure
		add = "modifier:country_bubble_pressure_monthly_add"
	}
	change_variable = {
		name = finance_cycle_value
		add = "modifier:country_finance_value_monthly_add"
	}

	# Monetary stance (monetary_policy_design.md §8, §16.3).
	#
	# A tight policy rate leans against the cycle and lets air out of the
	# bubble; a loose one does the opposite. This is deliberately NOT a
	# modifier: a stance modifier scaled by the gap would print
	# -0.125 x (policy rate - r*) in its tooltip, and the hidden neutral rate
	# could then be read off it to two decimals. The push therefore lands
	# straight in the variables, and the player is told the stance BAND
	# (banking_dash_momentum_tt, and the banded journal-entry modifier applied
	# by banking_cycle_apply_stance_band) but never the number.
	#
	# Both script values are guarded and clamp the gap to -2..+4, so a country
	# whose monetary update has not run yet — or which has no policy dial at
	# all — is pushed by exactly 0, and a gap one month old is tolerated by
	# design (§16.3). Placed with the other monthly pushes and ahead of the
	# clamps below, so the bubble term cannot leave the 0-100 range behind.
	change_variable = {
		name = finance_cycle_momentum
		add = te_mon_stance_momentum_add
	}
	change_variable = {
		name = bubble_pressure
		add = te_mon_stance_bubble_add
	}

	# Random mean-reversion nudge: biased towards equilibrium
	# Magnitude scales with country_banking_random_momentum_mult modifier
	random_list = {
		25 = {
			modifier = {
				value = var:finance_cycle_value
				subtract = 50
				divide = 5
			}
			change_variable = { name = finance_cycle_momentum add = banking_random_nudge_down_value }
		}
		50 = { }
		25 = {
			modifier = {
				value = 50
				subtract = var:finance_cycle_value
				divide = 5
			}
			change_variable = { name = finance_cycle_momentum add = banking_random_nudge_up_value }
		}
	}

	# Apply momentum to cycle value
	change_variable = {
		name = finance_cycle_value
		add = var:finance_cycle_momentum
	}

	# Clamp cycle value 0–100
	if = { limit = { var:finance_cycle_value < 0 }   set_variable = { name = finance_cycle_value value = 0 } }
	if = { limit = { var:finance_cycle_value > 100 } set_variable = { name = finance_cycle_value value = 100 } }

	# Clamp bubble pressure 0–100
	if = { limit = { var:bubble_pressure < 0 }   set_variable = { name = bubble_pressure value = 0 } }
	if = { limit = { var:bubble_pressure > 100 } set_variable = { name = bubble_pressure value = 100 } }
}

# 3. Check for a sudden crash (asymmetric probability).
#    If crash triggers, fire the origin event and spread contagion to
#    trade-linked countries weighted by GDP.
banking_cycle_check_and_execute_crash = {
	set_variable = { name = crash value = 0 }
	random_list = {
		0 = {
			modifier = {
				add = var:bubble_pressure
				if = {
					limit = { var:finance_cycle_value >= 88 } # frenzy
					subtract = 15
					multiply = 0.35
				}
				else_if = {
					limit = { var:finance_cycle_value >= 75 } # boom
					subtract = 25
					multiply = 0.15
				}
				else_if = {
					limit = { var:finance_cycle_value >= 60 } # expansion
					subtract = 50
					multiply = 0.05
				}
				else_if = {
					limit = { var:finance_cycle_value >= 40 } # stable
					subtract = 75
					multiply = 0.02
				}
				else = {
					# stagnation and below. Without this branch the weight fell
					# through as RAW bubble pressure, so a country dropping out of
					# a boom through 40 with its bubble intact crashed at
					# bubble / (bubble + 100) a month — a third at bubble 50 —
					# while two points higher it was zero
					# (docs/audits/banking_cycle_simulation.md F4).
					subtract = 90
					multiply = 0.01
				}
				if = {
					limit = { has_law = law_type:law_command_economy }
					multiply = 0.35
				}
				else_if = {
					limit = { has_law = law_type:law_cooperative_ownership }
					multiply = 0.65
				}
				min = 0
				# Scale crash probability by the crash chance modifier
				multiply = banking_crash_chance_multiplier_value
			}
			set_variable = { name = crash value = 1 }
		}
		100 = { }
	}

	if = {
		limit = { var:crash > 0 }
		save_scope_as = banking_crash_origin_country

		# History-chart marker on this month's bar (te_history_effects.txt)
		te_history_record_banking_marker = { MARK = crash }

		# Severity scales with bubble pressure (random multiplier 0.5–2.0).
		# Seeded from banking_crash_severity_scale_value x bubble, not bubble
		# itself: the crash PROBABILITY is already proportional to bubble, so a
		# crash only ever fires off a big bubble, and seeding severity from the
		# same number made two thirds of all crashes panic-tier resets
		# (docs/audits/banking_cycle_simulation.md F2).
		set_variable = {
			name = crash_severity
			value = {
				value = var:bubble_pressure
				multiply = banking_crash_severity_scale_value
			}
		}
		random_list = {
			10 = { change_variable = { name = crash_severity multiply = 0.5 } }
			20 = { change_variable = { name = crash_severity multiply = 0.75 } }
			40 = { }
			20 = { change_variable = { name = crash_severity multiply = 1.5 } }
			10 = { change_variable = { name = crash_severity multiply = 2 } }
		}
		if = {
			limit = { has_law = law_type:law_command_economy }
			change_variable = { name = crash_severity multiply = 0.6 }
		}
		else_if = {
			limit = { has_law = law_type:law_cooperative_ownership }
			change_variable = { name = crash_severity multiply = 0.8 }
		}
		if = {
			limit = { var:crash_severity > 100 }
			set_variable = { name = crash_severity value = 100 }
		}

		# Initialize crisis wave tracking for Great Depression mechanic
		banking_crisis_wave_init = yes

		trigger_event = { id = minor_events_timelineextended.6 popup = yes }

		# Spread contagion to trade-linked countries
		banking_cycle_spread_contagion = yes
	}
}

# 3b. Spread financial contagion to eligible countries.
#     Probability depends on trade linkage, severity, AND relative GDP.
#     Countries where the source is less than ~1% of target GDP are immune.
banking_cycle_spread_contagion = {
	custom_tooltip = {
		text = BANKING_CONTAGION_SPREAD_TT
		every_country = {
			limit = {
				NOT = { THIS = scope:banking_crash_origin_country }
				# Already hit by this crisis wave? Skip (prevents double-hits during cascades)
				NOT = { has_variable = banking_crisis_wave_crashed }
				has_journal_entry = je_banking_cycle
				# Trade-reachable
				OR = {
					this.market = scope:banking_crash_origin_country.market
					is_market_reachable_for_trade = scope:banking_crash_origin_country.market
				}
				# At least one trade channel
				OR = {
					this.market = scope:banking_crash_origin_country.market
					is_in_customs_union_with = scope:banking_crash_origin_country
					# [Monetary policy 5b, §15A.3] A shared currency is a channel too.
					te_mon_shares_currency_with = { OTHER = scope:banking_crash_origin_country }
					is_adjacent_to_country = scope:banking_crash_origin_country
					economic_dependence = {
						target = scope:banking_crash_origin_country
						value > 0.05
					}
					market = {
						market_trade_reliance = {
							target = scope:banking_crash_origin_country.market
							value > 0.01
						}
					}
					market = {
						market_production_share = {
							target = scope:banking_crash_origin_country
							value > 0.01
						}
					}
					market = {
						market_consumption_share = {
							target = scope:banking_crash_origin_country
							value > 0.01
						}
					}
				}
				# GDP filter: source must have at least 1% of target's GDP
				# Tiny economies cannot destabilise giants
				scope:banking_crash_origin_country.gdp >= banking_contagion_min_source_gdp
			}

			random = {
				chance = 25
				# GDP weight tiers: larger source -> higher contagion chance (approximates [0,50] range)
				# origin GDP >= affected GDP (same size or bigger)
				modifier = {
					trigger = { scope:banking_crash_origin_country.gdp >= gdp }
					add = 5
				}
				# origin GDP >= 2x affected
				modifier = {
					trigger = { scope:banking_crash_origin_country.gdp >= banking_contagion_gdp_threshold_2x }
					add = 10
				}
				# origin GDP >= 5x affected
				modifier = {
					trigger = { scope:banking_crash_origin_country.gdp >= banking_contagion_gdp_threshold_5x }
					add = 15
				}
				# origin GDP >= 10x affected
				modifier = {
					trigger = { scope:banking_crash_origin_country.gdp >= banking_contagion_gdp_threshold_10x }
					add = 20
				}
				# Same market = strongest contagion channel
				modifier = {
					trigger = { this.market = scope:banking_crash_origin_country.market }
					add = 10
				}
				# Hard crash spreads more easily
				modifier = {
					trigger = { scope:banking_crash_origin_country = { var:crash_severity >= 60 } }
					add = 10
				}
				modifier = {
					trigger = { scope:banking_crash_origin_country = { var:crash_severity >= 30 var:crash_severity < 60 } }
					add = 5
				}
				modifier = {
					trigger = { is_in_customs_union_with = scope:banking_crash_origin_country }
					add = 10
				}
				# [Monetary policy 5b, §15A.3] One currency, one banking system's worth
				# of exposure.
				modifier = {
					trigger = { te_mon_shares_currency_with = { OTHER = scope:banking_crash_origin_country } }
					add = 10
				}
				# [Monetary policy 5b tier 4] ...unless the currency area has one
				# supervisor and one deposit guarantee over all of it, which is most
				# of what the shared currency was carrying. Not all of it: a common
				# supervisor is not a wall, and 2008 and 2011 both crossed one.
				modifier = {
					trigger = {
						te_mon_shares_currency_with = { OTHER = scope:banking_crash_origin_country }
						te_mon_bloc_union_tier_4_plus = yes
					}
					add = te_mon_union_contagion_damp
				}
				modifier = {
					trigger = { is_adjacent_to_country = scope:banking_crash_origin_country }
					add = 5
				}
				modifier = {
					trigger = { economic_dependence = { target = scope:banking_crash_origin_country value > 0.10 } }
					add = 5
				}
				modifier = {
					trigger = { economic_dependence = { target = scope:banking_crash_origin_country value > 0.25 } }
					add = 10
				}
				modifier = {
					trigger = { market = { market_trade_reliance = { target = scope:banking_crash_origin_country.market value > 0.05 } } }
					add = 5
				}
				modifier = {
					trigger = { market = { market_trade_reliance = { target = scope:banking_crash_origin_country.market value > 0.1 } } }
					add = 10
				}
				modifier = {
					trigger = { scope:banking_crash_origin_country.relations:THIS >= relations_threshold:cordial }
					add = 1.5
				}
				modifier = {
					trigger = { has_law_or_variant = law_type:law_free_trade }
					add = 1.5
				}

				scope:banking_crash_origin_country = {
					save_scope_as = banking_crash_origin_country
				}

				# Mark target immediately so cascading spreads from other countries
				# in this wave cannot hit the same country twice
				set_variable = { name = banking_crisis_wave_crashed value = 1 }

				trigger_event = { id = minor_events_timelineextended.7 popup = yes days = 7 }
			}
		}
	}
}

# 4. Apply phase modifiers based on current cycle values.
#    Removes old phase & bubble-inertia modifiers, then applies current ones.
banking_cycle_apply_phase_modifiers = {
	remove_all_banking_phase_modifiers = yes

	# Bubble inertia: high speculation sustains expansion (positive feedback)
	if = {
		limit = {
			NOR = {
				has_law = law_type:law_command_economy
				has_law = law_type:law_cooperative_ownership
			}
		}
		if = {
			limit = { var:bubble_pressure >= 65 }
			je:je_banking_cycle = { add_modifier = { name = bubble_inertia_extreme multiplier = bubble_inertia_multiplier_script_value } }
		}
		else_if = {
			limit = { var:bubble_pressure >= 35 }
			je:je_banking_cycle = { add_modifier = { name = bubble_inertia_high multiplier = bubble_inertia_multiplier_script_value } }
		}
		else_if = {
			limit = { var:bubble_pressure > 0.5 }
			je:je_banking_cycle = { add_modifier = { name = bubble_inertia_moderate multiplier = bubble_inertia_multiplier_script_value } }
		}
	}

	# Phase modifier by economic law type
	if = {
		limit = { has_law = law_type:law_command_economy }
		if = { limit = { var:finance_cycle_value < 10 }  je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_panic_cmd } } }
		else_if = { limit = { var:finance_cycle_value < 25 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_downturn_cmd } } }
		else_if = { limit = { var:finance_cycle_value < 40 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_stagnation_cmd } } }
		else_if = { limit = { var:finance_cycle_value < 60 } } # Stable = no modifier
		else_if = { limit = { var:finance_cycle_value < 75 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_expansion_cmd } } }
		else_if = { limit = { var:finance_cycle_value < 88 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_boom_cmd } } }
		else = { je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_frenzy_cmd } } }
	}
	else_if = {
		limit = { has_law = law_type:law_cooperative_ownership }
		if = { limit = { var:finance_cycle_value < 10 }  je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_panic_coop } } }
		else_if = { limit = { var:finance_cycle_value < 25 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_downturn_coop } } }
		else_if = { limit = { var:finance_cycle_value < 40 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_stagnation_coop } } }
		else_if = { limit = { var:finance_cycle_value < 60 } } # Stable = no modifier
		else_if = { limit = { var:finance_cycle_value < 75 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_expansion_coop } } }
		else_if = { limit = { var:finance_cycle_value < 88 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_boom_coop } } }
		else = { je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_frenzy_coop } } }
	}
	else = {
		if = { limit = { var:finance_cycle_value < 10 }  je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_panic } } }
		else_if = { limit = { var:finance_cycle_value < 25 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_downturn } } }
		else_if = { limit = { var:finance_cycle_value < 40 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_stagnation } } }
		else_if = { limit = { var:finance_cycle_value < 60 } } # Stable = no modifier
		else_if = { limit = { var:finance_cycle_value < 75 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_expansion } } }
		else_if = { limit = { var:finance_cycle_value < 88 } je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_boom } } }
		else = { je:je_banking_cycle = { add_modifier = { name = financial_cycle_phase_frenzy } } }
	}
}

# 4b. Apply the displayed monetary-stance band as a journal-entry modifier.
#     (monetary_policy_design.md §8, §16.3.)
#
#     Five static modifiers, banking_stance_band_1..5, carry the investment-pool
#     side of the stance: +0.04 / +0.02 / nothing / -0.02 / -0.04 on
#     state_capitalists_investment_pool_contribution_add. Cheap money feeds the
#     pool, dear money starves it. Exactly one of them is on the journal entry at
#     any time, so the modifier list always NAMES the stance — which is the whole
#     point: the band is the only window onto the hidden gap (§8), and it leaks
#     nothing the band does not already show.
#
#     BAND 3 IS APPLIED EVEN THOUGH IT IS EMPTY. Vanilla does the same for the
#     neutral member of its own banded families — tax_modifier_medium,
#     government_wages_medium and infamy_reputable are all zero-field static
#     modifiers the engine applies so the neutral step still has a name in the
#     tooltip. Applying it also keeps the invariant above exactly true, and means
#     every country with the journal entry but no policy dial (bankless, command
#     economy, commodity money, crypto) reads "Monetary Stance: Neutral" rather
#     than showing nothing at all — which is what §8 says they are.
#
#     DECIDED FROM VARIABLES, NEVER FROM has_modifier. add_modifier and
#     remove_modifier results are not visible inside the effect block that issues
#     them (scripting_best_practices.md, "Modifier Values Are NOT Recalculated
#     Within a Single Effect Block"), so asking `has_modifier` here would read the
#     state this effect is in the middle of changing. te_mon_stance_band_applied
#     remembers which band was last put on instead. It initialises behind
#     has_variable, is never removed, and 0 means "none applied yet".
#
#     THE FIRST RUN REMOVES ALL FIVE. je_banking_cycle is can_revolution_inherit,
#     so the journal entry and its modifiers move to the successor tag while the
#     country variables do not. A successor would otherwise start at the 0
#     sentinel and add a second band on top of the inherited one. Removing all
#     five once, on the pulse that creates the bookkeeping variable, closes that
#     hole for the cost of five no-op removals per country per campaign.
#
#     A country without the journal entry never gets here: this is called from
#     je_banking_cycle's own monthly pulse.
banking_cycle_apply_stance_band = {
	if = {
		limit = { NOT = { has_variable = te_mon_stance_band_applied } }
		set_variable = { name = te_mon_stance_band_applied value = 0 }
		je:je_banking_cycle = { remove_modifier = banking_stance_band_1 }
		je:je_banking_cycle = { remove_modifier = banking_stance_band_2 }
		je:je_banking_cycle = { remove_modifier = banking_stance_band_3 }
		je:je_banking_cycle = { remove_modifier = banking_stance_band_4 }
		je:je_banking_cycle = { remove_modifier = banking_stance_band_5 }
	}

	if = {
		limit = {
			# te_mon_stance_band is written by te_monetary_update_stance
			# (common/scripted_effects/te_monetary_effects.txt); a country whose
			# monetary update has not run yet simply keeps whatever it has.
			has_variable = te_mon_stance_band
			NOT = { var:te_mon_stance_band_applied = var:te_mon_stance_band }
		}

		# Take off the band we put on last month.
		if = { limit = { var:te_mon_stance_band_applied = 1 } je:je_banking_cycle = { remove_modifier = banking_stance_band_1 } }
		else_if = { limit = { var:te_mon_stance_band_applied = 2 } je:je_banking_cycle = { remove_modifier = banking_stance_band_2 } }
		else_if = { limit = { var:te_mon_stance_band_applied = 3 } je:je_banking_cycle = { remove_modifier = banking_stance_band_3 } }
		else_if = { limit = { var:te_mon_stance_band_applied = 4 } je:je_banking_cycle = { remove_modifier = banking_stance_band_4 } }
		else_if = { limit = { var:te_mon_stance_band_applied = 5 } je:je_banking_cycle = { remove_modifier = banking_stance_band_5 } }

		# Put on this month's.
		if = { limit = { var:te_mon_stance_band = 1 } je:je_banking_cycle = { add_modifier = { name = banking_stance_band_1 } } }
		else_if = { limit = { var:te_mon_stance_band = 2 } je:je_banking_cycle = { add_modifier = { name = banking_stance_band_2 } } }
		else_if = { limit = { var:te_mon_stance_band = 3 } je:je_banking_cycle = { add_modifier = { name = banking_stance_band_3 } } }
		else_if = { limit = { var:te_mon_stance_band = 4 } je:je_banking_cycle = { add_modifier = { name = banking_stance_band_4 } } }
		else_if = { limit = { var:te_mon_stance_band = 5 } je:je_banking_cycle = { add_modifier = { name = banking_stance_band_5 } } }

		set_variable = { name = te_mon_stance_band_applied value = var:te_mon_stance_band }
	}
}

# 5. Update the four JE progress bars (the fourth, the policy-stance band,
#    is the monetary system's — see banking_policy_stance_bar).
#    Called from the monthly pulse (where scope:journal_entry is available).
banking_cycle_update_progress_bars = {
	scope:journal_entry = {
		set_bar_progress = { name = banking_cycle_value_bar value = ROOT.var:finance_cycle_value }
		set_bar_progress = { name = banking_bubble_pressure_bar value = ROOT.var:bubble_pressure }
		if = {
			limit = { ROOT = { has_variable = te_mon_stance_bar_pos } }
			set_bar_progress = { name = banking_policy_stance_bar value = ROOT.var:te_mon_stance_bar_pos }
		}
		if = {
			limit = { ROOT.var:finance_cycle_momentum < -5 }
			set_bar_progress = { name = banking_cycle_momentum_bar value = -5 }
		}
		else_if = {
			limit = { ROOT.var:finance_cycle_momentum > 5 }
			set_bar_progress = { name = banking_cycle_momentum_bar value = 5 }
		}
		else = {
			set_bar_progress = { name = banking_cycle_momentum_bar value = ROOT.var:finance_cycle_momentum }
		}
	}
}

# 6. Refresh phase modifiers and progress bars after a banking event option.
#    Called from banking event `after` blocks where scope:journal_entry
#    is NOT available, so we scope via the je: accessor instead.
banking_cycle_post_event_refresh = {
	hidden_effect = {
		# Clamp before bar update — events may push vars below 0 / above 100
		# and set_bar_progress with out-of-range values spams "clamping to min".
		if = { limit = { ROOT.var:finance_cycle_value < 0 }   ROOT = { set_variable = { name = finance_cycle_value value = 0 } } }
		if = { limit = { ROOT.var:finance_cycle_value > 100 } ROOT = { set_variable = { name = finance_cycle_value value = 100 } } }
		if = { limit = { ROOT.var:bubble_pressure < 0 }       ROOT = { set_variable = { name = bubble_pressure value = 0 } } }
		if = { limit = { ROOT.var:bubble_pressure > 100 }     ROOT = { set_variable = { name = bubble_pressure value = 100 } } }
		banking_cycle_apply_phase_modifiers = yes
		je:je_banking_cycle = {
			set_bar_progress = { name = banking_cycle_value_bar value = ROOT.var:finance_cycle_value }
			set_bar_progress = { name = banking_bubble_pressure_bar value = ROOT.var:bubble_pressure }
			if = {
				limit = { ROOT = { has_variable = te_mon_stance_bar_pos } }
				set_bar_progress = { name = banking_policy_stance_bar value = ROOT.var:te_mon_stance_bar_pos }
			}
			if = {
				limit = { ROOT.var:finance_cycle_momentum < -5 }
				set_bar_progress = { name = banking_cycle_momentum_bar value = -5 }
			}
			else_if = {
				limit = { ROOT.var:finance_cycle_momentum > 5 }
				set_bar_progress = { name = banking_cycle_momentum_bar value = 5 }
			}
			else = {
				set_bar_progress = { name = banking_cycle_momentum_bar value = ROOT.var:finance_cycle_momentum }
			}
		}
	}
}

# 7b. Directed credit: shed sectors beyond the cap. A law change can leave two
#     sectors running under a cap of one (leaving Directed Credit & Development
#     Banks). Lifted in reverse roster order, newest sector first, so the
#     long-standing Infrastructure tool is the last to go. Monthly rather than
#     on_law_enactment_pass: no same-tick modifier read, and it also catches a
#     law swapped by anything other than enactment.
#
#     banking_directed_credit_slots_free is read ONCE into a local: a
#     remove_modifier is not visible to the modifier reads it is built from
#     until the block ends, so each shed counts itself up by hand.
banking_directed_credit_enforce_cap = {
	if = {
		limit = { banking_directed_credit_slots_free < 0 }
		set_local_variable = { name = dc_slots value = banking_directed_credit_slots_free }
		if = {
			limit = { local_var:dc_slots < 0 banking_tool_directed_credit_electrification_active = yes }
			te_history_record_banking_marker = { MARK = off_cb_directed_credit_electrification }
			je:je_banking_cycle = { remove_modifier = banking_directed_credit_electrification }
			change_local_variable = { name = dc_slots add = 1 }
		}
		if = {
			limit = { local_var:dc_slots < 0 banking_tool_directed_credit_armaments_active = yes }
			te_history_record_banking_marker = { MARK = off_cb_directed_credit_armaments }
			je:je_banking_cycle = { remove_modifier = banking_directed_credit_armaments }
			change_local_variable = { name = dc_slots add = 1 }
		}
		if = {
			limit = { local_var:dc_slots < 0 banking_tool_directed_credit_agriculture_active = yes }
			te_history_record_banking_marker = { MARK = off_cb_directed_credit_agriculture }
			je:je_banking_cycle = { remove_modifier = banking_directed_credit_agriculture }
			change_local_variable = { name = dc_slots add = 1 }
		}
		if = {
			limit = { local_var:dc_slots < 0 banking_tool_directed_credit_heavy_industry_active = yes }
			te_history_record_banking_marker = { MARK = off_cb_directed_credit_heavy_industry }
			je:je_banking_cycle = { remove_modifier = banking_directed_credit_heavy_industry }
			change_local_variable = { name = dc_slots add = 1 }
		}
		if = {
			limit = { local_var:dc_slots < 0 banking_tool_directed_active = yes }
			te_history_record_banking_marker = { MARK = off_cb_directed_credit_infrastructure }
			je:je_banking_cycle = { remove_modifier = banking_directed_credit_infrastructure }
		}
	}
}

# 8. Command economy: keep investment pool net income at zero.
#    Under law_command_economy the state should be funding all planned investment
#    directly from the treasury. This effect removes and re-applies the
#    planning_treasury_pool_balance modifier on the JE with a multiplier chosen
#    so that investment_pool_net_income + multiplier = 0 next tick.
#
#    Note: no prior-multiplier subtraction is needed because
#    country_weekly_investment_pool_add is a mod-created modifier key that the
#    engine does not include in investment_pool_net_income, so our contribution
#    is never double-counted in the read.
#
#    Called from:
#      - je_banking_cycle on_monthly_pulse (monthly recalibration)
#      - ce_invest_pool_inject / ce_invest_pool_withdraw button effects
#        (refresh after one-shot treasury<->pool transfers)
banking_cycle_update_ce_pool_balance = {
	if = {
		limit = { has_law = law_type:law_command_economy }
		je:je_banking_cycle = {
			if = {
				limit = { has_modifier = planning_treasury_pool_balance }
				remove_modifier = planning_treasury_pool_balance
			}
			add_modifier = {
				name = planning_treasury_pool_balance
				multiplier = ce_treasury_pool_balance_multiplier
			}
		}
	}
	else = {
		# Law is no longer command economy — strip the balance modifier.
		if = {
			limit = { je:je_banking_cycle ?= { has_modifier = planning_treasury_pool_balance } }
			je:je_banking_cycle = { remove_modifier = planning_treasury_pool_balance }
		}
	}
}
