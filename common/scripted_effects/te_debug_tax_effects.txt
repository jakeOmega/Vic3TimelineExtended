# Scope country. Arm is read-only economically, and does not reset an existing run.
te_tp_arm = {
	if = {
		limit = { NOT = { has_variable = te_tp_armed } }
		set_variable = { name = te_tp_armed value = 1 }
		set_variable = { name = te_tp_version value = 0 }
		set_variable = { name = te_tp_revision value = 0 }
		set_variable = { name = te_tp_operative value = 0 }
		set_variable = { name = te_tp_stage value = 0 }
		set_variable = { name = te_tp_snapshot_seq value = 0 }
	}
	te_tp_snapshot = yes
}

# Scope country. Parameter CLAUSE is the explicit suffix of a probe amendment.
# Idempotent per clause; preserves other clauses for the two-clause experiment.
te_tp_add_clause = {
	active_law:lawgroup_taxation = {
		if = {
			limit = { NOT = { has_amendment = amendment_type:amendment_te_tp_$CLAUSE$ } }
			add_amendment = { type = amendment_te_tp_$CLAUSE$ sponsor = ROOT.ig:ig_intelligentsia cooldown = 0 }
		}
	}
}

					type = amendment_type:amendment_te_tp_rural
					type = amendment_type:amendment_te_tp_head

					type = amendment_type:amendment_te_tp_consumption_high
				} }
			remove_amendment = yes

	}
	remove_modifier = te_tp_static_wage
}

# Bounded records test storage, iteration, missing native inheritance and reload.
# No timed variables on containers; the date is an absolute calendar-month index.
te_tp_append_record = {
	if = {
		limit = { OR = { NOT = { has_variable_list = te_tp_records } NOT = { any_in_list = { variable = te_tp_records count >= 2 } } } }
		save_scope_as = te_tp_owner
		create_container = { tags = { te_tp_record } save_scope_as = te_tp_new }
		scope:te_tp_new = {
			set_variable = { name = te_tp_record_owner value = scope:te_tp_owner }
			set_variable = { name = te_tp_record_month value = te_history_month_index }
		}
		add_to_variable_list = { name = te_tp_records target = scope:te_tp_new }
		debug_log = "TE_TAX_PROBE record appended"
	}
	else = { debug_log = "TE_TAX_PROBE record rejected: capacity two" }
	te_tp_inspect_records = yes
}
te_tp_inspect_records = {
	set_variable = { name = te_tp_record_count value = 0 }
	save_scope_as = te_tp_owner
	if = {
		limit = { has_variable_list = te_tp_records }
		every_in_list = {
			variable = te_tp_records
			if = {
				limit = { var:te_tp_record_owner = scope:te_tp_owner }
				scope:te_tp_owner = { change_variable = { name = te_tp_record_count add = 1 } }
			}
			else = { debug_log = "TE_TAX_PROBE record owner mismatch" }
		}
	}
	debug_log = "TE_TAX_PROBE owned records=[SCOPE.ScriptValue('te_tp_view_record_count')|0]"
}

# These are deliberately simplified approval commands, not a production legislature.
te_tp_draft = {
	if = {
		limit = { has_variable = te_tp_armed has_law = law_type:law_te_probe_carrier NOT = { has_variable = te_tp_pending } }
		change_variable = { name = te_tp_revision add = 1 }
		set_variable = { name = te_tp_draft_revision value = var:te_tp_revision }
		set_variable = { name = te_tp_draft_base value = var:te_tp_version }
		set_variable = { name = te_tp_stage value = 1 }
		debug_log = "TE_TAX_PROBE draft only: no collection writes"
	}
}
te_tp_approve = {
	if = {
		limit = { te_tp_can_approve = yes }
		set_variable = { name = te_tp_approved_revision value = var:te_tp_draft_revision }
		set_variable = { name = te_tp_approved_base value = var:te_tp_draft_base }
		set_variable = { name = te_tp_due value = te_history_month_index }
		change_variable = { name = te_tp_due add = 1 }
		set_variable = { name = te_tp_expiry value = var:te_tp_due }
		change_variable = { name = te_tp_expiry add = 1 }
		set_variable = { name = te_tp_pending value = 1 }
		if = { limit = { has_variable = te_tp_customs_clause } set_variable = { name = te_tp_approved_customs value = 1 } }
		set_variable = { name = te_tp_stage value = 2 }
		remove_variable = te_tp_draft_revision
		debug_log = "TE_TAX_PROBE approved: both clauses due next month, zero successor one month later"
	}
}
# Single writer for the integration experiment. Both adds run in one effect;
# whether economy accounting is atomic is a runtime question, not asserted here.
te_tp_rebuild = {
	if = {
		limit = { has_variable = te_tp_armed has_law = law_type:law_te_probe_carrier }
		te_tp_clear_rates = yes
		if = {
			limit = { var:te_tp_operative = 1 }
			te_tp_add_clause = { CLAUSE = wage }
			te_tp_add_clause = { CLAUSE = dividend }
			if = { limit = { has_variable = te_tp_approved_customs } set_import_tariff_level = { goods = g:grain level = low_tariffs } }
		}
	}
}
te_tp_process_queue = {
	# Callers are armed country events; never call from GUI preview.
	if = {
		limit = { has_variable = te_tp_armed has_variable = te_tp_pending te_history_month_index >= var:te_tp_due }
		if = {
			limit = { var:te_tp_approved_base = var:te_tp_version has_law = law_type:law_te_probe_carrier }
			set_variable = { name = te_tp_operative value = 1 }
			change_variable = { name = te_tp_version add = 1 }
			set_variable = { name = te_tp_expiry_version value = var:te_tp_version }
			remove_variable = te_tp_pending
			set_variable = { name = te_tp_stage value = 3 }
			te_tp_rebuild = yes
			debug_log = "TE_TAX_PROBE commenced both clauses"
		}
		else = {
			set_variable = { name = te_tp_stage value = 4 }
			debug_log = "TE_TAX_PROBE held entire package: baseline or carrier conflict"
		}
	}
	if = {
		limit = { has_variable = te_tp_armed has_variable = te_tp_expiry NOT = { has_variable = te_tp_pending } has_variable = te_tp_expiry_version te_history_month_index >= var:te_tp_expiry }
		if = {
			limit = { var:te_tp_version = var:te_tp_expiry_version }
			set_variable = { name = te_tp_operative value = 0 }
			change_variable = { name = te_tp_version add = 1 }
			set_variable = { name = te_tp_stage value = 5 }
			te_tp_rebuild = yes
			if = { limit = { has_variable = te_tp_approved_customs } set_import_tariff_level = { goods = g:grain level = no_tariffs_or_subventions } }
			debug_log = "TE_TAX_PROBE expired to explicit zero successor"
		}
		else = { debug_log = "TE_TAX_PROBE stale expiry ignored" }
		remove_variable = te_tp_expiry
	}
}
te_tp_monthly = {
	if = {
		limit = { has_variable = te_tp_armed }
		# A repeated event in the same month must not advance AI twice.
		if = { limit = { NOT = { has_variable = te_tp_last_month } } set_variable = { name = te_tp_last_month value = -1 } }
		if = {
			limit = { var:te_tp_last_month < te_history_month_index }
			set_variable = { name = te_tp_last_month value = te_history_month_index }
			te_tp_process_queue = yes
			if = {
				limit = { is_ai = yes has_variable = te_tp_ai_step }
				if = {
					limit = { var:te_tp_ai_step = 1 }
					te_tp_draft = yes
					set_variable = { name = te_tp_ai_step value = 2 }
				}
				else_if = {
					limit = { var:te_tp_ai_step = 2 te_tp_can_approve = yes }
					te_tp_approve = yes
					remove_variable = te_tp_ai_step
				}
			}
			te_tp_snapshot = yes
		}
	}
}
