
# Set the building to exactly SPEC_LEVEL.
# - If absent: create at SPEC_LEVEL.
# - If present at the right level: do nothing (avoid destroy/recreate
#   churn that would fire all employees every weekly/recalc tick).
# - If present at a different level: remove, then create at SPEC_LEVEL.
#
# Previous form was the no-op-when-level-already-matches case BUT with the
# inverted comparison (only removed when current < target), and unconditional
# create_building afterwards. Because create_building stacks levels onto an
# existing building of the same type, any call where current >= target added
# a level instead of resetting. Border-change recalcs in
# te_construction_market_on_actions fire this with SPEC_LEVEL = 1 repeatedly
# (every war / formable / secession / market change / yearly pulse), so an
# existing level-N site would creep up by 1 each event — directly responsible
# for the multi-level sites that triggered the staffing-compensator
# throughput inflation.
#
# The "skip when level already matches" guard avoids the inverse failure
# mode: unconditional remove+create on every recalc destroys the site,
# fires all employees, and the next tick repeats the cycle — leaving the
# state with no effective construction throughput at all.
te_construction_market_set_specified_level = {
	if = {
		limit = {
			has_building = $SPEC_TYPE$
			NOT = { b:$SPEC_TYPE$.level = $SPEC_LEVEL$ }
		}
		remove_building = $SPEC_TYPE$
	}

	if = {
		limit = {
			NOT = { has_building = $SPEC_TYPE$ }
		}
		te_construction_market_build_specified_level = {
			SPEC_TYPE = $SPEC_TYPE$
			SPEC_LEVEL = $SPEC_LEVEL$
		}
	}
}

# Add ADD_LEVEL levels of SPEC_TYPE on top of whatever the state already has.
# Call from STATE scope. ADD_LEVEL is a state-relative value reference (the
# live callers pass `var:lvl_tmp`, a state variable): the arithmetic below runs
# inside `owner = { }`, so the helper prefixes `prev.` itself to hop back to the
# state. Pass a bare `var:X`, never `prev.var:X` / `owner.var:X`.
te_construction_market_add_specified_level = {
	owner = {
		set_variable = {
			name = add_level
			value = 0
		}
		if = {
			limit = {
				prev = {
					has_building = $SPEC_TYPE$
				}

			}
			change_variable = {
				name = add_level
				add = prev.b:$SPEC_TYPE$.level
			}
		}
		change_variable = {
			name = add_level
			add = prev.$ADD_LEVEL$
		}
	}
	te_construction_market_build_specified_level = {
		SPEC_TYPE = $SPEC_TYPE$
		SPEC_LEVEL = owner.var:add_level
	}

}
