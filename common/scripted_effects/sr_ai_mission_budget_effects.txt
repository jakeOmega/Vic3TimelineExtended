# ============================================================================
# THE AI'S SPACE PROGRAM BUDGET (docs/systems/mod_systems/ai.md § Space Program Mission Budget)
# ============================================================================
# Country scope, once a month for every country (space_race_on_action), with
# or without the Space Race rule: the program buys its Launch Capacity either
# way. An AI country's Space Program runs the dearest mission its government
# can pay for, as a player would choose; a player's is never touched, beyond
# lifting the AI's ceiling from a program a player has taken over.
#
# Why: the mission group's ai_selection = most_productive climbed every AI
# program to the dearest mission its technology allowed. In the owner's 1958
# save nine AI programs ran Solar Colonization, and the eight full-staffed
# ones paid 94% to 791% of their government's own revenue for it: 62% of all
# AI government goods spending in that save (study of 2026-11-15).
#
# The rule (sr_ai_mission_budget_values.txt for the numbers):
# - In default or under the bankruptcy modifier: Earth Orbit, at once.
# - The mission's own bill (unfunded, at full staff, at our market's price)
#   over 20% of fixed income: hold, no climbing; after six monthly reviews in
#   a row over it, step down to the dearest mission within 10%.
# - Otherwise: hold the mission, or climb to the dearest one within 10% that
#   the country's technology allows; not within five years of a step-down.
# - Funding is shed first (the AI's funding buttons, space_race_buttons.txt:
#   no increase past 10% with its funding, a decrease while over 20%), so a
#   mission steps down only when the mission itself is too dear.
#
# How: the pass records the ceiling it chose (var:sr_ai_mission_ceiling, a
# tier 1 to 6) and puts the program on the Mission Limit method of that tier
# (sr_mission_ceiling_pms.txt). The missions above it are then unavailable to
# the engine's AI, which would otherwise climb back past it under
# most_productive. The pass then sets the mission itself
# (var:sr_ai_mission_target): stepping down, the mission first and then the
# ceiling; climbing, the ceiling first, so neither move ever leaves the
# program on a mission its ceiling doesn't allow.
#
# What a step down costs: the Space Program buys no construction, so the
# mod's retooling surcharge (pm_retooling, a construction input multiplier)
# costs it nothing. But a lower mission fails every milestone the higher one
# carried (space_race.md, Programme loss), and that progress is lost; hence
# the six months' wait and the five years before it may climb again.
#
# Self-check sr_ai_mission_ceiling (sr_ai_mission_self_check, below).
# ============================================================================

sr_ai_mission_budget_monthly = {
	if = {
		limit = { sr_has_space_program = yes }
		if = {
			limit = { is_ai = yes }
			sr_ai_mission_budget_review = yes
		}
		else = {
			# A player's program: Any Mission (a country the player has just
			# taken over from the AI), and no record.
			if = {
				limit = {
					NOT = { sr_space_program_runs = { PM = pm_sr_mission_ceiling_none } }
				}
				activate_production_method = {
					building_type = building_space_program
					production_method = pm_sr_mission_ceiling_none
				}
			}
			sr_ai_mission_budget_forget = yes
		}
	}
	else_if = {
		limit = { has_variable = sr_ai_mission_ceiling }
		sr_ai_mission_budget_forget = yes
	}
}

sr_ai_mission_budget_forget = {
	if = {
		limit = { has_variable = sr_ai_mission_ceiling }
		remove_variable = sr_ai_mission_ceiling
	}
	if = {
		limit = { has_variable = sr_ai_mission_target }
		remove_variable = sr_ai_mission_target
	}
	if = {
		limit = { has_variable = sr_ai_mission_over_months }
		remove_variable = sr_ai_mission_over_months
	}
	if = {
		limit = { has_variable = sr_ai_mission_checked_months }
		remove_variable = sr_ai_mission_checked_months
	}
	if = {
		limit = { has_variable = sr_ai_mission_cooldown }
		remove_variable = sr_ai_mission_cooldown
	}
	if = {
		limit = { has_variable = sr_ai_mission_none_seen }
		remove_variable = sr_ai_mission_none_seen
	}
}

# An AI country's monthly review.
sr_ai_mission_budget_review = {
	# 1. Last month's settings, before anything moves.
	sr_ai_mission_self_check = yes

	# 2. This month's ceiling.
	if = {
		limit = { sr_ai_space_budget_in_trouble = yes }
		set_variable = { name = sr_ai_mission_ceiling value = 1 }
		if = {
			limit = { has_variable = sr_ai_mission_over_months }
			remove_variable = sr_ai_mission_over_months
		}
	}
	else_if = {
		limit = { sr_ai_mission_overrun > 0 }
		if = {
			limit = { has_variable = sr_ai_mission_over_months }
			change_variable = { name = sr_ai_mission_over_months add = 1 }
		}
		else = {
			set_variable = { name = sr_ai_mission_over_months value = 1 }
		}
		if = {
			limit = { var:sr_ai_mission_over_months >= sr_ai_mission_over_months_limit }
			set_variable = { name = sr_ai_mission_ceiling value = sr_ai_mission_affordable_tier }
			remove_variable = sr_ai_mission_over_months
		}
		else = {
			# Over, not yet for long enough: hold, and climb no further.
			set_variable = { name = sr_ai_mission_ceiling value = sr_ai_mission_tier_current }
		}
	}
	else = {
		if = {
			limit = { has_variable = sr_ai_mission_over_months }
			remove_variable = sr_ai_mission_over_months
		}
		if = {
			limit = { has_variable = sr_ai_mission_cooldown }
			set_variable = { name = sr_ai_mission_ceiling value = sr_ai_mission_tier_current }
		}
		else = {
			set_variable = { name = sr_ai_mission_ceiling value = sr_ai_mission_hold_or_climb_tier }
		}
	}

	# 3. The mission: the dearest under the ceiling that the technology allows.
	set_variable = { name = sr_ai_mission_target value = sr_ai_mission_target_tier }
	if = {
		# A step down: the mission first, then the ceiling, and five years
		# before the program may climb again (as long as the mod's retooling,
		# RETOOLING_WEEKS = 260), so an income or a price that swings around
		# a line can't fail the same milestones over and over.
		limit = { sr_ai_mission_target_gap > 0 }
		set_variable = { name = sr_ai_mission_cooldown value = yes days = 1825 }
		sr_ai_mission_set_mission = yes
		sr_ai_mission_set_ceiling = yes
	}
	else = {
		sr_ai_mission_set_ceiling = yes
		sr_ai_mission_set_mission = yes
	}
}

# The program onto mission var:sr_ai_mission_target, unless it runs it.
sr_ai_mission_set_mission = {
	if = {
		limit = {
			var:sr_ai_mission_target ?= 1
			NOT = { sr_space_program_runs = { PM = pm_earth_orbit } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_earth_orbit
		}
	}
	else_if = {
		limit = {
			var:sr_ai_mission_target ?= 2
			NOT = { sr_space_program_runs = { PM = pm_moon_mission } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_moon_mission
		}
	}
	else_if = {
		limit = {
			var:sr_ai_mission_target ?= 3
			NOT = { sr_space_program_runs = { PM = pm_mars_mission } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_mars_mission
		}
	}
	else_if = {
		limit = {
			var:sr_ai_mission_target ?= 4
			NOT = { sr_space_program_runs = { PM = pm_solar_colonization } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_solar_colonization
		}
	}
	else_if = {
		limit = {
			var:sr_ai_mission_target ?= 5
			NOT = { sr_space_program_runs = { PM = pm_deep_space_exploration } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_deep_space_exploration
		}
	}
	else_if = {
		limit = {
			var:sr_ai_mission_target ?= 6
			NOT = { sr_space_program_runs = { PM = pm_interstellar_mission } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_interstellar_mission
		}
	}
}

# Self-check sr_ai_mission_ceiling, once a game for each verdict. Three engine
# facts nobody had read: that script can set a government building's method
# for the AI, that it then stays, and that the engine's own method AI leaves a
# Mission Limit ceiling alone and climbs no mission past it.
#   BROKEN  the ceiling or the mission the pass set last month is not the one
#           the program runs now: the engine's AI moved one, or the switch
#           did not take; or the program is still on Any Mission at a second
#           review in a row, so the switch to a ceiling never took.
#   OK      twelve monthly reviews in which both held while the country's
#           technology allowed a dearer mission than its ceiling (so the
#           engine's AI, under most_productive, had a mission to climb to).
# Judged only for a program the pass has set. One on Any Mission at a single
# review is not judged: a Space Program founded since the last review starts
# there, and the record names the program it replaced.
sr_ai_mission_self_check = {
	if = {
		limit = {
			has_variable = sr_ai_mission_ceiling
			has_variable = sr_ai_mission_target
			sr_space_program_runs = { PM = pm_sr_mission_ceiling_none }
		}
		if = {
			limit = { has_variable = sr_ai_mission_none_seen }
			if = {
				limit = { NOT = { has_global_variable = sr_ai_mission_ceiling_broken_logged } }
				set_global_variable = { name = sr_ai_mission_ceiling_broken_logged value = yes }
				debug_log = "TE_SELF_CHECK BROKEN sr_ai_mission_ceiling country=[THIS.GetCountry.GetTagName] date=[TimeKeeper.GetCurrentDate.GetString] ceiling_set=[THIS.Var('sr_ai_mission_ceiling').GetValue|0] ceiling_active=0 mission_set=[THIS.Var('sr_ai_mission_target').GetValue|0] mission_now=[THIS.ScriptValue('sr_ai_mission_tier_current')|0] (tiers: 1 Earth Orbit, 2 Moon, 3 Mars, 4 Solar, 5 Deep Space, 6 Interstellar; ceiling 0 Any Mission) (the AI's Space Program is still on Any Mission at a second monthly review: activate_production_method did not move it onto a Mission Limit ceiling; docs/testing/space-race-playtest.md)"
			}
		}
		else = {
			set_variable = { name = sr_ai_mission_none_seen value = yes }
		}
	}
	else_if = {
		limit = {
			has_variable = sr_ai_mission_ceiling
			has_variable = sr_ai_mission_target
		}
		if = {
			limit = { has_variable = sr_ai_mission_none_seen }
			remove_variable = sr_ai_mission_none_seen
		}
		if = {
			limit = {
				OR = {
					sr_ai_mission_ceiling_gap > 0
					sr_ai_mission_ceiling_gap < 0
					sr_ai_mission_target_gap > 0
					sr_ai_mission_target_gap < 0
				}
			}
			if = {
				limit = { NOT = { has_global_variable = sr_ai_mission_ceiling_broken_logged } }
				set_global_variable = { name = sr_ai_mission_ceiling_broken_logged value = yes }
				debug_log = "TE_SELF_CHECK BROKEN sr_ai_mission_ceiling country=[THIS.GetCountry.GetTagName] date=[TimeKeeper.GetCurrentDate.GetString] ceiling_set=[THIS.Var('sr_ai_mission_ceiling').GetValue|0] ceiling_active=[THIS.ScriptValue('sr_ai_mission_ceiling_tier_active')|0] mission_set=[THIS.Var('sr_ai_mission_target').GetValue|0] mission_now=[THIS.ScriptValue('sr_ai_mission_tier_current')|0] tech=[THIS.ScriptValue('sr_ai_mission_tech_tier')|0] (tiers: 1 Earth Orbit, 2 Moon, 3 Mars, 4 Solar, 5 Deep Space, 6 Interstellar; ceiling 0 Any Mission) fixed_income=[THIS.ScriptValue('sr_ai_mission_fixed_income_read')|0] bill_now=[THIS.ScriptValue('sr_ai_mission_bill_now')|0] share_now_pct=[THIS.ScriptValue('sr_ai_mission_share_now_pct')|1] throughput_modifier=[THIS.ScriptValue('sr_ai_mission_throughput_read')|2] funding_levels=[THIS.ScriptValue('sr_ai_program_funding_levels')|0] (the Mission Limit ceiling or the mission the AI's budget pass set last month is not the one the Space Program runs: the engine's AI moved it, or the switch did not take; docs/testing/space-race-playtest.md)"
			}
		}
		else_if = {
			limit = { sr_ai_mission_tech_headroom > 0 }
			if = {
				limit = { has_variable = sr_ai_mission_checked_months }
				change_variable = { name = sr_ai_mission_checked_months add = 1 }
			}
			else = {
				set_variable = { name = sr_ai_mission_checked_months value = 1 }
			}
			if = {
				limit = {
					var:sr_ai_mission_checked_months >= 12
					NOT = { has_global_variable = sr_ai_mission_ceiling_ok_logged }
				}
				set_global_variable = { name = sr_ai_mission_ceiling_ok_logged value = yes }
				debug_log = "TE_SELF_CHECK OK sr_ai_mission_ceiling country=[THIS.GetCountry.GetTagName] date=[TimeKeeper.GetCurrentDate.GetString] ceiling=[THIS.Var('sr_ai_mission_ceiling').GetValue|0] mission=[THIS.Var('sr_ai_mission_target').GetValue|0] tech=[THIS.ScriptValue('sr_ai_mission_tech_tier')|0] (tiers: 1 Earth Orbit, 2 Moon, 3 Mars, 4 Solar, 5 Deep Space, 6 Interstellar) fixed_income=[THIS.ScriptValue('sr_ai_mission_fixed_income_read')|0] bill_now=[THIS.ScriptValue('sr_ai_mission_bill_now')|0] share_now_pct=[THIS.ScriptValue('sr_ai_mission_share_now_pct')|1] share_mission_pct=[THIS.ScriptValue('sr_ai_mission_share_current_pct')|1] throughput_modifier=[THIS.ScriptValue('sr_ai_mission_throughput_read')|2] funding_levels=[THIS.ScriptValue('sr_ai_program_funding_levels')|0] (for twelve monthly reviews the Space Program kept the mission and the Mission Limit ceiling the AI's budget pass set, though its technology allowed a dearer mission: script sets a government building's method for the AI, and the engine's AI keeps the ceiling and climbs no mission past it)"
			}
		}
	}
}
