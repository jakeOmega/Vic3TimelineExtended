# ============================================================================
# SOLAR SYSTEM COLONIZATION — colony modules: the queue, fabrication, shipments
# ============================================================================
# Design: docs/superpowers/specs/2026-10-25-solar-colonization-modules-and-bases-design.md
# (slice 4 of the scope, §§ 5-10). Simple and full colonies only.
#
# A world's module levels are on its global record (sr_site_lvl_<a|b>_<key>,
# sr_site_<hab|def|port>_<key>), so a cession or an heir takes them along. The
# holder queues modules in one national queue, first in first out (country
# variables sr_mq_<n>_site and sr_mq_<n>_type, places 1 to 20, with a count per
# world and type, sr_mq_<type>_<key>, for the caps and the panel); the Space
# Program's Module Fabrication method gives points a month
# (country_sr_module_fabrication_add), kept in sr_fab_stock; each 100 builds
# the queue's first entry. A built module fills one of eight shipment slots
# (sr_ms_<n>_site, _type, _arrive; sr_ms_count), waits for its body's window
# and lands after the transit its drive buys, from the global monthly step.
# Every world's shipments on their way are counted in sr_site_ship_<type>_<key>
# (global), rebuilt monthly from the slots. The per-world parts are generated
# (sr_sites_generated.txt, scripts/generators/gen_solar_sites.py); this file
# holds the rest. Module types: a / b the specializations (shipment code 1 / 2),
# hab 3, def 4, port 5.
# ============================================================================

# ---- The month's fabrication (the entry's pulse) ----------------------------------
# Country scope. The queue's counts rebuilt; the month's points into the stock,
# kept up to its cap so a stall (nothing queued, every slot taken) loses nothing
# short of it, or up to the queue's first module's price where that is more, so a
# module queued while the cap was higher is still built (sr_mod_fab_stock_fill_cap;
# review of #1847); a stock already above them is kept, not cut. Then up
# to sr_mod_fab_per_month (4) modules from the head of the queue, each into a free shipment slot, or
# for a lunar world into none (the owner, 2026-11-15), and only a head that can
# go (sr_mq_head_ready, generated): a module for a world with launch windows is
# built once its window is at most sr_mod_build_lead_months off, so none waits
# for a window in a slot, and nothing is built while launches are barred (the
# owner, 2026-11-23). While the head can't be built, the first module further
# down that can, and that the stock covers, goes ahead of it
# (sr_mq_promote_ready). Under Full Colonies an occupied launch site
# fabricates nothing (slice 6 fleets design § 5.1: the Space Program's method
# still gives its country points from a state it doesn't control); a blockade in
# orbit doesn't stop fabrication, and the modules wait at Earth.
sr_mod_month = {
	if = {
		limit = { sr_colonies_contested_enabled = yes }
		sr_mq_recount = yes
		# A level being dismantled at each world, a month on (owner, 2026-10-31).
		sr_md_month = yes
		sr_mod_fabrication_self_check = yes
		sr_mod_storage_self_check = yes
		sr_marker_self_check = yes
		if = {
			limit = {
				sr_mod_fab_rate > 0
				NOT = { sr_launch_site_lost = yes }
				sr_mod_fab_stock_value < sr_mod_fab_stock_fill_cap
			}
			set_variable = {
				name = sr_fab_stock
				value = {
					value = sr_mod_fab_stock_value
					add = sr_mod_fab_rate
					max = sr_mod_fab_stock_fill_cap
				}
			}
		}
		set_local_variable = { name = sr_fab_built value = 0 }
		sr_mq_promote_ready = yes
		while = {
			limit = {
				local_var:sr_fab_built < sr_mod_fab_per_month
				NOT = { sr_launch_site_lost = yes }
				sr_mod_fab_stock_value >= sr_mq_head_cost
				sr_mq_head_ready = yes
				has_variable = sr_mq_1_site
				has_variable = sr_mq_1_type
			}
			change_local_variable = { name = sr_fab_built add = 1 }
			sr_mod_fab_head = yes
			sr_mq_promote_ready = yes
		}
	}
}

# Country scope, from sr_mod_fab_head (the queue's first entry, already out of
# the queue): one module of TYPE (shipment code CODE) for the world KEY (index
# INDEX) costs its points (local_var:sr_mq_head_cost_now, the head's price
# before it left: past the world's slots it costs more, colony management
# design § 5) and fills a shipment slot; the world counts it on its way at
# once, so the queue's caps see it the same month.
sr_mod_fab_build = {
	change_variable = { name = sr_fab_stock subtract = local_var:sr_mq_head_cost_now }
	sr_ms_count_add = { VAR = sr_site_ship_$TYPE$_$KEY$ }
	set_local_variable = { name = sr_ms_new_site value = $INDEX$ }
	set_local_variable = { name = sr_ms_new_type value = $CODE$ }
	sr_ms_store_new = yes
}

# Country scope, from sr_mod_fab_head for a lunar world (the owner, 2026-11-15):
# its module takes no shipment slot. Earth's own Moon is reached the month a
# module is built (the transit is 0), so while launches aren't barred and the
# world is neither blockaded nor occupied it is installed at once, through the
# landing's own install (sr_mod_install_<key>; a full cap loses it, as a landing
# does). Since the owner's 2026-11-23 rule sr_mod_month builds a lunar head only
# when these three tests hold (sr_mq_head_ready reads them), so a module for a
# blockaded or occupied Moon colony waits in the queue; the slot below is a
# fallback for any other caller: the module waits there like any other and lands
# when it would have, lost if the world is still blockaded or occupied then
# (review of #1847: paying for a module lost the same month).
sr_mod_fab_build_lunar = {
	if = {
		limit = {
			NOT = { sr_launches_barred = yes }
			NOT = { has_global_variable = sr_site_blockaded_$KEY$ }
			NOT = { has_global_variable = sr_site_occupier_$KEY$ }
		}
		change_variable = { name = sr_fab_stock subtract = local_var:sr_mq_head_cost_now }
		set_local_variable = { name = sr_ms_type_now value = $CODE$ }
		set_local_variable = { name = sr_ms_installed value = 0 }
		sr_mod_install_$KEY$ = yes
		# Lost on a full cap: the console's record, and the log.
		if = {
			limit = { local_var:sr_ms_installed = 0 }
			set_variable = { name = sr_ms_last_lost_site value = $INDEX$ }
			set_variable = { name = sr_ms_last_lost_type value = $CODE$ }
			debug_log = "TE_SR_MODULES a module of type $CODE$ for world $INDEX$ was lost on installing (the world's cap full) country=[THIS.GetCountry.GetTagName]"
		}
	}
	else = {
		sr_mod_fab_build = { KEY = $KEY$ TYPE = $TYPE$ CODE = $CODE$ INDEX = $INDEX$ }
	}
}

# ---- The queue -----------------------------------------------------------------------
# Country scope. One more module of TYPE (code CODE) for the world KEY (index
# INDEX), at the end of the queue (sr_mq_append, generated), and in the world's
# count at once, so the next click's cap sees it. The caller has checked the cap
# and a free place (sr_mod_can_queue_<type>_<key>). Removals recount
# (sr_mq_recount, generated), as the month does.
sr_mod_queue = {
	set_local_variable = { name = sr_mq_new_site value = $INDEX$ }
	set_local_variable = { name = sr_mq_new_type value = $CODE$ }
	sr_mq_append = yes
	sr_mq_count_add = { VAR = sr_mq_$TYPE$_$KEY$ }
}

# ---- Dismantling (owner, 2026-10-31) ----------------------------------------------
# Country scope, from the dismantle scripted GUI (sr_md_start_from_scope,
# generated): one level of the module of code CODE at the world KEY starts to
# come down, sr_mod_dismantle_months from now. One at a time a world; the
# month's work, the removal and the record's end are generated
# (sr_md_month_<key>, sr_md_finish_<key>, sr_md_clear_<key>): losing the world
# or its occupation ends the order, a blockade holds it.
sr_md_start = {
	set_variable = { name = sr_md_type_$KEY$ value = $CODE$ }
	set_variable = { name = sr_md_left_$KEY$ value = sr_mod_dismantle_months }
}

# Country scope. One more of a world's queued modules of a type (VAR, a
# sr_mq_<type>_<key>).
sr_mq_count_add = {
	if = {
		limit = { NOT = { has_variable = $VAR$ } }
		set_variable = { name = $VAR$ value = 0 }
	}
	change_variable = { name = $VAR$ add = 1 }
}

# ---- Shipment slots -----------------------------------------------------------------
# Country scope, with local_var:sr_ms_new_site and local_var:sr_ms_new_type set:
# the first free slot takes the module, waiting at Earth (no _arrive yet). The
# caller has checked that one is free (sr_ms_has_free_slot).
sr_ms_store_new = {
	if = {
		limit = { NOT = { has_variable = sr_ms_1_site } }
		sr_ms_fill_slot = { N = 1 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_2_site } }
		sr_ms_fill_slot = { N = 2 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_3_site } }
		sr_ms_fill_slot = { N = 3 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_4_site } }
		sr_ms_fill_slot = { N = 4 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_5_site } }
		sr_ms_fill_slot = { N = 5 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_6_site } }
		sr_ms_fill_slot = { N = 6 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_7_site } }
		sr_ms_fill_slot = { N = 7 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_8_site } }
		sr_ms_fill_slot = { N = 8 }
	}
}

sr_ms_fill_slot = {
	set_variable = { name = sr_ms_$N$_site value = local_var:sr_ms_new_site }
	set_variable = { name = sr_ms_$N$_type value = local_var:sr_ms_new_type }
	if = {
		limit = { NOT = { has_variable = sr_ms_count } }
		set_variable = { name = sr_ms_count value = 0 }
	}
	change_variable = { name = sr_ms_count add = 1 }
}

sr_ms_clear_slot = {
	if = { limit = { has_variable = sr_ms_$N$_site } remove_variable = sr_ms_$N$_site }
	if = { limit = { has_variable = sr_ms_$N$_type } remove_variable = sr_ms_$N$_type }
	if = { limit = { has_variable = sr_ms_$N$_arrive } remove_variable = sr_ms_$N$_arrive }
	if = {
		limit = { has_variable = sr_ms_count }
		change_variable = { name = sr_ms_count subtract = 1 }
		if = {
			limit = { var:sr_ms_count <= 0 }
			remove_variable = sr_ms_count
		}
	}
}

# Any scope. One more of the world's shipments on their way (VAR, a global
# sr_site_ship_<type>_<key>).
sr_ms_count_add = {
	if = {
		limit = { NOT = { has_global_variable = $VAR$ } }
		set_global_variable = { name = $VAR$ value = 0 }
	}
	change_global_variable = { name = $VAR$ add = 1 }
}

# ---- Departures and landings (the global monthly step) ----------------------------
# Global scope (no ROOT), simple and full, before the worlds' step: every
# shipment waiting at Earth leaves if its body can be reached at no premium,
# every one whose month has come lands; then every world's count of shipments
# on their way is rebuilt from the slots, so a dead country's drop out.
sr_ms_global_month = {
	sr_launch_windows_ensure_cache = yes
	every_country = {
		limit = { has_variable = sr_ms_count }
		sr_ms_step_slot = { N = 1 }
		sr_ms_step_slot = { N = 2 }
		sr_ms_step_slot = { N = 3 }
		sr_ms_step_slot = { N = 4 }
		sr_ms_step_slot = { N = 5 }
		sr_ms_step_slot = { N = 6 }
		sr_ms_step_slot = { N = 7 }
		sr_ms_step_slot = { N = 8 }
	}
	sr_ms_clear_counters = yes
	every_country = {
		limit = { has_variable = sr_ms_count }
		sr_ms_count_slot = { N = 1 }
		sr_ms_count_slot = { N = 2 }
		sr_ms_count_slot = { N = 3 }
		sr_ms_count_slot = { N = 4 }
		sr_ms_count_slot = { N = 5 }
		sr_ms_count_slot = { N = 6 }
		sr_ms_count_slot = { N = 7 }
		sr_ms_count_slot = { N = 8 }
	}
}

# Country scope. The slot is cleared before its world's install runs.
sr_ms_step_slot = {
	if = {
		limit = {
			has_variable = sr_ms_$N$_site
			has_variable = sr_ms_$N$_type
		}
		set_local_variable = { name = sr_ms_site_now value = var:sr_ms_$N$_site }
		set_local_variable = { name = sr_ms_type_now value = var:sr_ms_$N$_type }
		if = {
			limit = { has_variable = sr_ms_$N$_arrive }
			if = {
				limit = { var:sr_ms_$N$_arrive <= te_history_month_index }
				set_local_variable = { name = sr_ms_installed value = 0 }
				sr_ms_clear_slot = { N = $N$ }
				sr_ms_land_by_index = yes
				# Lost on a world released or full: the console's record, and the log.
				if = {
					limit = { local_var:sr_ms_installed = 0 }
					set_variable = { name = sr_ms_last_lost_site value = local_var:sr_ms_site_now }
					set_variable = { name = sr_ms_last_lost_type value = local_var:sr_ms_type_now }
					debug_log = "TE_SR_MODULES a module of type [THIS.Var('sr_ms_last_lost_type').GetValue|0] for world [THIS.Var('sr_ms_last_lost_site').GetValue|0] was lost on landing (the world released, or its cap full) country=[THIS.GetCountry.GetTagName]"
				}
			}
		}
		else = {
			set_local_variable = { name = sr_ms_go value = 0 }
			# Full colonies: none leaves while Earth is blockaded or the launch
			# site is occupied (slice 6 fleets design § 15); it waits at Earth.
			if = {
				limit = { NOT = { sr_launches_barred = yes } }
				sr_ms_departure_by_index = yes
			}
			if = {
				limit = { local_var:sr_ms_go = 1 }
				set_variable = { name = sr_ms_$N$_arrive value = local_var:sr_ms_arrive_now }
			}
		}
	}
}

sr_ms_count_slot = {
	if = {
		limit = {
			has_variable = sr_ms_$N$_site
			has_variable = sr_ms_$N$_type
		}
		set_local_variable = { name = sr_ms_site_now value = var:sr_ms_$N$_site }
		set_local_variable = { name = sr_ms_type_now value = var:sr_ms_$N$_type }
		sr_ms_count_by_index = yes
	}
}

# ---- Fabrication methods ---------------------------------------------------------------
# Country scope. Puts the Space Program on the method PM, unless it runs it.
sr_mod_run_method = {
	if = {
		limit = { NOT = { sr_mod_fabrication_runs = { PM = $PM$ } } }
		activate_production_method = {
			building_type = building_space_program
			production_method = $PM$
		}
	}
}

# Country scope. The AI's Module Fabrication method, for its review (design
# § 10): the dearest one its technology allows whose bill is within its
# fabrication budget (the owner, 2026-11-20; sr_mod_ai_fab_room_<method>,
# sr_colony_module_values.txt): Fusion Shipyards with Compact Fusion Reactors,
# Orbital Foundries (pm_sr_fabrication_lunar_foundries) with Space
# Colonization. Otherwise the cheapest method it has, whatever the budget, as
# before the budget: Orbital Assembly with Cislunar Infrastructure, else
# Orbital Foundries, else Fusion Shipyards (neither later technology needs
# Cislunar Infrastructure, and a holder given its colony may have any of the
# three). With none of them, nothing.
sr_mod_activate_best_fabrication = {
	if = {
		limit = {
			has_technology_researched = compact_fusion_reactors
			sr_mod_ai_fab_room_fusion_shipyards >= 0
		}
		sr_mod_run_method = { PM = pm_sr_fabrication_fusion_shipyards }
	}
	else_if = {
		limit = {
			has_technology_researched = space_colonization
			sr_mod_ai_fab_room_lunar_foundries >= 0
		}
		sr_mod_run_method = { PM = pm_sr_fabrication_lunar_foundries }
	}
	else_if = {
		limit = { has_technology_researched = cislunar_infrastructure }
		sr_mod_run_method = { PM = pm_sr_fabrication_orbital_assembly }
	}
	else_if = {
		limit = { has_technology_researched = space_colonization }
		sr_mod_run_method = { PM = pm_sr_fabrication_lunar_foundries }
	}
	else_if = {
		limit = { has_technology_researched = compact_fusion_reactors }
		sr_mod_run_method = { PM = pm_sr_fabrication_fusion_shipyards }
	}
}

sr_mod_stop_fabrication = {
	sr_mod_run_method = { PM = pm_sr_fabrication_none }
}

sr_mod_stop_storage = {
	sr_mod_run_method = { PM = pm_sr_storage_none }
}

# ---- AI (design § 10) -----------------------------------------------------------------------
# Country scope, from the entry's pulse; once a year. A holder or top-three
# space power (sr_ai_tends_colonies) with money to spare queues one module at
# each colony with nothing on the way
# (sr_mod_ai_plan_<key>: a Habitat when its people fill three quarters of its
# room, else its next specialization level its crews can run; never the
# placeholder Defences or Spaceport), and runs the dearest fabrication method
# within its budget while anything is queued and the stock has room
# (sr_mod_activate_best_fabrication); otherwise fabrication goes off. The room
# test is the owner's (2026-11-23): a module waiting for its launch window stays
# queued (sr_mq_head_ready), so with the stock at its cap (sr_mod_fab_stock_fill_cap)
# the method would bill its goods for points the month throws away, for up to a
# Mars cycle. The review runs before the month's builds, so it reads last month's
# stock; one switched off keeps its points and is switched on again at the first
# review that finds the stock below the cap. Storage stays off.
sr_mod_ai_review = {
	if = {
		limit = {
			is_ai = yes
			sr_colonies_contested_enabled = yes
			sr_has_space_program = yes
			NOT = { has_variable = sr_mod_ai_reviewed }
		}
		set_variable = { name = sr_mod_ai_reviewed value = yes days = 360 }
		if = {
			limit = {
				sr_ai_tends_colonies = yes
				in_default = no
				net_fixed_income > 0
			}
			sr_mod_ai_plan_all = yes
		}
		if = {
			limit = {
				sr_mq_any = yes
				sr_mod_fab_stock_value < sr_mod_fab_stock_fill_cap
				in_default = no
				net_fixed_income > 0
			}
			sr_mod_activate_best_fabrication = yes
		}
		else = {
			sr_mod_stop_fabrication = yes
		}
		# No storage (the owner, 2026-11-15): the AI queues only within its
		# colonies' slots, where the base stock is always enough.
		sr_mod_stop_storage = yes
	}
}

# ---- Extraplanetary Bases from colonies (design § 9) ------------------------------------
# Country scope, from the monthly country pulse for every country with In-Situ
# Resource Utilization (the building's technology) or an Extraplanetary Base
# (one captured by a country without it), and from space_race_events.110 the day
# after a base is built or gains a level: the one refresh site of the base's
# modifiers. Ground types design § 5
# (docs/superpowers/specs/2026-11-05-extraplanetary-base-ground-types-design.md):
# two layers. The grade is throughput on the country: the average grade over
# the levels, past the room at -75%, is one country modifier, sr_ebase_room_bonus
# or _penalty, so only the levels past the room pay the penalty (owner,
# 2026-10-25). The ground's mix is an output multiplier on each base building,
# sr_ebase_mix_<good>_bonus or _penalty, so it never reaches the country's other
# producers of the good. Both are stripped first, every run, from every base the
# country owns; they come back only under Simple or Full Colonies with the
# technology. Under Races Only, with the rule switched off mid-game, or without
# the technology the refresh only strips, and the record (var:sr_eb_at_<k>,
# _target_<k>, _away_<k>, var:sr_eb_over, sr_eb_manual, the move record) waits
# in its variables for the rule to come back. Where the levels work is the
# month (sr_eb_month, § 8.2): by the checkbox, for the most profit, or by hand,
# toward the player's targets, every move paced. The generated parts are
# sr_ebase_generated.txt (scripts/generators/gen_ebase.py, from the model
# ebase_ground.py). It also stores where the next level goes, built and queued
# levels counted (var:sr_ebase_next_place, and its grade in
# var:sr_ebase_next_grade, sr_eb_next_place), for the overview's
# Extraplanetary Bases cell (UI pass design § 8, ground types design § 9.3).
sr_ebase_refresh = {
	# Before anything is stripped: the self-check reads a base's mix as the last
	# refresh left it.
	sr_eb_mix_self_check = yes
	if = {
		limit = { has_modifier = sr_ebase_room_bonus }
		remove_modifier = sr_ebase_room_bonus
	}
	if = {
		limit = { has_modifier = sr_ebase_room_penalty }
		remove_modifier = sr_ebase_room_penalty
	}
	sr_eb_strip_mix = yes
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_technology_researched = in_situ_resource_utilization
		}
		set_variable = { name = sr_ebase_room value = sr_ebase_room_total }
		# A country the AI now runs (its player left): the checkbox back on. The AI
		# never switches it off, so a player's assignments would hold for good.
		if = {
			limit = {
				is_ai = yes
				has_variable = sr_eb_manual
			}
			sr_eb_checkbox_on = yes
		}
		sr_eb_count_methods = yes
		# Where the levels work (§ 8.2's steps 1 to 5).
		sr_eb_month = yes
		# The Bases view's totals and the move notice (§ 9.1, § 9.4), players only: the
		# forecast runs from this month's result while the month's locals are set. Then
		# what switching the checkbox off would take off each place (the buttons read
		# it while the checkbox is on).
		if = {
			limit = { is_ai = no }
			sr_eb_forecast = yes
			sr_eb_totals = yes
			sr_eb_move_notice = yes
			sr_eb_switch_cuts = yes
		}
		sr_eb_compute_mix = yes
		if = {
			limit = { var:sr_ebase_grade > 0 }
			set_variable = { name = sr_ebase_units value = var:sr_ebase_grade }
			add_modifier = { name = sr_ebase_room_bonus multiplier = var:sr_ebase_units }
		}
		else_if = {
			limit = { var:sr_ebase_grade < 0 }
			set_variable = {
				name = sr_ebase_units
				value = {
					value = var:sr_ebase_grade
					multiply = -1
				}
			}
			add_modifier = { name = sr_ebase_room_penalty multiplier = var:sr_ebase_units }
		}
		sr_eb_apply_mix = yes
		# The self-check reads a mix at least two days old (a modifier sum can lag
		# a day): this run's lapses before the next monthly one.
		set_variable = { name = sr_eb_mix_fresh value = yes days = 2 }
		# Where the next level would go, counting queued levels (UI pass
		# design § 8, ground types design § 9.3): the overview's
		# Extraplanetary Bases cell.
		sr_eb_next_place = yes
	}
}

# Country scope, the owner of an Extraplanetary Base just built or expanded
# (te_on_building_built, te_on_building_expanded): Base Room's refresh the next
# day (space_race_events.110), when the new level surely counts. One event a
# day however many levels finish: each refresh removes and re-adds its
# modifier, and whether a second same-day run's has_modifier sees the first's
# add is unverified, so a batch must not run it twice. The marker lapses in
# two days if the event never fires.
sr_ebase_hook_fire = {
	if = {

# ---- Resort Colony room (#1948 design § 6.3) ----------------------------------------------
# Country scope, from the monthly country pulse, for every country with Space
# Colonization (the building's technology): the one refresh site of the two
# resort room modifiers, the twin of sr_ebase_refresh. Its built levels fill its
# room best draw first (sr_resort_fill: Earth orbit's 10, each colony's own and its
# Off-World Resorts'); the average draw over all of them, past the room at -75%,
# is one country modifier, so only the levels past the room pay the penalty. Both
# are removed first, before the rule's test, so under races only neither is
# added. It also stores the draw the next level would work at, built and queued
# levels counted (var:sr_resort_next_draw, sr_resort_next_fill), for the
# overview's Resort Room cell.
sr_resort_refresh = {
	if = {
		limit = { has_modifier = sr_resort_room_bonus }
		remove_modifier = sr_resort_room_bonus
	}
	if = {
		limit = { has_modifier = sr_resort_room_penalty }
		remove_modifier = sr_resort_room_penalty
	}
	if = {
		limit = { sr_colonies_contested_enabled = yes }
		set_local_variable = { name = sr_rs_levels value = sr_resort_levels }
		set_local_variable = { name = sr_rs_left value = local_var:sr_rs_levels }
		set_local_variable = { name = sr_rs_sum value = 0 }
		sr_resort_fill = yes
		sr_resort_next_fill = yes
		set_variable = { name = sr_resort_draw value = 0 }
		if = {
			limit = { local_var:sr_rs_levels > 0 }
			set_variable = {
				name = sr_resort_draw
				value = {
					value = local_var:sr_rs_sum
					divide = local_var:sr_rs_levels
				}
			}
		}
		if = {
			limit = { var:sr_resort_draw > 0 }
			set_variable = { name = sr_resort_units value = var:sr_resort_draw }
			add_modifier = { name = sr_resort_room_bonus multiplier = var:sr_resort_units }
		}
		else_if = {
			limit = { var:sr_resort_draw < 0 }
			set_variable = {
				name = sr_resort_units
				value = {
					value = var:sr_resort_draw
					multiply = -1
				}
			}
			add_modifier = { name = sr_resort_room_penalty multiplier = var:sr_resort_units }
		}
	}
}

# ---- The Resort Colony free of its state (#1953) -----------------------------------------
# State scope, ROOT the state, from sr_resort_state_on_action on the state's monthly
# pulse: the one refresh site of the four Resorts Off-World modifiers. The hotels and
# their guests are off-world, so the Earth state a Resort Colony is booked in mustn't
# move its output (owner, 2026-11-22; the scope
# docs/superpowers/specs/2026-11-22-colony-people-and-off-world-tourism-scope.md § 2).
# The Tourism Industry keeps every state bonus. For Tourism and Fine Art, the state's
# share of the good's output multiplier is the state's read less its owner's (a state's
# read carries its country's lines: scripting_best_practices.md, "A state's
# modifier:state_* read includes the country's modifiers"): the appeal
# (tourism_output), the transport methods, monuments, the film studios, pollution,
# turmoil and obstinance. The building takes it back off with a unit pair at the
# share's size in percent, and keeps its country's lines (laws, bloc principles,
# cultural hegemony), its own (a company's) and its throughput (the draw, the
# Spaceport's +1%). The read is the state's settled sum: a building's lines never
# reach its state's read, and a line the state gains reaches the resort for at most a
# month. Whatever the game rule, since the hotels are off-world under Races Only too.
# A Resort Colony's modifiers go with the building; the state's figures are dropped
# when it has none.
sr_resort_state_refresh = {
	if = {
		limit = {
			exists = owner
			has_building = building_space_resort
		}
		# Before anything is stripped: the self-check reads last month's cancel.
		sr_resort_state_self_check = yes
		sr_resort_state_good = { GOOD = tourism }
		sr_resort_state_good = { GOOD = fine_art }
	}
	else_if = {
		limit = { has_variable = sr_resort_state_tourism }
		remove_variable = sr_resort_state_tourism
		remove_variable = sr_resort_state_tourism_units
		remove_variable = sr_resort_state_fine_art
		remove_variable = sr_resort_state_fine_art_units
	}
}

# ROOT the state, in sr_resort_state_refresh: one good's cancel ($GOOD$ tourism or
# fine_art). var:sr_resort_state_$GOOD$ is the state's share in percent, signed;
# _units its size. A share above 0 (the state lifts the good) takes the penalty at
# its size, one below 0 (the state holds it down) the bonus. Removed first: a
# same-name add stacks.
sr_resort_state_good = {
	set_variable = {
		name = sr_resort_state_$GOOD$
		value = {
			value = modifier:goods_output_$GOOD$_mult
			subtract = owner.modifier:goods_output_$GOOD$_mult
			multiply = 100
		}
	}
	set_variable = { name = sr_resort_state_$GOOD$_units value = var:sr_resort_state_$GOOD$ }
	if = {
		limit = { var:sr_resort_state_$GOOD$ < 0 }
		set_variable = {
			name = sr_resort_state_$GOOD$_units
			value = {
				value = var:sr_resort_state_$GOOD$
				multiply = -1
			}
		}
	}
	b:building_space_resort = {
		if = {
			limit = { has_modifier = sr_resort_state_$GOOD$_bonus }
			remove_modifier = sr_resort_state_$GOOD$_bonus
		}
		if = {
			limit = { has_modifier = sr_resort_state_$GOOD$_penalty }
			remove_modifier = sr_resort_state_$GOOD$_penalty
		}
		if = {
			limit = { root.var:sr_resort_state_$GOOD$ > 0 }
			add_modifier = { name = sr_resort_state_$GOOD$_penalty multiplier = root.var:sr_resort_state_$GOOD$_units }
		}
		else_if = {
			limit = { root.var:sr_resort_state_$GOOD$ < 0 }
			add_modifier = { name = sr_resort_state_$GOOD$_bonus multiplier = root.var:sr_resort_state_$GOOD$_units }
		}
	}
}

# State scope, in sr_resort_state_refresh before its strip. TE_SELF_CHECK
# resort_state (#1953): the building's read of its Tourism output multiplier, less
# the state's read, plus last month's share (which the cancel took off), is the
# building's own lines: 0, or a company's +0.2 (Axiom Space, if its prosperity
# reaches the building and not the country). OK from -0.05 to +0.25: the state's
# read carries its country's lines, the building's read carries its state's, and
# the cancel reaches the building at its multiplier. Armed only where none of the
# three failures can land in the band (each is 0.3 or more off):
# - last month's share was 40% or more either way, on a building that carries its
#   cancel: a cancel that never arrived, or arrived at multiplier 1, is off by it;
# - the country's own line is 30% or more either way: were it missing from the
#   state's read, the share would be the state's lines less the country's, the
#   resort would take the country's lines twice, and the reading would be off by
#   the country's line;
# - the state's share has moved 5 points or less since: were the state's lines
#   missing from the building's read, the reading would be off by the share now,
#   still 35% or more.
# Otherwise it waits a month. The cancel is a month old by then (a modifier sum
# can lag a day). Once a game. It can't see whether the engine sums before it
# clamps at -100%: the play-test list's polluted state does.
sr_resort_state_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_resort_state_checked }
			has_variable = sr_resort_state_tourism_units
			var:sr_resort_state_tourism_units >= 40
			b:building_space_resort = {
				OR = {
					has_modifier = sr_resort_state_tourism_bonus
					has_modifier = sr_resort_state_tourism_penalty
				}
			}
			owner = {
				OR = {
					modifier:goods_output_tourism_mult >= 0.3
					modifier:goods_output_tourism_mult <= -0.3
				}
			}
		}
		# The state's share now less last month's, in points.
		set_local_variable = {
			name = sr_rsc_drift
			value = {
				value = modifier:goods_output_tourism_mult
				subtract = owner.modifier:goods_output_tourism_mult
				multiply = 100
				subtract = var:sr_resort_state_tourism
			}
		}
		if = {
			limit = {
				local_var:sr_rsc_drift <= 5
				local_var:sr_rsc_drift >= -5
			}
			set_global_variable = { name = sr_resort_state_checked value = yes }
			set_variable = { name = sr_rsc_check_state value = modifier:goods_output_tourism_mult }
			set_variable = { name = sr_rsc_check_country value = owner.modifier:goods_output_tourism_mult }
			b:building_space_resort = {
				root = {
					set_variable = { name = sr_rsc_check_building value = prev.modifier:goods_output_tourism_mult }
				}
			}
			set_variable = {
				name = sr_rsc_check_own
				value = {
					value = var:sr_rsc_check_building
					subtract = var:sr_rsc_check_state
					add = {
						value = var:sr_resort_state_tourism
						divide = 100
					}
				}
			}
			if = {
				limit = {
					var:sr_rsc_check_own >= -0.05
					var:sr_rsc_check_own <= 0.25
				}
				debug_log = "TE_SELF_CHECK OK resort_state building=[THIS.Var('sr_rsc_check_building').GetValue|3] state=[THIS.Var('sr_rsc_check_state').GetValue|3] country=[THIS.Var('sr_rsc_check_country').GetValue|3] share=[THIS.Var('sr_resort_state_tourism').GetValue|1] own=[THIS.Var('sr_rsc_check_own').GetValue|3] (a Resort Colony's Tourism output multiplier is its country's and its own, the state's share taken off: the state's read carries the country's lines, the building's the state's, and the cancel reaches it) state=[THIS.GetState.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN resort_state building=[THIS.Var('sr_rsc_check_building').GetValue|3] state=[THIS.Var('sr_rsc_check_state').GetValue|3] country=[THIS.Var('sr_rsc_check_country').GetValue|3] share=[THIS.Var('sr_resort_state_tourism').GetValue|1] own=[THIS.Var('sr_rsc_check_own').GetValue|3] (own near share/100: the cancel never reached the building; near country: the state's read leaves the country's lines out; near minus share/100: the building's read leaves the state's lines out; docs/testing/solar-colonies-playtest.md) state=[THIS.GetState.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			remove_variable = sr_rsc_check_state
			remove_variable = sr_rsc_check_country
			remove_variable = sr_rsc_check_building
			remove_variable = sr_rsc_check_own
		}
	}
}

# Country scope, from sr_ebase_refresh before it strips anything. TE_SELF_CHECK
# ebase_mix (ground types design § 10, stage 2): each base building's
# modifier:goods_output_tech_metals_mult must read the base's Tech-Critical
# Metals mix as the last refresh gave it, var:sr_ebase_mix_tech_metals / 100,
# within 0.001 (it matches to about 1e-5; a multiplier read as 1 would be off
# by 0.01 at the 1.5% arming floor): the unit modifiers on the building reach
# its goods output at their multiplier. Armed on the first country under Simple
# or Full Colonies with a base and a mix of at least 1.5% either way (smaller, a
# modifier that never arrived would read 0 and still pass); the verdict a month
# later, once the last refresh is two days old (sr_eb_mix_fresh, a modifier sum
# can lag a day) and no level is waiting for its refresh (sr_ebase_hook_pending:
# space_race_events.110 removes it after its refresh). Every base building is
# read: OK if all match, BROKEN if none does, and re-armed for a month later if
# some do and some don't (a building new to us, built or captured with its old
# owner's mix, that no refresh has reached). Re-armed too if it no longer
# qualifies by then. Once a game. Tech-Critical Metals, because nothing else in
# the mod or vanilla touches their output (Vale's and the unique mines'
# multipliers are on Structural and Conductive Metals), so no state needs
# skipping.
sr_eb_mix_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_eb_mix_checked }
			sr_colonies_contested_enabled = yes
			has_technology_researched = in_situ_resource_utilization
			has_variable = sr_ebase_mix_tech_metals
			OR = {
				var:sr_ebase_mix_tech_metals >= 1.5
				var:sr_ebase_mix_tech_metals <= -1.5
			}
			any_scope_building = {
				is_building_type = building_space_mine
				level >= 1
			}
		}
		if = {
			limit = {
				has_variable = sr_eb_mix_check_armed
				te_history_month_index > var:sr_eb_mix_check_armed
				NOT = { has_variable = sr_eb_mix_fresh }
				NOT = { has_variable = sr_ebase_hook_pending }
			}
			set_variable = {
				name = sr_eb_mix_check_want
				value = {
					value = var:sr_ebase_mix_tech_metals
					divide = 100
				}
			}
			set_local_variable = { name = sr_ebk_want value = var:sr_eb_mix_check_want }
			set_local_variable = { name = sr_ebk_read value = 0 }
			set_local_variable = { name = sr_ebk_match value = 0 }
			set_local_variable = { name = sr_ebk_miss value = 0 }
			# Each base building: matched or missed; `read` is a missed one's, or with none a matched one's.
			every_scope_building = {
				limit = {
					is_building_type = building_space_mine
					level >= 1
				}
				set_local_variable = {
					name = sr_ebk_gap
					value = {
						value = modifier:goods_output_tech_metals_mult
						subtract = local_var:sr_ebk_want
					}
				}
				if = {
					limit = {
						local_var:sr_ebk_gap <= 0.001
						local_var:sr_ebk_gap >= -0.001
					}
					change_local_variable = { name = sr_ebk_match add = 1 }
					if = {
						limit = { local_var:sr_ebk_miss = 0 }
						set_local_variable = { name = sr_ebk_read value = modifier:goods_output_tech_metals_mult }
					}
				}
				else = {
					change_local_variable = { name = sr_ebk_miss add = 1 }
					set_local_variable = { name = sr_ebk_read value = modifier:goods_output_tech_metals_mult }
				}
			}
			set_variable = { name = sr_eb_mix_check_read value = local_var:sr_ebk_read }
			set_variable = { name = sr_eb_mix_check_match value = local_var:sr_ebk_match }
			set_variable = { name = sr_eb_mix_check_miss value = local_var:sr_ebk_miss }
			if = {
				limit = { local_var:sr_ebk_miss = 0 }
				set_global_variable = { name = sr_eb_mix_checked value = yes }
				debug_log = "TE_SELF_CHECK OK ebase_mix country=[THIS.GetCountry.GetTagName] read=[THIS.Var('sr_eb_mix_check_read').GetValue|3] want=[THIS.Var('sr_eb_mix_check_want').GetValue|3] buildings=[THIS.Var('sr_eb_mix_check_match').GetValue|0] (every Extraplanetary Base building's Tech-Critical Metals output multiplier is the base's ground mix: the unit modifiers reach the building at their multiplier) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else_if = {
				limit = { local_var:sr_ebk_match = 0 }
				set_global_variable = { name = sr_eb_mix_checked value = yes }
				debug_log = "TE_SELF_CHECK BROKEN ebase_mix country=[THIS.GetCountry.GetTagName] read=[THIS.Var('sr_eb_mix_check_read').GetValue|3] want=[THIS.Var('sr_eb_mix_check_want').GetValue|3] buildings=[THIS.Var('sr_eb_mix_check_miss').GetValue|0] (no Extraplanetary Base building's Tech-Critical Metals output multiplier is the base's ground mix: read 0 means sr_ebase_mix_tech_metals_bonus or _penalty never reached the buildings; another figure, that the multiplier or another line is off; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				# Some match and some don't: a building no refresh has reached yet. A month later.
				set_variable = { name = sr_eb_mix_check_armed value = te_history_month_index }
			}
		}
		else_if = {
			limit = { NOT = { has_variable = sr_eb_mix_check_armed } }
			set_variable = { name = sr_eb_mix_check_armed value = te_history_month_index }
		}
	}
	else_if = {
		limit = { has_variable = sr_eb_mix_check_armed }
		remove_variable = sr_eb_mix_check_armed
	}
}

			NOT = { has_global_variable = sr_habitat_marker_checked }
			sr_has_space_colony = yes
			OR = {
				has_technology_researched = nuclear_thermal_propulsion
				has_technology_researched = orbital_manufacturing
				has_technology_researched = advanced_nanofabrication
				has_technology_researched = space_colonization
				has_technology_researched = molecular_assemblers
			}
		}
		if = {
			limit = {
				has_variable = sr_habitat_marker_armed
				te_history_month_index > var:sr_habitat_marker_armed
			}
			set_global_variable = { name = sr_habitat_marker_checked value = yes }
			if = {
				limit = {
					sr_habitat_room_modifier >= 0.09
					sr_habitat_marker_gap >= -0.01
				}
				debug_log = "TE_SELF_CHECK OK habitat_marker country=[THIS.GetCountry.GetTagName] modifier=[THIS.ScriptValue('sr_habitat_room_modifier')|2] seen=[THIS.ScriptValue('sr_site_hab_share_seen')|2] (a slot technology's Habitat Room reaches the country and its colony's Habitat share) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN habitat_marker country=[THIS.GetCountry.GetTagName] modifier=[THIS.ScriptValue('sr_habitat_room_modifier')|2] seen=[THIS.ScriptValue('sr_site_hab_share_seen')|2] (modifier 0: the technology's country_sr_habitat_room_add does not reach the country; seen below it: sr_site_hab_share_now_<key>'s read through the holder's scope fails; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		else_if = {
			limit = { NOT = { has_variable = sr_habitat_marker_armed } }
			set_variable = { name = sr_habitat_marker_armed value = te_history_month_index }
		}
	}
}
