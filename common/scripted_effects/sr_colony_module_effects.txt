# ============================================================================
# SOLAR SYSTEM COLONIZATION — colony modules: the queue, fabrication, shipments
# ============================================================================
# Design: docs/superpowers/specs/2026-10-25-solar-colonization-modules-and-bases-design.md
# (slice 4 of the scope, §§ 5-10). Simple and full colonies only.
#
# A world's module levels are on its global record (sr_site_lvl_<a|b>_<key>,
# sr_site_<hab|def|port>_<key>), so a cession or an heir takes them along. The
# holder queues modules in one national queue, first in first out (country
# variables sr_mq_<n>_site and sr_mq_<n>_type, places 1 to 20, with a count per
# world and type, sr_mq_<type>_<key>, for the caps and the panel); the Space
# Program's Module Fabrication method gives points a month
# (country_sr_module_fabrication_add), kept in sr_fab_stock; each 100 builds
# the queue's first entry. A built module fills one of eight shipment slots
# (sr_ms_<n>_site, _type, _arrive; sr_ms_count), waits for its body's window
# and lands after the transit its drive buys, from the global monthly step.
# Every world's shipments on their way are counted in sr_site_ship_<type>_<key>
# (global), rebuilt monthly from the slots. The per-world parts are generated
# (sr_sites_generated.txt, scripts/generators/gen_solar_sites.py); this file
# holds the rest. Module types: a / b the specializations (shipment code 1 / 2),
# hab 3, def 4, port 5.
# ============================================================================

# ---- The month's fabrication (the entry's pulse) ----------------------------------
# Country scope. The queue's counts rebuilt; the month's points into the stock,
# kept up to its cap so a stall (nothing queued, every slot taken) loses nothing
# short of it; then up to two modules from the head of the queue, each into a
# free shipment slot. Under Full Colonies an occupied launch site fabricates
# nothing (slice 6 fleets design § 5.1: the Space Program's method still gives
# its country points from a state it doesn't control); a blockade in orbit
# doesn't stop fabrication, and the modules wait at Earth.
sr_mod_month = {
	if = {
		limit = { sr_colonies_contested_enabled = yes }
		sr_mq_recount = yes
		# A level being dismantled at each world, a month on (owner, 2026-10-31).
		sr_md_month = yes
		sr_mod_fabrication_self_check = yes
		sr_marker_self_check = yes
		if = {
			limit = {
				sr_mod_fab_rate > 0
				NOT = { sr_launch_site_lost = yes }
			}
			set_variable = {
				name = sr_fab_stock
				value = {
					value = sr_mod_fab_stock_value
					add = sr_mod_fab_rate
					max = sr_mod_fab_stock_cap
				}
			}
		}
		set_local_variable = { name = sr_fab_built value = 0 }
		while = {
			limit = {
				local_var:sr_fab_built < sr_mod_fab_per_month
				NOT = { sr_launch_site_lost = yes }
				sr_mod_fab_stock_value >= sr_mq_head_cost
				sr_ms_has_free_slot = yes
				has_variable = sr_mq_1_site
				has_variable = sr_mq_1_type
			}
			change_local_variable = { name = sr_fab_built add = 1 }
			sr_mod_fab_head = yes
		}
	}
}

# Country scope, from sr_mod_fab_head (the queue's first entry, already out of
# the queue): one module of TYPE (shipment code CODE) for the world KEY (index
# INDEX) costs its points (local_var:sr_mq_head_cost_now, the head's price
# before it left: past the world's slots it costs more, colony management
# design § 5) and fills a shipment slot; the world counts it on its way at
# once, so the queue's caps see it the same month.
sr_mod_fab_build = {
	change_variable = { name = sr_fab_stock subtract = local_var:sr_mq_head_cost_now }
	sr_ms_count_add = { VAR = sr_site_ship_$TYPE$_$KEY$ }
	set_local_variable = { name = sr_ms_new_site value = $INDEX$ }
	set_local_variable = { name = sr_ms_new_type value = $CODE$ }
	sr_ms_store_new = yes
}

# ---- The queue -----------------------------------------------------------------------
# Country scope. One more module of TYPE (code CODE) for the world KEY (index
# INDEX), at the end of the queue (sr_mq_append, generated), and in the world's
# count at once, so the next click's cap sees it. The caller has checked the cap
# and a free place (sr_mod_can_queue_<type>_<key>). Removals recount
# (sr_mq_recount, generated), as the month does.
sr_mod_queue = {
	set_local_variable = { name = sr_mq_new_site value = $INDEX$ }
	set_local_variable = { name = sr_mq_new_type value = $CODE$ }
	sr_mq_append = yes
	sr_mq_count_add = { VAR = sr_mq_$TYPE$_$KEY$ }
}

# ---- Dismantling (owner, 2026-10-31) ----------------------------------------------
# Country scope, from the dismantle scripted GUI (sr_md_start_from_scope,
# generated): one level of the module of code CODE at the world KEY starts to
# come down, sr_mod_dismantle_months from now. One at a time a world; the
# month's work, the removal and the record's end are generated
# (sr_md_month_<key>, sr_md_finish_<key>, sr_md_clear_<key>): losing the world
# or its occupation ends the order, a blockade holds it.
sr_md_start = {
	set_variable = { name = sr_md_type_$KEY$ value = $CODE$ }
	set_variable = { name = sr_md_left_$KEY$ value = sr_mod_dismantle_months }
}

# Country scope. One more of a world's queued modules of a type (VAR, a
# sr_mq_<type>_<key>).
sr_mq_count_add = {
	if = {
		limit = { NOT = { has_variable = $VAR$ } }
		set_variable = { name = $VAR$ value = 0 }
	}
	change_variable = { name = $VAR$ add = 1 }
}

# ---- Shipment slots -----------------------------------------------------------------
# Country scope, with local_var:sr_ms_new_site and local_var:sr_ms_new_type set:
# the first free slot takes the module, waiting at Earth (no _arrive yet). The
# caller has checked that one is free (sr_ms_has_free_slot).
sr_ms_store_new = {
	if = {
		limit = { NOT = { has_variable = sr_ms_1_site } }
		sr_ms_fill_slot = { N = 1 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_2_site } }
		sr_ms_fill_slot = { N = 2 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_3_site } }
		sr_ms_fill_slot = { N = 3 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_4_site } }
		sr_ms_fill_slot = { N = 4 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_5_site } }
		sr_ms_fill_slot = { N = 5 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_6_site } }
		sr_ms_fill_slot = { N = 6 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_7_site } }
		sr_ms_fill_slot = { N = 7 }
	}
	else_if = {
		limit = { NOT = { has_variable = sr_ms_8_site } }
		sr_ms_fill_slot = { N = 8 }
	}
}

sr_ms_fill_slot = {
	set_variable = { name = sr_ms_$N$_site value = local_var:sr_ms_new_site }
	set_variable = { name = sr_ms_$N$_type value = local_var:sr_ms_new_type }
	if = {
		limit = { NOT = { has_variable = sr_ms_count } }
		set_variable = { name = sr_ms_count value = 0 }
	}
	change_variable = { name = sr_ms_count add = 1 }
}

sr_ms_clear_slot = {
	if = { limit = { has_variable = sr_ms_$N$_site } remove_variable = sr_ms_$N$_site }
	if = { limit = { has_variable = sr_ms_$N$_type } remove_variable = sr_ms_$N$_type }
	if = { limit = { has_variable = sr_ms_$N$_arrive } remove_variable = sr_ms_$N$_arrive }
	if = {
		limit = { has_variable = sr_ms_count }
		change_variable = { name = sr_ms_count subtract = 1 }
		if = {
			limit = { var:sr_ms_count <= 0 }
			remove_variable = sr_ms_count
		}
	}
}

# Any scope. One more of the world's shipments on their way (VAR, a global
# sr_site_ship_<type>_<key>).
sr_ms_count_add = {
	if = {
		limit = { NOT = { has_global_variable = $VAR$ } }
		set_global_variable = { name = $VAR$ value = 0 }
	}
	change_global_variable = { name = $VAR$ add = 1 }
}

# ---- Departures and landings (the global monthly step) ----------------------------
# Global scope (no ROOT), simple and full, before the worlds' step: every
# shipment waiting at Earth leaves if its body can be reached at no premium,
# every one whose month has come lands; then every world's count of shipments
# on their way is rebuilt from the slots, so a dead country's drop out.
sr_ms_global_month = {
	sr_launch_windows_ensure_cache = yes
	every_country = {
		limit = { has_variable = sr_ms_count }
		sr_ms_step_slot = { N = 1 }
		sr_ms_step_slot = { N = 2 }
		sr_ms_step_slot = { N = 3 }
		sr_ms_step_slot = { N = 4 }
		sr_ms_step_slot = { N = 5 }
		sr_ms_step_slot = { N = 6 }
		sr_ms_step_slot = { N = 7 }
		sr_ms_step_slot = { N = 8 }
	}
	sr_ms_clear_counters = yes
	every_country = {
		limit = { has_variable = sr_ms_count }
		sr_ms_count_slot = { N = 1 }
		sr_ms_count_slot = { N = 2 }
		sr_ms_count_slot = { N = 3 }
		sr_ms_count_slot = { N = 4 }
		sr_ms_count_slot = { N = 5 }
		sr_ms_count_slot = { N = 6 }
		sr_ms_count_slot = { N = 7 }
		sr_ms_count_slot = { N = 8 }
	}
}

# Country scope. The slot is cleared before its world's install runs.
sr_ms_step_slot = {
	if = {
		limit = {
			has_variable = sr_ms_$N$_site
			has_variable = sr_ms_$N$_type
		}
		set_local_variable = { name = sr_ms_site_now value = var:sr_ms_$N$_site }
		set_local_variable = { name = sr_ms_type_now value = var:sr_ms_$N$_type }
		if = {
			limit = { has_variable = sr_ms_$N$_arrive }
			if = {
				limit = { var:sr_ms_$N$_arrive <= te_history_month_index }
				set_local_variable = { name = sr_ms_installed value = 0 }
				sr_ms_clear_slot = { N = $N$ }
				sr_ms_land_by_index = yes
				# Lost on a world released or full: the console's record, and the log.
				if = {
					limit = { local_var:sr_ms_installed = 0 }
					set_variable = { name = sr_ms_last_lost_site value = local_var:sr_ms_site_now }
					set_variable = { name = sr_ms_last_lost_type value = local_var:sr_ms_type_now }
					debug_log = "TE_SR_MODULES a module of type [THIS.Var('sr_ms_last_lost_type').GetValue|0] for world [THIS.Var('sr_ms_last_lost_site').GetValue|0] was lost on landing (the world released, or its cap full) country=[THIS.GetCountry.GetTagName]"
				}
			}
		}
		else = {
			set_local_variable = { name = sr_ms_go value = 0 }
			# Full colonies: none leaves while Earth is blockaded or the launch
			# site is occupied (slice 6 fleets design § 15); it waits at Earth.
			if = {
				limit = { NOT = { sr_launches_barred = yes } }
				sr_ms_departure_by_index = yes
			}
			if = {
				limit = { local_var:sr_ms_go = 1 }
				set_variable = { name = sr_ms_$N$_arrive value = local_var:sr_ms_arrive_now }
			}
		}
	}
}

sr_ms_count_slot = {
	if = {
		limit = {
			has_variable = sr_ms_$N$_site
			has_variable = sr_ms_$N$_type
		}
		set_local_variable = { name = sr_ms_site_now value = var:sr_ms_$N$_site }
		set_local_variable = { name = sr_ms_type_now value = var:sr_ms_$N$_type }
		sr_ms_count_by_index = yes
	}
}

# ---- Fabrication methods ---------------------------------------------------------------
# Country scope. The best Module Fabrication method the country has, for the
# AI's review (design § 10): Fusion Shipyards with Muon-Catalyzed Fusion
# Reactors, else Orbital Foundries (pm_sr_fabrication_lunar_foundries) with
# Space Colonization, else Orbital Assembly with Cislunar Infrastructure; with
# none of them, nothing (a holder that was given its colony may have none).
sr_mod_activate_best_fabrication = {
	if = {
		limit = { has_technology_researched = compact_fusion_reactors }
		if = {
			limit = { NOT = { sr_mod_fabrication_runs = { PM = pm_sr_fabrication_fusion_shipyards } } }
			activate_production_method = {
				building_type = building_space_program
				production_method = pm_sr_fabrication_fusion_shipyards
			}
		}
	}
	else_if = {
		limit = { has_technology_researched = space_colonization }
		if = {
			limit = { NOT = { sr_mod_fabrication_runs = { PM = pm_sr_fabrication_lunar_foundries } } }
			activate_production_method = {
				building_type = building_space_program
				production_method = pm_sr_fabrication_lunar_foundries
			}
		}
	}
	else_if = {
		limit = {
			has_technology_researched = cislunar_infrastructure
			NOT = { sr_mod_fabrication_runs = { PM = pm_sr_fabrication_orbital_assembly } }
		}
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_sr_fabrication_orbital_assembly
		}
	}
}

sr_mod_stop_fabrication = {
	if = {
		limit = { NOT = { sr_mod_fabrication_runs = { PM = pm_sr_fabrication_none } } }
		activate_production_method = {
			building_type = building_space_program
			production_method = pm_sr_fabrication_none
		}
	}
}

# ---- AI (design § 10) -----------------------------------------------------------------------
# Country scope, from the entry's pulse; once a year. A holder or top-three
# space power (sr_ai_tends_colonies) with money to spare queues one module at
# each colony with nothing on the way
# (sr_mod_ai_plan_<key>: a Habitat when its people fill three quarters of its
# room, else its next specialization level its crews can run; never the
# placeholder Defences or Spaceport), and runs the best fabrication method
# while anything is queued; otherwise fabrication goes off.
sr_mod_ai_review = {
	if = {
		limit = {
			is_ai = yes
			sr_colonies_contested_enabled = yes
			sr_has_space_program = yes
			NOT = { has_variable = sr_mod_ai_reviewed }
		}
		set_variable = { name = sr_mod_ai_reviewed value = yes days = 360 }
		if = {
			limit = {
				sr_ai_tends_colonies = yes
				in_default = no
				net_fixed_income > 0
			}
			sr_mod_ai_plan_all = yes
		}
		if = {
			limit = {
				sr_mq_any = yes
				in_default = no
				net_fixed_income > 0
			}
			sr_mod_activate_best_fabrication = yes
		}
		else = {
			sr_mod_stop_fabrication = yes
		}
	}
}

# ---- Extraplanetary Bases from colonies (design § 9) ------------------------------------
# Country scope, from the monthly country pulse, for every country with
# In-Situ Resource Utilization (the building's technology): the one refresh
# site of the two room modifiers. Its built levels fill its room best grade
# first (sr_ebase_fill; the near-Earth asteroids' room needs Asteroid Mining,
# lunar colonization design § 6); the average grade over all of them, past the
# room at -75%, is one country modifier, so only the levels past the room pay
# the penalty (owner, 2026-10-25). Both are removed first; under races only
# neither is added. It also stores the grade the next level would work at,
# built and queued levels counted (var:sr_ebase_next_grade, sr_ebase_next_fill),
# for the overview's Extraplanetary Bases cell (UI pass design § 8).
sr_ebase_refresh = {
	if = {
		limit = { has_modifier = sr_ebase_room_bonus }
		remove_modifier = sr_ebase_room_bonus
	}
	if = {
		limit = { has_modifier = sr_ebase_room_penalty }
		remove_modifier = sr_ebase_room_penalty
	}
	if = {
		limit = { sr_colonies_contested_enabled = yes }
		set_variable = { name = sr_ebase_room value = sr_ebase_room_total }
		set_local_variable = { name = sr_eb_levels value = sr_ebase_levels }
		set_local_variable = { name = sr_eb_left value = local_var:sr_eb_levels }
		set_local_variable = { name = sr_eb_sum value = 0 }
		sr_ebase_fill = yes
		# Where the next level would work, counting queued levels (UI pass
		# design § 8): the overview's Extraplanetary Bases cell.
		sr_ebase_next_fill = yes
		set_variable = { name = sr_ebase_grade value = 0 }
		if = {
			limit = { local_var:sr_eb_levels > 0 }
			set_variable = {
				name = sr_ebase_grade
				value = {
					value = local_var:sr_eb_sum
					divide = local_var:sr_eb_levels
				}
			}
		}
		if = {
			limit = { var:sr_ebase_grade > 0 }
			set_variable = { name = sr_ebase_units value = var:sr_ebase_grade }
			add_modifier = { name = sr_ebase_room_bonus multiplier = var:sr_ebase_units }
		}
		else_if = {
			limit = { var:sr_ebase_grade < 0 }
			set_variable = {
				name = sr_ebase_units
				value = {
					value = var:sr_ebase_grade
					multiply = -1
				}
			}
			add_modifier = { name = sr_ebase_room_penalty multiplier = var:sr_ebase_units }
		}
	}
}

# ---- Self-checks (design § 12) -----------------------------------------------------------
# Country scope, from sr_mod_month. TE_SELF_CHECK module_fabrication: a Space
# Program that runs a fabrication method at half staff or more must give its
# country points. Armed on the first such country, the verdict a month later
# (a modifier sum can lag a day after a method changes); re-armed if it no
# longer qualifies by then. Once a game. A holder of Mare Imbrium's Mass Driver
# is skipped: its +3 points need no method, so its OK would prove nothing
# (sr_colony_imbrium_massdriver_mult is set on the country while it shows).
sr_mod_fabrication_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_mod_fab_checked }
			sr_mod_fab_qualifies = yes
			NOT = { has_variable = sr_colony_imbrium_massdriver_mult }
		}
		if = {
			limit = {
				has_variable = sr_mod_fab_check_armed
				te_history_month_index > var:sr_mod_fab_check_armed
			}
			set_global_variable = { name = sr_mod_fab_checked value = yes }
			# The method's points alone: a colony's Orbital Yards add their own.
			if = {
				limit = { sr_mod_fab_rate_method > 0 }
				debug_log = "TE_SELF_CHECK OK module_fabrication country=[THIS.GetCountry.GetTagName] points=[THIS.ScriptValue('sr_mod_fab_rate_method')|2] (a Space Program's fabrication method gives its country points) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN module_fabrication country=[THIS.GetCountry.GetTagName] points=[THIS.ScriptValue('sr_mod_fab_rate_method')|2] (a Space Program has run a fabrication method at half staff or more for a month and its country has no points: the method's country modifier does not reach the country; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		else_if = {
			limit = { NOT = { has_variable = sr_mod_fab_check_armed } }
			set_variable = { name = sr_mod_fab_check_armed value = te_history_month_index }
		}
	}
	else_if = {
		limit = { has_variable = sr_mod_fab_check_armed }
		remove_variable = sr_mod_fab_check_armed
	}
}

# Country scope, from sr_mod_month. The technology markers (colony management
# design § 11; scripting_best_practices.md § Technology markers), once a game
# each, a month after a country first qualifies, since a modifier's sum can lag:
# drive_marker, Fusion Power's country_sr_drive_fusion_bool reaches its country
# and sr_drive_level reads it; slot_marker, a slot technology's Colony Module
# Slots reach the country (at least 1, so OK can't be a false pass) and a
# colony's slots read them through its holder's scope (sr_slots_marker_gap 0).
sr_marker_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_drive_marker_checked }
			has_technology_researched = fusion_power
		}
		if = {
			limit = {
				has_variable = sr_drive_marker_armed
				te_history_month_index > var:sr_drive_marker_armed
			}
			set_global_variable = { name = sr_drive_marker_checked value = yes }
			if = {
				limit = {
					modifier:country_sr_drive_fusion_bool = yes
					sr_drive_level >= 1
				}
				debug_log = "TE_SELF_CHECK OK drive_marker country=[THIS.GetCountry.GetTagName] level=[THIS.ScriptValue('sr_drive_level')|0] (Fusion Power's marker reaches its country and sets the drive) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN drive_marker country=[THIS.GetCountry.GetTagName] level=[THIS.ScriptValue('sr_drive_level')|0] (a country with Fusion Power reads no fusion drive: modifier:country_sr_drive_fusion_bool does not read the technology's boolean; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		else_if = {
			limit = { NOT = { has_variable = sr_drive_marker_armed } }
			set_variable = { name = sr_drive_marker_armed value = te_history_month_index }
		}
	}
	if = {
		limit = {
			NOT = { has_global_variable = sr_slot_marker_checked }
			sr_has_space_colony = yes
			OR = {
				has_technology_researched = nuclear_thermal_propulsion
				has_technology_researched = orbital_manufacturing
				has_technology_researched = advanced_nanofabrication
				has_technology_researched = space_colonization
				has_technology_researched = molecular_assemblers
			}
		}
		if = {
			limit = {
				has_variable = sr_slot_marker_armed
				te_history_month_index > var:sr_slot_marker_armed
			}
			set_global_variable = { name = sr_slot_marker_checked value = yes }
			if = {
				limit = {
					sr_colony_slots_modifier >= 1
					sr_slots_marker_gap = 0
				}
				debug_log = "TE_SELF_CHECK OK slot_marker country=[THIS.GetCountry.GetTagName] modifier=[THIS.ScriptValue('sr_colony_slots_modifier')|0] seen=[THIS.ScriptValue('sr_site_slots_tech_seen')|0] (a slot technology's Colony Module Slots reach the country and its colony's slots) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN slot_marker country=[THIS.GetCountry.GetTagName] modifier=[THIS.ScriptValue('sr_colony_slots_modifier')|0] seen=[THIS.ScriptValue('sr_site_slots_tech_seen')|0] (modifier 0: the technology's country_sr_colony_slots_add does not reach the country; seen below it: sr_site_slots_<key>'s read through the holder's scope fails; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		else_if = {
			limit = { NOT = { has_variable = sr_slot_marker_armed } }
			set_variable = { name = sr_slot_marker_armed value = te_history_month_index }
		}
	}
}

# Journal entry scope, right after sr_site_show_modifier_at's add_modifier.
# TE_SELF_CHECK modifier_seen_same_effect: whether has_modifier sees a
# modifier added earlier in the same effect, which the mod's code has so far
# avoided relying on (slices 2-3 design § 15). Once a game, at the first add.
sr_mod_seen_self_check = {
	if = {
		limit = { NOT = { has_global_variable = sr_mod_seen_checked } }
		set_global_variable = { name = sr_mod_seen_checked value = yes }
		if = {
			limit = { has_modifier = $MODIFIER$ }
			debug_log = "TE_SELF_CHECK OK modifier_seen_same_effect $MODIFIER$ (has_modifier sees an add_modifier made earlier in the same effect) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN modifier_seen_same_effect $MODIFIER$ (has_modifier does not yet see an add_modifier made earlier in the same effect: keep every refresh to once per effect; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
}
