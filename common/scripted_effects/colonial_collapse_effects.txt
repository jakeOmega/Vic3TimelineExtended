# Colonial Collapse Effects
# When the decolonization era arrives, tiny non-player countries
# that lack population collapse: either reverting to decentralized
# (uncolonized) land or being absorbed by a culturally similar neighbor.
#
# Fired from colonial_collapse_on_action (on_yearly_pulse_country) — each
# country runs the limit block against itself on its own yearly pulse and
# self-collapses if it matches the tiny-and-poor + neutral-context criteria.
#
# Gates: the decolonization game rule, and the decolonization era — some
# country in the world has researched the decolonization technology
# (decol_era_begun; the collapsing country need not have it). The world check
# is last in the limit, so it only runs for the few countries that pass the
# cheap filters above it.
#
# Until 2026-09 the limit also held `any_civil_war = { always = no }`, added in
# db166830 ("colonial collapse checks tightened"), two days after the effect
# first ran, ungated, from 1836. An any_* iterator whose body is `always = no`
# is false for every country, so the effect never ran again. The owner
# confirmed on 2026-09-26 that collapse should be live, the decentralize branch
# included; the civil-war test now uses vanilla's idiom (00_autocracy.txt).
#
# Safety, since vanilla itself never sets a country decentralized:
# - a country freed in the last 20 years (recently_decolonized, set by every
#   release path) is exempt, so a colony the Colonial Empire entry just freed
#   cannot collapse the next year;
# - a country with subjects never collapses;
# - the absorber is at peace and not a subject;
# - the decentralize branch runs only for a country with nothing that ties it
#   to other countries or that a decentralized country cannot hold: no power
#   bloc, no treaty in force, no company, no warheads. Otherwise it waits and is
#   tested again next year;
# - the notice is posted before the annexation, while the collapsing country
#   still exists for its name.

colonial_collapse_effect = {
	if = {
		limit = {
			has_game_rule = decolonization_enabled

			# Must be alive and not player-controlled
			is_country_alive = yes
			is_player = no

			# Not already decentralized
			NOT = { is_country_type = decentralized }

			# Not a subject (puppet states are managed by overlords), and holding
			# no subjects of its own
			is_subject = no
			NOT = { any_subject_or_below = { always = yes } }

			# Freed in the last 20 years: exempt (owner ruling, 2026-09-26)
			NOT = { has_variable = recently_decolonized }

			# No civil war about to break out
			NOT = {
				any_civil_war = {
					civil_war_progress >= 0.75
				}
			}

			# Population threshold: very small, poor countries collapse
			OR = {
				AND = {
					total_population < 100000
					average_sol < 6
				}
				AND = {
					total_population < 200000
					average_sol < 5
				}
				AND = {
					total_population < 300000
					average_sol < 4
				}
				AND = {
					total_population < 500000
					average_sol < 3
				}
				AND = {
					total_population < 1000000
					average_sol < 2
				}
			}

			# Safety: don't collapse countries currently in a diplomatic play or at war
			NOR = {
				is_diplomatic_play_committed_participant = yes
				is_diplomatic_play_undecided_participant = yes
			}
			NOT = { is_at_war = yes }

			# The decolonization era has begun somewhere in the world
			decol_era_begun = yes
		}

		save_scope_as = collapsing_country

		# Try to find a culturally similar non-decentralized neighbor to absorb this country
		# Check if any of our states border a non-decentralized country that shares heritage
		ordered_scope_state = {
			limit = { always = yes }
			order_by = state_population
			max = 1

			save_scope_as = collapse_state

			# Look for neighboring state owned by a culturally similar country
			random_neighbouring_state = {
				limit = {
					owner = {
						is_country_alive = yes
						NOT = { is_country_type = decentralized }
						NOT = { this = scope:collapsing_country }
						# At peace, and free to annex (a subject's gains are its overlord's business)
						is_at_war = no
						is_subject = no
						any_primary_culture = {
							shares_heritage_trait_group_with_any_primary_culture = scope:collapsing_country
						}
					}
				}
				owner = {
					save_scope_as = absorbing_country
				}
			}
		}

		if = {
			limit = {
				exists = scope:absorbing_country
			}
			# Notify countries with strategic interest, first: the notice names
			# the collapsing country, which the annexation below removes.
			every_country = {
				limit = {
					is_country_alive = yes
					is_player = yes
					NOT = { is_country_type = decentralized }
					OR = {
						# Countries in the same strategic region
						any_scope_state = {
							region = scope:collapse_state.region
						}
						# Great powers always care
						country_rank >= rank_value:great_power
					}
				}
				post_notification = colonial_territory_absorbed_notice
			}

			# Culturally similar neighbor absorbs this country
			scope:absorbing_country = {
				annex = scope:collapsing_country
				# An armed country annexed: its arsenal settles to us (nuclear_custody_effects.txt).
				nd_custody_reconcile_soon = yes
			}
		}
		else_if = {
			# No culturally similar neighbor: revert to decentralized (uncolonized),
			# but only a country nothing ties to the rest of the world. Vanilla never
			# sets a country decentralized, so each of these is a case the engine may
			# not unwind cleanly; such a country waits and is tested next year.
			limit = {
				is_in_power_bloc = no
				NOT = { any_scope_treaty = { always = yes } }
				NOT = { any_company = { always = yes } }
				nd_is_armed = no
			}
			set_country_type = decentralized

			# Notify countries with strategic interest
			every_country = {
				limit = {
					is_country_alive = yes
					is_player = yes
					NOT = { is_country_type = decentralized }
					OR = {
						# Countries in the same strategic region
						any_scope_state = {
							region = scope:collapse_state.region
						}
						# Great powers always care
						country_rank >= rank_value:great_power
					}
				}
				post_notification = colonial_territory_collapsed_notice
			}
		}

		# Clear scopes for next iteration
		clear_saved_scope = absorbing_country
		clear_saved_scope = collapse_state
		clear_saved_scope = collapsing_country
	}
}
