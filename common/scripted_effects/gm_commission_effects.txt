# ============================================================================
# GRAND MONUMENTS — opportunities: scripted effects
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-12-grand-monument-opportunities-design.md,
# which reworks v2's commissions (docs/superpowers/archive/specs/2026-10-05-grand-
# monuments-v2-design.md §2). Players read "opportunity"; the script keeps v2's
# gm_com_* names, because a save carries an open commission in them.
#
# An opportunity is a moment the game has just produced (a new reign, a victory,
# a great power's rise, a technology) that an interest group wants marked in
# stone. Nothing is lost by letting one pass. One record per country, all
# country variables (a civil war's winner inherits variables, not modifiers or
# lists):
#   gm_com_offered        yes while the offer (monument_events.21) is open
#   gm_com_open           yes while seized and running
#   gm_com_source         flag: the moment (the spec's §1 table, one boon each)
#   gm_com_ig             flag: the asking IG's type (gm_ig_to_flag_on_owner)
#   gm_com_dedication     flag: one of the twelve dedication keys
#   gm_com_state          state, when the opportunity names one
#   gm_com_target         grandeur to add (5)
#   gm_com_baseline       counted grandeur on seizing (gm_com_fit_grandeur)
#   gm_com_months_left    gm_com_time_months on seizing (60, or 120 for a
#                         major power), counted down by gm_com_monthly
#   gm_com_namesake, gm_com_form, gm_com_occasion   flags: the name it asks for
#   gm_com_name_year, gm_com_name_char, gm_com_name_enemy_cap   its records
# Kept across opportunities: gm_com_last_state (the monument a level or a
# dedication last touched); the moment trackers (gm_ruler, gm_ruler_months,
# gm_regime_last, gm_com_faith_last, gm_com_seen_great_power,
# gm_com_seen_space_<m>, gm_com_seen_tech_<key>, gm_com_centenary_<n>,
# gm_founded_year, gm_com_opp_tracked); the boons (gm_boon_<source>, months
# left).
#
# Seized and completed, an opportunity pays three ways (spec §3): the monument
# that completed it gains five grandeur (monument_events.26), the
# asking IG's ledger rises (gm_ig_ledger_<ig>) and a boon runs for ten years.

# gm_com_gap_half are v2's, cleared for a save that still carries them.

	gm_remove_state_var = { VAR = gm_com_source }
	gm_remove_state_var = { VAR = gm_com_ig }
	gm_remove_state_var = { VAR = gm_com_dedication }
	gm_remove_state_var = { VAR = gm_com_state }

	gm_remove_state_var = { VAR = gm_com_baseline }

	gm_remove_state_var = { VAR = gm_com_extended }

	gm_remove_state_var = { VAR = gm_com_form }

	gm_remove_state_var = { VAR = gm_com_name_year }

	}
}

# Country scope. An opportunity closes (let pass, seized, run out or lapsed).
# Nothing follows it: no cooldown, no cost.
gm_com_close = {
	gm_com_clear = yes
}

# Country scope. Add $AMOUNT$ grandeur to the ledger of the IG whose type the
# country's variable $VAR$ names (gm_com_ig). Through the per-IG curve,
# 15 is about 2 approval.
gm_country_ledger_ig_by_flag = {
	gm_init_ledgers = yes
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = armed_forces }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = devout }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = industrialists }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = intelligentsia }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = landowners }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = petty_bourgeoisie }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = rural_folk }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = trade_unions }
}

gm_country_ledger_ig_by_flag_one = {
	if = {
		limit = {
			has_variable = $VAR$
			var:$VAR$ = flag:$IG$
		}
		change_variable = { name = gm_ig_ledger_$IG$ add = $AMOUNT$ }
	}
}

# Country scope. The strongest interest group in the government ($IN$ = yes)
# or outside it ($IN$ = no), as gm_com_ig; flag:none when there is none.
gm_com_ig_by_clout = {
	set_variable = { name = gm_com_ig value = flag:none }
	ordered_interest_group = {
		limit = { is_in_government = $IN$ }
		order_by = ig_clout
		position = 0
		check_range_bounds = no
		gm_ig_to_flag_on_owner = { VAR = gm_com_ig }
	}
}

# Country scope. The Petty Bourgeoisie, To the Nation's approving group, unless
# they are marginal; then the strongest group in government.
gm_com_ig_for_the_nation = {
	if = {
		limit = {
			ig:ig_petty_bourgeoisie ?= { is_marginal = no }
		}
		set_variable = { name = gm_com_ig value = flag:petty_bourgeoisie }
	}
	else = {
		gm_com_ig_by_clout = { IN = yes }
	}
}

# ---- Making an offer (spec §1) ------------------------------------------------------
# Country scope. The common part of every offer: what is asked for and how it
# would be named. The caller then sets gm_com_ig, any records and the state, and
# sends it (gm_com_send_offer). FORM = default leaves the form to the
# unveiling (the skin's first form).
gm_com_begin = {
	gm_com_clear = yes
	set_variable = { name = gm_com_source value = flag:$SOURCE$ }
	set_variable = { name = gm_com_dedication value = flag:$DEDICATION$ }
	set_variable = { name = gm_com_namesake value = flag:$NAMESAKE$ }
	set_variable = { name = gm_com_form value = flag:$FORM$ }
	set_variable = { name = gm_com_target value = 5 }
	set_variable = { name = gm_com_name_year value = year }
}

# Country scope. Fix the opportunity to the capital, if a monument of its
# dedication can stand there; otherwise it can be met anywhere (one Grand
# Monument per state: a capital holding another dedication could never meet it).
gm_com_fix_to_capital = {
	if = {
		limit = {
			capital ?= { gm_state_can_host_commission = yes }
		}
		capital = { save_scope_as = gm_com_capital }
		set_variable = { name = gm_com_state value = scope:gm_com_capital }
	}
}

# Country scope. Send the offer, once an asking group is known; with none (no
# interest group of the eight found), forget it.
gm_com_send_offer = {
	if = {
		limit = {
			has_variable = gm_com_ig
			NOT = { var:gm_com_ig = flag:none }
		}
		set_variable = gm_com_offered
		# The offer's watchdog: an offer still open after six months (its event
		# never fired, or never closed) is forgotten by the world pass, so it
		# cannot block every later opportunity.
		set_variable = { name = gm_com_offer_watch value = yes months = 6 }
		if = {
			limit = {
				NOT = { has_journal_entry = je_grand_monuments }
			}
			add_journal_entry = { type = je_grand_monuments }
		}
		trigger_event = { id = monument_events.21 }
	}
	else = {
		gm_com_clear = yes
	}
}

# Country scope. The scopes the offer's and the ceremony's text read: the
# asking interest group, the opportunity's state, the enemy (its capital's
# owner) and the ruler it names.
gm_com_save_scopes = {
	gm_com_save_ig_scope = { IG = armed_forces }
	gm_com_save_ig_scope = { IG = devout }
	gm_com_save_ig_scope = { IG = industrialists }
	gm_com_save_ig_scope = { IG = intelligentsia }
	gm_com_save_ig_scope = { IG = landowners }
	gm_com_save_ig_scope = { IG = petty_bourgeoisie }
	gm_com_save_ig_scope = { IG = rural_folk }
	gm_com_save_ig_scope = { IG = trade_unions }
	if = {
		limit = { has_variable = gm_com_state }
		var:gm_com_state ?= { save_scope_as = gm_com_where }
	}
	if = {
		limit = { has_variable = gm_com_name_enemy_cap }
		var:gm_com_name_enemy_cap ?= {
			owner ?= { save_scope_as = gm_com_enemy }
		}
	}
	if = {
		limit = { has_variable = gm_com_name_char }
		var:gm_com_name_char ?= { save_scope_as = gm_com_honoree }
	}
}

gm_com_save_ig_scope = {
	if = {
		limit = {
			has_variable = gm_com_ig
			var:gm_com_ig = flag:$IG$
		}
		ig:ig_$IG$ ?= { save_scope_as = gm_com_ig_scope }
	}
}

# ---- The offer's two answers (monument_events.21) -------------------------------------
# Country scope, ROOT = the country.
gm_com_accept = {
	gm_remove_state_var = { VAR = gm_com_offered }
	set_variable = gm_com_open
	set_variable = { name = gm_com_months_left value = gm_com_time_months }
	set_variable = { name = gm_com_baseline value = gm_com_fit_grandeur }
	set_variable = gm_active
	gm_init_ledgers = yes
	if = {
		limit = { is_ai = yes }
		gm_com_ai_queue = yes
	}
	trigger_event = { id = monument_events.20 }
}

	gm_com_close = yes
	trigger_event = { id = monument_events.20 }
}

# ---- Building it: the AI's queue and Raise It (v2 §2.6; usability §1) -------------------
# Country scope. Where an opportunity's missing levels go (sites spec
# 2026-10-18 §3.1: always onto a construction site, which the dedication
# makes the monument): the opportunity's state, if it has one; a state where
# a site already stands (its ceremony then offers the opportunity's
# dedication first); the capital; the most populous incorporated state. Each
# only where a site may be built or expanded (gm_state_can_build_site). None
# of these, no scope.
gm_com_pick_build_state = {
	clear_saved_scope = gm_com_build_state
	if = {
		limit = { has_variable = gm_com_state }
		var:gm_com_state ?= {
			if = {
				limit = { gm_state_can_build_site = yes }
				save_scope_as = gm_com_build_state
			}
		}
	}
	else_if = {
		limit = {
			any_scope_state = {
				gm_state_has_site = yes
				gm_state_can_build_site = yes
			}
		}
		ordered_scope_state = {
			limit = {
				gm_state_has_site = yes
				gm_state_can_build_site = yes
			}
			order_by = gm_state_site_level
			position = 0
			check_range_bounds = no
			save_scope_as = gm_com_build_state
		}
	}
	else_if = {
		limit = {
			capital ?= { gm_state_can_build_site = yes }
		}
		capital = { save_scope_as = gm_com_build_state }
	}
	else_if = {
		limit = {
			any_scope_state = {
				is_incorporated = yes
				gm_state_can_build_site = yes
			}
		}
		ordered_scope_state = {
			limit = {
				is_incorporated = yes
				gm_state_can_build_site = yes
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			save_scope_as = gm_com_build_state
		}
	}
}

# Country scope, with scope:gm_com_build_state and gm_com_raise_levels set:
# queue that many levels there as government construction
# (start_building_construction: the country pays for them, unlike
# create_building). One level a call, in a `while` with a limit: an <int>
# argument does not take a variable.
gm_com_queue_levels = {
	set_variable = { name = gm_com_queued value = 0 }
	while = {
		limit = { var:gm_com_queued < var:gm_com_raise_levels }
		scope:gm_com_build_state = {
			start_building_construction = building_grand_monument_site
		}
		change_variable = { name = gm_com_queued add = 1 }
	}
	remove_variable = gm_com_queued
}

# Country scope, AI only, on seizing: the levels still missing.
gm_com_ai_queue = {
	gm_com_pick_build_state = yes
	if = {
		limit = { exists = scope:gm_com_build_state }
		set_variable = { name = gm_com_raise_levels value = gm_com_levels_to_go }
		gm_com_queue_levels = yes
		remove_variable = gm_com_raise_levels
	}
}

# Country scope, from monument_events.27 (Raise It on the panel, or Seize It
# and Start Building on the offer): the levels still missing
# (gm_com_levels_to_go), where gm_com_pick_build_state puts them. An event,
# not the button's effect: the button's tooltip walks its effect without
# writing, and this sets a variable and reads it back.
gm_com_raise = {
	if = {
		limit = {
			has_variable = gm_com_open
			gm_com_levels_to_go > 0
		}
		gm_com_pick_build_state = yes
		if = {
			limit = { exists = scope:gm_com_build_state }
			set_variable = { name = gm_com_raise_levels value = gm_com_levels_to_go }
			gm_com_queue_levels = yes
			remove_variable = gm_com_raise_levels
		}
	}
}

# Country scope, from gm_country_refresh: where Raise It would build
# (gm_com_build_where), for the panel's Where line and the button's tooltip.
# Written, never read back here (the refresh runs inside tooltips).
gm_com_update_build_where = {
	if = {
		limit = { has_variable = gm_com_open }
		gm_com_pick_build_state = yes
		if = {
			limit = { exists = scope:gm_com_build_state }
			set_variable = { name = gm_com_build_where value = scope:gm_com_build_state }
		}
		else = {
			gm_remove_state_var = { VAR = gm_com_build_where }
		}
	}
	else = {
		gm_remove_state_var = { VAR = gm_com_build_where }
	}
}

# ---- The open opportunity (spec §2) ---------------------------------------------------
# Country scope. From gm_country_refresh, before the totals so its rewards
# count in the same refresh (a level finished, a dedication, the month), and
# from gm_country_monthly before the clock: met once the progress reaches the
# target. A counted monument that drops out (ceded, contested, demolished)
# lowers the baseline with it, so it stops counting instead of cancelling
# grandeur added since.
# Compared directly, never through a variable written here: the refresh also
# runs inside the policy buttons' tooltips, which walk the effect without
# writing anything, so a variable set and read in one pass reads as unset
# there (the old gm_com_drop logged ~150 errors a burst, 2026-10-07).
gm_com_check_progress = {
	if = {
		limit = { has_variable = gm_com_open }
		if = {
			limit = { var:gm_com_baseline > gm_com_fit_grandeur }
			set_variable = { name = gm_com_baseline value = gm_com_fit_grandeur }
		}
		if = {
			limit = { var:gm_com_target <= gm_com_progress }
			gm_com_fulfil = yes
		}
	}
}

# Country scope. From gm_country_monthly, after gm_com_check_progress and
# before the refresh: the clock, then a lapse before a run-out. Neither costs
# anything.
gm_com_monthly = {
	if = {
		limit = { has_variable = gm_com_open }
		change_variable = { name = gm_com_months_left add = -1 }
		if = {
			limit = { gm_com_has_lapsed = yes }
			gm_com_lapse = yes
		}
		else_if = {
			limit = { var:gm_com_months_left <= 0 }
			gm_com_miss = yes
		}
		# A year left and not yet met: a reminder (usability §5).
		else_if = {
			limit = {
				is_ai = no
				var:gm_com_months_left = 12
			}
			post_notification = gm_com_reminder_notice
		}
	}
	# The AI's next levels: one queue (on seizing, gm_com_ai_queue) takes at
	# most 100, which past twenty times 1836's wealth is short of the target,
	# so once its last site is dedicated and counted it queues what is still
	# missing. A player presses Raise It again.
	if = {
		limit = {
			has_variable = gm_com_open
			is_ai = yes
			gm_com_levels_to_go > 0
			NOT = { any_scope_state = { has_building = building_grand_monument_site } }
		}
		gm_com_ai_queue = yes
	}
}

# Country scope. Seized and completed (spec §3). The monument whose level or
# dedication completed it (gm_com_last_state, if it still counts), else the
# tallest counting monument, takes the opportunity's name -- unless it was
# unveiled while this opportunity was open, when the player already chose its
# name with the opportunity's offered first (gm_unveiled_for_com), or the
# opportunity asks only for the city's name and the monument already has a
# name: that would erase a chosen name for everyone's default -- and takes the
# gift in monument_events.26 (gm_gift; never here: this runs inside the
# refresh, which tooltips walk). Then the asking group's approval and the boon,
# and for a player monument_events.28, which names the monument and the boon
# from gm_com_seized_state and gm_com_seized_source (the record is gone by
# then; nothing here reads them back). monument_events.26 fires last, once the
# record is closed: a trigger_event with no delay runs the event at once, inside
# this effect, and the gift's refresh (monument_events.20) would find the
# opportunity still open and seize it again, without end (two stack-overflow
# crashes, 2026-10-17).
gm_com_fulfil = {
	clear_saved_scope = gm_com_named_state
	# Its special (signatures spec §5.2), if its moment has one: one monument
	# takes it, the one the gift lands on, or the country holds it.
	save_scope_as = gm_sig_granter
	gm_sig_special_for_source = yes
	gm_remove_state_var = { VAR = gm_com_seized_special }
	gm_remove_state_var = { VAR = gm_com_seized_special_held }
	if = {
		limit = { has_variable = gm_sig_special }
		set_variable = { name = gm_com_seized_special value = var:gm_sig_special }
	}
	if = {
		limit = {
			has_variable = gm_com_last_state
			var:gm_com_last_state ?= { gm_state_com_counts = yes }
		}
		var:gm_com_last_state = { save_scope_as = gm_com_named_state }
	}
	else = {
		ordered_scope_state = {
			limit = { gm_state_com_counts = yes }
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_com_named_state
		}
	}
	if = {
		limit = { exists = scope:gm_com_named_state }
		scope:gm_com_named_state = {
			if = {
				limit = {
					NOT = { has_variable = gm_unveiled_for_com }
					# A landmark keeps its historical name; the opportunity is
					# still completed (phase 4 §3.2).
					gm_state_has_historical_name = no
					# A historical opportunity's monument takes the landmark's
					# name instead (signatures spec §6.2).
					NOT = { owner = { has_variable = gm_com_landmark } }
					OR = {
						NOT = { has_variable = gm_namesake }
						owner = {
							NOT = { var:gm_com_namesake = flag:city }
						}
					}
				}
				gm_state_take_commission_name = yes
			}
			set_variable = gm_gift
			# A historical opportunity: the monument becomes its landmark, in
			# monument_events.26 with the gift (signatures spec §6.2).
			if = {
				limit = { scope:gm_sig_granter = { has_variable = gm_com_landmark } }
				set_variable = { name = gm_hist_become value = scope:gm_sig_granter.var:gm_com_landmark }
			}
			# The special lands with the gift (monument_events.26); a monument that
			# already carries a signature keeps it, and the country holds this one.
			if = {
				limit = { scope:gm_sig_granter = { has_variable = gm_sig_special } }
				if = {
					limit = { NOT = { has_variable = gm_signature } }
					set_variable = { name = gm_sig_grant value = scope:gm_sig_granter.var:gm_sig_special }
				}
				else = {
					scope:gm_sig_granter = { gm_sig_hold_special = yes }
				}
			}
		}
		# What monument_events.28 names after the record closes (usability §5).
		set_variable = { name = gm_com_seized_state value = scope:gm_com_named_state }
	}
	else = {
		gm_remove_state_var = { VAR = gm_com_seized_state }
		# A monument beside another completed it (ageing spec 2026-10-16 §5):
		# the gift on the largest that counts; it keeps its name, and .28 names
		# none.
		clear_saved_scope = gm_com_gift_mon
		if = {
			limit = { has_variable_list = gm_monuments_held }
			ordered_in_list = {
				variable = gm_monuments_held
				limit = { gm_mon_com_counts = yes }
				order_by = gm_mon_grandeur
				position = 0
				check_range_bounds = no
				save_scope_as = gm_com_gift_mon
			}
		}
		if = {
			limit = { exists = scope:gm_com_gift_mon }
			scope:gm_com_gift_mon = { gm_mon_add_age = { AGE = gm_com_gift } }
			# A historical opportunity's landmark (signatures spec §6.2).
			if = {
				limit = { has_variable = gm_com_landmark }
				scope:gm_com_gift_mon = { gm_mon_hist_take = yes }
			}
			# Its special beside the gift, variables only (signatures spec §5.2).
			if = {
				limit = { has_variable = gm_sig_special }
				if = {
					limit = { scope:gm_com_gift_mon = { gm_mon_takes_special = yes } }
					scope:gm_com_gift_mon = { gm_mon_sig_take_special = yes }
				}
				else = {
					gm_sig_hold_special = yes
				}
			}
		}
		# No monument could take it: held for the next of its dedication.
		else_if = {
			limit = { has_variable = gm_sig_special }
			gm_sig_hold_special = yes
		}
	}
	if = {
		limit = { has_variable = gm_com_source }
		set_variable = { name = gm_com_seized_source value = var:gm_com_source }
	}
	# The landmark a historical one made, for monument_events.28.
	gm_remove_state_var = { VAR = gm_com_seized_landmark }
	if = {
		limit = { has_variable = gm_com_landmark }
		set_variable = { name = gm_com_seized_landmark value = var:gm_com_landmark }
	}
	gm_remove_state_var = { VAR = gm_sig_special }
	gm_country_ledger_ig_by_flag = { VAR = gm_com_ig AMOUNT = 15 }
	gm_com_grant_boon = yes
	set_variable = gm_active
	if = {
		limit = { is_ai = no }
		trigger_event = { id = monument_events.28 popup = yes }
	}
	gm_com_close = yes
	scope:gm_com_named_state ?= { trigger_event = { id = monument_events.26 } }
}

gm_state_restore_dedication_one = {

		limit = {
			has_variable = gm_restore_pm
			var:gm_restore_pm = flag:$KEY$
		}

			limit = { gm_state_has_pm = { PM = pm_monument_$KEY$ } }

				limit = {

				}
				remove_variable = gm_created_check
			}
			remove_variable = gm_restore_pm
		}

			if = {
				limit = {
					has_variable = gm_created_check

			}

				building_type = building_grand_monument
				production_method = pm_monument_$KEY$
			}
		}
	}
}

# The ruler and how many months they have reigned, for a ruler's death (15
# years). A ruler already reigning in 1836 counts from the game's start. A
# ruler who is not last month's begins A New Reign, unless a regency is in
# force: a regent, appointed or replaced, is not a reign (vanilla's regency.1
# adds the regency modifier and makes the regent the ruler in one block, so
# has_gov_regency is already true when this pass first sees the regent).
gm_com_track_ruler = {
	if = {
		limit = { exists = ruler }
		ruler = { save_scope_as = gm_com_tracked_ruler }
		if = {
			# exists: the stored ruler can have died, and comparing a dead
			# character logged "Event target link 'var' returned an invalid
			# object" (2026-10-06 observer run). The else branch then starts
			# the new ruler's count, as it should.
			limit = {
				has_variable = gm_ruler
				exists = var:gm_ruler
				var:gm_ruler = scope:gm_com_tracked_ruler
				has_variable = gm_ruler_months
			}
			change_variable = { name = gm_ruler_months add = 1 }
		}
		else = {
			# A ruler was recorded before, and it is not this one: a new reign.
			# On a game's first pass nothing was recorded, so nothing is offered.
			# The regent is recorded all the same, so the monarch who takes over
			# when the regency ends is seen as the new ruler.
			if = {
				limit = {
					has_variable = gm_ruler
					NOT = {
						AND = {
							exists = var:gm_ruler
							var:gm_ruler = scope:gm_com_tracked_ruler
						}
					}
					has_gov_regency = no
				}
				gm_com_try_reign = yes
			}
			set_variable = { name = gm_ruler value = scope:gm_com_tracked_ruler }
			set_variable = { name = gm_ruler_months value = 0 }
		}
	}
}

# A New Reign (spec §1): To the Crown for a crowned ruler, else To the Leader
# where the government can raise one; neither, no opportunity. Named for the
# new ruler (scope:gm_com_tracked_ruler).
gm_com_try_reign = {
	if = {
		limit = {
			gm_com_can_offer = yes
			gm_country_can_raise_crown = yes
		}
		gm_com_begin = { SOURCE = reign DEDICATION = crown NAMESAKE = ruler FORM = column }
		set_variable = { name = gm_com_name_char value = scope:gm_com_tracked_ruler }
		set_variable = { name = gm_com_ig value = flag:landowners }
		gm_com_send_offer = yes
	}
	else_if = {
		limit = {
			gm_com_can_offer = yes
			gm_country_can_raise_leader = yes
		}
		gm_com_begin = { SOURCE = reign DEDICATION = leader NAMESAKE = ruler FORM = statue }
		set_variable = { name = gm_com_name_char value = scope:gm_com_tracked_ruler }
		set_variable = { name = gm_com_ig value = flag:none }
		scope:gm_com_tracked_ruler = {
			interest_group ?= {
				gm_ig_to_flag_on_owner = { VAR = gm_com_ig }
			}
		}
		if = {
			limit = { var:gm_com_ig = flag:none }
			gm_com_ig_by_clout = { IN = yes }
		}
		gm_com_send_offer = yes
	}
}

# The founding year, for a country that appears after the game starts (a
# release; formation sets it in gm_formed_on_action). Countries standing in 1836
# carry gm_com_start_country and have a founding year only from
# gm_seed_founding_years. Never for a side in a civil war: a revolutionary
# country's own value would beat the nation's when it wins.
gm_com_init_founded = {
	if = {
		limit = {
			NOT = { has_variable = gm_founded_year }
			NOT = { has_variable = gm_com_start_country }
			NOT = { has_variable = te_cw_role }
			is_revolutionary = no
		}
		set_variable = { name = gm_founded_year value = year }
	}
}

# A New Order (spec §1): the form of government a regime dedication honours
# changes to one that has no fitting monument. Read as an edge, month on month.
# Revolution is tested first: a Council Republic is also a republic.
gm_com_track_regime = {
	set_variable = { name = gm_regime_now value = flag:none }
	if = {
		limit = { gm_government_is_revolutionary = yes }
		set_variable = { name = gm_regime_now value = flag:revolution }
	}
	else_if = {
		limit = { monument_government_is_crowned = yes }
		set_variable = { name = gm_regime_now value = flag:crown }
	}
	else_if = {
		limit = { monument_government_is_republican = yes }
		set_variable = { name = gm_regime_now value = flag:republic }
	}
	if = {
		limit = {
			has_variable = gm_regime_last
			gm_com_can_offer = yes
		}
		gm_com_try_regime = { KEY = crown IG = landowners }
		gm_com_try_regime = { KEY = republic IG = intelligentsia }
		gm_com_try_regime = { KEY = revolution IG = trade_unions }
	}
	# Not while a side in a civil war: a winning revolution keeps its own
	# variables over the nation's, and one written all through the war would
	# already hold the new regime, so the edge it brings would never be seen.
	if = {
		limit = {
			NOT = { has_variable = te_cw_role }
			is_revolutionary = no
		}

# ---- War's end (spec §1) -----------------------------------------------------------------
# Country scope, from on_won_war / on_lost_war (monument_events_on_actions.txt)
# for a war leader after a year of war, scope:enemy_country = the other side's
# leader. Victory names the enemy (its capital, a state: a country stored in a
# variable does not print its name); without one, the year.

		limit = { gm_com_can_offer = yes }
		gm_com_begin = { SOURCE = victory DEDICATION = war_memorial NAMESAKE = occasion FORM = arch }

		}
		else = {
			set_variable = { name = gm_com_namesake value = flag:year }
		}

	}
}

}

# ---- Historical opportunities (signatures spec §6) ---------------------------------
# Country scope, from the world pass, last: a historical monument is a window,
# not a moment, so it waits while another opportunity is open and is offered
# when that closes, while its window lasts. One call per row of
# HISTORICAL_OPPORTUNITIES in test_grand_monument_registry.py, in its order.
gm_com_track_historical = {
	gm_com_try_historical = { KEY = walhalla STATE = STATE_BAVARIA FROM = 1836 UNTIL = 1876 DEDICATION = civic LEVEL = 5 IG = petty_bourgeoisie COND = german }
	gm_com_try_historical = { KEY = bunker_hill_monument STATE = STATE_MASSACHUSETTS FROM = 1836 UNTIL = 1876 DEDICATION = war_memorial LEVEL = 5 IG = armed_forces COND = none }
	gm_com_try_historical = { KEY = obelisk_of_the_second_of_may STATE = STATE_NEW_CASTILE FROM = 1836 UNTIL = 1876 DEDICATION = war_memorial LEVEL = 5 IG = armed_forces COND = none }
	gm_com_try_historical = { KEY = muhammad_ali_mosque STATE = STATE_LOWER_EGYPT FROM = 1836 UNTIL = 1876 DEDICATION = religious LEVEL = 10 IG = devout COND = sunni }
	gm_com_try_historical = { KEY = nelsons_column STATE = STATE_HOME_COUNTIES FROM = 1840 UNTIL = 1880 DEDICATION = war_memorial LEVEL = 5 IG = armed_forces COND = none }
	gm_com_try_historical = { KEY = washington_monument STATE = STATE_DISTRICT_OF_COLUMBIA FROM = 1848 UNTIL = 1888 DEDICATION = republic LEVEL = 10 IG = intelligentsia COND = republic }
}

# Offered once a game (gm_hist_offered_$KEY$, global), to whoever holds a part
# of $STATE$, from the year its construction began for forty years ($FROM$ to
# $UNTIL$), at any rank, while its holder condition holds ($COND$: none, a
# German primary culture, Sunni, a republic) and its dedication can be raised.
# Its asking group is the dedication's approving one; its target is the
# landmark's levels, and its state the holder's part of the region. It names
# nothing on its own (NAMESAKE = year): completing it makes the monument the
# landmark (gm_state_become_landmark). No boon: what it gives is the landmark.
gm_com_try_historical = {
	if = {
		limit = {
			NOT = { has_global_variable = gm_hist_offered_$KEY$ }
			game_date >= $FROM$.1.1
			game_date < $UNTIL$.1.1
			gm_com_can_offer = yes
			has_state_in_state_region = $STATE$
			gm_com_hist_cond_$COND$ = yes
			gm_country_can_raise_$DEDICATION$ = yes
		}
		set_global_variable = gm_hist_offered_$KEY$
		gm_com_begin = { SOURCE = historical DEDICATION = $DEDICATION$ NAMESAKE = year FORM = default }
		set_variable = { name = gm_com_landmark value = flag:$KEY$ }
		set_variable = { name = gm_com_hist_cond value = flag:$COND$ }
		set_variable = { name = gm_com_target value = $LEVEL$ }
		set_variable = { name = gm_com_ig value = flag:$IG$ }
		clear_saved_scope = gm_com_hist_state
		random_scope_state = {
			limit = { state_region = s:$STATE$ }
			save_scope_as = gm_com_hist_state
		}
		if = {
			limit = { exists = scope:gm_com_hist_state }
			set_variable = { name = gm_com_state value = scope:gm_com_hist_state }
		}
		gm_com_send_offer = yes
	}
}

# ---- Completion makes the landmark (§6.2) -------------------------------------------
# State scope, from monument_events.26 (gm_com_fulfil marked the completing
# monument with gm_hist_become beside its gift): it becomes the landmark the
# opportunity named. One branch per historical landmark.
gm_state_hist_take = {
	if = {
		limit = { has_variable = gm_hist_become }
		if = {
			limit = { var:gm_hist_become = flag:walhalla }
			gm_state_become_landmark = { KEY = walhalla FORM = hall_of_fame }
		}
		if = {
			limit = { var:gm_hist_become = flag:bunker_hill_monument }
			gm_state_become_landmark = { KEY = bunker_hill_monument FORM = obelisk }
		}
		if = {
			limit = { var:gm_hist_become = flag:obelisk_of_the_second_of_may }
			gm_state_become_landmark = { KEY = obelisk_of_the_second_of_may FORM = obelisk }
		}
		if = {
			limit = { var:gm_hist_become = flag:muhammad_ali_mosque }
			gm_state_become_landmark = { KEY = muhammad_ali_mosque FORM = faith }
		}
		if = {
			limit = { var:gm_hist_become = flag:nelsons_column }
			gm_state_become_landmark = { KEY = nelsons_column FORM = column }
		}
		if = {
			limit = { var:gm_hist_become = flag:washington_monument }
			gm_state_become_landmark = { KEY = washington_monument FORM = obelisk }
		}
		remove_variable = gm_hist_become
	}
}

# State scope. The monument becomes landmark $KEY$: its key, its own skin and
# form (the style and form chosen at its unveiling give way), its historical
# name (the landmark namesake, which the historical-name lock holds), its
# unique dedication and its picture. It stands beside whatever else stands in
# the state.
gm_state_become_landmark = {
	set_variable = { name = gm_landmark value = flag:$KEY$ }
	set_variable = { name = gm_skin value = flag:landmark_$KEY$ }
	set_variable = { name = gm_form value = flag:$FORM$ }
	gm_remove_state_var = { VAR = gm_skin_axis }
	gm_state_clear_namesake = yes
	set_variable = { name = gm_namesake value = flag:landmark }
	gm_remove_state_var = { VAR = gm_name }
	gm_state_sig_backfill = yes
	if = {
		limit = { NOT = { has_variable = gm_restore_pm } }
		gm_state_update_picture = yes
	}
	debug_log = "TE_MONUMENTS: a historical opportunity completed: the monument is now the landmark $KEY$"
}
