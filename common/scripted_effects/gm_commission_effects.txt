# ============================================================================
# GRAND MONUMENTS — opportunities: scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-10-12-grand-monument-opportunities-design.md,
# which reworks v2's commissions (docs/superpowers/specs/2026-10-05-grand-
# monuments-v2-design.md §2). Players read "opportunity"; the script keeps v2's
# gm_com_* names, because a save carries an open commission in them.
#
# An opportunity is a moment the game has just produced (a new reign, a victory,
# a great power's rise, a technology) that an interest group wants marked in
# stone. Nothing is lost by letting one pass. One record per country, all
# country variables (a civil war's winner inherits variables, not modifiers or
# lists):
#   gm_com_offered        yes while the offer (monument_events.21) is open
#   gm_com_open           yes while seized and running
#   gm_com_source         flag: the moment (the spec's §1 table, one boon each)
#   gm_com_ig             flag: the asking IG's type (gm_ig_to_flag_on_owner)
#   gm_com_dedication     flag: one of the twelve dedication keys
#   gm_com_state          state, when the opportunity names one
#   gm_com_target         grandeur to add (5)
#   gm_com_baseline       counted grandeur on seizing (gm_com_fit_grandeur)
#   gm_com_months_left    gm_com_time_months on seizing (60, or 120 for a
#                         major power), counted down by gm_com_monthly
#   gm_com_namesake, gm_com_form, gm_com_occasion   flags: the name it asks for
#   gm_com_name_year, gm_com_name_char, gm_com_name_enemy_cap   its records
# Kept across opportunities: gm_com_last_state (the monument a level or a
# dedication last touched); the moment trackers (gm_ruler, gm_ruler_months,
# gm_regime_last, gm_com_faith_last, gm_com_seen_great_power,
# gm_com_seen_space_<m>, gm_com_seen_tech_<key>, gm_com_centenary_<n>,
# gm_founded_year, gm_com_opp_tracked); the boons (gm_boon_<source>, months
# left).
#
# Seized and completed, an opportunity pays three ways (spec §3): the monument
# that completed it is rebuilt five levels taller (monument_events.26), the
# asking IG's ledger rises (gm_ig_ledger_<ig>) and a boon runs for ten years.

# Country scope. Forget the opportunity (offered or open). gm_com_extended and
# gm_com_gap_half are v2's, cleared for a save that still carries them.
gm_com_clear = {
	gm_remove_state_var = { VAR = gm_com_offered }
	gm_remove_state_var = { VAR = gm_com_open }
	gm_remove_state_var = { VAR = gm_com_source }
	gm_remove_state_var = { VAR = gm_com_ig }
	gm_remove_state_var = { VAR = gm_com_dedication }
	gm_remove_state_var = { VAR = gm_com_state }
	gm_remove_state_var = { VAR = gm_com_target }
	gm_remove_state_var = { VAR = gm_com_baseline }
	gm_remove_state_var = { VAR = gm_com_months_left }
	gm_remove_state_var = { VAR = gm_com_extended }
	gm_remove_state_var = { VAR = gm_com_namesake }
	gm_remove_state_var = { VAR = gm_com_form }
	gm_remove_state_var = { VAR = gm_com_occasion }
	gm_remove_state_var = { VAR = gm_com_name_year }
	gm_remove_state_var = { VAR = gm_com_name_char }
	gm_remove_state_var = { VAR = gm_com_name_enemy_cap }
	gm_remove_state_var = { VAR = gm_com_gap_half }
	gm_remove_state_var = { VAR = gm_com_offer_watch }
	every_scope_state = {
		limit = { has_variable = gm_unveiled_for_com }
		remove_variable = gm_unveiled_for_com
	}
}

# Country scope. An opportunity closes (let pass, seized, run out or lapsed).
# Nothing follows it: no cooldown, no cost.
gm_com_close = {
	gm_com_clear = yes
}

# Country scope. Add $AMOUNT$ grandeur to the ledger of the IG whose type the
# country's variable $VAR$ names (gm_com_ig). Through the per-IG curve,
# 15 is about 2 approval.
gm_country_ledger_ig_by_flag = {
	gm_init_ledgers = yes
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = armed_forces }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = devout }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = industrialists }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = intelligentsia }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = landowners }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = petty_bourgeoisie }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = rural_folk }
	gm_country_ledger_ig_by_flag_one = { VAR = $VAR$ AMOUNT = $AMOUNT$ IG = trade_unions }

gm_country_ledger_ig_by_flag_one = {
	if = {
		limit = {
			has_variable = $VAR$
			var:$VAR$ = flag:$IG$
		}
		change_variable = { name = gm_ig_ledger_$IG$ add = $AMOUNT$ }
	}
}

# Country scope. The strongest interest group in the government ($IN$ = yes)
# or outside it ($IN$ = no), as gm_com_ig; flag:none when there is none.
gm_com_ig_by_clout = {
	set_variable = { name = gm_com_ig value = flag:none }
	ordered_interest_group = {
		limit = { is_in_government = $IN$ }
		order_by = ig_clout
		position = 0
		check_range_bounds = no
		gm_ig_to_flag_on_owner = { VAR = gm_com_ig }
	}
}

# Country scope. The Petty Bourgeoisie, To the Nation's approving group, unless
# they are marginal; then the strongest group in government.
gm_com_ig_for_the_nation = {
	if = {
		limit = {
			ig:ig_petty_bourgeoisie ?= { is_marginal = no }
		}
		set_variable = { name = gm_com_ig value = flag:petty_bourgeoisie }
	}
	else = {
		gm_com_ig_by_clout = { IN = yes }
	}
}

# ---- Making an offer (spec §1) ------------------------------------------------------
# Country scope. The common part of every offer: what is asked for and how it
# would be named. The caller then sets gm_com_ig, any records and the state, and
# sends it (gm_com_send_offer). FORM = default leaves the form to the
# unveiling (the skin's first form).
gm_com_begin = {
	gm_com_clear = yes
	set_variable = { name = gm_com_source value = flag:$SOURCE$ }
	set_variable = { name = gm_com_dedication value = flag:$DEDICATION$ }
	set_variable = { name = gm_com_namesake value = flag:$NAMESAKE$ }
	set_variable = { name = gm_com_form value = flag:$FORM$ }
	set_variable = { name = gm_com_target value = 5 }
	set_variable = { name = gm_com_name_year value = year }
}

# Country scope. Fix the opportunity to the capital, if a monument of its
# dedication can stand there; otherwise it can be met anywhere (one Grand
# Monument per state: a capital holding another dedication could never meet it).
gm_com_fix_to_capital = {
	if = {
		limit = {
			capital ?= { gm_state_can_host_commission = yes }
		}
		capital = { save_scope_as = gm_com_capital }
		set_variable = { name = gm_com_state value = scope:gm_com_capital }
	}
}

# Country scope. Send the offer, once an asking group is known; with none (no
# interest group of the eight found), forget it.
gm_com_send_offer = {
	if = {
		limit = {
			has_variable = gm_com_ig
			NOT = { var:gm_com_ig = flag:none }
		}
		set_variable = gm_com_offered
		# The offer's watchdog: an offer still open after six months (its event
		# never fired, or never closed) is forgotten by the world pass, so it
		# cannot block every later opportunity.
		set_variable = { name = gm_com_offer_watch value = yes months = 6 }
		if = {
			limit = {
				NOT = { has_journal_entry = je_grand_monuments }
			}
			add_journal_entry = { type = je_grand_monuments }
		}
		trigger_event = { id = monument_events.21 }
	}
	else = {
		gm_com_clear = yes
	}
}

# Country scope. The scopes the offer's and the ceremony's text read: the
# asking interest group, the opportunity's state, the enemy (its capital's
# owner) and the ruler it names.
gm_com_save_scopes = {
	gm_com_save_ig_scope = { IG = armed_forces }
	gm_com_save_ig_scope = { IG = devout }
	gm_com_save_ig_scope = { IG = industrialists }
	gm_com_save_ig_scope = { IG = intelligentsia }
	gm_com_save_ig_scope = { IG = landowners }
	gm_com_save_ig_scope = { IG = petty_bourgeoisie }
	gm_com_save_ig_scope = { IG = rural_folk }
	gm_com_save_ig_scope = { IG = trade_unions }
	if = {
		limit = { has_variable = gm_com_state }
		var:gm_com_state ?= { save_scope_as = gm_com_where }
	}
	if = {
		limit = { has_variable = gm_com_name_enemy_cap }
		var:gm_com_name_enemy_cap ?= {
			owner ?= { save_scope_as = gm_com_enemy }
		}
	}
	if = {
		limit = { has_variable = gm_com_name_char }
		var:gm_com_name_char ?= { save_scope_as = gm_com_honoree }
	}
}

gm_com_save_ig_scope = {
	if = {
		limit = {
			has_variable = gm_com_ig
			var:gm_com_ig = flag:$IG$
		}
		ig:ig_$IG$ ?= { save_scope_as = gm_com_ig_scope }
	}
}

# ---- The offer's two answers (monument_events.21) -------------------------------------
# Country scope, ROOT = the country.
gm_com_accept = {
	gm_remove_state_var = { VAR = gm_com_offered }
	set_variable = gm_com_open
	set_variable = { name = gm_com_months_left value = gm_com_time_months }
	set_variable = { name = gm_com_baseline value = gm_com_fit_grandeur }
	set_variable = gm_active
	gm_init_ledgers = yes
	if = {
		limit = { is_ai = yes }
		gm_com_ai_queue = yes
	}
	trigger_event = { id = monument_events.20 }
}

# Let It Pass: from the offer, or from the open opportunity's panel

gm_com_decline = {
	gm_com_close = yes
	trigger_event = { id = monument_events.20 }
}

# ---- The AI's construction (v2 §2.6) ----------------------------------------------------
# Country scope, AI only, on seizing. Queues the target's levels as
# government construction (start_building_construction: the AI pays for them,
# unlike create_building) in the first of: the opportunity's state; the tallest
# fitting monument of its dedication; an undedicated monument (the ceremony's
# AI weight then picks the opportunity's dedication); the capital, if it has no
# monument; the most populous incorporated state without one. One level a call,
# in a `while` with a limit: an <int> argument does not take a variable.
gm_com_ai_queue = {
	clear_saved_scope = gm_com_build_state
	if = {
		limit = { has_variable = gm_com_state }
		var:gm_com_state ?= { save_scope_as = gm_com_build_state }
	}
	else_if = {
		limit = {
			any_scope_state = {
				gm_state_status_fits = yes
				gm_state_has_dedication_flag = { VAR = gm_com_dedication }
			}
		}
		ordered_scope_state = {
			limit = {
				gm_state_status_fits = yes
				gm_state_has_dedication_flag = { VAR = gm_com_dedication }
			}
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_com_build_state
		}
	}
	else_if = {
		limit = {
			any_scope_state = {
				has_building = building_grand_monument
				gm_state_is_dedicated = no
			}
		}
		random_scope_state = {
			limit = {
				has_building = building_grand_monument
				gm_state_is_dedicated = no
			}
			save_scope_as = gm_com_build_state
		}
	}
	else_if = {
		limit = {
			capital ?= {
				NOT = { has_building = building_grand_monument }
			}
		}
		capital = { save_scope_as = gm_com_build_state }
	}
	else_if = {
		limit = {
			any_scope_state = {
				is_incorporated = yes
				NOT = { has_building = building_grand_monument }
			}
		}
		ordered_scope_state = {
			limit = {
				is_incorporated = yes
				NOT = { has_building = building_grand_monument }
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			save_scope_as = gm_com_build_state
		}
	}
	if = {
		limit = { exists = scope:gm_com_build_state }
		set_variable = { name = gm_com_queued value = 0 }
		while = {
			limit = { var:gm_com_queued < var:gm_com_target }
			scope:gm_com_build_state = {
				start_building_construction = building_grand_monument
			}
			change_variable = { name = gm_com_queued add = 1 }
		}
		remove_variable = gm_com_queued
	}
}

# ---- The open opportunity (spec §2) ---------------------------------------------------
# Country scope. From gm_country_refresh, before the totals so its rewards
# count in the same refresh (a level finished, a dedication, the month), and
# from gm_country_monthly before the clock: met once the progress reaches the
# target. A counted monument that drops out (ceded, contested, demolished)
# lowers the baseline with it, so it stops counting instead of cancelling
# grandeur added since.
# Compared directly, never through a variable written here: the refresh also
# runs inside the policy buttons' tooltips, which walk the effect without
# writing anything, so a variable set and read in one pass reads as unset
# there (the old gm_com_drop logged ~150 errors a burst, 2026-10-07).
gm_com_check_progress = {
	if = {
		limit = { has_variable = gm_com_open }
		if = {
			limit = { var:gm_com_baseline > gm_com_fit_grandeur }
			set_variable = { name = gm_com_baseline value = gm_com_fit_grandeur }
		}
		if = {
			limit = { var:gm_com_target <= gm_com_progress }
			gm_com_fulfil = yes
		}
	}
}

# Country scope. From gm_country_monthly, after gm_com_check_progress and
# before the refresh: the clock, then a lapse before a run-out. Neither costs
# anything.
gm_com_monthly = {
	if = {
		limit = { has_variable = gm_com_open }
		change_variable = { name = gm_com_months_left add = -1 }
		if = {
			limit = { gm_com_has_lapsed = yes }
			gm_com_lapse = yes
		}
		else_if = {
			limit = { var:gm_com_months_left <= 0 }
			gm_com_miss = yes
		}
	}
}

# Country scope. Seized and completed (spec §3). The monument whose level or
# dedication completed it (gm_com_last_state, if it still counts), else the
# tallest counting monument, takes the opportunity's name -- unless it was
# unveiled while this opportunity was open, when the player already chose its
# name with the opportunity's offered first (gm_unveiled_for_com), or the
# opportunity asks only for the city's name and the monument already has a
# name: that would erase a chosen name for everyone's default -- and is rebuilt
# taller by monument_events.26 (gm_gift; never here: this runs inside the
# refresh, which tooltips walk). Then the asking group's approval and the boon.
gm_com_fulfil = {
	clear_saved_scope = gm_com_named_state
	if = {
		limit = {
			has_variable = gm_com_last_state
			var:gm_com_last_state ?= { gm_state_com_counts = yes }
		}
		var:gm_com_last_state = { save_scope_as = gm_com_named_state }
	}
	else = {
		ordered_scope_state = {
			limit = { gm_state_com_counts = yes }
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_com_named_state
		}
	}
	if = {
		limit = { exists = scope:gm_com_named_state }
		scope:gm_com_named_state = {
			if = {
				limit = {
					NOT = { has_variable = gm_unveiled_for_com }
					OR = {
						NOT = { has_variable = gm_namesake }
						owner = {
							NOT = { var:gm_com_namesake = flag:city }
						}
					}
				}
				gm_state_take_commission_name = yes
			}
			set_variable = gm_gift
			trigger_event = { id = monument_events.26 }
		}
	}
	gm_country_ledger_ig_by_flag = { VAR = gm_com_ig AMOUNT = 15 }
	gm_com_grant_boon = yes
	set_variable = gm_active
	if = {
		limit = { is_ai = no }
		post_notification = gm_com_fulfilled_notice
	}
	gm_com_close = yes
}

# Country scope. Time ran out: the moment passes, at no cost.
gm_com_miss = {
	if = {
		limit = { is_ai = no }
		post_notification = gm_com_missed_notice
	}
	gm_com_close = yes
}

# Country scope. It can no longer be met (its state lost, its dedication no
# longer raisable): it lapses, at no cost.
gm_com_lapse = {
	if = {
		limit = { is_ai = no }
		post_notification = gm_com_lapsed_notice
	}
	gm_com_close = yes
}

# ---- The monument's gift (spec §3.1) -------------------------------------------------
# State scope, from monument_events.26 (the monument gm_com_fulfil marked with
# gm_gift). Rebuilt gm_com_gift levels taller, at most 200 (the rebuild
# ladder's reach), keeping everything the state records: its dedication's
# raiser, honoree and faith, its style, form, name, picture and gm_seen. No
# effect adds levels to a building the state already has predictably
# (scripting_best_practices.md, "Building Lifecycle Effect Gotchas"), so it is
# removed and built again, as Rededicate does. The rebuilt building starts on
# the engine's copy of a dedication (grand_monument_pms.txt), so the state
# records its own (gm_restore_pm) for gm_state_restore_dedication to put back;
# monument_events.20 runs it in a refresh of its own, once the new building is
# readable. gm_rebuilding spares the rebuild's own on_building_built vanity
# backlash, as for Rededicate.
gm_state_gift_grandeur = {
	gm_remove_state_var = { VAR = gm_gift }
	if = {
		limit = {
			has_building = building_grand_monument
			gm_state_is_dedicated = yes
			NOT = { has_variable = gm_restore_pm }
			gm_state_grandeur < 200
		}
		set_variable = { name = gm_grandeur value = gm_state_grandeur }
		set_variable = {
			name = gm_rebuild_level
			value = {
				value = var:gm_grandeur
				add = gm_com_gift
				max = 200
			}
		}
		gm_state_mark_restore_pm = yes
		set_variable = { name = gm_rebuilding days = 1 }
		remove_building = building_grand_monument

# ---- The world pass (spec §1): every country, every month -------------------------------
# Country scope, from gm_com_world_on_action. Bookkeeping first, always (so a
# moment seen while busy passes, not queued); each tracker offers only while
# gm_com_can_offer holds, and the first to offer takes the slot, so the order
# is the priority: a new order, a new reign, a great power, a new faith, the
# space firsts, the centenary, the technologies. The war and death
# on-actions offer at once, ahead of all of these.
gm_com_world_monthly = {
	if = {
		limit = {
			NOT = { has_global_variable = gm_com_seeded }
		}
		gm_com_seed = yes
	}
	if = {
		limit = {
			has_variable = gm_com_offered
			NOT = { has_variable = gm_com_offer_watch }
		}
		gm_com_clear = yes
	}
	if = {
		limit = {
			NOT = { has_variable = gm_com_opp_tracked }
		}
		gm_com_first_sight = yes
	}
	gm_com_init_founded = yes
	gm_com_track_regime = yes
	gm_com_track_ruler = yes
	gm_com_track_great_power = yes
	gm_com_track_faith = yes
	gm_com_track_space = yes
	gm_com_track_centenary = yes
	gm_com_track_tech = yes
}

# Country scope. The first time the world pass sees a country (a new game, a

# cooldown go.

# The ruler and how many months they have reigned, for a ruler's death (15
# years). A ruler already reigning in 1836 counts from the game's start. A
# ruler who is not last month's begins A New Reign.
gm_com_track_ruler = {
	if = {
		limit = { exists = ruler }
		ruler = { save_scope_as = gm_com_tracked_ruler }
		if = {
			# exists: the stored ruler can have died, and comparing a dead
			# character logged "Event target link 'var' returned an invalid
			# object" (2026-10-06 observer run). The else branch then starts
			# the new ruler's count, as it should.
			limit = {
				has_variable = gm_ruler
				exists = var:gm_ruler
				var:gm_ruler = scope:gm_com_tracked_ruler
				has_variable = gm_ruler_months
			}
			change_variable = { name = gm_ruler_months add = 1 }
		}
		else = {
			# A ruler was recorded before, and it is not this one: a new reign.
			# On a game's first pass nothing was recorded, so nothing is offered.
			if = {
				limit = {
					has_variable = gm_ruler
					NOT = {
						AND = {
							exists = var:gm_ruler
							var:gm_ruler = scope:gm_com_tracked_ruler
						}
					}
				}
				gm_com_try_reign = yes
			}
			set_variable = { name = gm_ruler value = scope:gm_com_tracked_ruler }
			set_variable = { name = gm_ruler_months value = 0 }
		}
	}
}

# A New Reign (spec §1): To the Crown for a crowned ruler, else To the Leader
# where the government can raise one; neither, no opportunity. Named for the
# new ruler (scope:gm_com_tracked_ruler).
gm_com_try_reign = {
	if = {
		limit = {
			gm_com_can_offer = yes
			gm_country_can_raise_crown = yes
		}
		gm_com_begin = { SOURCE = reign DEDICATION = crown NAMESAKE = ruler FORM = column }
		set_variable = { name = gm_com_name_char value = scope:gm_com_tracked_ruler }
		set_variable = { name = gm_com_ig value = flag:landowners }
		gm_com_send_offer = yes
	}
	else_if = {
		limit = {
			gm_com_can_offer = yes
			gm_country_can_raise_leader = yes
		}
		gm_com_begin = { SOURCE = reign DEDICATION = leader NAMESAKE = ruler FORM = statue }
		set_variable = { name = gm_com_name_char value = scope:gm_com_tracked_ruler }
		set_variable = { name = gm_com_ig value = flag:none }
		scope:gm_com_tracked_ruler = {
			interest_group ?= {
				gm_ig_to_flag_on_owner = { VAR = gm_com_ig }
			}
		}
		if = {
			limit = { var:gm_com_ig = flag:none }
			gm_com_ig_by_clout = { IN = yes }
		}
		gm_com_send_offer = yes
	}
}

# The founding year, for a country that appears after the game starts (a
# release; formation sets it in gm_formed_on_action). Countries standing in 1836
# carry gm_com_start_country and have a founding year only from
# gm_seed_founding_years. Never for a side in a civil war: a revolutionary
# country's own value would beat the nation's when it wins.
gm_com_init_founded = {
	if = {
		limit = {
			NOT = { has_variable = gm_founded_year }
			NOT = { has_variable = gm_com_start_country }
			NOT = { has_variable = te_cw_role }
			is_revolutionary = no
		}
		set_variable = { name = gm_founded_year value = year }
	}
}

# A New Order (spec §1): the form of government a regime dedication honours
# changes to one that has no fitting monument. Read as an edge, month on month.
# Revolution is tested first: a Council Republic is also a republic.
gm_com_track_regime = {
	set_variable = { name = gm_regime_now value = flag:none }
	if = {
		limit = { gm_government_is_revolutionary = yes }
		set_variable = { name = gm_regime_now value = flag:revolution }
	}
	else_if = {
		limit = { monument_government_is_crowned = yes }
		set_variable = { name = gm_regime_now value = flag:crown }
	}
	else_if = {
		limit = { monument_government_is_republican = yes }
		set_variable = { name = gm_regime_now value = flag:republic }
	}
	if = {
		limit = {
			has_variable = gm_regime_last
			gm_com_can_offer = yes
		}
		gm_com_try_regime = { KEY = crown IG = landowners }
		gm_com_try_regime = { KEY = republic IG = intelligentsia }
		gm_com_try_regime = { KEY = revolution IG = trade_unions }
	}
	# Not while a side in a civil war: a winning revolution keeps its own
	# variables over the nation's, and one written all through the war would
	# already hold the new regime, so the edge it brings would never be seen.
	if = {
		limit = {
			NOT = { has_variable = te_cw_role }
			is_revolutionary = no
		}
		set_variable = { name = gm_regime_last value = var:gm_regime_now }
	}
	remove_variable = gm_regime_now
}

gm_com_try_regime = {
	if = {
		limit = {
			var:gm_regime_now = flag:$KEY$
			NOT = { var:gm_regime_last = flag:$KEY$ }
			gm_com_can_offer = yes
			NOT = {
				any_scope_state = {
					gm_state_status_fits = yes
					gm_state_has_pm = { PM = pm_monument_$KEY$ }
				}
			}
		}
		gm_com_begin = { SOURCE = regime DEDICATION = $KEY$ NAMESAKE = year FORM = column }
		set_variable = { name = gm_com_ig value = flag:$IG$ }
		gm_com_send_offer = yes
	}
}

# A New Faith (spec §1): the state religion is not last month's. First sight

# ---- Seeding (history, and old saves) -----------------------------------------------------
# Any scope. Every country standing now is a start country: it has a founding
# year only if gm_seed_founding_years gives it one, so 1936 does not bring a
# centenary to every country of 1836 at once. A space first already won is
# marked seen: no opportunity for an old achievement.
gm_com_seed = {
	set_global_variable = gm_com_seeded
	every_country = {

		}
	}
	gm_seed_founding_years = yes
}
