# GRAND MONUMENTS — the landmarks standing in 1836 (phase 4)
# ============================================================================
# Design: docs/superpowers/specs/2026-10-12-grand-monuments-landmarks-design.md.
# A landmark is an ordinary Grand Monument with five records on its state:
#   gm_landmark        flag: its key. Locks the skin and form (gm_state_is_landmark,
#                      monument_triggers.txt) while the monument stands
#   gm_skin            flag:landmark_<key>, its own skin: its name and its picture
#   gm_form            its one form, from the roster
#   gm_namesake        flag:landmark: the name prints the skin's (gm_name_landmark_row)
#   gm_ceremony_level  its level, so it never reads as the engine's copy
# The first refresh (monument_events.20) writes the rest: gm_seen, and for a
# bound dedication gm_raised_by (gm_state_first_sight). It has its dedication's
# usual effects and nothing more, and takes no inscription.
#
# test_grand_monument_registry.py's LANDMARKS is the table: one call per row,
# in its order, which is also the picture codes' (1,000,000 + the row number,
# gm_state_picture_landmarks).

# The part of $STATE$ that $TAG$ holds in the 1836 history. An owner that
# proves wrong, or a Grand Monument already standing there (create_building
# would keep the larger level), places nothing and says so in debug.log.
gm_seed_landmark = {
	if = {
		limit = {
			s:$STATE$.region_state:$TAG$ ?= {
				NOT = { has_building = building_grand_monument }
			}
		}
		s:$STATE$.region_state:$TAG$ = {
			# The records first, as Rededicate sets its guard first: if
			# create_building fires on_building_built at once, monument_events.1
			# then charges no vanity (gm_rebuilding) and sees no copy to send to
			# the ceremony (gm_ceremony_level).
			set_variable = { name = gm_rebuilding days = 1 }
			set_variable = { name = gm_landmark value = flag:$KEY$ }
			set_variable = { name = gm_skin value = flag:landmark_$KEY$ }
			set_variable = { name = gm_form value = flag:$FORM$ }
			set_variable = { name = gm_namesake value = flag:landmark }
			set_variable = { name = gm_ceremony_level value = $LEVEL$ }
			create_building = {
				building = building_grand_monument
				level = $LEVEL$
				activate_production_methods = { pm_monument_$DEDICATION$ } # REVIEWED 2026-10-11: a create_building parameter (effects.log 1.14.5 § create_building), not an effect
			}
			debug_log = "TE_MONUMENTS: landmark $KEY$ placed in $STATE$ for $TAG$: level $LEVEL$, pm_monument_$DEDICATION$, form $FORM$"
			# The refresh records it (gm_state_first_sight), opens the
			# journal entry and sets its picture.
			owner = { trigger_event = { id = monument_events.20 } }
		}
	}
	else = {
		debug_log = "TE_MONUMENTS: landmark $KEY$ not placed: no part of $STATE$ held by $TAG$, or a Grand Monument already stands there"
	}
}

		limit = {

				has_variable = gm_landmark
				var:gm_landmark = flag:$KEY$
			}
		}
		every_state = {
			limit = {
				has_variable = gm_landmark
				var:gm_landmark = flag:$KEY$
			}
			debug_log = "TE_MONUMENTS: landmark $KEY$: level [SCOPE.ScriptValue('gm_state_grandeur')|0], status [SCOPE.ScriptValue('gm_status_code')|0], picture [SCOPE.ScriptValue('gm_disp_picture_code')|0]"
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = undedicated }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = crown }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = republic }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = revolution }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = leader }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = religious }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = civic }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = war_memorial }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = artistic }
			gm_debug_log_landmark_pm = { KEY = $KEY$ PM = naturalist }
