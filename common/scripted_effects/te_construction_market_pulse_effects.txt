# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

te_construction_market_on_weekly_pulse = {
	# free_market_construction_rule disabled: vanilla-style direct construction
	# (te_construction_market_direct_effects.txt). Runs for decentralized
	# countries too; it skips their capacity itself.
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		te_direct_construction_on_weekly_pulse = yes
	}
	else_if = {
		limit = {
			NOT = {
				is_country_type = decentralized
			}
			#this = market.owner
		}
		te_construction_market_update_price_stabilizer = yes
		te_construction_market_country_update_data = yes
		if = {
			limit = {
				is_ai = yes
			}
			te_construction_market_ai_update_buy = yes
		}
		te_construction_market_reset_direct_sites = yes
	}
}


te_construction_market_on_border_changes = {
	if = {
		limit = {
			NOT = {
				is_country_type = decentralized
			}
			#this = market.owner
		}
		te_construction_market_country_update_sites = yes
		# Direct construction: a country released or reshaped mid-week gets its
		# modifiers, capacity and site PMs now rather than at the next pulse.
		if = {
			limit = {
				te_free_market_construction_off = yes
			}
			te_direct_construction_on_weekly_pulse = yes
		}
	}
}
