# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

te_construction_market_on_weekly_pulse = {
	# free_market_construction_rule disabled: vanilla-style direct construction
	# (te_construction_market_direct_effects.txt). The rule's production-method
	# flags do the construction; the pulse only keeps the law-set private share.
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		te_direct_construction_on_weekly_pulse = yes
	}
	else_if = {
		limit = {
			NOT = {
				is_country_type = decentralized
			}
			#this = market.owner
		}
		te_construction_market_update_price_stabilizer = yes
		te_construction_market_country_update_data = yes
		if = {
			limit = {
				is_ai = yes
			}
			te_construction_market_ai_update_buy = yes
		}
		# Last, after the sites are sized; see the effect for why.
		te_construction_market_ensure_site = yes
	}
}


te_construction_market_on_border_changes = {
	if = {
		limit = {
			te_free_market_construction_on = yes
			NOT = {
				is_country_type = decentralized
			}
			#this = market.owner
		}
		# Places sites only. Capacity is left to the weekly pulse, its one
		# refresh site: re-running it here, right after sites are created,
		# would size a not-yet-staffed site at 10x (see
		# te_construction_market_employment_adjusted_per_site). Direct games
		# have nothing to place: their sites are built by their owners.
		te_construction_market_country_update_sites = yes
	}
}
