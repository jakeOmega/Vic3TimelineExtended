# the province iterators don't run from a state. Where one region
# holds both of a strait's shores (Panama, Suez, Kiel, Lombok) a state holding

# Self-check, around te_seize_strait_shores in the goal's on_enforced
# (root = holder; scope:target_country, scope:te_ss_region set). OK when the
# shore provinces the holder owns in the region went up, BROKEN when they
# didn't; the holder usually owned part of the region already (Spain beside
# Gibraltar, Johore and Siam beside Singapore), so owning land there proves
# nothing. Nothing in the base game settles how `set_owner_of_provinces` splits
# a state's people, so the line also logs the population each side holds in
# the region before and after. The figures go through global variables because
# a saved scope doesn't print reliably in debug_log (scripting_best_practices.md
# § Runtime Debugging with debug_log). docs/testing/strait-wargoal-playtest.md.
te_seize_strait_selfcheck_before = {
	set_global_variable = { name = te_ss_chk_holder_before value = te_ss_holder_region_pop }
	set_global_variable = { name = te_ss_chk_target_before value = te_ss_target_region_pop }
	set_global_variable = { name = te_ss_chk_shores_before value = te_ss_holder_region_shores }
}

te_seize_strait_selfcheck_after = {
	set_global_variable = { name = te_ss_chk_holder_after value = te_ss_holder_region_pop }
	set_global_variable = { name = te_ss_chk_target_after value = te_ss_target_region_pop }
	set_global_variable = { name = te_ss_chk_shores_after value = te_ss_holder_region_shores }
	if = {
		limit = {
			global_var:te_ss_chk_shores_after > global_var:te_ss_chk_shores_before
		}
		debug_log = "TE_SELF_CHECK OK te_seize_strait holder=[THIS.GetCountry.GetNameNoFormatting] ai=[SCOPE.ScriptValue('te_ss_chk_holder_is_ai')|0] shores=[SCOPE.ScriptValue('te_ss_chk_shores_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_shores_after_g')|0] holder_pop=[SCOPE.ScriptValue('te_ss_chk_holder_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_holder_after_g')|0] target_pop=[SCOPE.ScriptValue('te_ss_chk_target_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_target_after_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (the holder owns more of the strait's shore after the transfer; the populations show how the engine split the state's people)"
	}
	else = {
		debug_log = "TE_SELF_CHECK BROKEN te_seize_strait holder=[THIS.GetCountry.GetNameNoFormatting] ai=[SCOPE.ScriptValue('te_ss_chk_holder_is_ai')|0] shores=[SCOPE.ScriptValue('te_ss_chk_shores_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_shores_after_g')|0] holder_pop=[SCOPE.ScriptValue('te_ss_chk_holder_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_holder_after_g')|0] target_pop=[SCOPE.ScriptValue('te_ss_chk_target_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_target_after_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (the holder owns no more of the strait's shore after the transfer: set_owner_of_provinces did not run, found no province, or takes effect later in the day; see docs/testing/strait-wargoal-playtest.md)"
	}
	remove_global_variable = te_ss_chk_holder_before
	remove_global_variable = te_ss_chk_target_before
	remove_global_variable = te_ss_chk_shores_before
	remove_global_variable = te_ss_chk_holder_after
	remove_global_variable = te_ss_chk_target_after
	remove_global_variable = te_ss_chk_shores_after
}
