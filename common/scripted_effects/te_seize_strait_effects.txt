# Self-check, around te_seize_strait_shores in the goal's on_enforced
# (root = holder; scope:target_country, scope:te_ss_region set). Nothing in
# the base game settles how `set_owner_of_provinces` splits a state's people,
# so each enforcement logs the population both sides hold in the region before
# and after: OK when the holder now owns land there, BROKEN when the transfer
# left it nothing. The figures go through global variables because a saved
# scope doesn't print reliably in debug_log (scripting_best_practices.md
# § Runtime Debugging with debug_log). docs/testing/strait-wargoal-playtest.md.
te_seize_strait_selfcheck_before = {
	set_global_variable = { name = te_ss_chk_holder_before value = te_ss_holder_region_pop }
	set_global_variable = { name = te_ss_chk_target_before value = te_ss_target_region_pop }
}

te_seize_strait_selfcheck_after = {
	set_global_variable = { name = te_ss_chk_holder_after value = te_ss_holder_region_pop }
	set_global_variable = { name = te_ss_chk_target_after value = te_ss_target_region_pop }
	if = {
		limit = {
			scope:te_ss_region = {
				any_scope_state = { owner = root }
			}
		}
		debug_log = "TE_SELF_CHECK OK te_seize_strait holder=[THIS.GetCountry.GetNameNoFormatting] ai=[SCOPE.ScriptValue('te_ss_chk_holder_is_ai')|0] holder_pop=[SCOPE.ScriptValue('te_ss_chk_holder_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_holder_after_g')|0] target_pop=[SCOPE.ScriptValue('te_ss_chk_target_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_target_after_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (the holder owns land in the strait's state region after the transfer; the populations show how the engine split the state's people)"
	}
	else = {
		debug_log = "TE_SELF_CHECK BROKEN te_seize_strait holder=[THIS.GetCountry.GetNameNoFormatting] ai=[SCOPE.ScriptValue('te_ss_chk_holder_is_ai')|0] holder_pop=[SCOPE.ScriptValue('te_ss_chk_holder_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_holder_after_g')|0] target_pop=[SCOPE.ScriptValue('te_ss_chk_target_before_g')|0]->[SCOPE.ScriptValue('te_ss_chk_target_after_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (the holder owns nothing in the strait's state region after the transfer: set_owner_of_provinces did not run, or ran on a province the state no longer held; see docs/testing/strait-wargoal-playtest.md)"
	}
	remove_global_variable = te_ss_chk_holder_before
	remove_global_variable = te_ss_chk_target_before
	remove_global_variable = te_ss_chk_holder_after
	remove_global_variable = te_ss_chk_target_after
}
