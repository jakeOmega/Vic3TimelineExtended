# ============================================================================
# COVERT WARFARE TEST CONSOLE — helpers
# ============================================================================
# Called only from events/te_debug_covert_events.txt, which is itself only ever
# fired by hand from the console. Nothing in the mod calls anything here.
#
# These deliberately go through the shipping effects wherever they can
# (covert_refresh_funding_state, covert_op_start, covert_op_refresh_phase), so
# testing a state also tests the code that maintains it.
# ============================================================================

# Put funding at an exact level and bring every dependent modifier and display
# value with it, exactly as a stepper click would.
# Parameters: $LEVEL$ = 0 .. iw_funding_level_max
# Scope: country
te_debug_covert_set_funding = {
	set_variable = { name = iw_funding_level value = $LEVEL$ }
	clamp_variable = { name = iw_funding_level min = 0 max = iw_funding_level_max }
	covert_refresh_funding_state = yes
}

# Start one operation of $TYPE$ against scope:te_debug_covert_target and set its
# duration, so a phase can be reached without waiting for it.
# covert_op_start creates the container; the pact is added separately because
# the container is what the widget draws, and covert_ops_sync_all would collect
# a container with no pact at the next monthly pulse.
# Parameters: $TYPE$ = short name, $ACTION$ = diplomatic_action_type key,
#             $DEFENSE_MOD$ = type-specific defense modifier,
#             $CODE$ = operation type code 0-12, matching the branches of
#                      covert_last_exposed_type_name
#             $MONTHS$ = duration to plant
# Scope: country; scope:te_debug_covert_target must be set
te_debug_covert_plant_op = {
	scope:te_debug_covert_target = { save_scope_as = target_country }
	create_diplomatic_pact = {
		country = scope:te_debug_covert_target
		type = $ACTION$
	}
	covert_op_start = { TYPE = $TYPE$ DEFENSE_MOD = $DEFENSE_MOD$ CODE = $CODE$ }
	if = {
		limit = { has_variable_list = iw_ops }
		every_in_list = {
			variable = iw_ops
			limit = {
				has_tag = iw_op_$TYPE$
				var:iw_target ?= scope:te_debug_covert_target
			}
			set_variable = { name = iw_duration value = $MONTHS$ }
			covert_op_refresh_phase = yes
		}
	}
}

# Four operations against the same target, spanning every phase (two land in
# fully operational), each one month short of its next boundary so the
# countdown can be watched tipping over.
#   industrial espionage   month 5   preparatory, 1 month to establishing
#   influence campaign     month 11  establishing, 1 month to fully operational
#   military espionage     month 14  fully operational
#   destabilization        month 14  fully operational, severe tier — the only
#                           plant that can trigger the third-party blowback
#                           when caught, so that branch is reachable from
#                           the console without hand-building a save
# Scope: country
te_debug_covert_seed_phases = {
	if = {
		limit = { NOT = { has_variable = iw_funding_level } }
		te_debug_covert_set_funding = { LEVEL = 2 }
	}
	else_if = {
		limit = { var:iw_funding_level < 1 }
		te_debug_covert_set_funding = { LEVEL = 2 }
	}
	random_country = {
		limit = {
			NOT = { this = ROOT }
			covert_action_valid_target = yes
		}
		save_scope_as = te_debug_covert_target
	}
	if = {
		limit = { exists = scope:te_debug_covert_target }
		te_debug_covert_plant_op = {
			TYPE = industrial_espionage
			ACTION = covert_industrial_espionage_action
			DEFENSE_MOD = country_covert_defense_economic_add
			CODE = 4
			MONTHS = 5
		}
		te_debug_covert_plant_op = {
			TYPE = influence_campaign
			ACTION = covert_influence_campaign_action
			DEFENSE_MOD = country_covert_defense_ideological_add
			CODE = 6
			MONTHS = 11
		}
		te_debug_covert_plant_op = {
			TYPE = military_espionage
			ACTION = covert_military_espionage_action
			DEFENSE_MOD = country_covert_defense_military_add
			CODE = 5
			MONTHS = 14
		}
		te_debug_covert_plant_op = {
			TYPE = destabilization
			ACTION = covert_destabilization_action
			DEFENSE_MOD = country_covert_defense_ideological_add
			CODE = 8
			MONTHS = 14
		}
	}
}

# Age every running operation by a year and restate its phase.
# Scope: country
te_debug_covert_age_year = {
	if = {
		limit = { has_variable_list = iw_ops }
		every_in_list = {
			variable = iw_ops
			change_variable = { name = iw_duration add = 12 }
			covert_op_refresh_phase = yes
		}
	}
}

# Re-derive every container's display state, which is what the first monthly
# pulse after loading an older save does. Use it to close the window in which
# covert_ops_phase_ready_sgui hides the phase lines.
# Scope: country
te_debug_covert_refresh_display = {
	covert_ops_sync_all = yes
	if = {
		limit = { has_variable_list = iw_ops }
		every_in_list = {
			variable = iw_ops
			covert_op_refresh_phase = yes
		}
	}
}

# Make the detection roll land on one specific operation: pin a random running
# operation's iw_detect to 100 and run the shipping roll effect, so the roll,
# the temporary-list pick and the scope hand-off to covert_warfare.1 are all
# exercised. The next monthly sync restores the honest figure.
# Scope: country
te_debug_covert_force_detection = {
	if = {
		limit = { has_variable_list = iw_ops }
		random_in_list = {
			variable = iw_ops
			set_variable = { name = iw_detect value = 100 }
		}
		covert_ops_roll_detection_all = yes
	}
}

# Raise one running operation to the top priority through the shipping step
# effect, twice — the same code the row stepper's scripted GUI runs, minus
# the GUI. Exercises the priority variable, upkeep and detection immediately;
# the scaled effect modifiers only pick up the new priority at the next
# monthly pulse (covert_apply_priority_change deliberately doesn't re-apply
# them on the click — see its own comment). Still works if the stepper's
# saved scope turns out not to arrive.
# The pick is held as scope:iw_debug_op and re-saved as scope:iw_op before
# each step: a step runs covert_refresh_funding_state -> covert_ops_sync_all,
# whose covert_op_refresh_detection re-saves scope:iw_op per container, so
# without the re-save the second step would move a different operation.
# Scope: country
te_debug_covert_max_priority = {
	if = {
		limit = { has_variable_list = iw_ops }
		random_in_list = {
			variable = iw_ops
			save_scope_as = iw_debug_op
		}
		scope:iw_debug_op = { save_scope_as = iw_op }
		covert_effect_priority_up = yes
		scope:iw_debug_op = { save_scope_as = iw_op }
		covert_effect_priority_up = yes
	}
}

# Put every network at an exact strength (slice 4). Runs the real network pass
# first so every running operation's target has one, then overwrites the
# strength and re-runs the operation sync so each row's detection line shows
# the new cover at once. The head start only reaches operations started
# after this — it is fixed at creation.
# Note: the network pass also ticks once, which is why the strength is
# written after it, not before. The intelligence report (slice 7) is refreshed
# after the write too, as the pass itself does, so the row shows the report
# the new strength earns at once.
# Parameters: $VALUE$ = 0 .. covert_net_max
# Scope: country
te_debug_covert_set_net_strength = {
	covert_nets_sync = yes
	if = {
		limit = { has_variable_list = iw_nets }
		save_scope_as = iw_net_operator
		every_in_list = {
			variable = iw_nets
			set_variable = { name = iw_net_strength value = $VALUE$ }
			covert_net_clamp = yes
			covert_net_refresh_intel = yes
		}
	}
	covert_ops_sync_all = yes
}

# Every network's target starts a cultivate-assets operation against us
# (slice 7), so a tier-2 network's "operations against us" line has something
# to count. Goes through te_debug_covert_plant_op in the target's own scope,
# like any operation it launched itself; funding is floored at 1 on the target
# (without touching its journal entry, which it may not have), or the pact's
# requirement_to_maintain ends it at the next check. Targets already running
# one against us are skipped (is_initiator, so our own cultivate-assets
# operation against them does not count). Then refreshes every network's
# report.
# Scope: country
te_debug_covert_hostile_cultivate = {
	save_scope_as = te_debug_covert_victim
	if = {
		limit = { has_variable_list = iw_nets }
		every_in_list = {
			variable = iw_nets
			var:iw_target ?= {
				if = {
					limit = {
						is_country_alive = yes
						NOT = {
							has_diplomatic_pact = {
								who = scope:te_debug_covert_victim
								type = covert_cultivate_assets_action
								is_initiator = yes
							}
						}
					}
					add_to_temporary_list = te_debug_covert_hostiles
				}
			}
		}
		every_in_list = {
			list = te_debug_covert_hostiles
			if = {
				limit = {
					OR = {
						NOT = { has_variable = iw_funding_level }
						var:iw_funding_level < 1
					}
				}
				set_variable = { name = iw_funding_level value = 1 }
			}
			scope:te_debug_covert_victim = { save_scope_as = te_debug_covert_target }
			te_debug_covert_plant_op = {
				TYPE = cultivate_assets
				ACTION = covert_cultivate_assets_action
				DEFENSE_MOD = country_covert_defense_ideological_add
				CODE = 12
				MONTHS = 0
			}
		}
	}
	te_debug_covert_refresh_net_intel = yes
}

# Refresh every network's intelligence report now (slice 7), without a tick.
# Re-saves scope:iw_net_operator first: planting an operation from another
# country's scope (above) re-saves it as that country.
# Scope: country
te_debug_covert_refresh_net_intel = {
	save_scope_as = iw_net_operator
	if = {
		limit = { has_variable_list = iw_nets }
		every_in_list = {
			variable = iw_nets
			covert_net_refresh_intel = yes
		}
	}
}

# One month of network growth / decay without waiting for the pulse. Stand an
# operation down first to watch its network decay instead.
# Scope: country
te_debug_covert_tick_nets = {
	covert_nets_sync = yes
	covert_ops_sync_all = yes
}

# Put the agency's Tradecraft at an exact score (slice 5) and re-apply the tier
# bonus at once, so the command centre's line, the capacity breakdown and the
# unlocks can be checked without months of play. Records no reason: the line
# keeps showing whatever last moved the score for real.
# Parameters: $VALUE$ = 0 .. covert_tradecraft_max
# Scope: country
te_debug_covert_set_tradecraft = {
	covert_tradecraft_init = yes
	set_variable = { name = iw_tradecraft value = $VALUE$ }
	covert_tradecraft_clamp = yes
	covert_tradecraft_refresh_bonus = yes
}

# One monthly Tradecraft pass without waiting for the pulse (gain, or idle
# decay with no operations running).
# Scope: country
te_debug_covert_tick_tradecraft = {
	covert_tradecraft_monthly = yes
}

# Slice 6's four operations, each planted fully operational (month 14) so
# their effects land at the next monthly pulse. Each needs a target that meets
# its requirement_to_maintain, or the pact lapses at the next check:
#   regime change + cultivate assets  one valid target; a rivalry is declared
#                                      on it first (regime change needs one)
#   nuclear sabotage                   a country whose programme is funded and
#                                      running, if there is one
#   space espionage                    a country ahead of us in the space race,
#                                      if there is one
# Funding is floored at 2, like te_debug_covert_seed_phases. Also sets
# Tradecraft to Seasoned (65), so both severe operations can be launched by
# hand from the diplomacy panel too.
# Scope: country
te_debug_covert_seed_new_ops = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = iw_funding_level }
				var:iw_funding_level < 1
			}
		}
		te_debug_covert_set_funding = { LEVEL = 2 }
	}
	te_debug_covert_set_tradecraft = { VALUE = 65 }
	random_country = {
		limit = {
			NOT = { this = ROOT }
			covert_action_valid_target = yes
			NOT = { is_subject_of = ROOT }
		}
		save_scope_as = te_debug_covert_target
	}
	if = {
		limit = { exists = scope:te_debug_covert_target }
		if = {
			limit = { NOT = { has_diplomatic_pact = { who = scope:te_debug_covert_target type = rivalry } } }
			create_diplomatic_pact = {
				country = scope:te_debug_covert_target
				type = rivalry
			}
		}
		te_debug_covert_plant_op = {
			TYPE = regime_change
			ACTION = covert_regime_change_action
			DEFENSE_MOD = country_covert_defense_ideological_add
			CODE = 9
			MONTHS = 14
		}
		te_debug_covert_plant_op = {
			TYPE = cultivate_assets
			ACTION = covert_cultivate_assets_action
			DEFENSE_MOD = country_covert_defense_ideological_add
			CODE = 12
			MONTHS = 14
		}
	}
	random_country = {
		limit = {
			NOT = { this = ROOT }
			covert_action_valid_target = yes
			nuclear_program_is_proliferating = yes
		}
		save_scope_as = te_debug_covert_nuke_target
	}
	if = {
		limit = { exists = scope:te_debug_covert_nuke_target }
		scope:te_debug_covert_nuke_target = { save_scope_as = te_debug_covert_target }
		te_debug_covert_plant_op = {
			TYPE = nuclear_sabotage
			ACTION = covert_nuclear_sabotage_action
			DEFENSE_MOD = country_covert_defense_military_add
			CODE = 10
			MONTHS = 14
		}
	}
	random_country = {
		limit = {
			NOT = { this = ROOT }
			covert_action_valid_target = yes
			ROOT = { covert_target_ahead_in_space = { TARGET = PREV } }
		}
		save_scope_as = te_debug_covert_space_target
	}
	if = {
		limit = { exists = scope:te_debug_covert_space_target }
		scope:te_debug_covert_space_target = { save_scope_as = te_debug_covert_target }
		te_debug_covert_plant_op = {
			TYPE = space_espionage
			ACTION = covert_space_espionage_action
			DEFENSE_MOD = country_covert_defense_economic_add
			CODE = 11
			MONTHS = 14
		}
	}
}
