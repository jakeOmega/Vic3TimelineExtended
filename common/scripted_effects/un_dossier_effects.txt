# UN GROUNDS AND VOTES — THE DOSSIER, THE LEAN, THE AI BALLOT, THE RECESS
# ============================================================================
# Redesign phase 3 (docs/systems/un_redesign_design.md §5.1, §6, §8.3; as
# shipped, §0.3). The figures and the lean's terms are in
# common/script_values/un_dossier_values.txt, the tests in
# common/scripted_triggers/un_dossier_triggers.txt.
#
# WHO WRITES WHAT
#   un_dos_<record> (country)     un_dossier_record, and the monthly decay
#   un_dos_war_logged (country)   un_dossier_monthly_update (a war already
#                                 recorded, so a long war is recorded once)
#   un_dos_war_mandate_only (country)
#                                 un_dossier_monthly_update: the logged war is
#                                 the holder's bound mandate's own, so a war it
#                                 begins beside it is still recorded
#   un_lean_* (country)           un_lean_snapshot, in un_vote.4 only
#   un_vote_draw_res, un_vote_cast_month, un_lean_misread (country)
#                                 un_vote_ai_draws, in un_vote.4: an AI member's
#                                 casting month and misreading, once per resolution
#   un_res_why_* (resolution)     the reason tallies, from un_resolution_record_vote
#                                 and un_resolution_record_veto
#   un_res_case (resolution)      un_resolution_open: the target's case when tabled
#   un_res_last_call (resolution) un_vote.6, or the monthly pulse from month 12:
#                                 every AI member still without a vote casts now
#   un_res_ai_cast (resolution)   legacy: written before the late ballots; inert
#   un_recess_months (global)     un_resolution_archive sets it,
#                                 un_recess_monthly_update counts it down
#
# RECORD CALL SITES (un_dossier_record = { RECORD = x POINTS = n }):
#   aggression  un_dossier_monthly_update (a war begun outside a bound mandate's play)
#   defiance    un_vote.3 option C (a binding resolution refused),
#               un_events.5 option C (sanctions busted), un_events.8 option C
#               (the court defied)
#   violation   un_mandate_on_violated (un_mandate_effects.txt)
#   covert      covert_warfare.1's immediate (a severe operation exposed)
#   nuclear     nuclear_first_strike / _tactical_strike / _response_strike
#               (extra_effects.txt)
# ============================================================================

# ---- The dossier -------------------------------------------------------------------

# Add $POINTS$ to this country's $RECORD$, capped at un_dossier_record_cap. A
# no-op while no UN exists: the dossier is the Assembly's file on a country,
# and there is no Assembly to keep one.
# Parameters: $RECORD$ = aggression | defiance | violation | covert | nuclear
#             $POINTS$ = a number or a script value (evaluated in this scope)
# Scope: the country the record is against
un_dossier_record = {
	hidden_effect = {
		if = {
			limit = { has_global_variable = un_founded }
			if = {
				limit = { NOT = { has_variable = un_dos_$RECORD$ } }
				set_variable = { name = un_dos_$RECORD$ value = 0 }
			}
			change_variable = { name = un_dos_$RECORD$ add = $POINTS$ }
			if = {
				limit = { var:un_dos_$RECORD$ > un_dossier_record_cap }
				set_variable = { name = un_dos_$RECORD$ value = un_dossier_record_cap }
			}
		}
	}
}

# Scope: country
un_dossier_decay_record = {
	if = {
		limit = { has_variable = un_dos_$RECORD$ }
		change_variable = { name = un_dos_$RECORD$ multiply = un_dossier_decay_factor }
		if = {
			limit = { var:un_dos_$RECORD$ < un_dossier_floor }
			remove_variable = un_dos_$RECORD$
		}
	}
}

# Global monthly upkeep (no-country scope; from un_global_authority_on_action,
# UN founded). NOT IDEMPOTENT WITHIN A MONTH: the records decay and the recess
# counts down on every call.
#   1. Every record fades.
#   2. A war begun without a mandate is recorded once, when it is first seen:
#      the initiator of a war it is fighting. The flag un_dos_war_logged holds
#      the war as recorded until the country is at peace again. A bound
#      mandate excuses only its own war (un_mandate_began_other_war): when
#      that is the only war the holder began, the war is logged unrecorded
#      with un_dos_war_mandate_only, and a war the holder begins beside it
#      while the mandate is bound is recorded then. Until 2026-10-02 a bound
#      mandate excused every war its holder began.
#   3. The recess counts down.
un_dossier_monthly_update = {
	every_country = {
		limit = { un_dossier_has_records = yes }
		un_dossier_decay_record = { RECORD = aggression }
		un_dossier_decay_record = { RECORD = defiance }
		un_dossier_decay_record = { RECORD = violation }
		un_dossier_decay_record = { RECORD = covert }
		un_dossier_decay_record = { RECORD = nuclear }
	}
	every_country = {
		limit = {
			has_variable = un_dos_war_logged
			OR = {
				is_at_war = no
				is_diplomatic_play_initiator = no
			}
		}
		remove_variable = un_dos_war_logged
		remove_variable = un_dos_war_mandate_only
	}
	every_country = {
		limit = {
			is_at_war = yes
			is_diplomatic_play_initiator = yes
			OR = {
				NOT = { has_variable = un_dos_war_logged }
				has_variable = un_dos_war_mandate_only
			}
		}
		if = {
			limit = { NOT = { has_variable = un_dos_war_logged } }
			set_variable = un_dos_war_logged
			if = {
				limit = {
					var:un_mandate_current ?= { has_tag = un_mandate_bound }
					NOT = { un_mandate_began_other_war = yes }
				}
				set_variable = un_dos_war_mandate_only
			}
			else = {
				un_dossier_record = { RECORD = aggression POINTS = un_dos_points_aggression }
			}
		}
		# Logged as the mandate's war alone. Only while the mandate is bound:
		# once it closes (complied mid-war by its goal's timer, say), its own
		# war no longer reads as the mandate's and would be recorded.
		else_if = {
			limit = {
				var:un_mandate_current ?= { has_tag = un_mandate_bound }
				un_mandate_began_other_war = yes
			}
			remove_variable = un_dos_war_mandate_only
			un_dossier_record = { RECORD = aggression POINTS = un_dos_points_aggression }
		}
	}
	un_recess_monthly_update = yes
}

# ---- The recess ----------------------------------------------------------------------

# Count the recess down; when it ends, tell the human members the floor is open.
# Scope: any
un_recess_monthly_update = {
	if = {
		limit = { has_global_variable = un_recess_months }
		change_global_variable = { name = un_recess_months add = -1 }
		if = {
			limit = { global_var:un_recess_months <= 0 }
			remove_global_variable = un_recess_months
			every_country = {
				limit = {
					is_ai = no
					je:je_united_nations ?= { has_modifier = un_member_modifier }
				}
				post_notification = un_floor_open_notice
			}
		}
	}
}

# ---- The lean --------------------------------------------------------------------------

# Write this voter's lean on scope:un_resolution, term by term, as country
# variables the chamber prints (un_lean_total and one variable per term), and
# note which resolution they belong to. The ONLY writer of those variables.
# Called from un_vote.4, whose ROOT is the voter, because the relations terms
# read relations:root.
# Scope: the voter (= ROOT), with scope:un_resolution, scope:un_vote_proposer
# and scope:un_vote_target saved
un_lean_snapshot = {
	set_variable = { name = un_lean_res value = scope:un_resolution }
	set_variable = { name = un_lean_grounds value = un_lean_grounds }
	set_variable = { name = un_lean_self value = un_lean_self }
	set_variable = { name = un_lean_proposer value = un_lean_proposer }
	set_variable = { name = un_lean_target value = un_lean_target }
	set_variable = { name = un_lean_glass value = un_lean_glass }
	set_variable = { name = un_lean_standing value = un_lean_standing }
	set_variable = { name = un_lean_pledge value = un_lean_pledge }
	set_variable = { name = un_lean_accepted value = un_lean_accepted }
	set_variable = { name = un_lean_burden value = un_lean_burden }
	set_variable = { name = un_lean_interests value = un_lean_interests }
	set_variable = { name = un_lean_lobbying value = un_lean_lobbying }
	set_variable = { name = un_lean_total value = un_vote_lean }
	# Derived from the two lines above (un_lobbying_values.txt): the lean the
	# other members read, misreading included, and the lean the veto is
	# decided on, without the pledge.
	set_variable = { name = un_lean_shown value = un_lean_shown_value }
	set_variable = { name = un_lean_veto_basis value = un_lean_veto_basis_value }
}

# An AI member's casting month and misreading on scope:un_resolution, drawn
# once per resolution (docs/superpowers/specs/2026-09-27-un-vote-lobbying-design.md
# §1). It votes at the monthly pulse of its month, by the lean it has then; the
# chamber shows the other members its lean plus the misreading, as a band.
# Drawn at its first un_vote.4 of the session: on the day the resolution opens
# for a member at that time, later for a member that joins mid-session or a
# resolution already open when the save was loaded — which then votes no
# earlier than next month.
# Scope: the AI member (= ROOT, un_vote.4), with scope:un_resolution saved
un_vote_ai_draws = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = un_vote_draw_res }
				NOT = { var:un_vote_draw_res ?= scope:un_resolution }
			}
		}
		set_variable = { name = un_vote_draw_res value = scope:un_resolution }
		random_list = {
			1 = { set_variable = { name = un_vote_cast_month value = 9 } }
			1 = { set_variable = { name = un_vote_cast_month value = 10 } }
			1 = { set_variable = { name = un_vote_cast_month value = 11 } }
		}
		random_list = {
			1 = { set_variable = { name = un_lean_misread value = -10 } }
			1 = { set_variable = { name = un_lean_misread value = 0 } }
			1 = { set_variable = { name = un_lean_misread value = 10 } }
		}
		if = {
			limit = { var:un_vote_cast_month < un_res_next_month }
			set_variable = { name = un_vote_cast_month value = un_res_next_month }
		}
	}
}

# The AI's ballot: the lean, plus noise, decides. A permanent member vetoes a
# binding resolution when its lean without the pledge term (and without noise)
# is at or below the veto line (un_vote_ai_should_veto); otherwise it votes yes
# when lean + noise > 0. The same three effects a human's ballot runs
# (un_vote_cast_effects.txt) record the vote, so the tally, lobbying
# settlement and IG reactions are identical. Cast from un_vote.4 at the
# member's casting month (un_vote_ai_cast_due), never earlier.
# Scope: the voter (= ROOT), after un_lean_snapshot, with the same saved scopes
un_vote_ai_cast = {
	set_variable = { name = un_lean_roll value = var:un_lean_total }
	random_list = {
		20 = { change_variable = { name = un_lean_roll add = -20 } }
		20 = { change_variable = { name = un_lean_roll add = -10 } }
		20 = { }
		20 = { change_variable = { name = un_lean_roll add = 10 } }
		20 = { change_variable = { name = un_lean_roll add = 20 } }
	}
	if = {
		limit = { un_vote_ai_should_veto = yes }
		un_vote_cast_veto = yes
	}
	else_if = {
		limit = { var:un_lean_roll > 0 }
		un_vote_cast_yes = yes
	}
	else = {
		un_vote_cast_no = yes
	}
	remove_variable = un_lean_roll
}

# Send every member its un_vote.4: a human member has its lean written for the
# chamber; an AI member draws its timing, has its lean written, and casts if
# its month has come (or the last call has been made). From the day the
# resolution opens, every monthly pulse while it votes, and the last call
# (un_vote.6). Only while a resolution is voting, so the fan-out is at most
# a year of monthly events per resolution.
# Scope: any, with scope:un_resolution saved
un_vote_refresh_leans = {
	every_country = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		trigger_event = { id = un_vote.4 }
	}
	# The next day, with every snapshot and cast in, the chamber's Delegations
	# rows are sorted once (un_vote.7). Only a human country ever looks at
	# them, so it goes to each human country.
	every_country = {
		limit = { is_ai = no }
		trigger_event = { id = un_vote.7 days = 1 }
	}
}

# ---- The chamber's Delegations rows (spec §6) -------------------------------------------
# Every AI member with a vote that has not voted is on the resolution's
# un_res_delegates, with a sort key (un_deleg_key, un_lobbying_values.txt)
# from its latest snapshot. The day after each fan-out (un_vote.7, sent by
# un_vote_refresh_leans to each human country, the only ones that look), the
# resolution re-assigns its 24 rows, un_deleg_row_0..23, one ordered_in_list
# position each, so the rows and their buttons read the same member; the
# summary line's counts are written at the same time. Between sorts the order
# stands: a campaign started mid-month moves its member up at the next pulse,
# and a member that votes drops out of its row at once (un_deleg_listed).

# Scope: the AI member (= ROOT, un_vote.4), after the snapshot and any cast,
# with scope:un_resolution saved
un_deleg_refresh_self = {
	if = {
		limit = { un_vote_can_cast_ballot = yes }
		set_variable = { name = un_deleg_key value = un_deleg_key_value }
		scope:un_resolution = {
			if = {
				limit = { NOT = { un_lobby_in_list = { LIST = un_res_delegates COUNTRY = ROOT } } }
				add_to_variable_list = { name = un_res_delegates target = ROOT }
			}
		}
	}
	else = {
		scope:un_resolution = {
			if = {
				limit = { un_lobby_in_list = { LIST = un_res_delegates COUNTRY = ROOT } }
				remove_list_variable = { name = un_res_delegates target = ROOT }
			}
		}
	}
}

# Parameters: $N$ = 0-23
# Scope: resolution container
un_deleg_clear_row = {
	if = {
		limit = { has_variable = un_deleg_row_$N$ }
		remove_variable = un_deleg_row_$N$
	}
}

# Row $N$ (0-based) gets the member at position $N$ of the rows' order, if
# there are $COUNT$ ($N$ + 1) to show. The guard keeps a position past the end
# from leaving the previous row's pick in the temporary scope.
# Parameters: $N$, $COUNT$
# Scope: resolution container
un_deleg_assign_row = {
	if = {
		limit = {
			has_variable_list = un_res_delegates
			any_in_list = {
				variable = un_res_delegates
				count >= $COUNT$
				un_deleg_in_rows = yes
			}
		}
		ordered_in_list = {
			variable = un_res_delegates
			limit = { un_deleg_in_rows = yes }
			order_by = un_deleg_order
			position = $N$
			check_range_bounds = no
			save_temporary_scope_as = un_deleg_pick
		}
		set_variable = { name = un_deleg_row_$N$ value = scope:un_deleg_pick }
	}
}

# ---- Why they voted: the reason tallies -------------------------------------------------
# When a ballot is recorded, each term of the voter's lean that pushed it the way
# it voted, by at least un_lean_reason_size, is counted on the resolution, so
# the chamber can say "Opposed: ties to the target (12 members)". The voter's
# snapshot must be for this resolution (un_lean_res); a ballot cast with none
# (a vote from a save made before phase 3) counts no reasons.
# Called as un_lean_tally_$LIST$ from un_resolution_record_vote (so the list
# name picks the side) and from un_resolution_record_veto.
# Scope: the voter, with scope:un_resolution saved

un_lean_tally_un_res_yes = {
	if = {
		limit = {
			has_variable = un_lean_res
			var:un_lean_res ?= scope:un_resolution
		}
		if = {
			limit = { var:un_lean_grounds >= un_lean_reason_size }
			un_lean_count_reason = { NAME = un_res_why_yes_grounds }
		}
		if = {
			limit = { var:un_lean_proposer >= un_lean_reason_size }
			un_lean_count_reason = { NAME = un_res_why_yes_proposer }
		}
		if = {
			limit = { var:un_lean_target >= un_lean_reason_size }
			un_lean_count_reason = { NAME = un_res_why_yes_target }
		}
		if = {
			limit = { var:un_lean_pledge >= un_lean_reason_size }
			un_lean_count_reason = { NAME = un_res_why_yes_pledge }
		}
		if = {
			limit = { var:un_lean_accepted >= un_lean_reason_size }
			un_lean_count_reason = { NAME = un_res_why_yes_accepted }
		}
		if = {
			limit = { var:un_lean_interests >= un_lean_reason_size }
			un_lean_count_reason = { NAME = un_res_why_yes_interests }
		}
		if = {
			limit = {
				has_variable = un_lean_lobbying
				var:un_lean_lobbying >= un_lean_reason_size
			}
			un_lean_count_reason = { NAME = un_res_why_yes_lobbying }
		}
	}
}

un_lean_tally_un_res_no = {
	if = {
		limit = {
			has_variable = un_lean_res
			var:un_lean_res ?= scope:un_resolution
		}
		if = {
			limit = { var:un_lean_grounds <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_grounds }
		}
		if = {
			limit = { var:un_lean_self <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_self }
		}
		if = {
			limit = { var:un_lean_pledge <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_pledge }
		}
		if = {
			limit = { var:un_lean_proposer <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_proposer }
		}
		if = {
			limit = { var:un_lean_target <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_target }
		}
		if = {
			limit = { var:un_lean_glass <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_glass }
		}
		if = {
			limit = { var:un_lean_burden <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_burden }
		}
		if = {
			limit = { var:un_lean_interests <= un_lean_reason_size_negative }
			un_lean_count_reason = { NAME = un_res_why_no_interests }
		}
		if = {
			limit = {
				has_variable = un_lean_lobbying
				var:un_lean_lobbying <= un_lean_reason_size_negative
			}
			un_lean_count_reason = { NAME = un_res_why_no_lobbying }
		}
	}
}

# Scope: any, with scope:un_resolution saved
un_lean_count_reason = {
