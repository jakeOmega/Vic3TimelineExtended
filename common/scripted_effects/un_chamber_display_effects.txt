# ============================================================================
# UN CHAMBER — DISPLAY-ONLY TOOLTIP BUILDERS
# ============================================================================
# Every effect in this file is READ-ONLY. They contain nothing but
# custom_tooltip / custom_tooltip_no_bullet lines and scope navigation, and
# exist purely so gui/journal_entry_widgets/un_chamber_widget.gui can render
# the General Assembly's state. They are called from
# common/scripted_guis/un_chamber_sguis.txt through
# `[GetScriptedGui('...').ExecuteTooltip(...)]`, which renders an effect
# without executing it (vanilla's documented "SGUIs to build lists in loc"
# pattern — see game/common/scripted_guis/scripted_guis.md and
# game/common/scripted_guis/journal_entry_sguis.txt).
#
# NOTHING HERE MAY WRITE GAMEPLAY STATE. Resolution creation, vote recording,
# the pass/fail decision, archiving and pruning all live in
# common/scripted_effects/un_vote_effects.txt and are untouched by this file.
#
# Scope for every effect below: a UN resolution script container (the tags and
# variables are documented at the top of un_vote_effects.txt).
#
# Why tooltip text rather than a GUI datamodel over the containers: the
# resolution containers store *countries* (un_res_proposer / un_res_target /
# un_res_vetoer, and the un_res_yes / un_res_no voter lists). The GUI chain
# `Var('x').GetCountry.GetName` is recorded as not working in this mod
# (docs/guides/scripting_best_practices.md "Saved Scopes vs Variables in
# Localization", gui_modding_guide.md gotcha #11) — which is exactly why the
# covert-operations widget stores a *state* (iw_target_capital) instead.
# Navigating to the country in script and letting loc print the name is the
# shape vanilla itself ships for this case, in two flavours, both used here:
#   - a list, printed per entry as `[THIS.GetCountry.GetName]` —
#     je_hispanoamerica_not_recognized_countries_sgui +
#     HISPANOAMERICA_RECOGNITION_COUNTRIES_LIST_ENTRY (ip4_spain_l_english.yml),
#     which also supplies the AddLocalizationIf(...Exists) annexed-country guard;
#   - a single country, stashed with save_temporary_scope_as and printed as
#     `[SCOPE.sCountry('...').GetName]` from the *container's* own line —
#     je_india_railway_*_sgui + INDIAN_RAILWAY_ENTRY (ip2_misc_02_l_english.yml).
#     Preferred for the singletons (proposer / target / vetoer) because the
#     tooltip line stays at container scope: wrapping a custom_tooltip in a
#     country scope-change block risks the engine prefixing the line with a
#     redundant "<Country>:" header.
# ============================================================================

# The territory and beneficiary a military_mandate resolution names. Printed
# under the parties line, and only for that topic — every other topic leaves
# both variables unset, so the `has_variable` guard doubles as the topic test.
# Scope: resolution container
un_chamber_mandate_terms_line = {
	if = {
		limit = {
			has_variable = un_res_mandate_region
			exists = var:un_res_mandate_region
		}
		var:un_res_mandate_region ?= { save_temporary_scope_as = un_chamber_res_region }
		custom_tooltip_no_bullet = je_un_chamber_res_mandate_territory
	}
	if = {
		limit = {
			has_variable = un_res_mandate_beneficiary
			exists = var:un_res_mandate_beneficiary
		}
		var:un_res_mandate_beneficiary ?= { save_temporary_scope_as = un_chamber_res_beneficiary }
		custom_tooltip_no_bullet = je_un_chamber_res_mandate_beneficiary
	}
	# A divided region: while the vote is open, say which part is at stake.
	if = {
		limit = {
			has_tag = un_res_voting
			exists = var:un_res_mandate_region
			exists = var:un_res_target
			exists = var:un_res_mandate_beneficiary
		}
		var:un_res_mandate_region = { save_temporary_scope_as = un_chamber_split_region }
		var:un_res_target = { save_temporary_scope_as = un_chamber_split_target }
		var:un_res_mandate_beneficiary = { save_temporary_scope_as = un_chamber_split_holder }
		un_chamber_mandate_split_lines = yes
	}
}

# A mandate names a whole state region, but claims cover a whole region too,
# so in a split region the beneficiary may already hold part of it, and the
# target may claim that part in turn: the holders of a split region usually
# claim each other's part (the previous owner of a conquered incorporated
# state gets a claim on its region). Say so, so "Territory named: Ceylon, to
# be restored to India" is not read as India's own Ceylon. Found in play,
# 2026-09-25: the rule is right, the card was ambiguous.
# Expects temporary scopes un_chamber_split_region, un_chamber_split_target
# and un_chamber_split_holder (the beneficiary).
# Scope: any
un_chamber_mandate_split_lines = {
	if = {
		limit = {
			exists = scope:un_chamber_split_region
			exists = scope:un_chamber_split_target
			exists = scope:un_chamber_split_holder
			scope:un_chamber_split_region = {
				any_scope_state = { owner = scope:un_chamber_split_holder }
			}
		}
		custom_tooltip_no_bullet = je_un_chamber_mandate_split
		if = {
			limit = {
				scope:un_chamber_split_region = {
					any_scope_state = {
						owner = scope:un_chamber_split_holder
						has_claim_by = scope:un_chamber_split_target
					}
				}
			}
			custom_tooltip_no_bullet = je_un_chamber_mandate_split_mutual
		}
	}
}

# Proposer and (when the topic names one) target. Both reads are guarded
# twice: `exists = var:X` catches a reference that no longer resolves, and the
# loc line's AddLocalizationIf(SCOPE.sCountry('...').Exists, ...) catches a
# country object that survives as a dead tag after annexation.
# Scope: resolution container
un_chamber_parties_line = {
	if = {
		limit = { exists = var:un_res_proposer }
		var:un_res_proposer ?= { save_temporary_scope_as = un_chamber_proposer }
		custom_tooltip_no_bullet = je_un_chamber_proposer
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_proposer_gone
	}
	if = {
		limit = { has_variable = un_res_target }
		if = {
			limit = { exists = var:un_res_target }
			var:un_res_target ?= { save_temporary_scope_as = un_chamber_target }
			custom_tooltip_no_bullet = je_un_chamber_target
		}
		else = {
			custom_tooltip_no_bullet = je_un_chamber_target_gone
		}
	}
	if = {
		limit = { has_tag = un_res_target_accepted }
		custom_tooltip_no_bullet = je_un_chamber_target_accepted
	}
}

# Vote tally plus the passage rule the resolve step actually applies.
# Mirrors un_resolution_decide (un_vote_effects.txt): expulsion and charter
# reform (un_resolution_needs_supermajority) need un_vote_supermajority_passed
# >= 0 (a 2/3 supermajority of all current members); every other topic needs
# un_res_margin >= 1 (a simple majority of the votes cast). The same trigger and
# the same two script values are evaluated here, so the projection line can
# never drift from the rule.
# Scope: resolution container
un_chamber_tally_line = {
	custom_tooltip_no_bullet = je_un_chamber_tally
	# Votes pledged but not yet cast, stated next to the tally and kept out of
	# it. un_res_commit_pending is a variable rather than a list size because
	# loc can print a variable and cannot print a list size; it is decremented
	# at the same single site that settles the commitment.
	# Only while the vote is in session: on a closed resolution in the archive
	# an unsettled pledge is a promise nobody ever got to keep, and "N more are
	# expected to vote" would be a lie. The Commitments list below still shows
	# those entries, worded so it reads correctly either way.
	if = {
		limit = {
			has_tag = un_res_voting
			has_variable = un_res_commit_pending
			var:un_res_commit_pending >= 1
		}
		custom_tooltip_no_bullet = je_un_chamber_tally_committed
	}
	if = {
		limit = { un_resolution_needs_supermajority = yes }
		custom_tooltip_no_bullet = je_un_chamber_rule_supermajority
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_rule_majority
	}
}

# Live projection of the same pass test, for a vote still in session.
# Scope: resolution container
un_chamber_projection_line = {
	if = {
		limit = { un_resolution_needs_supermajority = yes }
		if = {
			limit = { un_vote_supermajority_passed >= 0 }
			custom_tooltip_no_bullet = je_un_chamber_projection_pass
		}
		else = {
			custom_tooltip_no_bullet = je_un_chamber_projection_fail
		}
	}
	else_if = {
		limit = { un_res_margin >= 1 }
		custom_tooltip_no_bullet = je_un_chamber_projection_pass
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_projection_fail
	}
}

# Final outcome of a closed resolution. A veto is orthogonal to pass/fail —
# un_resolution_decide has no veto branch, and un_vote.2 carries dedicated
# "passed + vetoed" descriptions (events/un_vote_events.txt:1616-1628,
# 1665-1682) for the graduated forms. Reform is the one binding topic with no
# graduated fallback, so a vetoed reform that carried the Assembly is still
# blocked outright (un_vote.2.d_reform_passed_vetoed).
# `un_res_lapsed` means un_vote.2 never fired and the watchdog in
# un_resolutions_monthly_update closed the vote after 14 months.
# Scope: resolution container (closed)
un_chamber_outcome_line = {
	if = {
		limit = { has_tag = un_res_lapsed }
		custom_tooltip_no_bullet = je_un_chamber_outcome_lapsed
	}
	else_if = {
		limit = {
			has_tag = un_res_passed
			has_tag = un_res_vetoed
			has_tag = un_topic_reform
		}
		custom_tooltip_no_bullet = je_un_chamber_outcome_passed_vetoed_reform
	}
	else_if = {
		limit = {
			has_tag = un_res_passed
			has_tag = un_res_vetoed
			has_tag = un_topic_military_mandate
		}
		custom_tooltip_no_bullet = je_un_chamber_outcome_passed_vetoed_mandate
	}
	else_if = {
		limit = {
			has_tag = un_res_passed
			has_tag = un_res_vetoed
		}
		custom_tooltip_no_bullet = je_un_chamber_outcome_passed_vetoed
	}
	else_if = {
		limit = { has_tag = un_res_passed }
		custom_tooltip_no_bullet = je_un_chamber_outcome_passed
	}
	else_if = {
		limit = {
			has_tag = un_res_failed
			has_tag = un_res_vetoed
		}
		custom_tooltip_no_bullet = je_un_chamber_outcome_failed_vetoed
	}
	else_if = {
		limit = { has_tag = un_res_failed }
		custom_tooltip_no_bullet = je_un_chamber_outcome_failed
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_outcome_unknown
	}
	if = {
		limit = { has_tag = un_res_vetoed }
		if = {
			limit = { exists = var:un_res_vetoer }
			var:un_res_vetoer ?= { save_temporary_scope_as = un_chamber_vetoer }
			custom_tooltip_no_bullet = je_un_chamber_vetoed_by
		}
		else = {
			custom_tooltip_no_bullet = je_un_chamber_vetoed_unknown
		}
	}
}

# The recorded ballot, followed by the lobbying commitments made on the same
# resolution. Every list is iterated in place — nothing is copied into a
# country or global variable. Entries whose country has since been annexed
# print the "a former member" fallback instead of a broken name.
#
# Called from un_chamber_active_voters_sgui (the resolution in session) and
# from un_chamber_history_details_sgui (the archive's voting details), so the
# commitments appear in both without a second builder.
# Scope: resolution container
un_chamber_ballot_lines = {
	# Tooltip generation prints a scope's own lines first and its nested
	# country blocks afterwards, so a "In favour:" header followed by a list
	# renders as every header on top and every country underneath (seen in the
	# first play test). Each entry therefore says which side it is on, and the
	# only line printed at this level is one sentence of introduction.
	custom_tooltip_no_bullet = je_un_chamber_ballot_intro
	if = {
		limit = { has_variable_list = un_res_yes }
		every_in_list = {
			variable = un_res_yes
			custom_tooltip = je_un_chamber_voter_entry_yes
		}
	}
	if = {
		limit = { has_variable_list = un_res_no }
		every_in_list = {
			variable = un_res_no
			custom_tooltip = je_un_chamber_voter_entry_no
		}
	}
	un_chamber_commitment_lines = yes
}

# Vote commitments secured by the resolution's proposer — SEPARATE from the two
# ballot lists above, and never folded into them: a commitment is a promise, a
# vote is a ballot, and the chamber has to be able to tell the player apart.
# Nothing here is counted by un_chamber_tally_line's figures or by
# un_chamber_projection_line, both of which read un_res_support / un_res_oppose
# and therefore see votes actually cast and nothing else.
#
# The three lists are disjoint by construction (un_lobby_settle_commitment moves
# a recipient off the pending list as it adds it to a settled one), so the
# entries below never repeat. Country names print through the same guarded
# accessor the ballot uses, so a recipient annexed since it pledged reads as
# "a former member".
#
# The whole block is hidden unless a commitment was ever made on this
# resolution, which is also what keeps it invisible on every resolution in a
# save that predates the feature.
# Scope: resolution container
un_chamber_commitment_lines = {
	if = {
		limit = { un_lobby_resolution_has_commitments = yes }
		if = {
			limit = { un_lobby_list_has_entries = { LIST = un_res_committed_yes } }
			every_in_list = {
				variable = un_res_committed_yes
				custom_tooltip = je_un_chamber_commitment_pending
			}
		}
		if = {
			limit = { un_lobby_list_has_entries = { LIST = un_res_commit_kept } }
			every_in_list = {
				variable = un_res_commit_kept
				custom_tooltip = je_un_chamber_commitment_kept
			}
		}
		if = {
			limit = { un_lobby_list_has_entries = { LIST = un_res_commit_broken } }
			every_in_list = {
				variable = un_res_commit_broken

# The chamber's record of authorized military mandates. Same contract as the
# rest of this file: read-only, built out of custom_tooltip lines and scope

# (global list un_mandate_registry) is read in place — no index, length or cap
# is assumed, so the monthly pruner in un_mandate_effects.txt is invisible here
# and a save that has never issued a mandate simply shows the empty state.

# Countries are printed the same way the resolution lines print them: navigated
# to in script, stashed with save_temporary_scope_as, and rendered from the
# container's own line as [SCOPE.sCountry('...').GetName] behind an
# AddLocalizationIf(...Exists, ...) guard, so an actor or target annexed since
# the mandate was issued reads as "a former member" rather than a broken name.
# The territory is a state_region, which never stops existing, and prints
# through [SCOPE.sStateRegion('...').GetName].

# One mandate, as rendered in the register.
# Scope: mandate container
un_chamber_mandate_entry = {
	custom_tooltip_no_bullet = je_un_chamber_mandate_header
	if = {
		limit = { exists = var:un_mnd_actor }
		var:un_mnd_actor ?= { save_temporary_scope_as = un_chamber_mnd_actor }
		custom_tooltip_no_bullet = je_un_chamber_mandate_actor
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_mandate_actor_gone
	}
	if = {
		limit = { exists = var:un_mnd_target }
		var:un_mnd_target ?= { save_temporary_scope_as = un_chamber_mnd_target }
		custom_tooltip_no_bullet = je_un_chamber_mandate_target
	}
	else = {
		custom_tooltip_no_bullet = je_un_chamber_mandate_target_gone
	}
	if = {
		limit = { exists = var:un_mnd_region }
		var:un_mnd_region ?= { save_temporary_scope_as = un_chamber_mnd_region }
		custom_tooltip_no_bullet = je_un_chamber_mandate_territory
	}
	if = {
		limit = { exists = var:un_mnd_beneficiary }
		var:un_mnd_beneficiary ?= { save_temporary_scope_as = un_chamber_mnd_beneficiary }
		custom_tooltip_no_bullet = je_un_chamber_mandate_beneficiary
	}
	# A divided region (un_chamber_mandate_split_lines), while the mandate is live.
	if = {
		limit = {
			OR = {
				has_tag = un_mandate_active
				has_tag = un_mandate_bound
			}
			exists = var:un_mnd_region
			exists = var:un_mnd_target
			exists = var:un_mnd_beneficiary
		}
		var:un_mnd_region = { save_temporary_scope_as = un_chamber_split_region }
		var:un_mnd_target = { save_temporary_scope_as = un_chamber_split_target }
		var:un_mnd_beneficiary = { save_temporary_scope_as = un_chamber_split_holder }
		un_chamber_mandate_split_lines = yes
	}
	# Status. Exactly one of these tags is set at a time; the live pair comes
	# first so a closed mandate can never be read as still standing.
	if = {
		limit = { has_tag = un_mandate_active }
		custom_tooltip_no_bullet = je_un_chamber_mandate_status_active
		custom_tooltip_no_bullet = je_un_chamber_mandate_expiry
	}
	else_if = {
		limit = { has_tag = un_mandate_bound }
		custom_tooltip_no_bullet = je_un_chamber_mandate_status_bound
		if = {
			limit = { has_tag = un_mandate_forfeit }
			custom_tooltip_no_bullet = je_un_chamber_mandate_forfeit
		}
	}
	else_if = {
		limit = { has_tag = un_mandate_complied }
		custom_tooltip_no_bullet = je_un_chamber_mandate_status_complied
	}
	else_if = {
		limit = { has_tag = un_mandate_violated }
		custom_tooltip_no_bullet = je_un_chamber_mandate_status_violated
	}
	else_if = {
		limit = { has_tag = un_mandate_abandoned }
		custom_tooltip_no_bullet = je_un_chamber_mandate_status_abandoned
	}
	else_if = {
		limit = { has_tag = un_mandate_expired }
		custom_tooltip_no_bullet = je_un_chamber_mandate_status_expired
	}
	else = {

# The recorded ballot of the resolution in session, each voter with the lean
# it voted by where its snapshot is for this resolution: what the target (and
# everyone else) most needs to see, why each member voted as it did. The
# archive keeps un_chamber_ballot_lines, which prints no lean: a closed
# resolution's voters have since been snapshotted for other business.
# Scope: resolution container (the one in session)
un_chamber_ballot_lines_active = {
	custom_tooltip_no_bullet = je_un_chamber_ballot_intro
	if = {
		limit = { has_variable_list = un_res_yes }
		every_in_list = {
			variable = un_res_yes
			if = {
				limit = {
					has_variable = un_lean_res
					var:un_lean_res ?= PREV
				}
				custom_tooltip = je_un_chamber_voter_entry_yes_lean
			}
			else = {
				custom_tooltip = je_un_chamber_voter_entry_yes
			}
		}
	}
	if = {
		limit = { has_variable_list = un_res_no }
		every_in_list = {
			variable = un_res_no
			if = {
				limit = {
					has_variable = un_lean_res
					var:un_lean_res ?= PREV
				}
				custom_tooltip = je_un_chamber_voter_entry_no_lean
			}
			else = {
				custom_tooltip = je_un_chamber_voter_entry_no
			}
		}
	}
	un_chamber_commitment_lines = yes
}
