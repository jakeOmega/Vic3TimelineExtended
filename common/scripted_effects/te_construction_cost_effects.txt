# Construction Cost Scaling's yearly reading of a country's slack, and its self-check.
# Values and the constants: common/script_values/extra_script_values.txt (CONSTRUCTION COST SCALING). Wiring:
# construction_cost_scaling_on_action (common/on_actions/extra_on_actions.txt), which calls te_cc_yearly_update
# before it sets the modifier. Why: docs/systems/economy_growth_sim.md (#1034).

# THIS = a country. Set the scaling's modifier for the construction model the game is in. From the yearly pulse
# (construction_cost_scaling_on_action, after te_cc_yearly_update), and from a switch of
# free_market_construction_rule in a running game (te_construction_market_follow_rule), right after
# te_direct_construction_clear_cost_scaling has removed both modifiers and the variable: without that call the
# market's price and goods per point (te_construction_market_price, te_construction_market_goods_per_point), which
# multiply by construction_cost_scaling_mult, would run up to a year ahead of what the sites consume.
#
# free_market_construction_rule disabled: there is no construction good to scale, so the scaling goes on vanilla's
# own lever for the cost of the goods construction consumes (construction_cost_scaling_direct). The two branches
# never share var:prior_construction_scaling: a switch clears it, so each branch starts from its first run, which
# reads no modifier the same effect has just removed.
#
# X (construction_cost_scaling_mult) is negative for a small economy under te_small_economy_construction_rule, so
# a modifier is added whenever its multiplier is not 0. The game-start pass calls this too
# (te_cc_game_start_country). The multipliers resolve against ROOT, so every caller has the country as ROOT.
te_cc_apply_scaling = {
	if = {
		limit = { te_free_market_construction_off = yes }
		# Compute before removing, so the modifier read certainly still holds
		# the contribution var:prior_construction_scaling backs out.
		set_variable = { name = prior_construction_scaling value = construction_cost_scaling_direct_adjusted_mult }
		remove_modifier = construction_cost_scaling_direct
		if = {
			limit = { NOT = { var:prior_construction_scaling = 0 } }
			add_modifier = { name = construction_cost_scaling_direct multiplier = var:prior_construction_scaling }
		}
	}
	else = {
		remove_modifier = construction_cost_scaling
		if = {
			limit = { NOT = { has_variable = prior_construction_scaling } }
			# First run (new save or pre-adjustment saves) - use raw scaling
			if = {
				limit = { NOT = { construction_cost_scaling_mult = 0 } }
				add_modifier = { name = construction_cost_scaling multiplier = construction_cost_scaling_mult }
				set_variable = { name = prior_construction_scaling value = construction_cost_scaling_mult }
			}
		}
		else = {
			# Subsequent runs - use adjusted scaling (multiplicative with other modifiers)
			if = {
				limit = { NOT = { construction_cost_scaling_adjusted_mult = 0 } }
				add_modifier = { name = construction_cost_scaling multiplier = construction_cost_scaling_adjusted_mult }
				set_variable = { name = prior_construction_scaling value = construction_cost_scaling_adjusted_mult }
			}
			else = {
				set_variable = { name = prior_construction_scaling value = 0 }
			}
		}
	}
}

# THIS = a country, its yearly pulse (and the game-start pass, te_cc_game_start_country). Store its slack and
# peasants from its states' last walks (te_unemp_share_refresh) and move the share of the GDP part that applies
# (var:te_cc_slack_factor) a third of the way toward its target. A country none of whose states has been walked
# yet keeps the share it had (te_cc_slack_factor_now: on a save from before the rule, the share that reproduces
# the scaling in force). Then store the GDP per capita and the GDP part the scaling uses this year, for the Nation
# tab's tooltip, and move the scaling's own factor (var:te_cc_scale) toward one plus the GDP part times the share,
# by at most construction_cost_max_yearly_move of itself. A country with no stored factor first gets the scaling
# in force, read before the share moves, so a save from before the cap moves from where it stood. Last, the
# small-economy factor (var:te_cc_small_factor, 1 unless te_small_economy_construction_rule is on), which the cap
# does not touch. te_cc_had_reading tells the self-check whether the country had a reading before this one.
te_cc_yearly_update = {
	if = {
		limit = { has_variable = te_cc_slack }
		set_local_variable = { name = te_cc_had_reading value = 1 }
	}
	if = {
		limit = { NOT = { has_variable = te_cc_scale } }
		set_variable = { name = te_cc_scale value = te_cc_scale_target }
	}
	if = {
		limit = { te_country_walked_workforce > 0 }
		set_variable = { name = te_cc_slack value = te_country_slack_share }
		set_variable = { name = te_cc_peasants value = te_country_peasant_share }
		set_variable = { name = te_cc_slack_factor value = te_cc_slack_factor_eased }
	}
	set_variable = { name = te_cc_gdppc value = country_gdp_per_capita }
	set_variable = { name = te_cc_gdp_part value = construction_cost_scaling_gdp_mult }
	set_variable = { name = te_cc_scale value = te_cc_scale_next }
	set_variable = { name = te_cc_small_factor value = te_cc_small_economy_factor }
	te_cc_slack_selfcheck = yes
}

# Any scope, from on_game_started_after_lobby (te_construction_market_on_actions.txt), when the game rules are
# final. With te_small_economy_construction_rule on, every country gets the scaling at once, so a small economy
# builds cheaply from the first day and not from its first yearly pulse, up to a year on. ROOT is none there and
# add_modifier's multiplier resolves against ROOT, so each country is reached through a hidden country event
# (te_construction_market_recalc_events.3; scripting_best_practices.md, "add_modifier { multiplier = var:X }
# Resolves Against ROOT"). With the rule off there is nothing to apply in a new game: the scaling starts at 0.
te_cc_game_start = {
	if = {
		limit = { te_small_economy_construction_on = yes }
		every_country = {
			limit = {
				NOT = { is_country_type = decentralized }
			}
			trigger_event = { id = te_construction_market_recalc_events.3 }
		}
	}
}

# THIS = ROOT = a country, in te_construction_market_recalc_events.3. The yearly pulse's own steps, once, for a
# country the scaling has not reached yet. Its GDP must be known: the small-economy factor is read from it, and
# a GDP of 0 would give the largest discount to every country. Whether gdp reads before the first day is not
# established, so a country whose GDP reads 0 tries again a week later, and the self-check says which happened.
te_cc_game_start_country = {
	if = {
		limit = {
			te_small_economy_construction_on = yes
			NOT = { has_variable = te_cc_small_factor }
		}
		if = {
			limit = { gdp > 0 }
			te_cc_yearly_update = yes
			te_cc_apply_scaling = yes
			te_cc_small_start_selfcheck = yes
		}
		else = {
			set_variable = { name = te_cc_start_retried value = 1 }
			trigger_event = { id = te_construction_market_recalc_events.3 days = 7 }
		}
	}
}

# TE_SELF_CHECK small_economy_start (docs/testing/ai-economy-playtest.md, "Construction Cost Scaling: the yearly
# cap and the small-economy rule"). Once a game, the first country the game-start pass applies the small-economy
# factor to whose factor is below 1 (a GDP under construction_cost_small_economy_full_gdp, so the modifier is
# there to see) and above the minimum (so its GDP was not read as nothing).
#   OK      applied at the first try and the scaling it stored is a discount: gdp reads at game start, the
#           hidden event had the country as ROOT, and the cost is in force from the first day.
#   BROKEN  the country had to try again (gdp read 0 on the first day, so small economies paid full price for
#           the first week), or what it stored is not a discount.
te_cc_small_start_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_cc_small_start_check_done }
			var:te_cc_small_factor < 1
			var:te_cc_small_factor > construction_cost_small_economy_min_factor
		}
		set_global_variable = { name = te_cc_small_start_check_done value = yes }
		set_variable = { name = te_cc_chk_gdp value = gdp }
		if = {
			limit = {
				NOT = { has_variable = te_cc_start_retried }
				has_variable = prior_construction_scaling
				var:prior_construction_scaling < 0
			}
			debug_log = "TE_SELF_CHECK OK small_economy_start country=[THIS.GetCountry.GetTagName] gdp=[THIS.Var('te_cc_chk_gdp').GetValue|0] factor=[THIS.Var('te_cc_small_factor').GetValue|3] applied=[THIS.Var('prior_construction_scaling').GetValue|3]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN small_economy_start country=[THIS.GetCountry.GetTagName] gdp=[THIS.Var('te_cc_chk_gdp').GetValue|0] factor=[THIS.Var('te_cc_small_factor').GetValue|3] (the country tried again a week later, or what it stored is not a discount)"
		}
		debug_log_scopes = yes
		remove_variable = te_cc_chk_gdp
	}
	if = {
		limit = { has_variable = te_cc_start_retried }
		remove_variable = te_cc_start_retried
	}
}

# TE_SELF_CHECK construction_slack_read (docs/testing/ai-economy-playtest.md, Self-checks). Once a game, the first
# country at its yearly pulse with a reading at an earlier pulse (so every state has been walked at least once: no
# old save's half-walked first year), a walked workforce of at least 100,000 and a slack between 2% and 98%: the
# slack the pulse just stored (te_country_slack_share, summed from the states' stored counts) against a fresh walk
# of every pop in the country, counted by the same rule (te_unemp_count_pop).
#   OK      within 10 points, with some peasants in the fresh count: the states' counts are stored and summed as
#           meant (each is up to a year old: in a new game's first year, as the economy hires, that moved Orta
#           Zhuz's peasant share 5.3 points, and a dropped state or a peasant branch that never matches moves it
#           tens of points or to 0). The recount shares the walk's rule, so it can't catch a rule that never
#           matches; a peasant count of 0 can, since every country has subsistence farming, so it reads BROKEN.
#   BROKEN  anything else: the sum or the stored counts aren't what the walk counted, or the peasant branch never
#           matches. The fallback is in the playtest row.
te_cc_slack_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_cc_slack_check_done }
			has_local_variable = te_cc_had_reading
			has_variable = te_cc_slack
			te_country_walked_workforce >= 100000
			var:te_cc_slack >= 0.02
			var:te_cc_slack <= 0.98
		}
		set_global_variable = { name = te_cc_slack_check_done value = yes }
		set_local_variable = { name = te_unemp_w_pop value = 0 }
		set_local_variable = { name = te_unemp_w_wf value = 0 }
		set_local_variable = { name = te_unemp_w_idle value = 0 }
		set_local_variable = { name = te_unemp_w_slack value = 0 }
		set_local_variable = { name = te_unemp_w_peasant value = 0 }
		every_scope_state = {
			every_scope_pop = {
				te_unemp_count_pop = yes
			}
		}
		set_variable = {
			name = te_cc_chk_fresh
			value = { value = local_var:te_unemp_w_slack divide = { value = local_var:te_unemp_w_wf min = 1 } }
		}
		set_variable = {
			name = te_cc_chk_fresh_peasants
			value = { value = local_var:te_unemp_w_peasant divide = { value = local_var:te_unemp_w_wf min = 1 } }
		}
		set_variable = { name = te_cc_chk_wf value = local_var:te_unemp_w_wf }
		set_variable = { name = te_cc_chk_walked_wf value = te_country_walked_workforce }
		set_variable = { name = te_cc_chk_target value = te_cc_slack_target }
		set_variable = {
			name = te_cc_chk_diff
			value = { value = var:te_cc_slack subtract = var:te_cc_chk_fresh }
		}
		if = {
			limit = {
				var:te_cc_chk_diff <= 0.10
				var:te_cc_chk_diff >= -0.10
				var:te_cc_chk_fresh_peasants > 0
			}
			debug_log = "TE_SELF_CHECK OK construction_slack_read country=[THIS.GetCountry.GetTagName] slack=[THIS.Var('te_cc_slack').GetValue|3] fresh=[THIS.Var('te_cc_chk_fresh').GetValue|3] peasants=[THIS.Var('te_cc_peasants').GetValue|3] fresh_peasants=[THIS.Var('te_cc_chk_fresh_peasants').GetValue|3] walked_wf=[THIS.Var('te_cc_chk_walked_wf').GetValue|0] fresh_wf=[THIS.Var('te_cc_chk_wf').GetValue|0] target=[THIS.Var('te_cc_chk_target').GetValue|3] factor=[THIS.Var('te_cc_slack_factor').GetValue|3] gdppc=[THIS.Var('te_cc_gdppc').GetValue|2] gdp_part=[THIS.Var('te_cc_gdp_part').GetValue|3] (the states' stored counts sum to the country's slack, so the scaling reads the slack it means) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN construction_slack_read country=[THIS.GetCountry.GetTagName] slack=[THIS.Var('te_cc_slack').GetValue|3] fresh=[THIS.Var('te_cc_chk_fresh').GetValue|3] peasants=[THIS.Var('te_cc_peasants').GetValue|3] fresh_peasants=[THIS.Var('te_cc_chk_fresh_peasants').GetValue|3] walked_wf=[THIS.Var('te_cc_chk_walked_wf').GetValue|0] fresh_wf=[THIS.Var('te_cc_chk_wf').GetValue|0] target=[THIS.Var('te_cc_chk_target').GetValue|3] factor=[THIS.Var('te_cc_slack_factor').GetValue|3] gdppc=[THIS.Var('te_cc_gdppc').GetValue|2] gdp_part=[THIS.Var('te_cc_gdp_part').GetValue|3] (the stored slack isn't the pops' own, or no peasants were counted: take the fallback in docs/testing/ai-economy-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
		remove_variable = te_cc_chk_fresh
		remove_variable = te_cc_chk_fresh_peasants
		remove_variable = te_cc_chk_wf
		remove_variable = te_cc_chk_walked_wf
		remove_variable = te_cc_chk_target
		remove_variable = te_cc_chk_diff
	}
}
