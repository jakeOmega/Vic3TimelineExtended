# ============================================================================
# SOLAR SYSTEM COLONIZATION — fleets: combat, blockade, seizure, destruction
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-26-solar-colonization-fleets-design.md
# (slice 6b, §§ 7-11, 14; 6c, §§ 12-14; 6d, § 15). Full Colonies only.
#
# Two steps (2026-11-25 design, docs/superpowers/specs/
# 2026-11-25-space-fleet-clock-and-battles-design.md):
#   - the battle, on each of the fleets' ticks, four a month (sr_battle_at,
#     below: § 2): fleets of countries at war meet at a location and fight it
#     out within the tick, stances (§ 5.1), pace and ties deciding who fights
#     whom, each fleet breaking for the nearest other dock at the share its
#     stance keeps; the ground-to-orbit batteries fire once a tick at Earth;
#   - the siege, each month, from the global step after the 1st's battle
#     (sr_combat_at, with LOC and CODE: § 2.6):
#   0. every power's forces there by posture into country variables (sr_cb_*):
#      strength by posture (1 station, 2 stand off, 3 close in; 4 interdict,
#      6c, counts as standing off; at Earth every fleet counts as stationed
#      except one closing in on a lunar world (lunar colonization § 8.1), and
#      the Orbital Battlestation joins as stationed), and for each world its
#      closing-in fleets target, their strength, fire on defences, marines and
#      fire to destroy;
#   1. each world's closing-in strength (global sr_cb_c_<key>);
#   2. each power's hostile stationed strength (sr_cb_hs), which shares a
#      closing-in fleet's fire on Defences and bars a seizure;
#   3. the Defences' fire on the fleets closing in, shared by strength
#      (sr_cb_ld_<key>), applied slot by slot (screens first, half destroyed,
#      half damage: sr_fl_take_losses_at), debris at Earth;
#   4. each world: integrity, seizure, destruction, blockade (sr_site_*_step;
#      a lunar world's blockade follows its holder's Earth's, § 8.2);
#   5. the records cleared.
# Hostile means at war (has_war_with); a fleet closing in on a free colony is
# hostile to its Defences alone (§ 9.5). An interdiction (6c, § 12) fires no
# shot: it blockades the worlds of the country it interdicts, a country at
# peace with its interdictor (sr_icpt_<loc>, sr_site_blockade_step). The per-location and per-world blocks
# are generated (sr_fleets_generated.txt); the numbers are sr_fleet_values.txt's.
# ============================================================================

# ---- The siege's month (global scope) ------------------------------------------------------
sr_combat_global_monthly = {
	sr_combat_at = { LOC = earth CODE = 1 }
	sr_combat_at = { LOC = inner CODE = 2 }
	sr_combat_at = { LOC = mars CODE = 3 }
	sr_combat_at = { LOC = asteroids CODE = 4 }
	sr_combat_at = { LOC = jupiter CODE = 5 }
	sr_combat_at = { LOC = saturn CODE = 6 }
	sr_combat_at = { LOC = uranus CODE = 7 }
	sr_combat_at = { LOC = neptune CODE = 8 }
	sr_combat_at = { LOC = kuiper CODE = 9 }
}

sr_combat_at = {
	set_local_variable = { name = sr_cb_code value = $CODE$ }
	# 0. Forces.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { is_country_alive = yes }
		sr_cb_clear_$LOC$ = yes
		if = {
			limit = { has_variable = sr_fl_at_$LOC$ }
			sr_fl_each_slot = { DO = gather }
		}
		if = {
			limit = { local_var:sr_cb_code = 1 }
			sr_cb_battlestation_gather = yes
		}
	}
	# 1. Each world's closing-in strength.
	sr_cb_world_sums_$LOC$ = yes
	# 2. The hostiles stationed against each power.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_cb_present }
		set_variable = { name = sr_cb_hs value = sr_cb_hs_calc }
	}
	# 3. The Defences' fire on the fleets closing in; debris at Earth.
	set_local_variable = { name = sr_cb_debris value = 0 }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_cb_present }
		sr_cb_def_losses_$LOC$ = yes
		set_variable = { name = sr_cb_lost value = sr_cb_ld_total_$LOC$ }
		if = {
			limit = { var:sr_cb_lost > 0 }
			sr_fl_each_slot = { DO = apply_losses }
			if = {
				limit = { local_var:sr_cb_code = 1 }
				change_local_variable = { name = sr_cb_debris add = var:sr_cb_lost }
			}
		}
	}
	if = {
		limit = { local_var:sr_cb_debris > 0 }
		# An argument is one token (a block there breaks the file: #1096), so the
		# two shares go through local variables.
		set_local_variable = {
			name = sr_cb_debris_low
			value = { value = local_var:sr_cb_debris multiply = sr_battle_debris_low }
		}
		set_local_variable = {
			name = sr_cb_debris_high
			value = { value = local_var:sr_cb_debris multiply = sr_battle_debris_high }
		}
		te_debris_add = { LOW = local_var:sr_cb_debris_low HIGH = local_var:sr_cb_debris_high }
	}
	# 4. Each world.
	sr_cb_worlds_after_$LOC$ = yes
	# 5. Cleared.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_cb_present }
		sr_cb_clear_$LOC$ = yes
	}
	sr_cb_world_clear_$LOC$ = yes
	sr_loc_rebuild = yes
}

# ---- The battle (2026-11-25 design § 2; each of the fleets' ticks, global scope) --------------
# docs/superpowers/specs/2026-11-25-space-fleet-clock-and-battles-design.md. From
# each tick (sr_fl_tick_step; on the 1st, sr_fl_global_month, before the siege),
# after the landings and the location records, at each location where two
# powers at war meet (sr_battle_at = { LOC CODE }):
#   1. each power's units there join (sr_fl_<n>_bt_*; its Orbital Battlestation
#      at Earth as sr_fl_bs_bt_*): in the battle, its pace, tied or not (§ 2.3:
#      sr_bt_fleet_tied), its stance (sr_fl_<n>_stance, § 5.1: engaging or
#      not, and the share it keeps before it breaks), and the power's
#      strength before (sr_bt_before);
#   2. at Earth, the ground-to-orbit batteries fire once (§ 2.5);
#   3. the strength each unit brings (the break's measure) and each power's
#      totals; an AI sets Engage or Avoid by its side's win test (§§ 2.7, 5.3);
#   4. the battle (§ 2.4): steps of the square law per unit, each unit's fire
#      shared among the hostile units it is paired with (sr_bt_paired) by their
#      strength, until no pair is left or sr_bt_max_steps; a unit breaks once
#      it keeps no more than its stance's share of what it brought, or is
#      destroyed below sr_fl_minimum_strength, and drops out (the battlestation
#      and a fleet fighting to the last never break);
#   5. the broken fleets leave for the nearest dock they may, never the one
#      they are at (sr_fl_retreat_at, § 5.2);
#   6. the records: war support, the report (.101), the location's battle
#      stamp, debris at Earth, the self-check; everything cleared, and the
#      location records rebuilt.
# A unit is a slot 1-9 or bs; the passes that read only its battle figures go
# through sr_bt_each_unit, the ones that read or cut a fleet through
# sr_fl_each_slot and the battlestation's own.
sr_battle_global_tick = {
	sr_battle_at = { LOC = earth CODE = 1 }
	sr_battle_at = { LOC = inner CODE = 2 }
	sr_battle_at = { LOC = mars CODE = 3 }
	sr_battle_at = { LOC = asteroids CODE = 4 }
	sr_battle_at = { LOC = jupiter CODE = 5 }
	sr_battle_at = { LOC = saturn CODE = 6 }
	sr_battle_at = { LOC = uranus CODE = 7 }
	sr_battle_at = { LOC = neptune CODE = 8 }
	sr_battle_at = { LOC = kuiper CODE = 9 }
}

sr_battle_at = {
	set_local_variable = { name = sr_cb_code value = $CODE$ }
	set_local_variable = { name = sr_bt_war value = 0 }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = {
			is_country_alive = yes
			sr_bt_armed_here = { LOC = $LOC$ }
		}
		if = {
			limit = {
				any_in_global_list = {
					variable = sr_fleet_powers
					is_country_alive = yes
					has_war_with = prev
					sr_bt_here = { LOC = $LOC$ }
				}
			}
			set_local_variable = { name = sr_bt_war value = 1 }
		}
	}
	if = {
		limit = { local_var:sr_bt_war = 1 }
		sr_battle_run = { LOC = $LOC$ CODE = $CODE$ }
	}
}

sr_battle_run = {
	set_local_variable = { name = sr_bt_battle value = 0 }
	set_local_variable = { name = sr_bt_fired value = 0 }
	set_local_variable = { name = sr_bt_debris value = 0 }
	set_local_variable = { name = sr_bt_kinetic value = 0 }
	set_local_variable = { name = sr_bt_ends value = 0 }
	set_local_variable = { name = sr_bt_steps value = 0 }
	# 1. The units.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = {
			is_country_alive = yes
			sr_bt_here = { LOC = $LOC$ }
		}
		set_variable = { name = sr_bt_before value = 0 }
		set_variable = { name = sr_bt_lost value = 0 }
		set_variable = { name = sr_bt_dead value = 0 }
		set_local_variable = { name = sr_bt_held value = 0 }
		if = {
			limit = { sr_loc_tied_$LOC$ = yes }
			set_local_variable = { name = sr_bt_held value = 1 }
		}
		set_local_variable = { name = sr_bt_units value = 0 }
		sr_fl_each_slot = { DO = bt_setup }
		if = {
			limit = { local_var:sr_cb_code = 1 }
			sr_bt_setup_bs = yes
		}
		if = {
			limit = { local_var:sr_bt_units > 0 }
			set_variable = { name = sr_bt_present value = yes }
		}
	}
	# 2. The batteries, at Earth.
	if = {
		limit = { local_var:sr_cb_code = 1 }
		every_in_global_list = {
			variable = sr_fleet_powers
			limit = {
				is_country_alive = yes
				sr_has_battery = yes
			}
			sr_bt_battery_fire = yes
		}
	}
	# 3. What each brings; the AI's Engage or Avoid.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_present }
		set_variable = { name = sr_bt_s value = 0 }
		set_variable = { name = sr_bt_f value = 0 }
		# Its fleets' quality x readiness, once a battle (readiness reads its
		# Space Command; review of #2069).
		set_variable = { name = sr_bt_qr value = { value = sr_fleet_quality multiply = sr_fleet_readiness } }
		sr_fl_each_slot = { DO = bt_ready }
		sr_bt_ready_bs = yes
	}
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = {
			has_variable = sr_bt_present
			is_player = no
			var:sr_bt_s > 0
			# Without a Space Command its fleets keep the setting they had, as a
			# player's do (design § 2.9; review of #2069).
			sr_sc_operational = yes
		}
		sr_bt_ai_choose = yes
	}
	# 4. The battle: until no pair is left, at most sr_bt_max_steps steps (a
	# counted ceiling: the engine stops a while at 1,000 passes).
	set_local_variable = { name = sr_bt_pairs value = 1 }
	while = {
		limit = {
			local_var:sr_bt_pairs > 0
			local_var:sr_bt_steps < sr_bt_max_steps
		}
		sr_bt_step = yes
	}
	# 5. The broken fleets leave.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_present }
		sr_fl_each_slot = { DO = retreat }
	}
	# 6. The records.
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_present }
		sr_bt_record = { CODE = $CODE$ }
	}
	if = {
		limit = {
			OR = {
				local_var:sr_bt_battle = 1
				local_var:sr_bt_fired = 1
			}
		}
		set_global_variable = { name = sr_loc_battle_$LOC$ value = te_history_month_index }
	}
	if = {
		limit = {
			local_var:sr_cb_code = 1
			local_var:sr_bt_debris > 0
		}
		# An argument is one token (a block there breaks the file: #1096), so the
		# two shares go through local variables.
		set_local_variable = {
			name = sr_cb_debris_low
			value = { value = local_var:sr_bt_debris multiply = sr_battle_debris_low }
		}
		set_local_variable = {
			name = sr_cb_debris_high
			value = { value = local_var:sr_bt_debris multiply = sr_battle_debris_high }
		}
		te_debris_add = { LOW = local_var:sr_cb_debris_low HIGH = local_var:sr_cb_debris_high }
	}
	if = {
		limit = { local_var:sr_bt_kinetic > 0 }
		set_local_variable = {
			name = sr_cb_debris_low
			value = { value = local_var:sr_bt_kinetic multiply = sr_battery_debris_low }
		}
		te_debris_add = { LOW = local_var:sr_cb_debris_low HIGH = 0 }
	}
	sr_bt_self_check = { CODE = $CODE$ }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_before }
		sr_fl_each_slot = { DO = bt_end }
		sr_bt_end_power = yes
	}
	if = {
		limit = {
			OR = {
				local_var:sr_bt_battle = 1
				local_var:sr_bt_fired = 1
			}
		}
		sr_loc_rebuild = yes
	}
}

# Every unit: slots 1-9 and the battlestation (bs), for the passes that read
# only a unit's battle figures (sr_fl_<n>_bt_*).
sr_bt_each_unit = {
	sr_bt_$DO$ = { N = 1 }
	sr_bt_$DO$ = { N = 2 }
	sr_bt_$DO$ = { N = 3 }
	sr_bt_$DO$ = { N = 4 }
	sr_bt_$DO$ = { N = 5 }
	sr_bt_$DO$ = { N = 6 }
	sr_bt_$DO$ = { N = 7 }
	sr_bt_$DO$ = { N = 8 }
	sr_bt_$DO$ = { N = 9 }
	sr_bt_$DO$ = { N = bs }
}

# 1. Slot N at the location (local_var:sr_cb_code), not all but destroyed,
# joins: its pace, tied or not, whether it engages (every stance but Avoid),
# and the share of what it brings it keeps before it breaks (sr_fl_<n>_bt_keep,
# 2026-11-25 design § 5.1): Engage, a new fleet's, half; Avoid and Hit and Run
# three quarters; Fight to the Last none, so it never breaks.
sr_fl_bt_setup_at = {
	if = {
		limit = {
			has_variable = sr_fl_$N$_loc
			var:sr_fl_$N$_loc = local_var:sr_cb_code
			sr_fl_$N$_strength >= sr_fl_minimum_strength
		}
		set_variable = { name = sr_fl_$N$_bt_in value = yes }
		set_variable = { name = sr_fl_$N$_bt_pace value = sr_fl_$N$_pace }
		sr_bt_set_stance = { N = $N$ }
		set_variable = { name = sr_fl_$N$_bt_tied value = 0 }
		if = {
			limit = { sr_bt_fleet_tied = { N = $N$ } }
			set_variable = { name = sr_fl_$N$_bt_tied value = 1 }
		}
		change_variable = { name = sr_bt_before add = sr_fl_$N$_strength }
		change_local_variable = { name = sr_bt_units add = 1 }
	}
}

# Slot N's stance into its battle figures: engaging or not, and the share it
# keeps before it breaks (sr_fl_bt_setup_at; again after an AI's choice).
sr_bt_set_stance = {
	set_variable = { name = sr_fl_$N$_bt_eng value = 1 }
	set_variable = { name = sr_fl_$N$_bt_keep value = { value = 1 subtract = sr_bt_break_share } }
	if = {
		limit = { sr_fl_$N$_stance_value = 1 }
		set_variable = { name = sr_fl_$N$_bt_eng value = 0 }
		set_variable = { name = sr_fl_$N$_bt_keep value = { value = 1 subtract = sr_bt_break_share_early } }
	}
	else_if = {
		limit = { sr_fl_$N$_stance_value = 2 }
		set_variable = { name = sr_fl_$N$_bt_keep value = { value = 1 subtract = sr_bt_break_share_early } }
	}
	else_if = {
		limit = { sr_fl_$N$_stance_value = 4 }
		set_variable = { name = sr_fl_$N$_bt_keep value = 0 }
	}
}

# The Orbital Battlestation, at Earth, for its owner while its state is held
# (design § 7.5): it always engages, is always tied, and is the slowest.
sr_bt_setup_bs = {
	if = {
		limit = { sr_battlestation_held = yes }
		if = {
			limit = { NOT = { has_variable = sr_battlestation_strength } }
			set_variable = { name = sr_battlestation_strength value = sr_battlestation_full }
		}
		if = {
			limit = { var:sr_battlestation_strength >= sr_fl_minimum_strength }
			set_variable = { name = sr_fl_bs_bt_in value = yes }
			set_variable = { name = sr_fl_bs_bt_pace value = 0 }
			set_variable = { name = sr_fl_bs_bt_eng value = 1 }
			set_variable = { name = sr_fl_bs_bt_tied value = 1 }
			change_variable = { name = sr_bt_before add = var:sr_battlestation_strength }
			change_local_variable = { name = sr_bt_units add = 1 }
		}
	}
}

# 2. Country scope, a battery owner at Earth: a tick's fire (sr_fleet_k x
# sr_battery_tick_share x its fire) on the fleets in the battle of every power
# at war with it, shared by strength. Its kinetic part's debris, a quarter of a
# month's, only when there was a fleet to fire on. It is no fleet: nothing
# fires back (design § 15).
sr_bt_battery_fire = {
	set_local_variable = { name = sr_bt_bf value = sr_battery_fire }
	if = {
		limit = { local_var:sr_bt_bf > 0 }
		set_local_variable = { name = sr_bt_targets value = 0 }
		every_in_global_list = {
			variable = sr_fleet_powers
			limit = {
				has_variable = sr_bt_present
				has_war_with = prev
			}
			sr_fl_each_slot = { DO = bt_target }
		}
		if = {
			limit = { local_var:sr_bt_targets > 0 }
			set_local_variable = {
				name = sr_bt_per
				value = {
					value = local_var:sr_bt_bf
					multiply = sr_fleet_k
					multiply = sr_battery_tick_share
					divide = local_var:sr_bt_targets
				}
			}
			every_in_global_list = {
				variable = sr_fleet_powers
				limit = {
					has_variable = sr_bt_present
					has_war_with = prev
				}
				sr_fl_each_slot = { DO = bt_battery }
			}
			change_local_variable = {
				name = sr_bt_kinetic
				add = { value = sr_battery_fire_kinetic multiply = sr_battery_tick_share }
			}
			set_local_variable = { name = sr_bt_fired value = 1 }
			if = {
				limit = { NOT = { has_variable = sr_bt_present } }
				set_variable = { name = sr_bt_present value = yes }
			}
		}
	}
}

sr_fl_bt_target_at = {
	if = {
		limit = { has_variable = sr_fl_$N$_bt_in }
		change_local_variable = { name = sr_bt_targets add = sr_fl_$N$_strength }
	}
}

sr_fl_bt_battery_at = {
	if = {
		limit = { has_variable = sr_fl_$N$_bt_in }
		set_local_variable = {
			name = sr_fl_loss
			value = {
				value = sr_fl_$N$_strength
				multiply = local_var:sr_bt_per
				max = sr_fl_$N$_strength
			}
		}
		sr_bt_cut_slot = { N = $N$ }
		if = {
			limit = { sr_fl_$N$_strength < sr_fl_minimum_strength }
			remove_variable = sr_fl_$N$_bt_in
		}
	}
}

# Slot N loses local_var:sr_fl_loss (screens first, half destroyed, half
# damage: sr_fl_take_losses_at); its owner's record takes the loss and the
# crews that died with the hulls (sr_pers_note_people adds them to
# local_var:sr_pers_noted).
sr_bt_cut_slot = {
	set_local_variable = { name = sr_pers_noted value = 0 }
	sr_fl_take_losses_at = { N = $N$ }
	change_variable = { name = sr_bt_lost add = local_var:sr_fl_loss }
	change_variable = { name = sr_bt_dead add = local_var:sr_pers_noted }
	change_local_variable = { name = sr_bt_debris add = local_var:sr_fl_loss }
}

# 3. What slot N brings to the battle (its break is measured against it), and
# its owner's totals for the AI's win test.
sr_fl_bt_ready_at = {
	if = {
		limit = { has_variable = sr_fl_$N$_bt_in }
		set_variable = { name = sr_fl_$N$_bt_brought value = sr_fl_$N$_strength }
		change_variable = { name = sr_bt_s add = sr_fl_$N$_strength }
		change_variable = {
			name = sr_bt_f
			add = {
				value = sr_fl_$N$_fire_fleets
				# Set by step 3 before this; a tooltip's walk writes nothing (tooltip_walk).
				if = { limit = { has_variable = sr_bt_qr } multiply = var:sr_bt_qr }
			}
		}
	}
}

sr_bt_ready_bs = {
	if = {
		limit = { has_variable = sr_fl_bs_bt_in }
		set_variable = { name = sr_fl_bs_bt_brought value = var:sr_battlestation_strength }
		change_variable = { name = sr_bt_s add = var:sr_battlestation_strength }
		change_variable = { name = sr_bt_f add = sr_battlestation_fire }
	}
}

# Country scope, an AI power in the battle (§§ 2.7, 5.3): it engages when its
# side's strength there x its side's fire is at least the same product for
# every power at war with it there (x sr_fl_ai_engage_margin), and avoids
# otherwise. Its side is it and its friends there: each power in the battle at
# peace with it and at war with a power there at war with it, whatever their
# stances (a player's Avoid can still be caught). Battlestations count. The
# stance is its fleets' own, kept after the battle: Engage or Avoid; the AI
# leaves Hit and Run and Fight to the Last to players.
sr_bt_ai_choose = {
	save_temporary_scope_as = sr_bt_side
	set_local_variable = { name = sr_bt_us_s value = var:sr_bt_s }
	set_local_variable = { name = sr_bt_us_f value = var:sr_bt_f }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = {
			has_variable = sr_bt_present
			has_variable = sr_bt_s
			NOT = { this = scope:sr_bt_side }
			NOT = { has_war_with = scope:sr_bt_side }
			any_in_global_list = {
				variable = sr_fleet_powers
				has_variable = sr_bt_present
				has_war_with = scope:sr_bt_side
				has_war_with = prev
			}
		}
		change_local_variable = { name = sr_bt_us_s add = var:sr_bt_s }
		change_local_variable = { name = sr_bt_us_f add = var:sr_bt_f }
	}
	set_local_variable = { name = sr_bt_them_s value = 0 }
	set_local_variable = { name = sr_bt_them_f value = 0 }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = {
			has_variable = sr_bt_present
			has_variable = sr_bt_s
			has_war_with = prev
		}
		change_local_variable = { name = sr_bt_them_s add = var:sr_bt_s }
		change_local_variable = { name = sr_bt_them_f add = var:sr_bt_f }
	}
	set_local_variable = { name = sr_bt_ours value = { value = local_var:sr_bt_us_s multiply = local_var:sr_bt_us_f } }
	set_local_variable = {
		name = sr_bt_theirs
		value = {
			value = local_var:sr_bt_them_s
			multiply = local_var:sr_bt_them_f
			multiply = sr_fl_ai_engage_margin
		}
	}
	set_local_variable = { name = sr_bt_ai_eng value = 0 }
	if = {
		limit = { local_var:sr_bt_ours >= local_var:sr_bt_theirs }
		set_local_variable = { name = sr_bt_ai_eng value = 1 }
	}
	sr_fl_each_slot = { DO = bt_ai }
}

sr_fl_bt_ai_at = {
	if = {
		limit = { has_variable = sr_fl_$N$_bt_in }
		if = {
			limit = { local_var:sr_bt_ai_eng = 1 }
			if = {
				limit = { has_variable = sr_fl_$N$_stance }
				remove_variable = sr_fl_$N$_stance
			}
		}
		else = {
			set_variable = { name = sr_fl_$N$_stance value = 1 }
		}
		sr_bt_set_stance = { N = $N$ }
	}
}

# 4. One step (global scope, the location in local_var:sr_cb_code): every
# unit's strength and fire now; each unit's engaged strength (the hostile
# units it is paired with) and its fire per point engaged; each unit's loss,
# sr_bt_k x its strength x the fire per point engaged of every unit paired
# with it, so each unit's fire is dealt out exactly; then the losses, all
# reckoned before any is taken, and the breaks. local_var:sr_bt_pairs counts
# the units with a pair.
sr_bt_step = {
	set_local_variable = { name = sr_bt_pairs value = 0 }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_present }
		sr_fl_each_slot = { DO = bt_snap }
		sr_bt_snap_bs = yes
	}
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_present }
		sr_bt_each_unit = { DO = engaged }
	}
	# Fleets keep a battle going: they can break. Battlestations, which can't,
	# alone exchange one step a tick, at a quarter of a month's fire (the
	# batteries' share), not sixty (review of #2069).
	set_local_variable = { name = sr_bt_fleet_pairs value = 0 }
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_bt_present }
		sr_fl_each_slot = { DO = bt_count }
	}
	set_local_variable = { name = sr_bt_step_k value = sr_bt_k }
	if = {
		limit = {
			local_var:sr_bt_pairs > 0
			local_var:sr_bt_fleet_pairs = 0
		}
		if = {
			limit = { local_var:sr_bt_steps > 0 }
			set_local_variable = { name = sr_bt_pairs value = 0 }
		}
		else = {
			set_local_variable = { name = sr_bt_step_k value = { value = sr_fleet_k multiply = sr_battery_tick_share } }
		}
	}
	if = {
		limit = { local_var:sr_bt_pairs > 0 }
		set_local_variable = { name = sr_bt_battle value = 1 }
		every_in_global_list = {
			variable = sr_fleet_powers
			limit = { has_variable = sr_bt_present }
			sr_bt_each_unit = { DO = loss }
		}
		every_in_global_list = {
			variable = sr_fleet_powers
			limit = { has_variable = sr_bt_present }
			sr_fl_each_slot = { DO = bt_apply }
			sr_bt_apply_bs = yes
		}
		change_local_variable = { name = sr_bt_steps add = 1 }
	}
}

sr_fl_bt_snap_at = {
	if = {
		limit = { has_variable = sr_fl_$N$_bt_in }
		set_variable = { name = sr_fl_$N$_bt_s value = sr_fl_$N$_strength }
		set_variable = {
			name = sr_fl_$N$_bt_f
			value = {
				value = sr_fl_$N$_fire_fleets
				if = { limit = { has_variable = sr_bt_qr } multiply = var:sr_bt_qr }
			}
		}
	}
}

sr_bt_snap_bs = {
	if = {
		limit = { has_variable = sr_fl_bs_bt_in }
		set_variable = { name = sr_fl_bs_bt_s value = var:sr_battlestation_strength }
		set_variable = { name = sr_fl_bs_bt_f value = sr_battlestation_fire }
	}
}

# Unit N's figures as the pair test's "mine" (sr_bt_paired reads them).
sr_bt_me = {

	set_local_variable = { name = sr_bt_my_tied value = var:sr_fl_$N$_bt_tied }
	set_local_variable = { name = sr_bt_my_pace value = var:sr_fl_$N$_bt_pace }
}

sr_bt_engaged = {
	if = {
		limit = { has_variable = sr_fl_$N$_bt_in }
		sr_bt_me = { N = $N$ }
		set_local_variable = { name = sr_bt_sum value = 0 }
		every_in_global_list = {
			variable = sr_fleet_powers
			limit = {
				has_variable = sr_bt_present
				has_war_with = prev
			}
			sr_bt_each_unit = { DO = add_engaged }
		}
		set_variable = { name = sr_fl_$N$_bt_r value = 0 }
		set_variable = { name = sr_fl_$N$_bt_p value = 0 }

			limit = { local_var:sr_bt_sum > 0 }
			set_variable = {
				name = sr_fl_$N$_bt_r
				value = { value = var:sr_fl_$N$_bt_f divide = local_var:sr_bt_sum }
			}

	}

# Slot N's loss taken; destroyed below sr_fl_minimum_strength, or broken at
# the share of what it brought its stance keeps (sr_fl_<n>_bt_keep; never at
# Fight to the Last's none): either way out of the battle, a broken one marked
# to leave (sr_fl_retreat_at).
sr_fl_bt_apply_at = {
	if = {
		limit = {
			has_variable = sr_fl_$N$_bt_in
			has_variable = sr_fl_$N$_bt_l
			var:sr_fl_$N$_bt_l > 0
		}
		set_local_variable = {
			name = sr_fl_loss
			value = { value = var:sr_fl_$N$_bt_l max = sr_fl_$N$_strength }
		}
		sr_bt_cut_slot = { N = $N$ }
		if = {
			limit = { sr_fl_$N$_strength < sr_fl_minimum_strength }
			remove_variable = sr_fl_$N$_bt_in
			change_local_variable = { name = sr_bt_ends add = 1 }
		}
		else_if = {
			limit = {
				sr_fl_$N$_strength <= {
					value = var:sr_fl_$N$_bt_keep
					multiply = var:sr_fl_$N$_bt_brought
				}
			}
			remove_variable = sr_fl_$N$_bt_in
			set_variable = { name = sr_fl_$N$_bt_broke value = yes }
			change_local_variable = { name = sr_bt_ends add = 1 }
		}
	}
}

# The battlestation's loss, from its strength (not the building); it fights
# until that is gone, and regains 10% a month (sr_cb_battlestation_month).
sr_bt_apply_bs = {
	if = {
		limit = {
			has_variable = sr_fl_bs_bt_in
			has_variable = sr_fl_bs_bt_l
			var:sr_fl_bs_bt_l > 0
		}
		set_local_variable = {
			name = sr_bt_cut
			value = { value = var:sr_fl_bs_bt_l max = var:sr_battlestation_strength }
		}
		change_variable = { name = sr_battlestation_strength subtract = local_var:sr_bt_cut }
		change_variable = { name = sr_bt_lost add = local_var:sr_bt_cut }
		change_local_variable = { name = sr_bt_debris add = local_var:sr_bt_cut }
		if = {
			limit = { var:sr_battlestation_strength < sr_fl_minimum_strength }
			remove_variable = sr_fl_bs_bt_in
			change_local_variable = { name = sr_bt_ends add = 1 }
		}
	}
}

# 6. Country scope, a power in the battle: war support's record, for thirty
# days, the later battle's verdict standing (+0.1 a beat when the enemy lost
# the larger share of its strength, -0.1 when we did; +-0.25 when a side's
# force there was destroyed outright), and a player's report (.101): one a
# battle, at most one a tick (sr_battle_notice_pending, 6 days); after a tick
# of the batteries' fire alone, at most one a month (sr_battery_notice_pending).
sr_bt_record = {
	if = {
		limit = { sr_bt_fought = yes }
		set_local_variable = { name = sr_cb_ours value = 0 }
		if = {
			limit = { var:sr_bt_before > 0 }
			set_local_variable = { name = sr_cb_ours value = { value = var:sr_bt_lost divide = var:sr_bt_before } }
		}
		set_local_variable = { name = sr_cb_theirs value = sr_bt_enemy_lost_share }
		if = {
			limit = { local_var:sr_cb_theirs >= 0.99 }
			set_variable = { name = sr_ws_battle value = sr_ws_battle_rout days = 30 }
		}
		else_if = {
			limit = { local_var:sr_cb_ours >= 0.99 }
			set_variable = { name = sr_ws_battle value = { value = sr_ws_battle_rout multiply = -1 } days = 30 }
		}
		else_if = {
			limit = { local_var:sr_cb_theirs > local_var:sr_cb_ours }
			set_variable = { name = sr_ws_battle value = sr_ws_battle days = 30 }
		}
		else = {
			set_variable = { name = sr_ws_battle value = { value = sr_ws_battle multiply = -1 } days = 30 }
		}
		# One report a tick: a battle's, or the batteries' fire alone at most once a
		# month. The report reads its figures a day later, so a battle elsewhere
		# in the same tick takes over a batteries' report already queued instead
		# of queuing a second (sr_battle_notice_batteries, a day; review of #2069).
		if = {
			limit = {
				is_player = yes
				local_var:sr_bt_battle = 1
				has_variable = sr_battle_notice_batteries
			}
			remove_variable = sr_battle_notice_batteries
			set_variable = { name = sr_battle_loc value = $CODE$ }
			sr_bt_note_figures = yes
		}
		else_if = {
			limit = {
				is_player = yes
				NOT = { has_variable = sr_battle_notice_pending }
				OR = {
					local_var:sr_bt_battle = 1
					NOT = { has_variable = sr_battery_notice_pending }
				}
			}
			set_variable = { name = sr_battle_notice_pending value = yes days = 6 }
			if = {
				limit = { local_var:sr_bt_battle = 0 }
				set_variable = { name = sr_battery_notice_pending value = yes days = 25 }
				set_variable = { name = sr_battle_notice_batteries value = yes days = 1 }
			}
			set_variable = { name = sr_battle_loc value = $CODE$ }
			sr_bt_note_figures = yes
			trigger_event = { id = space_race_events.101 days = 1 }
		}
	}
}
# The report's figures beside sr_battle_loc, read by space_race_events.101 a
# day later.
sr_bt_note_figures = {
	set_variable = { name = sr_battle_lost value = var:sr_bt_lost }
	set_variable = { name = sr_battle_dealt value = sr_bt_enemy_lost }
	set_variable = { name = sr_battle_left value = { value = var:sr_bt_before subtract = var:sr_bt_lost min = 0 } }
	# The crews that died with the destroyed half (space forces personnel
	# design § 7): what sr_fl_cut_role booked in this battle, so the
	# Orbital Battlestation's share of the losses (no crew) and the
	# full-strength floor are left out (review of #2020).
	set_variable = { name = sr_battle_dead value = var:sr_bt_dead }
}

# Global scope, after the battle (§ 2.10): TE_SELF_CHECK fleet_battle, once a
# game each way. OK at the first battle that ends with a break or a unit
# destroyed inside the step ceiling. BROKEN at the first that reaches the
# ceiling with units still paired whose fire is at least half their strength:
# fleets with real fire that the steps didn't settle, so sr_bt_k or
# sr_bt_max_steps is wrong (fleets with almost no fire, tenders on tenders,
# may reach it rightly).
sr_bt_self_check = {
	if = {
		limit = {
			local_var:sr_bt_steps >= sr_bt_max_steps
			local_var:sr_bt_pairs > 0
			NOT = { has_global_variable = sr_bt_broken_logged }
		}
		set_local_variable = { name = sr_bt_pf value = 0 }
		set_local_variable = { name = sr_bt_ps value = 0 }
		every_in_global_list = {
			variable = sr_fleet_powers
			limit = { has_variable = sr_bt_present }
			sr_fl_each_slot = { DO = bt_left }
		}
		if = {
			limit = {
				local_var:sr_bt_ps > 0
				local_var:sr_bt_pf >= { value = local_var:sr_bt_ps multiply = 0.5 }
			}
			set_global_variable = { name = sr_bt_broken_logged value = yes }
			set_global_variable = { name = sr_bt_log_loc value = $CODE$ }
			set_global_variable = { name = sr_bt_log_steps value = local_var:sr_bt_steps }
			set_global_variable = { name = sr_bt_log_fire value = local_var:sr_bt_pf }
			set_global_variable = { name = sr_bt_log_strength value = local_var:sr_bt_ps }
			debug_log = "TE_SELF_CHECK BROKEN fleet_battle location=[GetGlobalVariable('sr_bt_log_loc').GetValue|0] steps=[GetGlobalVariable('sr_bt_log_steps').GetValue|0] fire=[GetGlobalVariable('sr_bt_log_fire').GetValue|1] strength=[GetGlobalVariable('sr_bt_log_strength').GetValue|1] (the step ceiling, sr_bt_max_steps, reached with fleets still paired whose fire is at least half their strength: raise sr_bt_k or the ceiling; docs/testing/solar-colonies-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}
	else_if = {
		limit = {
			local_var:sr_bt_battle = 1
			local_var:sr_bt_ends > 0
			local_var:sr_bt_steps < sr_bt_max_steps
			NOT = { has_global_variable = sr_bt_ok_logged }
		}
		set_global_variable = { name = sr_bt_ok_logged value = yes }
		set_global_variable = { name = sr_bt_log_loc value = $CODE$ }
		set_global_variable = { name = sr_bt_log_steps value = local_var:sr_bt_steps }
		set_global_variable = { name = sr_bt_log_ends value = local_var:sr_bt_ends }
		debug_log = "TE_SELF_CHECK OK fleet_battle location=[GetGlobalVariable('sr_bt_log_loc').GetValue|0] steps=[GetGlobalVariable('sr_bt_log_steps').GetValue|0] ended=[GetGlobalVariable('sr_bt_log_ends').GetValue|0] (a battle settled within its tick: units broke or were destroyed inside the ceiling of sr_bt_max_steps; an even fight takes about 14) date=[TimeKeeper.GetCurrentDate.GetString]"
	}
}

# The verdict from the sums (local_var:sr_fl_ts, _tf, _os, _of): the square
# law's ratio of ours to theirs, strength x fire (local_var:sr_fc_r);
# local_var:sr_fc_tier 0 nothing of theirs, 1 a clear win (ours at least
# sr_fc_clear times theirs), 2 a costly win, 3 a defeat, 4 a rout (theirs at
# least sr_fc_clear times ours, or nothing of ours); and local_var:sr_fc_loss,
# the share of its strength the winner loses while the loser loses half:
# 1 - sqrt(1 - sr_fc_break_loss / ratio), by the square law with both sides
# engaging. Country scope (te_set_sqrt's variable is set and removed here).
sr_fc_judge = {
	set_local_variable = { name = sr_fc_tier value = 0 }
	set_local_variable = { name = sr_fc_loss value = 0 }
	set_local_variable = { name = sr_fc_r value = 0 }
	if = {
		limit = {
			local_var:sr_fl_ts > 0
			local_var:sr_fl_tf > 0
		}
		if = {
			limit = {
				local_var:sr_fl_os > 0
				local_var:sr_fl_of > 0
			}
			set_local_variable = {
				name = sr_fc_r
				value = {
					value = local_var:sr_fl_os
					divide = local_var:sr_fl_ts
					multiply = { value = local_var:sr_fl_of divide = local_var:sr_fl_tf }
				}
			}
		}
		if = {
			limit = { local_var:sr_fc_r >= 1 }
			set_local_variable = { name = sr_fc_tier value = 2 }
			if = {
				limit = { local_var:sr_fc_r >= sr_fc_clear }
				set_local_variable = { name = sr_fc_tier value = 1 }
			}
			set_local_variable = {
				name = sr_fc_sq
				value = { value = 1 subtract = { value = sr_fc_break_loss divide = local_var:sr_fc_r } min = 0 }
			}
		}
		else = {
			set_local_variable = { name = sr_fc_tier value = 3 }
			if = {
				limit = { local_var:sr_fc_r <= { value = 1 divide = sr_fc_clear } }
				set_local_variable = { name = sr_fc_tier value = 4 }
			}
			set_local_variable = {
				name = sr_fc_sq
				value = { value = 1 subtract = { value = sr_fc_break_loss multiply = local_var:sr_fc_r } min = 0 }
			}
		}
		te_set_sqrt = { IN = local_var:sr_fc_sq OUT = sr_fc_rest }
		if = {
			limit = { has_variable = sr_fc_rest }
			set_local_variable = { name = sr_fc_loss value = { value = 1 subtract = var:sr_fc_rest } }
			remove_variable = sr_fc_rest
		}
	}
}

# ---- One slot's part ($N$; the power's scope) ---------------------------------------------------
# 0. A slot at the location (local_var:sr_cb_code) adds its strength and fire
# to its posture's; closing in, to its target world's too. At Earth every
# fleet is stationed, except one closing in on a lunar world (lunar
# colonization § 8.1): any other posture there is put back to Station. A target
# no longer hostile (a peace made, the world taken) turns it to Stand off first
# (at Earth, to Station).
sr_fl_gather_at = {
	if = {
		limit = {
			has_variable = sr_fl_$N$_loc
			var:sr_fl_$N$_loc = local_var:sr_cb_code
			sr_fl_$N$_strength > 0
		}
		set_local_variable = { name = sr_cb_p value = sr_fl_$N$_posture_value }
		if = {
			limit = {
				local_var:sr_cb_code = 1
				NOT = {
					AND = {
						local_var:sr_cb_p = 3
						sr_fl_$N$_target_lunar = yes
					}
				}
			}
			set_local_variable = { name = sr_cb_p value = 1 }
			if = {
				limit = { NOT = { sr_fl_$N$_posture_value = 1 } }
				set_variable = { name = sr_fl_$N$_posture value = 1 }
			}
		}
		if = {
			limit = { local_var:sr_cb_p = 3 }
			sr_fl_check_target_at = { N = $N$ }
			set_local_variable = { name = sr_cb_p value = sr_fl_$N$_posture_value }
		}
		# Interdicting (6c) fights as standing off; a target no longer to be
		# interdicted (a war, an alliance, the world gone) ends it first.
		if = {
			limit = { local_var:sr_cb_p = 4 }
			sr_fl_check_icpt_at = { N = $N$ }
			set_local_variable = { name = sr_cb_p value = 2 }
		}
		if = { limit = { local_var:sr_cb_p < 1 } set_local_variable = { name = sr_cb_p value = 1 } }
		set_local_variable = { name = sr_cb_s value = sr_fl_$N$_strength }
		set_local_variable = { name = sr_cb_f value = { value = sr_fl_$N$_fire_fleets multiply = sr_fleet_quality multiply = sr_fleet_readiness } }
		if = {
			limit = { local_var:sr_cb_p = 1 }
			sr_cb_add = { VAR = sr_cb_s1 AMOUNT = local_var:sr_cb_s }
			sr_cb_add = { VAR = sr_cb_f1 AMOUNT = local_var:sr_cb_f }
		}
		else_if = {
			limit = { local_var:sr_cb_p = 2 }
			sr_cb_add = { VAR = sr_cb_s2 AMOUNT = local_var:sr_cb_s }
			sr_cb_add = { VAR = sr_cb_f2 AMOUNT = local_var:sr_cb_f }
		}
		else = {
			sr_cb_add = { VAR = sr_cb_s3 AMOUNT = local_var:sr_cb_s }
			sr_cb_add = { VAR = sr_cb_f3 AMOUNT = local_var:sr_cb_f }
			sr_fl_gather_target_at = { N = $N$ }
		}
		if = {
			limit = { NOT = { has_variable = sr_cb_present } }
			set_variable = { name = sr_cb_present value = yes }
		}
	}
}

# The slot's target world KEY (it closes in on it): its strength, its fire on

sr_cb_add_target = {

	}
	sr_cb_add = { VAR = sr_cb_fd3_$KEY$ AMOUNT = local_var:sr_cb_fd }
	if = {
		limit = { has_variable = sr_fl_$N$_destroy }
		sr_cb_add = { VAR = sr_cb_dfd_$KEY$ AMOUNT = local_var:sr_cb_fd }
	}
	else_if = {
		limit = { sr_fl_$N$_marines_value > 0 }
		sr_cb_add = { VAR = sr_cb_mar_$KEY$ AMOUNT = sr_fl_$N$_marines_eff }
	}
}

# An interdicting slot (6c) whose target world KEY may no longer be interdicted

sr_cb_check_icpt = {

}

# ---- Interdiction in peace (6c, design § 12) ----------------------------------------------------
# Country scope, the interdictor; slot N interdicts the holder of the world KEY.
# When it begins (per fleet and target: sr_fl_<n>_icpt holds the country it
# interdicts), it costs 5 infamy and 10 on the UN dossier's hostile acts in
# space, and gives the target its casus belli: the interdictor keeps the list
# of countries it interdicted (sr_interdiction_targets) while
# sr_interdiction_recent (five years, renewed by each new interdiction) stands,
# and Cede Colony's war goal costs them no infamy against it. The target hears
# of it (.105).
sr_cb_charge_interdict = {
	global_var:sr_site_holder_$KEY$ ?= { save_scope_as = sr_icpt_target }
	if = {
		limit = {
			exists = scope:sr_icpt_target
			OR = {
				NOT = { has_variable = sr_fl_$N$_icpt }
				NOT = { var:sr_fl_$N$_icpt = scope:sr_icpt_target }
			}
		}
		set_variable = { name = sr_fl_$N$_icpt value = scope:sr_icpt_target }
		change_infamy = sr_interdiction_infamy
		un_dossier_record = { RECORD = space POINTS = sr_interdiction_dossier }
		if = {
			limit = {
				NOT = { has_variable = sr_interdiction_recent }
				has_variable_list = sr_interdiction_targets
			}
			clear_variable_list = sr_interdiction_targets
		}
		if = {
			limit = { NOT = { sr_interdicted_recently = { TARGET = scope:sr_icpt_target } } }
			add_to_variable_list = { name = sr_interdiction_targets target = scope:sr_icpt_target }
		}
		set_variable = { name = sr_interdiction_recent value = yes days = 1825 }
		save_scope_as = sr_icpt_power
		set_local_variable = { name = sr_icpt_loc value = sr_fl_$N$_loc_value }
		# A debug line reads no local variable or saved scope: the location
		# goes through a global, and each side logs its own tag.
		set_global_variable = { name = sr_cb_log_loc value = local_var:sr_icpt_loc }
		debug_log = "TE_SR_COMBAT interdiction begun at location [GetGlobalVariable('sr_cb_log_loc').GetValue|0] by [THIS.GetCountry.GetTagName]"
		scope:sr_icpt_target = {
			debug_log = "TE_SR_COMBAT interdiction at location [GetGlobalVariable('sr_cb_log_loc').GetValue|0] against [THIS.GetCountry.GetTagName]"
			if = {
				limit = {
					is_player = yes
					NOT = { has_variable = sr_icpt_alert_loc }
				}
				set_variable = { name = sr_icpt_alert_loc value = local_var:sr_icpt_loc }
				set_variable = { name = sr_icpt_alert_by value = scope:sr_icpt_power }
				trigger_event = { id = space_race_events.105 days = 1 }
			}
		}
	}
}

# A revolution's winner continues the nation (ROOT, from sr_repair_after_civil_war,
# scope:te_cw_loser the old government): variables merge in, variable lists
# don't. So the loser's casus belli list comes to the winner (it keeps
# sr_interdiction_recent, a variable), the winner joins every list that named
# the loser, and every fleet that interdicted the loser now names the winner,
# so the next month's check charges no new price.
sr_icpt_repair_after_civil_war = {
	if = {
		limit = { exists = scope:te_cw_loser }
		if = {
			limit = { scope:te_cw_loser = { has_variable_list = sr_interdiction_targets } }
			scope:te_cw_loser = {
				every_in_list = {
					variable = sr_interdiction_targets
					save_temporary_scope_as = sr_icpt_carried
					ROOT = {
						if = {
							limit = { NOT = { sr_interdicted_recently = { TARGET = scope:sr_icpt_carried } } }
							add_to_variable_list = { name = sr_interdiction_targets target = scope:sr_icpt_carried }
						}
					}
				}
			}
		}
		if = {
			limit = { has_global_variable_list = sr_fleet_powers }
			every_in_global_list = {
				variable = sr_fleet_powers
				limit = { is_country_alive = yes }
				if = {
					limit = {
						has_variable_list = sr_interdiction_targets
						is_target_in_variable_list = { name = sr_interdiction_targets target = scope:te_cw_loser }
						NOT = { is_target_in_variable_list = { name = sr_interdiction_targets target = ROOT } }
					}
					add_to_variable_list = { name = sr_interdiction_targets target = ROOT }
				}
				sr_fl_each_slot = { DO = icpt_repoint }
			}
		}
	}
}

sr_fl_icpt_repoint_at = {

			var:sr_fl_$N$_icpt = scope:te_cw_loser
		}

	}
}

# ---- The Orbital Battlestation (design § 7.5) -------------------------------------------------
# Country scope, at Earth: 25 warship strength a level, fighting for its

# In the siege it counts as stationed (it bars a seizure and holds orbit); in
# the battle it is a unit of its own (sr_bt_setup_bs).
sr_cb_battlestation_gather = {
	if = {
		limit = { sr_battlestation_held = yes }

			set_variable = { name = sr_battlestation_strength value = sr_battlestation_full }
		}
		if = {
			limit = { var:sr_battlestation_strength > 0 }
			sr_cb_add = { VAR = sr_cb_s1 AMOUNT = var:sr_battlestation_strength }
			sr_cb_add = { VAR = sr_cb_f1 AMOUNT = sr_battlestation_fire }
			set_variable = { name = sr_cb_bs value = var:sr_battlestation_strength }

# Country scope, the fleets' month: the battlestation regains strength while

# ---- Ground-to-orbit batteries (design § 15, 6d) -----------------------------------------------
# They fire once a tick in the battle at Earth (sr_bt_battery_fire, 2026-11-25

# ---- A world's month (global scope; KEY, INDEX; generated sr_cb_worlds_after_<loc>) --------------
# Integrity falls by 100 x k x the defence fire aimed at it / its hardness; at
# 0, the closing-in power with the most marines above its garrison and militia,
# and no hostile fleet stationed against it, seizes it (§ 9.1); a power
# closing in to destroy with its fire above the Defences' for three months
# running destroys it (§ 9.3).
sr_site_assault_step = {
	set_local_variable = { name = sr_cb_fd value = sr_cb_fd_total_$KEY$ }
	if = {
		limit = { local_var:sr_cb_fd > 0 }
		set_global_variable = { name = sr_site_under_assault_$KEY$ value = te_history_month_index }
		set_global_variable = {
			name = sr_site_integrity_$KEY$
			value = {
				value = local_var:sr_cb_fd
				multiply = sr_fleet_k
				multiply = 100
				divide = sr_site_hardness_$KEY$
				multiply = -1
				add = sr_site_integrity_value_$KEY$
				min = 0
			}
		}
	}
	# Destruction.
	if = {
		limit = {
			has_global_variable_list = sr_fleet_powers
			any_in_global_list = {
				variable = sr_fleet_powers
				sr_cb_destroys = { KEY = $KEY$ }
			}
		}
		ordered_in_global_list = {
			variable = sr_fleet_powers
			limit = { sr_cb_destroys = { KEY = $KEY$ } }
			order_by = var:sr_cb_dfd_$KEY$
			position = 0
			save_temporary_scope_as = sr_cb_destroyer_$KEY$
		}
		if = {
			limit = { NOT = { has_global_variable = sr_site_bombard_$KEY$ } }
			set_global_variable = { name = sr_site_bombard_$KEY$ value = 0 }
		}
		change_global_variable = { name = sr_site_bombard_$KEY$ add = 1 }
		if = {
			limit = { global_var:sr_site_bombard_$KEY$ >= sr_destroy_months }
			scope:sr_cb_destroyer_$KEY$ = { sr_site_destroy = { KEY = $KEY$ INDEX = $INDEX$ } }
		}
	}
	else_if = {
		limit = { has_global_variable = sr_site_bombard_$KEY$ }
		remove_global_variable = sr_site_bombard_$KEY$
	}
	# Seizure.
	if = {
		limit = {
			has_global_variable = sr_colony_$KEY$
			sr_site_integrity_value_$KEY$ <= 0
			has_global_variable_list = sr_fleet_powers
			any_in_global_list = {
				variable = sr_fleet_powers
				sr_cb_seizes = { KEY = $KEY$ }
			}
		}
		ordered_in_global_list = {
			variable = sr_fleet_powers
			limit = { sr_cb_seizes = { KEY = $KEY$ } }
			order_by = var:sr_cb_mar_$KEY$
			position = 0
			save_temporary_scope_as = sr_cb_seizer_$KEY$
		}
		scope:sr_cb_seizer_$KEY$ = { sr_site_seize = { KEY = $KEY$ INDEX = $INDEX$ } }
	}
}

# Country scope, the seizer (THIS), at the world KEY: its marines needed land
# (one more battalion than the garrison and militia, at most what it brought)
# and the world is taken. A free colony is ours at once (§ 9.4); an occupied
# world taken by its holder or one not at war with the holder is liberated
# (§ 9.2); any other is occupied.
#
# Space forces personnel (design §§ 4, 5.2): the figures compared are at the
# crewed share (sr_cb_mar_<key> holds the closing-in marines at our share, the
# garrison need its defenders' at theirs), and the defence costs the attacker
# the assault's toll (sr_site_assault_toll_<key>, half the need) on top: the
# seizure pays the need, one more and the toll, the toll dies, the rest land.
# The marines spent and landed go back to battalions at our share, since an
# uncrewed battalion fights for nothing and lands all the same.
#
# Training Grounds (the owner, 2026-11-24): the figures are weighed by each
# side's sr_marine_training_mult too, so what we pay and land goes back to
# battalions at our share and our weight (sr_marine_weight_value), and the dead
# are the toll over our weight: trained marines take a world with fewer
# battalions, and fewer of them die for it.
sr_site_seize = {
	set_local_variable = {
		name = sr_cb_pay
		value = {
			value = sr_site_garrison_need_$KEY$
			add = 1
			add = sr_site_assault_toll_$KEY$
			max = var:sr_cb_mar_$KEY$
		}
	}
	set_local_variable = {
		name = sr_cb_toll
		value = {
			value = sr_site_assault_toll_$KEY$
			max = local_var:sr_cb_pay
		}
	}
	set_local_variable = {
		name = sr_cb_landed
		value = {
			value = local_var:sr_cb_pay
			subtract = local_var:sr_cb_toll
		}
	}
	set_local_variable = {
		name = sr_pers_lost
		value = {
			value = local_var:sr_cb_toll
			multiply = sr_pers_per_battalion
			divide = sr_marine_training_mult
		}
	}
	sr_pers_note_people = yes
	# The occupation notice (.102) names them; a free colony's seizure has its own (.104).
	if = {
		limit = {
			local_var:sr_pers_lost > 0
			NOT = { has_global_variable = sr_site_free_$KEY$ }
		}
		sr_pers_note_occ_dead = yes
	}
	if = {
		limit = {
			sr_marine_weight_value > 0
			NOT = { sr_marine_weight_value = 1 }
		}
		set_local_variable = { name = sr_cb_pay value = { value = local_var:sr_cb_pay divide = sr_marine_weight_value } }
		set_local_variable = { name = sr_cb_landed value = { value = local_var:sr_cb_landed divide = sr_marine_weight_value } }
	}
	sr_fl_spend_marines_$KEY$ = yes
	if = {
		limit = { has_global_variable = sr_site_free_$KEY$ }
		sr_site_take_free = { KEY = $KEY$ INDEX = $INDEX$ }
	}
	else_if = {
		limit = {
			has_global_variable = sr_site_occupier_$KEY$
			OR = {
				global_var:sr_site_holder_$KEY$ ?= this
				global_var:sr_site_holder_$KEY$ ?= { NOT = { has_war_with = prev } }
			}
		}
		sr_site_liberate = { KEY = $KEY$ INDEX = $INDEX$ }
	}
	else = {
		sr_site_occupy = { KEY = $KEY$ INDEX = $INDEX$ }
	}
}

# THIS occupies the world KEY: its landed marines are the garrison, the
# holder's are lost (on the Moon and Mars, all its marines there, which
# defended it: sr_site_lose_shared_marines_<key>, generated, before the
# occupier is set); the world gives its modifiers to no one (the sync), and
# counts in war support (§ 10).
sr_site_occupy = {
	sr_site_note_garrison_dead = { KEY = $KEY$ }
	sr_site_lose_shared_marines_$KEY$ = yes
	set_global_variable = { name = sr_site_occupier_$KEY$ value = this }
	set_global_variable = { name = sr_site_marines_$KEY$ value = local_var:sr_cb_landed }
	if = { limit = { has_global_variable = sr_site_bombard_$KEY$ } remove_global_variable = sr_site_bombard_$KEY$ }
	sr_site_sync_$KEY$ = yes
	sr_site_occupation_notice = { KEY = $KEY$ INDEX = $INDEX$ KIND = 1 }
	sr_colony_un_on_seizure = yes
	sr_sites_refresh_players = yes
	set_global_variable = { name = sr_cb_log_world value = $INDEX$ }
	debug_log = "TE_SR_COMBAT world [GetGlobalVariable('sr_cb_log_world').GetValue|0] occupied by [THIS.GetCountry.GetTagName]"
}

# THIS (the holder, or one not at war with it) frees the world KEY: the
# occupier's garrison is gone, and dies (space forces personnel design § 5.1),
# the liberators' marines stand there.
sr_site_liberate = {
	sr_site_note_garrison_dead = { KEY = $KEY$ }
	sr_site_occupation_notice = { KEY = $KEY$ INDEX = $INDEX$ KIND = 3 }
	remove_global_variable = sr_site_occupier_$KEY$
	set_global_variable = { name = sr_site_marines_$KEY$ value = local_var:sr_cb_landed }
	sr_site_sync_$KEY$ = yes
	sr_sites_refresh_players = yes
	set_global_variable = { name = sr_cb_log_world value = $INDEX$ }
	debug_log = "TE_SR_COMBAT world [GetGlobalVariable('sr_cb_log_world').GetValue|0] liberated by [THIS.GetCountry.GetTagName]"
}

# THIS takes the free colony KEY (§ 9.4): ours at once, at autonomy 60, every
# country's ties to it cleared, its suppliers told (.104).
sr_site_take_free = {
	every_country = {
		limit = {
			is_player = yes
			has_variable = sr_free_supply_$KEY$
			NOT = { this = prev }
		}
		sr_site_event_open = { VAR = sr_seized_free_site INDEX = $INDEX$ }
		trigger_event = { id = space_race_events.104 days = 1 }
	}
	remove_global_variable = sr_site_free_$KEY$
	if = { limit = { has_global_variable = sr_site_suppliers_$KEY$ } remove_global_variable = sr_site_suppliers_$KEY$ }
	sr_site_clear_ties_$KEY$ = yes
	set_global_variable = { name = sr_site_holder_$KEY$ value = this }
	set_global_variable = { name = sr_site_autonomy_$KEY$ value = sr_free_seized_autonomy }
	set_global_variable = { name = sr_site_contact_$KEY$ value = te_history_month_index }
	set_global_variable = { name = sr_site_marines_$KEY$ value = local_var:sr_cb_landed }
	sr_site_sync_$KEY$ = yes
	sr_site_gained_quiet = { INDEX = $INDEX$ }
	if = {
		limit = { is_player = yes }
		sr_site_event_open = { VAR = sr_seized_free_site INDEX = $INDEX$ }
		trigger_event = { id = space_race_events.104 days = 1 }
	}
	sr_colony_un_on_seizure = yes
	sr_sites_refresh_players = yes
	set_global_variable = { name = sr_cb_log_world value = $INDEX$ }
	debug_log = "TE_SR_COMBAT world [GetGlobalVariable('sr_cb_log_world').GetValue|0] (a free colony) seized by [THIS.GetCountry.GetTagName]"
}

# Country scope, the destroyer (THIS), the world KEY (§ 9.3): the colony and its
# people lost and the world ruined for 25 years; 20 infamy (twice for a party
# to the outer-space convention), 20 on the UN dossier's hostile acts in
# space, and its own colonies' resentment: +10 autonomy at once and a drift
# term for ten years. Both sides are told (.103).
sr_site_destroy = {
	global_var:sr_site_holder_$KEY$ ?= {
		if = {
			limit = {
				is_country_alive = yes
				is_player = yes
			}
			sr_site_event_open = { VAR = sr_destroyed_site INDEX = $INDEX$ }
			trigger_event = { id = space_race_events.103 days = 1 }
		}
	}
	if = {
		limit = { is_player = yes }
		sr_site_event_open = { VAR = sr_destroyed_site INDEX = $INDEX$ }
		trigger_event = { id = space_race_events.103 days = 1 }
	}
	change_infamy = sr_destroy_infamy_cost
	un_dossier_record = { RECORD = space POINTS = sr_destroy_dossier_cost }
	sr_colony_un_on_seizure = yes
	set_local_variable = { name = sr_aut_delta value = sr_destroy_autonomy_now }
	sr_site_aut_add_all = yes
	set_variable = {
		name = sr_destroyed_colony_until
		value = {
			value = te_history_month_index
			add = 120
		}
	}
	# The garrison dies with the colony (space forces personnel design § 5.1).
	sr_site_note_destroyed_dead = { KEY = $KEY$ }
	sr_site_release_$KEY$ = yes
	set_global_variable = { name = sr_site_ruined_$KEY$ value = yes days = 9125 }
	sr_sites_recompute_stage = yes
	sr_sites_refresh_players = yes
	set_global_variable = { name = sr_cb_log_world value = $INDEX$ }
	debug_log = "TE_SR_COMBAT world [GetGlobalVariable('sr_cb_log_world').GetValue|0] destroyed by [THIS.GetCountry.GetTagName]"
}

# Global scope, the world's step (KEY, INDEX): an occupation goes back at the
# peace (§ 9.2): the occupier and the holder no longer at war, or the occupier
# gone. Its garrison comes home.
sr_site_occupation_step = {
	if = {
		limit = {
			OR = {
				NOT = { global_var:sr_site_occupier_$KEY$ ?= { is_country_alive = yes } }
				NOT = { has_global_variable = sr_site_holder_$KEY$ }
				global_var:sr_site_occupier_$KEY$ ?= global_var:sr_site_holder_$KEY$
				global_var:sr_site_occupier_$KEY$ ?= {
					NOT = { has_war_with = global_var:sr_site_holder_$KEY$ }
				}
			}
		}
		sr_site_end_occupation = { KEY = $KEY$ INDEX = $INDEX$ KIND = 2 }
	}
}

sr_site_end_occupation = {
	sr_site_occupation_notice = { KEY = $KEY$ INDEX = $INDEX$ KIND = $KIND$ }
	if = {
		limit = {
			has_global_variable = sr_site_marines_$KEY$
			NOT = { global_var:sr_site_occupier_$KEY$ ?= global_var:sr_site_holder_$KEY$ }
		}
		global_var:sr_site_occupier_$KEY$ ?= {
			if = {
				limit = { is_country_alive = yes }
				sr_fl_marines_home = { AMOUNT = sr_site_marines_value_$KEY$ }
			}
		}
		remove_global_variable = sr_site_marines_$KEY$
	}
	remove_global_variable = sr_site_occupier_$KEY$
	sr_site_sync_$KEY$ = yes
	sr_sites_refresh_players = yes
}

# Global scope, the world's step: integrity comes back while nothing assaults
# it (5 a month, 10 with a Spaceport; § 7.3). The assault mark is the month it
# was last assaulted, kept for the panel; a month's step after one regains.
sr_site_integrity_step = {
	if = {
		limit = {
			has_global_variable = sr_site_integrity_$KEY$
			NOT = {
				AND = {
					has_global_variable = sr_site_under_assault_$KEY$
					global_var:sr_site_under_assault_$KEY$ >= te_history_month_index
				}
			}
		}
		if = {
			limit = { sr_site_port_value_$KEY$ >= 1 }
			set_global_variable = { name = sr_site_integrity_$KEY$ value = { value = global_var:sr_site_integrity_$KEY$ add = sr_integrity_regain_port } }
		}
		else = {
			set_global_variable = { name = sr_site_integrity_$KEY$ value = { value = global_var:sr_site_integrity_$KEY$ add = sr_integrity_regain } }
		}
		if = {
			limit = { global_var:sr_site_integrity_$KEY$ >= sr_integrity_full }
			remove_global_variable = sr_site_integrity_$KEY$
		}
	}
}

# Global scope, from the location's step (KEY, LOC): a world is blockaded (§ 8)
# when a fleet at war with its holder stands off there, or (6c, § 12) a fleet
# interdicting its holder does (sr_icpt_<loc>), and no fleet of its holder is
# stationed or standing off there: an interdiction that meets the target's
# fleet is a standoff. Not at Earth, where nobody stands off: the Moon's worlds
# take sr_site_lunar_blockade_step instead. An occupied world, or a free one,
# is never blockaded. sr_site_interdicted_<key> marks a blockade by
# interdiction alone, no enemy standing off, for the panel.
sr_site_blockade_step = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_site_occupier_$KEY$ }
			global_var:sr_site_holder_$KEY$ ?= {
				is_country_alive = yes
				NOT = { sr_cb_holds_orbit = yes }
				has_global_variable_list = sr_fleet_powers
				save_temporary_scope_as = sr_icpt_victim
				any_in_global_list = {
					variable = sr_fleet_powers
					has_variable = sr_cb_present
					sr_cb_s2_value > 0
					OR = {
						has_war_with = prev
						sr_icpt_interdicts_at = { LOC = $LOC$ }
					}
				}
			}
		}
		if = {
			limit = { NOT = { has_global_variable = sr_site_blockaded_$KEY$ } }
			set_global_variable = { name = sr_site_blockaded_$KEY$ value = yes }
			sr_sites_refresh_players = yes
		}
		if = {
			limit = {
				global_var:sr_site_holder_$KEY$ ?= {
					NOT = {
						any_in_global_list = {
							variable = sr_fleet_powers
							has_variable = sr_cb_present
							sr_cb_s2_value > 0
							has_war_with = prev
						}
					}
				}
			}
			if = {
				limit = { NOT = { has_global_variable = sr_site_interdicted_$KEY$ } }
				set_global_variable = { name = sr_site_interdicted_$KEY$ value = yes }
			}
		}
		else_if = {
			limit = { has_global_variable = sr_site_interdicted_$KEY$ }
			remove_global_variable = sr_site_interdicted_$KEY$
		}
	}
	else_if = {
		limit = {
			OR = {
				has_global_variable = sr_site_blockaded_$KEY$
				has_global_variable = sr_site_interdicted_$KEY$
			}
		}
		if = { limit = { has_global_variable = sr_site_blockaded_$KEY$ } remove_global_variable = sr_site_blockaded_$KEY$ }
		if = { limit = { has_global_variable = sr_site_interdicted_$KEY$ } remove_global_variable = sr_site_interdicted_$KEY$ }
		sr_sites_refresh_players = yes
	}
}

# Global scope, from Earth's step (KEY, a lunar world; lunar colonization
# § 8.2): blockaded while its holder's Earth is (sr_earth_blockaded: a fleet at
# war with it at Earth and none of its own; its batteries and Orbital
# Battlestation don't lift it). It replaces the generic step at Earth, whose
# test (an enemy standing off) never holds there, since every fleet at Earth
# is stationed or closing in. Never interdicted (§ 8.3), so nothing reads or
# clears sr_site_interdicted_<key> for a lunar world: a read of a variable
# nothing sets is logged at load (#1243), and none was ever set, since build
# 2a's generic step at Earth never held either. An occupied world, or a free
# one, is never blockaded.
sr_site_lunar_blockade_step = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_site_occupier_$KEY$ }
			global_var:sr_site_holder_$KEY$ ?= {
				is_country_alive = yes
				sr_earth_blockaded = yes
			}
		}
		if = {
			limit = { NOT = { has_global_variable = sr_site_blockaded_$KEY$ } }
			set_global_variable = { name = sr_site_blockaded_$KEY$ value = yes }
			sr_sites_refresh_players = yes
		}
	}
	else_if = {
		limit = { has_global_variable = sr_site_blockaded_$KEY$ }
		remove_global_variable = sr_site_blockaded_$KEY$
		sr_sites_refresh_players = yes
	}
}

# ---- Orders (design §§ 7, 9, 12) ------------------------------------------------------------------
# Country scope (sr_fleet_set_posture_sgui, UI pass design § 7; $N$): the
# posture buttons set scope:pick's posture at once. Close in takes the first
# hostile world here as its target unless its target is one (the hostile step,
# sr_fl_step_hostile_target_at), begins a free colony's price (§ 9.4, once per
# fleet and world) and drops the interdiction's record; Interdict (6c, § 12)
# drops Destroy, takes the first world here it may interdict unless its target
# is one, and begins the interdiction's price (sr_cb_charge_interdict, once per
# fleet and target); Station and Stand off drop Destroy and the interdiction's
# record, as leaving Close in or Interdict does. The scripted GUI holds each
# pick to its rules (at Earth, lunar colonization § 8.1, Station and Close in
# alone), so none lands where it can't.
sr_fl_set_posture_at = {
	if = {
		limit = { scope:pick = 3 }
		set_variable = { name = sr_fl_$N$_posture value = 3 }
		if = { limit = { has_variable = sr_fl_$N$_icpt } remove_variable = sr_fl_$N$_icpt }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = sr_fl_$N$_target }
					NOT = { sr_fl_$N$_target_hostile = yes }
				}
			}
			sr_fl_step_hostile_target_at = { N = $N$ }
		}
		sr_fl_charge_free_at = { N = $N$ }
	}
	else_if = {
		limit = { scope:pick = 4 }
		set_variable = { name = sr_fl_$N$_posture value = 4 }
		if = { limit = { has_variable = sr_fl_$N$_destroy } remove_variable = sr_fl_$N$_destroy }
		if = {
			limit = { NOT = { sr_fl_$N$_target_interdictable = yes } }
			sr_fl_step_icpt_target_at = { N = $N$ }
		}
		sr_fl_charge_icpt_at = { N = $N$ }
	}
	else_if = {
		limit = {
			OR = {
				scope:pick = 1
				scope:pick = 2
			}
		}
		if = {
			limit = { scope:pick = 1 }
			set_variable = { name = sr_fl_$N$_posture value = 1 }
		}
		else = {
			set_variable = { name = sr_fl_$N$_posture value = 2 }
		}
		if = { limit = { has_variable = sr_fl_$N$_destroy } remove_variable = sr_fl_$N$_destroy }
		if = { limit = { has_variable = sr_fl_$N$_icpt } remove_variable = sr_fl_$N$_icpt }
	}
	sr_icpt_rebuild = yes
}

sr_fl_toggle_destroy_at = {
	if = {
		limit = { has_variable = sr_fl_$N$_destroy }
		remove_variable = sr_fl_$N$_destroy
	}
	else = {
		set_variable = { name = sr_fl_$N$_destroy value = yes }
	}
}

# ---- The AI's defence (design § 14, 6b) -------------------------------------------------------------
# Country scope, an AI fleet power's month. At war the mix is warships 3,
# siege 1, carriers 1, tenders 1; at peace the start mix. A fleet short of
# supply or of the propellant to come home is recalled first; then each
# location where it holds colonies and an enemy fleet is present or on its way
# gets its strongest available fleet, stationed, if that fleet arrives within
# six months or before the first hostile landing, and can come home again or
# dock there (sr_fl_ai_defend_<loc>, generated). Every order passes the same
# sr_fl_<n>_may_depart_<loc> a player's does.
sr_fl_ai_defend = {
	if = {
		limit = { is_at_war = yes }
		set_variable = { name = sr_sc_mix_war value = 3 }
		set_variable = { name = sr_sc_mix_siege value = 1 }
		set_variable = { name = sr_sc_mix_carrier value = 1 }
		set_variable = { name = sr_sc_mix_tender value = 1 }
		set_variable = { name = sr_sc_mix_rapid value = 0 }
	}
	else_if = {
		limit = { has_variable = sr_sc_mix_war }
		remove_variable = sr_sc_mix_war
		if = { limit = { has_variable = sr_sc_mix_siege } remove_variable = sr_sc_mix_siege }
		if = { limit = { has_variable = sr_sc_mix_carrier } remove_variable = sr_sc_mix_carrier }
		if = { limit = { has_variable = sr_sc_mix_tender } remove_variable = sr_sc_mix_tender }
		if = { limit = { has_variable = sr_sc_mix_rapid } remove_variable = sr_sc_mix_rapid }
	}
	sr_fl_each_slot = { DO = ai_recall }
	sr_fl_ai_defend_all = yes
}

# ---- The AI's offence at war (design § 14, 6c) ----------------------------------------------------
# Country scope, an AI fleet power's month, after the defence: at war, one
# docked fleet free of the defence goes to attack a location within twelve
# months' leg where it outmatches the enemy's force 1.5 times over, with the
# propellant for the leg home or a dock of ours there (sr_fl_ai_attack_all,
# generated); every fleet at a location with an enemy world, and no colony of
# ours threatened there, closes in on the world of most worth to seize it when
# its marines suffice, else stands off (sr_fl_ai_assault_at). It never destroys
# a colony, never closes in on a free one, and never interdicts: Interdict is
# no AI posture. Every order passes the same sr_fl_<n>_may_depart_<loc> a
# player's does, and the recall (sr_fl_ai_recall_at) brings home first a fleet
# short of supply or of the propellant home. First, at Earth (lunar
# colonization § 8.4, sr_fl_ai_lunar_assault, generated): its strongest fleet
# there closes in on the enemy's lunar world of most worth while the enemy's
# Earth is blockaded, to seize it when its marines suffice; else it stays
# stationed. The dispatch then leaves that fleet where it is.
sr_fl_ai_attack = {
	# No Space Command, no orders (UI pass design § 10): the lunar assault and
	# the offensive set postures directly, so they wait as a player does.
	if = {
		limit = { is_at_war = yes sr_sc_operational = yes }
		sr_fl_ai_lunar_assault = yes
		sr_fl_ai_attack_all = yes
		sr_fl_each_slot = { DO = ai_assault }
	}
}

# A fleet away and undocked, short of supply (under three months) or of the
# propellant home with a margin, goes home now.
sr_fl_ai_recall_at = {
	if = {
		limit = {
			has_variable = sr_fl_$N$_loc
			NOT = { var:sr_fl_$N$_loc = 1 }
			NOT = { sr_fl_$N$_docked = yes }
			sr_fl_$N$_may_depart_earth = yes
			OR = {
				sr_fl_$N$_supply_value < 3
				sr_fl_$N$_dv_value < { value = sr_fl_$N$_leg_dv_earth multiply = 1.25 }
			}
		}
		sr_fl_dispatch = { N = $N$ LOC = earth CODE = 1 }
	}
}
