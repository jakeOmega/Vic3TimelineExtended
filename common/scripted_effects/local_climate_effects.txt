# Local Climate (docs/superpowers/specs/2026-11-01-local-climate-design.md § 5.1).
# State scope, called only from local_climate_on_actions.txt: yearly on every state
# (local_climate_yearly_on_action), and monthly on a state that has never had it
# (local_climate_first_on_action: a new game's first month, a save from before this,
# a newly colonized or split-off state). Both are state pulses, so ROOT is the state
# and add_modifier's multiplier reads it; on_game_started has no ROOT and would
# apply x1 (scripting_best_practices.md § "`add_modifier { multiplier = var:X }`
# Resolves Against ROOT").
#
# te_climate_penalty_applied is the penalty in force and te_climate_bonus_applied its
# negative part as a positive percent (a mild winter, § 12), both set on every refresh and
# never removed: the modifiers' multipliers read them on later ticks too, and the state
# panel's Comfort rows show them. te_climate_base_penalty is the penalty before any relief,
# for the Map Room's Base Comfort map (te_journal_maps.txt); the first application keys on
# it, the newest of the three, so a save from before it fills the map within a month. At most one of the two modifiers applies. At 0.5 and under the state carries no modifier: the game shows
# the percentage whole, and the attraction breakdown would list "Local Climate -0%". The tile's
# Comfort rows gate on the same > 0.5 (a GUI has no >=), so the two agree at exactly 0.5.
te_update_local_climate_modifier = {
	remove_modifier_if_exists_effect = { MODIFIER = local_climate_discomfort }
	remove_modifier_if_exists_effect = { MODIFIER = local_climate_amenity }
	set_variable = { name = te_climate_penalty_applied value = te_climate_penalty }
	# te_climate_bonus is the penalty's negative part as a positive percent, read here from the
	# penalty just stored: evaluating it would compute the penalty again, a pass over eight
	# state_region chains on every state each year (a third of this refresh's time in a 1931
	# game, docs/audits/late_game_performance.md).
	set_variable = {
		name = te_climate_bonus_applied
		value = {
			value = var:te_climate_penalty_applied
			multiply = -1
			min = 0
		}
	}
	set_variable = { name = te_climate_base_penalty value = te_climate_penalty_raw }
	if = {
		limit = { var:te_climate_penalty_applied > 0.5 }
		add_modifier = { name = local_climate_discomfort multiplier = var:te_climate_penalty_applied }
	}
	else_if = {
		limit = { var:te_climate_bonus_applied > 0.5 }
		add_modifier = { name = local_climate_amenity multiplier = var:te_climate_bonus_applied }
	}
}
