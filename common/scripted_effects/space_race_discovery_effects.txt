# Rare colony discoveries. Country scope; flags are rolled once at the actual
# claim, before the founding event, independently of the specialization chosen.
# Existing colonies are not retroactively seeded. Game odds are not scientific
# estimates. The survey and spacing clocks expire in calendar days.
sr_seed_colony_discovery = {
	if = {
		limit = { NOT = { has_variable = sr_discovery_$DISCOVERY$_rolled } }
		set_variable = { name = sr_discovery_$DISCOVERY$_rolled value = yes }
		random = {
			chance = 10
			set_variable = { name = sr_discovery_$DISCOVERY$_pending value = yes }
			set_variable = { name = sr_discovery_$DISCOVERY$_survey value = yes days = 365 }
		}
	}
}

sr_roll_colony_discovery = {
	if = {
		limit = {
			has_variable = sr_discovery_$DISCOVERY$_pending
			NOT = { has_variable = sr_discovery_$DISCOVERY$_done }
			NOT = { has_variable = sr_discovery_$DISCOVERY$_survey }
			NOT = { has_variable = sr_discovery_cooldown }
		}
		random = {
			chance = $CHANCE$
			# Reserve the spacing clock before dispatch, so another candidate
			# cannot queue a second discovery in this monthly pulse.
			set_variable = { name = sr_discovery_cooldown value = yes days = 180 }
			trigger_event = { id = $EVENT$ }
		}
	}
}

sr_finish_colony_discovery = {
	remove_variable = sr_discovery_$DISCOVERY$_pending
	set_variable = { name = sr_discovery_$DISCOVERY$_done value = yes }
}

# Colony surveys continue in passive mode and after all sites are claimed.
