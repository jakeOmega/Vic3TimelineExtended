# ============================================================================
# UN AUTHORITY — DISPLAY-ONLY TOOLTIP BUILDERS
# ============================================================================
# Read-only, like un_chamber_display_effects.txt: nothing but custom_tooltip
# lines and scope navigation, rendered by the scripted GUIs in
# common/scripted_guis/un_authority_sguis.txt through ExecuteTooltip, which
# renders an effect without running it. NOTHING HERE MAY WRITE GAMEPLAY STATE.
#
# Every figure is an O(1) read of a snapshot the monthly update wrote (the
# un_disp_* script values in un_authority_values.txt), never a live world
# sweep: the journal entry re-renders every frame it is on screen.
#
# Ordering: tooltip generation prints a scope's own lines before the lines
# its nested scope blocks produce (found in the chamber's first play test).
# So each builder below either prints only root-level lines, or only nested
# lines that are self-describing — never a header meant to sit above a list.
# ============================================================================

# The empty states only: the UN not founded (never, or since a dissolution,
# or while a founding conference sits), or founded but not yet through its
# first monthly update. Once the pillars exist the widget draws them as a GUI
# table instead (gated by un_authority_ready_sgui), so this prints nothing.
# Scope: country (the viewer)
un_authority_summary_block = {
	if = {
		limit = {
			has_global_variable = un_founded
			NOT = { has_global_variable = un_authority_target }
		}
		custom_tooltip_no_bullet = je_un_auth_pending
	}
	else_if = {
		limit = { un_conference_in_session = yes }
		custom_tooltip_no_bullet = je_un_auth_conference
	}
	else_if = {
		limit = { un_founding_is_refounding = yes }
		if = {
			limit = { has_global_variable = un_refounding_cooldown }
			custom_tooltip_no_bullet = je_un_auth_dissolved_cooldown
		}
		else = {
			custom_tooltip_no_bullet = je_un_auth_dissolved_open
		}
	}
	else_if = {
		limit = { NOT = { has_global_variable = un_founded } }
		custom_tooltip_no_bullet = je_un_auth_not_founded
	}
}

# Redesign phase 2: the tier and its enforcement, the charter's ceiling, and
# how far the next charter reform has ripened. Root-level lines only.
# Scope: country (the viewer)
un_authority_ladder_block = {
	if = {
		limit = {
			has_global_variable = un_founded
			has_global_variable = un_tier
		}
		if = {
			limit = { un_tier_is_supranational = yes }
			custom_tooltip_no_bullet = je_un_auth_tier_supranational
		}
		else_if = {
			limit = { un_tier_at_least_strong = yes }
			custom_tooltip_no_bullet = je_un_auth_tier_strong
		}
		else_if = {
			limit = { un_tier_at_least_established = yes }
			custom_tooltip_no_bullet = je_un_auth_tier_established
		}
		else_if = {
			limit = { un_tier_at_least_contested = yes }
			custom_tooltip_no_bullet = je_un_auth_tier_contested
		}
		else = {
			custom_tooltip_no_bullet = je_un_auth_tier_moribund
		}
		if = {
			limit = { un_charter_level_value >= 2 }
			custom_tooltip_no_bullet = je_un_auth_charter_2
		}
		else_if = {
			limit = { un_charter_level_value >= 1 }
			custom_tooltip_no_bullet = je_un_auth_charter_1
		}
		else = {
			custom_tooltip_no_bullet = je_un_auth_charter_0
		}
		if = {
			limit = { un_charter_reform_ripe = yes }
			custom_tooltip_no_bullet = je_un_auth_reform_ripe
		}
		else_if = {
			limit = { un_charter_can_be_reformed = yes }
			custom_tooltip_no_bullet = je_un_auth_reform_clock
		}
	}
}

# Redesign phase 2: the crisis — its clock, and every great power that could
# save the organisation, with what its act would add to the target. Shown only
# while the crisis is open (un_authority_crisis_sgui's is_shown). The clock
# line is root-level and prints first; the powers are nested, each line
# self-describing.
# Scope: country (the viewer)
un_authority_crisis_block = {
	if = {
		limit = { un_crisis_is_open = yes }
		if = {
			limit = { un_crisis_clock_running = yes }
			custom_tooltip_no_bullet = je_un_auth_crisis_clock
		}
		else = {
			custom_tooltip_no_bullet = je_un_auth_crisis_linger
		}
		every_country = {
			limit = {
				country_rank >= rank_value:great_power
				is_subject = no
				NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
				has_variable = un_power_share
			}
			custom_tooltip = je_un_auth_saviour_join
		}
		every_country = {
			limit = {
				je:je_united_nations ?= { has_modifier = un_undermine_order_cost }
				has_variable = un_power_share
			}
			custom_tooltip = je_un_auth_saviour_stop_undermining
		}
		every_country = {
			limit = {
				country_rank >= rank_value:great_power
				je:je_united_nations ?= { has_modifier = un_member_modifier }
				NOT = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
				NOT = { je:je_united_nations ?= { has_modifier = un_undermine_order_cost } }
				has_variable = un_power_share
			}
			custom_tooltip = je_un_auth_saviour_champion
		}
	}
}

# Who is driving the commitment and participation pillars: every champion,
# every underminer, and every great power outside the UN, each line saying
# which it is. Scope: country (the viewer)
un_authority_powers_block = {
	if = {
		limit = {
			has_global_variable = un_founded
			any_country = {
				OR = {
					je:je_united_nations ?= { has_modifier = un_champion_order_cost }
					je:je_united_nations ?= { has_modifier = un_undermine_order_cost }
					AND = {
						country_rank >= rank_value:great_power
						NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
					}
				}
			}
		}
		every_country = {
			limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
			custom_tooltip = je_un_auth_power_champion
		}
		every_country = {
			limit = { je:je_united_nations ?= { has_modifier = un_undermine_order_cost } }
			custom_tooltip = je_un_auth_power_underminer
		}
		every_country = {
			limit = {
				country_rank >= rank_value:great_power
				NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
			}
			custom_tooltip = je_un_auth_power_outsider
		}
	}
	else_if = {
		limit = { has_global_variable = un_founded }
		custom_tooltip_no_bullet = je_un_auth_powers_none
	}
}

# The recent ledger entries, newest first, two lines each. Scope: country
un_authority_log_block = {
	if = {
		limit = { has_global_variable_list = un_ledger_log }
		ordered_in_global_list = {
			variable = un_ledger_log
			order_by = un_ledger_log_recency_order
			max = 12
			un_authority_log_entry = yes
		}
	}
	else = {
		custom_tooltip_no_bullet = je_un_auth_log_empty
	}
}

# One entry: "<year> <who>: <pillar> <amount>", then the reason. The actor is
# stashed with save_temporary_scope_as and printed from the container's own
# line, as the chamber prints a resolution's proposer, so an annexed actor
# reads as "a former member" instead of a blank.
# Scope: ledger log container
un_authority_log_entry = {
	if = {
		limit = { exists = var:un_led_actor }
		var:un_led_actor ?= { save_temporary_scope_as = un_led_actor_disp }
		if = {
			limit = { has_tag = un_led_credibility }
			custom_tooltip = je_un_led_actor_credibility
		}
		else_if = {
			limit = { has_tag = un_led_delivery }
			custom_tooltip = je_un_led_actor_delivery
		}
		else_if = {
			limit = { has_tag = un_led_order }
			custom_tooltip = je_un_led_actor_order
		}
		else = {
			custom_tooltip = je_un_led_actor_direct
		}
	}
	else = {
		if = {
			limit = { has_tag = un_led_credibility }
			custom_tooltip = je_un_led_assembly_credibility
		}
		else_if = {
			limit = { has_tag = un_led_delivery }
			custom_tooltip = je_un_led_assembly_delivery
		}
		else_if = {
			limit = { has_tag = un_led_order }
			custom_tooltip = je_un_led_assembly_order
		}
		else = {
			custom_tooltip = je_un_led_assembly_direct
		}
	}
	un_authority_log_reason_line = yes
}
