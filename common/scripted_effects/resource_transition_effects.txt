# ============================================================================
# RESOURCE TRANSITION — effects (issue #660)
# ============================================================================
# Triggers and the industry table: common/scripted_triggers/resource_transition_triggers.txt.
# Values and tuning: common/script_values/resource_transition_values.txt.
#
# RETIREMENT PROGRAMMES. Under Managed Fossil Phaseout a country may run one
# programme per industry (coal mines, oil rigs, power plants). A running
# programme retires the industry's smallest remaining building every
# RT_RETIRE_INTERVAL_MONTHS months, the first one that long after it starts:
#   - remove_building takes the whole building, all its levels and whoever owns
#     them (government, a company, its own investors, a foreign investor).
#     Vanilla's law_extraction_economy dismantles privately owned industry the
#     same way on enactment. Partial, level-by-level retirement is not
#     available: no effect downsizes a building by a level.
#   - the country pays the owners compensation (rt_compensation_<key>_here):
#     the building's construction value at the construction good's base price,
#     times RT_COMPENSATION_SHARE. It leaves the treasury; nothing reaches a
#     foreign owner's treasury.
#   - the state gets Transition Assistance (rt_transition_assistance,
#     RT_ASSISTANCE_YEARS years): retraining and income support for the workers
#     the closure lays off.
#   - a player is told in the feed (rt_<key>_retired_notice).
#
# The programme pulse is a country on_action gated on the law
# (common/on_actions/resource_transition_on_actions.txt), not je_global_warming's
# pulse: the laws work whether or not the entry is active and whatever the
# Global Warming game rule says.
# ============================================================================

# ----------------------------------------------------------------------------
# One retirement. $KEY$ coal / oil / power, $BT$ the building type. Picks the
# state where the building is smallest. Scope: country.
# ----------------------------------------------------------------------------
rt_retire_smallest = {
	if = {
		limit = { any_scope_state = { has_building = $BT$ } }
		ordered_scope_state = {
			limit = { has_building = $BT$ }
			order_by = rt_retire_order_$KEY$
			save_scope_as = rt_retire_state
		}
		scope:rt_retire_state = {
			set_variable = {
				name = rt_retired_cost
				value = rt_compensation_$KEY$_here
				days = 60
			}
			set_variable = {
				name = rt_retired_levels
				value = rt_state_level_$KEY$
				days = 60
			}
		}
		add_treasury = {
			value = scope:rt_retire_state.var:rt_retired_cost
			multiply = -1
		}
		scope:rt_retire_state = {
			remove_building = $BT$
			if = {
				limit = { has_modifier = rt_transition_assistance }
				remove_modifier = rt_transition_assistance
			}
			add_modifier = {
				name = rt_transition_assistance
				years = 5	# RT_ASSISTANCE_YEARS
			}
		}
		if = {
			limit = { is_player = yes }
			post_notification = rt_$KEY$_retired_notice
		}
		clear_saved_scope = rt_retire_state
	}
}

# ----------------------------------------------------------------------------
# Start and stop. Scope: country.
# ----------------------------------------------------------------------------
rt_start_programme = {
	custom_tooltip = {
		text = rt_start_$KEY$_tt
		set_variable = {
			name = rt_prog_$KEY$
			value = 0
		}
	}
	hidden_effect = {
		rt_refresh_display_effect = yes
	}
}

rt_stop_programme = {
	custom_tooltip = {
		text = rt_stop_$KEY$_tt
		remove_variable = rt_prog_$KEY$
	}
	hidden_effect = {
		rt_refresh_display_effect = yes
	}
}

# Every programme ends when the phaseout does. Scope: country.
rt_stop_all_programmes = {
	if = {
		limit = { has_variable = rt_prog_coal }
		remove_variable = rt_prog_coal
	}
	if = {
		limit = { has_variable = rt_prog_oil }
		remove_variable = rt_prog_oil
	}
	if = {
		limit = { has_variable = rt_prog_power }
		remove_variable = rt_prog_power
	}
}

# ----------------------------------------------------------------------------
# The monthly pulse. Scope: country.
# ----------------------------------------------------------------------------
rt_country_monthly_update = {
	if = {
		limit = {
			rt_any_programme_running = yes
			NOT = { has_law = law_type:law_managed_fossil_phaseout }
		}
		rt_stop_all_programmes = yes
	}
	else_if = {
		limit = { rt_any_programme_running = yes }
		rt_programme_tick = { KEY = coal BT = building_coal_mine }
		rt_programme_tick = { KEY = oil BT = building_oil_rig }
		rt_programme_tick = { KEY = power BT = building_power_plant }
	}
	if = {
		limit = { is_player = yes }
		rt_refresh_display_effect = yes
	}
}

# A running programme's month: count it, and retire at the interval.
rt_programme_tick = {
	if = {
		limit = { has_variable = rt_prog_$KEY$ }
		change_variable = {
			name = rt_prog_$KEY$
			add = 1
		}
		if = {
			limit = { var:rt_prog_$KEY$ >= RT_RETIRE_INTERVAL_MONTHS }
			set_variable = {
				name = rt_prog_$KEY$
				value = 0
			}
			rt_retire_smallest = { KEY = $KEY$ BT = $BT$ }
		}
	}
}

# ----------------------------------------------------------------------------
# The panel's snapshot (the rt_d_* variables rt_disp_* reads). Players only:
# nobody else looks. Also run on a law change and on every Start or Stop, so
# the section is never a month behind a click. Scope: country.
# ----------------------------------------------------------------------------
rt_refresh_display_effect = {
	if = {
		limit = { is_player = yes }
		set_variable = { name = rt_d_lv_coal value = rt_levels_coal }
		set_variable = { name = rt_d_lv_oil value = rt_levels_oil }
		set_variable = { name = rt_d_lv_power value = rt_levels_power }
		set_variable = { name = rt_d_n_coal value = rt_states_coal }
		set_variable = { name = rt_d_n_oil value = rt_states_oil }
		set_variable = { name = rt_d_n_power value = rt_states_power }
		set_variable = { name = rt_d_lv_clean value = rt_levels_clean_power }
		set_variable = { name = rt_d_lv_synth value = rt_levels_synthetics }
		set_variable = { name = rt_d_all_coal value = rt_compensation_all_coal }
		set_variable = { name = rt_d_all_oil value = rt_compensation_all_oil }
		set_variable = { name = rt_d_all_power value = rt_compensation_all_power }
		rt_refresh_next_target = { KEY = coal BT = building_coal_mine }
		rt_refresh_next_target = { KEY = oil BT = building_oil_rig }
		rt_refresh_next_target = { KEY = power BT = building_power_plant }
	}
}

# The building a programme would retire next, its state and compensation.
# Scope: country.
rt_refresh_next_target = {
	if = {
		limit = { any_scope_state = { has_building = $BT$ } }
		ordered_scope_state = {
			limit = { has_building = $BT$ }
			order_by = rt_retire_order_$KEY$
			save_temporary_scope_as = rt_next_state
		}
		set_variable = {
			name = rt_d_next_$KEY$
			value = scope:rt_next_state.rt_compensation_$KEY$_here
		}
		set_variable = {
			name = rt_d_next_lv_$KEY$
			value = scope:rt_next_state.rt_state_level_$KEY$
		}
		set_variable = {
			name = rt_d_next_state_$KEY$
			value = scope:rt_next_state
		}
	}
	else = {
		if = {
			limit = { has_variable = rt_d_next_$KEY$ }
			remove_variable = rt_d_next_$KEY$
		}
		if = {
			limit = { has_variable = rt_d_next_lv_$KEY$ }
			remove_variable = rt_d_next_lv_$KEY$
		}
		if = {
			limit = { has_variable = rt_d_next_state_$KEY$ }
			remove_variable = rt_d_next_state_$KEY$
		}
	}
}
