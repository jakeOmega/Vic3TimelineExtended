# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The effects behind the map
# buttons: the scripted GUIs in common/scripted_guis/te_journal_map_sguis.txt. Country scope: the player's country.

# The Map Room's pick, read by te_jmap_selector (common/script_values/te_journal_map_values.txt). Global, because a
# state's map_color reaches neither the entry nor a country (the scope probe painted blue, 2026-11-05), so every
# player in a multiplayer game shares it.
te_jmap_set_selector = {
	set_global_variable = { name = te_jmap_sel value = $SEL$ }
	# Company Headquarters paints stored figures: store them now, so a first pick of the month has them.
	if = {
		limit = { global_var:te_jmap_sel ?= 34 } # false when unset: the map buttons' tooltips walk this writing nothing
		te_jmap_store_hqs = yes
	}
	# Investment paints stored figures too.
	if = {
		limit = { global_var:te_jmap_sel ?= 46 }
		te_jmap_store_invest = yes
	}
	# The player who picks is the maps' viewer, and the Map Room's area is drawn around the viewer (its market, its
	# power bloc, its country), so the area is marked again before the range that reads the marks.
	te_jmap_note_viewer = yes
	te_jmap_mark_area = yes
	# The picked map's colors run over the area as it stands now (te_journal_map_ranges_generated.txt), measured
	# here so the reopened page paints on them; the monthly pulse measures them again.
	te_jmap_refresh_range = yes
	debug_log = "TE_JMAP pick $SEL$ date=[TimeKeeper.GetCurrentDate.GetString] (a Map Room pick ran; a pick on the page closes it a tenth of a second later and reopens it a fifth after that, which repaints the map: TE_JMAP reopen)"
}

# The Map Room's options (2026-11-14): Area and Outliers
# ---------------------------------------------------------------------------
# te_jmap_area, te_jmap_continent and te_jmap_trim (te_journal_map_values.txt) read the three globals set here.
# Each option button (te_jmap_option_button, gui/te_map_widgets.gui) runs one of these through its scripted GUI with
# the player as root, then fires the picks' own repaint (te_jmap_repaint): the option, the area's marks and the
# pick's range have landed before the page reopens, so a paused map repaints on them.

# AREA: 0 the world, 1 a continent, 2 the viewer's market, 3 its power bloc, 4 its country.
te_jmap_set_area = {
	set_global_variable = { name = te_jmap_opt_area value = $AREA$ }
	# The first pick of Continent takes the player's capital's.
	if = {
		limit = {
			global_var:te_jmap_opt_area ?= 1 # false when unset: a tooltip walks this writing nothing
			NOT = { has_global_variable = te_jmap_opt_cont }
		}
		te_jmap_default_continent = yes
	}
	te_jmap_apply_options = yes
	debug_log = "TE_JMAP area $AREA$ date=[TimeKeeper.GetCurrentDate.GetString] (a Map Room option ran: the area's states are marked and the pick's range measured again before the page reopens)"
}

# The continent the player's capital stands in; Europe for a country with no capital. Country scope.
te_jmap_default_continent = {
	set_global_variable = { name = te_jmap_opt_cont value = 1 }
	if = {
		limit = { capital ?= { te_jmap_in_continent_asia = yes } }
		set_global_variable = { name = te_jmap_opt_cont value = 2 }
	}
	else_if = {
		limit = { capital ?= { te_jmap_in_continent_africa = yes } }
		set_global_variable = { name = te_jmap_opt_cont value = 3 }
	}
	else_if = {
		limit = { capital ?= { te_jmap_in_continent_north_america = yes } }
		set_global_variable = { name = te_jmap_opt_cont value = 4 }
	}
	else_if = {
		limit = { capital ?= { te_jmap_in_continent_south_america = yes } }
		set_global_variable = { name = te_jmap_opt_cont value = 5 }
	}
	else_if = {
		limit = { capital ?= { te_jmap_in_continent_oceania = yes } }
		set_global_variable = { name = te_jmap_opt_cont value = 6 }
	}
}

# CONT: 1 Europe, 2 Asia, 3 Africa, 4 North America, 5 South America, 6 Oceania. The continent buttons show only
# while the area is a continent.
te_jmap_set_continent = {
	set_global_variable = { name = te_jmap_opt_cont value = $CONT$ }
	te_jmap_apply_options = yes
	debug_log = "TE_JMAP continent $CONT$ date=[TimeKeeper.GetCurrentDate.GetString]"
}

# TRIM: 0 four states at each end (the default), 1 none, 2 a twentieth of the states with a figure at each end.
te_jmap_set_trim = {
	set_global_variable = { name = te_jmap_opt_trim value = $TRIM$ }
	te_jmap_apply_options = yes
	debug_log = "TE_JMAP outliers $TRIM$ date=[TimeKeeper.GetCurrentDate.GetString]"
}

# An option changed. Country scope: the player's, who becomes the viewer the area is drawn around.
te_jmap_apply_options = {
	te_jmap_note_viewer = yes
	te_jmap_mark_area = yes
	te_jmap_refresh_range = yes
}

# The area's marks: te_jmap_out on every state outside the Map Room's area, which the paint greys
# (te_jmap_outside) and the ranges leave out (te_jmap_in_area). The paint may run once per state per frame, so it
# asks only for the mark; what the area is (te_jmap_state_in_area: a continent's strategic regions, the viewer's
# market, power bloc or country) is asked here, once per state, at each option click, each pick and each month from
# the global pulse (te_journal_map_on_actions.txt). So a state that changes hands, a market that splits or a bloc
# that gains a member shows within a month, and at once at the next pick. Under the world no state is marked and
# nothing is walked: te_jmap_marks records that some state may carry a mark to clear. Where the area has lost what
# it is drawn around (te_jmap_area_holds: the viewer left its power bloc, or holds no state), the option goes back
# to the world rather than grey every state. No root: any scope.
te_jmap_mark_area = {
	if = {
		limit = { te_jmap_area_holds = no }
		debug_log = "TE_JMAP area reset from=[GetGlobalVariable('te_jmap_opt_area').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString] (the Map Room's area lost what it is drawn around, the viewer's power bloc or the viewer's last state: back to the world)"
		set_global_variable = { name = te_jmap_opt_area value = 0 }
	}
	if = {
		limit = { te_jmap_area = 0 }
		if = {
			limit = { has_global_variable = te_jmap_marks }
			every_state = {
				limit = { has_variable = te_jmap_out }
				remove_variable = te_jmap_out
			}
			remove_global_variable = te_jmap_marks
		}
	}
	else = {
		set_global_variable = te_jmap_marks
		every_state = {
			if = {
				limit = { te_jmap_state_in_area = yes }
				if = {
					limit = { has_variable = te_jmap_out }
					remove_variable = te_jmap_out
				}
			}
			else = {
				set_variable = te_jmap_out
			}
		}
	}
}

# TE_SELF_CHECK area_mark (docs/testing/journal-map-modes-playtest.md, check 29). The area rests on three reads
# nobody has seen in game: a state's strategic region against a list (region = sr:...), a state's market against the
# viewer's, and a state's owner against the viewer's power bloc. Once a game, the first month's pulse that finds an
# area other than the world while at least 20 states exist (the global monthly pulse, after it marks the area again):
# OK when some states are inside and some outside, and, for an area drawn around the viewer, the viewer's own capital
# is inside; BROKEN when every state is inside (the test passes everything, so the map is not limited), every state
# outside (it passes nothing, so the whole map is greyed), or the viewer's capital is outside its own market, power
# bloc or country. An area that really holds every state would read BROKEN too; none does in a game with 20 states.
# The fallbacks, if BROKEN, by the area the line names: 1, mark by is_in_geographic_region against regions of the
# mod's own (the formables' are the pattern, te_formable_regions.txt); 2, compare the owners' markets (owner.market)
# or walk the viewer's market's states; 3, walk the bloc's members (every_power_bloc_member) and mark their states;
# 4, walk the viewer's own states (every_scope_state). The check speaks once, for the area set at the time; try the
# other areas by eye (check 29). No root: the counts are globals. Not called from a click: a tooltip that walked the
# option effects would read these counts unset.
te_jmap_area_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jmap_area_selfcheck_done }
			te_jmap_area > 0
			any_state = {
				count >= 20
				exists = owner
			}
		}
		set_global_variable = te_jmap_area_selfcheck_done
		set_global_variable = { name = te_jmap_sc_area_in value = 0 }
		set_global_variable = { name = te_jmap_sc_area_out value = 0 }
		every_state = {
			if = {
				limit = { has_variable = te_jmap_out }
				change_global_variable = { name = te_jmap_sc_area_out add = 1 }
			}
			else = {
				change_global_variable = { name = te_jmap_sc_area_in add = 1 }
			}
		}
		if = {
			limit = {
				global_var:te_jmap_sc_area_in > 0
				global_var:te_jmap_sc_area_out > 0
				OR = {
					te_jmap_area = 1
					global_var:te_jmap_viewer ?= {
						capital ?= { te_jmap_in_area = yes }
					}
				}
			}
			debug_log = "TE_SELF_CHECK OK area_mark area=[GetGlobalVariable('te_jmap_opt_area').GetValue|0] inside=[GetGlobalVariable('te_jmap_sc_area_in').GetValue|0] outside=[GetGlobalVariable('te_jmap_sc_area_out').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN area_mark area=[GetGlobalVariable('te_jmap_opt_area').GetValue|0] inside=[GetGlobalVariable('te_jmap_sc_area_in').GetValue|0] outside=[GetGlobalVariable('te_jmap_sc_area_out').GetValue|0] (the Map Room's area test put every state on one side, or left the viewer's own capital outside its market, power bloc or country: the map is not limited, greyed everywhere, or limited to the wrong states; take the fallback for this area in docs/testing/journal-map-modes-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_global_variable = te_jmap_sc_area_in
		remove_global_variable = te_jmap_sc_area_out
	}
}

# TE_SELF_CHECK range_count (docs/testing/journal-map-modes-playtest.md, check 29). Since 2026-11-14 a range's
# measure counts its states first and takes the trim from the count, through a ladder of literal positions
# (te_jmap_measure_at, te_journal_map_ranges_generated.txt), where it took position = 4 outright before. Once a
# game, the first month GDP per capita has a figure in at least 20 states (every state with people has one, and
# they differ from the first day), from the global monthly pulse: it measures GDP per capita's range the new way,
# then takes the fifth highest and fifth lowest the old way, and compares. OK when both ends agree and the high is
# above the low; BROKEN when they differ (the count, the trim or the ladder went wrong, and every fitted map is on
# a wrong range or its fixed curve's ends). It runs only under the default options, the one case with an old answer
# to compare; a game that starts with an option set waits for a month without one. The fallback, if BROKEN: have
# te_jmap_range_<map> call te_jmap_measure_<map> = { K = 4 } outright under the default options, as it did before.
te_jmap_range_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jmap_range_selfcheck_done }
			te_jmap_area = 0
			te_jmap_trim = 0
			any_state = {
				count >= 20
				gdp_per_capita > 0
			}
		}
		set_global_variable = te_jmap_range_selfcheck_done
		te_jmap_range_gdppc = yes
		ordered_state = {
			limit = { gdp_per_capita > 0 }
			order_by = te_jmap_gdppc_input
			position = 4
			check_range_bounds = no
			set_global_variable = { name = te_jmap_sc_rng_hi value = te_jmap_gdppc_input }
		}
		ordered_state = {
			limit = { gdp_per_capita > 0 }
			order_by = te_jmap_gdppc_input_neg
			position = 4
			check_range_bounds = no
			set_global_variable = { name = te_jmap_sc_rng_lo value = te_jmap_gdppc_input }
		}
		if = {
			limit = {
				has_global_variable = te_jmap_rng_gdppc_hi
				has_global_variable = te_jmap_rng_gdppc_lo
				has_global_variable = te_jmap_sc_rng_hi
				has_global_variable = te_jmap_sc_rng_lo
				global_var:te_jmap_rng_gdppc_hi = global_var:te_jmap_sc_rng_hi
				global_var:te_jmap_rng_gdppc_lo = global_var:te_jmap_sc_rng_lo
				global_var:te_jmap_rng_gdppc_hi > global_var:te_jmap_rng_gdppc_lo
			}
			debug_log = "TE_SELF_CHECK OK range_count states=[GetGlobalVariable('te_jmap_n').GetValue|0] cut=[GetGlobalVariable('te_jmap_k').GetValue|0] low=[GetGlobalVariable('te_jmap_rng_gdppc_lo').GetValue|3] high=[GetGlobalVariable('te_jmap_rng_gdppc_hi').GetValue|3] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN range_count states=[GetGlobalVariable('te_jmap_n').GetValue|0] cut=[GetGlobalVariable('te_jmap_k').GetValue|0] (GDP per capita's range, measured through the count and the ladder of positions, is not the fifth highest and fifth lowest state's: every fitted map is on a wrong range; take the fallback in docs/testing/journal-map-modes-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		if = {
			limit = { has_global_variable = te_jmap_sc_rng_hi }
			remove_global_variable = te_jmap_sc_rng_hi
		}
		if = {
			limit = { has_global_variable = te_jmap_sc_rng_lo }
			remove_global_variable = te_jmap_sc_rng_lo
		}
		# The pick's own range, which the monthly pulse measured just before, counted into te_jmap_n and
		# te_jmap_k too; the page reads its copy (te_jmap_shown_n, te_jmap_shown_k), so measure the pick again.
		te_jmap_refresh_range = yes
	}
}

# The maps' viewer: the country a map of "your" deterrent or "your" networks is drawn for (te_jmap_viewer_*,
# common/scripted_triggers/te_journal_map_triggers.txt). A state's map_color can't reach the player (is_player drew
# nothing under owner, 2026-11-05), so the player's country is a global variable, set by every pick and map button and
# each month for a player (te_journal_map_on_actions.txt), so a page opened from the journal has it too. One viewer for
# every player in a multiplayer game, as the pick is.
te_jmap_note_viewer = {
	set_global_variable = { name = te_jmap_viewer value = root }
}

# The Company Headquarters map's figures (34; te_journal_map_values.txt): each state's headquarters, counted by building
# group, their weekly profit summed, and its regional headquarters, as state variables the map, its hover and its range
# read (te_jmap_hq_n, te_jmap_hq_wp, te_jmap_rhq_n), removed where none is left. From the global monthly pulse and a
# pick of the map. No root: any scope.
te_jmap_store_hqs = {
	every_state = {
		if = {
			limit = { te_jmap_hq_walk_count > 0 }
			set_variable = { name = te_jmap_hq_n value = te_jmap_hq_walk_count }
			set_variable = { name = te_jmap_hq_wp value = te_jmap_hq_walk_profit }
		}
		else_if = {
			limit = { has_variable = te_jmap_hq_n }
			remove_variable = te_jmap_hq_n
			remove_variable = te_jmap_hq_wp
		}
		if = {
			limit = { te_jmap_rhq_walk_count > 0 }
			set_variable = { name = te_jmap_rhq_n value = te_jmap_rhq_walk_count }
		}
		else_if = {
			limit = { has_variable = te_jmap_rhq_n }
			remove_variable = te_jmap_rhq_n
		}
	}
}

# TE_SELF_CHECK hq_found (docs/testing/journal-map-modes-playtest.md, check 24). The store finds a headquarters by its
# building group (bg_company_headquarter), because the engine's per-company building types match no has_building on
# the template type, which drew the first build's map all grey and kept hq_profit below from ever arming (2026-11-08).
# Once a game, the first month three countries have a company (each company has a headquarters, so the store must find
# some): OK when it found any, with the count of states; BROKEN when it found none, and the map is grey everywhere.
# Global monthly pulse, after the store: no root, so the counts are globals. The fallback, if BROKEN: read the
# headquarters through the companies instead (every_country, every_company, and the regional ones through
# every_scope_regional_hqs, which reach their buildings).
te_jmap_hq_found_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jmap_hq_found_selfcheck_done }
			any_country = {
				count >= 3
				any_company = { always = yes }
			}
		}
		set_global_variable = te_jmap_hq_found_selfcheck_done
		set_global_variable = { name = te_jmap_sc_hq_found value = 0 }
		every_state = {
			limit = { has_variable = te_jmap_hq_n }
			change_global_variable = { name = te_jmap_sc_hq_found add = 1 }
		}
		if = {
			limit = { global_var:te_jmap_sc_hq_found > 0 }
			debug_log = "TE_SELF_CHECK OK hq_found states=[GetGlobalVariable('te_jmap_sc_hq_found').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString] (the store found company headquarters by building group)"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN hq_found states=0 (three countries have a company, but no state holds a building of group bg_company_headquarter: the Company Headquarters map is grey everywhere; take the fallback in docs/testing/journal-map-modes-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_global_variable = te_jmap_sc_hq_found
	}
}

# TE_SELF_CHECK hq_profit (docs/testing/journal-map-modes-playtest.md, check 24). The Company Headquarters map (34)
# colors a state by its headquarters' weekly_profit (te_jmap_hq_profit), on the unverified fact that an ownership
# building books what the levels it owns pay it, as the building panel's profit. Once a game, the first month at least
# three states hold a headquarters (fewer and an all-zero read proves nothing): OK when any reads a profit other than
# 0, BROKEN when every one reads exactly 0, and the map paints every headquarters as a loss. It arms on the store's
# count, which hq_found checks, not on the profit it tests. Global monthly pulse (te_journal_map_on_actions.txt), after
# the store: no root, so the counts are globals. The fallback, if BROKEN: color by the company's prosperity or its owned
# levels instead (a company is reached from its country, every_company).
te_jmap_hq_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jmap_hq_selfcheck_done }
			any_state = {
				count >= 3
				has_variable = te_jmap_hq_n
			}
		}
		set_global_variable = te_jmap_hq_selfcheck_done
		set_global_variable = { name = te_jmap_sc_hq_states value = 0 }
		set_global_variable = { name = te_jmap_sc_hq_nonzero value = 0 }
		every_state = {
			limit = { has_variable = te_jmap_hq_n }
			change_global_variable = { name = te_jmap_sc_hq_states add = 1 }
			if = {
				limit = {
					NOT = { te_jmap_hq_profit = 0 }
				}
				change_global_variable = { name = te_jmap_sc_hq_nonzero add = 1 }
			}
		}
		if = {
			limit = { global_var:te_jmap_sc_hq_nonzero > 0 }
			debug_log = "TE_SELF_CHECK OK hq_profit states=[GetGlobalVariable('te_jmap_sc_hq_states').GetValue|0] nonzero=[GetGlobalVariable('te_jmap_sc_hq_nonzero').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN hq_profit states=[GetGlobalVariable('te_jmap_sc_hq_states').GetValue|0] nonzero=0 (every company headquarters reads a weekly_profit of 0: an ownership building doesn't book its dividends as profit, so the Company Headquarters map paints every one as a loss; take the fallback in docs/testing/journal-map-modes-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_global_variable = te_jmap_sc_hq_states
		remove_global_variable = te_jmap_sc_hq_nonzero
	}
}

# The Investment map's figures (46; te_journal_map_values.txt): what each state's owners pay into the Investment Pool
# a week, estimated, in two state variables the map, its hover and its range read (te_jmap_inv_cap from its Financial
# Districts and company headquarters, te_jmap_inv_ari from its Manor Houses), removed where both are 0. At a pick of
# the map and from the global monthly pulse while it is the pick (te_jmap_monthly_invest): nothing else reads them, so
# the world's buildings aren't walked for a map nobody is looking at. No root: any scope.
te_jmap_store_invest = {
	every_state = {
		if = {
			limit = {
				OR = {
					te_jmap_invest_walk_cap > 0
					te_jmap_invest_walk_ari > 0
				}
			}
			set_variable = { name = te_jmap_inv_cap value = te_jmap_invest_walk_cap }
			set_variable = { name = te_jmap_inv_ari value = te_jmap_invest_walk_ari }
		}
		else_if = {
			limit = { has_variable = te_jmap_inv_cap }
			remove_variable = te_jmap_inv_cap
			remove_variable = te_jmap_inv_ari
		}
	}
}

# The monthly store, behind the pick: the global monthly pulse calls it before the ranges (te_journal_map_on_actions.txt).
te_jmap_monthly_invest = {
	if = {
		limit = {
			has_global_variable = te_jmap_sel
			global_var:te_jmap_sel = 46
		}
		te_jmap_store_invest = yes
		te_jmap_invest_selfcheck = yes
	}
}

# TE_SELF_CHECK invest_rate (docs/testing/journal-map-modes-playtest.md, check 28). The Investment map (46) multiplies
# an owners' building's profit by the share its owners reinvest, read in the state (modifier:state_capitalists_
# investment_pool_contribution_add), on the unverified fact that the state's read carries the country's base (0.30
# for capitalists, 0.20 for aristocrats: base_values). Once a game, the first month the map is the pick and at least
# three states hold a Financial District, a Manor House or a company headquarters in profit (fewer and an all-zero
# read proves nothing): OK when any of them stores a contribution, BROKEN when none does, and the map is pale
# everywhere. It arms on the buildings' profit, not on the figure it tests. After the store: no root, so the counts
# are globals. The fallback, if BROKEN: read the two shares from the owner (owner.modifier:...), or take the bases.
te_jmap_invest_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jmap_invest_selfcheck_done }
			any_state = {
				count >= 3
				OR = {
					te_jmap_invest_walk_cap_profit > 0
					te_jmap_invest_walk_ari_profit > 0
				}
			}
		}
		set_global_variable = te_jmap_invest_selfcheck_done
		set_global_variable = { name = te_jmap_sc_inv_states value = 0 }
		set_global_variable = { name = te_jmap_sc_inv_paying value = 0 }
		every_state = {
			limit = {
				OR = {
					te_jmap_invest_walk_cap_profit > 0
					te_jmap_invest_walk_ari_profit > 0
				}
			}
			change_global_variable = { name = te_jmap_sc_inv_states add = 1 }
			if = {
				limit = { te_jmap_invest_value > 0 }
				change_global_variable = { name = te_jmap_sc_inv_paying add = 1 }
			}
		}
		if = {
			limit = { global_var:te_jmap_sc_inv_paying > 0 }
			debug_log = "TE_SELF_CHECK OK invest_rate states=[GetGlobalVariable('te_jmap_sc_inv_states').GetValue|0] paying=[GetGlobalVariable('te_jmap_sc_inv_paying').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN invest_rate states=[GetGlobalVariable('te_jmap_sc_inv_states').GetValue|0] paying=0 (owners' buildings are in profit, but every state reads a reinvested share of 0: the state's modifier read doesn't carry the country's base, so the Investment map is pale everywhere; take the fallback in docs/testing/journal-map-modes-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_global_variable = te_jmap_sc_inv_states
		remove_global_variable = te_jmap_sc_inv_paying
	}
}

# The class maps' figures (41-44; te_journal_map_values.txt): each state's Capitalists, Aristocrats, Clergymen and
# Academics, counted in Demographics' yearly walk (te_demog_walks, te_demog_effects.txt), whatever the rule, as Armed
# Forces' soldiers are, and stored as te_dg_capitalists, te_dg_aristocrats, te_dg_clergy and te_dg_academics. Three calls
# of their own, not the census's helpers, which set nothing outside the census (test_demographics_registry.py,
# TestWorkWalk); test_demographics_golden.py leaves their three lines out of the walk it pins.
# State scope, before the walk's pop loop.
te_jmap_class_walk_init = {
	set_local_variable = { name = te_jmap_w_cap value = 0 }
	set_local_variable = { name = te_jmap_w_ari value = 0 }
	set_local_variable = { name = te_jmap_w_cle value = 0 }
	set_local_variable = { name = te_jmap_w_aca value = 0 }
}

# Pop scope, in the walk's pop loop.
te_jmap_class_walk_pop = {
	if = {
		limit = { is_pop_type = capitalists }
		change_local_variable = { name = te_jmap_w_cap add = total_size }
	}
	else_if = {
		limit = { is_pop_type = aristocrats }
		change_local_variable = { name = te_jmap_w_ari add = total_size }
	}
	else_if = {
		limit = { is_pop_type = clergymen }
		change_local_variable = { name = te_jmap_w_cle add = total_size }
	}
	else_if = {
		limit = { is_pop_type = academics }
		change_local_variable = { name = te_jmap_w_aca add = total_size }
	}
}

# State scope, after the walk's pop loop.
te_jmap_class_walk_store = {
	set_variable = { name = te_dg_capitalists value = local_var:te_jmap_w_cap }
	set_variable = { name = te_dg_aristocrats value = local_var:te_jmap_w_ari }
	set_variable = { name = te_dg_clergy value = local_var:te_jmap_w_cle }
	set_variable = { name = te_dg_academics value = local_var:te_jmap_w_aca }
}
