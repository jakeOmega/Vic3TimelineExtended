# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The effects behind the map
# buttons: the scripted GUIs in common/scripted_guis/te_journal_map_sguis.txt. Country scope: the player's country.

# The Map Room's pick, read by te_jmap_selector (common/script_values/te_journal_map_values.txt). Global, because a
# state's map_color reaches neither the entry nor a country (the scope probe painted blue, 2026-11-05), so every
# player in a multiplayer game shares it.
te_jmap_set_selector = {
	set_global_variable = { name = te_jmap_sel value = $SEL$ }
	# The picked map's colors run over the world as it stands now (te_journal_map_ranges_generated.txt), measured
	# here so the reopened page paints on them; the monthly pulse measures them again.
	te_jmap_refresh_range = yes
	te_jmap_note_viewer = yes
	debug_log = "TE_JMAP pick $SEL$ date=[TimeKeeper.GetCurrentDate.GetString] (a Map Room pick ran; a pick on the page closes it a tenth of a second later and reopens it a fifth after that, which repaints the map: TE_JMAP reopen)"
}

# The maps' viewer: the country a map of "your" deterrent or "your" networks is drawn for (te_jmap_viewer_*,
# common/scripted_triggers/te_journal_map_triggers.txt). A state's map_color can't reach the player (is_player drew
# nothing under owner, 2026-11-05), so the player's country is a global variable, set by every pick and map button and
# each month for a player (te_journal_map_on_actions.txt), so a page opened from the journal has it too. One viewer for
# every player in a multiplayer game, as the pick is.
te_jmap_note_viewer = {
	set_global_variable = { name = te_jmap_viewer value = root }
}

# TE_SELF_CHECK hq_profit (docs/testing/journal-map-modes-playtest.md, check 24). The Company Headquarters map (34)
# colors a state by its headquarters' weekly_profit (te_jmap_hq_profit), on the unverified fact that an ownership
# building books what the levels it owns pay it, as the building panel's profit. Once a game, the first month at least
# three states hold a headquarters (fewer and an all-zero read proves nothing): OK when any reads a profit other than
# 0, BROKEN when every one reads exactly 0, and the map paints every headquarters as a loss. Global monthly pulse
# (te_journal_map_on_actions.txt): no root, so the counts are globals. The fallback, if BROKEN: color by the company's
# prosperity or its owned levels instead (a company is reached from its country, every_company).
te_jmap_hq_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jmap_hq_selfcheck_done }
			any_state = {
				count >= 3
				has_building = building_company_headquarter
			}
		}
		set_global_variable = te_jmap_hq_selfcheck_done
		set_global_variable = { name = te_jmap_sc_hq_states value = 0 }
		set_global_variable = { name = te_jmap_sc_hq_nonzero value = 0 }
		every_state = {
			limit = { has_building = building_company_headquarter }
			change_global_variable = { name = te_jmap_sc_hq_states add = 1 }
			if = {
				limit = {
					NOT = { te_jmap_hq_profit = 0 }
				}
				change_global_variable = { name = te_jmap_sc_hq_nonzero add = 1 }
			}
		}
		if = {
			limit = { global_var:te_jmap_sc_hq_nonzero > 0 }
			debug_log = "TE_SELF_CHECK OK hq_profit states=[GetGlobalVariable('te_jmap_sc_hq_states').GetValue|0] nonzero=[GetGlobalVariable('te_jmap_sc_hq_nonzero').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN hq_profit states=[GetGlobalVariable('te_jmap_sc_hq_states').GetValue|0] nonzero=0 (every company headquarters reads a weekly_profit of 0: an ownership building doesn't book its dividends as profit, so the Company Headquarters map paints every one as a loss; take the fallback in docs/testing/journal-map-modes-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		remove_global_variable = te_jmap_sc_hq_states
		remove_global_variable = te_jmap_sc_hq_nonzero
	}
}
