# THIS = a state; reads the wealth-band sums te_demog_walks wrote a line earlier.
# Writes var:te_dg_gini, the Gini the panel shows (spec §4.1), with the formula
# it shares with the country (te_demog_gini_from_locals, generated: the bands are
# in income order, so it needs no sort, and the panel shows it with no map).
te_demog_state_gini = {
	te_demog_gini_band_load = yes
	te_demog_gini_from_locals = yes
}

# ============================================================================
# Wealth Concentration (spec §4.2): how much of a state's property belongs to a
# few great fortunes rather than to many owners, 0-100, kept per state.
#
# State variables:
#   te_dg_wc            the score; drifts 3% a year toward te_dg_wc_target and
#                       takes the shocks below
#   te_dg_wc_target     its target (te_demog_wc_target, te_demog_values.txt), as
#                       of the state's last yearly pulse
#   te_dg_wc_sown       the state-owned share of its capital at the last pulse, 0-1
#                       (the ownership shock; absent under 20 capital levels and
#                       after a change of owner)
#   te_dg_ineq          under Full, the Inequality effect's multiplier (the end of
#                       this file)
# Country variables (te_demog_wc_national, the one refresh site):
#   te_inh_concentration  the national figure: the states' scores weighted by
#                       where property is held. #822's name, so its loc,
#                       tooltips and two modifiers keep reading it. A
#                       country with no scored state keeps its own, which
#                       te_inh_yearly_update, te_inh_shift_concentration and
#                       te_demog_wc_shock move
#   te_dg_wc_target     the states' targets, weighted the same way
#   te_dg_wc_t_law, te_dg_wc_t_land, te_dg_wc_t_own, te_dg_wc_t_ineq,
#   te_dg_wc_t_tax, te_dg_wc_t_econ, te_dg_wc_t_rg  the target's seven terms,
#                       signed around 50 and weighted the same way, so they add up
#                       to the target (the panel's bars)
#   te_dg_wc_top, te_dg_wc_bottom  the most and least concentrated states
#   te_dg_gini          the Gini over all the states' wealth bands
#   te_dg_war_shock     the war shocks taken since the country was last at peace
#                       at its yearly pulse
#   te_dg_wc_boom       the financial boom's shock taken in this cycle
#   te_dg_wc_land       the land-tenure law's role last month (te_demog_wc_land_code)
#   te_dg_wc_gdp, te_dg_wc_g, te_dg_wc_r, te_dg_wc_rg  r > g's inputs and term
#                       (te_demog_wc_growth_update)
#
# Shocks jump the scores instead of moving the targets (Scheidel's levellers):
# a year at war with heavy dead -1, at most -10 until a yearly pulse finds peace
# (te_demog_wc_war_shock); a lost war that cost it a war goal -5 (on_lost_war,
# te_demog_on_actions.txt);
# a won left revolution -20 (te_demog_wc_civil_war_won); a banking crash -5
# (banking_cycle_effects.txt); devastation, in the state hit, up to -10 a year
# (te_demog_wc_state_yearly); land reform, -10 x each state's agrarian share
# (te_demog_wc_land_reform_check). Two raise them: the banking cycle's Frenzy
# +5, once a cycle (te_demog_wc_boom_check), and a state's ownership shock, a
# privatisation +5 to +10 or a nationalisation -5 to -10
# (te_demog_wc_ownership_shock). Nothing here depends on the demographics rule.
# ============================================================================

# THIS = a state at game start, after its walk (te_inh_game_start): the score
# starts at its own target, so nothing drifts in 1836.
te_demog_wc_seed_state = {
	set_variable = { name = te_dg_wc value = te_demog_wc_target }
	set_variable = { name = te_dg_wc_target value = var:te_dg_wc }
}

# THIS = a state, on its yearly pulse after the walk (te_demog_state_yearly). A
# state without a score (an old save, a state new to the game) starts at its
# owner's current figure; otherwise it closes 3% of the gap to its target.
# Then the ownership shock (te_demog_wc_ownership_shock). Devastation destroys
# local capital: the read is a 0-1 share, so a fully devastated state (1)
# loses 10 a year.
te_demog_wc_state_yearly = {
	set_variable = { name = te_dg_wc_target value = te_demog_wc_target }
	if = {
		limit = { NOT = { has_variable = te_dg_wc } }
		if = {
			limit = { owner = { has_variable = te_inh_concentration } }
			set_variable = { name = te_dg_wc value = owner.var:te_inh_concentration }
		}
		else = {
			set_variable = { name = te_dg_wc value = var:te_dg_wc_target }
		}
	}
	else = {
		change_variable = {
			name = te_dg_wc
			add = { value = var:te_dg_wc_target subtract = var:te_dg_wc multiply = 0.03 }
		}
	}
	# who owns the state's capital, against last year (the ownership shock, below)
	te_demog_wc_ownership_shock = yes
	if = {
		limit = { devastation > 0 }
		change_variable = { name = te_dg_wc add = { value = devastation multiply = -10 } }
	}
	clamp_variable = { name = te_dg_wc min = 0 max = 100 }
}

# THIS = a country: every scored state's score jumps by $AMOUNT$, clamped to
# 0-100. The national figure is the caller's to refresh.
te_demog_wc_jump = {
	every_scope_state = {
		limit = { has_variable = te_dg_wc }
		change_variable = { name = te_dg_wc add = $AMOUNT$ }
		clamp_variable = { name = te_dg_wc min = 0 max = 100 }
	}
}

# THIS = ROOT = a country (te_inh_apply_concentration_modifiers needs ROOT to
# be it): a shock to every state's score, then the national figure and #822's
# modifiers follow at once. A country with no scored state (an old save before
# its states' first pulse, a landless country) takes the shock on its own
# figure, as te_inh_shift_concentration does.
te_demog_wc_shock = {
	if = {
		limit = { any_scope_state = { has_variable = te_dg_wc } }
		te_demog_wc_jump = { AMOUNT = $AMOUNT$ }
	}
	else = {
		if = {
			limit = { NOT = { has_variable = te_inh_concentration } }
			set_variable = { name = te_inh_concentration value = te_inh_concentration_target }
		}
		change_variable = { name = te_inh_concentration add = $AMOUNT$ }
		clamp_variable = { name = te_inh_concentration min = 0 max = 100 }
	}
	te_demog_wc_national = yes
}

# THIS = a country at its yearly pulse (te_demog_country_yearly, before the year's
# war dead are shared out or cleared), whatever the rule. A year at war whose
# dead (te_dg_war_year, the monthly snapshot's) pass 0.2% of the people levels
# every state by 1: mass mobilisation. At most 10 such years count until a
# yearly pulse finds the country at peace. That pulse's national figure follows.
te_demog_wc_war_shock = {
	if = {
		limit = { is_at_war = yes }
		if = {
			limit = { NOT = { has_variable = te_dg_war_shock } }
			set_variable = { name = te_dg_war_shock value = 0 }
		}
		set_local_variable = { name = te_dg_wc_war_floor value = { value = total_population multiply = 0.002 } }
		if = {
			limit = {
				has_variable = te_dg_war_year
				var:te_dg_war_year > local_var:te_dg_wc_war_floor
				var:te_dg_war_shock < 10
			}
			te_demog_wc_jump = { AMOUNT = -1 }
			change_variable = { name = te_dg_war_shock add = 1 }
		}
	}
	else_if = {
		limit = { has_variable = te_dg_war_shock }
		remove_variable = te_dg_war_shock
	}
}

# THIS = ROOT = the winner of a civil war, from inh_repair_after_civil_war (in
# te_civil_war_on_won, between te_civil_war_resolve_sides and
# te_civil_war_clear). A left revolution that wins levels the great fortunes:
# -20 to every state. It reads te_cw_rebels_won, not te_cw_origin:
# te_civil_war_clear removes te_cw_origin, the order of two files'
# on_civil_war_won entries is undefined, and a loyalist winner can carry the
# rebels' te_cw_origin after the merge. The caller's national figure follows.
te_demog_wc_civil_war_won = {
	if = {
		limit = {
			has_variable = te_cw_rebels_won
			var:te_cw_rebels_won = 1
			OR = {
				has_law = law_type:law_council_republic
				any_interest_group = {
					is_in_government = yes
					is_interest_group_type = ig_trade_unions
				}
			}
		}
		te_demog_wc_jump = { AMOUNT = -20 }
	}
}

# THIS = ROOT = a country (te_inh_apply_concentration_modifiers needs ROOT to
# be it). The one refresh site of the national figure and #822's modifiers:
# the yearly pulse (te_demog_country_yearly, before the census reads it), game
# start, the inheritance events' shifts, the shocks and the civil-war repair.
#
# The states weigh by where property is held (spec §4.2): each level of an
# ownership building (te_dg_lv_own) and of self-owned capital (te_dg_lv_self)
# counts 1, and the country's state-owned levels count a tenth, shared out by
# each state's share of the country's bureaucrats. Under one level of property
# in all, the states weigh by people; with no people either, one each. A
# country with no scored state keeps its own figure (#822's drift in
# te_inh_yearly_update) and only has its modifiers put back. Each scored
# state's target is rewritten here too, from the law and taxes as they stand,
# so a state that pulsed before a law change does not hold the national
# target back until its next pulse.
te_demog_wc_national = {
	# the scored states: how many, their state-owned levels, bureaucrats,
	# owned and cooperative levels, and people
	set_local_variable = { name = te_dg_wc_n value = 0 }
	set_local_variable = { name = te_dg_wc_ctry value = 0 }
	set_local_variable = { name = te_dg_wc_bur value = 0 }
	set_local_variable = { name = te_dg_wc_held value = 0 }
	set_local_variable = { name = te_dg_wc_ppl value = 0 }
	every_scope_state = {
		limit = { has_variable = te_dg_wc }
		change_local_variable = { name = te_dg_wc_n add = 1 }
		change_local_variable = { name = te_dg_wc_ctry add = var:te_dg_lv_ctry }
		change_local_variable = { name = te_dg_wc_bur add = var:te_dg_bureaucrats }
		change_local_variable = { name = te_dg_wc_held add = { value = var:te_dg_lv_own add = var:te_dg_lv_self } }
		change_local_variable = { name = te_dg_wc_ppl add = var:te_dg_walk_pop }
	}
	if = {
		limit = { local_var:te_dg_wc_n > 0 }
		# the weighting: 0 by property, 1 by people, 2 one each
		set_local_variable = { name = te_dg_wc_total value = local_var:te_dg_wc_held }
		if = {
			limit = { local_var:te_dg_wc_bur >= 1 }
			change_local_variable = { name = te_dg_wc_total add = { value = local_var:te_dg_wc_ctry multiply = 0.1 } }
		}
		set_local_variable = { name = te_dg_wc_mode value = 0 }
		if = {
			limit = { local_var:te_dg_wc_total < 1 }
			set_local_variable = { name = te_dg_wc_mode value = 1 }
			if = {
				limit = { local_var:te_dg_wc_ppl < 1 }
				set_local_variable = { name = te_dg_wc_mode value = 2 }
			}
		}
		# the weighted sums, and the wealth bands summed for the country's Gini
		set_local_variable = { name = te_dg_wc_wsum value = 0 }
		set_local_variable = { name = te_dg_wc_s_score value = 0 }
		set_local_variable = { name = te_dg_wc_s_target value = 0 }
		set_local_variable = { name = te_dg_wc_s_land value = 0 }
		set_local_variable = { name = te_dg_wc_s_own value = 0 }
		set_local_variable = { name = te_dg_wc_s_ineq value = 0 }
		te_demog_gini_band_zero = yes
		set_local_variable = { name = te_dg_g_states value = 0 }
		every_scope_state = {
			limit = { has_variable = te_dg_wc }
			set_local_variable = { name = te_dg_wc_w value = 1 }
			if = {
				limit = { local_var:te_dg_wc_mode = 0 }
				set_local_variable = {
					name = te_dg_wc_w
					value = {
						value = var:te_dg_bureaucrats
						multiply = local_var:te_dg_wc_ctry
						multiply = 0.1
						divide = { value = local_var:te_dg_wc_bur min = 1 }
						add = var:te_dg_lv_own
						add = var:te_dg_lv_self
					}
				}
			}
			else_if = {
				limit = { local_var:te_dg_wc_mode = 1 }
				set_local_variable = { name = te_dg_wc_w value = var:te_dg_walk_pop }
			}
			# the state's target as the law and taxes stand now, not as they stood at its pulse
			set_variable = { name = te_dg_wc_target value = te_demog_wc_target }
			change_local_variable = { name = te_dg_wc_wsum add = local_var:te_dg_wc_w }
			change_local_variable = { name = te_dg_wc_s_score add = { value = var:te_dg_wc multiply = local_var:te_dg_wc_w } }
			change_local_variable = { name = te_dg_wc_s_target add = { value = var:te_dg_wc_target multiply = local_var:te_dg_wc_w } }
			change_local_variable = { name = te_dg_wc_s_land add = { value = te_demog_wc_land_term multiply = local_var:te_dg_wc_w } }
			change_local_variable = { name = te_dg_wc_s_own add = { value = te_demog_wc_ownership_term multiply = local_var:te_dg_wc_w } }
			change_local_variable = { name = te_dg_wc_s_ineq add = { value = te_demog_wc_inequality_term multiply = local_var:te_dg_wc_w } }
			# the state's wealth bands, if they follow the current layout; it counts itself in
			# te_dg_g_states. A state from an older save (strata sums, or the 14 bands to 61+)
			# adds nothing until its own pulse stores new ones (generated)
			te_demog_gini_band_add_state = yes
		}
		# every weighting sums to at least 1
		set_variable = { name = te_inh_concentration value = { value = local_var:te_dg_wc_s_score divide = { value = local_var:te_dg_wc_wsum min = 1 } min = 0 max = 100 } }
		set_variable = { name = te_dg_wc_target value = { value = local_var:te_dg_wc_s_target divide = { value = local_var:te_dg_wc_wsum min = 1 } min = 0 max = 100 } }
		set_variable = { name = te_dg_wc_t_law value = te_demog_wc_law_term }
		set_variable = { name = te_dg_wc_t_land value = { value = local_var:te_dg_wc_s_land divide = { value = local_var:te_dg_wc_wsum min = 1 } } }
		set_variable = { name = te_dg_wc_t_own value = { value = local_var:te_dg_wc_s_own divide = { value = local_var:te_dg_wc_wsum min = 1 } } }
		set_variable = { name = te_dg_wc_t_ineq value = { value = local_var:te_dg_wc_s_ineq divide = { value = local_var:te_dg_wc_wsum min = 1 } } }
		set_variable = { name = te_dg_wc_t_tax value = te_demog_wc_tax_term }
		set_variable = { name = te_dg_wc_t_econ value = te_demog_wc_econ_term }
		set_variable = { name = te_dg_wc_t_rg value = te_demog_wc_rg_term }
		# only when a state has current bands: after loading an old save, before any state's pulse,
		# the country keeps its last figure rather than showing 0
		if = {
			limit = { local_var:te_dg_g_states > 0 }
			te_demog_gini_from_locals = yes
		}
		# the most and least concentrated states, for the panel
		ordered_scope_state = {
			limit = { has_variable = te_dg_wc }
			order_by = te_demog_state_wc
			max = 1
			check_range_bounds = no
			save_temporary_scope_as = te_dg_wc_it
			root = { set_variable = { name = te_dg_wc_top value = scope:te_dg_wc_it } }
		}
		ordered_scope_state = {
			limit = { has_variable = te_dg_wc }
			order_by = te_demog_neg_wc
			max = 1
			check_range_bounds = no
			save_temporary_scope_as = te_dg_wc_it
			root = { set_variable = { name = te_dg_wc_bottom value = scope:te_dg_wc_it } }
		}
	}
	if = {
		limit = { has_variable = te_inh_concentration }
		te_inh_apply_concentration_modifiers = yes
	}
}

# ============================================================================
# The remaining shocks and the r > g term (owner, 2026-10-19; spec §4.2). Like the
# others they run whatever the Demographics rule: Wealth Concentration is the
# Inheritance system's (#822).
# ============================================================================

# THIS = a state, on its yearly pulse inside te_demog_wc_state_yearly, after its walk.
# Ownership shocks, whatever the cause (a law, the building panel's nationalise or
# privatise, decolonisation's nationalisations, a war goal): when the state-owned share
# of the state's capital levels (te_demog_wc_state_owned_share, the ownership term's
# walk and capital filter) moves 10 points or more since the last pulse, the score jumps
# half a point per point moved, at most 10: a privatisation raises it, a nationalisation
# lowers it. It measures the share (the owner's rule), so state or private capital
# growing faster than the other moves it too. te_dg_wc_sown keeps last year's share. No shock in the first year a share is
# stored; after a change of owner, which removes it (te_demog_wc_owner_change, on
# on_state_owner_change); or while the state holds under 20 capital levels, which also
# removes it: in 1849 most states held under 10, where one state-owned level is 10 points
# or more (kept on the owner's call, 2026-10-20). A move of the state's people skips
# nothing (owner, 2026-10-20): capital levels don't move with people, so a migration jump
# makes no share change spurious. Until then a move of more than a quarter in the year
# skipped it, the census's old re-seed rule.
te_demog_wc_ownership_shock = {
	if = {
		limit = { has_variable = te_dg_lv_ctry }
		if = {
			limit = { te_demog_wc_capital_levels >= 20 }
			set_local_variable = { name = te_dg_wc_so value = te_demog_wc_state_owned_share }
			if = {
				limit = { has_variable = te_dg_wc_sown }
				set_local_variable = { name = te_dg_wc_so_move value = { value = local_var:te_dg_wc_so subtract = var:te_dg_wc_sown } }
				if = {
					limit = {
						OR = {
							local_var:te_dg_wc_so_move >= 0.1
							local_var:te_dg_wc_so_move <= -0.1
						}
					}
					change_variable = {
						name = te_dg_wc
						add = { value = local_var:te_dg_wc_so_move multiply = -50 min = -10 max = 10 }
					}
					clamp_variable = { name = te_dg_wc min = 0 max = 100 }
				}
			}
			set_variable = { name = te_dg_wc_sown value = local_var:te_dg_wc_so }
		}
		else_if = {
			limit = { has_variable = te_dg_wc_sown }
			remove_variable = te_dg_wc_sown
		}
	}
}

# THIS = ROOT = a state that changed hands (on_state_owner_change: conquest, cession,
# release, annexation, a revolution's sides): its share is the old owner's, so the next
# pulse stores the new owner's without a shock.
te_demog_wc_owner_change = {
	if = {
		limit = { has_variable = te_dg_wc_sown }
		remove_variable = te_dg_wc_sown
	}
}

# THIS = ROOT = a country, monthly (te_demog_country_monthly_on_action), as
# te_demog_wc_shock needs. The financial boom: +5 to every state once its banking cycle
# reaches Frenzy, once a cycle. te_dg_wc_boom marks it taken and goes when the cycle falls
# below Boom. Every crash lowers the cycle to 40 or less (banking_crash_lower_cycle_to)
# once its event's option lands, so a crash re-arms it then; it isn't re-armed at the
# crash itself, when the cycle can still read Frenzy until the option.
te_demog_wc_boom_check = {
	if = {
		limit = { te_banking_cycle_running = yes }
		if = {
			limit = {
				has_variable = te_dg_wc_boom
				NOT = { banking_cycle_is_boom = yes }
				NOT = { banking_cycle_is_frenzy = yes }
			}
			remove_variable = te_dg_wc_boom
		}
		else_if = {
			limit = {
				NOT = { has_variable = te_dg_wc_boom }
				banking_cycle_is_frenzy = yes
			}
			set_variable = { name = te_dg_wc_boom value = 1 }
			te_demog_wc_shock = { AMOUNT = 5 }
		}
	}
}

# THIS = ROOT = a country, monthly (te_demog_country_monthly_on_action). Land reform:
# enacting Peasant Proprietorship or Collectivized Agriculture from Serfdom or Tenant
# Farmers (each with its variants: te_demog_wc_land_code) jumps every scored state's score
# by -10 x its agrarian share, and the national figure follows. The law is compared with
# last month's (te_dg_wc_land), so every way a law changes counts (enactment, an event's
# activate_law, a won revolution: te_demog_wc_civil_war_inherit gives the winner the old
# government's role, so the next month compares it with the rebels' law) and no vanilla
# law needs an on_activate. The first month only stores it. A move by way of Commercialized Agriculture is no land reform, and
# a country with no scored state takes none: the jump is by each state's farmers.
te_demog_wc_land_reform_check = {
	set_local_variable = { name = te_dg_wc_land_now value = te_demog_wc_land_code }
	if = {
		limit = {
			has_variable = te_dg_wc_land
			var:te_dg_wc_land >= 1
			var:te_dg_wc_land <= 2
			local_var:te_dg_wc_land_now >= 4
		}
		every_scope_state = {
			limit = {
				has_variable = te_dg_wc
				has_variable = te_dg_agr_share
			}
			change_variable = { name = te_dg_wc add = { value = var:te_dg_agr_share multiply = -10 } }
			clamp_variable = { name = te_dg_wc min = 0 max = 100 }
		}
		te_demog_wc_national = yes
	}
	set_variable = { name = te_dg_wc_land value = local_var:te_dg_wc_land_now }
}

# THIS = ROOT = the winner of a civil war the rebels won, from inh_repair_after_civil_war
# (scope:te_cw_loser the old government). The winner continues the nation, but its own
# variables take precedence over the loser's, and the rebels kept their own through the
# war. So it takes the loser's land-law role, and a revolution that brought in Peasant
# Proprietorship or Collectivized Agriculture takes the land-reform shock at the next
# monthly check, as the same law enacted in peace does; and r > g's GDP baseline and
# growth average, so growth neither restarts nor reads the rebels' wartime GDP. The
# baseline's state count goes with it (te_demog_wc_growth_update), so the next sample is
# taken only if the winner holds as many states as that GDP was measured on.
te_demog_wc_civil_war_inherit = {
	if = {
		limit = { scope:te_cw_loser = { has_variable = te_dg_wc_land } }
		set_variable = { name = te_dg_wc_land value = scope:te_cw_loser.var:te_dg_wc_land }
	}
	if = {
		limit = { scope:te_cw_loser = { has_variable = te_dg_wc_gdp } }
		set_variable = { name = te_dg_wc_gdp value = scope:te_cw_loser.var:te_dg_wc_gdp }
		if = {
			limit = { scope:te_cw_loser = { has_variable = te_dg_wc_states } }
			set_variable = { name = te_dg_wc_states value = scope:te_cw_loser.var:te_dg_wc_states }
		}
		else_if = {
			limit = { has_variable = te_dg_wc_states }
			remove_variable = te_dg_wc_states
		}
	}
	if = {
		limit = { scope:te_cw_loser = { has_variable = te_dg_wc_g } }
		set_variable = { name = te_dg_wc_g value = scope:te_cw_loser.var:te_dg_wc_g }
	}
}

# THIS = a country, in te_demog_wc_growth_update, with te_dg_wc_gdp above 0: the year's growth in log points,
# 100 x ln(GDP now / GDP last year), into local te_dg_wc_g_year, so a fall and the recovery to the same GDP cancel
# (owner, 2026-10-20). Script has no logarithm, so ln r = 2 atanh z = 2 (z + z^3/3 + z^5/5 + z^7/7 + z^9/9) with
# z = (r - 1) / (r + 1): under the fixed point, within 0.0002 log points of 100 ln r for r from 0.5 to 2, 0.2%
# short at r = 0.2 or 5 and 2% at 0.1 or 10, and rising with r throughout (test_demographics_wc_shocks). Each odd
# power is kept as 100 z^k (te_dg_wc_t, within -100..100) and taken to the next by multiplying by 100 z^2 before
# dividing by 100, so the five decimals keep their precision; 100 z itself divides the sum by 100 first, since GDP
# passes 7 x 10^11 late.
te_demog_wc_log_growth = {
	set_local_variable = {
		name = te_dg_wc_z
		value = {
			value = local_var:te_dg_wc_gdp_now
			subtract = var:te_dg_wc_gdp
			divide = { value = local_var:te_dg_wc_gdp_now add = var:te_dg_wc_gdp divide = 100 min = 0.00001 }
		}
	}
	set_local_variable = { name = te_dg_wc_z2 value = { value = local_var:te_dg_wc_z multiply = local_var:te_dg_wc_z divide = 100 } }
	set_local_variable = { name = te_dg_wc_t value = local_var:te_dg_wc_z }
	set_local_variable = { name = te_dg_wc_g_year value = local_var:te_dg_wc_z }
	set_local_variable = { name = te_dg_wc_t value = { value = local_var:te_dg_wc_t multiply = local_var:te_dg_wc_z2 divide = 100 } }
	change_local_variable = { name = te_dg_wc_g_year add = { value = local_var:te_dg_wc_t divide = 3 } }
	set_local_variable = { name = te_dg_wc_t value = { value = local_var:te_dg_wc_t multiply = local_var:te_dg_wc_z2 divide = 100 } }
	change_local_variable = { name = te_dg_wc_g_year add = { value = local_var:te_dg_wc_t divide = 5 } }
	set_local_variable = { name = te_dg_wc_t value = { value = local_var:te_dg_wc_t multiply = local_var:te_dg_wc_z2 divide = 100 } }
	change_local_variable = { name = te_dg_wc_g_year add = { value = local_var:te_dg_wc_t divide = 7 } }
	set_local_variable = { name = te_dg_wc_t value = { value = local_var:te_dg_wc_t multiply = local_var:te_dg_wc_z2 divide = 100 } }
	change_local_variable = { name = te_dg_wc_g_year add = { value = local_var:te_dg_wc_t divide = 9 } }
	change_local_variable = { name = te_dg_wc_g_year multiply = 2 }
}

# THIS = a country at its yearly pulse, before te_demog_country_yearly (te_demog_country_yearly_on_action), so
# its national refresh reads this year's figure. r > g's inputs and term (te_demog_wc_rg_term):
#   te_dg_wc_gdp  GDP at the last yearly pulse (te_mon_gdp_current). Its own copy: the monetary system's
#                 te_mon_gdp_last_year is rewritten by an on_yearly_pulse_country entry of its own, in an
#                 order against this one that is undefined
#   te_dg_wc_states  the states it held then (te_demog_wc_state_count), beside the GDP they made
#   te_dg_wc_g    growth averaged over about five years, in log points (te_demog_wc_log_growth: 100 x ln of
#                 the year's GDP ratio, which reads as percent in a normal year: 5% is 4.88), each year's taken
#                 in at a weight of 0.2 whatever its size. A year in which the country gained or lost states (or
#                 an old save's first, with no count) is taken too, unless its GDP moved more than a tenth
#                 (the ratio under 0.9 or over 1.1): that jump is the territory's, so it takes no sample and the
#                 GDP is re-based, and next year's growth is measured on the new territory (owner, 2026-10-20;
#                 until then each year's percent growth was clamped to -10..+10)
#   te_dg_wc_r    the real policy rate at the yearly pulse (te_demog_wc_real_rate), in percent
#   te_dg_wc_rg   the term, 2 x (te_dg_wc_r - te_dg_wc_g), at most 5 either way; only while the banking
#                 cycle runs (te_banking_cycle_running), removed otherwise
# Every country keeps the average, so it is warm when a cycle starts.
te_demog_wc_growth_update = {
	set_local_variable = { name = te_dg_wc_gdp_now value = te_mon_gdp_current }
	set_local_variable = { name = te_dg_wc_states_now value = te_demog_wc_state_count }
	if = {
		limit = {
			has_variable = te_dg_wc_gdp
			var:te_dg_wc_gdp > 0
		}
		# a sample, unless the territory changed and the GDP jumped by more than a tenth either way
		set_local_variable = { name = te_dg_wc_take value = 1 }
		if = {
			limit = {
				NOT = {
					AND = {
						has_variable = te_dg_wc_states
						var:te_dg_wc_states = local_var:te_dg_wc_states_now
					}
				}
			}
			set_local_variable = {
				name = te_dg_wc_ratio
				value = { value = local_var:te_dg_wc_gdp_now divide = { value = var:te_dg_wc_gdp min = 1 } }
			}
			if = {
				limit = {
					OR = {
						local_var:te_dg_wc_ratio < 0.9
						local_var:te_dg_wc_ratio > 1.1
					}
				}
				set_local_variable = { name = te_dg_wc_take value = 0 }
			}
		}
		if = {
			limit = { local_var:te_dg_wc_take = 1 }
			te_demog_wc_log_growth = yes
			if = {
				limit = { has_variable = te_dg_wc_g }
				change_variable = {
					name = te_dg_wc_g
					add = { value = local_var:te_dg_wc_g_year subtract = var:te_dg_wc_g multiply = 0.2 }
				}
			}
			else = {
				set_variable = { name = te_dg_wc_g value = local_var:te_dg_wc_g_year }
			}
		}
	}
	set_variable = { name = te_dg_wc_gdp value = local_var:te_dg_wc_gdp_now }
	set_variable = { name = te_dg_wc_states value = local_var:te_dg_wc_states_now }
	if = {
		limit = { has_variable = te_policy_rate }
		set_variable = { name = te_dg_wc_r value = te_demog_wc_real_rate }
	}
	if = {
		limit = {
			te_banking_cycle_running = yes
			has_variable = te_dg_wc_r
			has_variable = te_dg_wc_g
		}
		set_variable = { name = te_dg_wc_rg value = te_demog_wc_rg_now }
	}
	else_if = {
		limit = { has_variable = te_dg_wc_rg }
		remove_variable = te_dg_wc_rg
	}
}

# ============================================================================
# The Inequality effect (spec §4.1, §7; owner, 2026-10-19), under Full only: relative
# deprivation. The state's shown Gini above 0.45 raises its radicals, by up to 10% at
# 0.65 and above: multiplier (Gini - 0.45) / 0.20, 0-1, on te_demog_census_inequality
# (te_demog_inequality_modifiers.txt). The engine has no lower-strata radicals type, so it
# takes the state-wide one inh_concentrated_property and the youth bulge carry.
# ============================================================================

# THIS = ROOT = a state, on its yearly pulse (te_demog_state_yearly, after Wealth
# Concentration's refresh), so the multiplier reads the state's own variable: the one
# refresh site, removed and re-applied by the dynamic-modifier pattern. Outside Full the
# modifier and te_dg_ineq (its multiplier) go.
te_demog_inequality_refresh = {
	if = {
		limit = { has_modifier = te_demog_census_inequality }
		remove_modifier = te_demog_census_inequality
	}
	if = {
		limit = {
			te_demog_effects_run = yes
			has_variable = te_dg_gini
		}
		set_variable = { name = te_dg_ineq value = { value = var:te_dg_gini subtract = 0.45 multiply = 5 min = 0 max = 1 } }
		if = {
			limit = { var:te_dg_ineq > 0.01 }
			add_modifier = { name = te_demog_census_inequality multiplier = var:te_dg_ineq }
		}
	}
	else_if = {
		limit = { has_variable = te_dg_ineq }
		remove_variable = te_dg_ineq # REVIEWED 2026-10-19: outside Full only, after the modifier came off at the top of this effect; nothing applies it there
	}
}
