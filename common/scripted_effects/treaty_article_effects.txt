# Treaty Article Scripted Effects

# Intelligence Sharing Defense Shield
# Gives the country with lower covert defense half the difference between its
# base defense and its strongest intel-sharing partner's base defense, spread
# evenly over the three defense types (a sixth of the gap on each).
#
# IMPORTANT: Modifier values are NOT recalculated within a single effect block.
# remove_modifier / add_modifier changes only take effect after the block completes.
# Reading modifier:X within the same effect still sees the old values.
# We use a persistent variable (intel_shield_mult) to track the prior shield
# multiplier and subtract its contribution from modifier: reads to get base values.
# This follows the same pattern as prior_pollution_reduction and prior_free_port_import_cap.
#
# Country scope. Called from on_yearly_pulse_country.
update_intel_sharing_defense = {
	# Only proceed if country has an intel sharing pact
	if = {
		limit = {
			any_scope_treaty = {
				any_scope_article = { has_type = intelligence_sharing_pact }
			}
		}

		# Read own total covert defense (includes current shield modifier if any)
		set_variable = { name = _own_def value = modifier:country_covert_defense_economic_add }
		change_variable = { name = _own_def add = modifier:country_covert_defense_military_add }
		change_variable = { name = _own_def add = modifier:country_covert_defense_ideological_add }

		# Subtract own shield contribution to get base defense
		# Shield modifier gives (1 * multiplier) to each of 3 defense types = 3 * multiplier total
		if = {
			limit = { has_variable = intel_shield_mult }
			set_variable = { name = _shield_sub value = var:intel_shield_mult }
			change_variable = { name = _shield_sub multiply = 3 }
			change_variable = { name = _own_def subtract = var:_shield_sub }
			remove_variable = _shield_sub
		}

		set_variable = { name = _best_boost value = 0 }
		save_scope_as = _self

		every_scope_treaty = {
			limit = { any_scope_article = { has_type = intelligence_sharing_pact } }

			# Determine which country is the partner
			if = {
				limit = { first_country = scope:_self }
				# Partner is second_country
				second_country = {
					# Enter ROOT to set variables, PREV = partner country
					ROOT = {
						set_variable = { name = _p_def value = PREV.modifier:country_covert_defense_economic_add }
						change_variable = { name = _p_def add = PREV.modifier:country_covert_defense_military_add }
						change_variable = { name = _p_def add = PREV.modifier:country_covert_defense_ideological_add }

						# Subtract partner's shield contribution to get their base defense
						if = {
							limit = { PREV = { has_variable = intel_shield_mult } }
							set_variable = { name = _p_shield value = PREV.var:intel_shield_mult }
							change_variable = { name = _p_shield multiply = 3 }
							change_variable = { name = _p_def subtract = var:_p_shield }
							remove_variable = _p_shield
						}

						# boost = (partner_base - own_base) * 0.5 / 3, min 0: the modifier adds the
						# boost to each of the three defense types, so dividing by 3 makes the total
						# gain half the gap (owner decision, PR #505). The floor is
						# _best_boost's starting 0: a candidate must beat it (strict >)
						# to count, so a partner weaker than us never lowers the boost.
						set_variable = { name = _cand value = var:_p_def }
						change_variable = { name = _cand subtract = var:_own_def }
						change_variable = { name = _cand multiply = 0.5 }
						change_variable = { name = _cand divide = 3 }

						if = {
							limit = { var:_cand > var:_best_boost }
							set_variable = { name = _best_boost value = var:_cand }
						}
						remove_variable = _cand
						remove_variable = _p_def
					}
				}
			}
			else = {
				# Partner is first_country
				first_country = {
					ROOT = {
						set_variable = { name = _p_def value = PREV.modifier:country_covert_defense_economic_add }
						change_variable = { name = _p_def add = PREV.modifier:country_covert_defense_military_add }
						change_variable = { name = _p_def add = PREV.modifier:country_covert_defense_ideological_add }

						if = {
							limit = { PREV = { has_variable = intel_shield_mult } }
							set_variable = { name = _p_shield value = PREV.var:intel_shield_mult }
							change_variable = { name = _p_shield multiply = 3 }
							change_variable = { name = _p_def subtract = var:_p_shield }
							remove_variable = _p_shield
						}

						set_variable = { name = _cand value = var:_p_def }
						change_variable = { name = _cand subtract = var:_own_def }
						change_variable = { name = _cand multiply = 0.5 }
						change_variable = { name = _cand divide = 3 }

						if = {
							limit = { var:_cand > var:_best_boost }
							set_variable = { name = _best_boost value = var:_cand }
						}
						remove_variable = _cand
						remove_variable = _p_def
					}
				}
			}
		}

		# Remove old modifier, apply new one
		if = {
			limit = { has_modifier = intelligence_sharing_defense_shield_modifier }
			remove_modifier = intelligence_sharing_defense_shield_modifier
		}

		# Persist the multiplier BEFORE add_modifier and point the modifier at the
		# persistent copy: the modifier has no duration, so the engine re-reads
		# `multiplier` on later ticks and the backing variable must outlive this
		# effect (scripting_best_practices.md, "Variables backing add_modifier
		# multiplier must persist"). `_best_boost` is a scratch var removed below.
		# intel_shield_mult is also what partners read next year to strip our
		# shield out of our base defense.
		set_variable = { name = intel_shield_mult value = var:_best_boost }
		if = {
			limit = { var:intel_shield_mult > 0 }
			add_modifier = {
				name = intelligence_sharing_defense_shield_modifier
				multiplier = var:intel_shield_mult
			}
		}

		remove_variable = _own_def
		remove_variable = _best_boost
	}
	else = {
		# No intel sharing pact — clean up
		if = {
			limit = { has_modifier = intelligence_sharing_defense_shield_modifier }
			remove_modifier = intelligence_sharing_defense_shield_modifier
		}
		if = {
			limit = { has_variable = intel_shield_mult }
			remove_variable = intel_shield_mult
		}
	}
}

# ============================================================================
# STATE-INPUT ARTICLES: the state's effects follow the treaty
# ============================================================================
# Free Port Concession, Minority Protection, Religious Mission Rights and
# Demilitarized Zone each name one of the conceding country's states and put
# their effects on that state: a static modifier and a variable, which
# te_update_free_port_tariff and religious_mission_conversion_on_action read
# on the yearly state pulse. One apply and one clear effect per article (state
# scope) serve the article's own hooks and the two checks below.
#
# A treaty can end without on_break or on_withdrawal undoing them. A player
# who annexed Belgium kept a free port in Flanders whose treaty was gone, and
# te_update_free_port_tariff renewed it every year off the stale has_free_port.
# A state can also change hands while the treaty stands. So the treaty is the
# record (te_state_article_concession_in_force, misc_triggers.txt), and the
# state is checked against it:
#   te_sync_state_treaty_articles     clears an article's effects from a state
#                                     no longer conceded under it. Monthly
#                                     (te_state_treaty_article_sync_on_action)
#                                     and on every change of owner.
#   te_restore_state_treaty_articles  puts them back on a state that returns to
#                                     the country that conceded it while the
#                                     treaty still stands. On a change of owner.
# ============================================================================

te_free_port_state_apply = {
	add_modifier = {
		name = free_port_state_modifier
	}
	set_variable = has_free_port
	# The tariff modifiers wait for te_update_free_port_tariff on the yearly
	# pulse: the article's entry hook can't run its has_variable reads.
}

te_free_port_state_clear = {
	remove_modifier = free_port_state_modifier
	remove_modifier = free_port_tariff_import_modifier
	remove_modifier = free_port_tariff_export_modifier
	remove_variable = has_free_port
}

te_minority_protection_state_apply = {
	add_modifier = {
		name = minority_protection_state_modifier
	}
	set_variable = has_minority_protection
}

te_minority_protection_state_clear = {
	remove_modifier = minority_protection_state_modifier
	remove_variable = has_minority_protection
}

# $MISSION_COUNTRY$ = the article's target, whose religion the missions spread.
te_religious_mission_state_apply = {
	add_modifier = {
		name = religious_mission_state_modifier
	}
	set_variable = {
		name = religious_mission_country
		value = $MISSION_COUNTRY$
	}
}

te_religious_mission_state_clear = {
	remove_modifier = religious_mission_state_modifier
	remove_variable = religious_mission_country
}

te_demilitarized_zone_state_apply = {
	add_modifier = {
		name = demilitarized_zone_state_modifier
	}
	# The modifier caps a base at no levels; take down one already
	# standing rather than rely on the engine to shrink it.
	if = { limit = { has_building = building_military_base } remove_building = building_military_base }
	set_variable = has_demilitarized_zone
}

te_demilitarized_zone_state_clear = {
	remove_modifier = demilitarized_zone_state_modifier
	remove_variable = has_demilitarized_zone
}

# State scope. Gated on the variable or the modifier, so a state with neither
# never walks a treaty.
te_sync_state_treaty_articles = {
	if = {
		limit = {
			OR = {
				has_variable = has_free_port
				has_modifier = free_port_state_modifier
			}
			NOT = { te_state_article_concession_in_force = { ARTICLE_TYPE = free_port_concession } }
		}
		te_free_port_state_clear = yes
	}
	if = {
		limit = {
			OR = {
				has_variable = has_minority_protection
				has_modifier = minority_protection_state_modifier
			}
			NOT = { te_state_article_concession_in_force = { ARTICLE_TYPE = minority_protection } }
		}
		te_minority_protection_state_clear = yes
	}
	if = {
		limit = {
			OR = {
				has_variable = religious_mission_country
				has_modifier = religious_mission_state_modifier
			}
			NOT = { te_state_article_concession_in_force = { ARTICLE_TYPE = religious_mission_rights } }
		}
		te_religious_mission_state_clear = yes
	}
	if = {
		limit = {
			OR = {
				has_variable = has_demilitarized_zone
				has_modifier = demilitarized_zone_state_modifier
			}
			NOT = { te_state_article_concession_in_force = { ARTICLE_TYPE = demilitarized_zone } }
		}
		te_demilitarized_zone_state_clear = yes
	}
}

# State scope, ROOT = the state (on_state_owner_change). Walks the new owner's
# treaties for an article it conceded on this state, which is only there when
# the state has come back to the country that conceded it.
te_restore_state_treaty_articles = {
	save_scope_as = te_rsta_state
	owner = {
		save_scope_as = te_rsta_owner
		every_scope_treaty = {
			every_scope_article = {
				limit = {
					OR = {
						has_type = free_port_concession
						has_type = minority_protection
						has_type = religious_mission_rights
						has_type = demilitarized_zone
					}
					source_country = scope:te_rsta_owner
					input_state = scope:te_rsta_state
				}
				if = {
					limit = { has_type = free_port_concession }
					scope:te_rsta_state = {
						if = {
							limit = { NOT = { has_variable = has_free_port } }
							te_free_port_state_apply = yes
						}
					}
				}
				else_if = {
					limit = { has_type = minority_protection }
					scope:te_rsta_state = {
						if = {
							limit = { NOT = { has_variable = has_minority_protection } }
							te_minority_protection_state_apply = yes
						}
					}
				}
				else_if = {
					limit = { has_type = religious_mission_rights }
					target_country = {
						save_scope_as = te_rsta_mission_country
					}
					scope:te_rsta_state = {
						if = {
							limit = { NOT = { has_variable = religious_mission_country } }
							te_religious_mission_state_apply = { MISSION_COUNTRY = scope:te_rsta_mission_country }
						}
					}
				}
				else_if = {
					limit = { has_type = demilitarized_zone }
					scope:te_rsta_state = {
						if = {
							limit = { NOT = { has_variable = has_demilitarized_zone } }
							te_demilitarized_zone_state_apply = yes
						}
					}
				}
			}
		}
	}
}
