# Monthly country-scope settlement, after anchor discovery and before premium/FX.
# Bank purchases run after gold's initialization/flows, so its seed cannot
# overwrite a treasury-funded deposit. Modifiers update after this effect block;
# eligibility is checked directly wherever their mechanical benefits are read.
banking_external_monthly_update = {
	set_variable = { name = banking_external_sterilization_funded value = 0 }
	if = {
		limit = { NOT = { has_variable = banking_external_fx_protection } }
		set_variable = { name = banking_external_fx_protection value = 0 }
	}
	if = {
		limit = { banking_external_eligible_foreign_borrowing_limits = yes banking_tool_foreign_borrowing_limits_active = yes }
		change_variable = { name = banking_external_fx_protection add = banking_external_fx_protection_step }
	}
	else = {
		change_variable = { name = banking_external_fx_protection subtract = banking_external_fx_protection_step }
	}
	clamp_variable = { name = banking_external_fx_protection min = 0 max = banking_external_fx_protection_cap }
	if = {
		limit = { NOT = { has_variable = banking_external_import_months } }
		set_variable = { name = banking_external_import_months value = 0 }
	}
	if = {
		limit = { banking_tool_sterilize_inflows_active = yes }
		if = {
			limit = { banking_external_eligible_sterilize_inflows = yes gold_reserves >= banking_external_sterilization_cost }
			set_variable = { name = banking_external_sterilization_funded value = 1 }
			banking_external_charge_sterilization = yes
		}
		else = { banking_effect_cb_disable_sterilize_inflows = yes }
	}
	if = {
		limit = { banking_tool_emergency_import_financing_active = yes }
		if = {
			limit = {
				banking_external_eligible_emergency_import_financing = yes
				te_mon_in_financial_crisis = yes
				gold_reserves >= banking_external_combined_import_cost
			}
			# Charged at this month's rate, then the month counts towards the rise.
			banking_external_charge_import = yes
			change_variable = { name = banking_external_import_months add = 1 }
		}
		else = { banking_effect_cb_disable_emergency_import_financing = yes }
	}
	else = { change_variable = { name = banking_external_import_months subtract = 1 } }
	clamp_variable = { name = banking_external_import_months min = 0 max = banking_external_import_fatigue_cap }
	banking_external_cleanup = yes
}

# THE TWO BILLS ARE WEEKLY EXPENSES (owner, 2026-10-27, GUI audit BNK-14):
# Sterilize Capital Inflows and Emergency Import Financing were paid once a
# month with add_treasury, which no weekly figure includes, so neither the
# Overview nor the Breakdown showed them. Each is now a charge modifier on the
# entry (`country_expenses_add = 1` with the Breakdown's mirror field),
# applied at the month's cost ÷ 4 a week: a year of weekly charges is 13
# months' worth, where 12 monthly payments were 12 (and the activation's
# payment a thirteenth). The banking_external_charge_* pair is each charge's one
# refresh site: the monthly update above re-applies it while the tool stays
# funded, the Enable effect applies it at once, and the Disable effect and a
# change of economic system (remove_banking_market_modifiers_effect) take it
# off with the tool. The treasury test (gold_reserves against a month's cost)
# still decides whether a tool keeps running.
#
# A revolution's winner inherits neither the tool nor its charge (no
# modifiers cross), and the next update finds no tool to charge for: the
# refresh is driven by the tool and the inherited variables, never by the
# charge itself.
#
# FX Surrender is NOT moved: its debit is the same amount, the same day, as
# the gold the vault receives (banking_external_purchase_reserves below), and
# the purchase is capped by the cash in hand. A weekly expense would pay a
# different sum at a different time, into debt if the cash ran out.
banking_external_charge_sterilization = {
	je:je_banking_cycle ?= {
		remove_modifier_if_exists_effect = { MODIFIER = banking_external_sterilization_charge }
		add_modifier = {
			name = banking_external_sterilization_charge
			multiplier = banking_external_sterilization_weekly_charge_for_je
		}
	}
}

banking_external_charge_import = {
	je:je_banking_cycle ?= {
		remove_modifier_if_exists_effect = { MODIFIER = banking_external_import_charge }
		add_modifier = {
			name = banking_external_import_charge
			multiplier = banking_external_import_weekly_charge_for_je
		}
	}
}

banking_external_end_charges = {
	je:je_banking_cycle ?= {
		remove_modifier_if_exists_effect = { MODIFIER = banking_external_sterilization_charge }
		remove_modifier_if_exists_effect = { MODIFIER = banking_external_import_charge }
	}
}

banking_external_purchase_reserves = {
	if = {
		limit = {
			banking_external_eligible_fx_surrender = yes
			banking_tool_fx_surrender_active = yes
			has_variable = te_bank_gold
			banking_external_surrender_purchase > 0
		}
		set_variable = { name = banking_external_purchase value = banking_external_surrender_purchase }
		add_treasury = banking_external_surrender_payment
		change_variable = { name = te_bank_gold add = var:banking_external_purchase }
		remove_variable = banking_external_purchase
	}
}

# Unavailable tools never occupy intervention points indefinitely. Disable
# helpers also record the policy change in history; state stocks persist.
banking_external_cleanup = {
	if = {
		limit = { banking_tool_restrict_inflows_active = yes banking_external_eligible_restrict_inflows = no }
		banking_effect_cb_disable_restrict_inflows = yes
	}
	if = {
		limit = { banking_tool_foreign_borrowing_limits_active = yes banking_external_eligible_foreign_borrowing_limits = no }
		banking_effect_cb_disable_foreign_borrowing_limits = yes
	}
	if = {
		limit = { banking_tool_fx_surrender_active = yes banking_external_eligible_fx_surrender = no }
		banking_effect_cb_disable_fx_surrender = yes
	}
}
