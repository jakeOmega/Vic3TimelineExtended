# Plan 2026-10-03, Tasks 18-22; spec
# docs/superpowers/specs/2026-10-03-tax-code-ai-and-release-design.md §2;
# docs/systems/tax_code_schema.md, "AI legislation".
#
#   STREAKS   te_tax_ai_update_streaks, processor rule 1b, every migrated
#             country: the fiscal streaks the need tests read, after the
#             fiscal record of the 1st; draws the country's phase once.
#   DISPATCH  te_tax_ai_dispatch, processor rule 1b, AI countries only: when
#             a step is due (te_tax_ai_step_due) it raises the hidden te_tax.8
#             on the country's bucket day (5, 12, 19 or 26 days on), or the
#             next day on a retry. The processor itself never drafts.
#   STEP      te_tax_ai_step, from te_tax.8 only, with ROOT = THIS = the
#             country: held packages (te_tax_ai_manage_packages), promises at
#             risk (te_tax_ai_manage_promises), then the open bill
#             (te_tax_ai_manage_bill) or, with none, a new one
#             (te_tax_ai_initiative: one template bill, introduced and judged
#             in the same execution).
#   RESET     te_tax_ai_reset_bill_state, at the outbreak (on the rebels),
#             the win repair and a release (te_tax_civil_war_effects.txt).
#   LOGS      te_tax_ai_log_*, one summary line per AI action.
#
# The AI acts only through the commands the player's buttons run
# (te_tax_cmd_<c> inside te_tax_can_<c>); nothing here writes native fiscal
# state. The one owner-approved exception: a kind-1 promise the native AI is
# about to miss is enacted, by the generated te_tax_gen_ai_enact_<o> (spec
# §2.8). Every te_tax_ai_* variable is a schema token: initialised by
# te_tax_init_country, copied at the outbreak, never removed.
# ============================================================================

# Processor rule 1b
# ----------------------------------------------------------------------------

# The fiscal streaks (spec §2.3): consecutive 1sts with a recorded deficit, and
# with a recorded surplus at or above the cut ratio; any other 1st resets the
# streak. Every migrated country, so a revolution's player-side tokens stay
# meaningful. Called only by te_tax_process_month, after the fiscal record
# (te_tax_record_fiscal_month) and once per processed month. The phase is drawn
# once, by the first run that finds the sentinel; it is never redrawn.
te_tax_ai_update_streaks = {
	if = {
		limit = {
			te_tax_code_on = yes
			has_variable = te_tax_schema
		}
		if = {
			limit = { te_tax_fisc_rec_deficit = yes }
			change_variable = { name = te_tax_ai_def_streak add = 1 }
		}
		else = {
			set_variable = { name = te_tax_ai_def_streak value = 0 }
		}
		if = {
			limit = {
				te_tax_fisc_rec_surplus = yes
				te_tax_ai_ratio >= te_tax_ai_cut_ratio_now
			}
			change_variable = { name = te_tax_ai_sur_streak add = 1 }
		}
		else = {
			set_variable = { name = te_tax_ai_sur_streak value = 0 }
		}
		if = {
			limit = { var:te_tax_ai_phase < 0 }
			te_tax_ai_phase_draw = yes
		}
	}
}

# The country's phase, 0 to 11, drawn once and stored: the UN AI's stored
# random draw (un_vote_ai_draws, un_dossier_effects.txt), a random_list of
# equal weights whose branches each set a literal. It spreads the AI's
# initiatives over the cadence (te_tax_ai_cadence_pos) and its steps over the
# month (te_tax_ai_bucket).
te_tax_ai_phase_draw = {
	random_list = {
		1 = { set_variable = { name = te_tax_ai_phase value = 0 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 1 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 2 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 3 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 4 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 5 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 6 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 7 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 8 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 9 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 10 } }
		1 = { set_variable = { name = te_tax_ai_phase value = 11 } }
	}
}

# Raises the AI step for an AI country with something due this month: an
# initiative on its phase, an emergency, or something to manage
# (te_tax_ai_step_due). A retry (te_tax_ai_retry, set by the step when the
# native tax level alone blocks passage) raises it for tomorrow, the day after
# the writer pins the level; otherwise the bucket picks the day, off the
# month-boundary burst. trigger_event takes a literal delay, hence the chain.
# Called only by te_tax_process_month, after the streaks.
te_tax_ai_dispatch = {
	if = {
		limit = {
			te_tax_code_on = yes
			is_ai = yes
			te_tax_ai_step_due = yes
		}
		if = {
			limit = { var:te_tax_ai_retry = 1 }
			set_variable = { name = te_tax_ai_retry value = 0 }
			trigger_event = { id = te_tax.8 days = 1 }
		}
		else_if = {
			limit = { te_tax_ai_bucket = 0 }
			trigger_event = { id = te_tax.8 days = 5 }
		}
		else_if = {
			limit = { te_tax_ai_bucket = 1 }
			trigger_event = { id = te_tax.8 days = 12 }
		}
		else_if = {
			limit = { te_tax_ai_bucket = 2 }
			trigger_event = { id = te_tax.8 days = 19 }
		}
		else = {
			trigger_event = { id = te_tax.8 days = 26 }
		}
	}
}

# The AI step, run only by te_tax.8 (te_tax_ai_can_act holds), with ROOT = THIS
# = the country (spec §2.4): held packages, then a promise at risk, then the
# bill under debate or, with none, a new one when the initiative is ready.
# First, when the need has ended (te_tax_ai_need_ended), the episode's
# "no viable bill" marker is cleared, so the next episode logs its line again;
# here rather than in the initiative, so a need that ends during a cooldown
# still ends the episode.
te_tax_ai_step = {
	if = {
		limit = {
			te_tax_code_on = yes
			te_tax_ai_can_act = yes
		}
		if = {
			limit = { te_tax_ai_need_ended = yes }
			set_variable = { name = te_tax_ai_noviable value = 0 }
		}
		te_tax_ai_manage_packages = yes
		te_tax_ai_manage_promises = yes
		if = {
			limit = { te_tax_bill_active = yes }
			te_tax_ai_manage_bill = yes
		}
		else_if = {
			limit = { te_tax_ai_initiative_ready = yes }
			te_tax_ai_initiative = yes
		}
	}
}

# Step 1, held packages (spec §2.4): slot a, then b. A missed package (state
# 3) is rescheduled while it may be (te_tax_can_package_reschedule: at most
# te_tax_reschedule_max_delay months late, and untouched by a later bill);
# otherwise it, or a conflicting one (state 2), is released, so held packages
# never fill both slots and block every later bill. Neither command refreshes
# support. The managers test no is_ai: the step's te_tax_ai_can_act does, and
# the console (Task 22) runs them on its own country.
te_tax_ai_manage_packages = {
	if = {
		limit = {
			te_tax_code_on = yes
			te_tax_code_in_force = yes
		}
		if = {
			limit = { te_tax_can_package_reschedule = { SLOT = a } }
			te_tax_cmd_package_reschedule = { SLOT = a }
			te_tax_ai_log_rescheduled = yes
		}
		else_if = {
			limit = { te_tax_can_package_release = { SLOT = a } }
			te_tax_cmd_package_release = { SLOT = a }
			te_tax_ai_log_released = yes
		}
		if = {
			limit = { te_tax_can_package_reschedule = { SLOT = b } }
			te_tax_cmd_package_reschedule = { SLOT = b }
			te_tax_ai_log_rescheduled = yes
		}
		else_if = {
			limit = { te_tax_can_package_release = { SLOT = b } }
			te_tax_cmd_package_release = { SLOT = b }
			te_tax_ai_log_released = yes
		}
	}
}

# Step 2, promises (spec §2.4, §2.8): at most one a step, the first of slots 1
# to 4 that the next monthly check would breach (te_tax_ai_promise_at_risk;
# never a pending or bound one). A kind-1 promise the AI can enact
# (te_tax_ai_can_enact) is set to exactly its target by the generated
# te_tax_gen_ai_enact_<o>, the owner's one exception to the institution's
# native expansion; its native costs apply from then on and the monthly check
# still verifies it. Any other promise, a balance (kinds 2 and 4), a wage cut
# copied from a player at an outbreak (kind 3) or an institution that can no
# longer be enacted, is renegotiated: -2 approval and -1 trust instead of the
# breach's -5. Each is logged first, per slot (ai_obl_enacted,
# ai_renegotiated, with the promise and its measure): the setter's result is
# not assumed visible later in the block, and the renegotiation frees the
# slot, whose views then read nothing.
te_tax_ai_manage_promises = {
	if = {
		limit = {
			te_tax_code_on = yes
			te_tax_code_in_force = yes
		}
		if = {
			limit = { te_tax_ai_promise_at_risk = { N = 1 } }
			if = {
				limit = { te_tax_ai_can_enact = { N = 1 } }
				te_tax_gen_ai_log_enacted_1 = yes
				te_tax_gen_ai_enact_1 = yes
			}
			else_if = {
				limit = { te_tax_can_obl_renegotiate = { N = 1 } }
				te_tax_gen_ai_log_renegotiated_1 = yes
				te_tax_cmd_obl_renegotiate = { N = 1 }
			}
		}
		else_if = {
			limit = { te_tax_ai_promise_at_risk = { N = 2 } }
			if = {
				limit = { te_tax_ai_can_enact = { N = 2 } }
				te_tax_gen_ai_log_enacted_2 = yes
				te_tax_gen_ai_enact_2 = yes
			}
			else_if = {
				limit = { te_tax_can_obl_renegotiate = { N = 2 } }
				te_tax_gen_ai_log_renegotiated_2 = yes
				te_tax_cmd_obl_renegotiate = { N = 2 }
			}
		}
		else_if = {
			limit = { te_tax_ai_promise_at_risk = { N = 3 } }
			if = {
				limit = { te_tax_ai_can_enact = { N = 3 } }
				te_tax_gen_ai_log_enacted_3 = yes
				te_tax_gen_ai_enact_3 = yes
			}
			else_if = {
				limit = { te_tax_can_obl_renegotiate = { N = 3 } }
				te_tax_gen_ai_log_renegotiated_3 = yes
				te_tax_cmd_obl_renegotiate = { N = 3 }
			}
		}
		else_if = {
			limit = { te_tax_ai_promise_at_risk = { N = 4 } }
			if = {
				limit = { te_tax_ai_can_enact = { N = 4 } }
				te_tax_gen_ai_log_enacted_4 = yes
				te_tax_gen_ai_enact_4 = yes
			}
			else_if = {
				limit = { te_tax_can_obl_renegotiate = { N = 4 } }
				te_tax_gen_ai_log_renegotiated_4 = yes
				te_tax_cmd_obl_renegotiate = { N = 4 }
			}
		}
	}
}

# Step 3, the open bill (spec §2.4 step 3, §2.7, §2.10; plan Task 20). The
# AI introduced it (Task 21), or a player did before the country passed to
# the AI (te_tax_ai_tpl 0), which the same rules manage; its patience then
# starts the first time the step sees it. Then, once per step:
#   - a bill whose due month came during debate is moved to next month
#     (te_tax_cmd_reschedule, commitments kept), so a debate an accepted offer
#     restarted can still end in a pass;
#   1. pass it when it may, recording the reverse-window marks and the
#      cooldown first: the pass closes the bill record they read;
#   2. wait when only legitimacy blocks it and the native tax level is off
#      medium: the writer pins it on the 1st, and the next processor raises
#      te_tax.8 on the 2nd (te_tax_ai_retry). Logged each time, so at most one
#      line a month;
#   3. accept the best offer (te_tax_gen_ai_accept_offer), at most
#      te_tax_ai_max_offers a month, each raising the step again the next day;
#   4. in an emergency, force it through;
#   5. withdraw it when it is hopeless or has been open te_tax_ai_bill_patience
#      months (te_tax_ai_withdraw, with a reason).
# Each branch ends in one summary line; a pass, a force-through and a
# withdrawal reset the bill state. Rule 3's limit holds only when some offer is
# acceptable, so an else-chain never stops at a branch that does nothing.
te_tax_ai_manage_bill = {
	if = {
		limit = {
			te_tax_code_on = yes
			te_tax_code_in_force = yes
			te_tax_bill_active = yes
		}
		if = {
			limit = { var:te_tax_ai_bill_month < 0 }
			set_variable = { name = te_tax_ai_bill_month value = te_tax_ai_now }
		}
		if = {
			limit = { te_tax_can_reschedule = yes }
			te_tax_cmd_reschedule = yes
			te_tax_ai_log_bill_rescheduled = yes
		}
		if = {
			limit = { te_tax_can_pass = yes }
			te_tax_gen_ai_record_marks = yes
			set_variable = { name = te_tax_ai_next_month value = te_tax_ai_cooldown_after_pass }
			te_tax_cmd_pass = yes
			te_tax_ai_log_passed = yes
			te_tax_ai_reset_bill_state = yes
		}
		else_if = {
			limit = { te_tax_ai_blocked_by_native_level = yes }
			set_variable = { name = te_tax_ai_retry value = 1 }
			te_tax_ai_log_waiting_legitimacy_native_level = yes
		}
		else_if = {
			limit = { te_tax_ai_offer_available = yes }
			te_tax_gen_ai_accept_offer = yes
		}
		else_if = {
			limit = {
				te_tax_ai_emergency = yes
				te_tax_can_force_through = yes
			}
			if = {
				limit = { te_tax_can_force_through = yes }
				te_tax_gen_ai_record_marks = yes
				set_variable = { name = te_tax_ai_next_month value = te_tax_ai_cooldown_after_pass }
				te_tax_cmd_force_through = yes
				te_tax_ai_log_forced = yes
				te_tax_ai_reset_bill_state = yes
			}
		}
		else_if = {
			limit = {
				OR = {
					te_tax_ai_bill_hopeless = yes
					te_tax_ai_bill_age >= te_tax_ai_bill_patience
				}
			}
			te_tax_ai_withdraw = yes
		}
	}
}

# Accepts group IG's offer through the Accept button's command, called only by
# te_tax_gen_ai_accept_offer in its order. The acceptance opens a new revision
# and refreshes support; the step then runs again the next day, so a bargain
# completes within days. The month's count starts over in a new month.
te_tax_ai_accept_offer = {
	if = {
		limit = { te_tax_can_accept_offer = { IG = $IG$ } }
		te_tax_cmd_accept_offer = { IG = $IG$ }
		if = {
			limit = { NOT = { var:te_tax_ai_offer_month = te_tax_ai_now } }
			set_variable = { name = te_tax_ai_offer_count value = 0 }
		}
		change_variable = { name = te_tax_ai_offer_count add = 1 }
		set_variable = { name = te_tax_ai_offer_month value = te_tax_ai_now }
		te_tax_ai_log_accepted = yes
		trigger_event = { id = te_tax.8 days = 1 }
	}
}

# Withdraws the bill with the reason that applies (spec §2.4 step 3):
# legitimacy below the passage line; no free slot; in an emergency, enough
# committed to force it but no override capacity or Authority; hopeless on
# support; open te_tax_ai_bill_patience months; otherwise support, so a
# caller outside the manager never logs patience for a bill just opened. The
# initiative calls it for a bill that is hopeless on introduction
# (te_tax_ai_introduce_and_judge).
te_tax_ai_withdraw = {
	if = {
		limit = { te_tax_code_on = yes }
		if = {
			limit = { legitimacy < te_tax_passage_legitimacy }
			te_tax_ai_withdraw_legitimacy = yes
		}
		else_if = {
			limit = {
				NOT = {
					OR = {
						var:te_tax_pa_on = 0
						var:te_tax_pb_on = 0
					}
				}
			}
			te_tax_ai_withdraw_slots = yes
		}
		else_if = {
			limit = { te_tax_ai_force_blocked_by_authority = yes }
			te_tax_ai_withdraw_authority = yes
		}
		else_if = {
			limit = { te_tax_ai_bill_hopeless = yes }
			te_tax_ai_withdraw_support = yes
		}
		else_if = {
			limit = { te_tax_ai_bill_age >= te_tax_ai_bill_patience }
			te_tax_ai_withdraw_patience = yes
		}
		else = {
			te_tax_ai_withdraw_support = yes
		}
	}
}

# One effect per reason, since debug_log cannot print a parameter. Each closes
# the bill through the Withdraw command, starts the failure cooldown and logs
# ai_withdrawn, and ai_no_viable once per episode: the marker is read before
# the reset (which writes 0) and set after it, so only a pass or the need
# ending (te_tax_ai_step) starts a new episode. The marker set with the
# failure cooldown is what holds an emergency to that cooldown
# (te_tax_ai_initiative_ready).

# Withdrawal: legitimacy below the passage line.
te_tax_ai_withdraw_legitimacy = {
	if = {
		limit = { te_tax_code_on = yes }
		if = {
			limit = { te_tax_can_withdraw = yes }
			te_tax_cmd_withdraw = yes
			set_variable = { name = te_tax_ai_next_month value = te_tax_ai_cooldown_after_withdrawal }
			te_tax_ai_log_withdrawn_legitimacy = yes
			if = {
				limit = { var:te_tax_ai_noviable = 0 }
				te_tax_ai_log_no_viable_legitimacy = yes
			}
			te_tax_ai_reset_bill_state = yes
			set_variable = { name = te_tax_ai_noviable value = 1 }
		}
	}
}

# Step 4, the initiative (plan 2026-10-03, Task 21; spec §2.4 step 4, §2.5, §2.6)

# A new bill, when the step finds none under debate and
# te_tax_ai_initiative_ready holds: the first template that applies, T2 (an
# emergency), T3 (a war levy), T6 (luxury goods), T1 (a raise) or T5 (a cut),
# built in a fresh draft with the draft commands, then introduced and judged in
# the same execution. Exactly one bill; none when no template applies. Not
# gated on is_ai, like the managers: the step's te_tax_ai_can_act is, and the
# console (Task 22) may run it on its own country.
te_tax_ai_initiative = {
	if = {
		limit = {
			te_tax_code_on = yes
			te_tax_code_in_force = yes
			NOT = { te_tax_bill_active = yes }
		}
		if = {
			limit = { te_tax_ai_template_2_applies = yes }
			te_tax_ai_open_draft = yes
			te_tax_ai_build_t2 = yes
			te_tax_ai_introduce_and_judge = { TPL = 2 }
		}
		else_if = {
			limit = { te_tax_ai_template_3_applies = yes }
			te_tax_ai_open_draft = yes
			te_tax_ai_build_t3 = yes
			te_tax_ai_introduce_and_judge = { TPL = 3 }
		}
		else_if = {
			limit = { te_tax_ai_template_6_applies = yes }
			te_tax_ai_open_draft = yes
			te_tax_ai_build_t6 = yes
			te_tax_ai_introduce_and_judge = { TPL = 6 }
		}
		else_if = {
			limit = { te_tax_ai_template_1_applies = yes }
			te_tax_ai_open_draft = yes
			te_tax_ai_build_t1 = yes
			te_tax_ai_introduce_and_judge = { TPL = 1 }
		}
		else_if = {
			limit = { te_tax_ai_template_5_applies = yes }
			te_tax_ai_open_draft = yes
			te_tax_ai_build_t5 = yes
			te_tax_ai_introduce_and_judge = { TPL = 5 }
		}
	}
}

# A fresh draft: a stale one (a player's, left when the country passed to the
# AI) is discarded first, so the picks and exclusions read only the AI's own,
# untouched draft (te_tax_dr_eff_<key>: existing law in its due month).
te_tax_ai_open_draft = {
	if = {
		limit = { te_tax_code_on = yes }
		if = {
			limit = { te_tax_can_draft_discard = yes }
			te_tax_cmd_draft_discard = yes
		}
		if = {
			limit = { te_tax_can_draft_new = yes }
			te_tax_cmd_draft_new = yes
		}
	}
}

# The templates (spec §2.5). Each picks its instrument with the generated
# pre-score (te_tax_gen_ai_pick_raise / _cut, into the locals te_tax_pick and
# te_tax_pick2) and moves it through the generated per-instrument parts, every
# command behind its own trigger. A pick of 0 (every instrument excluded)
# leaves the draft untouched, and its introduction is refused.

# T1, a raise: +1 index step on the cheapest instrument and +1 on the next
# cheapest when there is one, due at the draft's default, now + 3.
te_tax_ai_build_t1 = {
	if = {
		limit = { te_tax_code_on = yes }
		te_tax_gen_ai_pick_raise = yes
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick DIR = 1 }
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick2 DIR = 1 }
	}
}

# T2, the emergency bill: +2 on the cheapest instrument, a minor bill (15
# days' debate), due a month before the default, now + 2.
te_tax_ai_build_t2 = {
	if = {
		limit = { te_tax_code_on = yes }
		te_tax_gen_ai_pick_raise = yes
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick DIR = 1 }
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick DIR = 1 }
		if = {
			limit = { te_tax_can_draft_due = { DIR = 0 } }
			te_tax_cmd_draft_due = { DIR = 0 }
		}
	}
}

# T3, a war levy: +2 on the cheapest instrument with a sunset of
# te_tax_ai_levy_sunset months (24), so it reverts by itself.
te_tax_ai_build_t3 = {
	if = {
		limit = { te_tax_code_on = yes }
		te_tax_gen_ai_pick_raise = yes
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick DIR = 1 }
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick DIR = 1 }
		te_tax_gen_ai_levy_sunset = { INST = local_var:te_tax_pick }
	}
}

# T5, a cut: -1 on the instrument with the largest clout-weighted grievance,
# the highest pre-score.
te_tax_ai_build_t5 = {
	if = {
		limit = { te_tax_code_on = yes }
		te_tax_gen_ai_pick_cut = yes
		te_tax_gen_ai_step_inst = { INST = local_var:te_tax_pick DIR = 0 }
	}
}

# T6, luxury goods: one or two untaxed luxuries, in the generator's order.
te_tax_ai_build_t6 = {
	if = {
		limit = { te_tax_code_on = yes }
		te_tax_gen_ai_tax_luxury = yes
	}
}

# Introduces the draft and judges the bill in the same execution (spec §2.6,
# introduce and observe: the bill's own support refresh scores it). TPL: the
# template's code. The draft is discarded either way: the bill holds its copy,
# and an AI draft lives for one step.
#   - Introduced: the template and the month are recorded (te_tax_ai_tpl,
#     te_tax_ai_bill_month), ai_introduced is logged, and a bill hopeless
#     already (te_tax_ai_bill_hopeless) is withdrawn at once with its reason
#     (te_tax_ai_withdraw: failure cooldown, ai_withdrawn, ai_no_viable once
#     per episode). Introduction and withdrawal cost nothing politically.
#   - Refused (te_tax_can_introduce failed; e.g. every instrument excluded, or
#     Traditionalism's rule): no bill to withdraw. The failure cooldown, so it
#     is not retried every month, ai_withdrawn reason=draft for the attempt,
#     and ai_no_viable reason=draft once per episode, with the same marker.
te_tax_ai_introduce_and_judge = {
	if = {
		limit = { te_tax_code_on = yes }
		if = {
			limit = { te_tax_can_introduce = yes }
			te_tax_cmd_introduce = yes
		}
		if = {
			limit = { te_tax_can_draft_discard = yes }
			te_tax_cmd_draft_discard = yes
		}
		if = {
			limit = { te_tax_bill_active = yes }
			set_variable = { name = te_tax_ai_tpl value = $TPL$ }
			set_variable = { name = te_tax_ai_bill_month value = te_tax_ai_now }
			te_tax_ai_log_introduced = yes
			if = {
				limit = { te_tax_ai_bill_hopeless = yes }
				te_tax_ai_withdraw = yes
			}
		}
		else = {
			set_variable = { name = te_tax_ai_next_month value = te_tax_ai_cooldown_after_withdrawal }
			te_tax_ai_log_withdrawn_draft = yes
			if = {
				limit = { var:te_tax_ai_noviable = 0 }
				te_tax_ai_log_no_viable_draft = yes
			}
			set_variable = { name = te_tax_ai_noviable value = 1 }
		}
	}
}

# No AI bill, offer chain or retry survives a civil war or a release: the
# outbreak leaves the bill with the government that wrote it, the win repair
# closes the winner's, and a released country starts without one. The
# streaks, the phase and the cooldown are the country's own and stay. Called
# by te_tax_copy_code (on the rebels), te_tax_repair_after_civil_war and
# te_tax_init_released_country.

	if = {
		limit = { te_tax_code_on = yes }
		set_variable = { name = te_tax_ai_tpl value = 0 }
		set_variable = { name = te_tax_ai_bill_month value = -1 }
		set_variable = { name = te_tax_ai_offer_month value = -1 }
		set_variable = { name = te_tax_ai_offer_count value = 0 }
		set_variable = { name = te_tax_ai_retry value = 0 }
		set_variable = { name = te_tax_ai_noviable value = 0 }
	}
}
