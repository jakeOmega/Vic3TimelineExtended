# Nuclear weapon system scripted effects

# ============================================================================
# PROGRAMME FUNDING — the two button effects, and the one refresh site
# ============================================================================
# nuclear_program_effect_increase_funding / _decrease_funding hold the verbatim
# bodies that used to sit inline on the two scripted buttons in
# common/scripted_buttons/nuclear_program_buttons.txt, with one call appended.
# The button and the widget's nuclear_program_funding_sgui both call the helper,
# so the hidden grid and the panel cannot drift apart.
#
# The funding modifier sits on the journal entry, and the engine re-evaluates a
# journal-entry modifier's multiplier in JE scope, where the bare
# `var:nuclear_weapons_program_funding` inside nuclear_weapons_program_current_cost
# reads 'none' ("Value of wrong type ... Got value of type 'none'" on the
# add_modifier line, every recalculation). So the cost is cached on the country
# and read through root, as the UN and banking entries do
# (scripting_best_practices.md, "JE scope can't read country properties").
# nuclear_program_refresh_state_effect re-applies a modifier left over from the
# old form once.

nuclear_program_effect_increase_funding = {
	custom_tooltip = {
		text = "INCREASE_NUCLEAR_WEAPON_PROGRAM_FUNDING_DESC"
		change_variable = {
			name = nuclear_weapons_program_funding
			add = 1
		}
		set_variable = {
			name = nuclear_weapons_program_cost_cached
			value = nuclear_weapons_program_current_cost
		}
		je:je_nuclear_program = {
			if = {
				limit = { has_modifier = nuclear_weapon_program_funding }
				remove_modifier = nuclear_weapon_program_funding
			}
			add_modifier = {
				name = nuclear_weapon_program_funding
				multiplier = root.var:nuclear_weapons_program_cost_cached
			}
		}
	}
	nuclear_program_refresh_state_effect = yes
}

nuclear_program_effect_decrease_funding = {
	custom_tooltip = {
		text = "DECREASE_NUCLEAR_WEAPON_PROGRAM_FUNDING_DESC"
		change_variable = {
			name = nuclear_weapons_program_funding
			subtract = 1
		}
		set_variable = {
			name = nuclear_weapons_program_cost_cached
			value = nuclear_weapons_program_current_cost
		}
		je:je_nuclear_program = {
			if = {
				limit = { has_modifier = nuclear_weapon_program_funding }
				remove_modifier = nuclear_weapon_program_funding
			}
			add_modifier = {
				name = nuclear_weapon_program_funding
				multiplier = root.var:nuclear_weapons_program_cost_cached
			}
		}
	}
	nuclear_program_refresh_state_effect = yes
}

# ----------------------------------------------------------------------------
# The programme's weekly step: progress, warheads, development events.
# ----------------------------------------------------------------------------
# CALL-SITE CONTRACT. Advances state and is NOT idempotent within a week (it
# adds progress and may build a warhead). Its one caller is je_nuclear_program's
# on_weekly_pulse, which calls it only while nuclear_program_has_programme
# holds, and runs nuclear_program_refresh_state_effect straight after it.
# Never call it from an event, a scripted GUI or a diplomatic action. ROOT is
# the country.
#
# Moved here verbatim from the entry when posture and crises were merged into
# it (2026-09-25), so the entry's pulse reads as programme, then posture.
nuclear_program_weekly_progress = {
	# A pause treaty zeroes funding. Taking the funding modifier back off is
	# left to nuclear_program_refresh_state_effect, which the entry's pulse runs
	# straight after this effect — the one site that owns that modifier. The
	# remove_modifier that used to sit here ran in *country* scope, while the
	# buttons apply the modifier in `je:je_nuclear_program` scope — so it was a
	# silent no-op and a paused programme went on paying its full weekly
	# innovation bill for ever.
	if = {
		limit = {
			modifier:country_nuclear_program_pause_bool = yes
		}
		set_variable = {
			name = nuclear_weapons_program_funding
			value = 0
		}
	}
	change_variable = {
		name = nuclear_weapon_program_progress
		add = {
			value = nuclear_weapon_program_monthly_progress
			divide = 4
			multiply = {
				value = 1
				add = modifier:country_nuclear_program_progress_mult
			}
		}
	}
	while = {
		limit = {var:nuclear_weapon_program_progress >= nuclear_weapon_program_goal_value}
		change_variable = {
			name = nuclear_weapon_program_progress
			subtract = nuclear_weapon_program_goal_value
		}
		change_variable = {
			name = nuclear_weapon_stockpile
			add = 1
		}
		if = {
			limit = { var:nuclear_weapons_program_first_nuke_done = 0 }
			set_variable = {
				name = nuclear_weapons_program_first_nuke_done
				value = 1
			}
			change_variable = {
				name = nuclear_weapon_program_progress
				multiply = nuclear_weapon_program_additional_nuke_progress_multiplier
			}
			# The year of the first device, for the panel's "First
			# device" row. `year` is a valid script value (see
			# common/script_values/te_history_values.txt).
			set_variable = {
				name = nuclear_program_first_device_year
				value = year
			}
			# Track whether this is the first nuclear power in the world
			if = {
				limit = { NOT = { has_global_variable = world_first_nuclear_weapon } }
				set_variable = { name = is_world_first_nuclear_power value = yes }
			}
			set_global_variable = { name = world_first_nuclear_weapon value = yes }
			trigger_event = { id = nuclear_weapon_events.10 }
			save_scope_as = first_nuke_country
			every_country = {
				limit = {
					NOT = { THIS = scope:first_nuke_country }
				}
				trigger_event = { id = nuclear_weapon_events.9 days = 14 }
			}
		}
	}

	# ---- Development events (only during first nuke, with active funding) ----
	if = {
		limit = {
			var:nuclear_weapons_program_first_nuke_done = 0
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding >= 1
			var:nuclear_weapon_program_progress >= 5
		}
		random_list = {
			490 = { } # nothing happens
			5 = {
				trigger_event = { id = nuclear_weapon_events.19 }
			}
			5 = {
				trigger_event = { id = nuclear_weapon_events.20 }
			}
		}
	}

	# ---- Proliferation alert (non-first country nearing completion) ----
	if = {
		limit = {
			has_global_variable = world_first_nuclear_weapon
			NOT = { has_variable = is_world_first_nuclear_power }
			var:nuclear_weapons_program_first_nuke_done = 0
			var:nuclear_weapon_program_progress >= 75
			NOT = { has_variable = nuclear_proliferation_alert_sent }
		}
		set_variable = { name = nuclear_proliferation_alert_sent value = 1 }
		save_scope_as = proliferating_country
		every_country = {
			limit = {
				NOT = { THIS = ROOT }
				NOT = { is_country_type = decentralized }
			}
			trigger_event = { id = nuclear_weapon_events.18 days = 7 }
		}
	}
}

# ----------------------------------------------------------------------------
# The single derivation site for everything the panel displays.
# ----------------------------------------------------------------------------
# Called from exactly four places in play, which between them cover every way the
# programme's state can move: je_nuclear_program's `immediate` (so an active
# entry always has a status code), its on_weekly_pulse right after the
# programme step, and the two button effects above — and therefore both stepper
# ops in the widget, which delegate to those effects. (events/te_debug_nuclear_events.txt also calls it,
# from the console only, after forcing state by hand.)
# In `immediate` the journal entry is still being
# activated, which is why the modifier guard below reaches it through `?=`:
# whether `je:je_nuclear_program` resolves that early is untested, and either
# answer is safe (unresolved is skipped; resolved has no modifier yet on a fresh
# entry, and on a re-activation the old entry's modifier died with it). Nothing
# else may write nuclear_program_last_status, and neither the .gui nor the
# localization may re-derive it — see
# docs/guides/scripting_best_practices.md, "Journal-Entry Widgets: Derive Display
# State Once, in Script".
#
# WHY THE FUNDING MODIFIER IS DECIDED FROM THE VARIABLE, NOT FROM has_modifier.
# add_modifier / remove_modifier results are not visible inside the same effect
# block, and a scripted effect call is inlined into its caller's block, so a
# has_modifier read here may see the state as it was *before* the button body
# ran. change_variable results, by contrast, are visible immediately. The
# decision is therefore taken from var:nuclear_weapons_program_funding, which
# this block's caller has just written. Walking every caller, with the invariant
# "funding >= 1 implies the modifier is on the journal entry":
#   increase, F -> F+1       funding >= 1, guard skipped; the button body already
#                            re-added the modifier at the new multiplier.
#   decrease, F -> F-1 >= 1  same: nothing to do.
#   decrease, 1 -> 0         funding = 0. Pre-block the modifier was present
#                            (funding was 1), so the has_modifier read is true,
#                            and the body removed and re-added it at multiplier
#                            0 — so it really is present and the remove lands.
#                            Without this the entry would keep a "-0 weekly
#                            innovation" row for ever.
#   pulse, pause in force    the pulse zeroes funding earlier in the same block;
#                            the modifier is genuinely still present, so the read
#                            is true and the innovation drain finally stops.
#   pulse, standing lost     the same, from the entry's own else_if: an armed
#                            power demoted below nuclear_program_has_programme
#                            keeps the entry but stops paying.
#   pulse, already at 0      modifier absent, read false, so no remove_modifier
#                            on a missing modifier (which logs an error).
#   pulse, steady funding>=1 funding /= 0, guard skipped, modifier left exactly
#                            as it is rather than re-added every week.
# The guard is correct whether the has_modifier read sees the pre-block or the
# live state, because the only caller where the two differ is "decrease 1 -> 0",
# and there both readings say "present".
nuclear_program_refresh_state_effect = {
	# Migration: a save funded before the cost was cached carries the modifier
	# with the old JE-scope multiplier; re-apply it once in the cached form.
	if = {
		limit = {
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding >= 1
			NOT = { has_variable = nuclear_weapons_program_cost_cached }
		}
		set_variable = {
			name = nuclear_weapons_program_cost_cached
			value = nuclear_weapons_program_current_cost
		}
		je:je_nuclear_program ?= {
			if = {
				limit = { has_modifier = nuclear_weapon_program_funding }
				remove_modifier = nuclear_weapon_program_funding
			}
			add_modifier = {
				name = nuclear_weapon_program_funding
				multiplier = root.var:nuclear_weapons_program_cost_cached
			}
		}
	}
	if = {
		limit = {
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding = 0
			je:je_nuclear_program ?= { has_modifier = nuclear_weapon_program_funding }
		}
		je:je_nuclear_program ?= { remove_modifier = nuclear_weapon_program_funding }
	}

	# The status code the panel and the status line read. Order matters: a pause
	# treaty forces funding to 0, so without testing the pause first a frozen
	# programme would report itself as merely unfunded.
	if = {
		limit = { modifier:country_nuclear_program_pause_bool = yes }
		set_variable = { name = nuclear_program_last_status value = 3 }
	}
	else_if = {
		limit = {
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding = 0
		}
		set_variable = { name = nuclear_program_last_status value = 0 }
	}
	else_if = {
		limit = {
			has_variable = nuclear_weapons_program_first_nuke_done
			var:nuclear_weapons_program_first_nuke_done > 0
		}
		set_variable = { name = nuclear_program_last_status value = 2 }
	}
	else = {
		set_variable = { name = nuclear_program_last_status value = 1 }
	}
}

# Updates global ranking of nuclear powers (top 10 by stockpile, AS THE WORLD
# ESTIMATES IT — nd_display_public_estimate, re-observed yearly with an error;
# docs/systems/nuclear_crisis_design.md §0.6). The panel shows these
# estimates for everyone, the viewer included; our own exact count is the
# programme panel's "Warheads held" row. Called from je_nuclear_program
# on_weekly_pulse.
update_nuclear_powers_ranking = {
	# Clear stale rank slots
	remove_global_variable = nuke_rank_1
	remove_global_variable = nuke_rank_2
	remove_global_variable = nuke_rank_3
	remove_global_variable = nuke_rank_4
	remove_global_variable = nuke_rank_5
	remove_global_variable = nuke_rank_6
	remove_global_variable = nuke_rank_7
	remove_global_variable = nuke_rank_8
	remove_global_variable = nuke_rank_9
	remove_global_variable = nuke_rank_10

	# Single ordered pass: top 10 countries by nuclear stockpile
	set_global_variable = { name = nuke_ldr_idx value = 0 }
	ordered_country = {
		limit = {
			has_modifier = nuclear_power
			has_variable = nuclear_weapon_stockpile
			var:nuclear_weapon_stockpile >= 1
		}
		order_by = nd_display_public_estimate
		check_range_bounds = no

		if = {
			limit = { global_var:nuke_ldr_idx < 1 }
			set_global_variable = { name = nuke_rank_1 value = THIS }
			set_global_variable = { name = nuke_rank_1_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 2 }
			set_global_variable = { name = nuke_rank_2 value = THIS }
			set_global_variable = { name = nuke_rank_2_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 3 }
			set_global_variable = { name = nuke_rank_3 value = THIS }
			set_global_variable = { name = nuke_rank_3_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 4 }
			set_global_variable = { name = nuke_rank_4 value = THIS }
			set_global_variable = { name = nuke_rank_4_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 5 }
			set_global_variable = { name = nuke_rank_5 value = THIS }
			set_global_variable = { name = nuke_rank_5_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 6 }
			set_global_variable = { name = nuke_rank_6 value = THIS }
			set_global_variable = { name = nuke_rank_6_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 7 }
			set_global_variable = { name = nuke_rank_7 value = THIS }
			set_global_variable = { name = nuke_rank_7_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 8 }
			set_global_variable = { name = nuke_rank_8 value = THIS }
			set_global_variable = { name = nuke_rank_8_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 9 }
			set_global_variable = { name = nuke_rank_9 value = THIS }
			set_global_variable = { name = nuke_rank_9_stockpile value = nd_display_public_estimate }
		}
		else_if = {
			limit = { global_var:nuke_ldr_idx < 10 }
			set_global_variable = { name = nuke_rank_10 value = THIS }
			set_global_variable = { name = nuke_rank_10_stockpile value = nd_display_public_estimate }
		}

		change_global_variable = { name = nuke_ldr_idx add = 1 }
	}
}
