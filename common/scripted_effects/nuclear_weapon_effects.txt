# Nuclear weapon system scripted effects

# ============================================================================
# PROGRAMME FUNDING — the two button effects, and the one refresh site
# ============================================================================
# nuclear_program_effect_increase_funding / _decrease_funding hold the verbatim
# bodies that used to sit inline on the two scripted buttons in
# common/scripted_buttons/nuclear_program_buttons.txt, with one call appended.
# The button and the widget's nuclear_program_funding_sgui both call the helper,
# so the hidden grid and the panel cannot drift apart.
#
# They only move the funding variable. The weekly innovation it costs is
# je_nuclear_program's declared nuclear_weapon_program_funding (value
# nuclear_program_display_weekly_cost), which follows the variable at the
# entry's next update. The custom tooltip is the click's preview: it names the
# step's cost, since change_variable renders nothing.

nuclear_program_effect_increase_funding = {
	custom_tooltip = {
		text = "INCREASE_NUCLEAR_WEAPON_PROGRAM_FUNDING_DESC"
		change_variable = {
			name = nuclear_weapons_program_funding
			add = 1
		}
	}
	nuclear_program_refresh_state_effect = yes
}

nuclear_program_effect_decrease_funding = {
	custom_tooltip = {
		text = "DECREASE_NUCLEAR_WEAPON_PROGRAM_FUNDING_DESC"
		change_variable = {
			name = nuclear_weapons_program_funding
			subtract = 1
		}
	}
	nuclear_program_refresh_state_effect = yes
}

# ----------------------------------------------------------------------------
# The programme's weekly step: progress, warheads, development events.
# ----------------------------------------------------------------------------
# CALL-SITE CONTRACT. Advances state and is NOT idempotent within a week (it
# adds progress and may build a warhead). Its one caller is je_nuclear_program's
# on_weekly_pulse, which calls it only while nuclear_program_has_programme
# holds, and runs nuclear_program_refresh_state_effect straight after it.
# Never call it from an event, a scripted GUI or a diplomatic action. ROOT is
# the country.
#
# Moved here verbatim from the entry when posture and crises were merged into
# it (2026-09-25), so the entry's pulse reads as programme, then posture.
nuclear_program_weekly_progress = {
	# A pause treaty zeroes funding, and with it the entry's declared funding
	# cost.
	if = {
		limit = {
			modifier:country_nuclear_program_pause_bool = yes
		}
		set_variable = {
			name = nuclear_weapons_program_funding
			value = 0
		}
	}
	change_variable = {
		name = nuclear_weapon_program_progress
		add = {
			value = nuclear_weapon_program_monthly_progress
			divide = 4
			multiply = {
				value = 1
				add = modifier:country_nuclear_program_progress_mult
			}
		}
	}
	while = {
		limit = {var:nuclear_weapon_program_progress >= nuclear_weapon_program_goal_value}
		change_variable = {
			name = nuclear_weapon_program_progress
			subtract = nuclear_weapon_program_goal_value
		}
		change_variable = {
			name = nuclear_weapon_stockpile
			add = 1
		}
		if = {
			limit = { var:nuclear_weapons_program_first_nuke_done = 0 }
			set_variable = {
				name = nuclear_weapons_program_first_nuke_done
				value = 1
			}
			change_variable = {
				name = nuclear_weapon_program_progress
				multiply = nuclear_weapon_program_additional_nuke_progress_multiplier
			}
			# The year of the first device, for the panel's "First
			# device" row. `year` is a valid script value (see
			# common/script_values/te_history_values.txt).
			set_variable = {
				name = nuclear_program_first_device_year
				value = year
			}
			# Built outside the Non-Proliferation Treaty while it stood: a
			# member that stays outside it keeps the Supranational standing
			# case its programme carried (un_case_npt_standing_applies, #804).
			if = {
				limit = {
					un_propose_npt_in_force = yes
					NOT = { has_variable = un_nonproliferation_modifier_on }
				}
				set_variable = un_npt_built_outside
			}
			# Track whether this is the first nuclear power in the world
			if = {
				limit = { NOT = { has_global_variable = world_first_nuclear_weapon } }
				set_variable = { name = is_world_first_nuclear_power value = yes }
				# The nuclear taboo is born with the world's first warhead, here
				# and only here (M1, final-review.md): every later first device
				# (a second, third, … power's) must pass through harmlessly and
				# leave the birth to the monthly seed (nd_taboo_seed_existing_world)
				# for a save that predates the taboo — the seed reads
				# world_first_nuclear_weapon, which this branch is the one to set.
				nd_taboo_birth = yes
			}
			set_global_variable = { name = world_first_nuclear_weapon value = yes }
			trigger_event = { id = nuclear_weapon_events.10 }
			save_scope_as = first_nuke_country
			every_country = {
				limit = {
					NOT = { THIS = scope:first_nuke_country }
				}
				trigger_event = { id = nuclear_weapon_events.9 days = 14 }
			}
		}
	}

	# ---- Development events (only during first nuke, with active funding) ----
	if = {
		limit = {
			var:nuclear_weapons_program_first_nuke_done = 0
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding >= 1
			var:nuclear_weapon_program_progress >= 5
		}
		random_list = {
			490 = { } # nothing happens
			5 = {
				trigger_event = { id = nuclear_weapon_events.19 }
			}
			5 = {
				trigger_event = { id = nuclear_weapon_events.20 }
			}
		}
	}

	# ---- Proliferation alert (non-first country nearing completion) ----
	if = {
		limit = {
			has_global_variable = world_first_nuclear_weapon
			NOT = { has_variable = is_world_first_nuclear_power }
			var:nuclear_weapons_program_first_nuke_done = 0
			var:nuclear_weapon_program_progress >= 75
			NOT = { has_variable = nuclear_proliferation_alert_sent }
		}
		set_variable = { name = nuclear_proliferation_alert_sent value = 1 }
		save_scope_as = proliferating_country
		# Only countries with a reason to care: great powers, rivals in
		# either direction, and neighbours. Every recipient's reply can
		# shave progress off this programme, so sending it to the whole
		# world let ~90 bystanders each chip at it.
		every_country = {
			limit = {
				NOT = { THIS = ROOT }
				NOT = { is_country_type = decentralized }
				OR = {
					country_rank >= rank_value:great_power
					any_rival_country = { THIS = scope:proliferating_country }
					any_rivaling_country = { THIS = scope:proliferating_country }
					any_neighbouring_state = { owner = scope:proliferating_country }
				}
			}
			trigger_event = { id = nuclear_weapon_events.18 days = 7 }
		}
		# ...and the proliferator learns that the world has noticed.
		trigger_event = { id = nuclear_weapon_events.23 days = 7 }
	}
}

# ----------------------------------------------------------------------------
# The single derivation site for everything the panel displays.
# ----------------------------------------------------------------------------
# Called from exactly four places in play, which between them cover every way the
# programme's state can move: je_nuclear_program's `immediate` (so an active
# entry always has a status code), its on_weekly_pulse right after the
# programme step, and the two button effects above — and therefore both stepper
# ops in the widget, which delegate to those effects. (events/te_debug_nuclear_events.txt also calls it,
# from the console only, after forcing state by hand.)
# Nothing else may write nuclear_program_last_status, and neither the .gui nor
# the localization may re-derive it — see
# docs/guides/scripting_best_practices.md, "Journal-Entry Widgets: Derive Display
# State Once, in Script".
#
# It no longer touches the funding modifier: since vanilla 1.15 the entry
# declares it (scaling_modifiers_while_active), with a value that reads
# var:nuclear_weapons_program_funding, so every caller that moves the variable
# moves the cost.
nuclear_program_refresh_state_effect = {
	# The status code the panel and the status line read. Order matters: a pause
	# treaty forces funding to 0, so without testing the pause first a frozen
	# programme would report itself as merely unfunded.
	# 5 and 4 before 3: the taboo's hold carries the same pause flag.
	if = {
		limit = { nd_taboo_is_dismantling = yes }
		set_variable = { name = nuclear_program_last_status value = 5 }
	}
	else_if = {
		limit = { has_modifier = nd_taboo_programme_held }
		set_variable = { name = nuclear_program_last_status value = 4 }
	}
	else_if = {
		limit = { modifier:country_nuclear_program_pause_bool = yes }
		set_variable = { name = nuclear_program_last_status value = 3 }
	}
	else_if = {
		limit = {
			has_variable = nuclear_weapons_program_funding
			var:nuclear_weapons_program_funding = 0
		}
		set_variable = { name = nuclear_program_last_status value = 0 }
	}
	else_if = {
		limit = {
			has_variable = nuclear_weapons_program_first_nuke_done
			var:nuclear_weapons_program_first_nuke_done > 0
		}
		set_variable = { name = nuclear_program_last_status value = 2 }
	}
	else = {
		set_variable = { name = nuclear_program_last_status value = 1 }
	}
}

# Rebuilds nd_powers, the ten largest arsenals AS THE WORLD ESTIMATES THEM
# (nd_display_public_estimate, re-observed yearly with an error;
# docs/systems/nuclear_crisis_design.md §0.6), largest first: a variable list
# of their capitals on the panel's global store (nd_lists_store_ensure,
# nuclear_crisis_effects.txt). The Nuclear Powers section draws it as a
# datamodel (NUC-14 of the 2026-10-27 GUI audit), reading each country through
# its capital (gui_modding_guide.md gotcha #15): its flag, the estimate and its
# public doctrine, our own row tinted. Our exact count is the overview's.
# Called once a month for the world (nd_taboo_monthly_update), and by the
# console harness (te_debug_nuclear_effects.txt).
update_nuclear_powers_ranking = {
	nd_lists_store_ensure = yes
	global_var:nd_lists_store ?= {
		if = {
			limit = { has_variable_list = nd_powers }
			clear_variable_list = nd_powers
		}
	}
	set_global_variable = { name = nuke_ldr_idx value = 0 }
	ordered_country = {
		limit = {
			has_modifier = nuclear_power
			has_variable = nuclear_weapon_stockpile
			var:nuclear_weapon_stockpile >= 1
		}
		order_by = nd_display_public_estimate
		check_range_bounds = no

		if = {
			limit = { global_var:nuke_ldr_idx < 10 }
			capital ?= { save_temporary_scope_as = nd_pw_cap }
			if = {
				limit = { exists = scope:nd_pw_cap }
				global_var:nd_lists_store ?= {
					add_to_variable_list = { name = nd_powers target = scope:nd_pw_cap }
				}
			}
		}
		change_global_variable = { name = nuke_ldr_idx add = 1 }
	}
}
