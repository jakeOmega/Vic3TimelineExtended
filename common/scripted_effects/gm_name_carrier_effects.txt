# PROTOTYPE (v2 spec §3.4, test build 9): the naming's carrier is an empty
# army, created for the naming and disbanded after it. The player types the
# monument's name in the base game's army-rename popup, and
# MilitaryFormation.GetNameNoFormatting returns it as a CString. Earlier
# carriers: a company works but trips the company-limit alerts (its slot
# modifier lands a tick after the company); a state's typed name can't be
# undone by script (reset_state_name resets only the script name), nor by
# clearing the popup's field from the GUI.
#
# Script can create an army but not remove one; the .gui disbands it with
# the player's own command (MilitaryFormation.Disband).

# Country scope. scope:gm_state = the monument's state. Creates the army and
# finds it: every army we had is tagged first, so the new one is the untagged
# one.
gm_name_carrier_open = {
	every_military_formation = {
		set_variable = gm_formation_seen
	}
	create_military_formation = {
		type = army
		hq_region = capital.region
		name = gm_name_carrier_army
	}
	random_military_formation = {
		limit = {
			NOT = { has_variable = gm_formation_seen }
		}
		save_scope_as = gm_new_army
	}
	every_military_formation = {
		limit = { has_variable = gm_formation_seen }
		remove_variable = gm_formation_seen
	}
	if = {
		limit = { exists = scope:gm_new_army }
		scope:gm_new_army = {
			set_variable = gm_carries_name
		}
		set_variable = {
			name = gm_name_carrier_army
			value = scope:gm_new_army
		}
		scope:gm_state = {
			set_variable = gm_being_named
		}
	}
}

# Country scope. Ends a naming, used or cancelled. The .gui disbands the army
# in the same click.
gm_name_carrier_close = {
	remove_variable = gm_name_carrier_army
	every_scope_state = {
		limit = { has_variable = gm_being_named }
		remove_variable = gm_being_named
	}
}
