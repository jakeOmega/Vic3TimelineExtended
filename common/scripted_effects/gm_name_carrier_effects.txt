# PROTOTYPE (v2 spec §3.4, test build 4): end a naming, whether the name was
# used or the naming cancelled. Country scope. Removes company_gm_name_carrier
# and its slot, then the flags that showed it (gm_naming_active, the company
# type's potential) and marked the state being named (gm_being_named).
gm_name_carrier_close = {
	if = {
		limit = { has_company = company_type:company_gm_name_carrier }
		remove_company = company_type:company_gm_name_carrier
	}
	if = {
		limit = { has_modifier = gm_name_carrier_slot }
		remove_modifier = gm_name_carrier_slot
	}
	remove_variable = gm_name_carrier
	remove_variable = gm_naming_active
	every_scope_state = {
		limit = { has_variable = gm_being_named }
		remove_variable = gm_being_named
	}
}
