# Space Race Scripted Effects

# Per-JE on_monthly_pulse body for the milestone progress tick. Country scope.
# Drives 7 of the 8 milestone JEs (solar_colonization has a structurally
# different progress loop and stays inline). The failure roll reads
# sr_risk_pct_$MILESTONE$, the same script value the journal-entry widget shows
# as "Setback risk" — there is no longer a sr_base_risk proxy variable and no
# RISK parameter.
#
# NOTE ON THE COOLDOWN (existing behaviour, displayed as-is by the widget): the
# cooldown suppresses only the roll. Progress is added above it, so a programme
# inside its cooldown runs at full speed with zero chance of a setback.
#
# Every running milestone moves and rolls, whatever its approach: Standard (no
# sr_safe_ / sr_ambitious_ variable) is an approach with no modifier and no
# cost. Until 2026-09-29 that state drifted at a flat 0.5 a month with no roll.
#
# Parameters:
#   $MILESTONE$ — milestone token (suborbital, orbital, moon_landing, probe,
#                 moon_base, mars_landing, interstellar_probe). Resolves the
#                 sr_safe_/sr_ambitious_/sr_progress_/sr_failed_ variable
#                 namespace and the sr_risk_pct_/sr_base_risk_ script values.
sr_monthly_progress_update_effect = {
	if = {
		limit = { sr_ai_should_participate = yes }
		change_variable = { name = sr_progress_$MILESTONE$ add = sr_progress }
		if = {
			limit = { NOT = { sr_has_failure_cooldown = yes } }
			random = {
				chance = sr_risk_pct_$MILESTONE$
				set_variable = { name = sr_failed_$MILESTONE$ value = yes }
				sr_count_setback_base = { MILESTONE = $MILESTONE$ }
				# Ambitious fails big; Safe and Standard suffer the minor setbacks.
				if = {
					limit = { has_variable = sr_ambitious_$MILESTONE$ }
					sr_fire_ambitious_failure_event = { MILESTONE = $MILESTONE$ }
				}
				else = {
					sr_fire_safe_setback_event = { MILESTONE = $MILESTONE$ }
				}
			}
		}
	}
	sr_set_milestone_status_base = { MILESTONE = $MILESTONE$ }
}

# Recalculate space program cost modifiers on each active JE
# Called whenever funding/approach changes or a milestone completes
sr_recalculate_cost = {
	# Remove per-JE cost and funding progress modifiers from all JEs.
	# ?= silently no-ops when the JE isn't currently active.
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = suborbital         }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = orbital            }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = moon_landing       }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = probe              }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = moon_base          }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = mars_landing       }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = interstellar_probe }
	sr_clear_milestone_cost_modifiers_base = { MILESTONE = solar_colonization }
	# Apply per-JE cost and funding progress modifiers for active milestones.
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = suborbital         }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = orbital            }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = moon_landing       }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = probe              }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = moon_base          }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = mars_landing       }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = interstellar_probe }
	sr_apply_milestone_cost_modifiers_base = { MILESTONE = solar_colonization }
	# The participation the approach costs above were computed with (sr_<m>_cost
	# gates them on sr_ai_should_participate). space_race_on_action compares it
	# with the live value each month and recalculates when it is stale.
	if = {
		limit = { sr_ai_should_participate = yes }
		set_variable = { name = sr_cost_participation value = 1 } # REVIEWED 2026-09-26: a cache of live state (global ranking), re-derived on every call; nothing a revolution's winner could lose
	}
	else = {
		set_variable = { name = sr_cost_participation value = 0 } # REVIEWED 2026-09-26: a cache of live state (global ranking), re-derived on every call; nothing a revolution's winner could lose
	}
	# The funding cap the pace multipliers above were computed with
	# (sr_funding_<m>_effective reads sr_max_funding_level; lunar colonization
	# design § 5.1). Technologies and a power bloc principle move the cap with
	# no button press, so space_race_on_action recalculates when this is stale.
	set_variable = { name = sr_cost_funding_cap value = sr_max_funding_level } # REVIEWED 2026-10-29: a cache of live state (the funding cap), re-derived on every call; nothing a revolution's winner could lose
	# Whether solar colonization was billed (sr_milestone_billed_solar_colonization).
	# Its production method can be switched off with no button press, so
	# space_race_on_action recalculates when this is stale.
	if = {
		limit = { sr_milestone_billed_solar_colonization = yes }
		set_variable = { name = sr_cost_solar_running value = 1 } # REVIEWED 2026-09-29: a cache of live state (the production method), re-derived on every call; nothing a revolution's winner could lose
	}
	else = {
		set_variable = { name = sr_cost_solar_running value = 0 } # REVIEWED 2026-09-29: a cache of live state (the production method), re-derived on every call; nothing a revolution's winner could lose
	}
	# Refresh the widget's display state. This effect is the shared tail of every
	# approach / funding button, every journal entry's `immediate`, milestone
	# completion and the monthly cleanup, so the panel's status label reacts to a
	# click instead of lagging a month. Status is derived from variables only
	# (never from a modifier), so the add_modifier churn above cannot affect it.
	sr_refresh_milestone_statuses = yes # REVIEWED 2026-09-26: each milestone's status is derived from its own variables; one not running has its status and setback count cleared, as the monthly cleanup does
}

	# programme to restart (sr_launch_bank_staged_fleets, sr_launch_window_effects.txt).
	if = {
		limit = { NOT = { has_journal_entry = je_space_race_solar_colonization } }
		sr_launch_bank_staged_fleets = yes
	}
	sr_cleanup_milestone_if_inactive_base = { MILESTONE = solar_colonization }
	# interstellar_results uses a different variable set (only sr_active_interstellar_results
	# and sr_interstellar_transit_progress) — cannot share the standard 5-var base.
	if = {
		limit = { NOT = { has_journal_entry = je_space_race_interstellar_results } }
		if = {
			limit = {
				OR = {
					has_variable = sr_active_interstellar_results
					has_variable = sr_interstellar_transit_progress
				}
			}
			set_variable = { name = sr_space_race_cleanup_changed value = yes }
			if = { limit = { has_variable = sr_active_interstellar_results } remove_variable = sr_active_interstellar_results }
			if = { limit = { has_variable = sr_interstellar_transit_progress } remove_variable = sr_interstellar_transit_progress }
		}
	}
	# Old saves: sr_active_milestone was retired for sr_has_running_milestone
	# (#480). Nothing reads it any more; strip what a save still carries.
	if = {
		limit = { has_variable = sr_active_milestone }
		remove_variable = sr_active_milestone
	}
	if = {
		limit = { has_variable = sr_space_race_cleanup_changed }
		remove_variable = sr_space_race_cleanup_changed
		sr_recalculate_cost = yes
	}
}

# --- The setback a failure event reports -------------------------------------
# A failure event is about ONE milestone: the one whose monthly roll failed.
# sr_fire_ambitious_failure_event / sr_fire_safe_setback_event save the country
# as scope:sr_setback_<milestone> before they dispatch, and saved scopes travel
# with trigger_event, so the event and each of its options know which milestone
# failed. The four helpers below act on that marked milestone only — never on
# "every running milestone", and never on the sr_failed_<milestone> flags, which
# two journal entries can raise in the same month (each entry's pulse is its own
# script run, so markers never mix between entries).
#
# sr_failed_<milestone> is still raised by the pulse: it is the "setback not yet
# answered" bookkeeping, and each option clears the flag of its own milestone.
#
# A failure event that carries no marker — one already open in a save from
# before the marker existed, or one fired from the console — falls back to the
# pre-marker behaviour: every helper first runs sr_resolve_setback_markers, which
# (only when no marker exists) marks each milestone whose sr_failed_ flag is
# raised. The first helper of an option does it; the rest then see the markers
# even after the safety review has cleared the flags. te_debug_space_race.1
# option j fires a setback through the real dispatcher;
# te_debug_sr_force_setback saves the markers itself.

# Apply a temporary safety review to the failed milestone's journal entry.
sr_temporary_safety_review_effect = {
	sr_resolve_setback_markers = yes
	sr_apply_safety_review_to_milestone_base = { MILESTONE = suborbital         }

	sr_apply_safety_review_to_milestone_base = { MILESTONE = moon_landing       }

}

# --- Progress in months -------------------------------------------------------
# Every event that moves a milestone's progress moves it by whole months of the
# programme's pace: sr_progress, what the monthly pulse adds to each running
# milestone (2026-10-25). Until then a setback kept x0.75 (Ambitious) or x0.85
# (Safe, Standard) of the progress, most options took a flat 3 to 10 points on
# top, and the in-progress events gave flat 2 to 30. Both scaled badly. The risk
# is a monthly roll, so a milestone sees setbacks in proportion to the time it
# takes, and each one took a share of a progress that grows toward a goal up to
# 13 times larger by the end of the race: a slow programme settled where its
# pace equalled its losses. Simulated on Standard, the Mars Landing at a pace of
# 4 a month took 880 months on average against 113 without setbacks (151 now),
# and at 2 never finished. The flat amounts meant nothing past the first
# milestones.
# In months, a setback costs every milestone the same share of its time,
# whatever its goal or pace, and the risk alone makes later milestones harder.
#
# A month is the pace sr_record_event_pace stored when the event fired (each
# such event calls it from its immediate), so changing funding while an event
# waits changes neither what it costs nor what it pays (sr_event_pace_value).
# Two such events open at once share the variable; the second records the pace
# of a moment later, which differs only if funding changed in between.
sr_record_event_pace = {
	set_variable = { name = sr_event_pace value = sr_progress }
}

# Add $MONTHS$ months of progress to one milestone. Country scope.
sr_gain_progress_months = {
	if = {
		limit = { has_variable = sr_progress_$MILESTONE$ }
		change_variable = {
			name = sr_progress_$MILESTONE$
			add = {
				value = sr_event_pace_value
				multiply = $MONTHS$
			}
		}
	}
}

# Take $MONTHS$ months of progress from one milestone: never more than it has,
# nor more than one goal. Progress past one goal is Solar System Colonization's
# fleets staged in orbit (Global Commons § 5.2), which a setback never touches,
# so a large staged bank doesn't make every setback ruinous (and gives no
# reason to cut funding while waiting); every other milestone completes at its
# goal. Country scope.
sr_lose_progress_months = {
	if = {
		limit = { has_variable = sr_progress_$MILESTONE$ }
		change_variable = {
			name = sr_progress_$MILESTONE$
			subtract = {
				value = sr_event_pace_value
				multiply = $MONTHS$
				max = sr_$MILESTONE$_goal
				max = var:sr_progress_$MILESTONE$
			}
		}
	}
}

# Notify all countries about a space race milestone achievement
# Scope: Country that achieved the milestone
# $MILESTONE$ = suborbital, orbital, moon_landing, probe, moon_base, mars_landing, interstellar_probe
#
# Each receiver gets one entry per completion on a per-milestone LIST
# (sr_notify_achievers_<m>, holding the achiever's capital — the capital chain is
# save-reload safe for loc display) and one queued space_race_events.20. Each
# popup takes exactly one entry, so two completions inside the 14-day delay —
# of different milestones, or the same milestone by two countries — give two
# popups, each naming its own achiever. A single shared receiver variable used to
# be overwritten by the second completion, crediting the wrong country in the
# first popup and blanking the second. "World first" is read at fire time from
# the achiever's own sr_was_first_<m>.
sr_notify_milestone_achievement = {
	save_scope_as = sr_achiever
	custom_tooltip = {
		text = sr_notify_achievement_tt
		every_country = {
			limit = {
				NOT = { THIS = scope:sr_achiever }
				is_country_type = recognized
			}
			add_to_variable_list = { name = sr_notify_achievers_$MILESTONE$ target = scope:sr_achiever.capital }
			trigger_event = { id = space_race_events.20 days = 14 }
		}
	}
}

# space_race_events.20's side of the list above. Country scope (the receiver).
# Takes one pending entry for $MILESTONE$ and turns it into the scopes the
# event's text and options read.
sr_take_milestone_notice_base = {
	# The world first's entry goes first. The list has no order to rely on
	# (remove_list_variable swaps the last element into the hole), so with the
	# first and a follower both pending an unfiltered pick would, half the time,
	# show the follower on day 14 and "No nation has done this before" days later.
	if = {
		limit = {
			any_in_list = {
				variable = sr_notify_achievers_$MILESTONE$
				owner ?= { has_variable = sr_was_first_$MILESTONE$ }
			}
		}
		random_in_list = {
			variable = sr_notify_achievers_$MILESTONE$
			limit = { owner ?= { has_variable = sr_was_first_$MILESTONE$ } }
			save_scope_as = sr_notice_capital
		}
	}
	else = {
		random_in_list = {
			variable = sr_notify_achievers_$MILESTONE$
			save_scope_as = sr_notice_capital
		}
	}
	remove_list_variable = { name = sr_notify_achievers_$MILESTONE$ target = scope:sr_notice_capital }
	scope:sr_notice_capital = { owner = { save_scope_as = sr_achiever } }
	save_scope_as = sr_milestone_$MILESTONE$
	if = {
		limit = { has_variable = sr_completed_$MILESTONE$ }
		save_scope_as = sr_already_achieved
	}
	if = {
		limit = { scope:sr_achiever = { has_variable = sr_was_first_$MILESTONE$ } }
		save_scope_as = sr_is_first
	}
}

# --- Per-milestone completion / reward / cleanup helpers ---------------------
# Each milestone exposes three named entry points (sr_complete_<X>_effect,
# sr_apply_<X>_reward, and the JE-driven sr_cleanup_milestone_variables_effect)
# so callers in events/ and journal_entries/ can grep for the concrete name.
# All three delegate to a single $MILESTONE$-parameterized base.
#
# Supported $MILESTONE$ values (must match a corresponding sr_<X> /
# sr_first_<X> static modifier and the per-JE variable namespace):
#   suborbital | orbital | moon_landing | probe | moon_base | mars_landing |
#   interstellar_probe

sr_complete_milestone_effect_base = {
	set_variable = { name = sr_completed_$MILESTONE$ value = yes }
	if = {
		limit = { NOT = { has_global_variable = sr_global_first_$MILESTONE$ } }
		set_global_variable = { name = sr_global_first_$MILESTONE$ value = yes }
		set_variable = { name = sr_was_first_$MILESTONE$ value = yes }
	}
	sr_notify_milestone_achievement = { MILESTONE = $MILESTONE$ }
	if = { limit = { has_variable = sr_funding_$MILESTONE$ } remove_variable = sr_funding_$MILESTONE$ }
	if = { limit = { has_variable = sr_safe_$MILESTONE$ } remove_variable = sr_safe_$MILESTONE$ }
	if = { limit = { has_variable = sr_ambitious_$MILESTONE$ } remove_variable = sr_ambitious_$MILESTONE$ }
	if = { limit = { has_variable = sr_progress_$MILESTONE$ } remove_variable = sr_progress_$MILESTONE$ }
	sr_complete_milestone_effect = yes
}

# Adds the reward modifier and nothing else. Keep it that way:
# sr_restore_milestone_reward_base runs it again for a revolution's winner, so
# a one-off added here (prestige, an event) would be paid twice.
sr_apply_milestone_reward_base = {
	if = {
		limit = { has_variable = sr_was_first_$MILESTONE$ }
		add_modifier = { name = sr_first_$MILESTONE$ }

# Strip the cost and the two funding modifiers (pace and throughput) from one
# milestone's JE. Country scope. Uses ?= so an inactive JE silently no-ops
# instead of erroring. sr_recalculate_cost runs it on every entry before
# sr_apply_milestone_cost_modifiers_base, so each modifier is removed before it
# is added again (a same-name add_modifier stacks).
sr_clear_milestone_cost_modifiers_base = {
	je:je_space_race_$MILESTONE$ ?= {
		if = { limit = { has_modifier = sr_space_program_cost } remove_modifier = sr_space_program_cost }
		if = { limit = { has_modifier = sr_funding_progress   } remove_modifier = sr_funding_progress   }
		if = { limit = { has_modifier = sr_funding_throughput } remove_modifier = sr_funding_throughput }
	}
}

# Re-apply the cost and the two funding modifiers to one milestone's JE,
# gated on sr_milestone_billed_<X> (the active flag; for solar colonization,
# its programme actually running). The pace (sr_funding_progress) takes the
# effective level, which counts the levels past the cap for less; the
# throughput (sr_funding_throughput) and the cost take the level itself, so a
# level past the cap still costs in full and adds its full +25% (lunar
# colonization design § 5.1). Multiplier names interpolate from $MILESTONE$
# (sr_<X>_cost, sr_funding_<X>_effective and sr_funding_<X>_level — verified to
# exist for all 8 tokens). Country scope. ?= because each milestone entry's immediate
# sets sr_active_<X> and then runs sr_recalculate_cost, and when a
# revolution's winner inherits the entry that immediate runs before je: can
# find it ("Failed to scope to journal entry by tag", Spain's moon landing in
# the 2026-10-10 observer run). The monthly pulse then puts them back
# (sr_cost_modifiers_missing, space_race_on_actions.txt).
sr_apply_milestone_cost_modifiers_base = {
	if = {
		limit = { sr_milestone_billed_$MILESTONE$ = yes }
		je:je_space_race_$MILESTONE$ ?= {
			add_modifier = { name = sr_space_program_cost multiplier = sr_$MILESTONE$_cost              }
			add_modifier = { name = sr_funding_progress   multiplier = sr_funding_$MILESTONE$_effective }
			add_modifier = { name = sr_funding_throughput multiplier = sr_funding_$MILESTONE$_level     }
		}
	}
}

# When a milestone JE is no longer active, clear its 5-var family (active /
# progress / funding / safe / ambitious) and flag a cleanup-changed signal so
# the orchestrator runs sr_recalculate_cost once at the end. Country scope.
# interstellar_results does NOT use this base — it has only 2 vars (handled
# inline in sr_cleanup_inactive_space_race_milestones).
sr_cleanup_milestone_if_inactive_base = {
	if = {
		limit = { NOT = { has_journal_entry = je_space_race_$MILESTONE$ } }
		if = {
			limit = {
				OR = {
					has_variable = sr_active_$MILESTONE$
					has_variable = sr_progress_$MILESTONE$
					has_variable = sr_funding_$MILESTONE$
					has_variable = sr_safe_$MILESTONE$
					has_variable = sr_ambitious_$MILESTONE$
					has_variable = sr_$MILESTONE$_last_status
					has_variable = sr_$MILESTONE$_setbacks
				}
			}
			set_variable = { name = sr_space_race_cleanup_changed value = yes }
			if = { limit = { has_variable = sr_active_$MILESTONE$    } remove_variable = sr_active_$MILESTONE$    }
			if = { limit = { has_variable = sr_progress_$MILESTONE$  } remove_variable = sr_progress_$MILESTONE$  }
			if = { limit = { has_variable = sr_funding_$MILESTONE$   } remove_variable = sr_funding_$MILESTONE$   }
			if = { limit = { has_variable = sr_safe_$MILESTONE$      } remove_variable = sr_safe_$MILESTONE$      }
			if = { limit = { has_variable = sr_ambitious_$MILESTONE$ } remove_variable = sr_ambitious_$MILESTONE$ }
			if = { limit = { has_variable = sr_$MILESTONE$_last_status } remove_variable = sr_$MILESTONE$_last_status }
			if = { limit = { has_variable = sr_$MILESTONE$_setbacks    } remove_variable = sr_$MILESTONE$_setbacks    }
		}
		# The rivals estimate (display only; sr_rivals_estimate_refresh): this
		# country's offset as a rival, and its own snapshot as a viewer. Outside
		# the block above, so it never asks for a cost recalculation.
		if = { limit = { has_variable = sr_fuzz_$MILESTONE$       } remove_variable = sr_fuzz_$MILESTONE$       }
		if = { limit = { has_variable = sr_fuzz_until_$MILESTONE$ } remove_variable = sr_fuzz_until_$MILESTONE$ }
		sr_clear_rival_slots = { MILESTONE = $MILESTONE$ }
	}
}

# --- Display state for the journal-entry widget ------------------------------
# gui/journal_entry_widgets/space_race_widget.gui and
# common/customizable_localization/space_race_custom_loc.txt only READ these two
# variables. Neither the GUI nor localization re-derives them.

# The one derivation site for sr_<MILESTONE>_last_status, the integer the widget
# and the profile custom loc branch on. Country scope. Called from two places:
# sr_refresh_milestone_statuses (the tail of sr_recalculate_cost — every button,
# every journal entry's `immediate`, completion and cleanup — and of the two
# mission-failure effects, which start the cooldown) and the tail of the monthly
# progress pulse.
#
#   (no variable)  the milestone is not running — the panel shows it as Idle
#   0              standard approach (neither variable), exposed to setbacks
#   1              safe approach, exposed to setbacks
#   2              ambitious approach, exposed to setbacks
#   3              safe approach, inside the post-setback cooldown
#   4              ambitious approach, inside the post-setback cooldown
#   5              standard approach, inside the post-setback cooldown
#
# States 3 to 5 are "shielded, not paused": the cooldown suppresses only the
# failure roll, so progress continues at full rate. The widget says exactly that.
# Deliberately no "programme incompatible" state — that is the journal entry's
# own `fail` block, which already explains itself through
# sr_space_program_incompatible_tt, and it would need a second parameter because
# the eight milestones gate on five different country_sr_*_program_bool modifiers.
sr_set_milestone_status_base = {
	if = {
		limit = { NOT = { has_variable = sr_active_$MILESTONE$ } }
		if = { limit = { has_variable = sr_$MILESTONE$_last_status } remove_variable = sr_$MILESTONE$_last_status }
		if = { limit = { has_variable = sr_$MILESTONE$_setbacks    } remove_variable = sr_$MILESTONE$_setbacks    }
	}
	else_if = {
		limit = { has_variable = sr_safe_$MILESTONE$ }
		sr_ensure_setback_counter_base = { MILESTONE = $MILESTONE$ }
		if = {
			limit = { sr_has_failure_cooldown = yes }
			set_variable = { name = sr_$MILESTONE$_last_status value = 3 }
		}
		else = {
			set_variable = { name = sr_$MILESTONE$_last_status value = 1 }
		}
	}
	else_if = {
		limit = { has_variable = sr_ambitious_$MILESTONE$ }
		sr_ensure_setback_counter_base = { MILESTONE = $MILESTONE$ }
		if = {
			limit = { sr_has_failure_cooldown = yes }
			set_variable = { name = sr_$MILESTONE$_last_status value = 4 }
		}
		else = {
			set_variable = { name = sr_$MILESTONE$_last_status value = 2 }
		}
	}
	else = {
		sr_ensure_setback_counter_base = { MILESTONE = $MILESTONE$ }
		if = {
			limit = { sr_has_failure_cooldown = yes }
			set_variable = { name = sr_$MILESTONE$_last_status value = 5 }
		}
		else = {
			set_variable = { name = sr_$MILESTONE$_last_status value = 0 }
		}
	}
}

# The panel's pace line prints the setback count beside the risk, so the counter
# has to exist for as long as the milestone does — including in a save where the

# status derivation above, which runs from every button, every `immediate`, the
# monthly pulse and the cleanup.
sr_ensure_setback_counter_base = {
	if = {
		limit = { NOT = { has_variable = sr_$MILESTONE$_setbacks } }
		set_variable = { name = sr_$MILESTONE$_setbacks value = 0 }
	}
}

# Lifetime setback counter for one milestone, shown by the widget as "Setbacks so
# far". Called from the single site that raises sr_failed_$MILESTONE$ — the
# monthly progress pulse for seven milestones, and the inline solar colonization
# pulse. Self-initialising so an old save that already has the milestone running

sr_count_setback_base = {

}

# Dispatches a milestone-themed ambitious failure event for the milestone whose
# monthly roll just failed. Called from sr_monthly_progress_update_effect and the
# solar colonization pulse, once per failed roll.
#
# $MILESTONE$ is the failed milestone. It is saved as scope:sr_setback_$MILESTONE$
# first; the themed entries below read that marker, not "any running milestone",
# and the event and its options inherit it (saved scopes travel with
# trigger_event). A theme whose milestones did not fail is simply not eligible:
# a probe failure can no longer play out as a habitat breach, and a Moon landing
# failure can no longer say the probe was lost.
sr_fire_ambitious_failure_event = {
	save_scope_as = sr_setback_$MILESTONE$
	random_list = {
		# Generic mission failure (always available)
		10 = {
			trigger_event = { id = space_race_events.10 }
		}
		# Launchpad Explosion (suborbital/orbital)
		10 = {
			trigger = {
				OR = {
					exists = scope:sr_setback_suborbital
					exists = scope:sr_setback_orbital
				}
			}
			trigger_event = { id = space_race_events.12 }
		}
		# Deep Space Signal Lost (probe/interstellar). Same gate as the event's
		# own trigger, so the roll can never pick an event that then refuses to fire.
		10 = {
			trigger = { sr_setback_is_deep_space_probe = yes }
			trigger_event = { id = space_race_events.13 }
		}
		# Habitat Depressurization (moon base/mars/colonization)
		10 = {
			trigger = { sr_setback_is_crewed_outpost = yes }
			trigger_event = { id = space_race_events.14 }
		}
		# Guidance System Malfunction (moon landing)
		10 = {
			trigger = {
				exists = scope:sr_setback_moon_landing
			}
			trigger_event = { id = space_race_events.15 }
		}
		# Booster Separation Failure (orbital)
		10 = {
			trigger = {
				exists = scope:sr_setback_orbital
			}
			trigger_event = { id = space_race_events.16 }
		}
		# Crew Medical Emergency (moon base/mars/colonization)
		10 = {
			trigger = { sr_setback_is_crewed_outpost = yes }
			trigger_event = { id = space_race_events.17 }
		}
		# Program Mismanagement Exposed (always available)
		10 = {
			trigger_event = { id = space_race_events.18 }
		}
		# Suspected Sabotage: "a rival power that cannot afford to see us
		# succeed". With covert warfare on, only while a foreign operation is
		# really working against us — space espionage on the programme, or
		# infrastructure sabotage in one of our states. With the rule off it is

# --- Display-only: rivals' estimated progress --------------------------------
# The Rivals section (space_race_widget.gui, te_sr_sec_rivals) lists every
# other power whose programme is running this milestone and moving, each as an
# estimate of its progress inside a band. The owner asked for "a bar with some
# fuzz ... less if you have an intelligence network there".
#
# - Who: sr_is_rival, the participation test the monthly pulse uses
#   (sr_ai_should_participate). A great power outside the top three can hold
#   sr_active_$MILESTONE$ and never tick a single month of progress, and
#   listing it as a competitor would be a lie.
# - The fuzz is one random offset per rival and milestone, sr_fuzz_<m> (−1 to
#   +1 in nine steps), drawn again a year after it was drawn (sr_fuzz_until_<m>,
#   a month index). Script values cannot roll dice, so it is a variable.
# - The band around it is the viewer's: ±25% of the goal, ±10% or ±5% with a
#   covert network of intelligence tier 1 or 2 in the rival
#   (sr_disp_rival_band). The estimate is progress + offset × band, so the true
#   figure is always inside the band drawn around it. Programmes are announced;
#   their schedules are not, so no row ever reads a rival's sr_progress_<m>
#   straight.
# - The rows come from here, once a month, into the viewer's own variable
#   list sr_rivals_list_<m>: every rival's capital, by the open estimate
#   (sr_rival_est_open_<m>), furthest first (audit SPC-9). The panel draws it as
#   a datamodel of states and reaches each country through its capital
#   (gotcha #11); the numbers are read live, per frame, through script values
#   with the rival passed in as scope:sr_rival.
# - Solar colonization lists its rivals the same way; it has no first to race
#   for (no sr_first_solar_colonization reward, no sr_global_first_ flag: it
#   does not go through sr_complete_milestone_effect_base).
#
# Players only: nobody else looks, and the draws happen here, on the rivals a
# player can see, so no AI pulse carries any of it. Several players seeing one
# rival share its offset, and a month index (not a countdown) dates it, so a
# second player's pulse cannot age it twice. A revolution's winner inherits the
# offsets (variables) but not the list, which the next month rebuilds.
# Scope: country (a player); ROOT is the same country (the monthly pulse).
sr_rivals_estimate_refresh = {
	sr_rivals_estimate_refresh_base = { MILESTONE = suborbital         }
	sr_rivals_estimate_refresh_base = { MILESTONE = orbital            }
	sr_rivals_estimate_refresh_base = { MILESTONE = moon_landing       }
	sr_rivals_estimate_refresh_base = { MILESTONE = probe              }
	sr_rivals_estimate_refresh_base = { MILESTONE = moon_base          }
	sr_rivals_estimate_refresh_base = { MILESTONE = mars_landing       }
	sr_rivals_estimate_refresh_base = { MILESTONE = interstellar_probe }
	sr_rivals_estimate_refresh_base = { MILESTONE = solar_colonization }
}

sr_rivals_estimate_refresh_base = {
	sr_clear_rival_slots = { MILESTONE = $MILESTONE$ }
	if = {
		limit = { has_variable = sr_active_$MILESTONE$ }
		save_scope_as = sr_rivals_viewer
		# 1. Each rival's offset: drawn if it has none, again a year after.
		every_country = {
			limit = { sr_is_rival = { MILESTONE = $MILESTONE$ } }
			if = {
				limit = { NOT = { has_variable = sr_fuzz_until_$MILESTONE$ } }
				sr_roll_rival_fuzz = { MILESTONE = $MILESTONE$ }
			}
			else_if = {
				limit = { var:sr_fuzz_until_$MILESTONE$ <= te_history_month_index }
				sr_roll_rival_fuzz = { MILESTONE = $MILESTONE$ }
			}
		}
		# 2. Every rival's capital, by the open estimate, furthest first. 40 is
		# a cap no game reaches (players and the top three); min/max with
		# check_range_bounds walks the whole list (scripting_best_practices.md,
		# "remove_list_variable Does Not Preserve List Order").
		if = {
			limit = {
				any_country = { sr_is_rival = { MILESTONE = $MILESTONE$ } }
			}
			ordered_country = {
				limit = { sr_is_rival = { MILESTONE = $MILESTONE$ } }
				order_by = sr_rival_est_open_$MILESTONE$
				min = 0
				max = 40
				check_range_bounds = no
				capital = { save_scope_as = sr_rival_capital }
				scope:sr_rivals_viewer = {
					add_to_variable_list = { name = sr_rivals_list_$MILESTONE$ target = scope:sr_rival_capital }
				}
			}
		}
	}
}

# One draw of a rival's offset, in nine even steps from −1 to +1, dated a year
# ahead. random_list takes literal weights (scripting_best_practices.md).
# Scope: the rival.
sr_roll_rival_fuzz = {
	random_list = {
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = -1 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = -0.75 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = -0.5 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = -0.25 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = 0 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = 0.25 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = 0.5 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = 0.75 } }
		1 = { set_variable = { name = sr_fuzz_$MILESTONE$ value = 1 } }
	}
	set_variable = { name = sr_fuzz_until_$MILESTONE$ value = te_history_month_index }
	change_variable = { name = sr_fuzz_until_$MILESTONE$ add = 12 }
}

# The viewer's snapshot for a milestone: the list, and the five slots and the

	if = { limit = { has_variable = sr_rival_$MILESTONE$_2 } remove_variable = sr_rival_$MILESTONE$_2 }
	if = { limit = { has_variable = sr_rival_$MILESTONE$_3 } remove_variable = sr_rival_$MILESTONE$_3 }
	if = { limit = { has_variable = sr_rival_$MILESTONE$_4 } remove_variable = sr_rival_$MILESTONE$_4 }
	if = { limit = { has_variable = sr_rival_$MILESTONE$_5 } remove_variable = sr_rival_$MILESTONE$_5 }
}

# --- Display-only: the Space Powers standings --------------------------------
# The Timeline Extended window's Space Race tab stays open once a country is in
# the race (sr_in_space_race), and opens with Space Powers (te_sr_sec_powers,
# space_race_widget.gui): our programme, then every other power, each as its
# programme's icons and an estimate of its furthest milestone under way.
# Unlike the Rivals rows, which exist only inside a milestone we run, it shows
# between milestones and powers on milestones we never ran.
#
# - Who: every other power in the race (sr_is_space_power), in the order of
#   sr_power_rank_score (space_race_values.txt): what each has achieved and
#   which milestone it is on, never its progress.
# - Into the viewer's own variables, once a month: sr_power_total (how many;
#   missing until the first refresh, so the panel can say "next month"), and
#   the list sr_powers_list, each power's capital in that order (audit SPC-9;
#   gotcha #11), which the panel draws as a datamodel of states. The icons and
#   the estimate are read live, per frame.
# - The estimate is the Rivals rows' (sr_disp_rival_*_<m> with the power passed
#   in as scope:sr_rival), so each power's offset (sr_fuzz_<m>) is drawn here
#   for every milestone it runs, when missing or a year old, as
#   sr_rivals_estimate_refresh_base draws it: the same offset in both places.
#
# Players only, from the monthly pulse (space_race_on_actions.txt). Everything
# is rebuilt every month; a revolution's winner inherits the count, not the
# list, for that month.
# Scope: country (a player in the race); ROOT is the same country.
sr_powers_refresh = {
	save_scope_as = sr_powers_viewer
	every_country = {
		limit = { sr_is_space_power = yes }
		sr_power_fuzz_base = { MILESTONE = suborbital         }
		sr_power_fuzz_base = { MILESTONE = orbital            }
		sr_power_fuzz_base = { MILESTONE = moon_landing       }
		sr_power_fuzz_base = { MILESTONE = probe              }
		sr_power_fuzz_base = { MILESTONE = moon_base          }
		sr_power_fuzz_base = { MILESTONE = mars_landing       }
		sr_power_fuzz_base = { MILESTONE = interstellar_probe }
		sr_power_fuzz_base = { MILESTONE = solar_colonization }
	}
	sr_powers_clear = yes
	set_variable = { name = sr_power_total value = 0 } # REVIEWED 2026-10-25: a display snapshot rebuilt every month; nothing a revolution's winner could lose
	every_country = {
		limit = { sr_is_space_power = yes }
		scope:sr_powers_viewer = { change_variable = { name = sr_power_total add = 1 } }
	}
	# Every power's capital, furthest along first. 40 is a cap no game
	# reaches; min/max with check_range_bounds walks the whole list
	# (sr_rivals_estimate_refresh_base).
	if = {
		limit = { var:sr_power_total >= 1 }
		ordered_country = {
			limit = { sr_is_space_power = yes }
			order_by = sr_power_rank_score
			min = 0
			max = 40
			check_range_bounds = no
			capital = { save_scope_as = sr_power_capital }
			scope:sr_powers_viewer = {
				add_to_variable_list = { name = sr_powers_list target = scope:sr_power_capital }
			}
		}
	}
	sr_powers_list_selfcheck = yes
}

# TE_SELF_CHECK sr_powers_list (audit SPC-9, 2026-10-27), once a game: the
# first time a player's standings hold 2 to 40 other powers, the list must hold
# every one of them (sr_power_total, counted by every_country), which says the
# ordered_country walk with min/max reached the whole field rather than one
# power. Fewer than two can't tell a walk from a single pick, so it waits. That
# the panel draws the list (a datamodel of capitals, State.GetCountry on each)
# is the game's to show: docs/testing/space-race-playtest.md.
# Scope: the viewer (scope:sr_powers_viewer is set).
sr_powers_list_selfcheck = {
	if = {
		limit = {
			NOT = { has_global_variable = sr_selfcheck_powers_list_done }
			var:sr_power_total >= 2
			var:sr_power_total <= 40
		}
		set_global_variable = { name = sr_selfcheck_powers_list_done value = yes }
		set_variable = { name = sr_selfcheck_powers_listed value = 0 }
		if = {
			limit = { has_variable_list = sr_powers_list }
			every_in_list = {
				variable = sr_powers_list
				scope:sr_powers_viewer = { change_variable = { name = sr_selfcheck_powers_listed add = 1 } }
			}
		}
		if = {
			limit = {
				var:sr_selfcheck_powers_listed >= var:sr_power_total
				var:sr_power_total >= var:sr_selfcheck_powers_listed
			}
			debug_log = "TE_SELF_CHECK OK sr_powers_list country=[THIS.GetCountry.GetTagName] date=[TimeKeeper.GetCurrentDate.GetString] powers=[THIS.Var('sr_power_total').GetValue|0] listed=[THIS.Var('sr_selfcheck_powers_listed').GetValue|0] (the Space Powers list holds every other power in the race)"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN sr_powers_list country=[THIS.GetCountry.GetTagName] date=[TimeKeeper.GetCurrentDate.GetString] powers=[THIS.Var('sr_power_total').GetValue|0] listed=[THIS.Var('sr_selfcheck_powers_listed').GetValue|0] (the Space Powers and Rivals lists miss powers: ordered_country with min/max did not walk the whole field; take the fallback in docs/testing/space-race-playtest.md)"
		}
		remove_variable = sr_selfcheck_powers_listed
	}
}

# --- A launch complex that can no longer carry a milestone -------------------
# Every milestone journal entry fails when the space programme building stops
# granting its tier's country_sr_*_program_bool: the building was lost (the
# state it stood in changed hands, it was demolished) or its production method
# was lowered. One such change fails every milestone that tier was carrying, and
# the journal entries notice it on their own update schedule (every
# JOURNAL_ENTRY_UPDATE_ACTIVE = 4 days, not necessarily in step), so each
# on_fail used to fire its own identical space_race_events.76. Each on_fail now
# records its milestone, and only the first one queues the report — five days
# out, so every journal entry failing over the same change has recorded itself
# by the time it fires. Country scope. Called from the eight on_fail blocks.
sr_report_program_loss = {
	set_variable = { name = sr_program_lost_$MILESTONE$ value = yes }
	if = {
		limit = { NOT = { has_variable = sr_program_loss_report_queued } }
		# .76's immediate removes this; the timer only guards against a report
		# that never fires (the country gone within the five days).
		set_variable = { name = sr_program_loss_report_queued value = yes days = 30 }
		trigger_event = { id = space_race_events.76 days = 5 }
	}
}

# --- What a revolution's winner keeps ----------------------------------------
# A civil war's winner inherits the loser's variables (its own value wins where
# both sides hold one) but none of its modifiers, on the country or on a journal
# entry, and every entry it inherits active runs `immediate` again after the
# merge (docs/guides/scripting_best_practices.md § "What a Civil War's Winner
# Inherits"; docs/audits/civil_war_inheritance_audit.md F5). The winner
# continues the nation, so every space-race reward is recorded in a variable and
# rebuilt from it:
#
#   - sr_first_<m> / sr_<m> (country): from sr_completed_<m> and
#     sr_was_first_<m>, by sr_restore_milestone_rewards.
#   - sr_probe_*_data (country): from <modifier>_held, by sr_sync_probe_data.
#   - sr_safe_approach / sr_ambitious_approach (entry): from sr_safe_<m> /
#     sr_ambitious_<m>, by sr_sync_<m>_entry.
#   - sr_choice_* (entry): from the choice variable (sr_orbital_military, ...),
#     by sr_sync_<m>_entry.
#   - sr_colony_* (solar colonization entry): from <modifier>_held, by
#     sr_sync_solar_colonization_entry.
#   - sr_solar_system_trade (solar colonization entry): from
#     sr_completed_solar_colonization, by sr_sync_solar_colonization_entry.
#   - sr_commercial_space_dividend (country): from the variables
#     sr_space_race_concluded reads, by sr_refresh_space_dividend, which
#     re-applies it every month anyway (end of this file).
#   - sr_space_program_cost / sr_funding_progress / sr_funding_throughput
#     (entry): from sr_funding_<m>
#     and the approach variables, by sr_recalculate_cost, which every
#     `immediate` runs and the monthly pulse runs when an entry lacks one
#     (sr_cost_modifiers_missing). The `immediate` alone was not enough: an
#     inherited entry's `immediate` could not find its own entry by `je:`
#     (Spain's moon landing, 2026-10-10 observer run).
#
# Country modifiers come back in sr_repair_after_civil_war, which the shared
# civil-war hook runs for the winner (te_civil_war_on_won in
# common/on_actions/te_civil_war_on_actions.txt). Journal-entry modifiers do
# not: whether on_civil_war_won runs before or after an inherited entry's
# `immediate` is unknown. Each entry's own `immediate` rebuilds them, and the
# monthly country pulse (sr_sync_space_race_entries) is the backstop. Every
# step puts back only what a record says is missing (the one removal, of a
# stale approach modifier, is not about revolutions), so a loyalist winner,
# which still holds its modifiers, sees no change.
#
# Not rebuilt: timed modifiers (the completion events' themed rewards, the
# failure penalties, sr_temporary_safety_review), whose remaining time is
# recorded nowhere.

# Country scope. The milestone's reward, when the milestone is complete and
# neither reward modifier is on the country. sr_apply_milestone_reward_base adds
# the modifier and nothing else, so running it again replays no prestige, popup
# or notification. Only at the civil war's end (sr_repair_after_civil_war): a
# monthly check would also fire between on_complete, which sets
# sr_completed_<m>, and the completion event's option, which grants the reward.
sr_restore_milestone_reward_base = {
	if = {
		limit = {
			has_variable = sr_completed_$MILESTONE$
			NOT = { has_modifier = sr_first_$MILESTONE$ }
			NOT = { has_modifier = sr_$MILESTONE$ }
		}
		sr_apply_milestone_reward_base = { MILESTONE = $MILESTONE$ }
	}
}

sr_restore_milestone_rewards = {
	sr_restore_milestone_reward_base = { MILESTONE = suborbital         }
	sr_restore_milestone_reward_base = { MILESTONE = orbital            }
	sr_restore_milestone_reward_base = { MILESTONE = moon_landing       }
	sr_restore_milestone_reward_base = { MILESTONE = probe              }
	sr_restore_milestone_reward_base = { MILESTONE = moon_base          }
	sr_restore_milestone_reward_base = { MILESTONE = mars_landing       }
	sr_restore_milestone_reward_base = { MILESTONE = interstellar_probe }
}

# The space race's part of the civil-war repair (#464). ROOT = THIS = the
# winner, called from te_civil_war_on_won (te_civil_war_on_actions.txt) after
# the loser's variables have merged in. Puts back the country modifiers a
# revolution's winner does not inherit: the milestone rewards and the probe
# result. The journal-entry modifiers are rebuilt elsewhere (see "What a
# revolution's winner keeps" above).
#
# It reads no side information (te_cw_rebels_won, te_cw_role, the loser): it
# is stateless, adding only a reward whose record the country holds and whose
# modifier it lacks, and it never removes anything. So it is a no-op wherever
# nothing was lost:
# - a loyalist winner still holds its own modifiers and records;
# - a secession's winner, if this hook fires for one at all, is either the
#   original (as a loyalist) or the new country, which merged nothing and
#   holds only what it earned itself.
# One edge on any winner: a milestone completed while its completion event
# still waits for an answer already has its record, so the reward is added
# here and again by the option (assumed not to stack; untested).
sr_repair_after_civil_war = {
	sr_restore_milestone_rewards = yes
	sr_sync_probe_data = yes
	# The worlds the old government held are the winner's (sr_sites_generated.txt).
	sr_sites_repoint_loser = yes
	# Its fleets are the winner's (the sr_fl_* variables are inherited): it
	# joins the fleet powers now, not at its next monthly pulse; the loser's
	# entry is pruned (sr_fl_prune_powers).
	if = {
		limit = { sr_fleet_power = yes }
		sr_fl_join_powers = yes
	}
	# Interdiction's records that name the old government (6c).
	sr_icpt_repair_after_civil_war = yes
}

# The interstellar probe's result: one of four permanent country modifiers,
# granted by every option of space_race_events.601-630. Recorded as
# <modifier>_held in the same effect, so the record and the modifier cannot
# disagree. Country scope.
sr_grant_probe_data = {
	add_modifier = { name = $MODIFIER$ }
	set_variable = { name = $MODIFIER$_held value = yes }
}

# Country scope. One probe result, both ways: a modifier with no record was
# granted before the record existed (record it); a record with no modifier is a
# revolution's winner (put it back). Run at the civil war's end
# (sr_repair_after_civil_war) and monthly, so a save from before the record
# existed records itself within a month, a loser already fighting its civil war
# included: its record is a variable, which the winner inherits.
sr_sync_probe_data_base = {
	if = {
		limit = {
			has_modifier = $MODIFIER$
			NOT = { has_variable = $MODIFIER$_held }
		}
		set_variable = { name = $MODIFIER$_held value = yes }
	}
	else_if = {
		limit = {
			has_variable = $MODIFIER$_held
			NOT = { has_modifier = $MODIFIER$ }
		}
		add_modifier = { name = $MODIFIER$ }
	}
}

sr_sync_probe_data = {
	if = {
		limit = { has_variable = sr_interstellar_results_received }
		sr_sync_probe_data_base = { MODIFIER = sr_probe_dead_worlds_data   }
		sr_sync_probe_data_base = { MODIFIER = sr_probe_astro_wonders_data }
		sr_sync_probe_data_base = { MODIFIER = sr_probe_biological_data    }
		sr_sync_probe_data_base = { MODIFIER = sr_probe_intelligence_data  }
	}
}

# A colony's specialisation: one of 68 modifiers on the solar colonization
# entry, granted by an option of space_race_colony_events.1-34. Recorded as
# <modifier>_held in the same effect. Country scope.
sr_grant_colony_modifier = {
	je:je_space_race_solar_colonization = { add_modifier = { name = $MODIFIER$ } }
	set_variable = { name = $MODIFIER$_held value = yes }
	# The world's record (sr_sites_generated.txt): which specialization or
	# discovery outcome it carries, whoever holds it later. Bookkeeping behind
	# the modifier the option already shows, so hidden from its tooltip.
	hidden_effect = { sr_site_choose_$MODIFIER$ = yes }
}

# Country scope. One colony modifier, both ways, as sr_sync_probe_data_base.
# The entry is open whenever a colony is held (its `possible`), so both
# directions can always reach it. A discovery's outcome; a specialization
# takes sr_sync_colony_spec_base below.
sr_sync_colony_modifier_base = {
	if = {
		limit = {
			NOT = { has_variable = $MODIFIER$_held }
			je:je_space_race_solar_colonization ?= { has_modifier = $MODIFIER$ }
		}
		set_variable = { name = $MODIFIER$_held value = yes }
	}
	else_if = {
		limit = {
			has_variable = $MODIFIER$_held
			je:je_space_race_solar_colonization ?= { NOT = { has_modifier = $MODIFIER$ } }
		}
		je:je_space_race_solar_colonization = { add_modifier = { name = $MODIFIER$ } }
	}
}

# Country scope. One colony specialization, both ways, as the above. A record
# with no modifier (a revolution's winner, whose inherited entry arrives empty)
# gets it back at MULT, its level times the colony's staffing
# (sr_site_mult_<a|b>_<key>; colony modules design § 7), and the multiplier
# applied is recorded as the world's sync records it (sr_site_show_modifier_at).
sr_sync_colony_spec_base = {
	if = {
		limit = {
			NOT = { has_variable = $MODIFIER$_held }
			je:je_space_race_solar_colonization ?= { has_modifier = $MODIFIER$ }
		}
		set_variable = { name = $MODIFIER$_held value = yes }
	}
	else_if = {
		limit = {
			has_variable = $MODIFIER$_held
			je:je_space_race_solar_colonization ?= { NOT = { has_modifier = $MODIFIER$ } }
		}
		je:je_space_race_solar_colonization = { add_modifier = { name = $MODIFIER$ multiplier = $MULT$ } }
		set_variable = { name = $MODIFIER$_mult value = $MULT$ } # REVIEWED 2026-10-25: records the multiplier just applied, recomputed from the world's global record; nothing built up is lost
	}
}

# Country scope. The approach modifier on a running milestone's entry follows
# the approach variable, which sr_effect_safe / sr_effect_ambitious set together
# with it ($APPROACH$ = safe | ambitious). Added when the variable is set and
# the entry lacks it (an entry a revolution's winner inherited). Removed when
# the variable is gone and the entry still has it, which needs no revolution:
# the monthly cleanup clears a deactivated milestone's approach variable but
# cannot reach the inactive entry's modifiers, so the entry came back with its
# old approach modifier and no approach selected (the German-revolution saves'
# inactive orbital entry carried sr_safe_approach, with no sr_safe_orbital);
# and a finished solar colonization kept its approach modifier on the entry
# that stays open for the colonies.
sr_sync_approach_modifier_base = {
	if = {
		limit = {
			has_variable = sr_$APPROACH$_$MILESTONE$
			je:je_space_race_$MILESTONE$ ?= { NOT = { has_modifier = sr_$APPROACH$_approach } }
		}
		je:je_space_race_$MILESTONE$ = { add_modifier = sr_$APPROACH$_approach }
	}
	else_if = {
		limit = {
			NOT = { has_variable = sr_$APPROACH$_$MILESTONE$ }
			je:je_space_race_$MILESTONE$ ?= { has_modifier = sr_$APPROACH$_approach }
		}
		je:je_space_race_$MILESTONE$ = { remove_modifier = sr_$APPROACH$_approach }
	}
}

# Country scope. A running milestone's choice ($CHOICE$, the variable the choice
# event's option sets) puts $MODIFIER$ on its entry; put it back when the entry
# lacks it. Only while the milestone runs: these modifiers shape the run's risk
# and speed (see their header in space_race_modifiers.txt).
sr_sync_choice_modifier_base = {
	if = {
		limit = {
			has_variable = sr_active_$MILESTONE$
			has_variable = $CHOICE$
			je:je_space_race_$MILESTONE$ ?= { NOT = { has_modifier = $MODIFIER$ } }
		}
		je:je_space_race_$MILESTONE$ = { add_modifier = $MODIFIER$ }
	}
}

# ======= After the race: Commercial Space Industry =======
# Country scope; its one refresh site is the monthly country pulse
# (space_race_on_action). Once the race is over for the country
# (sr_space_race_concluded), the flat progress its sources still give is paid
# as Extraplanetary Base throughput: sr_commercial_space_dividend at
# multiplier = sr_space_dividend_points (+5% a point, capped; the reasoning is
# at the script value). Re-applied every month, so it follows its sources (a
# Space Elevator level built, a company losing prosperity), and it comes back
# on its own after a revolution: the winner inherits every variable the test
# reads, though no modifier. The first time it applies, a player gets
# space_race_events.77, which explains it; the variable that records the
# announcement is inherited too, so a revolution does not repeat it.
sr_refresh_space_dividend = {
	if = {
		limit = { has_modifier = sr_commercial_space_dividend }
		remove_modifier = sr_commercial_space_dividend
	}
	if = {
		limit = {
			sr_space_race_concluded = yes
			sr_space_dividend_points > 0
		}
		add_modifier = { name = sr_commercial_space_dividend multiplier = sr_space_dividend_points }
		if = {
			limit = { NOT = { has_variable = sr_space_dividend_announced } }
			set_variable = { name = sr_space_dividend_announced value = yes }
			if = {
				limit = { is_player = yes }
				trigger_event = { id = space_race_events.77 days = 1 }
			}
		}
	}
}
