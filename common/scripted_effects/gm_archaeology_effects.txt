# ============================================================================
# GRAND MONUMENTS — archaeology: scripted effects
# ============================================================================
# Design: docs/superpowers/specs/2026-10-14-grand-monument-archaeology-design.md (issue #939),
# reworked by 2026-10-16-grand-monument-archaeology-rework-design.md: costs by the
# site, sized digs, events that name the place, the find and its history, and
# foreign digs by treaty (an Excavation Concession, with partage, or a Permit).
#
# A buried site is a script container tagged gm_site, with no parent (it
# outlives every country), in the global list gm_arch_sites. It holds the
# monument it will become under the records a returned trophy carries
# (gm_tr_landmark, gm_tr_skin, gm_tr_form, gm_tr_levels, gm_tr_dedication,
# gm_tr_namesake, gm_tr_share, gm_tr_origin_region), so the trophy names and
# the monument machinery serve it unchanged, and the dig's own:
#   gm_site_tech, gm_site_vector     what opens it: a technology (a flag, the
#                                    hyphen an underscore) and `survey` (open as
#                                    soon as its owner has the technology),
#                                    `incidental` (turned up by a monthly chance)
#                                    or `excavating` (a dig already under way when
#                                    it is seeded, gm_arch_seed_dig: Pompeii)
#   gm_site_status                   1 buried, 2 known, 3 being dug, 4 unearthed
#   gm_site_sponsor, gm_site_method  who digs it, and how: `antiquarian`,
#                                    `systematic` or `forensic`
#   gm_site_progress                 0 to 100, kept when a dig is suspended
#   gm_site_terms                    where the sponsor digs: `own` ground, or
#                                    abroad by `concession` or `permit` (the
#                                    two treaty articles; rework spec § 3)
#   gm_site_told                     the players told of it (arch_events.1)
# No timed variable on a container: it would never expire
# (container_timed_variable_audit).
#
# A finished dig stands as a monument in the sponsor's part of the ground: the
# state's one building takes its levels (gm_state_tr_receive when it has none,
# else the container stays as a secondary monument beside the primary:
# gm_arch_stand_beside), the way a trophy comes home.
#
# The sites are seeded once, lazily, from the monthly pass (gm_arch_monthly),
# so a save from before this change has them too. One call per row of
# ARCH_SITES in test_grand_monument_archaeology.py, in LANDMARKS order.
#
# No order runs where a tooltip walks the effect: a scripted GUI or an event
# option only orders the work (gm_arch_order_launch, gm_arch_order_suspend) and
# a hidden event does it (arch_events.10, .11), as the trophies' do.

# ---- The monthly pass (from on_monthly_pulse) -------------------------------------------------
# No scope. Seed, turn up what a chance turns up, dig a month, finish what is
# done (after the walk: a container is taken off the list it is walked in
# only outside the walk), then the lists, the costs and the AI.
gm_arch_monthly = {
	if = {
		limit = { gm_system_enabled = yes }
		if = {
			limit = { NOT = { has_global_variable = gm_arch_seeded } }
			set_global_variable = gm_arch_seeded
			set_global_variable = { name = gm_arch_seed_round value = gm_arch_seed_rounds }
			gm_arch_seed_sites = yes
			gm_arch_self_check = yes
		}
		# A site added since a game was seeded is seeded the next month: the
		# seeding makes no site twice (gm_arch_seed_site), so running it again
		# adds only the new rows. Pompeii is round 2 (2026-10-16).
		if = {
			limit = { NOT = { has_global_variable = gm_arch_seed_round } }
			set_global_variable = { name = gm_arch_seed_round value = 1 }
		}
		if = {
			limit = { global_var:gm_arch_seed_round < gm_arch_seed_rounds }
			set_global_variable = { name = gm_arch_seed_round value = gm_arch_seed_rounds }
			gm_arch_seed_sites = yes
		}
		if = {
			limit = { has_global_variable_list = gm_arch_sites }
			every_in_global_list = {
				variable = gm_arch_sites
				gm_arch_site_monthly = yes
			}
			every_in_list = {
				list = gm_arch_done
				gm_arch_finish = yes
			}
		}
		gm_arch_pass = yes
		every_country = {
			limit = {
				has_variable = gm_arch_listed
				gm_arch_ai_may_dig = yes
			}
			gm_arch_ai_decide = yes
		}
		# A player the pass has listed is refreshed now: the cost and the drain
		# follow the digs this month's pass counted, a finished dig's included, and
		# a site waiting opens the journal entry.
		every_country = {
			limit = {
				has_variable = gm_arch_listed
				is_ai = no
			}
			trigger_event = { id = monument_events.20 }
		}
	}
}

# No scope, once a game after the seeding. The one engine fact nobody has seen
# hold: that a stored state region is still one a state can be found in. With the
# first site absent it logs nothing, so OK cannot be a false pass. The fallback is
# in docs/testing/grand-monuments-archaeology-playtest.md.
gm_arch_self_check = {
	clear_saved_scope = gm_arch_chk
	if = {
		limit = { has_global_variable_list = gm_arch_sites }
		random_in_global_list = {
			variable = gm_arch_sites
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:nineveh_library
			}
			save_scope_as = gm_arch_chk
		}
	}
	if = {
		limit = { exists = scope:gm_arch_chk }
		scope:gm_arch_chk = {
			if = {
				limit = {
					var:gm_tr_origin_region ?= {
						any_scope_state = { always = yes }
					}
				}
				debug_log = "TE_SELF_CHECK OK archaeology_site_region (a site's stored state region finds its states)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN archaeology_site_region (a site's stored state region finds no state, so no site will be listed, found or dug: take the fallback in docs/testing/grand-monuments-archaeology-playtest.md)"
			}
		}
	}
}

# Any scope. A site container with the records of the monument it will be.
# Never twice: the debug event runs the seeding again, so a site already on the
# list, standing as a monument or carried off as a trophy is not made again.
# The container has no parent, and is made in whatever scope this runs in.
gm_arch_seed_site = {
	if = {
		limit = {
			NOT = {
				AND = {
					has_global_variable_list = gm_arch_sites
					any_in_global_list = {
						variable = gm_arch_sites
						has_variable = gm_tr_landmark
						var:gm_tr_landmark = flag:$KEY$
					}
				}
			}
			# Not one already dug: it has left the list, and stands as a
			# monument (a state's own, or a container beside it) or is a trophy.
			NOT = {
				any_state = {
					has_variable = gm_landmark
					var:gm_landmark = flag:$KEY$
				}
			}
			NOT = {
				AND = {
					has_global_variable_list = gm_monuments
					any_in_global_list = {
						variable = gm_monuments
						has_variable = gm_tr_landmark
						var:gm_tr_landmark = flag:$KEY$
					}
				}
			}
			NOT = {
				AND = {
					has_global_variable_list = gm_trophies
					any_in_global_list = {
						variable = gm_trophies
						has_variable = gm_tr_landmark
						var:gm_tr_landmark = flag:$KEY$
					}
				}
			}
		}
		clear_saved_scope = gm_arch_new
		clear_saved_scope = gm_arch_seed_region
		# The region is saved first and stored from the saved scope, as every
		# stored region is elsewhere (gm_tr_origin_region).
		s:$STATE$ = { save_scope_as = gm_arch_seed_region }
		create_container = {
			tags = { gm_site }
			save_scope_as = gm_arch_new
			on_created = {
				save_scope_as = gm_arch_new
			}
		}
		if = {
			limit = {
				exists = scope:gm_arch_new
				exists = scope:gm_arch_seed_region
			}
			add_to_global_variable_list = { name = gm_arch_sites target = scope:gm_arch_new }
			scope:gm_arch_new = {
				set_variable = { name = gm_tr_landmark value = flag:$KEY$ }
				set_variable = { name = gm_tr_skin value = flag:landmark_$KEY$ }
				set_variable = { name = gm_tr_form value = flag:$FORM$ }
				set_variable = { name = gm_tr_levels value = $LEVEL$ }
				# When it was built, for the years its grandeur carries once it
				# is unearthed (gm_arch_place).
				set_variable = { name = gm_tr_built value = $BUILT$ }
				set_variable = { name = gm_tr_dedication value = flag:civic }
				set_variable = { name = gm_tr_namesake value = flag:landmark }
				# A whole landmark: its title is its name, with no part's.
				set_variable = { name = gm_tr_share value = flag:whole }
				set_variable = { name = gm_tr_origin_region value = scope:gm_arch_seed_region }
				set_variable = { name = gm_site_tech value = flag:$TECH$ }
				set_variable = { name = gm_site_vector value = flag:$VECTOR$ }
				set_variable = { name = gm_site_status value = 1 }
				# Being dug when the game begins (Pompeii): a dig already under way.
				if = {
					limit = { var:gm_site_vector = flag:excavating }
					gm_arch_seed_dig = yes
				}
			}
		}
	}
	# Round 3: a site seeded before it still buried, learning its year.
	else_if = {
		limit = { has_global_variable_list = gm_arch_sites }
		every_in_global_list = {
			variable = gm_arch_sites
			limit = {
				has_variable = gm_tr_landmark
				var:gm_tr_landmark = flag:$KEY$
				NOT = { has_variable = gm_tr_built }
			}
			set_variable = { name = gm_tr_built value = $BUILT$ }
		}
	}
}

# Container scope, a site just seeded with VECTOR = excavating: being dug when
# the game begins, as Pompeii was from 1748. The owner of its ground digs it by
# the Antiquarian method, part done (gm_arch_excavating_progress), and pays for
# it from the first month; it may suspend it or, with Empiricism, start it again
# by Systematic Stratigraphy, keeping the progress. With no owner, it waits as a
# known site with the progress made.
gm_arch_seed_dig = {
	clear_saved_scope = gm_arch_seed_sponsor
	var:gm_tr_origin_region ?= {
		random_scope_state = {
			limit = { exists = owner }
			owner = { save_scope_as = gm_arch_seed_sponsor }
		}
	}
	set_variable = { name = gm_site_progress value = gm_arch_excavating_progress }
	if = {
		limit = { exists = scope:gm_arch_seed_sponsor }
		set_variable = { name = gm_site_method value = flag:antiquarian }
		gm_arch_begin_dig = { WHO = scope:gm_arch_seed_sponsor }
	}
	else = {
		set_variable = { name = gm_site_status value = 2 }
	}
}

# ---- A site's month ---------------------------------------------------------------------------------
# Container scope, from gm_arch_monthly. A dig goes on a month; an incidental
# site that nobody has found may be.
gm_arch_site_monthly = {
	if = {
		limit = { gm_arch_site_digging = yes }
		gm_arch_dig_step = yes
	}
	else_if = {
		limit = {
			has_variable = gm_site_status
			var:gm_site_status = 1
			var:gm_site_vector = flag:incidental
		}
		gm_arch_try_find = yes
	}
}

# Container scope. An incidental site whose ground belongs to a country with the
# technology that gates it: a monthly chance (gm_arch_find_chance) that the
# plough, the railway cutting or a hiker turns it up. The finder is one of the
# ground's owners with the technology, and the state its part.
gm_arch_try_find = {
	save_scope_as = gm_arch_find_site
	clear_saved_scope = gm_arch_finder
	clear_saved_scope = gm_arch_find_state
	var:gm_tr_origin_region ?= {
		random_scope_state = {
			limit = {
				exists = owner
				owner = { save_temporary_scope_as = gm_arch_cand }
				scope:gm_arch_find_site = { gm_arch_has_gate_for = { WHO = scope:gm_arch_cand } }
			}
			save_scope_as = gm_arch_find_state
			owner = { save_scope_as = gm_arch_finder }
		}
	}
	if = {
		limit = {
			exists = scope:gm_arch_finder
			exists = scope:gm_arch_find_state
		}
		random = {
			chance = gm_arch_find_chance
			gm_arch_found = yes
		}
	}
}

# Container scope (scope:gm_arch_finder, scope:gm_arch_find_state). It is known
# now, and open to whoever may dig its ground with the technology. A player
# finder is told at once (arch_events.1); any other player who may dig it is
# told when the pass first lists it for them, and the AI finds it in its list.
gm_arch_found = {
	set_variable = { name = gm_site_status value = 2 }
	debug_log = "TE_MONUMENTS: site turned up (incidental)"
	gm_arch_tell_found = { WHO = scope:gm_arch_finder STATE = scope:gm_arch_find_state }
}

# Container scope. $WHO$, a player who may dig the site, is told of it once
# (gm_site_told): arch_events.1, with the site as scope:gm_arch_pick and the
# state it would dig ($STATE$) under the site's own name (gm_arch_ev_mark), so
# the popup names this site and this state whatever else is told this month.
gm_arch_tell_found = {
	if = {
		limit = {
			$WHO$ = { is_ai = no }
			OR = {
				NOT = { has_variable_list = gm_site_told }
				NOT = { is_target_in_variable_list = { name = gm_site_told target = $WHO$ } }
			}
		}
		add_to_variable_list = { name = gm_site_told target = $WHO$ }
		save_scope_as = gm_arch_pick
		clear_saved_scope = gm_arch_ev_survey
		gm_arch_ev_mark = { STATE = $STATE$ }
		if = {
			limit = { NOT = { var:gm_site_vector = flag:incidental } }
			$STATE$ = { save_scope_as = gm_arch_ev_survey }
		}
		$WHO$ = { trigger_event = { id = arch_events.1 } }
	}
}

# Container scope, a dig in progress. A month of progress by the method and the
# site's size (gm_arch_site_rate_<method>), from the state it is dug in
# (gm_arch_save_dig_state): a sponsor that may dig none of the ground any longer
# (it changed hands, the treaty that let it dig abroad ended, or the country
# is gone) suspends it, with its progress kept. A dig that reaches 100 is put
# on gm_arch_done, to be finished once the walk over the list is over.
gm_arch_dig_step = {
	clear_saved_scope = gm_arch_sponsor
	clear_saved_scope = gm_arch_dig_state
	var:gm_site_sponsor ?= { save_scope_as = gm_arch_sponsor }
	if = {
		limit = { exists = scope:gm_arch_sponsor }
		gm_arch_save_dig_state = yes
	}
	if = {
		limit = { exists = scope:gm_arch_dig_state }
		if = {
			limit = { var:gm_site_method = flag:antiquarian }
			change_variable = { name = gm_site_progress add = gm_arch_site_rate_antiquarian }
		}
		else_if = {
			limit = { var:gm_site_method = flag:systematic }
			change_variable = { name = gm_site_progress add = gm_arch_site_rate_systematic }
		}
		else = {
			change_variable = { name = gm_site_progress add = gm_arch_site_rate_forensic }
		}
		clamp_variable = { name = gm_site_progress max = 100 min = 0 }
		if = {
			limit = { var:gm_site_progress >= 100 }
			add_to_temporary_list = gm_arch_done
		}
	}
	else = {
		gm_arch_suspend = yes
	}
}

# Container scope, with scope:gm_arch_sponsor. The state the sponsor digs in:
# its own part of the ground if it holds one, else a part whose owner granted
# it the right to dig there, by an Excavation Concession before a Permit (the
# treaty read live, gm_arch_state_conceded_to). Writes the terms on the
# container (gm_site_terms); with none, scope:gm_arch_dig_state stays unset.
gm_arch_save_dig_state = {
	clear_saved_scope = gm_arch_dig_state
	var:gm_tr_origin_region ?= {
		random_scope_state = {
			limit = { owner = scope:gm_arch_sponsor }
			save_scope_as = gm_arch_dig_state
		}
	}
	if = {
		limit = { exists = scope:gm_arch_dig_state }
		set_variable = { name = gm_site_terms value = flag:own }
	}
	else = {
		var:gm_tr_origin_region ?= {
			random_scope_state = {
				limit = { gm_arch_state_conceded_to = { WHO = scope:gm_arch_sponsor ARTICLE = archaeological_concession } }
				save_scope_as = gm_arch_dig_state
			}
		}
		if = {
			limit = { exists = scope:gm_arch_dig_state }
			set_variable = { name = gm_site_terms value = flag:concession }
		}
		else = {
			var:gm_tr_origin_region ?= {
				random_scope_state = {
					limit = { gm_arch_state_conceded_to = { WHO = scope:gm_arch_sponsor ARTICLE = excavation_permit } }
					save_scope_as = gm_arch_dig_state
				}
			}
			if = {
				limit = { exists = scope:gm_arch_dig_state }
				set_variable = { name = gm_site_terms value = flag:permit }
			}
		}
	}
}

# Container scope. The dig stops; what was done is kept, and the site is open
# to whoever may dig its ground and has the technology.
gm_arch_suspend = {
	set_variable = { name = gm_site_status value = 2 }
	remove_variable = gm_site_sponsor
}

# ---- Finishing a dig -------------------------------------------------------------------------------
# Container scope, from gm_arch_monthly's walk of gm_arch_done. The method
# decides what the dig makes of it (gm_arch_settle_levels), the terms what the
# sponsor takes home from a dig abroad (gm_arch_settle_terms), then it stands
# in the state it was dug in, or the dig is suspended if the sponsor may dig
# there no longer.
gm_arch_finish = {
	clear_saved_scope = gm_arch_sponsor
	clear_saved_scope = gm_arch_dig_state
	var:gm_site_sponsor ?= { save_scope_as = gm_arch_sponsor }
	if = {
		limit = { exists = scope:gm_arch_sponsor }
		gm_arch_save_dig_state = yes
	}
	# Its state's building was rebuilt today (another dig finished there this
	# month, or a return): placing it rebuilds the building again, which must
	# wait (gm_tr_return_home's note). It stays at 100 and finishes next month,
	# before its method is settled, so a Forensic bonus is not added twice.
	if = {
		limit = {
			exists = scope:gm_arch_dig_state
			scope:gm_arch_dig_state = {
				NOT = { has_variable = gm_rebuilding }
			}
		}
		gm_arch_settle_levels = yes
		gm_arch_settle_terms = yes
		gm_arch_place = yes
	}
	else_if = {
		limit = {
			NOT = { exists = scope:gm_arch_dig_state }
		}
		gm_arch_suspend = yes
	}
}

# Container scope. What the method made of the dig. A Forensic dig adds a few
# levels; an Antiquarian one risks Stratigraphic Destruction, which halves what
# it finds (at least 3). gm_arch_ruined tells the sponsor which it was.
gm_arch_settle_levels = {
	if = {
		limit = { has_variable = gm_arch_ruined }
		remove_variable = gm_arch_ruined
	}
	if = {
		limit = { var:gm_site_method = flag:forensic }
		change_variable = { name = gm_tr_levels add = gm_arch_forensic_bonus }
	}
	else_if = {
		limit = { var:gm_site_method = flag:antiquarian }
		random = {
			chance = gm_arch_destruction_chance
			set_variable = gm_arch_ruined
			set_variable = {
				name = gm_tr_levels
				value = {
					value = var:gm_tr_levels
					divide = 2
					ceiling = yes
					min = 3
				}
			}
		}
	}
}

# Container scope (scope:gm_arch_sponsor, scope:gm_arch_dig_state). What a dig
# abroad leaves its sponsor (rework spec § 3). Under an Excavation Concession,
# the partage: half the levels go to the sponsor's capital as a trophy
# (gm_arch_partage), save at a rock-cut site, which has nothing to carry off.
# Under a Permit, or at a rock-cut site, the finds stay and the sponsor gains
# Foreign Scholarship. gm_arch_partaged and gm_arch_scholar tell the events.
gm_arch_settle_terms = {
	if = {
		limit = { has_variable = gm_arch_partaged }
		remove_variable = gm_arch_partaged
	}
	if = {
		limit = { has_variable = gm_arch_scholar }
		remove_variable = gm_arch_scholar
	}
	if = {
		limit = {
			has_variable = gm_site_terms
			var:gm_site_terms = flag:concession
			NOT = { var:gm_tr_form = flag:rock_cut }
			gm_arch_disp_levels >= 2
			scope:gm_arch_sponsor = { exists = capital }
		}
		gm_arch_partage = yes
	}
	else_if = {
		limit = {
			has_variable = gm_site_terms
			OR = {
				var:gm_site_terms = flag:concession
				var:gm_site_terms = flag:permit
			}
		}
		set_variable = gm_arch_scholar
		scope:gm_arch_sponsor = {
			add_modifier = { name = gm_arch_scholarship days = 3650 }
		}
	}
}

# Container scope (scope:gm_arch_sponsor, scope:gm_arch_dig_state). The
# partage: a trophy made from the dig's own records, as gm_state_tr_make_trophy
# makes one from a state's, standing in the sponsor's capital, its origin the
# host's state, a piece of half the levels (gm_arch_partage_levels), counting
# for half of them abroad, taken by the sponsor this year. The site keeps the
# rest, to stand in the host's state. Everything after is the trophies' own:
# the host may claim the piece back, and a returned piece rejoins the monument.
# No infamy and no anger: the host signed for it.
gm_arch_partage = {
	save_scope_as = gm_arch_partage_site
	gm_tr_sig_backfill = yes
	clear_saved_scope = gm_tr_new
	clear_saved_scope = gm_arch_partage_home
	set_variable = { name = gm_site_piece value = gm_arch_partage_levels }
	set_variable = { name = gm_site_piece_grandeur value = gm_arch_partage_grandeur }
	scope:gm_arch_sponsor = {
		capital = { save_scope_as = gm_arch_partage_home }
	}
	if = {
		limit = { exists = scope:gm_arch_partage_home }
		create_container = {
			tags = { gm_trophy }
			save_scope_as = gm_tr_new
			on_created = {
				save_scope_as = gm_tr_new
			}
		}
	}
	if = {
		limit = { exists = scope:gm_tr_new }
		add_to_global_variable_list = { name = gm_trophies target = scope:gm_tr_new }
		scope:gm_tr_new = {
			set_variable = { name = gm_tr_origin value = scope:gm_arch_dig_state }
			set_variable = { name = gm_tr_origin_region value = scope:gm_arch_partage_site.var:gm_tr_origin_region }
			set_variable = { name = gm_tr_site value = scope:gm_arch_partage_home }
			set_variable = { name = gm_tr_site_region value = scope:gm_arch_partage_home.state_region }
			set_variable = { name = gm_tr_levels value = scope:gm_arch_partage_site.var:gm_site_piece }
			# A find counts at the 1836 rate (sites spec 2026-10-18 §3.6), split
			# by the years since it was built (owner, 2026-10-19), as the rest
			# of the monument is in gm_arch_place: a taken piece carries its
			# share of the monument's years (ageing spec 2026-10-16 §5).
			if = {
				limit = { scope:gm_arch_partage_site = { has_variable = gm_tr_built } }
				set_variable = { name = gm_tr_built value = scope:gm_arch_partage_site.var:gm_tr_built }
			}
			gm_set_historical_grandeur = {
				LEVELS = scope:gm_arch_partage_site.var:gm_site_piece
				BUILT = gm_mon_built
				START = gm_tr_start_grandeur
				AGE = gm_tr_age
			}
			set_variable = { name = gm_tr_grandeur value = scope:gm_arch_partage_site.var:gm_site_piece_grandeur }
			set_variable = { name = gm_tr_year value = gm_this_year }
			set_variable = { name = gm_tr_taken_by value = scope:gm_arch_sponsor }
			set_variable = { name = gm_tr_share value = flag:piece }
			set_variable = { name = gm_tr_dedication value = scope:gm_arch_partage_site.var:gm_tr_dedication }
			set_variable = { name = gm_tr_skin value = scope:gm_arch_partage_site.var:gm_tr_skin }
			set_variable = { name = gm_tr_form value = scope:gm_arch_partage_site.var:gm_tr_form }
			set_variable = { name = gm_tr_landmark value = scope:gm_arch_partage_site.var:gm_tr_landmark }
			set_variable = { name = gm_tr_namesake value = scope:gm_arch_partage_site.var:gm_tr_namesake }
			# The piece carries the find's unique dedication (signatures spec §7).
			if = {
				limit = { scope:gm_arch_partage_site = { has_variable = gm_tr_signature } }
				set_variable = { name = gm_tr_signature value = scope:gm_arch_partage_site.var:gm_tr_signature }
				set_variable = { name = gm_tr_sig_parent value = scope:gm_arch_partage_site.var:gm_tr_sig_parent }
			}
			gm_tr_update_picture = yes
		}
		change_variable = { name = gm_tr_levels subtract = var:gm_site_piece }
		set_variable = gm_arch_partaged
		debug_log = "TE_MONUMENTS: partage, a piece shipped to the sponsor's capital"
	}
	remove_variable = gm_site_piece
	remove_variable = gm_site_piece_grandeur
}

# Container scope (scope:gm_arch_sponsor, scope:gm_arch_dig_state). The
# monument stands in the state it was dug in, the host's when the dig was
# abroad. With none there, the state's building is built from the records
# (gm_state_tr_receive, as a trophy's return is) and the container is done
# with. With one standing, the building takes the levels and the container
# stays as a monument beside it.
gm_arch_place = {
	save_scope_as = gm_tr_moving
	clear_saved_scope = gm_arch_host
	set_variable = { name = gm_site_status value = 4 }
	set_variable = { name = gm_tr_origin value = scope:gm_arch_dig_state }
	set_variable = { name = gm_tr_site value = scope:gm_arch_dig_state }
	set_variable = { name = gm_tr_site_region value = var:gm_tr_origin_region }
	set_variable = { name = gm_tr_year value = gm_this_year }
	# Its levels count at the 1836 rate (sites spec 2026-10-18 §3.6): what the
	# method and any partage left of them, once settled, split into starting
	# grandeur and the years since it was built (owner, 2026-10-19).
	gm_set_historical_grandeur = { LEVELS = var:gm_tr_levels BUILT = gm_mon_built START = gm_tr_start_grandeur AGE = gm_tr_age }
	# Its unique dedication (signatures spec §4.4), so the monument it becomes
	# has it from the day it stands.
	gm_tr_sig_backfill = yes
	gm_tr_update_picture = yes
	scope:gm_arch_sponsor = {
		set_variable = gm_active
	}
	scope:gm_arch_dig_state = {
		owner = { save_scope_as = gm_arch_host }
	}
	debug_log = "TE_MONUMENTS: site unearthed and placed"
	# The players are told before the container may go.
	gm_arch_tell_finished = yes
	if = {
		limit = { scope:gm_arch_dig_state = { has_building = building_grand_monument } }
		gm_arch_stand_beside = yes
	}
	else = {
		scope:gm_arch_dig_state = { gm_state_tr_receive = yes }
		gm_arch_retire = yes
	}
}

# Container scope (scope:gm_arch_sponsor, scope:gm_arch_host,

gm_arch_tell_finished = {

	else_if = {

			var:gm_site_terms = flag:permit
		}

	else_if = {
		limit = { has_variable = gm_arch_scholar }
		scope:gm_arch_dig_state = { save_scope_as = gm_arch_ev_scholar }
	}
	if = {
		limit = { scope:gm_arch_sponsor = { is_ai = no } }
		scope:gm_arch_sponsor = { trigger_event = { id = arch_events.2 } }
	}
	if = {

			exists = scope:gm_arch_host
			NOT = { scope:gm_arch_host = scope:gm_arch_sponsor }
			scope:gm_arch_host = { is_ai = no }
		}

	}

		$STATE$ = { save_scope_as = gm_arch_ev_sidon_necropolis }
	}
	else_if = {
		limit = { var:gm_tr_landmark = flag:villa_of_the_papyri }
		$STATE$ = { save_scope_as = gm_arch_ev_villa_of_the_papyri }
	}
	else_if = {
		limit = { var:gm_tr_landmark = flag:tomb_of_alexander }
		$STATE$ = { save_scope_as = gm_arch_ev_tomb_of_alexander }
	}
	else_if = {
		limit = { var:gm_tr_landmark = flag:pompeii }
		$STATE$ = { save_scope_as = gm_arch_ev_pompeii }
	}
}

# ---- Orders (the scripted GUI's and the events' side) ------------------------------------------------------
# Container scope. $WHO$ is to dig it by $METHOD$ (antiquarian, systematic or
# forensic): arch_events.10 checks again and does it.
gm_arch_order_launch = {
	set_variable = { name = gm_site_order_method value = flag:$METHOD$ }
	save_scope_as = gm_arch_pick
	$WHO$ = { trigger_event = { id = arch_events.10 } }
}

# Container scope. $WHO$ is to stop digging it (arch_events.11).
gm_arch_order_suspend = {
	save_scope_as = gm_arch_pick
	$WHO$ = { trigger_event = { id = arch_events.11 } }
}

# Container scope (scope:gm_arch_pick). The choice, with the tooltip that says
# what it costs, for an event's option.
gm_arch_choose_launch = {
	custom_tooltip = {
		text = gm_arch_launch_$METHOD$_tt
		gm_arch_order_launch = { WHO = $WHO$ METHOD = $METHOD$ }
	}
}

# Container scope, from arch_events.10 once it has checked the site again.
# $WHO$ starts the dig by the method it ordered, if it has that method's
# technology. Progress is kept from a dig suspended before.
gm_arch_start_dig = {
	if = {
		limit = {
			var:gm_site_order_method = flag:antiquarian
			$WHO$ = { gm_arch_method_antiquarian_ok = yes }
		}
		set_variable = { name = gm_site_method value = flag:antiquarian }
		gm_arch_begin_dig = { WHO = $WHO$ }
	}
	else_if = {
		limit = {
			var:gm_site_order_method = flag:systematic
			$WHO$ = { gm_arch_method_systematic_ok = yes }
		}
		set_variable = { name = gm_site_method value = flag:systematic }
		gm_arch_begin_dig = { WHO = $WHO$ }
	}
	else_if = {
		limit = {
			var:gm_site_order_method = flag:forensic
			$WHO$ = { gm_arch_method_forensic_ok = yes }
		}
		set_variable = { name = gm_site_method value = flag:forensic }
		gm_arch_begin_dig = { WHO = $WHO$ }
	}
	remove_variable = gm_site_order_method
}

gm_arch_begin_dig = {
	set_variable = { name = gm_site_status value = 3 }
	set_variable = { name = gm_site_sponsor value = $WHO$ }
	if = {
		limit = { NOT = { has_variable = gm_site_progress } }
		set_variable = { name = gm_site_progress value = 0 }
	}
	debug_log = "TE_MONUMENTS: dig begun"
}

# Container scope, from arch_events.11: the dig stops and its progress stays.
gm_arch_stop_dig = {
	gm_arch_suspend = yes
	if = {
		limit = { has_variable = gm_site_order_method }
		remove_variable = gm_site_order_method
	}
}

# ---- Lists, innovation, the national side --------------------------------------------------------------------
# No scope. Every country's lists of what it digs and what waits for it, and the
# innovation the digs divert, from the sites. Last month's are cleared first; the
# lists are rebuilt each time, after a click too (arch_events.10, .11).
gm_arch_pass = {
	every_country = {
		limit = { has_variable = gm_arch_listed }
		clear_variable_list = gm_arch_available
		clear_variable_list = gm_arch_digging
		set_variable = { name = gm_arch_dig_count value = 0 }
		set_variable = { name = gm_arch_open_count value = 0 }
		set_variable = { name = gm_arch_innovation_units value = 0 }
	}
	if = {
		limit = { has_global_variable_list = gm_arch_sites }
		every_in_global_list = {
			variable = gm_arch_sites
			gm_arch_list_one = yes
		}
	}
}

# Container scope. This site into the lists of the country that digs it, or of
# the countries that may: the owners of its ground with the technology, and
# whoever holds the right to dig a part of it by treaty (rework spec § 3).
gm_arch_list_one = {
	save_scope_as = gm_arch_listing
	if = {
		limit = { gm_arch_site_digging = yes }
		clear_saved_scope = gm_arch_sponsor
		var:gm_site_sponsor ?= { save_scope_as = gm_arch_sponsor }
		if = {
			limit = { exists = scope:gm_arch_sponsor }
			# The state the row names, and the terms it is dug on.
			gm_arch_save_dig_state = yes
			if = {
				limit = { exists = scope:gm_arch_dig_state }
				set_variable = { name = gm_site_view_state value = scope:gm_arch_dig_state }
			}
			scope:gm_arch_sponsor = {
				gm_arch_country_listed = yes
				add_to_variable_list = { name = gm_arch_digging target = scope:gm_arch_listing }
				change_variable = { name = gm_arch_dig_count add = 1 }
				if = {
					limit = { scope:gm_arch_listing = { var:gm_site_method = flag:antiquarian } }
					change_variable = { name = gm_arch_innovation_units add = gm_arch_innovation_antiquarian }
				}
				else_if = {
					limit = { scope:gm_arch_listing = { var:gm_site_method = flag:systematic } }
					change_variable = { name = gm_arch_innovation_units add = gm_arch_innovation_systematic }
				}
				else = {
					change_variable = { name = gm_arch_innovation_units add = gm_arch_innovation_forensic }
				}
			}
		}
	}
	else_if = {
		limit = { gm_arch_site_open = yes }
		var:gm_tr_origin_region ?= {
			# The owners of its ground, with the technology.
			every_scope_state = {
				limit = {
					exists = owner
					owner = { save_temporary_scope_as = gm_arch_cand }
					scope:gm_arch_listing = { gm_arch_has_gate_for = { WHO = scope:gm_arch_cand } }
				}
				save_scope_as = gm_arch_ls_state
				owner = { save_scope_as = gm_arch_ls_who }
				scope:gm_arch_listing = { gm_arch_list_for = yes }
			}
			# Whoever an owner granted the right to dig its part, by either article.
			every_scope_state = {
				limit = { exists = owner }
				save_scope_as = gm_arch_ls_state
				owner = {
					save_scope_as = gm_arch_ls_host
					every_scope_treaty = {
						every_scope_article = {
							limit = {
								OR = {
									has_type = archaeological_concession
									has_type = excavation_permit
								}
								source_country = scope:gm_arch_ls_host
								input_state = scope:gm_arch_ls_state
							}
							target_country = { save_scope_as = gm_arch_ls_who }
							if = {
								limit = {
									NOT = { scope:gm_arch_ls_who = scope:gm_arch_ls_host }
									scope:gm_arch_listing = { gm_arch_has_gate_for = { WHO = scope:gm_arch_ls_who } }
								}
								scope:gm_arch_listing = { gm_arch_list_for = yes }
							}
						}
					}
				}
			}
		}
	}
}

# Container scope (scope:gm_arch_listing, scope:gm_arch_ls_who the country,
# scope:gm_arch_ls_state the state it would dig). Into its list once: a country
# may hold a part of the ground and the right to dig another, or both articles.
# A player not yet told of the site is told (gm_arch_tell_found).
gm_arch_list_for = {
	if = {
		limit = {
			scope:gm_arch_ls_who = {
				OR = {
					NOT = { has_variable_list = gm_arch_available }
					NOT = { is_target_in_variable_list = { name = gm_arch_available target = scope:gm_arch_listing } }
				}
			}
		}
		scope:gm_arch_ls_who = {
			gm_arch_country_listed = yes
			add_to_variable_list = { name = gm_arch_available target = scope:gm_arch_listing }
			change_variable = { name = gm_arch_open_count add = 1 }
		}
		gm_arch_tell_found = { WHO = scope:gm_arch_ls_who STATE = scope:gm_arch_ls_state }
	}
}

# Country scope: listed this month, with its counts started at 0, and its
# national pulse running (gm_active keeps it going while it digs or waits).
gm_arch_country_listed = {
	if = {
		limit = { NOT = { has_variable = gm_arch_listed } }
		set_variable = gm_arch_listed
		set_variable = { name = gm_arch_dig_count value = 0 }
		set_variable = { name = gm_arch_open_count value = 0 }
		set_variable = { name = gm_arch_innovation_units value = 0 }
	}
	set_variable = gm_active
}

# Journal-entry scope (ROOT = the country), from gm_apply_national_modifiers.
# The digs' weekly innovation drain, the units gm_arch_pass summed: the
# modifier carries one unit. Digs cost no money (owner, 2026-10-16).
gm_arch_add_national = {
	# Only a country gm_arch_pass has listed has the variable, and
	# gm_add_national reads it unguarded (the boons' guard, gm_add_boon).
	if = {
		limit = { root = { has_variable = gm_arch_innovation_units } }
		gm_add_national = { NAME = gm_arch_diversion VAR = gm_arch_innovation_units }
	}
}

# Journal-entry scope, from gm_remove_national_modifiers. gm_arch_expense is
# retired: nothing adds it, and removing it here sheds it from a save made while
# digs cost money.
gm_arch_remove_national = {
	remove_modifier = gm_arch_diversion
	remove_modifier = gm_arch_expense
}

# ---- The AI ---------------------------------------------------------------------------------------------------------
# Country scope, monthly, an AI at peace and out of default (gm_arch_ai_may_dig)
# that is listed. With a site waiting, fewer than two digs going and the weekly
# innovation to spare for at least an Antiquarian dig (gm_arch_ai_affords_<method>:
# four times what the method diverts), one month in twelve it opens the grandest
# site waiting, by the best method it affords: Forensic, then Systematic, then
# Antiquarian (gm_arch_start_affordable). The diverted innovation is the brake,
# as for a player.
gm_arch_ai_decide = {
	if = {
		limit = {
			has_variable = gm_arch_open_count
			var:gm_arch_open_count > 0
			# any_in_list's count, not variable_list_size, whose key fails to load in 1.14;
			# a country that has never dug has no list
			OR = {
				NOT = { has_variable_list = gm_arch_digging }
				NOT = { any_in_list = { variable = gm_arch_digging count >= 2 } }
			}
			gm_arch_ai_affords_antiquarian = yes
		}
		random = {
			chance = 8
			clear_saved_scope = gm_arch_ai_pick
			save_scope_as = gm_arch_ai_country
			ordered_in_list = {
				variable = gm_arch_available
				order_by = gm_arch_disp_levels
				position = 0
				check_range_bounds = no
				save_scope_as = gm_arch_ai_pick
			}
			if = {
				limit = { exists = scope:gm_arch_ai_pick }
				scope:gm_arch_ai_pick = {
					gm_arch_start_affordable = { WHO = scope:gm_arch_ai_country }
				}
			}
		}
	}
}

# Container scope. $WHO$, an AI, digs the site by the best method whose
# innovation it can spare (gm_arch_ai_affords_<method>), Antiquarian at worst
# (gm_arch_ai_decide has checked that one).
gm_arch_start_affordable = {
	if = {
		limit = { $WHO$ = { gm_arch_ai_affords_forensic = yes } }
		set_variable = { name = gm_site_order_method value = flag:forensic }
	}
	else_if = {
		limit = { $WHO$ = { gm_arch_ai_affords_systematic = yes } }
		set_variable = { name = gm_site_order_method value = flag:systematic }
	}
	else = {
		set_variable = { name = gm_site_order_method value = flag:antiquarian }
	}
	if = {
		limit = { gm_arch_can_dig = { WHO = $WHO$ } }
		gm_arch_start_dig = { WHO = $WHO$ }
	}
	else = {
		remove_variable = gm_site_order_method
	}
}
