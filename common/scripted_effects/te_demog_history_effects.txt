# DEMOGRAPHICS — the yearly history store

# One sample container per calendar year for each tracked country
# (te_history_country_is_tracked), at most 100, kept in the country's variable
# list te_dg_hist oldest first. The Demographics panel's history charts draw
# it with GetList. It follows the Cultural Hegemony annual store
# (te_history_cultural_hegemony_effects.txt) and the country-owned monthly store
# (te_history_effects.txt), with the differences below.
#
# DATA MODEL
#   container  tag te_dg_hist_sample, parent the recording country (the engine
#              culls it when the country stops existing)
#   te_hist_i  the year. The chart template reads this name, and
#              te_history_sample_order negates it, so the shared eviction order
#              works on a year as it does on a month index
#   te_dg_h_median, te_dg_h_tfr, te_dg_h_e0, te_dg_h_gini, te_dg_h_wc
#              median age, children per woman, life expectancy at birth, Gini
#              and national Wealth Concentration as the census left them
#   te_dg_h_cbr, te_dg_h_cdr, te_dg_h_mig  (issue #1029) births, deaths and net
#              migration per 1,000 people, from a census with a step behind it
#              only (a seed counts none: te_dg_stepped)
#   te_dg_h_s<k>, te_dg_h_c<k>  (issue #1029) the year's deaths by cause: each
#              slice's share and the shares summed through it, k = 1-8 in the
#              pie's order (te_demog_cause_deaths_total's codes), from a census
#              with deaths by cause (te_dg_dc_total above 0)
#   te_dg_h_mk_war, te_dg_h_mk_famine, te_dg_h_mk_event  (issue #1029) the
#              charts' marker pips, 1 when set: the states' last counts took war
#              dead; a month of the year had a famine (te_dg_famine_year); a
#              demographic event fired (te_dg_wv_year, te_demog_history_note_event,
#              which marks a sample already stored too)
#   country    te_dg_hist (the list), te_dg_hist_year (the year last recorded),
#              te_dg_famine_year and te_dg_wv_year (the markers' years),
#              te_dg_h_e0_max (the Life Expectancy chart's axis, 100-10,000)
#
# ONE SAMPLE A YEAR. The census can run twice in the game's first year, at game
# start and at the country's yearly pulse (each country on its own day), so a
# container is created only when te_dg_hist_year is not the
# current year, and te_dg_hist_year is set once the container is stored. The
# figures are copied on every run: the yearly pulse's run overwrites the game-start
# sample. The Gini and Wealth Concentration come from te_demog_wc_national,
# which runs on the yearly pulse and at game start (te_inh_game_start, in
# inheritance_events.1); the two game-start events run in either order, so the
# game-start sample may lack te_dg_h_gini and te_dg_h_wc until the yearly pulse. A
# figure the country does not hold yet is left unset, never written as 0: a
# chart reads an unset value as "no data".
#
# A revolution's winner inherits te_dg_hist_year but not the list, so a
# country with no list creates a container whatever its year says: its chart
# starts again with the year it wins. The refill reads the list only once it
# exists.
#
# Scope: country, and only from te_demog_country_census (where THIS is the
# country). Everything is in hidden_effect: this is a store, not a tooltip.
# ============================================================================

te_demog_history_record = {
	hidden_effect = {
		save_scope_as = te_dg_hist_owner
		if = {
			limit = {
				OR = {
					NOT = { has_variable = te_dg_hist_year }
					NOT = { var:te_dg_hist_year = te_demog_year }
					NOT = { has_variable_list = te_dg_hist }
				}
			}
			create_container = {
				tags = { te_dg_hist_sample }
				parent = scope:te_dg_hist_owner
				save_scope_as = te_dg_hist_new
			}
			if = {
				limit = { exists = scope:te_dg_hist_new }
				scope:te_dg_hist_new = {
					set_variable = { name = te_hist_i value = te_demog_year }
				}
				add_to_variable_list = { name = te_dg_hist target = scope:te_dg_hist_new }
				te_demog_history_prune = yes
				# Commit the guard only after the sample has been stored.
				set_variable = { name = te_dg_hist_year value = te_demog_year }
			}
		}
		# This year's container, wherever the list keeps it: fill it, or refill it.
		if = {
			limit = { has_variable_list = te_dg_hist }
			every_in_list = {
				variable = te_dg_hist
				limit = { var:te_hist_i = te_demog_year }
				te_demog_history_copy = { FROM = te_dg_median TO = te_dg_h_median }
				te_demog_history_copy = { FROM = te_dg_tfr TO = te_dg_h_tfr }
				te_demog_history_copy = { FROM = te_dg_e0 TO = te_dg_h_e0 }
				te_demog_history_copy = { FROM = te_dg_gini TO = te_dg_h_gini }
				te_demog_history_copy = { FROM = te_inh_concentration TO = te_dg_h_wc }
				# issue #1029: the rates from a census a step stands behind (a seed's births and deaths are 0)
				if = {
					limit = {
						scope:te_dg_hist_owner = {
							has_variable = te_dg_stepped
							var:te_dg_stepped > 0
						}
					}
					te_demog_history_copy = { FROM = te_dg_cbr TO = te_dg_h_cbr }
					te_demog_history_copy = { FROM = te_dg_cdr TO = te_dg_h_cdr }
					te_demog_history_copy = { FROM = te_dg_mig_rate TO = te_dg_h_mig }
				}
				if = {
					limit = {
						scope:te_dg_hist_owner = {
							has_variable = te_dg_dc_total
							var:te_dg_dc_total > 0
						}
					}
					te_demog_history_causes = yes
				}
				te_demog_history_markers = yes
			}
			# the Life Expectancy chart's axis (2026-10-20)
			te_demog_history_e0_axis = yes
		}
	}
}

# Scope: a sample container, with scope:te_dg_hist_owner the country, which holds deaths by cause (issue #1029): each
# slice's share of the year's deaths and the shares summed through it, which the stacked chart's layers draw (largest
# first, as the pie's: gui_modding_guide.md, "Pie charts of script-held data").
te_demog_history_causes = {
	te_demog_history_share = { K = 1 SLICE = inf }
	te_demog_history_share = { K = 2 SLICE = chr }
	te_demog_history_share = { K = 3 SLICE = ext }
	te_demog_history_share = { K = 4 SLICE = work }
	te_demog_history_share = { K = 5 SLICE = mat }
	# slot 6 held the deaths added on top, which count under their causes since 2026-10-22: a save's older samples keep
	# theirs, so the chart's layers keep their colours
	set_variable = { name = te_dg_h_s6 value = 0 }
	te_demog_history_share = { K = 7 SLICE = kill }
	te_demog_history_share = { K = 8 SLICE = war }
	set_variable = { name = te_dg_h_c1 value = var:te_dg_h_s1 }
	set_variable = { name = te_dg_h_c2 value = { value = var:te_dg_h_c1 add = var:te_dg_h_s2 } }
	set_variable = { name = te_dg_h_c3 value = { value = var:te_dg_h_c2 add = var:te_dg_h_s3 } }
	set_variable = { name = te_dg_h_c4 value = { value = var:te_dg_h_c3 add = var:te_dg_h_s4 } }
	set_variable = { name = te_dg_h_c5 value = { value = var:te_dg_h_c4 add = var:te_dg_h_s5 } }
	set_variable = { name = te_dg_h_c6 value = { value = var:te_dg_h_c5 add = var:te_dg_h_s6 } }
	set_variable = { name = te_dg_h_c7 value = { value = var:te_dg_h_c6 add = var:te_dg_h_s7 } }
	set_variable = { name = te_dg_h_c8 value = 1 }
}

# Scope: a sample container. Slice $SLICE$'s share of the country's deaths, as te_dg_h_s$K$.
te_demog_history_share = {
	set_variable = {
		name = te_dg_h_s$K$
		value = {
			value = scope:te_dg_hist_owner.var:te_dg_dc_$SLICE$
			divide = { value = scope:te_dg_hist_owner.var:te_dg_dc_total min = 1 }
		}
	}
}

# Scope: a sample container, with scope:te_dg_hist_owner the country (issue #1029): the year's marker pips. War dead in
# the states' last counts; a famine in a month of the year (te_demog_history_note_famine); a demographic event
# (te_demog_history_note_event). A pip once set stays for the year: the yearly pulse's run only adds.
te_demog_history_markers = {
	if = {
		limit = {
			scope:te_dg_hist_owner = {
				has_variable = te_dg_dc_war
				var:te_dg_dc_war >= 1
			}
		}
		set_variable = { name = te_dg_h_mk_war value = 1 }
	}
	if = {
		limit = {
			scope:te_dg_hist_owner = {
				has_variable = te_dg_famine_year
				var:te_dg_famine_year = te_demog_year
			}
		}
		set_variable = { name = te_dg_h_mk_famine value = 1 }
	}
	if = {
		limit = {
			scope:te_dg_hist_owner = {
				has_variable = te_dg_wv_year
				var:te_dg_wv_year = te_demog_year
			}
		}
		set_variable = { name = te_dg_h_mk_event value = 1 }
	}
}

# Scope: a tracked country, on its monthly pulse (te_demog_country_monthly_on_action; issue #1029): a month with a
# famine marks the year's history sample, for the census's sample at the country's yearly pulse, and on this year's
# sample at once where it is stored already. The pulse falls on the country's own day, so a famine month after it in
# the calendar year would otherwise set a year whose sample is already stored (the event marker below does the same).
te_demog_history_note_famine = {
	if = {
		limit = {
			te_history_country_is_tracked = yes
			has_famine = yes
		}
		set_variable = { name = te_dg_famine_year value = te_demog_year }
		if = {
			limit = { has_variable_list = te_dg_hist }
			every_in_list = {
				variable = te_dg_hist
				limit = { var:te_hist_i = te_demog_year }
				set_variable = { name = te_dg_h_mk_famine value = 1 }
			}
		}
	}
}

# Scope: a country, as one of the event wave's events fires (te_demog_wave_effects.txt; issue #1029): the year's marker,
# for the census's sample at the yearly pulse, and on this year's sample at once where it is stored already (the yearly
# events fire after the census).
te_demog_history_note_event = {
	set_variable = { name = te_dg_wv_year value = te_demog_year }
	if = {
		limit = { has_variable_list = te_dg_hist }
		every_in_list = {
			variable = te_dg_hist
			limit = { var:te_hist_i = te_demog_year }
			set_variable = { name = te_dg_h_mk_event value = 1 }
		}
	}
}

# Scope: a sample container, with scope:te_dg_hist_owner the country. Copies one
# country variable into the sample when the country holds it.
te_demog_history_copy = {
	if = {
		limit = { scope:te_dg_hist_owner = { has_variable = $FROM$ } }
		set_variable = { name = $TO$ value = scope:te_dg_hist_owner.var:$FROM$ }
	}
}

# Scope: country. Evict the oldest sample once the list is over the cap (cap 100,
# so 101 trips it). `order_by`, not raw list position, so eviction is right
# whatever order the engine keeps the list in. `remove_list_variable` fills the
# hole with the list's LAST element, which breaks the oldest-first order, so
# every eviction sorts again.
te_demog_history_prune = {
	if = {
		limit = { any_in_list = { variable = te_dg_hist count >= 101 has_tag = te_dg_hist_sample } } # cap 100 + 1
		ordered_in_list = {
			variable = te_dg_hist
			position = 0
			order_by = te_history_sample_order
			save_scope_as = te_dg_hist_evicted
		}
		if = {
			limit = { exists = scope:te_dg_hist_evicted }
			remove_list_variable = { name = te_dg_hist target = scope:te_dg_hist_evicted }
			scope:te_dg_hist_evicted = { destroy_container = yes }
			te_demog_history_sort = yes
		}
	}
}

# Scope: country, called only from the pruner, with te_dg_hist at exactly the cap
# (which is why the completeness gate can be count >= 100). Build and verify the
# sorted copy before clearing the live list, so a short iterator cannot orphan
# every retained container. te_history_sample_order is negated because
# `ordered_*` sorts highest first: the oldest year comes out first.
te_demog_history_sort = {
	save_scope_as = te_dg_hist_sorter
	if = {
		limit = { has_variable_list = te_dg_hist_tmp }
		clear_variable_list = te_dg_hist_tmp
	}
	ordered_in_list = {
		variable = te_dg_hist
		order_by = te_history_sample_order
		min = 0
		max = 100
		check_range_bounds = no
		save_scope_as = te_dg_hist_moved
		scope:te_dg_hist_sorter = {
			add_to_variable_list = { name = te_dg_hist_tmp target = scope:te_dg_hist_moved }
		}
	}
	if = {
		limit = { any_in_list = { variable = te_dg_hist_tmp count >= 100 has_tag = te_dg_hist_sample } }
		clear_variable_list = te_dg_hist
		ordered_in_list = {
			variable = te_dg_hist_tmp
			order_by = te_history_sample_order
			min = 0
			max = 100
			check_range_bounds = no
			save_scope_as = te_dg_hist_moved
			scope:te_dg_hist_sorter = {
				add_to_variable_list = { name = te_dg_hist target = scope:te_dg_hist_moved }
			}
		}
	}
	if = {
		limit = { has_variable_list = te_dg_hist_tmp }
		clear_variable_list = te_dg_hist_tmp
	}
}

# Scope: the country, after its samples are stored (te_demog_history_record; the life table's open last interval,
# 2026-10-20): the Life Expectancy chart's axis, te_dg_h_e0_max, the largest figure its samples hold, at least
# te_demog_k_e0_axis_min (100) and at most te_demog_k_e0_shown_max (10,000), where a bar stops. Read over every sample, so
# it falls again once the long-lived years leave the chart.
te_demog_history_e0_axis = {
	set_local_variable = { name = te_dg_h_ax value = te_demog_k_e0_axis_min }
	every_in_list = {
		variable = te_dg_hist
		limit = { has_variable = te_dg_h_e0 }
		if = {
			limit = { local_var:te_dg_h_ax < var:te_dg_h_e0 }
			set_local_variable = { name = te_dg_h_ax value = var:te_dg_h_e0 }
		}
	}
	set_variable = { name = te_dg_h_e0_max value = { value = local_var:te_dg_h_ax max = te_demog_k_e0_shown_max } }
}
