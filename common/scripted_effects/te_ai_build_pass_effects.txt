# The AI construction pass (#1877; owner, 2026-11-16: "If low, no need [for a reserve], and you could even allow
# spending in (small amounts of) debt.")
#
# Country scope, an AI country, from the monthly country pulse (te_ai_build_pass_on_actions.txt). The engine queues
# nothing new for a government in debt: it holds no reserves, and the engine's reserve gate
# (NAI CONSTRUCTION_RESERVES_NEW_CONSTRUCTIONS) asks for some after a building's cost. Below the gate it queues only
# what it counts critical (a Government Administration while bureaucracy is short). So where the interest rate allows
# a government the means (te_ai_build_pass_means: its reserves above the reserve the rate asks for, plus the debt it
# may hold at its rate, less its debt), and it has the surplus the engine's own budget AI asks for before adding
# spending, the pass queues one level as government construction (start_building_construction, as
# te_grid_ai_build_effect and the Grand Monuments' gm_com_queue_levels do), after the queue has stood empty a month so
# that the engine has gone first. What it queues is the first entry of THE ORDER below that applies; each entry's
# conditions, site and reason are in te_ai_build_pass_triggers.txt. One level at a time: the pass follows the level it
# queued, in the state it queued it in, until it is built or gone (te_ai_build_pass_follow) and queues nothing more
# until then, and no level of a kind for a year after one of that kind was lost, or was built and gone again within
# two years (te_ai_build_pass_watch). The weekly purchase pays for the
# level (te_construction_market_ai_purchase_target). The numbers and their reasons:
# common/script_values/te_construction_market_ai_values.txt, THE RATE and THE PASS.
#
# Where a building has a `possible` or a can_build_government (the university's Polish closure, the power plant's
# banned-industry law and Resource Transition rule), the entry's trigger carries a copy, since whether
# start_building_construction honours them is unread.
#
# TE_SELF_CHECK ai_build_pass reads what became of the first 36 levels, administrations aside, the pass queued for a
# government the engine would not queue for (te_ai_build_pass_in_sample): built, still waiting after 18 to 36 months,
# or lost (its own spending AI removes "should-have" spending when income is under 1.05 times expenses), in all and by
# kind, once every one of them is read. Play-test rows: docs/testing/ai-economy-playtest.md.
te_ai_build_pass_effect = {
	# What became of last month's level, first, so that a level lost this month is backed off before the decision.
	te_ai_build_pass_follow = yes
	if = {
		limit = {
			te_ai_build_pass_may_queue = yes
		}
		# The country's manufacturing levels, once, for the highway's and the airport's major industrial states.
		if = {
			limit = {
				OR = {
					has_technology_researched = paved_roads
					has_technology_researched = commercial_aviation
				}
			}
			set_variable = { name = te_ai_build_pass_industry value = te_ai_build_pass_country_manufacturing }
		}
		# THE ORDER (owner, 2026-11-16). The one place the list is written: the first entry that applies is queued
		# and the rest are skipped (te_ai_build_pass_try). To reorder the list, reorder these lines.
		te_ai_build_pass_try = { KIND = administration }
		te_ai_build_pass_try = { KIND = power_plant }
		te_ai_build_pass_try = { KIND = railway }
		te_ai_build_pass_try = { KIND = university }
		te_ai_build_pass_try = { KIND = highway }
		te_ai_build_pass_try = { KIND = airport }
		if = {
			limit = {
				has_variable = te_ai_build_pass_industry
			}
			remove_variable = te_ai_build_pass_industry
		}
	}
	# Next month's "already empty a month".
	if = {
		limit = {
			construction_queue_num_queued_government_levels > 0
		}
		if = {
			limit = {
				has_variable = te_ai_build_pass_idle
			}
			remove_variable = te_ai_build_pass_idle
		}
	}
	else = {
		set_variable = te_ai_build_pass_idle
	}
}

# Country scope: one entry of THE ORDER. Skipped once an earlier entry has queued this month (it set the pending
# kind), else queued if the country wants it.
te_ai_build_pass_try = {
	if = {
		limit = {
			NOT = { has_variable = te_ai_build_pass_pending_kind }
			te_ai_build_pass_wants_$KIND$ = yes
		}
		te_ai_build_pass_queue_$KIND$ = yes
	}
}

# Country scope: queues one level of $BUILDING$ in the saved site and starts following it.
te_ai_build_pass_queue = {
	if = {
		limit = {
			exists = scope:te_ai_build_pass_site
		}
		set_variable = { name = te_ai_build_pass_pending_state value = scope:te_ai_build_pass_site }
		te_ai_build_pass_pending_start = { KIND = $KIND$ }
		scope:te_ai_build_pass_site = {
			start_building_construction = $BUILDING$
		}
		set_variable = { name = te_ai_build_pass_recent value = yes days = 45 }
		te_ai_build_pass_count_effect = yes
	}
}

# Country scope: the levels the pass has queued for the country since the game began, which the AI probe reads
# (cm_pass).
te_ai_build_pass_count_effect = {
	if = {
		limit = {
			NOT = { has_variable = te_ai_build_pass_count }
		}
		set_variable = { name = te_ai_build_pass_count value = 0 }
	}
	change_variable = { name = te_ai_build_pass_count add = 1 }
}

# Country scope, just before the pass queues: what it follows until the level is built or gone. The base is the
# levels of that kind built in the site (te_ai_build_pass_pending_state, set just before); the government queue is
# empty (the pass queues into none other). While it is followed the pass queues nothing (te_ai_build_pass_may_queue),
# and the reserve gate's self-check leaves the country out. A level for a country the engine would not queue for
# (te_ai_build_pass_in_sample), of any kind but an administration, joins the self-check's cohort while it has room
# (te_ai_build_pass_cohort_size): an administration is what the engine itself queues below its reserve gate while
# bureaucracy is short, so its fate says least about a level only the pass would queue, and at 100 points it would
# be decided first.
te_ai_build_pass_pending_start = {
	set_variable = { name = te_ai_build_pass_pending_kind value = $KIND$ }
	set_variable = { name = te_ai_build_pass_pending_base value = te_ai_build_pass_tracked_levels }
	set_variable = { name = te_ai_build_pass_pending_months value = 0 }
	if = {
		limit = {
			NOT = { has_global_variable = te_ai_build_pass_checked }
			NOT = { var:te_ai_build_pass_pending_kind = 2 }
			te_ai_build_pass_in_sample = yes
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_ai_build_pass_cohort }
			}
			set_global_variable = { name = te_ai_build_pass_cohort value = 0 }
			set_global_variable = { name = te_ai_build_pass_cohort_months value = 0 }
			set_global_variable = { name = te_ai_build_pass_built value = 0 }
			set_global_variable = { name = te_ai_build_pass_waiting value = 0 }
			set_global_variable = { name = te_ai_build_pass_lost value = 0 }
			te_ai_build_pass_tally_start = { KIND = university }
			te_ai_build_pass_tally_start = { KIND = railway }
			te_ai_build_pass_tally_start = { KIND = power_plant }
			te_ai_build_pass_tally_start = { KIND = highway }
			te_ai_build_pass_tally_start = { KIND = airport }
			te_ai_build_pass_down_start = { KIND = university }
			te_ai_build_pass_down_start = { KIND = administration }
			te_ai_build_pass_down_start = { KIND = railway }
			te_ai_build_pass_down_start = { KIND = power_plant }
			te_ai_build_pass_down_start = { KIND = highway }
			te_ai_build_pass_down_start = { KIND = airport }
		}
		if = {
			limit = {
				te_ai_build_pass_cohort_g < te_ai_build_pass_cohort_size
			}
			change_global_variable = { name = te_ai_build_pass_cohort add = 1 }
			set_variable = te_ai_build_pass_sampled
		}
	}
}

te_ai_build_pass_tally_start = {
	set_global_variable = { name = te_ai_build_pass_built_$KIND$ value = 0 }
	set_global_variable = { name = te_ai_build_pass_lost_$KIND$ value = 0 }
}

# The count of $KIND$ levels built and lost again (te_ai_build_pass_watch_one), from 0 where nothing has set it: the
# watch may count one before the cohort starts.
te_ai_build_pass_down_start = {
	if = {
		limit = {
			NOT = { has_global_variable = te_ai_build_pass_down_$KIND$ }
		}
		set_global_variable = { name = te_ai_build_pass_down_$KIND$ value = 0 }
	}
}

# Country scope, monthly: what became of the level the pass queued, after the watch on the levels it built
# (te_ai_build_pass_watch). Built: its state holds more levels of the kind than the base; it is then watched for two
# years. Lost: unbuilt, and either none of the kind is queued there or the government queue is empty (dropped from
# the queue, or the state lost). The pass queued into an empty government queue and adds nothing while it follows, so
# an empty one means its level is gone even where an investor's level of the kind is queued in the same state, which
# would otherwise hold the follow, and the pass, for as long as the investor's build takes. The pass then queues no
# level of that kind for the country for a year (THE PASS's BACKOFF_DAYS, written here as the literal a timed variable
# takes). A level of the kind taken down there lowers the base, so it reads as neither.
te_ai_build_pass_follow = {
	te_ai_build_pass_watch = yes
	if = {
		limit = {
			has_variable = te_ai_build_pass_pending_kind
		}
		change_variable = { name = te_ai_build_pass_pending_months add = 1 }
		if = {
			limit = {
				te_ai_build_pass_tracked_levels < var:te_ai_build_pass_pending_base
			}
			set_variable = { name = te_ai_build_pass_pending_base value = te_ai_build_pass_tracked_levels }
		}
		if = {
			limit = {
				te_ai_build_pass_tracked_levels > var:te_ai_build_pass_pending_base
			}
			if = {
				limit = {
					has_variable = te_ai_build_pass_sampled
					NOT = { has_global_variable = te_ai_build_pass_checked }
				}
				te_ai_build_pass_tally = { OUTCOME = built }
			}
			te_ai_build_pass_watch_start = yes
			te_ai_build_pass_pending_clear = yes
		}
		else_if = {
			limit = {
				OR = {
					te_ai_build_pass_tracked_queued <= 0
					NOT = { construction_queue_num_queued_government_levels > 0 }
				}
			}
			if = {
				limit = {
					has_variable = te_ai_build_pass_sampled
					NOT = { has_global_variable = te_ai_build_pass_checked }
				}
				te_ai_build_pass_tally = { OUTCOME = lost }
			}
			te_ai_build_pass_backoff = yes
			te_ai_build_pass_pending_clear = yes
		}
		else_if = {
			limit = {
				has_variable = te_ai_build_pass_sampled
				var:te_ai_build_pass_pending_months >= te_ai_build_pass_waiting_months
			}
			# Still queued after 18 to 36 months: counted as waiting, and out of the cohort's count of the undecided,
			# so its later fate is not counted too. The pass goes on following it.
			if = {
				limit = {
					NOT = { has_global_variable = te_ai_build_pass_checked }
				}
				change_global_variable = { name = te_ai_build_pass_waiting add = 1 }
			}
			remove_variable = te_ai_build_pass_sampled
		}
	}
}

# Country scope: $OUTCOME$ (built or lost) for the level followed, in all and for its kind (no administration:
# the cohort holds none, te_ai_build_pass_pending_start).
te_ai_build_pass_tally = {
	change_global_variable = { name = te_ai_build_pass_$OUTCOME$ add = 1 }
	if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 1
		}
		change_global_variable = { name = te_ai_build_pass_$OUTCOME$_university add = 1 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 3
		}
		change_global_variable = { name = te_ai_build_pass_$OUTCOME$_railway add = 1 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 4
		}
		change_global_variable = { name = te_ai_build_pass_$OUTCOME$_power_plant add = 1 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 5
		}
		change_global_variable = { name = te_ai_build_pass_$OUTCOME$_highway add = 1 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 6
		}
		change_global_variable = { name = te_ai_build_pass_$OUTCOME$_airport add = 1 }
	}
}

# Country scope, at "built": the level is watched for two years (730 days) in its state, by its kind, at the levels
# of the kind that state then holds.
te_ai_build_pass_watch_start = {
	if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 1
		}
		te_ai_build_pass_watch_set = { KIND = university }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 2
		}
		te_ai_build_pass_watch_set = { KIND = administration }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 3
		}
		te_ai_build_pass_watch_set = { KIND = railway }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 4
		}
		te_ai_build_pass_watch_set = { KIND = power_plant }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 5
		}
		te_ai_build_pass_watch_set = { KIND = highway }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 6
		}
		te_ai_build_pass_watch_set = { KIND = airport }
	}
}
te_ai_build_pass_watch_set = {
	set_variable = { name = te_ai_build_pass_watch_state_$KIND$ value = var:te_ai_build_pass_pending_state days = 730 }
	set_variable = { name = te_ai_build_pass_watch_level_$KIND$ value = te_ai_build_pass_tracked_levels days = 730 }
}

# Country scope, monthly: built, then gone again. Where a state the pass built a level in holds fewer levels of the
# kind within two years than once it was built (the engine's budget AI cuts what thin reserves bought, as Bengal's
# universities went from 7 to 1 in 1924-1939; or a private level of the kind there was downsized empty), the kind
# backs off for a year, as for a level lost while queued, so that the pass does not rebuild what the budget will not
# keep. The fall is counted for the self-check's line.
te_ai_build_pass_watch = {
	te_ai_build_pass_watch_one = { KIND = university BUILDING = building_university }
	te_ai_build_pass_watch_one = { KIND = administration BUILDING = building_government_administration }
	te_ai_build_pass_watch_one = { KIND = railway BUILDING = building_railway }
	te_ai_build_pass_watch_one = { KIND = power_plant BUILDING = building_power_plant }
	te_ai_build_pass_watch_one = { KIND = highway BUILDING = building_highway }
	te_ai_build_pass_watch_one = { KIND = airport BUILDING = building_airport }
}
te_ai_build_pass_watch_one = {
	if = {
		limit = {
			has_variable = te_ai_build_pass_watch_state_$KIND$
			has_variable = te_ai_build_pass_watch_level_$KIND$
			te_ai_build_pass_watch_fell = { KIND = $KIND$ BUILDING = $BUILDING$ }
		}
		set_variable = { name = te_ai_build_pass_backoff_$KIND$ value = yes days = 365 }
		te_ai_build_pass_down_start = { KIND = $KIND$ }
		change_global_variable = { name = te_ai_build_pass_down_$KIND$ add = 1 }
		remove_variable = te_ai_build_pass_watch_state_$KIND$
		remove_variable = te_ai_build_pass_watch_level_$KIND$
	}
}

# Country scope: no level of the lost level's kind for a year.
te_ai_build_pass_backoff = {
	if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 1
		}
		set_variable = { name = te_ai_build_pass_backoff_university value = yes days = 365 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 2
		}
		set_variable = { name = te_ai_build_pass_backoff_administration value = yes days = 365 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 3
		}
		set_variable = { name = te_ai_build_pass_backoff_railway value = yes days = 365 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 4
		}
		set_variable = { name = te_ai_build_pass_backoff_power_plant value = yes days = 365 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 5
		}
		set_variable = { name = te_ai_build_pass_backoff_highway value = yes days = 365 }
	}
	else_if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 6
		}
		set_variable = { name = te_ai_build_pass_backoff_airport value = yes days = 365 }
	}
}

te_ai_build_pass_pending_clear = {
	remove_variable = te_ai_build_pass_pending_kind
	remove_variable = te_ai_build_pass_pending_base
	remove_variable = te_ai_build_pass_pending_months
	if = {
		limit = {
			has_variable = te_ai_build_pass_pending_state
		}
		remove_variable = te_ai_build_pass_pending_state
	}
	if = {
		limit = {
			has_variable = te_ai_build_pass_sampled
		}
		remove_variable = te_ai_build_pass_sampled
	}
}
