# The AI construction pass (#1877; owner, 2026-11-16: "If low, no need [for a reserve], and you could even allow
# spending in (small amounts of) debt.")
#
# Country scope, an AI country, from the monthly country pulse (te_ai_build_pass_on_actions.txt). The engine queues
# nothing new for a government in debt: it holds no reserves, and the engine's reserve gate
# (NAI CONSTRUCTION_RESERVES_NEW_CONSTRUCTIONS) asks for some after a building's cost. Below the gate it queues only
# what it counts critical (a Government Administration while bureaucracy is short). So where the interest rate allows
# a government the means (te_ai_build_pass_means: its reserves above the reserve the rate asks for, plus the debt it
# may hold at its rate, less its debt), and it has the surplus the engine's own budget AI asks for before adding
# spending, the pass queues one level as government construction (start_building_construction, as
# te_grid_ai_build_effect and the Grand Monuments' gm_com_queue_levels do), after the queue has stood empty a month so
# that the engine has gone first. What does most good, in order (te_ai_build_pass_triggers.txt): a Government
# Administration while net bureaucracy is close to nothing or below it, in the incorporated state whose taxes come
# closest to its tax capacity; else a university while innovation is under nine-tenths of its cap, in the most
# populous incorporated state; each only where the surplus pays for the level within two years; else nothing. One level at a time: the pass follows the level it queued until it is built or gone
# (te_ai_build_pass_follow) and queues nothing more until then, and no level of a kind for a year after one of that
# kind was lost. The weekly purchase pays for the level (te_construction_market_ai_purchase_target).
# The numbers and their reasons: common/script_values/te_construction_market_ai_values.txt, THE RATE and THE PASS.
#
# Neither building has a `possible` block; the university's can_build_government (vanilla's Polish universities
# closure) is copied into te_ai_build_pass_wants_university, since whether start_building_construction honours it is
# unread.
#
# TE_SELF_CHECK ai_build_pass reads what became of the levels the pass queued for a government the engine would not
# queue for (te_ai_build_pass_in_sample): built, still waiting after 18 months, or lost (its own spending AI removes
# "should-have" spending when income is under 1.05 times expenses). Play-test rows: docs/testing/ai-economy-playtest.md.
te_ai_build_pass_effect = {
	# What became of last month's level, first, so that a level lost this month is backed off before the decision.
	te_ai_build_pass_follow = yes
	if = {
		limit = {
			te_ai_build_pass_may_queue = yes
		}
		if = {
			limit = {
				te_ai_build_pass_wants_administration = yes
			}
			te_ai_build_pass_pending_start = { KIND = 2 }
			ordered_scope_state = {
				limit = {
					te_ai_build_pass_administration_site = yes
				}
				order_by = te_ai_build_pass_tax_shortfall
				position = 0
				check_range_bounds = no
				start_building_construction = building_government_administration
			}
			set_variable = { name = te_ai_build_pass_recent value = yes days = 45 }
			te_ai_build_pass_count_effect = yes
		}
		else_if = {
			limit = {
				te_ai_build_pass_wants_university = yes
			}
			te_ai_build_pass_pending_start = { KIND = 1 }
			ordered_scope_state = {
				limit = {
					te_ai_build_pass_university_site = yes
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				start_building_construction = building_university
			}
			set_variable = { name = te_ai_build_pass_recent value = yes days = 45 }
			te_ai_build_pass_count_effect = yes
		}
	}
	# Next month's "already empty a month".
	if = {
		limit = {
			construction_queue_num_queued_government_levels > 0
		}
		if = {
			limit = {
				has_variable = te_ai_build_pass_idle
			}
			remove_variable = te_ai_build_pass_idle
		}
	}
	else = {
		set_variable = te_ai_build_pass_idle
	}
}

# Country scope: the levels the pass has queued for the country since the game began, which the AI probe reads
# (cm_pass).
te_ai_build_pass_count_effect = {
	if = {
		limit = {
			NOT = { has_variable = te_ai_build_pass_count }
		}
		set_variable = { name = te_ai_build_pass_count value = 0 }
	}
	change_variable = { name = te_ai_build_pass_count add = 1 }
}

# Country scope, just before the pass queues: what it follows until the level is built or gone. KIND 1 a university,
# 2 a Government Administration. The base is the country's built levels of that kind; the queue is empty (the pass
# queues into none other), so the level is the only one of its kind queued. While it is followed the pass queues
# nothing (te_ai_build_pass_may_queue), and the reserve gate's self-check leaves the country out. A country the engine
# would not queue for (te_ai_build_pass_in_sample) is also in the self-check's sample until it reads.
te_ai_build_pass_pending_start = {
	set_variable = { name = te_ai_build_pass_pending_kind value = $KIND$ }
	set_variable = { name = te_ai_build_pass_pending_base value = te_ai_build_pass_tracked_levels }
	set_variable = { name = te_ai_build_pass_pending_months value = 0 }
	if = {
		limit = {
			NOT = { has_global_variable = te_ai_build_pass_checked }
			te_ai_build_pass_in_sample = yes
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_ai_build_pass_built }
			}
			set_global_variable = { name = te_ai_build_pass_built value = 0 }
			set_global_variable = { name = te_ai_build_pass_waiting value = 0 }
			set_global_variable = { name = te_ai_build_pass_lost value = 0 }
		}
		set_variable = te_ai_build_pass_sampled
	}
}

# Country scope, monthly: what became of the level the pass queued. Built: the country holds more levels of the kind
# than the base. Lost: none of the kind is queued and none more is built (dropped from the queue, or its state lost);
# the pass then queues no level of that kind for the country for a year (THE PASS's BACKOFF_DAYS, written here as the
# literal a timed variable takes). A level of the kind taken down elsewhere lowers the base, so it reads as neither.
te_ai_build_pass_follow = {
	if = {
		limit = {
			has_variable = te_ai_build_pass_pending_kind
		}
		change_variable = { name = te_ai_build_pass_pending_months add = 1 }
		if = {
			limit = {
				te_ai_build_pass_tracked_levels < var:te_ai_build_pass_pending_base
			}
			set_variable = { name = te_ai_build_pass_pending_base value = te_ai_build_pass_tracked_levels }
		}
		if = {
			limit = {
				te_ai_build_pass_tracked_levels > var:te_ai_build_pass_pending_base
			}
			if = {
				limit = {
					has_variable = te_ai_build_pass_sampled
					NOT = { has_global_variable = te_ai_build_pass_checked }
				}
				change_global_variable = { name = te_ai_build_pass_built add = 1 }
			}
			te_ai_build_pass_pending_clear = yes
		}
		else_if = {
			limit = {
				te_ai_build_pass_tracked_queued <= 0
			}
			if = {
				limit = {
					has_variable = te_ai_build_pass_sampled
					NOT = { has_global_variable = te_ai_build_pass_checked }
				}
				change_global_variable = { name = te_ai_build_pass_lost add = 1 }
			}
			if = {
				limit = {
					var:te_ai_build_pass_pending_kind = 1
				}
				set_variable = { name = te_ai_build_pass_backoff_university value = yes days = 365 }
			}
			else = {
				set_variable = { name = te_ai_build_pass_backoff_administration value = yes days = 365 }
			}
			te_ai_build_pass_pending_clear = yes
		}
		else_if = {
			limit = {
				has_variable = te_ai_build_pass_sampled
				var:te_ai_build_pass_pending_months >= 18
			}
			# Still queued after 18 months: counted as waiting, and out of the sample, so its later fate is not
			# counted too. The pass goes on following it.
			if = {
				limit = {
					NOT = { has_global_variable = te_ai_build_pass_checked }
				}
				change_global_variable = { name = te_ai_build_pass_waiting add = 1 }
			}
			remove_variable = te_ai_build_pass_sampled
		}
	}
}

te_ai_build_pass_pending_clear = {
	remove_variable = te_ai_build_pass_pending_kind
	remove_variable = te_ai_build_pass_pending_base
	remove_variable = te_ai_build_pass_pending_months
	if = {
		limit = {
			has_variable = te_ai_build_pass_sampled
		}
		remove_variable = te_ai_build_pass_sampled
	}
}
