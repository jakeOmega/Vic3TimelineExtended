te_construction_market_init_target = {
	if = {
		limit = {
			not = {
				has_variable = te_construction_market_public_target
			}
		}
		set_variable = {
			name = te_construction_market_public_target
			value = 0
		}
	}
	# Seed price_stabilizer so the divide on var:te_construction_market_price_stabilizer
	# in te_construction_market_private_use_target doesn't return 'none' / Div/0
	# before the first stabilizer update tick.
	if = {
		limit = {
			NOT = { has_variable = te_construction_market_price_stabilizer }
		}
		set_variable = {
			name = te_construction_market_price_stabilizer
			value = TE_CONSTRUCTION_MARKET_SERVICES_PRICE
		}
	}
}

te_construction_market_country_setup = {
	te_construction_market_init_target = yes
	# country_replace_sectors = yes
	if = {
		limit = {
			te_free_market_construction_on = yes
			NOT = {
				any_scope_building = {
					is_building_type = te_construction_market_site
				}
			}
		}
		capital = {
			te_construction_market_place_site = yes
		}
	}
	# capital = {
	# 	create_building = {
	# 		building = te_construction_market_site
	# 		level = 1
	# 	}
	# }
	#te_construction_market_country_update_private_share = yes

	# Direct construction: the law-set private share from day one (the laws'
	# own allocations are cancelled in both modes). The weekly pulse keeps it
	# current, and adds the AI site guard and the construction minimum. Nothing
	# to place: the sites are the history's sectors, converted when the game
	# starts (te_direct_construction_convert_sectors).

		limit = {
			te_free_market_construction_off = yes
		}
		te_direct_construction_country_setup = yes
	}
}
