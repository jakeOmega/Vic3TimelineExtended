# ============================================================================
# DEMOGRAPHICS — the metropolitan share (phase 3; spec docs/superpowers/specs/2026-10-08-demographics-design.md §5.2)
# Values: te_demog_metro_values.txt (the target's eight terms, the drift, the label, the strengths, the panel's
# figures); modifiers: te_demog_metro_modifiers.txt; Python twin: scripts/analysis/demographics_metro.py.
#
# The share is computed and shown wherever the census runs (Full or Display only: te_demog_cohorts_run); its effects,
# the two modifiers and the census's urban fertility term (te_demog_mt_fertility_scale, te_demog_values.txt), act
# under Full only (te_demog_effects_run: the release precondition, nothing changes outside Full).
#
# Variables (state):
#   te_dg_mt       the metropolitan share, 0-100: the share of the state's urban people living in its largest city.
#                  Kept while the census runs and the state has te_demog_k_mt_min_urban urban people or more; seeded at
#                  its target the first time (a new game's first yearly count, a new state, a state that regained its
#                  towns), then moved te_demog_k_mt_drift of the way to the target at each yearly count. Removed when
#                  the state has too few urban people, or outside the census
#   te_dg_mt_mult  the multiplier of the state's metropolitan-share modifier (Full only; kept, since a modifier's
#                  multiplier is read again after it is applied)
# ============================================================================

# THIS = ROOT = a state, on its yearly pulse (te_demog_state_yearly_on_action), before te_demog_state_yearly: the share
# moves on last year's walk and its owner's last census, so the census step that follows reads this year's share in
# its urban fertility term and the subtab's tooltips match the step until the next count.
te_demog_mt_state_update = {
	if = {
		limit = {
			te_demog_cohorts_run = yes
			has_variable = te_dg_urban
			var:te_dg_urban >= te_demog_k_mt_min_urban
		}
		if = {
			limit = { has_variable = te_dg_mt }
			change_variable = { name = te_dg_mt add = te_demog_mt_step }
		}
		else = {
			set_variable = { name = te_dg_mt value = te_demog_mt_target }
		}
		te_debug_demog_mt_hub_selfcheck = yes
	}
	else_if = {
		limit = { has_variable = te_dg_mt }
		remove_variable = te_dg_mt
	}
}

# THIS = ROOT = a state, on its yearly pulse (te_demog_state_yearly_on_action, after the census), the one refresh site of
# the three modifiers: ROOT is the state, so the multiplier reads its own variable. Under Full, a share above Leading
# City's floor takes te_demog_mt_metropolis (its extended strength) and te_demog_mt_crowding (the 0-1 ramp: crowding
# feeds the share's target, a model term), and one below Market Towns' ceiling te_demog_mt_dispersed; below 0.01
# none. Outside Full all come off at the next pulse.
te_demog_mt_state_refresh = {
	if = {
		limit = { has_modifier = te_demog_mt_metropolis }
		remove_modifier = te_demog_mt_metropolis
	}
	if = {
		limit = { has_modifier = te_demog_mt_crowding }
		remove_modifier = te_demog_mt_crowding
	}
	if = {
		limit = { has_modifier = te_demog_mt_dispersed }
		remove_modifier = te_demog_mt_dispersed
	}
	if = {
		limit = {
			te_demog_effects_run = yes
			has_variable = te_dg_mt
		}
		if = {
			limit = { te_demog_mt_e_pos > 0.01 }
			set_variable = { name = te_dg_mt_mult value = te_demog_mt_e_pos_strength }
			add_modifier = {
				name = te_demog_mt_metropolis
				multiplier = var:te_dg_mt_mult
			}
			set_variable = { name = te_dg_mt_crowd_mult value = te_demog_mt_e_pos }
			add_modifier = {
				name = te_demog_mt_crowding
				multiplier = var:te_dg_mt_crowd_mult
			}
		}
		else_if = {
			limit = { te_demog_mt_e_neg > 0.01 }
			set_variable = { name = te_dg_mt_mult value = te_demog_mt_e_neg }
			add_modifier = {
				name = te_demog_mt_dispersed
				multiplier = var:te_dg_mt_mult
			}
		}
	}
}

# THIS = ROOT = a state, in te_demog_mt_state_update. The metro_world_market_hub self-check, once a game in development
# builds whatever the census's setting (docs/testing/demographics-playtest.md, Self-checks): the gateway term reads
# is_world_market_hub, the market area's hub, which the base game creates where a market area has at least one Port
# (concept_world_market_hub). So the world's hubs are counted against its Port states: OK when some states read yes,
# every hub has a Port and at least one Port state is not a hub (a market area holds several); BROKEN otherwise, with
# the counts: no hub (the trigger reads no in state scope, so no state takes the term), a hub without a Port (it reads
# something else) or every Port state a hub (it reads "has a Port"). Read at once: a trigger, not a modifier sum.
te_debug_demog_mt_hub_selfcheck = {
	if = {
		limit = {
			te_debug_demog_selfchecks_run = yes
			NOT = { has_global_variable = te_demog_mt_hub_selfcheck_done }
		}
		set_global_variable = { name = te_demog_mt_hub_selfcheck_done value = yes }
		set_local_variable = { name = te_dg_mt_chk_hubs value = 0 }
		set_local_variable = { name = te_dg_mt_chk_ports value = 0 }
		set_local_variable = { name = te_dg_mt_chk_hub_noport value = 0 }
		set_local_variable = { name = te_dg_mt_chk_port_nohub value = 0 }
		set_local_variable = { name = te_dg_mt_chk_countries value = 0 }
		every_state = {
			if = {
				limit = { b:building_port ?= { level > 0 } }
				change_local_variable = { name = te_dg_mt_chk_ports add = 1 }
				if = {
					limit = { is_world_market_hub = no }
					change_local_variable = { name = te_dg_mt_chk_port_nohub add = 1 }
				}
			}
			if = {
				limit = { is_world_market_hub = yes }
				change_local_variable = { name = te_dg_mt_chk_hubs add = 1 }
				if = {
					limit = {
						NOT = { b:building_port ?= { level > 0 } }
					}
					change_local_variable = { name = te_dg_mt_chk_hub_noport add = 1 }
				}
			}
		}
		every_country = {
			change_local_variable = { name = te_dg_mt_chk_countries add = 1 }
		}
		set_variable = { name = te_dg_mt_chk_hubs value = local_var:te_dg_mt_chk_hubs }
		set_variable = { name = te_dg_mt_chk_ports value = local_var:te_dg_mt_chk_ports }
		set_variable = { name = te_dg_mt_chk_hub_noport value = local_var:te_dg_mt_chk_hub_noport }
		set_variable = { name = te_dg_mt_chk_port_nohub value = local_var:te_dg_mt_chk_port_nohub }
		set_variable = { name = te_dg_mt_chk_countries value = local_var:te_dg_mt_chk_countries }
		if = {
			limit = {
				var:te_dg_mt_chk_hubs > 0
				var:te_dg_mt_chk_hub_noport = 0
				var:te_dg_mt_chk_port_nohub > 0
			}
			debug_log = "TE_SELF_CHECK OK metro_world_market_hub hubs=[THIS.Var('te_dg_mt_chk_hubs').GetValue|0] port_states=[THIS.Var('te_dg_mt_chk_ports').GetValue|0] hubs_without_port=[THIS.Var('te_dg_mt_chk_hub_noport').GetValue|0] ports_not_hub=[THIS.Var('te_dg_mt_chk_port_nohub').GetValue|0] countries=[THIS.Var('te_dg_mt_chk_countries').GetValue|0] first=[THIS.GetState.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN metro_world_market_hub hubs=[THIS.Var('te_dg_mt_chk_hubs').GetValue|0] port_states=[THIS.Var('te_dg_mt_chk_ports').GetValue|0] hubs_without_port=[THIS.Var('te_dg_mt_chk_hub_noport').GetValue|0] ports_not_hub=[THIS.Var('te_dg_mt_chk_port_nohub').GetValue|0] countries=[THIS.Var('te_dg_mt_chk_countries').GetValue|0] (expected hubs above 0, every hub with a Port and some Port states not hubs: is_world_market_hub doesn't read the market area's hub; take the fallback in docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
		remove_variable = te_dg_mt_chk_hubs
		remove_variable = te_dg_mt_chk_ports
		remove_variable = te_dg_mt_chk_hub_noport
		remove_variable = te_dg_mt_chk_port_nohub
		remove_variable = te_dg_mt_chk_countries
	}
}
