# ============================================================================
# DEMOGRAPHICS — the metropolitan share (phase 3; spec docs/superpowers/specs/2026-10-08-demographics-design.md §5.2)
# Values: te_demog_metro_values.txt (the target's eight terms, the drift, the label, the strengths, the panel's
# figures); modifiers: te_demog_metro_modifiers.txt; Python twin: scripts/analysis/demographics_metro.py.
#
# The share is computed and shown wherever the census runs (Full or Display only: te_demog_cohorts_run); its effects,
# the two modifiers and the census's urban fertility term (te_demog_mt_fertility_scale, te_demog_values.txt), act
# under Full only (te_demog_effects_run: the release precondition, nothing changes outside Full).
#
# Variables (state):
#   te_dg_mt       the metropolitan share, 0-100: the share of the state's urban people living in its largest city.
#                  Kept while the census runs and the state has te_demog_k_mt_min_urban urban people or more; seeded at
#                  its target the first time (a new game's first yearly count, a new state, a state that regained its
#                  towns), then moved te_demog_k_mt_drift of the way to the target at each yearly count. Removed when
#                  the state has too few urban people, or outside the census
#   te_dg_mt_mult  the multiplier of the state's metropolitan-share modifier (Full only; kept, since a modifier's
#                  multiplier is read again after it is applied)
# ============================================================================

# THIS = ROOT = a state, on its yearly pulse (te_demog_state_yearly_on_action), before te_demog_state_yearly: the share
# moves on last year's walk and its owner's last census, so the census step that follows reads this year's share in
# its urban fertility term and the subtab's tooltips match the step until the next count.
te_demog_mt_state_update = {
	if = {
		limit = {
			te_demog_cohorts_run = yes
			has_variable = te_dg_urban
			var:te_dg_urban >= te_demog_k_mt_min_urban
		}
		if = {
			limit = { has_variable = te_dg_mt }
			change_variable = { name = te_dg_mt add = te_demog_mt_step }
		}
		else = {
			set_variable = { name = te_dg_mt value = te_demog_mt_target }
		}
		te_debug_demog_mt_hub_selfcheck = yes
	}
	else_if = {
		limit = { has_variable = te_dg_mt }
		remove_variable = te_dg_mt
	}
}

# THIS = ROOT = a state, on its yearly pulse (te_demog_state_yearly_on_action, after the census), the one refresh site of
# both modifiers: ROOT is the state, so the multiplier reads its own variable. Under Full, a share above Leading City's
# floor takes te_demog_mt_metropolis and one below Market Towns' ceiling te_demog_mt_dispersed, each scaled by its
# strength; below 0.01 neither. Outside Full both come off at the next pulse.
te_demog_mt_state_refresh = {
	if = {
		limit = { has_modifier = te_demog_mt_metropolis }
		remove_modifier = te_demog_mt_metropolis
	}
	if = {
		limit = { has_modifier = te_demog_mt_dispersed }
		remove_modifier = te_demog_mt_dispersed
	}
	if = {
		limit = {
			te_demog_effects_run = yes
			has_variable = te_dg_mt
		}
		if = {
			limit = { te_demog_mt_e_pos > 0.01 }
			set_variable = { name = te_dg_mt_mult value = te_demog_mt_e_pos }
			add_modifier = {
				name = te_demog_mt_metropolis
				multiplier = var:te_dg_mt_mult
			}
		}
		else_if = {
			limit = { te_demog_mt_e_neg > 0.01 }
			set_variable = { name = te_dg_mt_mult value = te_demog_mt_e_neg }
			add_modifier = {
				name = te_demog_mt_dispersed
				multiplier = var:te_dg_mt_mult
			}
		}
	}
}

# THIS = ROOT = a state, in te_demog_mt_state_update. The metro_world_market_hub self-check, once a game in development
# builds whatever the census's setting (docs/testing/demographics-playtest.md, Self-checks): the gateway term reads
# is_world_market_hub, one hub a market area. The world's hubs are counted against its states: OK when some states read
# yes and at most a quarter do (a market area holds many states), BROKEN when none do (the trigger reads no in state
# scope, so no state takes the gateway term) or more than a quarter (it reads yes everywhere). Read at once: a trigger,
# not a modifier sum.
te_debug_demog_mt_hub_selfcheck = {
	if = {
		limit = {
			te_debug_demog_selfchecks_run = yes
			NOT = { has_global_variable = te_demog_mt_hub_selfcheck_done }
		}
		set_global_variable = { name = te_demog_mt_hub_selfcheck_done value = yes }
		set_local_variable = { name = te_dg_mt_chk_hubs value = 0 }
		set_local_variable = { name = te_dg_mt_chk_states value = 0 }
		every_state = {
			change_local_variable = { name = te_dg_mt_chk_states add = 1 }
			if = {
				limit = { is_world_market_hub = yes }
				change_local_variable = { name = te_dg_mt_chk_hubs add = 1 }
			}
		}
		set_variable = { name = te_dg_mt_chk_hubs value = local_var:te_dg_mt_chk_hubs }
		set_variable = { name = te_dg_mt_chk_states value = local_var:te_dg_mt_chk_states }
		if = {
			limit = {
				var:te_dg_mt_chk_hubs > 0
				var:te_dg_mt_chk_hubs <= te_debug_demog_mt_hub_most
			}
			debug_log = "TE_SELF_CHECK OK metro_world_market_hub hubs=[THIS.Var('te_dg_mt_chk_hubs').GetValue|0] states=[THIS.Var('te_dg_mt_chk_states').GetValue|0] first=[THIS.GetState.GetName] date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN metro_world_market_hub hubs=[THIS.Var('te_dg_mt_chk_hubs').GetValue|0] states=[THIS.Var('te_dg_mt_chk_states').GetValue|0] (some hubs and at most a quarter of the states expected: is_world_market_hub doesn't read a state's hub; take the fallback in docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		debug_log_scopes = yes
		remove_variable = te_dg_mt_chk_hubs
		remove_variable = te_dg_mt_chk_states
	}
}
