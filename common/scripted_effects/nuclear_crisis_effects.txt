# ============================================================================
# NUCLEAR CRISIS — opening, weekly tick, acts, settlements, outcomes
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0.3 is what shipped).
#
# DATA MODEL. A crisis is one pair of countries with one dispute. The record
# lives on the ISSUER (the country that warned; the single update owner, role
# 1); the TARGET (role 2) holds a mirror: nd_crisis_id, nd_crisis_opponent,
# nd_crisis_role. A country is in at most one crisis at a time.
#   issuer only: nd_crisis_dispute (1 play, 2 war, 3 guarantee, 4 proliferation,
#                5 alert), nd_crisis_beneficiary (dispute 3), nd_crisis_stage
#                (1 warning, 2 confrontation, 3 acute), nd_crisis_public,
#                nd_crisis_weeks, nd_crisis_deadline, nd_crisis_deadline_fired,
#                nd_crisis_danger, nd_crisis_counter, nd_crisis_talks,
#                nd_crisis_pressure_weeks, nd_crisis_backed (the target's
#                guarantor stands with it), nd_crisis_abandoned (it declined)
#   target only: nd_yield_pressure (written weekly by the tick)
#   after close: nd_crisis_last_outcome, nd_crisis_last_opponent,
#                nd_crisis_last_role (display); on the issuer
#                nd_crisis_cooldown_target / nd_crisis_cooldown_months
#
# SAVED SCOPES. Almost everything here expects scope:nd_issuer and
# scope:nd_target, saved by nd_crisis_save_parties from either party's scope,
# and hands them on to the events it fires (saved scopes survive trigger_event).
#
# OUTCOMES (nd_crisis_close = { OUTCOME = N }):
#   1  the target yielded                 6  the war ended
#   2  the issuer backed down             7  nuclear use
#   3  reciprocal stand-down              8  expired
#   4  the target backed down natively    9  invalid (a party vanished)
#   5  the issuer's side backed down      10 the dispute ended, no clear winner
#
# TOOLTIP PASSES. The diplomatic actions' accept_effect and the panel's
# ExecuteTooltip re-walk these effects every frame without running them, so
# every use of a scope saved inside this file is guarded by `exists`.
# ============================================================================

# ----------------------------------------------------------------------------
# PARTIES
# ----------------------------------------------------------------------------
# Either party's scope. Saves scope:nd_issuer and scope:nd_target.
nd_crisis_save_parties = {
	if = {
		limit = { nd_is_crisis_owner = yes }
		save_scope_as = nd_issuer
		var:nd_crisis_opponent ?= { save_scope_as = nd_target }
	}
	else_if = {
		limit = { nd_is_crisis_target = yes }
		save_scope_as = nd_target
		var:nd_crisis_opponent ?= { save_scope_as = nd_issuer }
	}
}

# Hands a decision event to $WHO$ (scope:nd_issuer or scope:nd_target) with
# the crisis's saved scopes. The token lets the event and its options confirm
# that the popup still belongs to the live crisis (nd_crisis_event_valid).
nd_crisis_send_event = {
	$WHO$ = {
		set_variable = { name = nd_crisis_event_token value = var:nd_crisis_id }
		trigger_event = { id = $EVENT$ popup = yes }
	}
}

# ----------------------------------------------------------------------------
# OPENING
# ----------------------------------------------------------------------------
# Issuer scope (ROOT = issuer: the diplomatic actions' accept_effect).
# TARGET = scope:target_country. PUBLIC = no (private warning) | yes
# (ultimatum) — a literal flag, so `always = $PUBLIC$` reads the same in a
# tooltip preview as in the run.
#
# TOOLTIP PASSES. The engine re-walks this effect every frame a diplomatic
# action's confirmation box (or an event option) is on screen, without running
# it, and it evaluates the limits and resolves the scopes inside hidden_effect
# too (scripting_best_practices.md § Per-Entity State rule 6). Every read of a
# crisis variable below would then hit a variable that does not exist yet. So
# the body is fenced by a scope saved on the first line: save_scope_as is not
# run by the preview, so scope:nd_crisis_opening only exists in the real run.
nd_crisis_open = {
	custom_tooltip = nd_tt_crisis_opens_$PUBLIC$
	hidden_effect = {
		save_scope_as = nd_crisis_opening
		if = {
			limit = {
				exists = scope:nd_crisis_opening
				exists = $TARGET$
				nd_crisis_parties_free = { TARGET = $TARGET$ }
			}
			save_scope_as = nd_issuer
			$TARGET$ = { save_scope_as = nd_target }
			if = {
				limit = { NOT = { has_global_variable = nd_crisis_counter } }
				set_global_variable = { name = nd_crisis_counter value = 0 }
			}
			change_global_variable = { name = nd_crisis_counter add = 1 }
			# ---- the issuer's record ----------------------------------------
			set_variable = { name = nd_crisis_id value = global_var:nd_crisis_counter }
			set_variable = { name = nd_crisis_opponent value = scope:nd_target }
			set_variable = { name = nd_crisis_role value = 1 }
			nd_crisis_classify_dispute = yes
			set_variable = { name = nd_crisis_weeks value = 0 }
			set_variable = { name = nd_crisis_pressure_weeks value = 0 }
			set_variable = { name = nd_crisis_counter value = 0 }
			set_variable = { name = nd_crisis_talks value = 0 }
			set_variable = { name = nd_crisis_backed value = 0 }
			set_variable = { name = nd_crisis_abandoned value = 0 }
			set_variable = { name = nd_crisis_deadline_fired value = 0 }
			if = {
				limit = { always = $PUBLIC$ }
				set_variable = { name = nd_crisis_public value = 1 }
				set_variable = { name = nd_crisis_stage value = 2 }
				set_variable = { name = nd_crisis_deadline value = nd_crisis_deadline_public_weeks }
			}
			else = {
				set_variable = { name = nd_crisis_public value = 0 }
				set_variable = { name = nd_crisis_stage value = 1 }
				set_variable = { name = nd_crisis_deadline value = nd_crisis_deadline_private_weeks }
			}
			set_variable = { name = nd_crisis_danger value = nd_crisis_danger_value }
			# ---- the target's mirror ----------------------------------------
			scope:nd_target = {
				set_variable = { name = nd_crisis_id value = scope:nd_issuer.var:nd_crisis_id }
				set_variable = { name = nd_crisis_opponent value = scope:nd_issuer }
				set_variable = { name = nd_crisis_role value = 2 }
				set_variable = { name = nd_yield_pressure value = nd_yield_pressure_value }
			}
			# ---- what the world sees ----------------------------------------
			if = {
				limit = { always = $PUBLIC$ }
				change_infamy = 5
				scope:nd_target = {
					change_relations = { country = scope:nd_issuer value = -20 }
				}
			}
			else = {
				scope:nd_target = {
					change_relations = { country = scope:nd_issuer value = -5 }
				}
			}
			# ---- both AIs reconsider their posture now ------------------------
			scope:nd_issuer = { nd_crisis_ai_review = yes }
			scope:nd_target = { nd_crisis_ai_review = yes }
			# ---- the target's guarantors are asked whether they stand by it ----
			nd_crisis_notify_guarantors = yes
			# ---- the target answers -----------------------------------------
			nd_crisis_send_event = { WHO = scope:nd_target EVENT = nuclear_crisis.1 }
		}
	}
}

# Issuer scope, scope:nd_target saved. Picks the dispute in priority order —
# war, play, guarantee, proliferation, alert — and for a guarantee dispute
# remembers the beneficiary.
nd_crisis_classify_dispute = {
	if = {
		limit = { nd_dispute_war_with = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 2 }
	}
	else_if = {
		limit = { nd_dispute_play_with = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 1 }
	}
	else_if = {
		limit = { nd_dispute_guarantee_against = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 3 }
		every_scope_treaty = {
			every_scope_article = {
				limit = {
					has_type = nuclear_guarantee
					source_country = scope:nd_issuer
					target_country = {
						OR = {
							has_war_with = scope:nd_target
							is_diplomatic_play_enemy_of = scope:nd_target
						}
					}
				}
				target_country = { save_scope_as = nd_crisis_ben }
			}
		}
		if = {
			limit = { exists = scope:nd_crisis_ben }
			set_variable = { name = nd_crisis_beneficiary value = scope:nd_crisis_ben }
		}
	}
	else_if = {
		limit = { nd_dispute_proliferation_by = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 4 }
	}
	else = {
		set_variable = { name = nd_crisis_dispute value = 5 }
	}
}

# Either party's scope: an AI party reviews its posture at once, in its own
# event so that its posture effects see it as ROOT
# (nuclear_deterrence_effects.txt header, "ROOT").
nd_crisis_ai_review = {
	if = {
		limit = {
			is_ai = yes
			nd_is_armed = yes
		}
		trigger_event = { id = nuclear_crisis.99 }
	}
}

# scope:nd_issuer and scope:nd_target saved. Every armed guarantor of the
# target other than the issuer is asked (nuclear_crisis.20), under the same
# scope names nd_record_nuclear_use uses for a strike.
nd_crisis_notify_guarantors = {
	scope:nd_issuer = { save_scope_as = nd_guarantee_attacker }
	scope:nd_target = { save_scope_as = nd_guarantee_beneficiary }
	scope:nd_target = {
		every_scope_treaty = {
			every_scope_article = {
				limit = {
					has_type = nuclear_guarantee
					target_country = scope:nd_target
					source_country = {
						NOT = { this = scope:nd_issuer }
						nd_believed_armed = yes
					}
				}
				source_country = {
					save_scope_as = nd_guarantor
					set_variable = { name = nd_guarantee_context value = 1 }
					trigger_event = { id = nuclear_crisis.20 }
				}
			}
		}
	}
}

# ----------------------------------------------------------------------------
# THE WEEKLY TICK — owner scope, from nd_weekly_update only
# ----------------------------------------------------------------------------
nd_crisis_weekly_tick = {
	nd_crisis_save_parties = yes
	if = {
		limit = {
			OR = {
				NOT = { exists = scope:nd_target }
				nd_crisis_pair_intact = no
			}
		}
		nd_crisis_clear_own_record = yes
	}
	else = {
		change_variable = { name = nd_crisis_weeks add = 1 }
		change_variable = { name = nd_crisis_pressure_weeks add = 1 }
		if = {
			limit = { has_variable = nd_crisis_exercise_weeks }
			change_variable = { name = nd_crisis_exercise_weeks subtract = 1 }
			if = {
				limit = { var:nd_crisis_exercise_weeks <= 0 }
				remove_variable = nd_crisis_exercise_weeks
			}
		}
		nd_crisis_revalidate_dispute = yes
	}
	# Still open after revalidation?
	if = {
		limit = { nd_is_crisis_owner = yes }
		set_variable = { name = nd_crisis_danger value = nd_crisis_danger_value }
		scope:nd_target = {
			set_variable = { name = nd_yield_pressure value = nd_yield_pressure_value }
		}
		# ---- confrontation becomes acute ------------------------------------
		if = {
			limit = {
				var:nd_crisis_stage = 2
				OR = {
					var:nd_crisis_danger >= 70
					AND = {
						nd_readiness_high_alert = yes
						scope:nd_target = { nd_readiness_heightened_or_higher = yes }
					}
				}
			}
			set_variable = { name = nd_crisis_stage value = 3 }
		}
		# ---- expiry -----------------------------------------------------------
		if = {
			limit = { var:nd_crisis_weeks >= nd_crisis_max_weeks }
			nd_crisis_close = { OUTCOME = 8 }
		}
		# ---- the deadline -----------------------------------------------------
		else_if = {
			limit = {
				var:nd_crisis_deadline_fired = 0
				var:nd_crisis_weeks >= var:nd_crisis_deadline
			}
			set_variable = { name = nd_crisis_deadline_fired value = 1 }
			nd_crisis_send_event = { WHO = scope:nd_issuer EVENT = nuclear_crisis.4 }
		}
		# ---- pressure on the target -------------------------------------------
		else_if = {
			limit = {
				var:nd_crisis_stage >= 2
				var:nd_crisis_talks = 0
				var:nd_crisis_pressure_weeks >= nd_crisis_pressure_interval_weeks
			}
			set_variable = { name = nd_crisis_pressure_weeks value = 0 }
			nd_crisis_send_event = { WHO = scope:nd_target EVENT = nuclear_crisis.5 }
		}
	}
}

# Owner scope, parties saved. Converts or closes a crisis whose dispute has
# moved on. The on_diplo_play_back_down hook closes a play crisis with a
# winner the moment it happens; this is the net for everything else.
nd_crisis_revalidate_dispute = {
	# ---- a play that became a war -----------------------------------------
	if = {
		limit = {
			var:nd_crisis_dispute = 1
			has_war_with = scope:nd_target
		}
		set_variable = { name = nd_crisis_dispute value = 2 }
		set_variable = { name = nd_crisis_stage value = 3 }
		nd_crisis_record_defiance = yes
		if = {
			limit = { nd_crisis_is_public = yes }
			nd_crisis_send_event = { WHO = scope:nd_issuer EVENT = nuclear_crisis.7 }
		}
	}
	# ---- a play that ended without a war and without the hook seeing it ------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 1
			NOT = { is_diplomatic_play_enemy_of = scope:nd_target }
		}
		nd_crisis_close = { OUTCOME = 10 }
	}
	# ---- a war that ended -------------------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 2
			NOT = { has_war_with = scope:nd_target }
		}
		nd_crisis_close = { OUTCOME = 6 }
	}
	# ---- a guarantee dispute ----------------------------------------------
	else_if = {
		limit = { var:nd_crisis_dispute = 3 }
		if = {
			limit = { has_war_with = scope:nd_target }
			set_variable = { name = nd_crisis_dispute value = 2 }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_variable = nd_crisis_beneficiary }
					NOT = {
						var:nd_crisis_beneficiary ?= {
							OR = {
								has_war_with = scope:nd_target
								is_diplomatic_play_enemy_of = scope:nd_target
							}
						}
					}
				}
			}
			nd_crisis_close = { OUTCOME = 10 }
		}
	}
	# ---- proliferation that stopped ---------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 4
			scope:nd_target = { nuclear_program_is_proliferating = no }
		}
		nd_crisis_close = { OUTCOME = 4 }
	}
	# ---- an alert that was stood down -------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 5
			scope:nd_target = {
				OR = {
					nd_is_armed = no
					NOT = { has_variable = nd_readiness_target }
					var:nd_readiness_target = 1
				}
			}
		}
		nd_crisis_close = { OUTCOME = 4 }
	}
	# ---- a disarmed issuer has no threat left -----------------------------
	if = {
		limit = {
			nd_is_crisis_owner = yes
			nd_is_armed = no
		}
		nd_crisis_close = { OUTCOME = 10 }
	}
}

# A target whose issuer vanished, or an owner whose target did: drop this
# side's record without outcomes (outcome 9). Either party's scope.
nd_crisis_clear_own_record = {
	set_variable = { name = nd_crisis_last_outcome value = 9 }
	nd_crisis_clear_vars = yes
}

nd_crisis_clear_vars = {
	remove_variable = nd_crisis_id
	remove_variable = nd_crisis_opponent
	remove_variable = nd_crisis_role
	if = {
		limit = { has_variable = nd_crisis_dispute }
		remove_variable = nd_crisis_dispute
	}
	if = {
		limit = { has_variable = nd_crisis_beneficiary }
		remove_variable = nd_crisis_beneficiary
	}
	if = {
		limit = { has_variable = nd_crisis_stage }
		remove_variable = nd_crisis_stage
	}
	if = {
		limit = { has_variable = nd_crisis_public }
		remove_variable = nd_crisis_public
	}
	if = {
		limit = { has_variable = nd_crisis_weeks }
		remove_variable = nd_crisis_weeks
	}
	if = {
		limit = { has_variable = nd_crisis_deadline }
		remove_variable = nd_crisis_deadline
	}
	if = {
		limit = { has_variable = nd_crisis_deadline_fired }
		remove_variable = nd_crisis_deadline_fired
	}
	if = {
		limit = { has_variable = nd_crisis_danger }
		remove_variable = nd_crisis_danger
	}
	if = {
		limit = { has_variable = nd_crisis_counter }
		remove_variable = nd_crisis_counter
	}
	if = {
		limit = { has_variable = nd_crisis_talks }
		remove_variable = nd_crisis_talks
	}
	if = {
		limit = { has_variable = nd_crisis_pressure_weeks }
		remove_variable = nd_crisis_pressure_weeks
	}
	if = {
		limit = { has_variable = nd_crisis_backed }
		remove_variable = nd_crisis_backed
	}
	if = {
		limit = { has_variable = nd_crisis_abandoned }
		remove_variable = nd_crisis_abandoned
	}
	if = {
		limit = { has_variable = nd_yield_pressure }
		remove_variable = nd_yield_pressure
	}
	if = {
		limit = { has_variable = nd_crisis_exercise_weeks }
		remove_variable = nd_crisis_exercise_weeks
	}
}

# Target scope, from the target's monthly update: an issuer that vanished
# (annexed, or its record gone) leaves nobody to run the tick.
nd_crisis_target_watchdog = {
	if = {
		limit = {
			nd_is_crisis_target = yes
			NOT = {
				var:nd_crisis_opponent ?= {
					nd_crisis_pair_intact = yes
				}
			}
		}
		nd_crisis_clear_own_record = yes
	}
}

# ----------------------------------------------------------------------------
# CLOSING — scope:nd_issuer and scope:nd_target saved, any scope
# ----------------------------------------------------------------------------
# Applies the outcome once and cleans both records. OUTCOME is a literal.
# Fenced like nd_crisis_open: an option tooltip walks this without running it,
# and nd_crisis_apply_outcome reads the nd_crisis_outcome_now written a line
# earlier, which a preview would find missing.
nd_crisis_close = {
	save_scope_as = nd_crisis_closing
	if = {
		limit = {
			exists = scope:nd_crisis_closing
			exists = scope:nd_issuer
			exists = scope:nd_target
			scope:nd_issuer = { nd_is_crisis_owner = yes }
		}
		scope:nd_issuer = {
			set_variable = { name = nd_crisis_outcome_now value = $OUTCOME$ }
		}
		nd_crisis_apply_outcome = yes
		scope:nd_issuer = {
			set_variable = { name = nd_crisis_last_outcome value = $OUTCOME$ }
			set_variable = { name = nd_crisis_last_opponent value = scope:nd_target }
			set_variable = { name = nd_crisis_last_role value = 1 }
			set_variable = { name = nd_crisis_cooldown_target value = scope:nd_target }
			set_variable = { name = nd_crisis_cooldown_months value = nd_crisis_cooldown_months_value }
			nd_crisis_clear_vars = yes
		}
		scope:nd_target = {
			if = {
				limit = {
					nd_is_crisis_target = yes
					var:nd_crisis_opponent ?= scope:nd_issuer
				}
				nd_crisis_clear_vars = yes
			}
			set_variable = { name = nd_crisis_last_outcome value = $OUTCOME$ }
			set_variable = { name = nd_crisis_last_opponent value = scope:nd_issuer }
			set_variable = { name = nd_crisis_last_role value = 2 }
		}
		if = {
			limit = { scope:nd_issuer = { NOT = { var:nd_crisis_last_outcome = 9 } } }
			scope:nd_issuer = { trigger_event = { id = nuclear_crisis.6 } }
			scope:nd_target = { trigger_event = { id = nuclear_crisis.6 } }
		}
	}
}

# From either party's scope.
nd_crisis_close_from_either = {
	nd_crisis_save_parties = yes
	nd_crisis_close = { OUTCOME = $OUTCOME$ }
}

# The consequences of each outcome (§6). Reads
# scope:nd_issuer.var:nd_crisis_outcome_now; the crisis variables are still set.
nd_crisis_apply_outcome = {
	# ---- the issuer's objective was achieved (1, 4) --------------------------
	if = {
		limit = {
			scope:nd_issuer = {
				OR = {
					var:nd_crisis_outcome_now = 1
					var:nd_crisis_outcome_now = 4
				}
			}
		}
		scope:nd_issuer = {
			nd_change_credibility = { AMOUNT = 10 }
			if = {
				limit = { nd_crisis_is_public = yes }
				nd_change_credibility = { AMOUNT = 5 }
			}
			add_modifier = {
				name = nd_crisis_coercive_success
				days = normal_modifier_time
				is_decaying = yes
			}
			nd_reward_hawks = { OTHER = scope:nd_target LIST = nd_lob_iw GOOD = yes }
			nd_lobby_react = { OTHER = scope:nd_target TYPE = anti AMOUNT = 3 EVENT = negative }
			nd_lobby_react = { OTHER = scope:nd_target TYPE = pro AMOUNT = -3 EVENT = negative }
		}
		scope:nd_target = {
			nd_change_credibility = { AMOUNT = -5 }
			add_modifier = {
				name = nd_crisis_yielded
				days = normal_modifier_time
				is_decaying = yes
			}
			nd_reward_hawks = { OTHER = scope:nd_issuer LIST = nd_lob_tl GOOD = no }
			nd_lobby_react = { OTHER = scope:nd_issuer TYPE = anti AMOUNT = -3 EVENT = positive }
			nd_lobby_react = { OTHER = scope:nd_issuer TYPE = pro AMOUNT = 2 EVENT = positive }
		}
	}
	# ---- the issuer backed down (2, 5) -------------------------------------
	else_if = {
		limit = {
			scope:nd_issuer = {
				OR = {
					var:nd_crisis_outcome_now = 2
					var:nd_crisis_outcome_now = 5
				}
			}
		}
		scope:nd_issuer = {
			if = {
				limit = { nd_crisis_is_public = yes }
				nd_change_credibility = { AMOUNT = -15 }
				add_modifier = {
					name = nd_crisis_climbdown
					days = normal_modifier_time
					is_decaying = yes
				}
				nd_reward_hawks = { OTHER = scope:nd_target LIST = nd_lob_il GOOD = no }
				nd_lobby_react = { OTHER = scope:nd_target TYPE = anti AMOUNT = -4 EVENT = positive }
				nd_lobby_react = { OTHER = scope:nd_target TYPE = pro AMOUNT = 2 EVENT = positive }
			}
			else = {
				nd_change_credibility = { AMOUNT = -5 }
			}
		}
		scope:nd_target = {
			nd_change_credibility = { AMOUNT = 10 }
			add_modifier = {
				name = nd_crisis_stood_firm
				days = normal_modifier_time
				is_decaying = yes
			}
			nd_reward_hawks = { OTHER = scope:nd_issuer LIST = nd_lob_tw GOOD = yes }
			nd_lobby_react = { OTHER = scope:nd_issuer TYPE = anti AMOUNT = 3 EVENT = negative }
			nd_lobby_react = { OTHER = scope:nd_issuer TYPE = pro AMOUNT = -2 EVENT = negative }
		}
	}
	# ---- reciprocal stand-down (3) -------------------------------------------
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_outcome_now = 3 } }
		scope:nd_issuer = {
			# A public demand traded away for a stand-down it did not ask for is
			# half an audience cost; a defensive issuer got what it wanted.
			if = {
				limit = {
					nd_crisis_is_public = yes
					nd_crisis_issuer_is_defensive = no
				}
				nd_change_credibility = { AMOUNT = -5 }
				nd_lobby_react = { OTHER = scope:nd_target TYPE = anti AMOUNT = -2 EVENT = positive }
			}
			else = {
				nd_change_credibility = { AMOUNT = 5 }
			}
			nd_lobby_react = { OTHER = scope:nd_target TYPE = pro AMOUNT = 2 EVENT = positive }
			add_modifier = {
				name = nd_crisis_mutual_restraint
				days = normal_modifier_time
				is_decaying = yes
			}
			nd_reward_doves = yes
		}
		scope:nd_target = {
			nd_change_credibility = { AMOUNT = 5 }
			add_modifier = {
				name = nd_crisis_mutual_restraint
				days = normal_modifier_time
				is_decaying = yes
			}
			nd_reward_doves = yes
			nd_lobby_react = { OTHER = scope:nd_issuer TYPE = pro AMOUNT = 2 EVENT = positive }
		}
	}
	# ---- an ultimatum left to lapse (8) --------------------------------------
	else_if = {
		limit = {
			scope:nd_issuer = {
				var:nd_crisis_outcome_now = 8
				nd_crisis_is_public = yes
			}
		}
		scope:nd_issuer = {
			nd_change_credibility = { AMOUNT = -10 }
		}
	}
}

# Country scope. The hawkish classes react to how we did against $OTHER$.
# Groups that belong to an anti-$OTHER$ lobby are paid through the lobby
# (nd_lobby_react), not twice (§7). LIST names the temporary list, distinct per

nd_reward_hawks = {

		}
	}
	every_interest_group = {
		limit = {
			OR = {
				nd_ig_class_warfighting = yes
				nd_ig_class_professional = yes
			}
			NOT = { is_in_list = $LIST$ }
		}

			limit = { always = $GOOD$ }
			add_modifier = {
				name = ig_approval_positive_modifier
				days = normal_modifier_time
				is_decaying = yes
			}

		else = {

				name = ig_approval_negative_modifier

}

# Country scope. Native lobby appeasement (§7): the lobbies in this country
# that are about $OTHER$. TYPE = anti | pro; AMOUNT is a signed literal.
# EVENT = positive | negative is how the event treats $OTHER$, not the sign of
# AMOUNT. It picks the vanilla "From Events" factor line, and a pro/anti pair
# reacting to one event passes the same EVENT with opposite-signed amounts, as
# vanilla's add_lobby_appeasement_from_diplomacy_unidirectional does with its
# FACTOR. The engine rejects a factor that the lobby type does not list for the
# sign of the amount (00_political_lobbies.txt): an anti lobby gains on
# _negative and loses on _positive.
nd_lobby_react = {
	every_political_lobby = {
		limit = {
			is_political_lobby_type = lobby_$TYPE$_country
			target = $OTHER$
		}
		change_appeasement = {
			amount = $AMOUNT$
			factor = appeasement_special_events_$EVENT$
		}
	}
}

# Target scope: concede (§6). The concession depends on the dispute and is
# applied while the crisis record still says what the dispute is; the close
# follows. If the concession is a play resolved for the other side, the
# engine's own on_diplo_play_back_down may close the crisis first (as a native
# win for the issuer, outcome 4) — nd_crisis_close then finds nothing to close,
# so either way it closes exactly once.
nd_crisis_act_yield = {
	custom_tooltip = nd_tt_act_yield
	hidden_effect = {
		nd_crisis_save_parties = yes
		if = {
			limit = {
				exists = scope:nd_issuer
				exists = scope:nd_target
				scope:nd_issuer = { nd_is_crisis_owner = yes }
			}
			scope:nd_issuer = {
				if = {
					limit = { has_variable = nd_crisis_beneficiary }
					var:nd_crisis_beneficiary ?= { save_scope_as = nd_crisis_other_side }
				}
				else = {
					save_scope_as = nd_crisis_other_side
				}
			}
			scope:nd_issuer = {
				if = {
					limit = {
						OR = {
							var:nd_crisis_dispute = 1
							var:nd_crisis_dispute = 3
						}
						exists = scope:nd_crisis_other_side
						scope:nd_target = { NOT = { has_war_with = scope:nd_crisis_other_side } }
					}
					nd_crisis_concede_play = yes
				}
				else_if = {
					limit = {
						OR = {
							var:nd_crisis_dispute = 2
							var:nd_crisis_dispute = 3
						}
						exists = scope:nd_crisis_other_side
					}
					nd_crisis_concede_war = yes
				}
				else_if = {
					limit = { var:nd_crisis_dispute = 4 }
					scope:nd_target = {
						add_modifier = {
							name = nd_crisis_programme_freeze
							years = 10
						}
					}
				}
				else = {
					scope:nd_target = {
						if = {
							limit = { has_variable = nd_readiness }
							nd_set_readiness_target_1 = yes
							set_variable = { name = nd_readiness_lock_months value = nd_readiness_lock_months_value }
						}
					}
				}
			}
			nd_crisis_close = { OUTCOME = 1 }
		}
	}
}

# scope:nd_target (the yielder, a lead of the play) and
# scope:nd_crisis_other_side (the issuer, or the beneficiary it protects).
# Ends the play for the other side with resolve_play_for — the engine's own
# back-down outcome, war goals and all. VERIFY IN-GAME (§0.9 check 5): this is
# the one engine hook in the system with no vanilla script precedent.
nd_crisis_concede_play = {
	random_diplomatic_play = {
		limit = {
			is_war = no
			OR = {
				initiator_is = scope:nd_target
				target_is = scope:nd_target
			}
			any_scope_play_involved = { this = scope:nd_crisis_other_side }
		}
		save_scope_as = nd_crisis_play
	}
	if = {
		limit = { exists = scope:nd_crisis_play }
		scope:nd_crisis_play = {
			if = {
				limit = { initiator_is = scope:nd_target }
				random_scope_play_involved = {
					limit = { scope:nd_crisis_play = { target_is = PREV } }
					save_scope_as = nd_crisis_play_winner
				}
				if = {
					limit = { exists = scope:nd_crisis_play_winner }
					resolve_play_for = scope:nd_crisis_play_winner
				}
			}
			else = {
				resolve_play_for = initiator
			}
		}
	}
}

# NATIVE OBSERVATION — on_diplo_play_back_down (nuclear_crisis_on_actions.txt)
# ----------------------------------------------------------------------------
# Diplomatic-play scope; scope:actor = the country that backed down. Closes a
# play crisis with a winner the moment one side yields natively: the target (or
# its side) backing down achieves the issuer's objective (4); the issuer's side
# backing down is its failure (5).
nd_crisis_on_back_down = {
	save_scope_as = nd_bd_play
	every_scope_play_involved = {
		limit = {
			nd_in_crisis = yes
		}
		# Re-checked in the body: closing one side's crisis earlier in this
		# loop clears the other side's record too.
		if = {
			limit = { nd_in_crisis = yes }
			nd_crisis_save_parties = yes
		}
		if = {
			limit = {
				nd_in_crisis = yes
				exists = scope:nd_issuer
				exists = scope:nd_target
				scope:nd_issuer = {
					OR = {
						var:nd_crisis_dispute = 1
						var:nd_crisis_dispute = 3
					}
				}
				scope:nd_bd_play = {
					any_scope_play_involved = { this = scope:nd_target }
				}
			}
			if = {
				limit = { scope:actor = scope:nd_target }
				nd_crisis_close = { OUTCOME = 4 }
			}
			else_if = {
				limit = {
					OR = {
						scope:actor = scope:nd_issuer
						scope:nd_issuer = { var:nd_crisis_beneficiary ?= scope:actor }
					}
				}
				nd_crisis_close = { OUTCOME = 5 }
			}
		}
	}
}

					nd_has_crisis_dispute_with = { TARGET = scope:nd_guarantee_attacker }
				}
				nd_crisis_open = { TARGET = scope:nd_guarantee_attacker PUBLIC = yes }
			}
			nd_crisis_ai_review = yes
			scope:nd_guarantee_beneficiary = {
				set_variable = { name = nd_guarantee_answer value = 1 }
				trigger_event = { id = nuclear_crisis.21 }
			}
		}
	}
}

# Answer a strike on the beneficiary in kind: at war with the attacker, with
# doctrine and pledges permitting. Through the one dispatch path.
nd_guarantee_act_retaliate = {
	custom_tooltip = nd_tt_guarantee_retaliate
	nd_change_credibility = { AMOUNT = 10 }
	hidden_effect = {
		if = {
			limit = { exists = scope:nd_guarantee_attacker }
			scope:nd_guarantee_attacker = { save_scope_as = nd_strike_victim }
			nd_dispatch_strategic_strike = yes
			scope:nd_guarantee_beneficiary ?= {
				set_variable = { name = nd_guarantee_answer value = 1 }
				trigger_event = { id = nuclear_crisis.21 }
			}
		}
	}
}

# Walk away (§6: abandoning a covered commitment shakes every beneficiary).
nd_guarantee_act_abandon = {
	custom_tooltip = nd_tt_guarantee_abandon
	nd_change_credibility = { AMOUNT = -15 }
	add_modifier = {
		name = nd_guarantee_abandoned
		years = 5
		is_decaying = yes
	}
	hidden_effect = {
		save_scope_as = nd_guarantor_self
		if = {
			limit = {
				exists = scope:nd_guarantee_attacker
				exists = scope:nd_guarantee_beneficiary
			}
			if = {
				limit = {
					var:nd_guarantee_context = 1
					scope:nd_guarantee_attacker = {
						nd_is_crisis_owner = yes
						var:nd_crisis_opponent ?= scope:nd_guarantee_beneficiary
					}
				}
				scope:nd_guarantee_attacker = {
					set_variable = { name = nd_crisis_abandoned value = 1 }
				}
			}
			scope:nd_guarantee_beneficiary = {
				change_relations = { country = scope:nd_guarantor_self value = -30 }
				set_variable = { name = nd_guarantee_answer value = 0 }
				trigger_event = { id = nuclear_crisis.21 }
			}
			# Every other beneficiary of ours takes note.
			every_scope_treaty = {
				every_scope_article = {
					limit = {
						has_type = nuclear_guarantee
						source_country = scope:nd_guarantor_self
						NOT = { target_country = scope:nd_guarantee_beneficiary }
					}
					target_country = {
						change_relations = { country = scope:nd_guarantor_self value = -10 }
					}
				}
			}
		}
	}
}
