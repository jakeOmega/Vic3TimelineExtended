# ============================================================================
# NUCLEAR CRISIS — opening, weekly tick, acts, settlements, outcomes
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0.3 is what shipped).
#
# DATA MODEL. A crisis is one pair of countries with one dispute. The record
# lives on the ISSUER (the country that warned; the single update owner, role
# 1); the TARGET (role 2) holds a mirror: nd_crisis_id, nd_crisis_opponent,
# nd_crisis_role. A country is in at most one crisis at a time.
#   issuer only: nd_crisis_dispute (1 play, 2 war, 3 guarantee, 4 proliferation,
#                5 alert), nd_crisis_beneficiary (dispute 3), nd_crisis_stage
#                (1 warning, 2 confrontation, 3 acute), nd_crisis_public,
#                nd_crisis_weeks, nd_crisis_deadline, nd_crisis_deadline_fired,
#                nd_crisis_danger, nd_crisis_counter, nd_crisis_talks,
#                nd_crisis_pressure_weeks, nd_crisis_backed (the target's
#                guarantor stands with it), nd_crisis_abandoned (it declined)
#   target only: nd_yield_pressure (written weekly by the tick)
#   after close: nd_crisis_last_outcome, nd_crisis_last_opponent,
#                nd_crisis_last_role (display); on the issuer
#                nd_crisis_cooldown_target / nd_crisis_cooldown_months
#
# SAVED SCOPES. Almost everything here expects scope:nd_issuer and
# scope:nd_target, saved by nd_crisis_save_parties from either party's scope,
# and hands them on to the events it fires (saved scopes survive trigger_event).
#
# OUTCOMES (nd_crisis_close = { OUTCOME = N }):
#   1  the target yielded                 6  the war ended
#   2  the issuer backed down             7  nuclear use
#   3  reciprocal stand-down              8  expired
#   4  the target backed down natively    9  invalid (a party vanished)
#   5  the issuer's side backed down      10 the dispute ended, no clear winner
#
# TOOLTIP PASSES. The diplomatic actions' accept_effect and the panel's
# ExecuteTooltip re-walk these effects every frame without running them, so
# every use of a scope saved inside this file is guarded by `exists`.
# ============================================================================

# ----------------------------------------------------------------------------
# PARTIES
# ----------------------------------------------------------------------------
# Either party's scope. Saves scope:nd_issuer and scope:nd_target.
nd_crisis_save_parties = {
	if = {
		limit = { nd_is_crisis_owner = yes }
		save_scope_as = nd_issuer
		var:nd_crisis_opponent ?= { save_scope_as = nd_target }
	}
	else_if = {
		limit = { nd_is_crisis_target = yes }
		save_scope_as = nd_target
		var:nd_crisis_opponent ?= { save_scope_as = nd_issuer }
	}
}

# Hands a decision event to $WHO$ (scope:nd_issuer or scope:nd_target) with
# the crisis's saved scopes. The token lets the event and its options confirm
# that the popup still belongs to the live crisis (nd_crisis_event_valid).
nd_crisis_send_event = {
	$WHO$ = {
		set_variable = { name = nd_crisis_event_token value = var:nd_crisis_id }
		trigger_event = { id = $EVENT$ popup = yes }
	}
}

# ----------------------------------------------------------------------------
# OPENING
# ----------------------------------------------------------------------------
# Issuer scope. What opening a crisis against $TARGET$ does and risks, for the
# diplomatic actions' confirmation box and the event options that open one
# (2026-09-25 spec §2.1). Reads only our own state and $TARGET$, which exist in
# a tooltip preview (a temporary scope saved inside a `limit` does exist
# there; a scope saved by an effect does not). Every line sits at this scope
# level so the lines print in order (gui_modding_guide.md gotcha #18). The infamy and relations lines are
# the real effects: every caller has already checked that the crisis opens
# (nd_crisis_parties_free), so they are not fenced.
nd_crisis_preview = {
	# ---- what the world sees ------------------------------------------------
	# Charged only if the crisis will open: nd_crisis_open's fence checks the
	# same (every caller checks it too; this keeps a future caller honest).
	if = {
		limit = { nd_crisis_parties_free = { TARGET = $TARGET$ } }
		if = {
			limit = { always = $PUBLIC$ }
			change_infamy = 5
			change_relations = { country = $TARGET$ value = -20 }
		}
		else = {
			change_relations = { country = $TARGET$ value = -5 }
		}
	}
	# ---- what it is about, and what conceding would cost them ----------------
	if = {
		limit = { nd_dispute_war_with = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_over_war
	}
	else_if = {
		limit = { nd_dispute_play_with = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_over_play
	}
	else_if = {
		limit = { nd_dispute_guarantee_against = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_over_guarantee
	}
	else_if = {
		limit = { nd_dispute_proliferation_by = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_over_proliferation
	}
	else = {
		custom_tooltip = nd_tt_open_over_alert
	}
	# ---- can we carry it out? ---------------------------------------------------
	if = {
		limit = { nd_threat_bluff_nfu = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_bluff_nfu
	}
	else_if = {
		limit = { NOT = { nd_threat_law_permits = { TARGET = $TARGET$ } } }
		custom_tooltip = nd_tt_open_bluff_law
	}
	else_if = {
		limit = { nd_threat_backed = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_backed
	}
	else_if = {
		limit = { nd_threat_uncertain = { TARGET = $TARGET$ } }
		custom_tooltip = nd_tt_open_uncertain
	}
	else_if = {
		limit = { nd_doctrine_flexible = yes }
		custom_tooltip = nd_tt_open_bluff_flexible
	}
	else = {
		custom_tooltip = nd_tt_open_bluff_existential
	}
	# ---- what else weighs on them (nd_yp_*; danger and exercises come later) --
	custom_tooltip = nd_tt_open_factors_header
	if = {
		limit = { $TARGET$ = { nd_believed_armed = no } }
		custom_tooltip = nd_tt_open_f_unarmed
	}
	else_if = {
		limit = {
			$TARGET$ = {
				has_variable = nd_survivability
				var:nd_survivability >= 50
			}
		}
		custom_tooltip = nd_tt_open_f_survivable
	}
	else = {
		custom_tooltip = nd_tt_open_f_armed
	}
	if = {
		limit = {
			save_temporary_scope_as = nd_pv_self
			$TARGET$ = { nd_has_armed_guarantor_against = { AGAINST = scope:nd_pv_self } }
		}
		custom_tooltip = nd_tt_open_f_protector
	}
	custom_tooltip = nd_tt_open_f_credibility
	if = {
		limit = { nd_readiness_high_alert = yes }
		custom_tooltip = nd_tt_open_f_alert
	}
	if = {
		limit = { $TARGET$ = { ruler_is_aggressive = yes } }
		custom_tooltip = nd_tt_open_f_aggressive
	}
	if = {
		limit = { $TARGET$ = { ruler_is_cautious = yes } }
		custom_tooltip = nd_tt_open_f_cautious
	}
	if = {
		limit = {
			save_temporary_scope_as = nd_pv_self
			$TARGET$ = {
				has_war_with = scope:nd_pv_self
				nd_is_losing_war_to = { ENEMY = scope:nd_pv_self }
			}
		}
		custom_tooltip = nd_tt_open_f_losing
	}
	if = {
		limit = {
			save_temporary_scope_as = nd_pv_self
			$TARGET$ = { nd_enemy_threatens_existence = { ENEMY = scope:nd_pv_self } }
		}
		custom_tooltip = nd_tt_open_f_existence
	}
	# ---- the stakes, and what happens next ------------------------------------
	if = {
		limit = { always = $PUBLIC$ }
		custom_tooltip = nd_tt_open_stakes_public
		if = {
			limit = {
				NOT = { nd_threat_backed = { TARGET = $TARGET$ } }
				NOT = { nd_threat_uncertain = { TARGET = $TARGET$ } }
			}
			custom_tooltip = nd_tt_open_stakes_bluff
		}
		custom_tooltip = nd_tt_open_next_public
	}
	else = {
		custom_tooltip = nd_tt_open_stakes_private
		custom_tooltip = nd_tt_open_next_private
	}
}

# Issuer scope (ROOT = issuer: the diplomatic actions' accept_effect).
# TARGET = scope:target_country. PUBLIC = no (private warning) | yes
# (ultimatum) — a literal flag, so `always = $PUBLIC$` reads the same in a
# tooltip preview as in the run.
#
# TOOLTIP PASSES. The engine re-walks this effect every frame a diplomatic
# action's confirmation box (or an event option) is on screen, without running
# it, and it evaluates the limits and resolves the scopes inside hidden_effect
# too (scripting_best_practices.md § Per-Entity State rule 6). Every read of a
# crisis variable below would then hit a variable that does not exist yet. So
# the body is fenced by a scope saved on the first line: save_scope_as is not
# run by the preview, so scope:nd_crisis_opening only exists in the real run.
# What the world sees — the infamy and relations — is nd_crisis_preview's,
# outside the fence, so the confirmation box shows it as a real effect.
nd_crisis_open = {
	nd_crisis_preview = { TARGET = $TARGET$ PUBLIC = $PUBLIC$ }
	hidden_effect = {
		save_scope_as = nd_crisis_opening
		if = {
			limit = {
				exists = scope:nd_crisis_opening
				exists = $TARGET$
				nd_crisis_parties_free = { TARGET = $TARGET$ }
			}
			save_scope_as = nd_issuer
			$TARGET$ = { save_scope_as = nd_target }
			if = {
				limit = { NOT = { has_global_variable = nd_crisis_counter } }
				set_global_variable = { name = nd_crisis_counter value = 0 }
			}
			change_global_variable = { name = nd_crisis_counter add = 1 }
			# ---- the issuer's record ----------------------------------------
			set_variable = { name = nd_crisis_id value = global_var:nd_crisis_counter }
			set_variable = { name = nd_crisis_opponent value = scope:nd_target }
			set_variable = { name = nd_crisis_role value = 1 }
			nd_crisis_classify_dispute = yes
			set_variable = { name = nd_crisis_weeks value = 0 }
			set_variable = { name = nd_crisis_pressure_weeks value = 0 }
			set_variable = { name = nd_crisis_counter value = 0 }
			set_variable = { name = nd_crisis_talks value = 0 }
			set_variable = { name = nd_crisis_backed value = 0 }
			set_variable = { name = nd_crisis_abandoned value = 0 }
			set_variable = { name = nd_crisis_deadline_fired value = 0 }
			if = {
				limit = { always = $PUBLIC$ }
				set_variable = { name = nd_crisis_public value = 1 }
				set_variable = { name = nd_crisis_stage value = 2 }
				set_variable = { name = nd_crisis_deadline value = nd_crisis_deadline_public_weeks }
			}
			else = {
				set_variable = { name = nd_crisis_public value = 0 }
				set_variable = { name = nd_crisis_stage value = 1 }
				set_variable = { name = nd_crisis_deadline value = nd_crisis_deadline_private_weeks }
			}
			# ---- the target's mirror ----------------------------------------
			scope:nd_target = {
				set_variable = { name = nd_crisis_id value = scope:nd_issuer.var:nd_crisis_id }
				set_variable = { name = nd_crisis_opponent value = scope:nd_issuer }
				set_variable = { name = nd_crisis_role value = 2 }
			}
			# ---- the crisis's figures, before anyone weighs them ------------
			nd_crisis_refresh_figures = yes
			# ---- both AIs reconsider their posture now ------------------------
			scope:nd_issuer = { nd_crisis_ai_review = yes }
			scope:nd_target = { nd_crisis_ai_review = yes }
			# ---- the target's guarantors are asked whether they stand by it ----
			nd_crisis_notify_guarantors = yes
			# ---- the target answers -----------------------------------------
			nd_crisis_send_event = { WHO = scope:nd_target EVENT = nuclear_crisis.1 }
		}
	}
}

# Issuer scope, scope:nd_target saved. Picks the dispute in priority order —
# war, play, guarantee, proliferation, alert — and for a guarantee dispute
# remembers the beneficiary.
nd_crisis_classify_dispute = {
	if = {
		limit = { nd_dispute_war_with = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 2 }
	}
	else_if = {
		limit = { nd_dispute_play_with = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 1 }
	}
	else_if = {
		limit = { nd_dispute_guarantee_against = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 3 }
		every_scope_treaty = {
			every_scope_article = {
				limit = {
					has_type = nuclear_guarantee
					source_country = scope:nd_issuer
					target_country = {
						OR = {
							has_war_with = scope:nd_target
							is_diplomatic_play_enemy_of = scope:nd_target
						}
					}
				}
				target_country = { save_scope_as = nd_crisis_ben }
			}
		}
		# ...or a subject under our umbrella (umbrella/recessed/dead-hand spec
		# §1.2).
		if = {
			limit = { NOT = { exists = scope:nd_crisis_ben } }
			random_direct_subject = {
				limit = {
					nd_under_umbrella_of = { OVERLORD = scope:nd_issuer }
					nd_beneficiary_threatened_by = { TARGET = scope:nd_target }
				}
				save_scope_as = nd_crisis_ben
			}
		}
		if = {
			limit = { exists = scope:nd_crisis_ben }
			set_variable = { name = nd_crisis_beneficiary value = scope:nd_crisis_ben }
		}
	}
	else_if = {
		limit = { nd_dispute_proliferation_by = { TARGET = scope:nd_target } }
		set_variable = { name = nd_crisis_dispute value = 4 }
	}
	else = {
		set_variable = { name = nd_crisis_dispute value = 5 }
	}
}

# Issuer scope. Records which follow-through branch holds against $TARGET$
# (nd_ft_reason, table in nuclear_deterrence_triggers.txt § FOLLOW-THROUGH):
#   1 at war, the doctrine permits a strike now     backed
#   2 Compellence or higher (defiance licenses it)  backed
#   3 not at war, Flexible first use                uncertain
#   4 not at war, Existential with a stake          uncertain
#   5 the doctrine does not allow it                bluff
#   6 the Rules of War law forbids it               bluff
#   7 No First Use                                  bluff
# Called by nd_crisis_refresh_figures on opening and every week, so a doctrine
# change or a war turning against us counts the same week.
nd_crisis_classify_follow_through = {
	if = {
		limit = { nd_threat_bluff_nfu = { TARGET = $TARGET$ } }
		set_variable = { name = nd_ft_reason value = 7 }
	}
	else_if = {
		limit = { NOT = { nd_threat_law_permits = { TARGET = $TARGET$ } } }
		set_variable = { name = nd_ft_reason value = 6 }
	}
	else_if = {
		limit = {
			has_war_with = $TARGET$
			nd_doctrine_permits_strike = { ENEMY = $TARGET$ }
		}
		set_variable = { name = nd_ft_reason value = 1 }
	}
	else_if = {
		limit = { nd_threat_backed = { TARGET = $TARGET$ } }
		set_variable = { name = nd_ft_reason value = 2 }
	}
	else_if = {
		limit = {
			nd_threat_uncertain = { TARGET = $TARGET$ }
			nd_doctrine_flexible = yes
		}
		set_variable = { name = nd_ft_reason value = 3 }
	}
	else_if = {
		limit = { nd_threat_uncertain = { TARGET = $TARGET$ } }
		set_variable = { name = nd_ft_reason value = 4 }
	}
	else = {
		set_variable = { name = nd_ft_reason value = 5 }
	}
}

# Either party's scope: an AI party reviews its posture at once, in its own
# event so that its posture effects see it as ROOT
# (nuclear_deterrence_effects.txt header, "ROOT").
nd_crisis_ai_review = {
	if = {
		limit = {
			is_ai = yes
			nd_is_armed = yes
		}
		trigger_event = { id = nuclear_crisis.99 }
	}
}

# scope:nd_issuer and scope:nd_target saved. Every armed guarantor of the
# target other than the issuer is asked (nuclear_crisis.20), under the same
# scope names nd_record_nuclear_use uses for a strike.
nd_crisis_notify_guarantors = {
	scope:nd_issuer = { save_scope_as = nd_guarantee_attacker }
	scope:nd_target = { save_scope_as = nd_guarantee_beneficiary }
	scope:nd_target = {
		every_scope_treaty = {
			every_scope_article = {
				limit = {
					has_type = nuclear_guarantee
					target_country = scope:nd_target
					source_country = {
						NOT = { this = scope:nd_issuer }
						nd_believed_armed = yes
					}
				}
				source_country = {
					save_scope_as = nd_guarantor
					set_variable = { name = nd_guarantee_context value = 1 }
					trigger_event = { id = nuclear_crisis.20 }
				}
			}
		}
		# The umbrella: our overlord, unless it is the issuer or the treaty
		# loop above has already asked it (umbrella/recessed/dead-hand spec
		# §1.2; siblings in nd_record_nuclear_use and nd_guarantee_act_abandon).
		if = {
			limit = {
				nd_under_an_umbrella = yes
				overlord = { NOT = { this = scope:nd_issuer } }
			}
			overlord = { save_scope_as = nd_umb_guarantor }
			if = {
				limit = { NOT = { nd_is_guaranteed_by_treaty = { GUARANTOR = scope:nd_umb_guarantor } } }
				scope:nd_umb_guarantor = {
					save_scope_as = nd_guarantor
					set_variable = { name = nd_guarantee_context value = 1 }
					trigger_event = { id = nuclear_crisis.20 }
				}
			}
		}
	}
}

# ----------------------------------------------------------------------------
# FIGURES — the one writer of the crisis's numbers (2026-09-25 spec §1.2)
# ----------------------------------------------------------------------------
# Issuer scope, parties saved. Classifies follow-through, then stores every
# danger part (issuer-scope values) and every pressure part (target-scope
# values) on BOTH parties, then the totals. The panel's breakdowns print the
# stored parts (nd_crisis_*_breakdown_sgui), so what it shows always sums to
# what the AI weighs.
nd_crisis_refresh_figures = {
	nd_crisis_classify_follow_through = { TARGET = scope:nd_target }
	scope:nd_target = {
		set_variable = { name = nd_ft_reason value = scope:nd_issuer.var:nd_ft_reason }
	}
	nd_crisis_store_cd = { C = nd_cd_stage V = nd_cd_stage_value }
	nd_crisis_store_cd = { C = nd_cd_issuer_readiness V = nd_cd_issuer_readiness_value }
	nd_crisis_store_cd = { C = nd_cd_target_readiness V = nd_cd_target_readiness_value }
	nd_crisis_store_cd = { C = nd_cd_public V = nd_cd_public_value }
	nd_crisis_store_cd = { C = nd_cd_counter V = nd_cd_counter_value }
	nd_crisis_store_cd = { C = nd_cd_reliability V = nd_cd_reliability_value }
	nd_crisis_store_cd = { C = nd_cd_weeks V = nd_cd_weeks_value }
	nd_crisis_store_cd = { C = nd_cd_talks V = nd_cd_talks_value }
	nd_crisis_store_cd = { C = nd_cd_backed V = nd_cd_backed_value }
	nd_crisis_store_cd = { C = nd_cd_exercise V = nd_cd_exercise_value }
	if = {
		limit = { nd_crisis_danger_dampened = yes }
		set_variable = { name = nd_cd_dampened value = 1 }
	}
	else = {
		set_variable = { name = nd_cd_dampened value = 0 }
	}
	scope:nd_target = {
		set_variable = { name = nd_cd_dampened value = scope:nd_issuer.var:nd_cd_dampened }
	}
	set_variable = { name = nd_crisis_danger value = nd_crisis_danger_value }
	scope:nd_target = {
		nd_crisis_store_yp = { C = nd_yp_base V = nd_yp_base_value }
		nd_crisis_store_yp = { C = nd_yp_answer V = nd_yp_answer_value }
		nd_crisis_store_yp = { C = nd_yp_protector V = nd_yp_protector_value }
		nd_crisis_store_yp = { C = nd_yp_credibility V = nd_yp_credibility_value }
		nd_crisis_store_yp = { C = nd_yp_alert V = nd_yp_alert_value }
		nd_crisis_store_yp = { C = nd_yp_recessed V = nd_yp_recessed_value }
		nd_crisis_store_yp = { C = nd_yp_danger V = nd_yp_danger_value }
		nd_crisis_store_yp = { C = nd_yp_exercise V = nd_yp_exercise_value }
		nd_crisis_store_yp = { C = nd_yp_temperament V = nd_yp_temperament_value }
		nd_crisis_store_yp = { C = nd_yp_war V = nd_yp_war_value }
		nd_crisis_store_yp = { C = nd_yp_follow_through V = nd_yp_follow_through_value }
		set_variable = { name = nd_yield_pressure value = nd_yield_pressure_value }
	}
}

# Issuer scope, parties saved: one danger part, stored and mirrored.
nd_crisis_store_cd = {
	set_variable = { name = $C$ value = $V$ }
	scope:nd_target = {
		set_variable = { name = $C$ value = scope:nd_issuer.var:$C$ }
	}
}

# Target scope, parties saved: one pressure part, stored and mirrored.
nd_crisis_store_yp = {
	set_variable = { name = $C$ value = $V$ }
	scope:nd_issuer = {
		set_variable = { name = $C$ value = scope:nd_target.var:$C$ }
	}
}

# ----------------------------------------------------------------------------
# THE WEEKLY TICK — owner scope, from nd_weekly_update only
# ----------------------------------------------------------------------------
nd_crisis_weekly_tick = {
	nd_crisis_save_parties = yes
	if = {
		limit = {
			OR = {
				NOT = { exists = scope:nd_target }
				nd_crisis_pair_intact = no
			}
		}
		nd_crisis_clear_own_record = yes
	}
	else = {
		change_variable = { name = nd_crisis_weeks add = 1 }
		change_variable = { name = nd_crisis_pressure_weeks add = 1 }
		if = {
			limit = { has_variable = nd_crisis_exercise_weeks }
			change_variable = { name = nd_crisis_exercise_weeks subtract = 1 }
			if = {
				limit = { var:nd_crisis_exercise_weeks <= 0 }
				remove_variable = nd_crisis_exercise_weeks
				if = {
					limit = { has_variable = nd_crisis_exercise_by_issuer }
					remove_variable = nd_crisis_exercise_by_issuer
				}
			}
		}
		nd_crisis_revalidate_dispute = yes
	}
	# Still open after revalidation?
	if = {
		limit = { nd_is_crisis_owner = yes }
		nd_crisis_refresh_figures = yes
		# ---- confrontation becomes acute ------------------------------------
		if = {
			limit = {
				var:nd_crisis_stage = 2
				OR = {
					var:nd_crisis_danger >= 70
					AND = {
						nd_readiness_high_alert = yes
						scope:nd_target = { nd_readiness_heightened_or_higher = yes }
					}
				}
			}
			set_variable = { name = nd_crisis_stage value = 3 }
		}
		# ---- expiry -----------------------------------------------------------
		if = {
			limit = { var:nd_crisis_weeks >= nd_crisis_max_weeks }
			nd_crisis_close = { OUTCOME = 8 }
		}
		# ---- the deadline -----------------------------------------------------
		else_if = {
			limit = {
				var:nd_crisis_deadline_fired = 0
				var:nd_crisis_weeks >= var:nd_crisis_deadline
			}
			set_variable = { name = nd_crisis_deadline_fired value = 1 }
			nd_crisis_send_event = { WHO = scope:nd_issuer EVENT = nuclear_crisis.4 }
		}
		# ---- pressure on the target -------------------------------------------
		else_if = {
			limit = {
				var:nd_crisis_stage >= 2
				var:nd_crisis_talks = 0
				var:nd_crisis_pressure_weeks >= nd_crisis_pressure_interval_weeks
			}
			set_variable = { name = nd_crisis_pressure_weeks value = 0 }
			nd_crisis_send_event = { WHO = scope:nd_target EVENT = nuclear_crisis.5 }
		}
	}
}

# Owner scope, parties saved. Converts or closes a crisis whose dispute has
# moved on. The on_diplo_play_back_down hook closes a play crisis with a
# winner the moment it happens; this is the net for everything else.
nd_crisis_revalidate_dispute = {
	# ---- a play that became a war -----------------------------------------
	if = {
		limit = {
			var:nd_crisis_dispute = 1
			has_war_with = scope:nd_target
		}
		set_variable = { name = nd_crisis_dispute value = 2 }
		set_variable = { name = nd_crisis_stage value = 3 }
		nd_crisis_record_defiance = yes
		if = {
			limit = { nd_crisis_is_public = yes }
			nd_crisis_send_event = { WHO = scope:nd_issuer EVENT = nuclear_crisis.7 }
		}
	}
	# ---- a play that ended without a war and without the hook seeing it ------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 1
			NOT = { is_diplomatic_play_enemy_of = scope:nd_target }
		}
		nd_crisis_close = { OUTCOME = 10 }
	}
	# ---- a war that ended -------------------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 2
			NOT = { has_war_with = scope:nd_target }
		}
		nd_crisis_close = { OUTCOME = 6 }
	}
	# ---- a guarantee dispute ----------------------------------------------
	else_if = {
		limit = { var:nd_crisis_dispute = 3 }
		if = {
			limit = { has_war_with = scope:nd_target }
			set_variable = { name = nd_crisis_dispute value = 2 }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_variable = nd_crisis_beneficiary }
					NOT = {
						var:nd_crisis_beneficiary ?= {
							OR = {
								has_war_with = scope:nd_target
								is_diplomatic_play_enemy_of = scope:nd_target
							}
						}
					}
				}
			}
			nd_crisis_close = { OUTCOME = 10 }
		}
	}
	# ---- proliferation that stopped ---------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 4
			scope:nd_target = { nuclear_program_is_proliferating = no }
		}
		nd_crisis_close = { OUTCOME = 4 }
	}
	# ---- an alert that was stood down -------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_dispute = 5
			scope:nd_target = {
				OR = {
					nd_is_armed = no
					NOT = { has_variable = nd_readiness_target }
					var:nd_readiness_target <= 1
				}
			}
		}
		nd_crisis_close = { OUTCOME = 4 }
	}
	# ---- a disarmed issuer has no threat left -----------------------------
	if = {
		limit = {
			nd_is_crisis_owner = yes
			nd_is_armed = no
		}
		nd_crisis_close = { OUTCOME = 10 }
	}
}

# ----------------------------------------------------------------------------
# CLOSING — scope:nd_issuer and scope:nd_target saved, any scope
# ----------------------------------------------------------------------------
# Records the outcome once, for each side's outcome notice to apply
# (nd_crisis_record_pending; nuclear_crisis.6), and cleans both records.
# OUTCOME is a literal. Fenced like nd_crisis_open: an option tooltip walks
# this without running it, and the record reads the nd_crisis_outcome_now
# written a line earlier, which a preview would find missing.
nd_crisis_close = {
	save_scope_as = nd_crisis_closing
	if = {
		limit = {
			exists = scope:nd_crisis_closing
			exists = scope:nd_issuer
			exists = scope:nd_target
			scope:nd_issuer = { nd_is_crisis_owner = yes }
		}
		# An outcome notice still unanswered from an earlier crisis settles
		# now, before this crisis's record replaces it (2026-09-25 spec §2.3).
		scope:nd_issuer = { nd_crisis_flush_pending = yes }
		scope:nd_target = { nd_crisis_flush_pending = yes }
		scope:nd_issuer = {
			set_variable = { name = nd_crisis_outcome_now value = $OUTCOME$ }
		}
		# Each side's consequences wait on its outcome notice's option
		# (nd_crisis_apply_outcome_side), so the option's tooltip shows them.
		if = {
			limit = { scope:nd_issuer = { NOT = { var:nd_crisis_outcome_now = 9 } } }
			nd_crisis_record_pending = yes
		}
		scope:nd_issuer = {
			set_variable = { name = nd_crisis_last_outcome value = $OUTCOME$ }
			set_variable = { name = nd_crisis_last_opponent value = scope:nd_target }
			set_variable = { name = nd_crisis_last_role value = 1 }
			set_variable = { name = nd_crisis_cooldown_target value = scope:nd_target }
			set_variable = { name = nd_crisis_cooldown_months value = nd_crisis_cooldown_months_value }
			nd_crisis_clear_vars = yes
		}
		scope:nd_target = {
			if = {
				limit = {
					nd_is_crisis_target = yes
					var:nd_crisis_opponent ?= scope:nd_issuer
				}
				nd_crisis_clear_vars = yes
			}
			set_variable = { name = nd_crisis_last_outcome value = $OUTCOME$ }
			set_variable = { name = nd_crisis_last_opponent value = scope:nd_issuer }
			set_variable = { name = nd_crisis_last_role value = 2 }
		}
		if = {
			limit = { scope:nd_issuer = { NOT = { var:nd_crisis_last_outcome = 9 } } }
			scope:nd_issuer = { trigger_event = { id = nuclear_crisis.6 popup = yes } }
			scope:nd_target = { trigger_event = { id = nuclear_crisis.6 popup = yes } }
		}
	}
}

# Country scope. Settles an outcome notice still waiting on a click, so a new
# close cannot overwrite it. The stale notice's option then finds its pair's
# record gone and says so (nuclear_crisis.6). Reached only through
# nd_crisis_close, which runs from pulses and hidden_effects: never call it
# from a visible option, or its credibility lines would render there.
nd_crisis_flush_pending = {
	if = {
		limit = { has_variable = nd_crisis_pending_outcome }
		var:nd_crisis_pending_opponent ?= { save_scope_as = nd_outcome_other }
		# If the opponent is gone, a scope saved by the other party's flush a
		# line earlier may still be set; nd_outcome_other_is_live keeps it
		# from being paid.
		nd_crisis_apply_outcome_side = yes
	}
}

# Country scope, the pending record set, scope:nd_outcome_other saved (the
# other party: nuclear_crisis.6's immediate, or nd_crisis_flush_pending).
# One side's consequences of a closed crisis (§6 of the nuclear design).
# Every line renders in the outcome notice's option tooltip.
nd_crisis_apply_outcome_side = {
	# ---- the objective was achieved (1, 4) ----------------------------------
	if = {
		limit = {
			OR = {
				var:nd_crisis_pending_outcome = 1
				var:nd_crisis_pending_outcome = 4
			}
		}
		if = {
			limit = { var:nd_crisis_pending_role = 1 }
			if = {
				limit = { var:nd_crisis_pending_public = 1 }
				custom_tooltip = {
					text = nd_tt_credibility_up_15
					nd_change_credibility = { AMOUNT = 15 }
				}
			}
			else = {
				custom_tooltip = {
					text = nd_tt_credibility_up_10
					nd_change_credibility = { AMOUNT = 10 }
				}
			}
			add_modifier = {
				name = nd_crisis_coercive_success
				days = normal_modifier_time
				is_decaying = yes
			}
			if = {
				limit = { nd_outcome_other_is_live = yes }
				custom_tooltip = {
					text = nd_tt_outcome_hawks_approve
					nd_reward_hawks = { OTHER = scope:nd_outcome_other LIST = nd_lob_iw GOOD = yes }
				}
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = anti AMOUNT = 3 EVENT = negative }
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = pro AMOUNT = -3 EVENT = negative }
			}
		}
		else = {
			custom_tooltip = {
				text = nd_tt_credibility_down_5
				nd_change_credibility = { AMOUNT = -5 }
			}
			add_modifier = {
				name = nd_crisis_yielded
				days = normal_modifier_time
				is_decaying = yes
			}
			if = {
				limit = { nd_outcome_other_is_live = yes }
				custom_tooltip = {
					text = nd_tt_outcome_hawks_disapprove
					nd_reward_hawks = { OTHER = scope:nd_outcome_other LIST = nd_lob_tl GOOD = no }
				}
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = anti AMOUNT = -3 EVENT = positive }
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = pro AMOUNT = 2 EVENT = positive }
			}
		}
	}
	# ---- the issuer backed down (2, 5) ----------------------------------------
	else_if = {
		limit = {
			OR = {
				var:nd_crisis_pending_outcome = 2
				var:nd_crisis_pending_outcome = 5
			}
		}
		if = {
			limit = { var:nd_crisis_pending_role = 1 }
			if = {
				limit = { var:nd_crisis_pending_public = 1 }
				custom_tooltip = {
					text = nd_tt_credibility_down_15
					nd_change_credibility = { AMOUNT = -15 }
				}
				nd_crisis_bluff_called = yes
				add_modifier = {
					name = nd_crisis_climbdown
					days = normal_modifier_time
					is_decaying = yes
				}
				if = {
					limit = { nd_outcome_other_is_live = yes }
					custom_tooltip = {
						text = nd_tt_outcome_hawks_disapprove
						nd_reward_hawks = { OTHER = scope:nd_outcome_other LIST = nd_lob_il GOOD = no }
					}
					nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = anti AMOUNT = -4 EVENT = positive }
					nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = pro AMOUNT = 2 EVENT = positive }
				}
			}
			else = {
				custom_tooltip = {
					text = nd_tt_credibility_down_5
					nd_change_credibility = { AMOUNT = -5 }
				}
			}
		}
		else = {
			custom_tooltip = {
				text = nd_tt_credibility_up_10
				nd_change_credibility = { AMOUNT = 10 }
			}
			add_modifier = {
				name = nd_crisis_stood_firm
				days = normal_modifier_time
				is_decaying = yes
			}
			if = {
				limit = { nd_outcome_other_is_live = yes }
				custom_tooltip = {
					text = nd_tt_outcome_hawks_approve
					nd_reward_hawks = { OTHER = scope:nd_outcome_other LIST = nd_lob_tw GOOD = yes }
				}
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = anti AMOUNT = 3 EVENT = negative }
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = pro AMOUNT = -2 EVENT = negative }
			}
		}
	}
	# ---- reciprocal stand-down (3) --------------------------------------------
	else_if = {
		limit = { var:nd_crisis_pending_outcome = 3 }
		# A public demand traded away for a stand-down it did not ask for is
		# half an audience cost; a defensive issuer got what it wanted.
		if = {
			limit = {
				var:nd_crisis_pending_role = 1
				var:nd_crisis_pending_public = 1
				var:nd_crisis_pending_defensive = 0
			}
			custom_tooltip = {
				text = nd_tt_credibility_down_5
				nd_change_credibility = { AMOUNT = -5 }
			}
			if = {
				limit = { nd_outcome_other_is_live = yes }
				nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = anti AMOUNT = -2 EVENT = positive }
			}
		}
		else = {
			custom_tooltip = {
				text = nd_tt_credibility_up_5
				nd_change_credibility = { AMOUNT = 5 }
			}
		}
		if = {
			limit = { nd_outcome_other_is_live = yes }
			nd_lobby_react = { OTHER = scope:nd_outcome_other TYPE = pro AMOUNT = 2 EVENT = positive }
		}
		add_modifier = {
			name = nd_crisis_mutual_restraint
			days = normal_modifier_time
			is_decaying = yes
		}
		custom_tooltip = {
			text = nd_tt_outcome_doves_approve
			nd_reward_doves = yes
		}
	}
	# ---- an ultimatum left to lapse (8) ---------------------------------------
	else_if = {
		limit = {
			var:nd_crisis_pending_outcome = 8
			var:nd_crisis_pending_role = 1
			var:nd_crisis_pending_public = 1
		}
		custom_tooltip = {
			text = nd_tt_credibility_down_10
			nd_change_credibility = { AMOUNT = -10 }
		}
		nd_crisis_bluff_called = yes
	}
	# ---- the cooldown, for the side that made the threat ---------------------
	if = {
		limit = { var:nd_crisis_pending_role = 1 }
		custom_tooltip = nd_tt_outcome_cooldown
	}
	nd_crisis_clear_pending = yes
}

# Country scope. Native lobby appeasement (§7): the lobbies in this country
# that are about $OTHER$. TYPE = anti | pro; AMOUNT is a signed literal.
# EVENT = positive | negative is how the event treats $OTHER$, not the sign of
# AMOUNT. It picks the vanilla "From Events" factor line, and a pro/anti pair
# reacting to one event passes the same EVENT with opposite-signed amounts, as
# vanilla's add_lobby_appeasement_from_diplomacy_unidirectional does with its
# FACTOR. The engine rejects a factor that the lobby type does not list for the
# sign of the amount (00_political_lobbies.txt): an anti lobby gains on
# _negative and loses on _positive.
nd_lobby_react = {
	every_political_lobby = {
		limit = {
			is_political_lobby_type = lobby_$TYPE$_country
			target = $OTHER$
		}
		change_appeasement = {
			amount = $AMOUNT$
			factor = appeasement_special_events_$EVENT$
		}
	}
}

# Target scope: concede (§6). The concession depends on the dispute and is
# applied while the crisis record still says what the dispute is; the close
# follows. If the concession is a play resolved for the other side, the
# engine's own on_diplo_play_back_down may close the crisis first (as a native

nd_crisis_act_yield = {
	custom_tooltip = nd_tt_act_yield
	nd_crisis_yield_lines = yes
	custom_tooltip = nd_tt_then_yield
	hidden_effect = {
		nd_crisis_save_parties = yes

			limit = {
				exists = scope:nd_issuer
				exists = scope:nd_target
				scope:nd_issuer = { nd_is_crisis_owner = yes }
			}
			scope:nd_issuer = {
				if = {
					limit = { has_variable = nd_crisis_beneficiary }
					var:nd_crisis_beneficiary ?= { save_scope_as = nd_crisis_other_side }
				}
				else = {
					save_scope_as = nd_crisis_other_side
				}
			}
			scope:nd_issuer = {
				if = {
					limit = {
						OR = {
							var:nd_crisis_dispute = 1
							var:nd_crisis_dispute = 3
						}
						exists = scope:nd_crisis_other_side
						scope:nd_target = { NOT = { has_war_with = scope:nd_crisis_other_side } }
					}
					nd_crisis_concede_play = yes
				}
				else_if = {
					limit = {
						OR = {
							var:nd_crisis_dispute = 2
							var:nd_crisis_dispute = 3
						}
						exists = scope:nd_crisis_other_side
					}
					nd_crisis_concede_war = yes
				}
				else_if = {
					limit = { var:nd_crisis_dispute = 4 }
					scope:nd_target = {
						add_modifier = {
							name = nd_crisis_programme_freeze
							years = 10
						}
					}
				}
				else = {
					scope:nd_target = {
						if = {
							limit = { has_variable = nd_readiness }
							# Down to Routine — never up to it: a Recessed target
							# already stands lower.
							if = {
								limit = { var:nd_readiness_target > 1 }
								nd_set_readiness_target_1 = yes
							}
							set_variable = { name = nd_readiness_lock_months value = nd_readiness_lock_months_value }
						}
					}
				}
			}
			nd_crisis_close = { OUTCOME = 1 }
		}
	}
}

# scope:nd_target (the yielder, a lead of the play) and
# scope:nd_crisis_other_side (the issuer, or the beneficiary it protects).
# Ends the play for the other side with resolve_play_for — the engine's own
# back-down outcome, war goals and all. VERIFY IN-GAME (§0.9 check 5): this is
# the one engine hook in the system with no vanilla script precedent.
nd_crisis_concede_play = {
	random_diplomatic_play = {
		limit = {
			is_war = no
			OR = {
				initiator_is = scope:nd_target
				target_is = scope:nd_target
			}
			any_scope_play_involved = { this = scope:nd_crisis_other_side }
		}
		save_scope_as = nd_crisis_play
	}
	if = {
		limit = { exists = scope:nd_crisis_play }
		scope:nd_crisis_play = {
			if = {
				limit = { initiator_is = scope:nd_target }
				random_scope_play_involved = {
					limit = { scope:nd_crisis_play = { target_is = PREV } }
					save_scope_as = nd_crisis_play_winner
				}
				if = {
					limit = { exists = scope:nd_crisis_play_winner }
					resolve_play_for = scope:nd_crisis_play_winner
				}
			}
			else = {
				resolve_play_for = initiator
			}
		}
	}
}

# NATIVE OBSERVATION — on_diplo_play_back_down (nuclear_crisis_on_actions.txt)
# ----------------------------------------------------------------------------
# Diplomatic-play scope; scope:actor = the country that backed down. Closes a
# play crisis with a winner the moment one side yields natively: the target (or
# its side) backing down achieves the issuer's objective (4); the issuer's side
# backing down is its failure (5).
nd_crisis_on_back_down = {
	save_scope_as = nd_bd_play
	every_scope_play_involved = {
		limit = {
			nd_in_crisis = yes
		}
		# Re-checked in the body: closing one side's crisis earlier in this
		# loop clears the other side's record too.
		if = {
			limit = { nd_in_crisis = yes }
			nd_crisis_save_parties = yes
		}
		if = {
			limit = {
				nd_in_crisis = yes
				exists = scope:nd_issuer
				exists = scope:nd_target
				scope:nd_issuer = {
					OR = {
						var:nd_crisis_dispute = 1
						var:nd_crisis_dispute = 3
					}
				}
				scope:nd_bd_play = {
					any_scope_play_involved = { this = scope:nd_target }
				}
			}
			if = {
				limit = { scope:actor = scope:nd_target }
				nd_crisis_close = { OUTCOME = 4 }
			}
			else_if = {
				limit = {
					OR = {
						scope:actor = scope:nd_issuer
						scope:nd_issuer = { var:nd_crisis_beneficiary ?= scope:actor }
					}
				}
				nd_crisis_close = { OUTCOME = 5 }
			}
		}
	}
}

# Guarantor scope. Expects scope:nd_guarantee_attacker and
# scope:nd_guarantee_beneficiary; nd_guarantee_context on the guarantor is 1
# when the beneficiary is under a nuclear ultimatum, 2 when it was struck.

# Stand with the beneficiary.
nd_guarantee_act_honour = {
	custom_tooltip = nd_tt_guarantee_honour
	custom_tooltip = {
		text = nd_tt_credibility_up_5
		nd_change_credibility = { AMOUNT = 5 }
	}
	# After a strike, with no war to answer in: a public ultimatum of our own
	# against the attacker, if neither of us is in a crisis already. Outside
	# the hidden block so the option shows what the ultimatum does
	# (nd_crisis_preview).
	if = {
		limit = {
			exists = scope:nd_guarantee_attacker
			exists = scope:nd_guarantee_beneficiary
			var:nd_guarantee_context = 2
			nd_is_armed = yes
			nd_crisis_parties_free = { TARGET = scope:nd_guarantee_attacker }
			nd_has_crisis_dispute_with = { TARGET = scope:nd_guarantee_attacker }
		}
		nd_crisis_open = { TARGET = scope:nd_guarantee_attacker PUBLIC = yes }
	}
	hidden_effect = {
		if = {
			limit = {
				exists = scope:nd_guarantee_attacker
				exists = scope:nd_guarantee_beneficiary
			}
			scope:nd_guarantee_attacker = {
				change_relations = { country = PREV value = -25 }
			}
			# Under an ultimatum: the beneficiary's crisis now has a named
			# protector (pressure −15, danger up through the tick).
			if = {
				limit = {
					var:nd_guarantee_context = 1
					scope:nd_guarantee_attacker = {
						nd_is_crisis_owner = yes
						var:nd_crisis_opponent ?= scope:nd_guarantee_beneficiary
					}
				}
				scope:nd_guarantee_attacker = {
					set_variable = { name = nd_crisis_backed value = 1 }
				}
			}
			nd_crisis_ai_review = yes
			scope:nd_guarantee_beneficiary = {
				set_variable = { name = nd_guarantee_answer value = 1 }
				trigger_event = { id = nuclear_crisis.21 }
			}
		}
	}
}

# Walk away (§6: abandoning a covered commitment shakes every beneficiary).
nd_guarantee_act_abandon = {
	custom_tooltip = nd_tt_guarantee_abandon
	custom_tooltip = {
		text = nd_tt_credibility_down_15
		nd_change_credibility = { AMOUNT = -15 }
	}
	# Abandoning a subject under our umbrella feeds its wish to be free
	# (umbrella/recessed/dead-hand spec §1.4). The umbrella stays. ROOT is the
	# guarantor (nuclear_crisis.20), and the tooltip preview needs a scope
	# that already exists.
	if = {
		limit = {
			exists = scope:nd_guarantee_beneficiary
			scope:nd_guarantee_beneficiary = { nd_under_umbrella_of = { OVERLORD = ROOT } }
		}
		scope:nd_guarantee_beneficiary = { add_liberty_desire = 10 }
	}
	add_modifier = {
		name = nd_guarantee_abandoned
		years = 5
		is_decaying = yes
	}
	hidden_effect = {
		save_scope_as = nd_guarantor_self
		if = {
			limit = {
				exists = scope:nd_guarantee_attacker
				exists = scope:nd_guarantee_beneficiary
			}
			if = {
				limit = {
					var:nd_guarantee_context = 1
					scope:nd_guarantee_attacker = {
						nd_is_crisis_owner = yes
						var:nd_crisis_opponent ?= scope:nd_guarantee_beneficiary
					}
				}
				scope:nd_guarantee_attacker = {
					set_variable = { name = nd_crisis_abandoned value = 1 }
				}
			}
			scope:nd_guarantee_beneficiary = {
				change_relations = { country = scope:nd_guarantor_self value = -30 }
				set_variable = { name = nd_guarantee_answer value = 0 }
				trigger_event = { id = nuclear_crisis.21 }
			}
			# Every other beneficiary of ours takes note.
			every_scope_treaty = {
				every_scope_article = {
					limit = {
						has_type = nuclear_guarantee
						source_country = scope:nd_guarantor_self
						NOT = { target_country = scope:nd_guarantee_beneficiary }
					}
					target_country = {
						change_relations = { country = scope:nd_guarantor_self value = -10 }
					}
				}
			}
			# ...and every subject under our umbrella (the treaty loop's
			# sibling; a subject with a treaty too is told once, above).
			every_direct_subject = {
				limit = {
					nd_under_umbrella_of = { OVERLORD = scope:nd_guarantor_self }
					NOT = { this = scope:nd_guarantee_beneficiary }
					NOT = { nd_is_guaranteed_by_treaty = { GUARANTOR = scope:nd_guarantor_self } }
				}
				change_relations = { country = scope:nd_guarantor_self value = -10 }
			}
		}
	}
}
