# ======================================================================
#  JUDICIARY — scripted effects (country scope)
#  Spec: docs/superpowers/archive/specs/2026-10-17-judiciary-design.md
#  The score lives in var:te_judicial_independence (0-100) and nowhere else,
#  so a revolution's winner inherits it (te_jud_monthly never seeds the
#  rebels, whose own score would win the merge) and te_jud_refresh rebuilds
#  the modifiers.
# ======================================================================

# An unset score starts at half the ceiling (spec § 5); history sets some
# first. Called from the monthly pulse only, where the laws' modifier sums
# are in place.
te_jud_seed_score = {
	if = {
		limit = { NOT = { has_variable = te_judicial_independence } }
		set_variable = {
			name = te_judicial_independence
			value = te_jud_seed_value
		}
	}
	set_variable = te_jud_seeded
}

# The yearly step toward the ceiling (spec § 3.1): +4 below it, -10 above it,
# stopping at it either way.
te_jud_yearly_update = {
	if = {
		limit = { has_variable = te_judicial_independence }
		if = {
			limit = { var:te_judicial_independence < te_jud_ceiling }
			change_variable = {
				name = te_judicial_independence
				add = 4
			}
			if = {
				limit = { var:te_judicial_independence > te_jud_ceiling }
				set_variable = {
					name = te_judicial_independence
					value = te_jud_ceiling
				}
			}
		}
		else_if = {
			limit = { var:te_judicial_independence > te_jud_ceiling }
			change_variable = {
				name = te_judicial_independence
				subtract = 10
			}
			if = {
				limit = { var:te_judicial_independence < te_jud_ceiling }
				set_variable = {
					name = te_judicial_independence
					value = te_jud_ceiling
				}
			}
		}
	}
}

# Spends AMOUNT of the score, floored at 0 (spec § 3.3). Callers wrap it in a
# custom_tooltip naming the cost; the modifier follows at the next monthly
# refresh.
te_jud_spend = {
	if = {
		limit = { has_variable = te_judicial_independence }
		change_variable = {
			name = te_judicial_independence
			subtract = $AMOUNT$
		}
		if = {
			limit = { var:te_judicial_independence < 0 }
			set_variable = {
				name = te_judicial_independence
				value = 0
			}
		}
	}
}

# Adds AMOUNT to the score, capped at 100 (a ruling obeyed, a promise of
# tenure, a landmark case). Callers wrap it in a custom_tooltip naming the
# gain; above the ceiling it erodes at the usual 10 a year.
te_jud_gain = {
	if = {
		limit = { has_variable = te_judicial_independence }
		change_variable = {
			name = te_judicial_independence
			add = $AMOUNT$
		}
		if = {
			limit = { var:te_judicial_independence > 100 }
			set_variable = {
				name = te_judicial_independence
				value = 100
			}
		}
	}
}

# ROOT the country (te_jud_monthly, on_monthly_pulse_country). The ONLY
# writer of the four scaled modifiers. Their multiplier variables are kept,
# never removed: the engine re-reads them.
te_jud_refresh = {
	# The score (spec § 3.2).
	if = {
		limit = { has_variable = te_judicial_independence }
		set_variable = {
			name = te_jud_modifier_mult
			value = te_jud_modifier_mult_value
		}
		if = {
			limit = { has_modifier = te_judicial_independence }
			remove_modifier = te_judicial_independence
		}
		if = {
			limit = { var:te_jud_modifier_mult > 0 }
			add_modifier = {
				name = te_judicial_independence
				multiplier = var:te_jud_modifier_mult
			}
		}
	}
	# Statutory Courts' two lines (spec § 2.3): legitimacy from votes under an
	# elected government, authority without one. Each multiplier is the law's
	# tooltip line, so 0 under any other court.
	set_variable = {
		name = te_jud_votes_mult
		value = te_jud_votes_mult_value
	}
	set_variable = {
		name = te_jud_authority_mult
		value = te_jud_authority_mult_value
	}
	if = {
		limit = { has_modifier = te_jud_votes_legitimacy }
		remove_modifier = te_jud_votes_legitimacy
	}
	if = {
		limit = { has_modifier = te_jud_unelected_authority }
		remove_modifier = te_jud_unelected_authority
	}
	if = {
		limit = {
			te_jud_elected_government = yes
			var:te_jud_votes_mult > 0
		}
		add_modifier = {
			name = te_jud_votes_legitimacy
			multiplier = var:te_jud_votes_mult
		}
	}
	else_if = {
		limit = {
			NOT = { te_jud_elected_government = yes }
			var:te_jud_authority_mult > 0
		}
		add_modifier = {
			name = te_jud_unelected_authority
			multiplier = var:te_jud_authority_mult
		}
	}
	# Algorithmic Jurisprudence's capture (spec § 2.6).
	set_variable = {
		name = te_jud_capture_mult
		value = te_jud_capture_mult_value
	}
	if = {
		limit = { has_modifier = te_jud_algorithmic_capture }
		remove_modifier = te_jud_algorithmic_capture
	}
	if = {
		limit = {
			has_law = law_type:law_algorithmic_governance
			var:te_jud_capture_mult > 0
		}
		add_modifier = {
			name = te_jud_algorithmic_capture
			multiplier = var:te_jud_capture_mult
		}
	}
}

# ROOT the country (te_jud_monthly). The amendments' standing conditions
# (phase 3, spec § 14.2): `possible` is read only when an amendment is
# added, so script lifts the Suspension of Habeas Corpus at peace (its
# tooltip line, country_amendment_lapses_at_peace_bool) and the People's
# Tribunals without a single-party state.
te_jud_amendment_upkeep = {
	if = {
		limit = {
			is_at_war = no
			modifier:country_amendment_lapses_at_peace_bool = yes
		}
		active_law:lawgroup_judiciary ?= {
			random_scope_amendment = {
				limit = { amendment_type:amendment_habeas_corpus_suspension ?= this.type }
				remove_amendment = yes
			}
		}
	}
	if = {
		limit = {
			NOT = { has_law = law_type:law_single_party_state }
			active_law:lawgroup_judiciary ?= {
				has_amendment = amendment_type:amendment_peoples_tribunals
			}
		}
		active_law:lawgroup_judiciary ?= {
			random_scope_amendment = {
				limit = { amendment_type:amendment_peoples_tribunals ?= this.type }
				remove_amendment = yes
			}
		}
	}
}

# The starting court by rule (spec § 5): history, and an old save's one-time
# seeding (te_repair_judiciary_seed). activate_law ignores unlock
# technologies, as history does.
te_jud_assign_court_by_rule = {
	if = {
		limit = {
			OR = {
				has_law = law_type:law_theocracy
				AND = {
					te_jud_is_islamic_state = yes
					OR = {
						has_law_or_variant = law_type:law_state_religion
						has_law = law_type:law_freedom_of_conscience
					}
				}
			}
		}
		activate_law = law_type:law_religious_courts
	}
	else_if = {
		limit = {
			OR = {
				has_law = law_type:law_landed_voting
				has_law = law_type:law_wealth_voting
				has_law = law_type:law_census_voting
				has_law = law_type:law_universal_suffrage
			}
		}
		activate_law = law_type:law_statutory_courts
	}
	else = {
		activate_law = law_type:law_executive_tribunals
	}
}

# Once a game (spec § 9; CLAUDE.md "In-game checks"): does
# currently_enacting_law resolve at on_law_enactment_started? Only a country
# whose court challenges, and only while an enactment runs, so a missing link
# can't read as OK. No line at all after such an enactment means the hook's
# ROOT isn't the country.
te_jud_selfcheck_enacting_scope = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jud_selfcheck_enacting_done }
			enacting_any_law = yes
			modifier:country_constitutional_law_enactment_success_add < 0
			NOT = { te_jud_paper_court = yes }
		}
		set_global_variable = te_jud_selfcheck_enacting_done
		if = {
			limit = { exists = currently_enacting_law }
			debug_log = "TE_SELF_CHECK OK judiciary_enacting_law_scope country=[THIS.GetCountry.GetNameNoFormatting]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN judiciary_enacting_law_scope country=[THIS.GetCountry.GetNameNoFormatting] currently_enacting_law missing at on_law_enactment_started"
		}
	}
}

# ======================================================================
#  Phase 2: defying the court (spec § 14.1)
#  A Forceful Legislation option under a court that reviews forced laws
#  opens a challenge (te_jud_open_challenge). The monthly pulse files it
#  once the enactment is over and the law is in force
#  (te_jud_challenge_monthly), and the court rules 180 days later
#  (te_judiciary_events.2): it strikes the law down when the score is at
#  least te_jud_strike_threshold, and the government complies, defies or
#  packs the court (te_judiciary_events.3).
#  Variables (country):
#    te_jud_challenge_pending   a forced enactment is under way
#    te_jud_challenge_law       its law type
#    te_jud_replaced_law        the law type it replaces (compliance
#                               restores it)
#    te_jud_challenge_tier      1-3, the option taken
#    te_jud_case_filed          a case is before the court (270 days, a
#                               backstop for a revolution that drops the
#                               scheduled ruling)
#    te_jud_case_law, te_jud_case_replaced, te_jud_case_tier
#                               the filed case, copied at filing so a
#                               second forced enactment can't overwrite it
# ======================================================================

# Called from minor_events_timelineextended.2's three bypass options, inside a
# custom_tooltip naming the review. TIER is the option (1 Expedite, 2 Push
# decisively, 3 Force it through). currently_enacting_law is the law being
# forced, and its group's active law the one it would replace.
te_jud_open_challenge = {
	if = {
		limit = {
			te_jud_court_can_strike_laws = yes
			exists = currently_enacting_law
		}
		currently_enacting_law = {
			law_type = { save_scope_as = te_jud_forced_type }
			currently_active_law_in_group ?= {
				law_type = { save_scope_as = te_jud_replaced_type }
			}
		}
		set_variable = {
			name = te_jud_challenge_law
			value = scope:te_jud_forced_type
		}
		if = {
			limit = { exists = scope:te_jud_replaced_type }
			set_variable = {
				name = te_jud_replaced_law
				value = scope:te_jud_replaced_type
			}
		}
		else_if = {
			limit = { has_variable = te_jud_replaced_law }
			remove_variable = te_jud_replaced_law
		}
		set_variable = {
			name = te_jud_challenge_tier
			value = $TIER$
		}
		set_variable = te_jud_challenge_pending
		te_jud_selfcheck_law_type_variable = yes
	}
}

# Once a game (CLAUDE.md "In-game checks"): can a country variable hold a law
# type? The challenge files only when has_law reads the stored type, and
# compliance restores the replaced law from one. Only where the forced law's
# type was found, so a missing scope can't read as OK.
te_jud_selfcheck_law_type_variable = {
	if = {
		limit = {
			NOT = { has_global_variable = te_jud_selfcheck_law_type_done }
			exists = scope:te_jud_forced_type
		}
		set_global_variable = te_jud_selfcheck_law_type_done
		if = {
			limit = {
				has_variable = te_jud_challenge_law
				var:te_jud_challenge_law ?= { is_same_law_group_as = scope:te_jud_forced_type }
			}
			debug_log = "TE_SELF_CHECK OK judiciary_law_type_variable country=[THIS.GetCountry.GetNameNoFormatting]"
		}
		else = {
			debug_log = "TE_SELF_CHECK BROKEN judiciary_law_type_variable country=[THIS.GetCountry.GetNameNoFormatting] a variable set to a law type does not read back as one"
		}
	}
}

# ROOT the country (te_jud_monthly). Once the forced enactment is over: a law
# that passed under a reviewing court goes to it, one case at a time; one
# that failed is forgotten.
te_jud_challenge_monthly = {
	if = {
		limit = {
			has_variable = te_jud_challenge_pending
			NOT = { enacting_any_law = yes }
		}
		remove_variable = te_jud_challenge_pending
		if = {
			limit = {
				NOT = { has_variable = te_jud_case_filed }
				has_variable = te_jud_challenge_law
				has_law = var:te_jud_challenge_law
				te_jud_court_can_strike_laws = yes
			}
			set_variable = { name = te_jud_case_filed days = 270 }
			set_variable = {
				name = te_jud_case_law
				value = var:te_jud_challenge_law
			}
			if = {
				limit = { has_variable = te_jud_replaced_law }
				set_variable = {
					name = te_jud_case_replaced
					value = var:te_jud_replaced_law
				}
			}
			else_if = {
				limit = { has_variable = te_jud_case_replaced }
				remove_variable = te_jud_case_replaced
			}
			set_variable = {
				name = te_jud_case_tier
				value = var:te_jud_challenge_tier
			}
			trigger_event = { id = te_judiciary_events.2 days = 180 }
		}
	}
}

# The case is decided, or can no longer be (te_judiciary_events.2-.4).
te_jud_close_case = {
	if = {
		limit = { has_variable = te_jud_case_filed }
		remove_variable = te_jud_case_filed
	}
	if = {
		limit = { has_variable = te_jud_case_law }
		remove_variable = te_jud_case_law
	}
	if = {
		limit = { has_variable = te_jud_case_replaced }
		remove_variable = te_jud_case_replaced
	}
	if = {
		limit = { has_variable = te_jud_case_tier }
		remove_variable = te_jud_case_tier
	}
}

# Packing the court (te_judiciary_events.3, a landmark case's answer): the
# authority it takes, ten years of a court that challenges nothing and is
# trusted less, and a year later the choice of what to do with the bench
# (te_judiciary_events.5). The score's cost is the caller's, in its own
# custom_tooltip.
te_jud_pack_court = {
	add_modifier = { name = te_jud_packing_cost years = 5 is_decaying = yes }
	if = {
		limit = { has_modifier = te_jud_court_packed }
		remove_modifier = te_jud_court_packed
	}
	add_modifier = { name = te_jud_court_packed years = 10 }
	hidden_effect = {
		trigger_event = { id = te_judiciary_events.5 days = 365 }
	}
}
