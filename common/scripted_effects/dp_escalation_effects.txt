# ============================================================================
# Diplomatic-play weekly escalation apply effect
# ============================================================================
# The mod's extra escalation, on top of vanilla's point a day. Each country in a
# play contributes its own weekly figure for its role (aggressor, defender or
# participant: dp_escalation_script_values.txt). Roles are mutually exclusive
# per (country, play): the initiator is the aggressor, the target the defender,
# every other committed member a participant.
#
# Only on the days the play escalates (owner, 2026-11-11). Vanilla pauses a
# play's escalation for some days after each demand or sway (5 a maneuver, up
# to 20), and until then the mod's weekly add ran straight through the pause.
# Script can't read the pause, but it can see what it does: the engine adds
# DIPLOMATIC_PLAY_ESCALATION_DAILY (1) on each day the play isn't paused, so
# the play's own rise since the last tick, net of what the mod added then, is
# the number of days it escalated. Each tick adds the week's figure times those
# days over 7, the fraction carried to the next tick (te_dp_esc_carry), so a
# play paused all week gets nothing and one paused for three of seven days gets
# four-sevenths. The rise is read with te_dp_escalation_now (a ladder of
# comparisons: the engine has no number for it) and kept on the play as
# te_dp_esc_mark. Capped at 10 days, the longest gap between two ticks, so an
# event's add_escalation is never read as a fortnight of escalation. A play's
# first tick has no mark: it reads the days since it began (its escalation, at
# most 7), which is right for a play that opened at 0.
#
# One tick per play, the initiator's (every play has one, and its weekly
# schedule runs as long as it exists), with every role's figure summed: a
# second tick within the week would split the days between two marks. The
# target is found among the involved by target_is, as nd_crisis_concede_play
# does (nuclear_crisis_effects.txt).
#
# Called from dp_escalation_events.1 (events/dp_escalation_events.txt) and
# inline at week 0 from st_dp_escalation_weekly_schedule (extra_on_actions.txt).
#
# Why a while-loop instead of `add_escalation = <script_value>`: vanilla's
# only `add_escalation` usage is a literal integer (italian_unification.txt:
# 613) and the engine doc lists the signature as `add_escalation = integer`,
# so feeding a script value risks a silent no-op. Looping `add_escalation = 1`
# N times uses only the literal form vanilla itself uses (`count =` takes a
# local variable, as resettlement_effects.txt's does); engine cap is 1000
# iterations, well above the 0-14 a tick adds.
#
# Unverified in game: that a play keeps a variable, and that the ladder reads.
# The self-check dp_esc_pause logs which (te_dp_esc_pause_self_check, below),
# and a ladder that reads nothing falls back to the flat weekly add by itself
# (te_dp_esc_flat): docs/testing/diplomatic-play-escalation-playtest.md lists both.
# ============================================================================

st_dp_country_apply_weekly_escalation = {
	if = {
		limit = { is_diplomatic_play_initiator = yes }
		save_temporary_scope_as = mover
		every_diplomatic_play = {
			limit = {
				initiator_is = scope:mover
				is_war = no
			}
			st_dp_play_apply_weekly_escalation = yes
		}
	}
}

# Diplomatic-play scope, scope:mover its initiator. The week's extra from every
# role, times the days the play escalated since the last tick over 7.
st_dp_play_apply_weekly_escalation = {
	save_temporary_scope_as = te_dp_play
	# The week's figure: the initiator's as aggressor, the target's as defender,
	# and each other committed member's as a participant.
	set_local_variable = { name = te_dp_week value = scope:mover.country_aggressor_dp_escalation_weekly_total }
	every_scope_play_involved = {
		limit = {
			NOT = { this = scope:mover }
		}
		save_temporary_scope_as = te_dp_member
		if = {
			limit = {
				scope:te_dp_play = { target_is = scope:te_dp_member }
			}
			change_local_variable = { name = te_dp_week add = country_defender_dp_escalation_weekly_total }
		}
		else_if = {
			limit = { is_diplomatic_play_committed_participant = yes }
			change_local_variable = { name = te_dp_week add = country_participant_dp_escalation_weekly_total }
		}
	}
	# The days the engine escalated it since the last tick (see the header).
	set_local_variable = { name = te_dp_now value = te_dp_escalation_now }
	if = {
		limit = { has_global_variable = te_dp_esc_flat }
		set_local_variable = { name = te_dp_days value = 7 }
	}
	else_if = {
		limit = { has_variable = te_dp_esc_mark }
		set_local_variable = {
			name = te_dp_days
			value = {
				value = local_var:te_dp_now
				subtract = var:te_dp_esc_mark
				min = 0
				max = 10
			}
		}
	}
	else = {
		set_local_variable = {
			name = te_dp_days
			value = {
				value = local_var:te_dp_now
				max = 7
			}
		}
	}
	te_dp_esc_pause_self_check = yes
	# The extra for those days, and what was left over last time.
	set_local_variable = {
		name = te_dp_add
		value = {
			value = local_var:te_dp_week
			multiply = local_var:te_dp_days
			divide = 7
		}
	}
	if = {
		limit = { has_variable = te_dp_esc_carry }
		change_local_variable = { name = te_dp_add add = var:te_dp_esc_carry }
	}
	set_local_variable = {
		name = te_dp_whole
		value = {
			value = local_var:te_dp_add
			floor = yes
		}
	}
	set_variable = {
		name = te_dp_esc_carry
		value = {
			value = local_var:te_dp_add
			subtract = local_var:te_dp_whole
		}
	}
	while = {
		count = local_var:te_dp_whole
		add_escalation = 1
	}
	# Where the play stands after this tick: what was read plus what was added
	# (not read again, in case an add lands after the effect).
	set_variable = {
		name = te_dp_esc_mark
		value = {
			value = local_var:te_dp_now
			add = local_var:te_dp_whole
			max = 100
		}
	}
}

# TE_SELF_CHECK dp_esc_pause (docs/testing/diplomatic-play-escalation-playtest.md). Diplomatic-play scope, from
# st_dp_play_apply_weekly_escalation, with local_var:te_dp_now (the ladder's reading) and local_var:te_dp_days set. Two
# engine facts nobody has seen: that a play keeps a variable from one tick to the next, and that `escalation >= N`
# reads in a script value. Once a game: OK at the first tick that finds a mark from the last one and reads the play
# above 0 (both hold). BROKEN mark after 20 ticks on plays with no mark and none with one (plays keep no variable: every
# tick is a first tick, days at most 7, about the old flat add once a play passes 7; fallback: keep the mark on the
# initiator). BROKEN ladder after 20 ticks with a mark and none above 0 (the comparator doesn't read: every play would
# get nothing), which also sets te_dp_esc_flat, the flat weekly add of before, from the next tick on. Counts are globals.
te_dp_esc_pause_self_check = {
	if = {
		limit = {
			NOT = { has_global_variable = te_dp_esc_sc_done }
		}
		if = {
			limit = {
				NOT = { has_global_variable = te_dp_esc_sc_marked }
			}
			set_global_variable = { name = te_dp_esc_sc_marked value = 0 }
			set_global_variable = { name = te_dp_esc_sc_unmarked value = 0 }
		}
		if = {
			limit = { has_variable = te_dp_esc_mark }
			change_global_variable = { name = te_dp_esc_sc_marked add = 1 }
			if = {
				limit = { local_var:te_dp_now > 0 }
				set_global_variable = te_dp_esc_sc_done
				# The numbers through globals, as te_jmap_hq_selfcheck prints its counts.
				set_global_variable = { name = te_dp_esc_sc_now value = local_var:te_dp_now }
				set_global_variable = { name = te_dp_esc_sc_days value = local_var:te_dp_days }
				set_global_variable = { name = te_dp_esc_sc_week value = local_var:te_dp_week }
				set_global_variable = { name = te_dp_esc_sc_mark value = var:te_dp_esc_mark }
				debug_log = "TE_SELF_CHECK OK dp_esc_pause now=[GetGlobalVariable('te_dp_esc_sc_now').GetValue|0] mark=[GetGlobalVariable('te_dp_esc_sc_mark').GetValue|0] days=[GetGlobalVariable('te_dp_esc_sc_days').GetValue|0] week=[GetGlobalVariable('te_dp_esc_sc_week').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString] (a play kept its mark and the escalation ladder read it: days is the engine's rise since the last tick, 7 a week unpaused)"
				remove_global_variable = te_dp_esc_sc_now
				remove_global_variable = te_dp_esc_sc_days
				remove_global_variable = te_dp_esc_sc_week
				remove_global_variable = te_dp_esc_sc_mark
			}
			else_if = {
				limit = { global_var:te_dp_esc_sc_marked >= 20 }
				set_global_variable = te_dp_esc_sc_done
				set_global_variable = te_dp_esc_flat
				debug_log = "TE_SELF_CHECK BROKEN dp_esc_pause ladder marked_ticks=[GetGlobalVariable('te_dp_esc_sc_marked').GetValue|0] (twenty ticks on plays with a mark and the escalation ladder read 0 every time: escalation >= N doesn't read in a script value; the flat weekly add is back, te_dp_esc_flat) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
		else = {
			change_global_variable = { name = te_dp_esc_sc_unmarked add = 1 }
			if = {
				limit = {
					global_var:te_dp_esc_sc_unmarked >= 20
					global_var:te_dp_esc_sc_marked = 0
				}
				set_global_variable = te_dp_esc_sc_done
				debug_log = "TE_SELF_CHECK BROKEN dp_esc_pause mark unmarked_ticks=[GetGlobalVariable('te_dp_esc_sc_unmarked').GetValue|0] (twenty ticks found no mark on their play and none ever did: a diplomatic play keeps no variable, so every tick reads the play as new; take the fallback in docs/testing/diplomatic-play-escalation-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
	}
}
