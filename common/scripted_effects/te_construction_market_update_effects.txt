# Country scope, market games only (te_construction_market_on_border_changes):
# re-place the sites after the map changes. te_cm_placing_site lets the site's
# can_build_government accept these create_building calls (see
# common/buildings/te_construction_market_site.txt).
te_construction_market_country_update_sites = {
	set_global_variable = {
		name = te_cm_placing_site
		value = 1
	}
	every_state = {
		if = {
			limit = {
				any_scope_building = {
					is_under_construction = yes
				}
			}
			te_construction_market_set_specified_level = {
				SPEC_LEVEL = 1
				SPEC_TYPE = te_construction_market_site
			}
		}
		te_construction_market_state_on_end = yes
	}
	if = {
		limit = {
			NOT = {
				any_scope_building = {
					is_building_type = te_construction_market_site
				}
			}
		}
		capital = {
			create_building = {
				building = te_construction_market_site
				level = 1
			}
		}
	}
	remove_global_variable = te_cm_placing_site
}


te_construction_market_country_update_data = {
	te_construction_market_country_update_private_share = yes
	te_construction_market_apply_per_site_modifier = yes
}


te_construction_market_country_update_private_share = {
	remove_modifier = te_construction_market_private_share_modifier
	add_modifier = {
		name = te_construction_market_private_share_modifier
		multiplier = te_construction_market_private_share_target
	}
}


te_construction_market_apply_per_site_modifier = {
	# Per-site capacity modifier (te_construction_market_capacity_modifier
	# = `building_throughput_add = 1`), scaled by the staffing-adjusted
	# per-site target.
	#
	# Why the country-variable handoff: `add_modifier = { multiplier = X }`
	# inside `every_scope_building` evaluates X in the OUTER (country)
	# scope, not the building scope of the iteration. The compensator
	# in te_construction_market_employment_adjusted_per_site reads
	# `occupancy`, which is undefined at country scope — the else branch
	# fires (multiply 10), modifier inflates to +1000%+.
	#
	# Several attempted workarounds (save_temporary_scope_value_as,
	# save_temporary_scope_as + scope:X.<script_value>, set_variable on
	# building + scope:X.var:Y) all silently fell back to multiplier=1.
	# Buildings appear not to support variables; saved-scope traversal
	# only works for value-triggers, not script values.
	#
	# Working pattern: write the building-scope-evaluated value into a
	# country variable (via `owner = { set_variable { value = prev.<script_value> } }`),
	# then have add_modifier read `owner.var:te_cm_capacity_mult`. The
	# country-scope variable lookup works even from add_modifier's
	# outer-scope evaluation context.
	#
	# The variable MUST persist past this effect. The engine stores the
	# `multiplier` EXPRESSION with the modifier and re-evaluates it on
	# later ticks (decay / tooltip / recalculation), not just at add time.
	# A `remove_variable` at the end of this effect therefore makes every
	# later re-evaluation read `'none'` (`jomini_scriptvalue.cpp` spam,
	# per-site scaling silently lost) until the next weekly pulse
	# re-applies it. Nothing needs cleaning up: the next call simply
	# overwrites the variable. See the "Variables backing
	# `add_modifier { multiplier = ... }` must persist beyond the call
	# site" bullet in docs/guides/scripting_best_practices.md.
	#
	# Multi-site caveat: one country variable is shared across all of the
	# country's sites, so between pulses every site's modifier re-evaluates
	# against the LAST site's value. That is pre-existing and unchanged by
	# the persistence fix (before it, those re-evaluations returned
	# `'none'`, which is strictly worse). A per-site redesign is separate
	# work and is deliberately not attempted here.
	#
	# Clear at start each call so a stale value never leaks across ticks
	# or partial iterations. Harmless for the persistence rule above: it is
	# re-set before `add_modifier` in the same block.
	remove_variable = te_cm_capacity_mult
	every_scope_building = {
		limit = {
			is_building_type = te_construction_market_site
		}
		remove_modifier = te_construction_market_capacity_modifier
		# Compute in building scope (prev = this building), write to
		# country variable on the owner.
		owner = {
			set_variable = {
				name = te_cm_capacity_mult
				value = prev.te_construction_market_employment_adjusted_per_site
			}
		}
		add_modifier = {
			name = te_construction_market_capacity_modifier
			multiplier = owner.var:te_cm_capacity_mult
		}
	}
}
