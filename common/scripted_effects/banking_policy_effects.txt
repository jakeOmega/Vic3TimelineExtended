# Banking policy dashboard — shared policy effects.
#
# Each `banking_effect_<button>` holds the body that used to live inline in the
# matching scripted_button's `effect` block. Both the journal-entry button and the
# dashboard scripted GUI call these, so the cost and the applied modifier cannot
# drift apart between the two surfaces.

# The "Raise Policy Rate" pair used to open this file. It is deleted
# (monetary_policy_design.md §18): the policy rate is a country variable the
# player or the delegated central bank sets directly, not an on/off intervention.

	te_history_record_banking_marker = { MARK = on_cb_open_market_ops }
	je:je_banking_cycle = { add_modifier = { name = banking_open_market_ops } }
	add_treasury = banking_omo_activation_cost
}

# Ninety days, then the banks reopen on their own: the modifier is TIMED, so
# the Disable action only ever ends a holiday early, and the history chart gets
# no "withdrawn" marker when one simply expires. The one-shot halving of a
# negative momentum is the run stopping; the cooldown is on the country, not the
# modifier, so ending the holiday early does not reset it.
banking_effect_cb_bank_holiday = {
	te_history_record_banking_marker = { MARK = on_cb_bank_holiday }
	je:je_banking_cycle = { add_modifier = { name = banking_bank_holiday days = 90 } }
	set_variable = { name = banking_bank_holiday_cooldown days = 1825 }
	custom_tooltip = {
		text = banking_bank_holiday_momentum_tt
		if = {
			limit = { var:finance_cycle_momentum < 0 }
			change_variable = { name = finance_cycle_momentum multiply = 0.5 }
		}
	}
	# Depositors who cannot reach their money.
	add_radicals = { value = banking_small_radicals_value strata = middle }
}

banking_effect_cb_disable_bank_holiday = {
	te_history_record_banking_marker = { MARK = off_cb_bank_holiday }
	je:je_banking_cycle = { remove_modifier = banking_bank_holiday }
}
