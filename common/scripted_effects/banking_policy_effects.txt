# Banking policy dashboard — shared policy effects.
#
# Each `banking_effect_<button>` holds the body that used to live inline in the
# matching scripted_button's `effect` block. Both the journal-entry button and the
# dashboard scripted GUI call these, so the cost and the applied modifier cannot
# drift apart between the two surfaces.

banking_effect_cb_policy_rate_hike = {
	# Spend 2 points (via the modifier) and cool the cycle
	je:je_banking_cycle = { add_modifier = { name = banking_policy_rate_hike } }
	add_radicals = {
		value = banking_medium_radicals_value
	}
}

banking_effect_cb_disable_policy_rate_hike = {
	je:je_banking_cycle = { remove_modifier = banking_policy_rate_hike }
	add_loyalists = {
		value = banking_small_radicals_value
	}
}
