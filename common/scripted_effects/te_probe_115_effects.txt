# ============================================================================
# te_probe_115: the vanilla 1.15 migration's engine probes, the steps (2026-11-05)
# ============================================================================
# What it answers: the open engine questions of docs/audits/vanilla_1_15_prep.md § 2 that script can read with no
# player watching, each as literal `TE_PROBE_115 <name> ...` lines in debug.log (which branch fired is the answer):
#   C7   scaling_modifiers_while_active: a value from a country-only read, a negative value, 0, how soon a
#        modifier: read sees a change, whether has_modifier sees the modifier on the entry
#   C19  a quoted $ARG$ inside a debug_log string (te_probe_115_quoted_arg_effect)
#   C20  fire_only_once (per country? a failed trigger?) and fire_only_once_globally (events te_probe_115.10, .11)
#   C21  scope:journal_entry in an entry's pulse and outcome effect; disable_evaluation; a concluded entry's reads
#   C3   which of two outcomes true on the same tick runs (the entry's outcomes; te_probe_115.20 reports)
# The entry (common/journal_entries/te_probe_115_probes.txt) runs on one AI major power once a game and calls
# te_probe_115_weekly_effect from its weekly pulse; each step runs once, a week or more after the last, and step 9
# makes both of the entry's C3 outcomes true together, which ends it. te_probe_115.20 removes everything the probe
# set, except the global te_probe_115_claimed, which keeps it from running again in this game.
#
# Delete once answered (see docs/testing/vanilla-1-15-playtest.md).
#
# C7's modifiers are the nuclear deterrent's display-only terms (common/static_modifiers/nuclear_deterrence_modifiers.txt):
# nd_readiness_terms_1 carries only country_nd_crew_strain_monthly_add = -6, nd_authority_terms_2 carries
# country_nd_launch_credit_add = 0.25 (and three display bools). Their types are script_only and nothing reads them
# (common/modifier_type_definitions/nuclear_deterrence_display_types.txt), so on the probe country they do nothing,
# and no other source adds to them, so a modifier: read is the probe's alone.
#
# THIS = ROOT = the probe country throughout, unless a block says otherwise.

te_probe_115_weekly_effect = {
	if = {
		limit = {
			has_variable = te_probe_115_owner
			NOT = { has_variable = te_probe_115_c7_sampling }
			NOT = { has_variable = te_probe_115_c3_go }
		}
		change_variable = { name = te_probe_115_step add = 1 }
		if = {
			limit = { var:te_probe_115_step = 1 }
			# A week's wait, so the entry has been updated (every 4 days, JOURNAL_ENTRY_UPDATE_ACTIVE) before any read.
			debug_log = "TE_PROBE_115 step 1 (waiting a week for the entry's first updates)"
		}
		else_if = {
			limit = { var:te_probe_115_step = 2 }
			te_probe_115_step_2_effect = yes
		}
		else_if = {
			limit = { var:te_probe_115_step = 3 }
			debug_log = "TE_PROBE_115 step 3 (C7 negative value)"
			te_probe_115_c7_start_effect = { PHASE = 2 SCALE = -1 }
		}
		else_if = {
			limit = { var:te_probe_115_step = 4 }
			debug_log = "TE_PROBE_115 step 4 (C7 zero after a non-zero value)"
			te_probe_115_c7_start_effect = { PHASE = 3 SCALE = 0 }
		}
		else_if = {
			limit = { var:te_probe_115_step = 5 }
			te_probe_115_c20_try_without_gate_effect = yes
		}
		else_if = {
			limit = { var:te_probe_115_step = 6 }
			te_probe_115_c20_try_with_gate_effect = yes
		}
		else_if = {
			limit = { var:te_probe_115_step = 7 }
			te_probe_115_c20_try_again_effect = yes
		}
		else_if = {
			limit = { var:te_probe_115_step = 8 }
			te_probe_115_c20_report_effect = yes
		}
		else_if = {
			limit = { var:te_probe_115_step = 9 }
			# C3: both visible outcomes test this one variable, so they come true on the same update.
			debug_log = "TE_PROBE_115 step 9 (C3: both outcomes come true at the entry's next update)"
			set_variable = te_probe_115_c3_go
		}
	}
}

# ---- Step 2: C21's pulse scope, C19, C7's first reads and its positive value ------------------------------------
te_probe_115_step_2_effect = {
	debug_log = "TE_PROBE_115 step 2 (C21 pulse scope, C19, C7 first reads, C7 positive value)"
	if = {
		limit = { exists = scope:journal_entry }
		debug_log = "TE_PROBE_115 c21_pulse_scope OK (scope:journal_entry is set in an entry's on_weekly_pulse and the scripted effects it calls)"
	}
	else = {
		debug_log = "TE_PROBE_115 c21_pulse_scope BROKEN (scope:journal_entry is not set in an entry's on_weekly_pulse: reach the entry with je:<key>)"
	}
	te_probe_115_quoted_arg_effect = { N = 7 WORD = probe }
	te_probe_115_c7_first_reads_effect = yes
	te_probe_115_c7_start_effect = { PHASE = 1 SCALE = 1 }
}

# ---- C19: a quoted $ARG$ in debug_log -----------------------------------------------------------------------------
# On 1.14.5 "(kind $KIND$)" logged "(kind1$)": the character before the argument went, the value went in and the
# closing dollar stayed (scripting_best_practices.md, "Runtime Debugging with debug_log"). Called once, with N = 7 and
# WORD = probe. Line a puts each argument after an equals sign, line b after a space, as 1.14.5's case had it.
# Substituted, they read "number=7 word=probe end" and "(kind probe)"; garbled as on 1.14.5, "number7" and
# "wordprobe" each with a dollar sign after, and "(kindprobe" with one; left alone, the argument names in dollar
# signs. The branch line says what the argument was. The two quoted lines are the only TE_PROBE_115 lines that are
# not literal (test_te_probe_115.py).
te_probe_115_quoted_arg_effect = {
	debug_log = "TE_PROBE_115 c19_quoted_arg_a number=$N$ word=$WORD$ end"
	debug_log = "TE_PROBE_115 c19_quoted_arg_b (kind $WORD$)"
	set_local_variable = { name = te_probe_115_c19_n value = $N$ }
	if = {
		limit = { local_var:te_probe_115_c19_n = 7 }
		debug_log = "TE_PROBE_115 c19_branch number=7 (the argument reached the effect as 7: compare the two quoted lines)"
	}
	else = {
		debug_log = "TE_PROBE_115 c19_branch number=other (the argument did not reach the effect as 7)"
	}
}

# ---- C7: scaling_modifiers_while_active ---------------------------------------------------------------------------
# The entry's control block scales nd_authority_terms_2 by a constant 4, so country_nd_launch_credit_add should read
# 4 x 0.25 = 1. Its test block scales nd_readiness_terms_1 by var:te_probe_115_c7_scale behind is_ai, a trigger only
# a country can answer, so country_nd_crew_strain_monthly_add should read scale x -6, and 0 if the value's root is not
# the owner. The scale is 0 from activation to step 2.
te_probe_115_c7_first_reads_effect = {
	if = {
		limit = {
			modifier:country_nd_launch_credit_add > 0.99
			modifier:country_nd_launch_credit_add < 1.01
		}
		debug_log = "TE_PROBE_115 c7_control OK (a constant value of 4 scales the modifier: launch credit reads 4 x 0.25)"
	}
	else_if = {
		limit = {
			modifier:country_nd_launch_credit_add > 0.24
			modifier:country_nd_launch_credit_add < 0.26
		}
		debug_log = "TE_PROBE_115 c7_control UNSCALED (the modifier applies at its own size: the value is ignored)"
	}
	else_if = {
		limit = {
			modifier:country_nd_launch_credit_add > -0.01
			modifier:country_nd_launch_credit_add < 0.01
		}
		debug_log = "TE_PROBE_115 c7_control ABSENT (no scaling modifier reaches the country's modifier: reads 0)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_control OTHER (launch credit reads neither 0, 0.25 nor 1)"
	}
	if = {
		limit = { je:te_probe_115 ?= { has_modifier = nd_authority_terms_2 } }
		debug_log = "TE_PROBE_115 c7_has_modifier_entry YES (has_modifier on the entry sees a scaling modifier)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_has_modifier_entry NO (has_modifier on the entry does not see a scaling modifier at value 4)"
	}
	if = {
		limit = { has_modifier = nd_authority_terms_2 }
		debug_log = "TE_PROBE_115 c7_has_modifier_country YES (has_modifier on the country sees the entry's scaling modifier)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_has_modifier_country NO (has_modifier on the country does not see the entry's scaling modifier)"
	}
	if = {
		limit = {
			modifier:country_nd_crew_strain_monthly_add > -0.01
			modifier:country_nd_crew_strain_monthly_add < 0.01
		}
		debug_log = "TE_PROBE_115 c7_zero_first_read reads 0 (value 0 since activation)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_zero_first_read reads non-zero (value 0 since activation, yet crew strain is not 0)"
	}
	# Whether a value of 0 keeps the test modifier on the entry (read against c7_has_modifier_entry above).
	if = {
		limit = { je:te_probe_115 ?= { has_modifier = nd_readiness_terms_1 } }
		debug_log = "TE_PROBE_115 c7_zero_first PRESENT (a scaling modifier at value 0 since activation is on the entry)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_zero_first ABSENT (a scaling modifier at value 0 since activation is not on the entry)"
	}
}

# Sets the test block's scale and samples the read daily (te_probe_115.1) until it shows, for up to 14 days. The
# weekly steps wait while te_probe_115_c7_sampling is set. PHASE: 1 positive, 2 negative, 3 zero.
te_probe_115_c7_start_effect = {
	set_variable = { name = te_probe_115_c7_scale value = $SCALE$ }
	set_variable = { name = te_probe_115_c7_phase value = $PHASE$ }
	set_variable = { name = te_probe_115_c7_days value = 0 }
	set_variable = te_probe_115_c7_sampling
	te_probe_115_c7_check_effect = yes
	if = {
		limit = { has_variable = te_probe_115_c7_sampling }
		trigger_event = { id = te_probe_115.1 days = 1 }
	}
}

# One sample: logs and stops when the read shows the phase's scale, or after 14 days.
te_probe_115_c7_check_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					var:te_probe_115_c7_phase = 1
					modifier:country_nd_crew_strain_monthly_add < -5.99
					modifier:country_nd_crew_strain_monthly_add > -6.01
				}
				AND = {
					var:te_probe_115_c7_phase = 2
					modifier:country_nd_crew_strain_monthly_add > 5.99
					modifier:country_nd_crew_strain_monthly_add < 6.01
				}
				AND = {
					var:te_probe_115_c7_phase = 3
					modifier:country_nd_crew_strain_monthly_add > -0.01
					modifier:country_nd_crew_strain_monthly_add < 0.01
				}
			}
		}
		te_probe_115_c7_seen_log_effect = yes
		te_probe_115_c7_end_phase_effect = yes
	}
	else_if = {
		limit = { var:te_probe_115_c7_days >= 14 }
		te_probe_115_c7_never_log_effect = yes
		te_probe_115_c7_end_phase_effect = yes
	}
}

te_probe_115_c7_end_phase_effect = {
	if = {
		limit = { var:te_probe_115_c7_phase = 3 }
		if = {
			limit = { je:te_probe_115 ?= { has_modifier = nd_readiness_terms_1 } }
			debug_log = "TE_PROBE_115 c7_zero_after PRESENT (a scaling modifier set back to 0 stays on the entry)"
		}
		else = {
			debug_log = "TE_PROBE_115 c7_zero_after ABSENT (a scaling modifier set back to 0 leaves the entry)"
		}
	}
	remove_variable = te_probe_115_c7_sampling
}

# The read never showed the phase's scale: which phase, then what it reads.
te_probe_115_c7_never_log_effect = {
	if = {
		limit = { var:te_probe_115_c7_phase = 1 }
		debug_log = "TE_PROBE_115 c7_positive NOT seen within 14 days (value 1 from a country-only read never read -6)"
	}
	else_if = {
		limit = { var:te_probe_115_c7_phase = 2 }
		debug_log = "TE_PROBE_115 c7_negative NOT seen within 14 days (value -1 never read +6)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_zero NOT seen within 14 days (value 0 never read 0 after -1)"
	}
	if = {
		limit = {
			modifier:country_nd_crew_strain_monthly_add > -0.01
			modifier:country_nd_crew_strain_monthly_add < 0.01
		}
		debug_log = "TE_PROBE_115 c7_read 0 (the test modifier contributes nothing)"
	}
	else_if = {
		limit = {
			modifier:country_nd_crew_strain_monthly_add < -5.99
			modifier:country_nd_crew_strain_monthly_add > -6.01
		}
		debug_log = "TE_PROBE_115 c7_read -6 (the test modifier at its own size, or still at value 1)"
	}
	else_if = {
		limit = {
			modifier:country_nd_crew_strain_monthly_add > 5.99
			modifier:country_nd_crew_strain_monthly_add < 6.01
		}
		debug_log = "TE_PROBE_115 c7_read +6 (the test modifier still at value -1)"
	}
	else = {
		debug_log = "TE_PROBE_115 c7_read other (crew strain reads neither 0, -6 nor +6)"
	}
}

# ---- C20: fire_only_once and fire_only_once_globally --------------------------------------------------------------
# te_probe_115.10 is fire_only_once with the trigger has_variable = te_probe_115_c20_gate; .11 is
# fire_only_once_globally with no trigger. Step 5 fires .10 here with the gate off and .11 here; step 6 sets the gate
# and fires .10 here, then .10 and .11 on one other AI country; step 7 fires both here again; step 8 reports.
te_probe_115_c20_try_without_gate_effect = {
	debug_log = "TE_PROBE_115 step 5 (C20: the once event with its trigger false, the globally event here)"
	set_variable = { name = te_probe_115_c20_once_here value = 0 }
	set_global_variable = { name = te_probe_115_c20_once_other value = 0 }
	set_global_variable = { name = te_probe_115_c20_globally_n value = 0 }
	trigger_event = { id = te_probe_115.10 }
	trigger_event = { id = te_probe_115.11 }
}

te_probe_115_c20_try_with_gate_effect = {
	debug_log = "TE_PROBE_115 step 6 (C20: the once event with its trigger true, here and on another country)"
	set_variable = te_probe_115_c20_gate
	trigger_event = { id = te_probe_115.10 }
	random_country = {
		limit = {
			is_ai = yes
			NOT = { has_variable = te_probe_115_owner }
			NOT = { is_country_type = decentralized }
			exists = capital
		}
		set_global_variable = { name = te_probe_115_c20_other value = this }
		set_variable = te_probe_115_c20_gate
		set_variable = te_probe_115_c20_other_mark
		trigger_event = { id = te_probe_115.10 }
		trigger_event = { id = te_probe_115.11 }
	}
}

te_probe_115_c20_try_again_effect = {
	debug_log = "TE_PROBE_115 step 7 (C20: both events here a second time)"
	trigger_event = { id = te_probe_115.10 }
	trigger_event = { id = te_probe_115.11 }
}

te_probe_115_c20_report_effect = {
	debug_log = "TE_PROBE_115 step 8 (C20 report, C21 disable_evaluation)"
	if = {
		limit = { var:te_probe_115_c20_once_here = 1 }
		debug_log = "TE_PROBE_115 c20_once_here ONCE (fire_only_once fired once here: not marked by the try whose trigger failed, and not again after)"
	}
	else_if = {
		limit = { var:te_probe_115_c20_once_here = 0 }
		debug_log = "TE_PROBE_115 c20_once_here NEVER (fire_only_once never fired here: the try whose trigger failed used it up, or it never fires; see c20_once_other)"
	}
	else = {
		debug_log = "TE_PROBE_115 c20_once_here TWICE (fire_only_once fired more than once on one country)"
	}
	if = {
		limit = { global_var:te_probe_115_c20_once_other = 1 }
		debug_log = "TE_PROBE_115 c20_once_other FIRED (fire_only_once is per country: it fired on a second country after this one)"
	}
	else = {
		debug_log = "TE_PROBE_115 c20_once_other NOT_FIRED (fire_only_once did not fire on a second country: once for the whole game, or no other country was found)"
	}
	if = {
		limit = { global_var:te_probe_115_c20_globally_n = 1 }
		debug_log = "TE_PROBE_115 c20_globally ONCE (fire_only_once_globally fired once across two countries and three tries)"
	}
	else_if = {
		limit = { global_var:te_probe_115_c20_globally_n = 2 }
		debug_log = "TE_PROBE_115 c20_globally TWICE (fire_only_once_globally fired twice: once per country, not once a game)"
	}
	else_if = {
		limit = { global_var:te_probe_115_c20_globally_n = 0 }
		debug_log = "TE_PROBE_115 c20_globally NEVER (fire_only_once_globally never fired)"
	}
	else = {
		debug_log = "TE_PROBE_115 c20_globally MORE (fire_only_once_globally fired on every try)"
	}
	global_var:te_probe_115_c20_other ?= {
		# C21: the entry's disable_evaluation (the claim) should have removed this country's inactive record by now,
		# seven weeks or more after the claim. The control is an entry no country holds this early.
		if = {
			limit = { has_inactive_journal_entry = je_space_race_interstellar_results }
			if = {
				limit = { has_inactive_journal_entry = te_probe_115 }
				debug_log = "TE_PROBE_115 c21_disable_evaluation KEPT (another country still holds the probe entry inactive weeks after disable_evaluation came true)"
			}
			else = {
				debug_log = "TE_PROBE_115 c21_disable_evaluation REMOVED (disable_evaluation removed another country's inactive record)"
			}
		}
		else = {
			debug_log = "TE_PROBE_115 c21_disable_evaluation INCONCLUSIVE (has_inactive_journal_entry is no even for an entry the country has never activated)"
		}
		remove_variable = te_probe_115_c20_gate
		remove_variable = te_probe_115_c20_other_mark
	}
}

# ---- C3 and C21: inside an outcome's effect -----------------------------------------------------------------------
# Called from both C3 outcomes. A second call (both outcomes ran) logs again but fires no second report.
te_probe_115_in_outcome_effect = {
	if = {
		limit = { has_journal_entry = te_probe_115 }
		debug_log = "TE_PROBE_115 c21_outcome_effect_active YES (has_journal_entry is still true inside an outcome's effect)"
	}
	else = {
		debug_log = "TE_PROBE_115 c21_outcome_effect_active NO (has_journal_entry is already false inside an outcome's effect)"
	}
	if = {
		limit = { exists = scope:journal_entry }
		debug_log = "TE_PROBE_115 c21_outcome_scope OK (scope:journal_entry is set in an outcome's effect)"
	}
	else = {
		debug_log = "TE_PROBE_115 c21_outcome_scope BROKEN (scope:journal_entry is not set in an outcome's effect)"
	}
	if = {
		limit = { NOT = { has_variable = te_probe_115_reported } }
		set_variable = te_probe_115_reported
		trigger_event = { id = te_probe_115.20 days = 3 }
	}
}

# ---- Cleanup: everything the probe set but the claim --------------------------------------------------------------

	remove_variable = te_probe_115_step
	remove_variable = te_probe_115_c7_scale
	remove_variable = te_probe_115_c7_phase
	remove_variable = te_probe_115_c7_days
	remove_variable = te_probe_115_c7_sampling
	remove_variable = te_probe_115_c20_gate
	remove_variable = te_probe_115_c20_once_here
	remove_variable = te_probe_115_c3_go
	remove_variable = te_probe_115_c3_ran_first
	remove_variable = te_probe_115_c3_ran_second
	remove_variable = te_probe_115_reported
	global_var:te_probe_115_c20_other ?= {
		remove_variable = te_probe_115_c20_gate
		remove_variable = te_probe_115_c20_other_mark
	}
	remove_global_variable = te_probe_115_c20_other
	remove_global_variable = te_probe_115_c20_once_other
	remove_global_variable = te_probe_115_c20_globally_n
	every_country = {

		remove_variable = te_probe_115_dupe
	}
}

# ---- C7's seen-after-days ladder (one literal line per phase and day) ----------------------------------------------
te_probe_115_c7_seen_log_effect = {
	if = {
		limit = { var:te_probe_115_c7_phase = 1 }
		if = {
			limit = { var:te_probe_115_c7_days = 0 }
			debug_log = "TE_PROBE_115 c7_positive seen in the same effect that set it (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 1 }
			debug_log = "TE_PROBE_115 c7_positive seen after 1 day (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 2 }
			debug_log = "TE_PROBE_115 c7_positive seen after 2 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 3 }
			debug_log = "TE_PROBE_115 c7_positive seen after 3 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 4 }
			debug_log = "TE_PROBE_115 c7_positive seen after 4 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 5 }
			debug_log = "TE_PROBE_115 c7_positive seen after 5 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 6 }
			debug_log = "TE_PROBE_115 c7_positive seen after 6 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 7 }
			debug_log = "TE_PROBE_115 c7_positive seen after 7 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
		else = {
			debug_log = "TE_PROBE_115 c7_positive seen after 8 to 14 days (value 1 read -6: a scaling modifier follows a country-only read)"
		}
	}
	else_if = {
		limit = { var:te_probe_115_c7_phase = 2 }
		if = {
			limit = { var:te_probe_115_c7_days = 0 }
			debug_log = "TE_PROBE_115 c7_negative seen in the same effect that set it (value -1 read +6: a negative value works)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 1 }
			debug_log = "TE_PROBE_115 c7_negative seen after 1 day (value -1 read +6: a negative value works)"
		}
		else_if = {
			limit = { var:te_probe_115_c7_days = 2 }
			debug_log = "TE_PROBE_115 c7_negative seen after 2 days (value -1 read +6: a negative value works)"
		}
