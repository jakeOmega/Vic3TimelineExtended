# ---- Condemnation --------------------------------------------------------------

# What a full condemnation adds above Established: shunned from Strong,
# restrained as well at Supranational. Both scale with E like every penalty.
# Scope: the condemned country (un_vote.2)
un_teeth_condemn_by_tier = {
	if = {
		limit = { un_tier_at_least_strong = yes }
		add_modifier = {
			name = un_condemned_isolation_modifier
			multiplier = un_enforcement
			days = long_modifier_time
			is_decaying = yes
		}
	}
	if = {
		limit = { un_tier_is_supranational = yes }
		add_modifier = {
			name = un_condemned_restraint_modifier
			multiplier = un_enforcement
			days = normal_modifier_time
			is_decaying = yes
		}
	}
}

# ---- Sanctions: embargoes ---------------------------------------------------------

# From Strong, a sanctions regime is also an embargo: every major-power member
# that voted for it (at Supranational, every major-power member) and has
# diplomatic relevance to the target embargoes it. The embargo pact is the
# vanilla one: it costs the embargoing member influence and trade, which is
# the enforcers' share of the cost (§5.2). Relevance is also the pact's own
# requirement to stand, so a member without it could not hold one.
# Every embargo imposed this way is listed on the target (un_embargo_by), so
# lifting the regime lifts exactly these and no embargo a member chose for
# itself.
# Scope: any; expects scope:un_resolution and scope:un_vote_target
un_teeth_sanctions_embargo = {
	scope:un_vote_target = { save_scope_as = un_embargo_target }
	every_country = {
		limit = {
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			NOT = { this = scope:un_embargo_target }
			country_rank >= rank_value:major_power
			has_diplomatic_relevance = scope:un_embargo_target
			NOT = { has_diplomatic_pact = { who = scope:un_embargo_target type = embargo } }
			OR = {
				un_tier_is_supranational = yes
				scope:un_resolution = { un_lobby_in_list = { LIST = un_res_yes COUNTRY = PREV } }
			}
		}
		create_diplomatic_pact = {
			country = scope:un_embargo_target
			type = embargo
		}
		scope:un_embargo_target = {
			add_to_variable_list = { name = un_embargo_by target = PREV }
		}
	}
}

# Lift every embargo the Assembly imposed on this country, and forget them.
# Called when the regime ends: un_lift_sanctions_button, a dissolution.
# Scope: the sanctioned country
un_teeth_lift_embargoes = {
	if = {
		limit = { has_variable_list = un_embargo_by }
		save_scope_as = un_embargo_target
		every_in_list = {
			variable = un_embargo_by
			if = {
				limit = { has_diplomatic_pact = { who = scope:un_embargo_target type = embargo } }
				remove_diplomatic_pact = {
					country = scope:un_embargo_target
					type = embargo
				}
			}
		}
		clear_variable_list = un_embargo_by
	}
}

# ---- War without a mandate: the surcharge ---------------------------------------

# Every war goal the initiator of a play adds without a bound mandate costs it
# infamy from the Established tier up (un_war_surcharge_value: 2 / 4 / 10).
# From Strong, a goal against a country under a UN peacekeeping deployment
# costs twice that. A subject fighting its way out, and a play against a
# decentralized people, are left alone; so is the holder of a bound mandate,
# whose other demands the mandate's own rules already police.
# The surcharge is the rule (owner decision 2026-09-24): the AI already weighs
# infamy before adding goals and starting wars.
# Scope: diplomatic play; expects scope:actor (from on_wargoal_added)
un_teeth_war_goal_surcharge = {
	if = {
		limit = {
			has_global_variable = un_founded
			un_war_surcharge_value > 0
			exists = scope:actor
			initiator ?= scope:actor
			scope:actor = {
				is_subject = no
				NOT = { var:un_mandate_current ?= { has_tag = un_mandate_bound } }
			}
			target ?= { NOT = { is_country_type = decentralized } }
		}
		scope:actor = {
			change_infamy = un_war_surcharge_value
		}
		if = {
			limit = {
				un_tier_at_least_strong = yes
				target ?= { has_modifier = un_peacekeeping_received_modifier }
			}
			scope:actor = {
				change_infamy = un_war_surcharge_value
			}
		}
		scope:actor = {
			if = {
				limit = { is_ai = no }
				post_notification = un_war_surcharge_notice
			}
		}
	}
}

# ---- Monthly --------------------------------------------------------------------

# Global monthly upkeep (no-country scope; from un_global_authority_on_action,
# UN founded).
#   1. Sanctions busting: a member that has dropped the embargo the Assembly
#      imposed, while the regime still stands, is recorded as busting it
#      (the dossier's +8, and a credibility debit weighted by the member), and
#      struck from the list. A regime that has ended forgets its list.
#   2. The mandate's war support (the single refresh site).
un_teeth_monthly_update = {
	every_country = {
		limit = { has_variable_list = un_embargo_by }
		save_scope_as = un_embargo_target
		if = {
			limit = {
				OR = {
					has_modifier = un_sanctions_target_modifier
					has_modifier = un_sanctions_voluntary_target_modifier
				}
			}
			# The list is rebuilt rather than edited while it is iterated:
			# the members still embargoing go to un_embargo_by_next, the
			# busters are recorded and left out.
			every_in_list = {
				variable = un_embargo_by
				if = {
					limit = { un_teeth_embargo_dropped = yes }
					un_dossier_record = { RECORD = defiance POINTS = un_dos_points_busting }
					un_ledger_actor_entry = { PILLAR = credibility POINTS = -0.5 REASON = 15 }
				}
				else = {
					scope:un_embargo_target = {
						add_to_variable_list = { name = un_embargo_by_next target = PREV }
					}
				}
			}
			clear_variable_list = un_embargo_by
			if = {
				limit = { has_variable_list = un_embargo_by_next }
				every_in_list = {
					variable = un_embargo_by_next
					scope:un_embargo_target = {
						add_to_variable_list = { name = un_embargo_by target = PREV }
					}
				}
				clear_variable_list = un_embargo_by_next
			}
		}
		else = {
			clear_variable_list = un_embargo_by
		}
	}
	every_country = {
		limit = {
			OR = {
				has_variable = un_mandate_current
				has_modifier = un_mandate_war_support_modifier
			}
		}
		if = {
			limit = { un_teeth_mandate_support_applies = yes }
			if = {
				limit = { NOT = { has_modifier = un_mandate_war_support_modifier } }
				add_modifier = {
					name = un_mandate_war_support_modifier
					multiplier = un_enforcement
				}
			}
		}
		else_if = {
			limit = { has_modifier = un_mandate_war_support_modifier }
			remove_modifier = un_mandate_war_support_modifier
		}
	}
}
