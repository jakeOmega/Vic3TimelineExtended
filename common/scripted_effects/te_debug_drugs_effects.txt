# ============================================================================
# DRUGS WEALTH PROBE — console-only (`event te_debug_drugs.1`)
# ============================================================================
# Drugs phase 2 § 1 (docs/superpowers/plans/2026-10-02-drugs-phase-2-healthcare-
# demand.md). Every country gets its population and the share of it living in
# pops at wealth 10, 15, 20, 25, 30, 35, 40 and 50 or more, its Drugs buildings,
# its market's Drugs price and whether it bans Drugs. The observer (ROOT) also
# gets the world totals. Every number is written to debug.log on TE_DRUGS_PROBE
# lines, so nothing needs saving: a save from before #500 crashed twice on saving
# after the probe ran (2026-10-02). The variables stay on the countries for the
# in-game variable view, and a save that does work can be read with
#   scripts/analysis/save_country_probe.py <save> --var '^te_dbg_drugs_'
# World totals sit on the observer, not in global variables, because that tool
# reads country variables only. Re-running overwrites every value.
# ============================================================================

# THIS = a country, saved as scope:te_dbg_drugs_country. Shares are percentages
# (0-100): save_country_probe.py rounds to three decimals, which would flatten a
# 0-1 share of the richest bands in early saves to zero.
te_debug_drugs_measure_band = {
	set_variable = { name = te_dbg_drugs_n value = te_dbg_drugs_pops_w$W$ }
	set_variable = {
		name = te_dbg_drugs_w$W$
		value = {
			value = var:te_dbg_drugs_n
			multiply = 100
			divide = {
				value = var:te_dbg_drugs_pop
				min = 1
			}
		}
	}
	root = {
		change_variable = { name = te_dbg_drugs_world_n$W$ add = scope:te_dbg_drugs_country.var:te_dbg_drugs_n }
	}
	remove_variable = te_dbg_drugs_n
}

# THIS = ROOT, after every country has been counted.
te_debug_drugs_world_band = {
	set_variable = {
		name = te_dbg_drugs_world_w$W$
		value = {
			value = var:te_dbg_drugs_world_n$W$
			multiply = 100
			divide = {
				value = var:te_dbg_drugs_world_pop
				min = 1
			}
		}
	}
}

# THIS = a country te_debug_drugs_measure has just measured, saved as
# scope:te_dbg_drugs_country. A debug_log ScriptValue evaluates against ROOT, so
# the numbers go onto ROOT first. The tag has its own line: GetTagName has never
# been printed by a debug_log here, and a bracket that fails takes its line with it.
te_debug_drugs_log_country = {
	root = {
		set_variable = { name = te_dbg_drugs_cur_pop value = scope:te_dbg_drugs_country.var:te_dbg_drugs_pop }
		set_variable = { name = te_dbg_drugs_cur_w10 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w10 }
		set_variable = { name = te_dbg_drugs_cur_w15 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w15 }
		set_variable = { name = te_dbg_drugs_cur_w20 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w20 }
		set_variable = { name = te_dbg_drugs_cur_w25 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w25 }
		set_variable = { name = te_dbg_drugs_cur_w30 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w30 }
		set_variable = { name = te_dbg_drugs_cur_w35 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w35 }
		set_variable = { name = te_dbg_drugs_cur_w40 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w40 }
		set_variable = { name = te_dbg_drugs_cur_w50 value = scope:te_dbg_drugs_country.var:te_dbg_drugs_w50 }
		set_variable = { name = te_dbg_drugs_cur_plant value = scope:te_dbg_drugs_country.var:te_dbg_drugs_plant_lv }
		set_variable = { name = te_dbg_drugs_cur_pharma value = scope:te_dbg_drugs_country.var:te_dbg_drugs_pharma_lv }
		set_variable = { name = te_dbg_drugs_cur_price value = -1 }
		set_variable = { name = te_dbg_drugs_cur_banned value = 0 }
	}
	if = {
		limit = { has_variable = te_dbg_drugs_price }
		root = {
			set_variable = { name = te_dbg_drugs_cur_price value = scope:te_dbg_drugs_country.var:te_dbg_drugs_price }
		}
	}
	if = {
		limit = { has_variable = te_dbg_drugs_banned }
		root = {
			set_variable = { name = te_dbg_drugs_cur_banned value = 1 }
		}
	}
	debug_log = "TE_DRUGS_PROBE TAG [THIS.GetCountry.GetTagName]"
	debug_log = "TE_DRUGS_PROBE COUNTRY [THIS.GetCountry.GetNameNoFormatting] | pop=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_pop')|0] w10=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w10')|3] w15=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w15')|3] w20=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w20')|3] w25=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w25')|3] w30=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w30')|3] w35=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w35')|3] w40=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w40')|3] w50=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_w50')|3] plant=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_plant')|0] pharma=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_pharma')|0] price=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_price')|3] banned=[SCOPE.ScriptValue('te_dbg_drugs_log_cur_banned')|0]"
}

te_debug_drugs_measure = {
	debug_log = "TE_DRUGS_PROBE RUN decade=[SCOPE.ScriptValue('te_dbg_drugs_log_decade')|0] observer=[THIS.GetCountry.GetNameNoFormatting]"
	set_variable = { name = te_dbg_drugs_world_pop value = 0 }
	set_variable = { name = te_dbg_drugs_world_n10 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n15 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n20 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n25 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n30 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n35 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n40 value = 0 }
	set_variable = { name = te_dbg_drugs_world_n50 value = 0 }

	every_country = {
		save_temporary_scope_as = te_dbg_drugs_country
		set_variable = { name = te_dbg_drugs_pop value = te_dbg_drugs_pops_all }
		root = {
			change_variable = { name = te_dbg_drugs_world_pop add = scope:te_dbg_drugs_country.var:te_dbg_drugs_pop }
		}
		te_debug_drugs_measure_band = { W = 10 }
		te_debug_drugs_measure_band = { W = 15 }
		te_debug_drugs_measure_band = { W = 20 }
		te_debug_drugs_measure_band = { W = 25 }
		te_debug_drugs_measure_band = { W = 30 }
		te_debug_drugs_measure_band = { W = 35 }
		te_debug_drugs_measure_band = { W = 40 }
		te_debug_drugs_measure_band = { W = 50 }

		set_variable = { name = te_dbg_drugs_plant_lv value = te_dbg_drugs_lv_plantation }
		set_variable = { name = te_dbg_drugs_pharma_lv value = te_dbg_drugs_lv_pharma }
		if = {
			limit = {
				market ?= {
					mg:opium ?= { always = yes }
				}
			}
			set_variable = { name = te_dbg_drugs_price value = te_dbg_drugs_market_price }
		}
		else_if = {
			limit = { has_variable = te_dbg_drugs_price }
			remove_variable = te_dbg_drugs_price
		}
		if = {
			limit = { is_banning_goods = g:opium }
			set_variable = { name = te_dbg_drugs_banned value = 1 }
		}
		else_if = {
			limit = { has_variable = te_dbg_drugs_banned }
			remove_variable = te_dbg_drugs_banned
		}
		te_debug_drugs_log_country = yes
	}

	te_debug_drugs_world_band = { W = 10 }
	te_debug_drugs_world_band = { W = 15 }
	te_debug_drugs_world_band = { W = 20 }
	te_debug_drugs_world_band = { W = 25 }
	te_debug_drugs_world_band = { W = 30 }
	te_debug_drugs_world_band = { W = 35 }
	te_debug_drugs_world_band = { W = 40 }
	te_debug_drugs_world_band = { W = 50 }

	set_variable = { name = te_dbg_drugs_world_lv_plant_default value = te_dbg_drugs_world_levels_plantation_default }
	set_variable = { name = te_dbg_drugs_world_lv_plant_irrigation value = te_dbg_drugs_world_levels_plantation_irrigation }
	set_variable = { name = te_dbg_drugs_world_lv_plant_factory_farm value = te_dbg_drugs_world_levels_plantation_factory_farm }
	set_variable = { name = te_dbg_drugs_world_lv_pharma_alkaloid value = te_dbg_drugs_world_levels_pharma_alkaloid }
	set_variable = { name = te_dbg_drugs_world_lv_pharma_synth_chem value = te_dbg_drugs_world_levels_pharma_synthetic_chemistry }
	set_variable = { name = te_dbg_drugs_world_lv_pharma_antibiotic value = te_dbg_drugs_world_levels_pharma_antibiotic }
	set_variable = { name = te_dbg_drugs_world_lv_pharma_synth_opium value = te_dbg_drugs_world_levels_pharma_synthetic_opium }
	set_variable = { name = te_dbg_drugs_world_lv_pharma_biologics value = te_dbg_drugs_world_levels_pharma_biologics }
	set_variable = { name = te_dbg_drugs_world_lv_pharma_precision value = te_dbg_drugs_world_levels_pharma_precision }
	set_variable = { name = te_dbg_drugs_world_lv_company value = te_dbg_drugs_world_levels_company }
	set_variable = { name = te_dbg_drugs_world_buy value = te_dbg_drugs_world_buy_orders }
	set_variable = { name = te_dbg_drugs_world_sell value = te_dbg_drugs_world_sell_orders }

	debug_log = "TE_DRUGS_PROBE WORLD pop=[SCOPE.ScriptValue('te_dbg_drugs_log_world_pop')|0] w10=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w10')|3] w15=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w15')|3] w20=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w20')|3] w25=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w25')|3] w30=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w30')|3] w35=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w35')|3] w40=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w40')|3] w50=[SCOPE.ScriptValue('te_dbg_drugs_log_world_w50')|3]"
	debug_log = "TE_DRUGS_PROBE SUPPLY plant_default=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_plantation_default')|0] plant_irrigation=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_plantation_irrigation')|0] plant_factory_farm=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_plantation_factory_farm')|0] pharma_alkaloid=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_pharma_alkaloid')|0] pharma_synth_chem=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_pharma_synthetic_chemistry')|0] pharma_antibiotic=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_pharma_antibiotic')|0] pharma_synth_opium=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_pharma_synthetic_opium')|0] pharma_biologics=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_pharma_biologics')|0] pharma_precision=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_pharma_precision')|0] company=[SCOPE.ScriptValue('te_dbg_drugs_world_levels_company')|0]"
	debug_log = "TE_DRUGS_PROBE MARKET buy=[SCOPE.ScriptValue('te_dbg_drugs_world_buy_orders')|0] sell=[SCOPE.ScriptValue('te_dbg_drugs_world_sell_orders')|0]"
	debug_log = "TE_DRUGS_PROBE END"
	remove_variable = te_dbg_drugs_cur_pop
	remove_variable = te_dbg_drugs_cur_w10
	remove_variable = te_dbg_drugs_cur_w15
	remove_variable = te_dbg_drugs_cur_w20
	remove_variable = te_dbg_drugs_cur_w25
	remove_variable = te_dbg_drugs_cur_w30
	remove_variable = te_dbg_drugs_cur_w35
	remove_variable = te_dbg_drugs_cur_w40
	remove_variable = te_dbg_drugs_cur_w50
	remove_variable = te_dbg_drugs_cur_plant
	remove_variable = te_dbg_drugs_cur_pharma
	remove_variable = te_dbg_drugs_cur_price
	remove_variable = te_dbg_drugs_cur_banned
}
