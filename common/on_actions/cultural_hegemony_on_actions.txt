# CULTURAL HEGEMONY — On-Actions
# ============================================================================

# ---- War end: force leaderboard recalculation ----
# When a war ends, countries may be annexed or borders may change.
# Remove the leaderboard lock so the next monthly pulse forces a full
# recalculation, ensuring dead/annexed countries are removed from rankings.
on_war_end = {
	on_actions = {
		ch_war_end_recalc_on_action
	}
}
ch_war_end_recalc_on_action = {
	effect = {
		if = {
			limit = {
				has_global_variable = ch_ldr_lock
				te_cultural_hegemony_on = yes
			}
			remove_global_variable = ch_ldr_lock
		}
	}
}

# ---- Monthly pulse: update per-country CH values and modifiers ----
# Runs from game start for all qualifying countries (even before JE activates).
# This ensures scores are stable by the time the JE appears (once any country
# has Mass Media; je_cultural_hegemony `possible`).
on_monthly_pulse_country = {
	on_actions = {
		ch_monthly_pulse_on_action
	}
}
ch_monthly_pulse_on_action = {
	effect = {
		# ---- The entry's net ----
		# An ended entry doesn't come back on its own, and a record
		# disable_evaluation removed while the rule was off is gone for good;
		# add_journal_entry brings a concluded type back (Colonial Empire's
		# ce_reentry, read OK 2026-11-12). So a country with no record of the
		# entry gets it once the rule is on and the gate of the setting in
		# force is met (ch_entry_due), whether its entry ended on the rule,
		# ended early to late, or it never had one. Covert Warfare's net
		# (covert_warfare_monthly_country) is the same, with the same
		# self-check:
		#   TE_SELF_CHECK OK ch_entry_return      the entry the net added last
		#                                         month is running (once a game)
		#   TE_SELF_CHECK BROKEN ch_entry_return  it is not, though the country
		#                                         is still due it; the net then
		#                                         waits a year for that country
		#                                         (ch_entry_net_hold)
		if = {
			limit = { has_variable = ch_entry_net_added }
			save_scope_as = ch_net_country
			if = {
				limit = { var:ch_entry_net_added ?= scope:ch_net_country }
				if = {
					limit = { has_journal_entry = je_cultural_hegemony }
					if = {
						limit = { NOT = { has_global_variable = ch_entry_return_ok_logged } }
						set_global_variable = ch_entry_return_ok_logged
						debug_log = "TE_SELF_CHECK OK ch_entry_return country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString] (the net added je_cultural_hegemony last month and it is running; logged once a game)"
					}
				}
				else_if = {
					limit = { ch_entry_due = yes }
					set_variable = { name = ch_entry_net_hold value = yes days = 365 }
					debug_log = "TE_SELF_CHECK BROKEN ch_entry_return country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString] (the net added je_cultural_hegemony last month; it is not running and the country is still due it, so it ended at once or the add failed: the net waits a year for this country; docs/testing/rules-off-midgame-playtest.md, Self-checks)"
				}
			}
			remove_variable = ch_entry_net_added
		}
		if = {
			limit = {
				NOT = { has_variable = ch_entry_net_hold }
				ch_entry_due = yes
			}
			if = {
				limit = { has_variable = ch_reopen_at_gate }
				debug_log = "TE_CH_ENTRY added country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] ended_before=yes date=[TimeKeeper.GetCurrentDate.GetString]"
				remove_variable = ch_reopen_at_gate
			}
			else = {
				debug_log = "TE_CH_ENTRY added country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] ended_before=no date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			add_journal_entry = { type = je_cultural_hegemony }
			save_scope_as = ch_net_country
			set_variable = { name = ch_entry_net_added value = scope:ch_net_country }
		}
		if = {
			limit = {
				te_cultural_hegemony_on = yes
			}
			ch_monthly_country_update = yes
			# A save from before the census kept each country's own pull
			# (ch_raw_at_census) recounts now rather than when the lock expires;
			# until then shares divide by the census total as before.
			if = {
				limit = {
					NOT = { has_global_variable = ch_census_own_raw }
					has_global_variable = ch_ldr_lock
				}
				remove_global_variable = ch_ldr_lock
			}
			# If leaderboard lock was cleared (e.g. by war end), rebuild immediately
			if = {
				limit = { NOT = { has_global_variable = ch_ldr_lock } }
				ch_yearly_global_update = yes
			}
		}
	}
}

# ---- Yearly pulse: refresh global caches + build leaderboard ----
# Runs from game start. Lock guard inside the effect ensures only one country
# per year does the expensive global computation.
on_yearly_pulse_country = {
	on_actions = {
		ch_yearly_pulse_on_action
	}
}
ch_yearly_pulse_on_action = {
	effect = {
		if = {
			limit = {
				te_cultural_hegemony_on = yes
			}
			ch_yearly_global_update = yes
		}
	}
}

# Track "World First" technology achievements for Cultural Pull
# Root scope = Country that just acquired the technology
# scope:technology = the technology just researched (from on_acquired_technology)
# If no other country has this tech yet, this is a world first.
cultural_hegemony_tech_first_on_action = {
	effect = {
		if = {
			limit = { te_cultural_hegemony_on = yes }
			if = {
				limit = {
					# True world first: no other country has researched this tech yet
					NOT = {
						any_country = {
							NOT = { this = ROOT }
							has_technology_researched = scope:technology
						}
					}
				}
				if = {
					limit = { NOT = { has_variable = ch_tech_firsts_recent } }
					set_variable = { name = ch_tech_firsts_recent value = 0 }
				}
				change_variable = { name = ch_tech_firsts_recent add = cultural_pull_tech_first_bonus }
			}
		}
	}
}
