# Copper mine group: a once-a-game self-check (vanilla 1.15 migration, #1402).
#
# extra_buildings.txt moves vanilla 1.15's building_copper_mine from bg_mining
# into the mod's bg_discoverable_mining with an INJECT of building_group, so the
# map's copper resource blocks (vanilla's caps discovered, the rest not) apply.
# No in-game check has shown an INJECTed building_group take. The first yearly
# country pulse of the game reads every copper mine in the world, once:
#   OK      one is in bg_discoverable_mining: the INJECT took.
#   BROKEN  copper mines exist but none is: the mine stayed capped in bg_mining,
#           the resource blocks are ignored, and no new copper mine can be built
#           anywhere (debug.log: "Copper Mines and can only support 0").
#   BROKEN  no copper mine exists anywhere: the same failure, after the engine
#           removed the 1836 mines it found no cap for. Vanilla's 1836 history
#           places copper mines, so a working game always has one to read.
# Fallback for BROKEN: a TRY_REPLACE of vanilla's whole building_copper_mine with
# the group changed; the maintenance and tailings INJECTs still apply after it.
# Play-test row: docs/testing/vanilla-1-15-playtest.md, "Copper and lead".
on_yearly_pulse_country = {
	on_actions = {
		te_copper_mine_group_selfcheck
	}
}

te_copper_mine_group_selfcheck = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = te_copper_mine_group_checked }
			}
			set_global_variable = te_copper_mine_group_checked
			if = {
				limit = {
					any_country = {
						any_scope_building = {
							is_building_type = building_copper_mine
							is_building_group = bg_discoverable_mining
						}
					}
				}
				debug_log = "TE_SELF_CHECK OK copper_mine_discoverable (vanilla's copper mine is in bg_discoverable_mining: the INJECTed building_group took)"
			}
			else_if = {
				limit = {
					any_country = {
						any_scope_building = { is_building_type = building_copper_mine }
					}
				}
				debug_log = "TE_SELF_CHECK BROKEN copper_mine_discoverable (copper mines exist but none is in bg_discoverable_mining: the INJECTed building_group did not take, so the map's copper resource blocks are ignored; fall back to a TRY_REPLACE of building_copper_mine, see docs/testing/vanilla-1-15-playtest.md)"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN copper_mine_discoverable (no copper mine exists anywhere: the INJECTed building_group most likely did not take and the engine removed the 1836 mines; fall back to a TRY_REPLACE of building_copper_mine, see docs/testing/vanilla-1-15-playtest.md)"
			}
		}
	}
}
