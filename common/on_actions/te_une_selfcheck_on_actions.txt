# United Nations of Earth supermajority: two once-a-game self-checks.
#
# une_formation_supermajority (formable_gating_triggers.txt) asks "at most one
# great power other than the candidate stands outside its bloc" as a share:
# `filter` over the other great powers, `percent >=` a script value half a
# great power below all but one (te_formation_gating_values.txt). The engine
# refused the first version (`filter` with `count >=`, 2026-11-12), and no
# launch has run this one, so each check compares it, on a real bloc leader,
# with the form it replaced, `NOT = { any_country = { <outside> count >= 2 } }`,
# which the engine has always taken.
#
# Judged only for a bloc leader with two or more other great powers, where the
# gate reads the share at all, and only at the boundary, so that an answer the
# share gets wrong one way can't read OK:
#   une_supermajority_pass  one or no other great power outside: the old form
#                           passes. OK when the gate passes too.
#   une_supermajority_fail  exactly two outside: the old form fails. OK when
#                           the gate fails too.
# BROKEN on either: the engine reads `filter` with `percent`, or the share,
# otherwise than vanilla's tsar buttons suggest. Fallback: drop the share and
# the list, and go back to the NOT form (`une_formation_supermajority` in
# git history before #1672), which lists nothing but counts right.
# Play-test rows: docs/testing/vanilla-1-15-playtest.md, `list_scopes` (#1436).
on_yearly_pulse_country = {
	on_actions = {
		te_une_supermajority_selfcheck
	}
}

te_une_supermajority_selfcheck = {
	effect = {
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable = te_sc_une_supermajority_pass }
					NOT = { has_global_variable = te_sc_une_supermajority_fail }
				}
				is_power_bloc_leader = yes
				any_country = {
					country_rank >= rank_value:great_power
					NOT = { this = ROOT }
					count >= 2
				}
			}
			if = {
				limit = {
					NOT = { has_global_variable = te_sc_une_supermajority_pass }
					NOT = {
						any_country = {
							country_rank >= rank_value:great_power
							NOT = { this = ROOT }
							NOT = { is_in_same_power_bloc = ROOT }
							count >= 2
						}
					}
				}
				set_global_variable = te_sc_une_supermajority_pass
				if = {
					limit = { une_formation_supermajority = yes }
					debug_log = "TE_SELF_CHECK OK une_supermajority_pass country=[THIS.GetCountry.GetTagName] (a bloc leader with at most one other great power outside its bloc passes the share gate, as the count it replaced does) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN une_supermajority_pass country=[THIS.GetCountry.GetTagName] (a bloc leader with at most one other great power outside its bloc fails the share gate though the count it replaced passes: the engine reads filter with percent, or the share, otherwise; see docs/testing/vanilla-1-15-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
			else_if = {
				limit = {
					NOT = { has_global_variable = te_sc_une_supermajority_fail }
					any_country = {
						country_rank >= rank_value:great_power
						NOT = { this = ROOT }
						NOT = { is_in_same_power_bloc = ROOT }
						count >= 2
					}
					NOT = {
						any_country = {
							country_rank >= rank_value:great_power
							NOT = { this = ROOT }
							NOT = { is_in_same_power_bloc = ROOT }
							count >= 3
						}
					}
				}
				set_global_variable = te_sc_une_supermajority_fail
				if = {
					limit = { une_formation_supermajority = no }
					debug_log = "TE_SELF_CHECK OK une_supermajority_fail country=[THIS.GetCountry.GetTagName] (a bloc leader with exactly two other great powers outside its bloc fails the share gate, as the count it replaced does) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN une_supermajority_fail country=[THIS.GetCountry.GetTagName] (a bloc leader with exactly two other great powers outside its bloc passes the share gate though the count it replaced fails: the engine reads filter with percent, or the share, otherwise; see docs/testing/vanilla-1-15-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
		}
	}
}
