# Root = State, a different day per state across the year. Runs whatever the
# rule is: the walks feed inheritance and Wealth Concentration even with the
# census off, and the orchestrator gates the census itself.
on_yearly_pulse_state = {
	on_actions = {
		te_demog_state_yearly_on_action
	}
}

te_demog_state_yearly_on_action = {
	effect = {
		te_demog_state_yearly = yes
		# the national urban pattern's state modifiers, from the owner's last census (Full only; phase 3,
		# te_demog_place_effects.txt): their one refresh site, with ROOT the state
		te_demog_up_state_refresh = yes
		# a planned capital's development ends once the state is no longer the capital
		te_demog_pc_state_check = yes
	}
}

# Root = Country, monthly. Not gated by the rule: the war-shock work (Task 11) reads
# the same running total of the owner's war dead.
on_monthly_pulse_country = {
	on_actions = {
		te_demog_country_monthly_on_action
	}
}

te_demog_country_monthly_on_action = {
	effect = {
		te_demog_country_monthly = yes
		# Wealth Concentration's financial boom and land reform, whatever the rule (te_demog_wealth_effects.txt)
		te_demog_wc_boom_check = yes
		te_demog_wc_land_reform_check = yes
		# the Crash Campaign ends itself after five years, to the month (Full only)
		te_demog_ms_monthly = yes
		# phase 2 step 1's once-a-game self-checks (te_debug_demog_effects.txt; they gate on the census themselves)
		te_debug_demog_wf_selfchecks = yes
		# calibration step 3's (te_debug_demog_effects.txt)
		te_debug_demog_step3_selfchecks = yes
		# phase 3's event wave: the war tracker for The Lost Generation and The Baby Boom (Full only;
		# te_demog_wave_effects.txt)
		te_demog_wave_war_monthly = yes
		# the History charts' famine pip (te_demog_history_effects.txt; issue #1029)
		te_demog_history_note_famine = yes
	}
}

# No ROOT, the 1st of every calendar month. Fast mode's census clock (console only,
# te_demog_fast_effects.txt): nothing runs unless it was started.
on_monthly_pulse = {
	on_actions = {
		te_demog_clock_on_action
	}
}

te_demog_clock_on_action = {
	effect = {
		if = {
			limit = { te_demog_clock_on = yes }
			te_demog_clock_tick = yes
		}
	}
}

# Root = Country, on 31 December. The war shock and Wealth Concentration's national
# figure run whatever the rule; the census gates itself on it.
on_yearly_pulse_country = {
	on_actions = {
		te_demog_country_yearly_on_action
	}
}

te_demog_country_yearly_on_action = {
	effect = {
		# Wealth Concentration's r > g inputs and term, before the national refresh in te_demog_country_yearly reads

		te_demog_wc_growth_update = yes

# Every 1836 state opens with a census. on_game_started has no ROOT, so each
# country walks and seeds its states in a hidden country event
# (te_demog_events.1); the fan-out is te_inh_on_game_started's.
on_game_started = {
	on_actions = {
		te_demog_on_game_started
	}
}

te_demog_on_game_started = {
	effect = {
		every_country = {
			trigger_event = {
				id = te_demog_events.1
			}
		}
	}
}

# Root = a country on the losing side of a war, capitulations included (scope:war the war;
# Grand Monuments' war commissions use the same hook, monument_events_on_actions.txt). A lost
# war that cost it a war goal takes 5 from every state's Wealth Concentration (spec §4.2),
# whatever the rule. scope:victim_of_wargoal exists only when one of the winners' war goals
# was imposed on this country, so a co-belligerent that dropped out having given up nothing
# is spared (owner's ruling on #830); long wars have their own shock, te_demog_wc_war_shock.
on_lost_war = {
	on_actions = {
		te_demog_lost_war
	}
}

te_demog_lost_war = {
	effect = {
		if = {
			limit = { exists = scope:victim_of_wargoal }
			te_demog_wc_shock = { AMOUNT = -5 }
		}
	}
}

# The engine's moved birth and mortality lines (owner, 2026-10-11; the engine-lines plan, Task 4): refreshed where their
# inputs change, and every month for institution levels, which have no on-action, and old saves.
# ROOT must be the state (the pairs' multipliers resolve against it): the state hooks call the refresh, the law,
# technology and global ones fire te_demog_events.4 in each state. That event runs the refresh with ROOT = the state, so
# the "pulse-only" note in extra_on_actions.txt (script values evaluated in a law's or country's scope chain) doesn't
# apply: these hooks only dispatch.
on_monthly_pulse_state = {
	on_actions = {
		te_demog_lines_on_action
	}
}
on_state_owner_change = {
	on_actions = {
		te_demog_lines_on_action
		te_demog_wc_owner_change_on_action
	}
}
on_state_incorporation = {
	on_actions = {
		te_demog_lines_on_action
	}
}
on_state_created = {
	on_actions = {
		te_demog_lines_on_action
	}
}
on_law_activated = {
	on_actions = {
		te_demog_lines_law_on_action
		te_demog_ms_law_on_action
		te_demog_pension_law_on_action
	}
}
on_acquired_technology = {
	on_actions = {

		te_demog_wave_technology_on_action
	}
}
# ROOT = the country (vanilla's comments on both hooks): an amendment repealed or timed out (the timeout's is already off
# its law) changes its law's working-adult lines, so the country's states refresh that day, as for a technology. An
# amendment attached has no hook; the monthly refresh takes it within a month (the design §1.9).
on_amendment_repealed = {
	on_actions = {
		te_demog_lines_country_on_action
	}
}
on_amendment_timeout = {
	on_actions = {
		te_demog_lines_country_on_action
	}
}
on_game_started_after_lobby = {
	on_actions = {
		te_demog_lines_world_on_action
	}
}

# THIS = ROOT = a state.
te_demog_lines_on_action = {
	effect = {
		te_demog_lines_refresh = yes
		te_demog_workforce_lines_refresh = yes
		# phase 2 step 2: a lost workforce term's modifier back within a month
		te_demog_workforce_heal = yes
		# the census's births and deaths terms likewise (#937)
		te_demog_rates_heal = yes
	}
}

# ROOT = the state that changed hands. Wealth Concentration's ownership shock compares a state's state-owned share with
# last year's; after a change of owner the next pulse stores the new owner's without a shock, whatever the rule
# (te_demog_wealth_effects.txt).
te_demog_wc_owner_change_on_action = {
	effect = {
		te_demog_wc_owner_change = yes
	}
}

# ROOT = the law just activated (an enactment, or a law script set; vanilla's note on the hook), so the states are its
# owner's. Not before the game starts: history's laws are te_demog_lines_world_on_action's, once the rule is settled.
te_demog_lines_law_on_action = {
	effect = {
		if = {
			limit = { has_game_started = yes }
			owner = {
				every_scope_state = {
					trigger_event = { id = te_demog_events.4 }
					# again a day later, in case the engine hasn't re-summed the families on the day (the refresh acts
					# only where the sum changed)
					trigger_event = { id = te_demog_events.4 days = 1 }
				}
			}
		}
	}
}

# ROOT = the law just activated: its owner's pension age, that day (phase 2 step 1). Under every rule, as the yearly
# refresh; a history law's activation sets it before the game starts, and the game start sets it again.
te_demog_pension_law_on_action = {
	effect = {
		owner = {
			te_demog_pension_age_refresh = yes
		}
	}
}

# ROOT = a country that acquired a technology: each of its states, as its own ROOT. Not before the game starts, as above.
te_demog_lines_country_on_action = {
	effect = {
		if = {
			limit = { has_game_started = yes }
			every_scope_state = {
				trigger_event = { id = te_demog_events.4 }
				trigger_event = { id = te_demog_events.4 days = 1 }
			}
		}
	}
}

# No ROOT (after the lobby, so the rule is settled): every state, as its own ROOT.
te_demog_lines_world_on_action = {
	effect = {
		every_state = {
			trigger_event = { id = te_demog_events.4 }
		}
	}
}

# ROOT = the law just activated, so the country is its owner: a Family Policy measure the law in force no longer
# allows switches off at once, with a notice (te_demog_ms_law_check; Full only). Not before the game starts: no measure
# is on then.
te_demog_ms_law_on_action = {
	effect = {
		if = {
			limit = { has_game_started = yes }
			owner = {
				te_demog_ms_law_check = yes
			}
		}
	}
}

# ROOT = a country that acquired a technology: phase 3's The Last Generation? at once when it is Biological Immortality
# (Full only; te_demog_wave_effects.txt, whose yearly check is the fallback).
te_demog_wave_technology_on_action = {
	effect = {
		te_demog_wave_technology = yes
	}
}
