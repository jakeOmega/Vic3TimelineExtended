# ============================================================================
# OLD-SAVE REPAIRS — one-off fixes for saves made under an older mod version.
# ============================================================================
# Each repair carries a marker line:
#   # OLD-SAVE REPAIR (#<issue>): remove after <YYYY-MM-DD> or at vanilla <X.Y>, whichever comes first.
# test_old_save_repairs.py fails CI once the committed vanilla snapshot
# (vanilla_parsed/manifest.json) reaches <X.Y>, so the repair goes in the
# vanilla-patch migration PR: saves from before a major vanilla version are not
# expected to carry over. A scheduled reminder covers the date. Delete the
# repair's block, and this file once it holds none.
# ============================================================================

on_monthly_pulse = {
	on_actions = {
		te_repair_double_inheritance_law
		te_repair_lunar_technologies
		te_repair_era_13_technologies
		te_repair_history_order
	}
}

# OLD-SAVE REPAIR (#821): remove after 2026-12-08 or at vanilla 1.15, whichever comes first.
#
# The 2026-10-08 inheritance redesign added law_free_testation and
# law_state_universal_heir to lawgroup_inheritance. They come after the old
# four in file order, unlike #520's Penal Labor Camps, which went first; whether
# that is enough to keep a pre-change save from switching a second law on in
# every country (the #547 fault above) was not tested in game. So the same net:
# a country holding two inheritance laws keeps one.
#
# The law kept: one of the four laws a pre-change save could hold over either
# new law (the load could not have enacted them), and any other old law over
# Primogeniture, the group's first law and the one a #520-style load would have
# switched on. The group is cleared, then the kept law activated (vanilla's
# order, as above). The laws' only on_activate sets te_inh_law_recent, which
# holds back the generation events for 20 years: a repaired country waits a
# generation for them, nothing worse. Monthly and idempotent: a country is touched only while it holds two.
# A TE_INH_REPAIR line for the same country in two months running means the
# repair does not hold.
te_repair_double_inheritance_law = {
	effect = {
		every_country = {
			limit = {
				OR = {
					AND = {
						OR = {
							has_law = law_type:law_free_testation
							has_law = law_type:law_state_universal_heir
						}
						OR = {
							has_law = law_type:law_primogeniture
							has_law = law_type:law_partible
							has_law = law_type:law_equal_inheritance
							has_law = law_type:law_non_inheritable_usage_rights
						}
					}
					AND = {
						has_law = law_type:law_free_testation
						has_law = law_type:law_state_universal_heir
					}
					AND = {
						has_law = law_type:law_primogeniture
						OR = {
							has_law = law_type:law_partible
							has_law = law_type:law_equal_inheritance
							has_law = law_type:law_non_inheritable_usage_rights
						}
					}
				}
			}
			# Decide before touching the group: 1 Customary, 2 Forced Heirship,
			# 3 Possession by Use, 4 Primogeniture, 5 Freedom of Testation,
			# 6 State as Universal Heir.
			if = {
				limit = { has_law = law_type:law_equal_inheritance }
				set_local_variable = { name = te_inh_keep value = 2 }
			}
			else_if = {
				limit = { has_law = law_type:law_non_inheritable_usage_rights }
				set_local_variable = { name = te_inh_keep value = 3 }
			}
			else_if = {
				limit = { has_law = law_type:law_partible }
				set_local_variable = { name = te_inh_keep value = 1 }
			}
			else_if = {
				limit = { has_law = law_type:law_primogeniture }
				set_local_variable = { name = te_inh_keep value = 4 }
			}
			else_if = {
				limit = { has_law = law_type:law_free_testation }
				set_local_variable = { name = te_inh_keep value = 5 }
			}
			else = {
				set_local_variable = { name = te_inh_keep value = 6 }
			}
			if = {
				limit = { has_law = law_type:law_primogeniture }
				deactivate_law = law_type:law_primogeniture
			}
			if = {
				limit = { has_law = law_type:law_partible }
				deactivate_law = law_type:law_partible
			}
			if = {
				limit = { has_law = law_type:law_equal_inheritance }
				deactivate_law = law_type:law_equal_inheritance
			}
			if = {
				limit = { has_law = law_type:law_non_inheritable_usage_rights }
				deactivate_law = law_type:law_non_inheritable_usage_rights
			}
			if = {
				limit = { has_law = law_type:law_free_testation }
				deactivate_law = law_type:law_free_testation
			}
			if = {
				limit = { has_law = law_type:law_state_universal_heir }
				deactivate_law = law_type:law_state_universal_heir
			}
			if = {
				limit = { local_var:te_inh_keep = 1 }
				activate_law = law_type:law_partible
				debug_log = "TE_INH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] held two inheritance laws; kept Customary Inheritance"
			}
			else_if = {
				limit = { local_var:te_inh_keep = 2 }
				activate_law = law_type:law_equal_inheritance
				debug_log = "TE_INH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] held two inheritance laws; kept Forced Heirship"
			}
			else_if = {
				limit = { local_var:te_inh_keep = 3 }
				activate_law = law_type:law_non_inheritable_usage_rights
				debug_log = "TE_INH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] held two inheritance laws; kept Possession by Use"
			}
			else_if = {
				limit = { local_var:te_inh_keep = 4 }
				activate_law = law_type:law_primogeniture
				debug_log = "TE_INH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] held two inheritance laws; kept Primogeniture"
			}
			else_if = {
				limit = { local_var:te_inh_keep = 5 }
				activate_law = law_type:law_free_testation
				debug_log = "TE_INH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] held two inheritance laws; kept Freedom of Testation"
			}
			else = {
				activate_law = law_type:law_state_universal_heir
				debug_log = "TE_INH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] held two inheritance laws; kept State as Universal Heir"
			}
		}
	}
}

# OLD-SAVE REPAIR (#1232): remove after 2026-12-31 or at vanilla 1.15, whichever comes first.
#
# The lunar colonization design (docs/superpowers/specs/2026-10-29-lunar-colonization-design.md
# §§ 5, 6) moved Solar System Colonization's method (pm_solar_colonization) and
# Orbital Assembly from Directed Energy Weapons to the new Cislunar
# Infrastructure, and the Extraplanetary Base to the new In-Situ Resource
# Utilization, which Asteroid Mining now requires. A save from before it has
# neither, so a colonizing program would lose its method's technology and a
# base holder its building's. Once, in the first month after the load, every
# country with Directed Energy Weapons and a Space Program is given Cislunar
# Infrastructure, and every country with Asteroid Mining In-Situ Resource
# Utilization. Once only: in a new game a country can hold Directed Energy
# Weapons without Cislunar Infrastructure, and a new game sets the flag in its
# first month with no one to repair. Whether the engine moves a Space Program
# off a method whose technology it lacks before this month comes is unread
# (docs/testing/solar-colonies-playtest.md).
te_repair_lunar_technologies = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = te_lunar_techs_repaired } }
			set_global_variable = te_lunar_techs_repaired
			every_country = {
				limit = {
					has_technology_researched = directed_energy_weapons
					has_technology_researched = reusable_rocketry # Cislunar Infrastructure's prerequisite
					NOT = { has_technology_researched = cislunar_infrastructure }
					sr_has_space_program = yes
				}
				add_technology_researched = cislunar_infrastructure
				debug_log = "TE_LUNAR_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Cislunar Infrastructure"
			}
			every_country = {
				limit = {
					has_technology_researched = asteroid_mining
					NOT = { has_technology_researched = in_situ_resource_utilization }
				}
				add_technology_researched = in_situ_resource_utilization
				debug_log = "TE_LUNAR_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given In-Situ Resource Utilization"
			}
		}
	}
}

# OLD-SAVE REPAIR (#1276): remove after 2027-01-31 or at vanilla 1.15, whichever comes first.
#
# Phase 1 of the era 13 design (docs/superpowers/specs/2026-10-31-era-13-technology-tree-design.md
# § 5.6) gave two moved technologies a new era 12 prerequisite: Biological
# Immortality now requires Rejuvenation Therapies, and Post-Scarcity Economy
# Post-Work Society. The engine grants no prerequisite after the fact, so a
# save from before it could hold the era 13 technology without the one under
# it. Once, in the first month after the load, each such country is given the
# missing prerequisite, and nothing else: the rule is that a repair grants a
# technology only where one the country holds now requires it. A new game sets
# the flag in its first month with no one to repair. An active Post-Scarcity
# Transition in an old save stays active and keeps its timer (§ 5.3).
#
# Phase 2 (the AI wave), under its own flag, since a save from between the
# phases has the first: Artificial General Intelligence now requires
# Self-Driving Laboratories and Humanoid Robotics, so a holder is given them and
# AI Agents under them. Content moved to the new technologies is kept where it
# is in use, by the same rule: a Brain-Computer Interfaces holder running one of
# its three old AI methods gets the method's new technology and AI Agents; a
# Generative AI holder under Automated Bureaucracy gets AI Agents; a JADC2
# holder fielding Robotic Soldiers gets Autonomous Weapons. Whether the engine
# moves a building off a method, or keeps an orphaned unit type's battalions,
# before this month comes is unread (docs/testing/era-13-playtest.md).
#
# Phase 3, Aneutronic Fusion, under its own flag: Fusion Batteries now requires
# Aneutronic Fusion, so a holder is given it. The Interstellar Probe (the
# Interstellar Mission method, its decision and the milestone's entry) moved
# from Compact Fusion Reactors to Aneutronic Fusion, so a Compact Fusion
# Reactors holder whose Space Program runs the Interstellar Mission is given it
# too. The method is the "under way" test: the running probe's entry needs its
# programme bool (possible and fail), which only this method carries, so the
# method covers every probe under way; and a programme kept on the method after
# its probe, for the colonization and Deep Space bools it also carries, is
# content the technology now gates in use as well. A probe finished and off the
# method gets nothing. If the entry deactivates before this month (its
# possible now names Aneutronic Fusion), the grant re-activates it, and its
# immediate keeps the progress unless the monthly milestone cleanup came in
# between: unread, on the play-test list.
te_repair_era_13_technologies = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = te_era_13_techs_repaired } }
			set_global_variable = te_era_13_techs_repaired
			every_country = {
				limit = {
					has_technology_researched = biological_immortality
					NOT = { has_technology_researched = rejuvenation_therapies }
				}
				add_technology_researched = rejuvenation_therapies
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Rejuvenation Therapies"
			}
			every_country = {
				limit = {
					has_technology_researched = post-scarcity_economy
					NOT = { has_technology_researched = post_work_society }
				}
				add_technology_researched = post_work_society
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Post-Work Society"
			}
		}
		if = {
			limit = { NOT = { has_global_variable = te_era_13_ai_techs_repaired } }
			set_global_variable = te_era_13_ai_techs_repaired
			# AI Agents first: the other three require it
			every_country = {
				limit = {
					NOT = { has_technology_researched = ai_agents }
					OR = {
						has_technology_researched = artificial_intelligence
						AND = {
							has_technology_researched = brain_computer_interfaces
							any_scope_building = {
								OR = {
									has_active_production_method = pm_fully_automated_content_creation
									has_active_production_method = pm_ai_assisted_research
									has_active_production_method = pm_neuromorphic_computing
								}
							}
						}
						AND = {
							has_technology_researched = generative_ai
							has_law = law_type:law_automated_bureaucracy
						}
					}
				}
				add_technology_researched = ai_agents
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given AI Agents"
			}
			every_country = {
				limit = {
					NOT = { has_technology_researched = self_driving_laboratories }
					OR = {
						has_technology_researched = artificial_intelligence
						AND = {
							has_technology_researched = brain_computer_interfaces
							any_scope_building = { has_active_production_method = pm_ai_assisted_research }
						}
					}
				}
				add_technology_researched = self_driving_laboratories
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Self-Driving Laboratories"
			}
			every_country = {
				limit = {
					NOT = { has_technology_researched = humanoid_robotics }
					OR = {
						has_technology_researched = artificial_intelligence
						AND = {
							has_technology_researched = brain_computer_interfaces
							any_scope_building = { has_active_production_method = pm_neuromorphic_computing }
						}
					}
				}
				add_technology_researched = humanoid_robotics
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Humanoid Robotics"
			}
			every_country = {
				limit = {
					has_technology_researched = jadc2
					NOT = { has_technology_researched = autonomous_weapons }
					any_military_formation = {
						any_combat_unit = { has_unit_type = unit_type:combat_unit_type_robotic_soldiers }
					}
				}
				add_technology_researched = autonomous_weapons
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Autonomous Weapons"
			}
		}
		if = {
			limit = { NOT = { has_global_variable = te_era_13_aneutronic_fusion_repaired } }
			set_global_variable = te_era_13_aneutronic_fusion_repaired
			every_country = {
				limit = {
					NOT = { has_technology_researched = aneutronic_fusion }
					OR = {
						has_technology_researched = fusion_batteries # now its prerequisite
						AND = {
							has_technology_researched = compact_fusion_reactors
							any_scope_building = { has_active_production_method = pm_interstellar_mission }
						}
					}
				}
				add_technology_researched = aneutronic_fusion
				debug_log = "TE_ERA13_TECH_REPAIR: [THIS.GetCountry.GetNameNoFormatting] given Aneutronic Fusion"
			}
		}
	}
}

# OLD-SAVE REPAIR (#1341): remove after 2027-01-31 or at vanilla 1.15, whichever comes first.
#
# National Info (docs/superpowers/specs/2026-11-02-charts-help-national-info-design.md
# § 6.6) shows another country's history charts, which draw a store (te_hist)
# in the order it is kept. Every eviction re-sorts it (te_history_prune_samples
# calls te_history_sort_samples, common/scripted_effects/te_history_effects.txt),
# so a store that stops recording is already in order. The only stores out of
# order are ones scrambled before that sort existed (#356, 2026-09-21) and not
# evicted since: a country that dropped below major power, or out of the
# entry its series belongs to, before then. Nobody could open another
# country's journal entry, so they were unreachable; National Info reaches
# them. Once, in the first month after the load, every country with a store
# sorts it. Safe for every country: a store under the cap has never evicted,
# so it is in order, and the sort's own completeness gate (count >= 240)
# leaves a shorter list as it is. The flag only saves the 2 x 240 iterations
# a month; a new game sets it in its first month with nothing to sort.
# scripts/analysis/check_save_history_order.py fails on any store out of
# order, frozen or not, so a save shows whether this held.
te_repair_history_order = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = te_history_order_repaired } }
			set_global_variable = te_history_order_repaired
			every_country = {
				limit = { has_variable_list = te_hist }
				te_history_sort_samples = yes
			}
		}
	}
}
