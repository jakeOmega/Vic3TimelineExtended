# The AI construction pass, monthly for each AI country (common/scripted_effects/te_ai_build_pass_effects.txt), and
# its self-check's verdict.
on_monthly_pulse_country = {
	on_actions = {
		te_ai_build_pass_on_action
	}
}

on_monthly_pulse = {
	on_actions = {
		te_ai_build_pass_selfcheck
	}
}

te_ai_build_pass_on_action = {
	trigger = {
		is_ai = yes
	}
	effect = {
		te_ai_build_pass_effect = yes
	}
}

# TEMPORARY self-check (2026-11-16, #1877): does a government construction the pass queues for an AI the engine would
# not queue for get built? The pass is the first thing in the mod that queues government construction (an
# administration, a power plant, a railway, a university, a highway, an airport) for an AI in debt or under the
# reserve gate, and the engine's own spending AI cuts "should-have" spending when income is
# under 1.05 times expenses with under 20 weeks of reserves (NAI MONEY_SPENDING_MAX_RATIO_TO_REMOVE_SHOULD_HAVE and
# MONEY_SPENDING_MIN_WEEKS_OF_GOLD_RESERVES_TO_NOT_REMOVE_SHOULD_HAVE), and such a government always has under 20
# weeks. Whether that removes a construction queued by script, or only lowers spending elsewhere, is unread.
#
# The sample: each level the pass queues for a country in debt or with reserves under 0.02 when it queues
# (te_ai_build_pass_in_sample), until the verdict. The pass queues nothing more for a country while its level is
# followed, so a re-queue cannot read as built (te_ai_build_pass_follow). Each is read in the state it was queued in:
# built (the state holds more levels of the kind than when it queued, a level of the kind taken down there lowering
# the base), lost (none of the kind queued there any more, unbuilt) or, still queued after 18 months (24 for a
# railway, 36 for a highway or an airport), waiting. Judged on the levels whose fate is known, built and lost, in
# all; the waiting are printed, and built and lost by kind (uni, adm, rail, power, road, air). At the first global
# monthly pulse with 12 built or lost, or 36 in all:
#   START       a plain marker, logged first: with no verdict line after it, the templated line was dropped.
#   OK          6 or more built, and lost a quarter or less of built and lost: what the pass queues is kept.
#   BROKEN      6 or more built or lost, and lost half or more of them: the engine drops what the pass queues.
#               Fallback: queue through the engine instead, lowering the gates that hold it (NAI
#               CONSTRUCTION_RESERVES_NEW_CONSTRUCTIONS to 0, and MONEY_SPENDING_MIN_GOLD_RESERVE_FRACTION_TO_ADD_SHOULD_HAVE),
#               which still cannot reach a country in debt; or place the level with create_building, as the Strategic
#               Reserve's decision does.
#   NO_VERDICT  otherwise. Mostly waiting: the purchase is not paying for the levels (a pause, or no surplus).
# Play-test row: docs/testing/ai-economy-playtest.md § The AI's reserve and credit by interest rate. Delete the
# verdict, the sample's lines in te_ai_build_pass_pending_start and te_ai_build_pass_follow, and the _g values once it
# reads (the follow itself stays: the back-off and the reserve gate's self-check read it).
te_ai_build_pass_selfcheck = {
	trigger = {
		NOT = { has_global_variable = te_ai_build_pass_checked }
		has_global_variable = te_ai_build_pass_built
		OR = {
			te_ai_build_pass_decided_g >= 12
			te_ai_build_pass_sampled_g >= 36
		}
	}
	effect = {
		set_global_variable = te_ai_build_pass_checked
		debug_log = "TE_SELF_CHECK START ai_build_pass (the verdict line follows; none after this one means the engine dropped the templated line)"
		random_country = {
			if = {
				limit = {
					te_ai_build_pass_built_g >= 6
					te_ai_build_pass_lost_share <= 0.25
				}
				debug_log = "TE_SELF_CHECK OK ai_build_pass built=[SCOPE.ScriptValue('te_ai_build_pass_built_g')|0] lost=[SCOPE.ScriptValue('te_ai_build_pass_lost_g')|0] waiting=[SCOPE.ScriptValue('te_ai_build_pass_waiting_g')|0] built/lost uni=[SCOPE.ScriptValue('te_ai_build_pass_built_university_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_university_g')|0] adm=[SCOPE.ScriptValue('te_ai_build_pass_built_administration_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_administration_g')|0] rail=[SCOPE.ScriptValue('te_ai_build_pass_built_railway_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_railway_g')|0] power=[SCOPE.ScriptValue('te_ai_build_pass_built_power_plant_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_power_plant_g')|0] road=[SCOPE.ScriptValue('te_ai_build_pass_built_highway_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_highway_g')|0] air=[SCOPE.ScriptValue('te_ai_build_pass_built_airport_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_airport_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (what the AI construction pass queued for governments in debt or under 0.02 reserves was built)"
			}
			else_if = {
				limit = {
					te_ai_build_pass_decided_g >= 6
					te_ai_build_pass_lost_share >= 0.5
				}
				debug_log = "TE_SELF_CHECK BROKEN ai_build_pass built=[SCOPE.ScriptValue('te_ai_build_pass_built_g')|0] lost=[SCOPE.ScriptValue('te_ai_build_pass_lost_g')|0] waiting=[SCOPE.ScriptValue('te_ai_build_pass_waiting_g')|0] built/lost uni=[SCOPE.ScriptValue('te_ai_build_pass_built_university_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_university_g')|0] adm=[SCOPE.ScriptValue('te_ai_build_pass_built_administration_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_administration_g')|0] rail=[SCOPE.ScriptValue('te_ai_build_pass_built_railway_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_railway_g')|0] power=[SCOPE.ScriptValue('te_ai_build_pass_built_power_plant_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_power_plant_g')|0] road=[SCOPE.ScriptValue('te_ai_build_pass_built_highway_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_highway_g')|0] air=[SCOPE.ScriptValue('te_ai_build_pass_built_airport_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_airport_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (half or more of what the pass queued was dropped unbuilt: the engine drops it; fallback in te_ai_build_pass_on_actions.txt)"
			}
			else = {
				debug_log = "TE_SELF_CHECK NO_VERDICT ai_build_pass built=[SCOPE.ScriptValue('te_ai_build_pass_built_g')|0] lost=[SCOPE.ScriptValue('te_ai_build_pass_lost_g')|0] waiting=[SCOPE.ScriptValue('te_ai_build_pass_waiting_g')|0] built/lost uni=[SCOPE.ScriptValue('te_ai_build_pass_built_university_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_university_g')|0] adm=[SCOPE.ScriptValue('te_ai_build_pass_built_administration_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_administration_g')|0] rail=[SCOPE.ScriptValue('te_ai_build_pass_built_railway_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_railway_g')|0] power=[SCOPE.ScriptValue('te_ai_build_pass_built_power_plant_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_power_plant_g')|0] road=[SCOPE.ScriptValue('te_ai_build_pass_built_highway_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_highway_g')|0] air=[SCOPE.ScriptValue('te_ai_build_pass_built_airport_g')|0]/[SCOPE.ScriptValue('te_ai_build_pass_lost_airport_g')|0] date=[TimeKeeper.GetCurrentDate.GetString] (too few built or lost, or between: read the probe's cm_pass rows)"
			}
		}
	}
}
