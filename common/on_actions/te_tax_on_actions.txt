# ============================================================================
# LEGISLATED TAX CODE — migration, monthly dispatch and watchdog
# ============================================================================
# Schedule rules: docs/systems/tax_code_schema.md, "Scheduler, package slots
# and history".
#
# Dispatch. The global on_monthly_pulse runs on the 1st of every calendar month
# (docs/guides/scripting_best_practices.md, "on_monthly_pulse_country Is a
# 30-Day Timer, Not a Calendar Month"), so a package due in a month commences
# on its 1st. The global pulse has no ROOT, and the collection writer needs the
# country as ROOT for its amendment sponsor, so each country gets the hidden
# country event te_tax.1, whose ROOT is that country. The processor claims the
# month first, so a second dispatch in the same month does nothing.
#
# Watchdog. A country's own on_monthly_pulse_country is a 30-day timer; it only
# looks for a month the dispatch has not processed while a transition is due,
# and then runs te_tax.2, which never commences and never claims the month.
#
# Migration. Nothing in the tax code runs for a country until te_tax.3 has
# migrated it (docs/systems/tax_code_schema.md, "Migration"): every country
# once the lobby is over, each formed country, a released country whose parent
# has no code, and any country the monthly dispatch still finds unmigrated. A
# released country whose parent holds the carrier law takes the parent's
# enacted code instead, and an uprising the original's whole code (the
# outbreak copy hangs off the shared te_civil_war_on_start,
# te_civil_war_on_actions.txt): docs/systems/tax_code_schema.md, "Civil wars
# and new countries".
#
# Relief eligibility. A state that changes owner keeps its regional-relief
# marks only while the code that named it went with it (on_state_owner_change,
# below).
#
# Re-assertion. A taxation law other than law_te_tax_code activated in a
# migrated country (activate_law from an event, a button or the law-consistency
# cascade) is replaced by the carrier again through te_tax.5
# (docs/systems/tax_code_schema.md, "Native controls and drift").
#
# Customs authority (plan Task 15; customs option only). A market merge, a
# new market and a change of subject status can make a country gain or lose
# its own market, so its customs authority is revalidated a day later
# (te_tax.7, te_tax_customs_revalidate); the processor also revalidates at
# every month's start, which catches any change no hook reports.
#
# Every handler checks te_tax_code_full first: with the rule off nothing here runs.
# The engine unions same-name on_action declarations across files.
#
# The light code (te_tax_code_light_on; te_tax_light_effects.txt) has its own
# handlers, te_tax_light_*, each gated on te_tax_code_light_on, on the same hooks:
# migration after the lobby, for a formed or released country and from the
# monthly dispatch (te_tax_light.2), the monthly processor (te_tax_light.1) and
# a sync after a taxation law is activated (te_tax_light.3). A released country
# starts from its own law, so its marker is cleared first. Each country's own
# monthly pulse also runs the entry's safety net (te_tax_light_entry_net, below):
# the light code's relief and reactions are je_tax_code's scaling modifiers.
# The full code has the same net (te_tax_full_entry_net): its tab opens only
# while the entry is active.
#
# Decentralized countries are left out of the three fan-outs (final review
# A-Minor 1; extra_on_actions.txt's global-warming pulse does the same): their
# type has no events, economy or politics, so te_tax.3 would never migrate
# them and the self-heal would raise it on each of them every month.
# ============================================================================

# ----------------------------------------------------------------------------
# Migration at game start
# ----------------------------------------------------------------------------
# After the lobby, not on_game_started: that fires while the players are still
# in the lobby, where the rule can still change (scripting_best_practices.md,
# "Convert what history placed after the lobby"). Migrating there and turning
# the rule off in the lobby would leave every country on the carrier law in a
# rule-off campaign. No scope here; the event makes each country its own ROOT.
on_game_started_after_lobby = {
	on_actions = {
		te_tax_on_game_started
		te_tax_light_on_game_started
	}
}

te_tax_on_game_started = {
	effect = {
		if = {
			limit = { te_tax_code_full = yes }
			every_country = {
				limit = { NOT = { is_country_type = decentralized } }
				trigger_event = { id = te_tax.3 }
			}
		}
	}
}

te_tax_light_on_game_started = {
	effect = {
		if = {
			limit = { te_tax_code_light_on = yes }
			every_country = {
				limit = { NOT = { is_country_type = decentralized } }
				trigger_event = { id = te_tax_light.2 }
			}
		}
	}
}

# The scope is the releasing country; scope:target is the released one.
# te_tax_init_released_country (te_tax_civil_war_effects.txt) runs in
# scope:target with the parent saved as scope:te_tax_source: a parent holding
# the carrier law passes on its enacted provisions, which a copy of variables
# can do from here; the carrier law and the first sync are dispatched to the
# new country as te_tax.6, and any other parent's released country gets the
# migration te_tax.3, so whatever runs as a country event has the new country
# as its ROOT and never touches the parent (te_monetary_on_actions.txt,
# "THE UPDATE IS DISPATCHED, NOT CALLED", does the same).
on_country_released_as_independent = {
	on_actions = {
		te_tax_on_country_released
		te_tax_light_on_country_released
	}
}

on_country_released_as_own_subject = {
	on_actions = {
		te_tax_on_country_released
		te_tax_light_on_country_released
	}
}

on_country_released_as_overlord_subject = {
	on_actions = {
		te_tax_on_country_released
		te_tax_light_on_country_released
	}
}

on_country_released_as_company_subject = {
	on_actions = {
		te_tax_on_country_released
		te_tax_light_on_country_released
	}
}

te_tax_on_country_released = {
	effect = {
		if = {
			limit = { te_tax_code_full = yes }
			save_scope_as = te_tax_source
			scope:target ?= {
				te_tax_init_released_country = yes
			}
		}
	}
}

# A released country starts from its own taxation law and tax level: a revived
# tag's marker from an earlier life is cleared, and te_tax_light.2 migrates it
# as its own ROOT.
te_tax_light_on_country_released = {
	effect = {
		if = {
			limit = { te_tax_code_light_on = yes }
			scope:target ?= {
				set_variable = { name = te_tax_light_on value = 0 }
				trigger_event = { id = te_tax_light.2 }
			}
		}
	}
}

# An uprising (on_revolution_start, on_secession_start) is not handled here: the
# shared te_civil_war_on_start calls te_tax_on_uprising after it has recorded
# the two sides. Same-name on_actions are unioned across files in no known
# order, so a second handler here would race the copy with a migration.

# ----------------------------------------------------------------------------
# Relief eligibility
# ----------------------------------------------------------------------------
# ROOT is the state that changed owner; the hook names no previous owner, so
# the country whose code named the state is stamped on it beforehand
# (te_tax_relief_holder). te_tax_relief_follow_owner
# (te_tax_civil_war_effects.txt) resets the state's relief marks unless that
# code went with it.
on_state_owner_change = {
	on_actions = {
		te_tax_on_state_owner_change
	}
}

te_tax_on_state_owner_change = {
	effect = {
		if = {
			limit = { te_tax_code_full = yes }
			te_tax_relief_follow_owner = yes
		}
	}
}

# ----------------------------------------------------------------------------
# Monthly dispatch and watchdog
# ----------------------------------------------------------------------------

on_monthly_pulse = {
	on_actions = {
		te_tax_monthly_dispatch
		te_tax_light_monthly_dispatch
	}
}

on_monthly_pulse_country = {
	on_actions = {
		te_tax_watchdog_on_action
		te_tax_light_entry_net
		te_tax_full_entry_net
		te_tax_rule_off_net
	}
}

# Global scope (no ROOT). The date only: the month index is printed by each
# country's te_tax.1, the same day.
te_tax_monthly_dispatch = {
	effect = {
		if = {
			limit = { te_tax_code_full = yes }
			debug_log = "TE_TAX dispatch date=[TimeKeeper.GetCurrentDate.GetString]"
			# Migrated countries: this month's processor.
			every_country = {
				limit = {
					NOT = { is_country_type = decentralized }
					has_variable = te_tax_migrated
					var:te_tax_migrated > 0
				}
				trigger_event = { id = te_tax.1 }
			}
			# Self-heal: a country no hook migrated (one created by an effect, or a
			# game-start dispatch that missed it) migrates now; its first te_tax.1
			# runs next month.
			every_country = {
				limit = {
					NOT = { is_country_type = decentralized }
					OR = { NOT = { has_variable = te_tax_migrated } var:te_tax_migrated < 1 }
				}
				trigger_event = { id = te_tax.3 }
			}
		}
	}
}

# Global scope (no ROOT). Migrated countries get this month's processor
# (te_tax_light.1); one no hook migrated migrates now.
te_tax_light_monthly_dispatch = {
	effect = {
		if = {
			limit = { te_tax_code_light_on = yes }
			every_country = {
				limit = {
					NOT = { is_country_type = decentralized }
					te_tax_light_in_force = yes
				}
				trigger_event = { id = te_tax_light.1 }
			}
			every_country = {
				limit = {
					NOT = { is_country_type = decentralized }
					NOT = { te_tax_light_in_force = yes }
				}
				trigger_event = { id = te_tax_light.2 }
			}
		}
	}
}

# Country scope (ROOT = the country).
te_tax_watchdog_on_action = {
	effect = {
		if = {
			limit = {
				te_tax_code_full = yes
				has_variable = te_tax_migrated
				var:te_tax_migrated > 0
				has_variable = te_tax_last_month
				var:te_tax_last_month < te_history_month_index
				has_variable = te_tax_next_month
				var:te_tax_next_month >= 0
				var:te_tax_next_month <= te_history_month_index
			}
			trigger_event = { id = te_tax.2 }
		}
	}
}

# Country scope. The light code's relief and the interest groups' reactions are
# je_tax_code's scaling modifiers (scaling_modifiers_while_active), which apply
# only while the entry is active, so a country with the code in force must hold
# it. The engine activates it from the inactive record every country holds
# under the rule, within two weeks of the migration; Banking's entry showed the
# engine can leave a country without a record (te_banking_entry_net,
# te_banking_entry_on_actions.txt: BROKEN missing=3 in the owner's EU game), and
# such a country would collect without its relief and lose its reactions. So
# each month this adds the entry to a migrated country with no record of it,
# active or inactive. Not while an inactive copy waits: add_journal_entry is
# refused then, with an error ("already has a journal entry of type",
# scripting_best_practices.md), and the engine activates that copy itself.
# Decentralized countries are left out, as in the dispatch: none is migrated.
te_tax_light_entry_net = {
	effect = {
		if = {
			limit = {
				te_tax_code_light_on = yes
				te_tax_light_in_force = yes
				NOT = { is_country_type = decentralized }
				NOT = { has_journal_entry = je_tax_code }
				NOT = { has_inactive_journal_entry = je_tax_code }
			}
			add_journal_entry = { type = je_tax_code }
			debug_log = "TE_TAX light_entry_net the safety net added je_tax_code; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
	}
}

# Country scope. The same net under the full code (2026-11-13). The entry is
# display only there, but the Budget panel's Tax Code tab and the Timeline
# Extended window's open only while it is active (te_budget_tax_tab_sgui), and
# the tab is where a tax is changed. A Workshop player with the full code had
# the tab greyed for ten years, its tooltip saying the conditions were met and
# the entry was not running: the code was in force and the country held no
# record of the entry, active or inactive. disable_evaluation prunes the
# inactive record while the rule is off, so a game whose rule was switched on
# during play is one way there (that this was the player's case is not
# confirmed); Banking's net found the engine leaving countries without a record
# in an ordinary game too. Nothing added the entry back, since only the light
# setting had a net. The conditions are the light net's.
te_tax_full_entry_net = {
	effect = {
		if = {
			limit = {
				te_tax_code_full = yes
				te_tax_code_in_force = yes
				NOT = { is_country_type = decentralized }
				NOT = { has_journal_entry = je_tax_code }
				NOT = { has_inactive_journal_entry = je_tax_code }
			}
			add_journal_entry = { type = je_tax_code }
			debug_log = "TE_TAX full_entry_net the safety net added je_tax_code; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
	}
}

# Country scope. The rule switched to Disabled mid-game, from the screen that
# changes country (2026-11-08). je_tax_code ends through its invalid outcome and
# takes the light code's relief and reactions along, but what the code put on
# the country stays: the carrier law under the full code, the interest groups'
# view bands, the relief modifiers and the light code's marker.
# te_tax_leave_code_for_rule takes them off, once: it clears both markers.
# This is the one negated rule check (te_tax_triggers.txt says why the others
# are positive): a save from before the rule reads as Disabled here, and is
# safe only because the check asks for a marker no such save has.
te_tax_rule_off_net = {
	effect = {
		if = {
			limit = {
				NOT = { te_tax_code_on = yes }
				OR = {
					has_law = law_type:law_te_tax_code
					AND = {
						has_variable = te_tax_migrated
						var:te_tax_migrated >= 1
					}
					AND = {
						has_variable = te_tax_light_on
						var:te_tax_light_on >= 1
					}
				}
			}
			te_tax_leave_code_for_rule = yes
			debug_log = "TE_TAX rule_off left the code; month=[SCOPE.ScriptValue('te_history_month_index')|0] date=[TimeKeeper.GetCurrentDate.GetString] country=[THIS.GetCountry.GetNameNoFormatting]"
		}
	}
}

# ----------------------------------------------------------------------------
# Re-assertion of the carrier
# ----------------------------------------------------------------------------
# on_law_activated fires for a law activated for any reason, script included
# (vanilla 00_code_on_actions.txt, the note on the hook), so it catches every
# activate_law of a vanilla taxation law: restore_peruvian_constitution
# (vanilla 00_victoria_scripted_effects.txt, Per-Capita whenever not held), the
# law-consistency cascade (extra_law_consistency_generated.txt), a taxation law
# the engine gives an uprising after the outbreak copy, and anything else.
# Hungary's Per-Capita button is not one: it activates Per-Capita only from
# Land-Based, which no country holds under the rule. The five vanilla taxation
# laws cannot be enacted under the rule (te_tax_vanilla_law_gates.txt), so
# script is the only way in. The vanilla files are left alone.
#
# ROOT is the law just activated; its owner is the country. The re-assert is
# dispatched as te_tax.5, a country event, never run here, so it sees the law
# this activation installed (precedent: te_collective_governance_from_law_scope,
# extra_on_actions.txt). Only a migrated country: at game start history and the
# political setup activate vanilla laws before the migration. The carrier's own
# activation (the migration's, or te_tax.5's) is not another taxation law, so it
# never re-dispatches.
on_law_activated = {
	on_actions = {
		te_tax_on_law_activated
		te_tax_light_on_law_activated
	}
}

te_tax_on_law_activated = {
	effect = {
		if = {
			limit = {
				te_tax_code_full = yes
				law_type = { is_same_law_group_as = law_type:law_te_tax_code }
				NOT = { law_type = law_type:law_te_tax_code }
				owner = {
					has_variable = te_tax_migrated
					var:te_tax_migrated > 0
				}
			}
			owner = {
				trigger_event = { id = te_tax.5 }
			}
		}
	}
}

# Light code: a taxation law activated in a migrated country, enacted or set by
# script. The sync is dispatched as te_tax_light.3, a country event, so it sees
# the new law and runs with the country as ROOT; it clamps the code into the
# new law's range (te_tax_light_sync, step 1).
te_tax_light_on_law_activated = {
	effect = {
		if = {
			limit = {
				te_tax_code_light_on = yes
				law_type = { is_same_law_group_as = law_type:law_proportional_taxation }
				owner = { te_tax_light_in_force = yes }
			}
			owner = {
				trigger_event = { id = te_tax_light.3 }
			}
		}
	}
}

# ----------------------------------------------------------------------------
# Customs authority (plan Task 15; docs/systems/tax_code_schema.md, "Customs
# schedule")
# ----------------------------------------------------------------------------
# Only under the customs option, and only for a country whose code is in force
# (te_tax_customs_revalidate tests both again). Each dispatches the hidden
# te_tax.7 a day later, once the merge or the change of status has settled
# (te_monetary_on_become_subject dispatches its own update the same way), so
# the country revalidates as its own ROOT.

# Root = the dissolving (joining) market; owner = the country whose market
# dissolves, which now trades in scope:market, whose owner leads it
# (gw_market_join_on_action, extra_on_actions.txt).
on_merge_markets = {
	on_actions = {
		te_tax_on_merge_markets
	}
}

te_tax_on_merge_markets = {
	effect = {
		if = {
			limit = { te_tax_code_full = yes te_tax_customs_on = yes }
			if = {
				limit = { exists = owner owner = { te_tax_code_in_force = yes } }
				owner = { trigger_event = { id = te_tax.7 days = 1 } }
			}
			if = {
				limit = { exists = scope:market.owner scope:market.owner = { te_tax_code_in_force = yes } }
				scope:market.owner = { trigger_event = { id = te_tax.7 days = 1 } }
			}
		}
	}
}

# Root = the new subject (vanilla's own block scopes top_overlord from it). A
# subject pact acts as a customs union, so the subject joins its overlord's
# market; the overlord is revalidated too.
on_become_subject = {
	on_actions = {
		te_tax_on_become_subject
	}
}

te_tax_on_become_subject = {
	effect = {
		if = {
			limit = { te_tax_code_full = yes te_tax_customs_on = yes }
			if = {
				limit = { te_tax_code_in_force = yes }
				trigger_event = { id = te_tax.7 days = 1 }
			}
			if = {
				limit = { exists = overlord overlord = { te_tax_code_in_force = yes } }
				overlord = { trigger_event = { id = te_tax.7 days = 1 } }
			}
		}
	}
}
