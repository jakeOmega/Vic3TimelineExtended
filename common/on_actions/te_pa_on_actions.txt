# Private Actors (docs/systems/mod_systems/private_actors.md): the firms' pulses. One named on-action a hook, as
# te_company_on_actions.txt.

# Each country's firms: records, turns, the quarterly step; the teardown once when the rule is off
# (te_pa_monthly, te_pa_core_effects.txt).
on_monthly_pulse_country = {
	on_actions = {
		te_pa_monthly_on_action
	}
}

te_pa_monthly_on_action = {
	effect = {
		te_pa_monthly = yes
	}
}

# Each state: the state modifiers' one refresh site (te_pa_state_monthly, te_pa_trust_effects.txt), a foundation's
# yield from what its firm's quarter wrote on the state. ROOT is the state, which a state modifier's multiplier
# resolves against (scripting_best_practices.md, "add_modifier { multiplier = var:X } Resolves Against ROOT").
on_monthly_pulse_state = {
	on_actions = {
		te_pa_state_monthly_on_action
	}
}

te_pa_state_monthly_on_action = {
	effect = {
		te_pa_state_monthly = yes
	}
}

# Once a year: which of the sectors' technologies some country holds (te_pa_gen_tech_cache, generated), read by
# every firm's technology gap. The first monthly pulse with the rule on fills it once (te_pa_monthly).
on_yearly_pulse = {
	on_actions = {
		te_pa_yearly_on_action
	}
}

te_pa_yearly_on_action = {
	trigger = {
		te_pa_on = yes
	}
	effect = {
		te_pa_gen_tech_cache = yes
		set_global_variable = { name = te_pa_tech_cached value = 1 }
	}
}

# A company disbanded: clear the firm (te_pa_clear_firm). Root = the country, scope:company = the company; beside
# the flagship cleanup in extra_on_actions.txt (te_on_company_disbanded).
on_company_disbanded = {
	on_actions = {
		te_pa_company_disbanded
	}
}

te_pa_company_disbanded = {
	effect = {
		save_scope_as = te_pa_home
		scope:company ?= {
			if = {
				limit = { has_variable = te_pa_init }
				te_pa_clear_firm = yes
			}
		}
	}
}
