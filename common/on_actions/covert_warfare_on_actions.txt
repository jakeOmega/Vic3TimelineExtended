# Covert warfare hooks into vanilla on_actions.
# on_election_campaign_end: ROOT = country that just held an election.
# We use on_actions = { } to add a named sub-action (safe, additive).
# Adding effect = { } would overwrite vanilla's BRZ election variable and coup modifier removal.

# ---- Monthly, every country: the defender's side of an exposure ----
# The target of an operation need not run a covert service of its own, so
# what an exposure leaves on it is aged here, not by je_covert_warfare's pulse
# (the single site for both; the entry used to age them, so a country without
# it saw covert_warfare.2 once per game and kept a two-year-old exposure
# "recent" for ever).
#   iw_last_exposed_age         months since covert_op_burn caught an
#                               operation against us. Only ever counts up; the
#                               command centre stops showing it past ten years
#                               (covert_last_exposed_known_sgui), and
#                               surveillance_events.5 reads it as "within two
#                               years".
#   iw_defender_event_cooldown  set with its _age by covert_warfare.2; both go
#                               after 24 months, so the event can come again.
#   iw_last_exposed_public      1 when that exposure turned the public against
#                               the interest groups friendly to the attacker
#                               (covert_backlash_public), for covert_warfare.2's
#                               text. Like the rest of the record it is only
#                               overwritten, never aged: every burn passes
#                               covert_backlash_public, which sets or removes it.
#   iw_cis_protector            the country running Counterintelligence Support
#                               here (covert_op_create). Dropped once it no longer
#                               does, whatever ended it without the break hook (a
#                               protector annexed, a pact removed by script): here
#                               and not in the protector's pulse, which a dead
#                               protector no longer runs. One protector per ally
#                               reads it, so a stale one would block the next.
#                               Asked from the protector's side with
#                               is_initiator = yes, as un_lobby_runs_campaign_on
#                               does (un_lobby_triggers.txt: the target's side
#                               did not see the initiator's one-sided pact in play;
#                               vanilla 03_lobbies.txt uses the same form).
on_monthly_pulse_country = {
	on_actions = {
		covert_warfare_monthly_country
	}
}

covert_warfare_monthly_country = {
	effect = {
		# ---- The entry's net (covert_warfare.md, "Coming back") ----
		# An ended entry doesn't come back on its own, and a record
		# disable_evaluation removed while the rule was off is gone for good;
		# add_journal_entry brings a concluded type back (Colonial Empire's
		# ce_reentry, read OK 2026-11-12). So a country with no record of the
		# entry gets it once the rule is on and the gate of the setting in
		# force is met (covert_warfare_entry_due), whether its entry ended on
		# the rule, ended early to late, or it never had one. The rule off
		# pauses the service; on again, it resumes (funding 0, Tradecraft
		# kept; the operations and networks ended with the rule).
		#
		#   1. The self-check, before anything is added this month, so it
		#      reads last month's add (iw_entry_net_added, naming the country
		#      it was set on, so a revolution's winner holding the loser's
		#      mark gives no verdict):
		#        TE_SELF_CHECK OK covert_entry_return      the entry the net
		#                                                  added is running
		#                                                  (once a game)
		#        TE_SELF_CHECK BROKEN covert_entry_return  it is not, though the
		#                                                  country is still due
		#                                                  it: it ended at once,
		#                                                  or the add failed
		#      BROKEN also holds the net off this country for a year
		#      (iw_entry_net_hold), so an entry that would end at every update
		#      comes back once a year, not once a month; the outcome's own line
		#      (covert_log_entry_end) says which branch ended it.
		#   2. The add, with one TE_COVERT_ENTRY line naming the country, the
		#      three settings as read here and whether its entry had ended
		#      before (iw_reopen_at_gate, set by the outcome).
		if = {
			limit = { has_variable = iw_entry_net_added }
			save_scope_as = iw_net_country
			if = {
				limit = { var:iw_entry_net_added ?= scope:iw_net_country }
				if = {
					limit = { has_journal_entry = je_covert_warfare }
					if = {
						limit = { NOT = { has_global_variable = iw_entry_return_ok_logged } }
						set_global_variable = iw_entry_return_ok_logged
						debug_log = "TE_SELF_CHECK OK covert_entry_return country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString] (the net added je_covert_warfare last month and it is running; logged once a game)"
					}
				}
				else_if = {
					limit = { covert_warfare_entry_due = yes }
					set_variable = { name = iw_entry_net_hold value = yes days = 365 }
					debug_log = "TE_SELF_CHECK BROKEN covert_entry_return country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] on=[THIS.ScriptValue('iw_log_rule_on')|0] late=[THIS.ScriptValue('iw_log_rule_late')|0] early=[THIS.ScriptValue('iw_log_rule_early')|0] off=[THIS.ScriptValue('iw_log_rule_off')|0] late_source=[THIS.ScriptValue('iw_log_late_source')|0] date=[TimeKeeper.GetCurrentDate.GetString] (the net added je_covert_warfare last month; it is not running and the country is still due it, so it ended at once or the add failed: the net waits a year for this country; read the TE_COVERT_ENTRY ended line before this; docs/testing/rules-off-midgame-playtest.md, Self-checks)"
				}
			}
			remove_variable = iw_entry_net_added
		}
		if = {
			limit = {
				NOT = { has_variable = iw_entry_net_hold }
				covert_warfare_entry_due = yes
			}
			if = {
				limit = { has_variable = iw_reopen_at_gate }
				debug_log = "TE_COVERT_ENTRY added country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] ended_before=yes on=[THIS.ScriptValue('iw_log_rule_on')|0] late=[THIS.ScriptValue('iw_log_rule_late')|0] early=[THIS.ScriptValue('iw_log_rule_early')|0] off=[THIS.ScriptValue('iw_log_rule_off')|0] late_source=[THIS.ScriptValue('iw_log_late_source')|0] date=[TimeKeeper.GetCurrentDate.GetString]"
				remove_variable = iw_reopen_at_gate
			}
			else = {
				debug_log = "TE_COVERT_ENTRY added country=[THIS.GetCountry.GetTagName] name=[THIS.GetCountry.GetNameNoFormatting] ended_before=no on=[THIS.ScriptValue('iw_log_rule_on')|0] late=[THIS.ScriptValue('iw_log_rule_late')|0] early=[THIS.ScriptValue('iw_log_rule_early')|0] off=[THIS.ScriptValue('iw_log_rule_off')|0] late_source=[THIS.ScriptValue('iw_log_late_source')|0] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			add_journal_entry = { type = je_covert_warfare }
			save_scope_as = iw_net_country
			set_variable = { name = iw_entry_net_added value = scope:iw_net_country }
		}
		if = {
			limit = { has_variable = iw_last_exposed_age }
			change_variable = { name = iw_last_exposed_age add = 1 }
		}
		if = {
			limit = { has_variable = iw_defender_event_cooldown_age }
			change_variable = { name = iw_defender_event_cooldown_age add = 1 }
			if = {
				limit = { var:iw_defender_event_cooldown_age >= 24 }
				remove_variable = iw_defender_event_cooldown
				remove_variable = iw_defender_event_cooldown_age
			}
		}
		if = {
			limit = {
				has_variable = iw_cis_protector
				NOT = {
					var:iw_cis_protector ?= {
						has_diplomatic_pact = { who = ROOT type = covert_counterintel_support_action is_initiator = yes }
					}
				}
			}
			remove_variable = iw_cis_protector
		}
	}
}

on_election_campaign_end = {
	on_actions = {
		covert_warfare_election_end
	}
}

covert_warfare_election_end = {
	effect = {
		# Check if any election interference operations are targeting this country
		if = {
			limit = {
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = covert_election_interference_action
					second_country = { this = ROOT }
				}
			}
			covert_op_election_confidence_effect = yes
		}
	}
}
