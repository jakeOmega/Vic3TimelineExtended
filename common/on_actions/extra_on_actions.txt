# ============================================================================
# LAW ENACTMENT PASS: banking cycle modifier cleanup
# When an economy-type law changes, remove modifiers that belonged to the old
# economy type so stale planning/council/CB-tool modifiers don't linger.
# Root = country that enacted the law. Scope is country.
# ============================================================================
on_law_enactment_pass = {
	on_actions = {
		te_banking_law_change_cleanup
	}
}

# Initialize globals before any journal-entry `possible` clause reads them.
# je_global_warming.possible reads global_var:greenhouse_gas_emissions, which
# would otherwise return 'none' until on_yearly_pulse_state first fires.
on_game_started = {
	on_actions = {
		te_init_global_state
	}
}

te_init_global_state = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = greenhouse_gas_emissions } }
			set_global_variable = {
				name = greenhouse_gas_emissions
				value = 0
			}
		}
	}
}

te_banking_law_change_cleanup = {
	effect = {
		# Only run if the banking cycle JE is active
		if = {
			limit = { has_journal_entry = je_banking_cycle }
			# Switched AWAY from command economy → remove all planning_* modifiers
			if = {
				limit = { NOT = { has_law = law_type:law_command_economy } }
				remove_banking_planning_modifiers_effect = yes
			}
			# Switched AWAY from cooperative ownership → remove all cooperative_* modifiers
			if = {
				limit = { NOT = { has_law = law_type:law_cooperative_ownership } }
				remove_banking_cooperative_modifiers_effect = yes
			}
			# Switched TO command economy or cooperative → remove all market CB-tool modifiers
			if = {
				limit = {
					OR = {
						has_law = law_type:law_command_economy
						has_law = law_type:law_cooperative_ownership
					}
				}
				remove_banking_market_modifiers_effect = yes
			}
			# Left fiat or digital currency with open-market operations still running.
			# OMO is fiat/digital only (monetary_policy_design.md §5.1 and the §0.1
			# ruling of 2026-09-20), enforced in banking_possible_cb_open_market_ops,
			# but `possible` only gates the click — a country that enacts commodity
			# money, the gold standard or crypto afterwards kept the tool, and its four
			# intervention points, until it switched it off by hand.
			#
			# WHY THE TWO-PART TEST. This on-action fires while the new law is still
			# "considered being enacted" (vanilla 00_code_on_actions.txt:4835-4841), so
			# the country still holds the OLD law here and
			# modifier:country_can_create_unbacked_money_bool still reads the OLD
			# regime's value on the very enactment that takes it away. is_enacting_law
			# names the law that is about to land, which is exactly how vanilla's own
			# block at this hook clears Brazil's slave-trade modifiers (old law via
			# has_law_or_variant + new law via is_enacting_law). The bool read then
			# covers every other route out of fiat/digital — a save from before this
			# rule, a law set directly by script, any later enactment. ADD A LAW HERE
			# if a new lawgroup_monetary_policy law that does NOT grant
			# country_can_create_unbacked_money_bool is ever introduced.
			#
			# It runs through banking_effect_cb_disable_open_market_ops, the same effect
			# cb_disable_open_market_ops and the dashboard call, not a bare
			# remove_modifier: the points come back either way (they are the modifier's
			# own country_banking_intervention_max_add = -4), but only the shared effect
			# also records the off_cb_open_market_ops history marker, so an automatic
			# switch-off draws on the chart exactly like a manual one.
			#
			# else_if on purpose: a country ALREADY IN a command or cooperative economy
			# (the branch above reads the old law, per the timing note, so that is the
			# case it catches) has just had banking_open_market_ops stripped there, and
			# that removal is not visible inside this same effect block — the
			# has_modifier read behind banking_tool_omo_active would still say yes and
			# post a second, spurious marker. A country *entering* one of those
			# economies falls through to here instead, which is also right: if it has a
			# legacy OMO and no unbacked-money regime, switching it off is the whole
			# point of this branch.
			#
			# te_mon_full_system: the regime rule this branch enforces is the monetary
			# layer's, and banking_possible_cb_open_market_ops drops it under the
			# simplified game rule. Switching the tool off here anyway would take it
			# away from a country the `possible` gate still lets click it.
			else_if = {
				limit = {
					te_mon_full_system = yes
					banking_tool_omo_active = yes
					OR = {
						modifier:country_can_create_unbacked_money_bool = no
						is_enacting_law = law_type:law_commodity_money
						is_enacting_law = law_type:law_gold_standard
						is_enacting_law = law_type:law_decentralized_cryptocurrency
					}
				}
				banking_effect_cb_disable_open_market_ops = yes
			}
		}

		# Monetary policy (§14). A command economy has no dial and no delegation
		# — the rate is administered — so entering one clears the delegation
		# toggle. This sits OUTSIDE the journal-entry check above on purpose:
		# the monetary plumbing runs for every country, with or without
		# je_banking_cycle and with or without banking_system_enabled.
		#
		# Only the toggle is cleared. No monetary variable is ever removed here:
		# te_rate_paid_applied backs a live add_modifier multiplier, and the
		# rest are re-derived by the next monthly update anyway.
		#
		# THE TWO-PART TEST is the OMO branch's above and the monetisation
		# branch's below, for the same timing reason: this on-action fires while
		# the new law is still "considered being enacted", so te_mon_is_command
		# reads the OLD law and on the very enactment that brings the plan in it
		# answers "no". is_enacting_law names the law about to land, which is the
		# case the comment above describes — "entering one clears the delegation
		# toggle" — and without it the toggle only cleared for a country that was
		# already a command economy and happened to enact some other law later.
		if = {
			limit = {
				OR = {
					te_mon_is_command = yes
					is_enacting_law = law_type:law_command_economy
				}
				has_variable = te_mon_delegated
			}
			set_variable = { name = te_mon_delegated value = 0 }
		}

		# Monetisation (§11, §14). Printing money to pay the government needs a
		# currency the state can create, a bank to create it through and a
		# government still able to give that bank the order, so a law that takes
		# any of the three away takes the stepper with it — and it should do so
		# the moment the law lands, not at the next monthly pulse, because the
		# level is charged in inflation pressure and in the risk premium every
		# month it stands.
		#
		# THE TWO-PART TEST is the OMO branch's, for the same timing reason: this
		# on-action fires while the new law is still "considered being enacted",
		# so te_mon_can_monetise still reads the OLD laws on the very enactment
		# that disqualifies the country. is_enacting_law names the law about to
		# land, and the list is the exact inverse of te_mon_can_monetise's four
		# conditions — the currency (three ways out of fiat/digital), the bank,
		# independence, and the planned economy. ADD A LAW HERE if a new one that
		# blocks monetisation is ever introduced; te_mon_can_monetise in
		# common/scripted_triggers/te_monetary_triggers.txt is the other half.
		#
		# The bare trigger read covers every other route — a save from before
		# phase 2, a law set directly by script, dollarisation (ruling P9, which
		# is a variable and no law at all).
		#
		# ONLY THE LEVEL IS WRITTEN. No variable is removed (§16.2), and the
		# te_monetisation_minting modifier is left to step 1b of the next pulse,
		# which owns it: that is the same "write the input and let the pulse pick
		# it up" rule every other control in this system follows, and the cost is
		# at most one month of minting the country has already been charged the
		# premium for.
		if = {
			limit = {
				has_variable = te_monetisation_level
				NOT = { var:te_monetisation_level = 0 }
				OR = {
					te_mon_can_monetise = no
					is_enacting_law = law_type:law_commodity_money
					is_enacting_law = law_type:law_gold_standard
					is_enacting_law = law_type:law_decentralized_cryptocurrency
					is_enacting_law = law_type:law_no_national_bank
					is_enacting_law = law_type:law_central_bank_independence
					is_enacting_law = law_type:law_command_economy
				}
			}
			set_variable = { name = te_monetisation_level value = 0 }
		}

		# Dollarisation (§9.2, ruling P9). A country that abandoned its own money
		# in a hyperinflation would carry te_mon_dollarised for ever otherwise:
		# the spec names no exit. THERE ARE TWO, and they are one if/else_if pair
		# because the flag can only be cleared once and the price for clearing it
		# is not the same:
		#
		#   * ENTERING A COMMAND ECONOMY — free (final-review finding M8);
		#   * ENACTING A MONETARY-POLICY LAW — pays te_mon_effect_new_currency.
		#
		# NO MODIFIER IS REMOVED IN EITHER. te_mon_dollarised_modifier is the
		# seventh member of the te_inflation_band_* family and belongs to step 6c
		# of the monthly update (te_monetary_apply_inflation_band), which takes it
		# off and puts the country's current band on in the same swap. A
		# remove_modifier here would be a second owner of that modifier, and would
		# leave te_inflation_band_applied still claiming it was applied. Clearing
		# the flag IS what releases it, on the next pulse — which is the whole of
		# finding M8: law_command_economy lives in lawgroup_economic_system, not
		# the monetary group, so the five-law list below never saw it and a
		# dollarised country that went over to a planned economy carried
		# te_inflation_band_applied = 7 and its -75% minting for the rest of the
		# campaign.
		#
		# THE COMMAND EXIT IS FREE, AND TESTED FIRST. An administered currency is
		# not a market re-founding: the plan decrees what money is and what it
		# buys, so there is nobody to persuade to hold it, no ten-year premium to
		# charge, and no reset to 5 to write (step 6's command branch holds the
		# inflation figures at 0 anyway, §14). Testing it first is also what makes
		# the overlap deterministic — a command economy enacting a monetary law is
		# still administering its currency, so it takes this exit rather than the
		# priced one. The te_mon_is_command half of the OR is the same two-part
		# test the delegation and monetisation blocks above use, and here it
		# catches only the leftovers (a save made before this rule, or a
		# dollarised planned economy enacting some other law); is_enacting_law is
		# the live case.
		if = {
			limit = {
				has_variable = te_mon_dollarised
				var:te_mon_dollarised = 1
				OR = {
					te_mon_is_command = yes
					is_enacting_law = law_type:law_command_economy
				}
			}
			set_variable = { name = te_mon_dollarised value = 0 }
		}
		# THE MONETARY-LAW EXIT PAYS THE NEW-CURRENCY PRICE. Enacting a
		# monetary-policy law is the smallest exit the spec leaves room for — it
		# is the act of declaring what the country's currency now is, which is
		# precisely the thing dollarisation gave up. All five laws of
		# lawgroup_monetary_policy count, because a law cannot be enacted by a
		# country that already holds it: any of them IS a different one. ADD A LAW
		# HERE if lawgroup_monetary_policy ever gains a sixth.
		#
		# Leaving dollarisation this way is not putting a currency back in a
		# drawer, it is issuing one: the state has to persuade people to hold
		# paper it prints again, having already failed to once. Without the price
		# the sequence dollarise -> wait out one law enactment -> re-enact fiat
		# would be a currency reform with no pool wipe, no radicals and no premium
		# — strictly the cheapest way out of a hyperinflation, and free, because
		# the dollarise option already reset expectations to the anchor for
		# nothing. te_mon_effect_new_currency is the same effect te_inflation.1's
		# reform option calls: inflation and expectations start at
		# te_mon_currency_reform_reset and lenders charge the ten-year premium.
		# NOT the pool wipe and NOT the radicals — the months of -75% minting
		# under dollarisation were that half of the price, and te_inflation.1.b.tt
		# tells the player so before they choose.
		#
		# It is the only write of inflation state outside the monthly update and
		# the crisis event; the variable contract in the header of
		# common/script_values/te_monetary_script_values.txt lists it.
		else_if = {
			limit = {
				has_variable = te_mon_dollarised
				var:te_mon_dollarised = 1
				OR = {
					is_enacting_law = law_type:law_commodity_money
					is_enacting_law = law_type:law_gold_standard
					is_enacting_law = law_type:law_fiat_currency
					is_enacting_law = law_type:law_digital_currency
					is_enacting_law = law_type:law_decentralized_cryptocurrency
				}
			}
			set_variable = { name = te_mon_dollarised value = 0 }
			te_mon_effect_new_currency = yes
		}
	}
}

# PM RETOOLING - schedule country events at day 0/7/14/21 from the monthly
# country pulse. Each event iterates all states and removes pm_retooling where
# it is waived: level-0 building sites, and every building in games whose
# free_market_construction_rule setting waives retooling
# (te_remove_waived_pm_retooling). The on_start_expanding_building and
# on_production_method_changed hooks in te_construction_market_on_actions.txt

	effect = {
		trigger_event = { id = minor_events_timelineextended.100 }
		trigger_event = { id = minor_events_timelineextended.100 days = 7 }
		trigger_event = { id = minor_events_timelineextended.100 days = 14 }
		trigger_event = { id = minor_events_timelineextended.100 days = 21 }
	}
}

# When command economy activates, force-enact state-owned banking.
# ROOT = law scope; owner = country.
banking_law_cascade_on_law_activated = {
	effect = {
		if = {
			limit = {
				type = law_type:law_command_economy
				owner = {
					NOT = { has_law = law_type:law_state_owned_banking }
				}
			}
			owner = {
				activate_law = law_type:law_state_owned_banking
			}
		}
	}
}

# NOTE: the building and investment hooks that used to refresh the state
# modifiers here were removed, and so were the law/technology ones (see the
# STATE MODIFIER UPDATES ARE PULSE-ONLY block further down).
# add_modifier = { multiplier = script_value } evaluates the script value
# through the parent scope chain, and building / institution / law scopes
# don't support the variables and modifiers those values read, which caused
# cascading errors. The periodic state pulses are the only refresh sites now:
# pollution and tourism monthly, migration crowding and free-port tariffs
# yearly.

# ============================================================================
# BUILDING BUILT (Root = Building)
# When a barracks is built, a new combat unit type enters a formation.
# Recalculate combined arms bonuses immediately for the owning country.
# ============================================================================
on_building_built = {
	on_actions = {
		te_on_building_built
	}
}
te_on_building_built = {
	effect = {
		if = {
			limit = {
				is_building_type = building_barrack
				exists = owner
				owner = { modifier:country_combined_arms_bonus_enabled_bool = yes }
			}
			owner = {
				apply_combined_arms_bonuses = yes
			}
		}
	}
}

# ============================================================================
# CHARACTER RECRUITMENT (Root = Character)
# When a character is recruited:
# 1. Apply tech-based traits immediately (immortality, logistics)
# 2. If a general/admiral, recalculate combined arms (new commander)
# ============================================================================
on_character_recruitment = {
	on_actions = {
		te_on_character_recruitment
	}
}
te_on_character_recruitment = {
	effect = {
		# Combined-arms recompute walks every general; non-general recruits
		# (politicians, agitators, admirals, executives) don't change the
		# output, so skip the work. Matches the narrowing already applied
		# in te_on_building_built (is_building_type = building_barrack) and
		# te_on_military_formation_created. Pure perf trim, behavior-neutral.
		if = {
			limit = { has_role_of_type = general }
			owner = {
				if = {
					limit = { modifier:country_combined_arms_bonus_enabled_bool = yes }
					apply_combined_arms_bonuses = yes
				}
			}
		}
	}
}

# ============================================================================
# STATE OWNER CHANGE (Root = state)
# When a state changes hands (e.g. conquest), remove building_space_program
# if it exists and this state is not the new owner's market capital.

	}
}

# Fires from on_law_activated (ROOT = law scope).
# Checks if the activated law is radical and the government is not exempt.
# Identifies the specific radical law via a variable so the event can
# react to exactly what was enacted (not what the country already has).
radical_law_backlash_on_action = {
	effect = {
		if = {
			limit = {
				is_radical_law_activated = yes
				owner = {
					is_exempt_from_radical_backlash = no
					NOT = { has_variable = ch_hegemonic_law_backlash_immunity }
				}
			}
			# Identify which radical law was activated (ROOT = law scope)
			if = {
				limit = { law_type = law_type:law_communal_child_rearing }
				owner = { set_variable = { name = radical_law_backlash_type value = 1 } }
			}
			else_if = {
				limit = { law_type = law_type:law_state_atheism }
				owner = { set_variable = { name = radical_law_backlash_type value = 2 } }
			}
			else_if = {
				limit = { law_type = law_type:law_cooperative_ownership }
				owner = { set_variable = { name = radical_law_backlash_type value = 3 } }
			}
			else_if = {
				limit = { law_type = law_type:law_command_economy }
				owner = { set_variable = { name = radical_law_backlash_type value = 4 } }
			}
			else_if = {
				limit = { law_type = law_type:law_collectivized_agriculture }
				owner = { set_variable = { name = radical_law_backlash_type value = 5 } }
			}
			owner = {
				trigger_event = { id = extra_law_events.36 }
			}
		}
	}
}

# Root = the dissolving (joining) market, scope:market = the absorbing market.
# owner = the country whose market is dissolving (new market member).
# scope:market.owner = the country that owns the absorbing market (the market leader).
gw_market_join_on_action = {
	effect = {
		if = {
			limit = {
				exists = owner
				exists = scope:market.owner
			}
			owner = {
				# Apply any active market-leader policies from the absorbing market's leader.
				gw_copy_market_policy_modifiers = yes
			}
		}
	}
}

remove_invalid_buildings = {
	effect = {
		# Two separate iterations on purpose: an iterator's `limit` filters the
		# whole iteration no matter where it sits in the block, so a single
		# every_scope_state with the space-program limit would gate the company
		# sweep as well (it did, until 2026-09-12).
		every_scope_state = {
			remove_invalid_company_buildings_effect = yes
		}
		every_scope_state = {
			limit = {
				AND = {
					NOT = { owner.market_capital = THIS }
					has_building = building_space_program
				}
			}
			remove_building = building_space_program
		}
	}
}

global_warming_update_on_action = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = greenhouse_gas_emissions }
			}
			set_global_variable = {
				name = greenhouse_gas_emissions
				value = 0
			}
		}
		# With the rule disabled there is no climate change: emissions never
		# accumulate, so temperature_anomaly_display stays 0 for every reader
		# (UN docket, election events, movements, treaty article 109).
		if = {
			limit = {
				has_game_rule = global_warming_enabled
				owner.market_capital = THIS
			}
			owner = {
				# Snapshot first, accumulate from the snapshot. The market-wide
				# emissions sweep is evaluated exactly once, as before, and the
				# journal-entry widget then reads the stored figure instead of
				# recomputing it in loc every frame. Variable writes are visible
				# later in the same effect block, so the number added to the
				# global total is identical to what this on_action added before.
				gw_snapshot_market_emissions_effect = yes
				change_global_variable = {
					name = greenhouse_gas_emissions
					add = var:gw_disp_market_emis
				}
			}
		}
	}
}

global_warming_events_on_action = {
	effect = {
		# Display snapshots for the journal-entry widget. Both are here because
		# this on_action is the one global-scope monthly pulse, so each runs
		# exactly once for the whole world rather than once per country.
		# gw_refresh_global_counts_effect is a single country sweep that fills
		# all eight adoption counters; gw_rebase_annual_emissions_effect is O(1)
		# and only does work in January. Neither has a gameplay effect.
		# Everything climate-specific in this pulse is gated on the rule: with it
		# disabled there are no greenhouse figures to show and no threshold events.
		if = {
			limit = { has_game_rule = global_warming_enabled }
			gw_rebase_annual_emissions_effect = yes
			gw_refresh_global_counts_effect = yes

			# Fire warming-threshold environmentalism events exactly once globally when
			# temperature_anomaly_display crosses each threshold upward. Runs in the
			# global (world) scope of on_monthly_pulse — every_country inside the
			# helper reaches all countries without per-country duplication.
			gw_fire_warming_threshold_event = { THRESHOLD = 0.5 FLAG = gw_mild_event_fired         EVENT_ID = environmentalism_events.1 }
			gw_fire_warming_threshold_event = { THRESHOLD = 1.0 FLAG = gw_moderate_event_fired     EVENT_ID = environmentalism_events.2 }
			gw_fire_warming_threshold_event = { THRESHOLD = 2.0 FLAG = gw_severe_event_fired       EVENT_ID = environmentalism_events.3 }
			gw_fire_warming_threshold_event = { THRESHOLD = 3.0 FLAG = gw_catastrophic_event_fired EVENT_ID = environmentalism_events.4 }

			# Cooling / reversal threshold events: fire once globally when temperatures
			# move back below a threshold, gated on the matching warming flag so cooling
			# only fires if the warming side already fired earlier in the run.
			gw_fire_cooling_threshold_event = { THRESHOLD = 3.0 PREREQ = gw_catastrophic_event_fired FLAG = gw_cooling_below_3_event_fired       EVENT_ID = environmentalism_events.17 }
			gw_fire_cooling_threshold_event = { THRESHOLD = 2.0 PREREQ = gw_severe_event_fired       FLAG = gw_cooling_below_2_event_fired       EVENT_ID = environmentalism_events.18 }
			gw_fire_cooling_threshold_event = { THRESHOLD = 1.0 PREREQ = gw_moderate_event_fired     FLAG = gw_cooling_below_1_event_fired       EVENT_ID = environmentalism_events.19 }
			gw_fire_cooling_threshold_event = { THRESHOLD = 0.5 PREREQ = gw_mild_event_fired         FLAG = gw_cooling_below_0_5_event_fired     EVENT_ID = environmentalism_events.20 }
			gw_fire_cooling_threshold_event = { THRESHOLD = 0.1 PREREQ = gw_mild_event_fired         FLAG = gw_cooling_near_baseline_event_fired EVENT_ID = environmentalism_events.21 }
		}
		else = {
			# Save compatibility: before the rule gated accumulation, a disabled-rule
			# game still built up emissions. Hold the global at 0 so no reader of
			# temperature_anomaly_display acts on that phantom warming. The cooling
			# events cannot fire from the drop, since they are gated above.
			set_global_variable = {
				name = greenhouse_gas_emissions
				value = 0
			}
		}

		# Recurring random events for countries experiencing global warming.
		# Each country has a small monthly chance of getting one event.
		# Event triggers are checked inside the event definitions themselves.
		every_country = {
			limit = {
				NOT = { is_country_type = decentralized }
				has_global_variable = greenhouse_gas_emissions
			}
			random_list = {
				1800 = { } # ~90% chance of nothing happening this month
				# Crop failures (severe+ warming, farming country)
				15 = {
					trigger = {
						temperature_anomaly_display >= 2.0
						any_scope_state = {
							is_incorporated = yes
							any_scope_building = { is_building_type = building_wheat_farm }
						}
					}
					trigger_event = { id = environmentalism_events.5 }
				}
				# Climate refugees (moderate+ warming)
				15 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 1.0
					}
					trigger_event = { id = environmentalism_events.6 }
				}
				# Pollution scandal (has coal/oil industry)
				15 = {
					trigger = {
						any_scope_state = {
							is_incorporated = yes
							any_scope_building = {
								OR = {
									is_building_type = building_coal_mine
									is_building_type = building_oil_rig
								}
								level >= 3
							}
						}
					}
					trigger_event = { id = environmentalism_events.7 }
				}
				# Environmentalist movement (mild+ warming, has pollution_control)
				15 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 0.5
						has_technology_researched = pollution_control
					}
					trigger_event = { id = environmentalism_events.8 }
				}
				# Coastal flooding (moderate+ warming, coastal ports)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 1.0
						any_scope_state = {
							is_incorporated = yes
							is_coastal = yes
							has_building = building_port
						}
					}
					trigger_event = { id = environmentalism_events.9 }
				}
				# Water shortage (severe+ warming)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 2.0
					}
					trigger_event = { id = environmentalism_events.10 }
				}
				# Green tech boom (environmental_movement tech + renewables)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						has_technology_researched = environmental_movement
						any_scope_state = {
							has_building = building_renewable_energy_plant
						}
					}
					trigger_event = { id = environmentalism_events.11 }
				}
				# Fossil fuel lobby (has emission policies + large fossil industry)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						OR = {
							je:je_global_warming = { has_modifier = carbon_tax_modifier }
							je:je_global_warming = { has_modifier = emission_standards_modifier }
							je:je_global_warming = { has_modifier = fossil_fuel_divestment_modifier }
						}
						any_scope_state = {
							any_scope_building = {
								OR = {
									is_building_type = building_coal_mine
									is_building_type = building_oil_rig
								}
								level >= 5
							}
						}
					}
					trigger_event = { id = environmentalism_events.12 }
				}
				# Climate summit (major+ power, moderate+ warming). The UN's climate
				# accord (un_events.17) owns this diplomacy while a UN exists, so the
				# summit is the fallback for a world without one.
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 1.0
						country_rank >= rank_value:major_power
						has_technology_researched = pollution_control
						OR = {
							has_game_rule = united_nations_disabled
							NOT = { has_global_variable = un_founded }
						}
					}
					trigger_event = { id = environmentalism_events.13 }
				}
				# Disease outbreak (severe+ warming)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 2.0
					}
					trigger_event = { id = environmentalism_events.14 }
				}
				# Political clash (clean_energy_technologies + industrialists in gov + catastrophic warming)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 3.0
						has_technology_researched = clean_energy_technologies
						ig:ig_industrialists ?= { is_in_government = yes }
					}
					trigger_event = { id = environmentalism_events.15 }
				}
				# Melting ice caps (high warming, coastal)
				10 = {
					trigger = {
						has_journal_entry = je_global_warming
						je:je_global_warming = { has_modifier = global_warming }
						temperature_anomaly_display >= 2.5
						any_scope_state = { is_coastal = yes }
					}
					trigger_event = { id = environmentalism_events.16 }
				}
			}
		}
	}
}

# an added guard) would silently fight. One modifier, one refresh site.

# months stale.

# requirement the old yearly comment cited: the tourism script values use
# state_region = s:STATE_X comparisons that only resolve in state scope
# (which is why tourism is refreshed here and never from a law / technology /
# building hook).

# ---- Construction Cost Scaling (country scope, yearly) ----
# Applies goods_input_construction_mult penalty based on GDP per capita.
# Uses adjusted multiplier that combines multiplicatively with other
# sources of goods_input_construction_mult (ministry, laws, etc.).
# See extra_script_values.txt for tuning constants.
construction_cost_scaling_on_action = {
	effect = {
		# free_market_construction_rule disabled: there is no construction good
		# to scale, so the scaling goes on vanilla's own lever for the cost of
		# the goods construction consumes (construction_cost_scaling_direct).
		# The rule is fixed for the whole game, so the two branches never share
		# var:prior_construction_scaling.
		if = {
			limit = { te_free_market_construction_off = yes }
			# Compute before removing, so the modifier read certainly still holds
			# the contribution var:prior_construction_scaling backs out.
			set_variable = { name = prior_construction_scaling value = construction_cost_scaling_direct_adjusted_mult }
			remove_modifier = construction_cost_scaling_direct
			if = {
				limit = { var:prior_construction_scaling > 0 }
				add_modifier = { name = construction_cost_scaling_direct multiplier = var:prior_construction_scaling }
			}
		}
		else = {
			remove_modifier = construction_cost_scaling
			if = {
				limit = { NOT = { has_variable = prior_construction_scaling } }
				# First run (new save or pre-adjustment saves) - use raw scaling
				if = {
					limit = { construction_cost_scaling_mult > 0 }
					add_modifier = { name = construction_cost_scaling multiplier = construction_cost_scaling_mult }
					set_variable = { name = prior_construction_scaling value = construction_cost_scaling_mult }
				}
			}
			else = {
				# Subsequent runs - use adjusted scaling (multiplicative with other modifiers)
				if = {
					limit = { construction_cost_scaling_adjusted_mult > 0 }
					add_modifier = { name = construction_cost_scaling multiplier = construction_cost_scaling_adjusted_mult }
					set_variable = { name = prior_construction_scaling value = construction_cost_scaling_adjusted_mult }
				}
				else = {
					set_variable = { name = prior_construction_scaling value = 0 }
				}
			}
		}
	}
}

# ============================================================================
# STATE MODIFIER UPDATES ARE PULSE-ONLY - no immediate tech/law/building hook
# ============================================================================
# There used to be te_modifier_update_on_technology_on_action and
# te_modifier_update_on_law_on_action here, dispatched from
# on_acquired_technology / on_law_activated so that migration crowding and
# friends responded the same tick a tech or law landed. Both had been
# emptied to `effect = { }` and were removed (2026-09-13) along with their
# dispatch entries; the reasons they were emptied still apply:
#
# - All state modifier updates (migration crowding, pollution, free-port
#   tariffs) use has_variable in their script values. On on_acquired_technology
#   (ROOT = country) the engine traces the parent scope chain through that root
#   and emits "Value of wrong type: none" for the has_variable checks; on
#   on_law_activated (ROOT = law type, owner = enacting country) the law scope
#   doesn't support variables at all, producing "has_variable trigger" errors.
# - Tourism additionally uses state_region = s:STATE_X comparisons, which only
#   resolve in genuine state scope - even wrapped in every_scope_state, the
#   engine still validates against the parent scope.
# - The investment hook was removed for the same reason (institution scope
#   supports neither variables nor modifiers).
#
# All of these are instead refreshed from the periodic state pulses:
# pollution_on_action on on_monthly_pulse_state, migration_crowding_on_action
# and free_port_tariff_update_on_action on on_yearly_pulse_state (within
# 1 year), and tourism from tourism_on_action on on_monthly_pulse_state.
# Do not re-add an immediate hook for them - add the work to a state pulse
# instead.

#==================================================
# Overbuild Protection - removes excess sub-buildings
# when slot modifier goes negative (race condition fix).
# Grace period: buildings are only removed after ~6 consecutive
# months over the limit. The counter resets if the modifier
# returns to zero or positive.
#==================================================

overbuild_protection_on_action = {
	effect = {
		# --- Solar receivers ---
		if = {
			limit = { modifier:country_solar_receiver_max_level_add < 0 }
			# Increment grace counter
			if = {
				limit = { NOT = { has_variable = overbuild_solar_receiver_months } }
				set_variable = { name = overbuild_solar_receiver_months value = 1 }
			}
			else = {
				change_variable = { name = overbuild_solar_receiver_months add = 1 }
			}
			# After 6 months, start removing 1 excess level per month
			if = {
				limit = { var:overbuild_solar_receiver_months >= 6 }
				ordered_scope_state = {
					limit = { has_building = building_solar_receiver }
					order_by = market.mg:electricity.market_goods_cheaper
					max = 1
					set_variable = {
						name = lvl_tmp
						value = b:building_solar_receiver.level
					}
					change_variable = {
						name = lvl_tmp
						add = -1
					}
					remove_building = building_solar_receiver
					if = {
						limit = { var:lvl_tmp > 0 }
						te_construction_market_add_specified_level = {
							ADD_LEVEL = var:lvl_tmp
							SPEC_TYPE = building_solar_receiver
						}
					}
					remove_variable = lvl_tmp
				}
			}
		}
		else = {
			remove_variable = overbuild_solar_receiver_months
		}

		# --- Antimatter sub-buildings ---
		if = {
			limit = { modifier:country_antimatter_facility_max_level_add < 0 }
			# Increment grace counter
			if = {
				limit = { NOT = { has_variable = overbuild_antimatter_months } }
				set_variable = { name = overbuild_antimatter_months value = 1 }
			}
			else = {
				change_variable = { name = overbuild_antimatter_months add = 1 }
			}
			# After 6 months, start removing 1 excess level per month
			if = {
				limit = { var:overbuild_antimatter_months >= 6 }
				random_scope_state = {
					limit = { has_building = building_antimatter_engine }
					set_variable = {
						name = lvl_tmp
						value = b:building_antimatter_engine.level
					}
					change_variable = {
						name = lvl_tmp
						add = -1
					}
					remove_building = building_antimatter_engine
					if = {
						limit = { var:lvl_tmp > 0 }
						te_construction_market_add_specified_level = {
							ADD_LEVEL = var:lvl_tmp
							SPEC_TYPE = building_antimatter_engine
						}
					}
					remove_variable = lvl_tmp
				}
				random_scope_state = {
					limit = { has_building = building_antimatter_warhead_plant }
					set_variable = {
						name = lvl_tmp
						value = b:building_antimatter_warhead_plant.level
					}
					change_variable = {
						name = lvl_tmp
						add = -1
					}
					remove_building = building_antimatter_warhead_plant
					if = {
						limit = { var:lvl_tmp > 0 }
						te_construction_market_add_specified_level = {
							ADD_LEVEL = var:lvl_tmp
							SPEC_TYPE = building_antimatter_warhead_plant
						}
					}
					remove_variable = lvl_tmp
				}
			}
		}
		else = {
			remove_variable = overbuild_antimatter_months
		}
	}
}

# IRREDENTIST PRESSURE EVENTS — monthly chance for eligible same-culture
# pairs. Fires Event 1 ("Diaspora Calls") for warlike/general candidates,
# Event 3 ("Opportunity for Union") for peaceful-driver candidates with
# high relations. Total fire rate ~3.5%/month per active country.
#
# See events/irredentism_events.txt and the design plan at
# /home/jakef/.claude/plans/can-we-make-a-sparkling-pine.md.

# Operation Polo dispatcher: when a formed India (BHA) finds an independent,
# sub-minor-power, Muslim-ruled state surviving bordered by Indian territory —
# the holdout the formation religion guard (can_be_unification_target) leaves
# behind — offer the annex/accede/decline event (formation_te.10). Gated to
# c:BHA only, so the any_country sweep is cheap. A 2-year cooldown variable is
# set when the event resolves.
operation_polo_on_action = {
	effect = {
		if = {
			limit = {
				c:BHA ?= this
				NOT = { has_variable = operation_polo_cooldown }
				any_country = {
					NOT = { this = c:BHA }
					NOT = { is_country_type = decentralized }
					is_subject = no
					NOT = { country_rank >= rank_value:minor_power }
					OR = {
						country_has_state_religion = rel:sunni
						country_has_state_religion = rel:shiite
						country_has_state_religion = rel:ibadi
					}
					any_scope_state = {
						any_neighbouring_state = { owner = c:BHA }
					}
				}
			}
			random_country = {
				limit = {
					NOT = { this = c:BHA }
					NOT = { is_country_type = decentralized }
					is_subject = no
					NOT = { country_rank >= rank_value:minor_power }
					OR = {
						country_has_state_religion = rel:sunni
						country_has_state_religion = rel:shiite
						country_has_state_religion = rel:ibadi
					}
					any_scope_state = {
						any_neighbouring_state = { owner = c:BHA }
					}
				}
				save_scope_as = polo_target_country
			}
			trigger_event = { id = formation_te.10 popup = yes }
		}
	}
}

# ============================================================================
# IRREDENTIST BLOC APPROVAL — fires Events 5a/5b on bloc leaders when one of
# their non-leader members would activate the war or peace path but lacks
# the corresponding approval var. Low fire rate (~1.5%/month combined per
# eligible non-leader member).
#
# 5a: bloc member is a strong irredentist candidate (warlike-tilted) without
#     war approval. Bloc leader endorses or vetoes.
# 5b: bloc member has just received an irr_offered_union_modifier (proposed
#     to via Event 3) without peace approval. Bloc leader blesses or forbids.
# ============================================================================
irredentism_bloc_approval_on_action = {
	effect = {
		# 5a: warlike client asks bloc leader to bless a war
		if = {
			limit = {
				NOT = { is_country_type = decentralized }
				has_technology_researched = decolonization
				is_in_power_bloc = yes
				is_power_bloc_leader = no
				irredentist_unifier_can_act = yes
				NOT = { has_modifier = irr_war_vetoed_modifier }
				# No standing approval for any target
				NOT = {
					AND = {
						has_variable = irredentist_war_approval
						exists = var:irredentist_war_approval
					}
				}
				# At least one strong candidate
				any_country = {
					NOT = { THIS = ROOT }
					NOT = { is_country_type = decentralized }
					ROOT = {
						any_primary_culture = { is_primary_culture_of = PREV }
					}
					any_scope_state = {
						is_homeland_of_country_cultures = ROOT
						has_claim_by = ROOT
					}
				}
			}
			random_list = {
				985 = { } # 98.5% nothing
				15 = {
					save_scope_as = irr_requester
					random_country = {
						limit = {
							NOT = { THIS = ROOT }
							NOT = { is_country_type = decentralized }
							ROOT = {
								any_primary_culture = { is_primary_culture_of = PREV }
							}
							any_scope_state = {
								is_homeland_of_country_cultures = ROOT
								has_claim_by = ROOT
							}
						}
						save_scope_as = target_country
					}
					power_bloc ?= {
						power_bloc_leader ?= {
							trigger_event = { id = irredentism.7 days = 3 }
						}
					}
				}
			}
		}

		# 5b: peaceful client received a union proposal -- ask bloc leader.
		# Skip for Diplomatic-Framework blocs: that identity has no leader
		# authority over members, so irredentist_bloc_check_peace already
		# passes without an approval var. Firing 5b would be wasted.
		if = {
			limit = {
				NOT = { is_country_type = decentralized }
				is_in_power_bloc = yes
				is_power_bloc_leader = no
				NOT = {
					power_bloc ?= {
						has_identity = identity:identity_diplomatic
					}
				}
				has_modifier = irr_open_to_union_modifier
				NOT = {
					AND = {
						has_variable = irredentist_peace_approval
						exists = var:irredentist_peace_approval
					}
				}
				# Some same-culture amicable+ neighbor exists -- the unifier
				NOT = { has_modifier = irr_union_forbidden_modifier }
				any_country = {
					NOT = { THIS = ROOT }
					ROOT = {
						any_primary_culture = { is_primary_culture_of = PREV }
					}
					relations:ROOT >= relations_threshold:amicable
				}
			}
			random_list = {
				985 = { }
				15 = {
					save_scope_as = irr_proposed_target
					random_country = {
						limit = {
							NOT = { THIS = ROOT }
							ROOT = {
								any_primary_culture = { is_primary_culture_of = PREV }
							}
							relations:ROOT >= relations_threshold:amicable
						}
						save_scope_as = irr_unifier
					}
					power_bloc ?= {
						power_bloc_leader ?= {
							trigger_event = { id = irredentism.8 days = 3 }
						}
					}
				}
			}
		}
	}
}

# ============================================================================
# MANDATE RESERVE - yearly cohesion bonus on the leader's bloc, scaled to the
# stockpile of unused mandates. Engine has no on_yearly_pulse_power_bloc and
# no add_mandate/remove_mandate effect, so we run from on_yearly_pulse_country
# gated on is_power_bloc_leader = yes and scope into the bloc to apply the
# modifier. remove_modifier first prevents stack drift across years.
# months = 13 outlives the yearly pulse so the modifier never blinks off.
# ============================================================================
te_unused_mandate_reserve_on_action = {
	effect = {
		if = {
			limit = { is_power_bloc_leader = yes }
			power_bloc = {
				remove_modifier = sm_te_unused_mandate_reserve
				# num_mandates is a power_bloc-scoped trigger comparator; `multiplier = <script_value>`
				# resolves in the outer (country) scope where num_mandates is invalid. Inline the
				# tier ladder here so the comparator runs in power_bloc scope and the multiplier is
				# a plain integer literal.
				if = {
					limit = { num_mandates >= 4 }
					add_modifier = { name = sm_te_unused_mandate_reserve multiplier = 12 months = 13 }
				}
				else_if = {
					limit = { num_mandates >= 3 }
					add_modifier = { name = sm_te_unused_mandate_reserve multiplier = 9 months = 13 }
				}
				else_if = {
					limit = { num_mandates >= 2 }
					add_modifier = { name = sm_te_unused_mandate_reserve multiplier = 6 months = 13 }
				}
				else_if = {
					limit = { num_mandates >= 1 }
					add_modifier = { name = sm_te_unused_mandate_reserve multiplier = 3 months = 13 }
				}
			}
		}
	}
}

# --- Agricultural diffusion: backfill for newly-formed countries -----------

# countries that exist at the moment of first research. Decolonized 1960+

	}

# Fire the bespoke India/Indonesia formation events as forced popups (matching
# vanilla's `trigger_event = { ... popup = yes }` formation idiom). An on_action
# `events = { }` list fires without `popup = yes`, which surfaces as a small
# notification card rather than the milestone popup these formation moments want.
te_bespoke_formation_dispatcher = {
	effect = {
		if = {
			limit = { c:BHA ?= this }

		}
		else_if = {
			limit = { c:NUS ?= this }
			trigger_event = { id = formation_te.2 popup = yes }
		}
	}
}
