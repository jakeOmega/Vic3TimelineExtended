# ============================================================================
# Power Grid: the monthly balancing of every grid
# ============================================================================
# See power_grid_effects.txt. A grid can span countries (a Grid
# Interconnection Agreement), so every grid is balanced in one run, on the 1st
# of the month. The global pulse has no ROOT, and the flow modifiers'
# multipliers want one, so it fires te_grid.1 (events/te_grid_events.txt) on
# one country that has a station, or a state a station has left its figures on
# (the run's own country filter): so the run after the world's last station is
# demolished still clears that state and the grid record the tab reads
# (te_grid_record_clear_effect). Its immediate runs the world-wide update
# (scripting_best_practices.md, "The global pulse has no ROOT, so dispatch
# through a country event").
# ============================================================================

# No root scope.
on_monthly_pulse = {
	on_actions = {
		te_grid_world_dispatch
	}
}

te_grid_world_dispatch = {
	effect = {
		random_country = {
			limit = {
				any_scope_state = {
					OR = {
						has_building = building_grid_transfer_station
						has_variable = te_grid_flow
					}
				}
			}
			trigger_event = { id = te_grid.1 }
		}
	}
}

# The AI's own station building (te_grid_ai_build_effect), and every
# country's generation by source for the Power Grids tab
# (te_grid_mix_record_effect). Never the balancing: that runs once a month
# for the world, above.
on_monthly_pulse_country = {
	on_actions = {
		te_grid_ai_build_on_action
		te_grid_mix_on_action
	}
}

te_grid_ai_build_on_action = {
	effect = {
		if = {
			limit = { is_ai = yes }
			te_grid_ai_build_effect = yes
		}
	}
}

# Once the tab is open (te_grid_tab_unlocked): the panel reads it for any
# country, National Info's included.
te_grid_mix_on_action = {
	effect = {
		if = {
			limit = { te_grid_tab_unlocked = yes }
			te_grid_mix_record_effect = yes
		}
	}
}
