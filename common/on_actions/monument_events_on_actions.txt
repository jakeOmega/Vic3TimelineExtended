# ============================================================================
# The local pulse (spec §2.5)
# ============================================================================
# State scope (ROOT = the state). A state modifier's multiplier resolves
# against ROOT, so each monument's local modifiers are refreshed here and never
# from the country pulse. Runs for a state with a monument, or one whose
# monument has just gone (records or a multiplier left to clear).
on_monthly_pulse_state = {
	on_actions = {
		gm_state_on_action
	}
}

gm_state_on_action = {
	trigger = {
		gm_system_enabled = yes
	}
	effect = {
		if = {
			limit = {
				OR = {
					has_building = building_grand_monument
					# A construction site, and its records once it goes.
					has_building = building_grand_monument_site
					has_variable = gm_site_counted
					has_variable = gm_seen
					AND = {
						has_variable = gm_local_steps
						var:gm_local_steps > 0
					}
				}
			}
			gm_state_monthly = yes
		}
	}
}

# ============================================================================
# The national pulse (spec §2.4), and the hooks that re-read "fits" at once
# ============================================================================
gm_country_on_action = {
	trigger = {
		gm_system_enabled = yes
	}
	effect = {
		if = {
			limit = {
				OR = {
					any_scope_state = { has_building = building_grand_monument }
					has_variable = gm_active
					has_variable = gm_com_open
					# A naming outlives its last monument until the sweep
					# closes it (gm_name_sweep).
					has_variable = gm_naming_active
				}
			}
			gm_country_monthly = yes
		}
	}
}

# ROOT = the law; its owner is the country. A new law can topple a crown, a
# republic or a revolution: refresh through the hidden country event.
on_law_activated = {
	on_actions = {
		gm_law_activated_on_action
	}
}

gm_law_activated_on_action = {
	trigger = {
		gm_system_enabled = yes
	}
	effect = {
		owner = {
			if = {
				limit = {
					any_scope_state = { has_building = building_grand_monument }
				}
				trigger_event = { id = monument_events.20 }
			}
		}
	}
}

# ROOT = a country on a side that has just won (on_won_war) or lost
# (on_lost_war) a war, capitulations included; scope:war = the war,
# scope:enemy_country = the other side's leader, scope:benefitted_from_wargoal /
# scope:victim_of_wargoal = set if a war goal of this country's was enforced, or
# one against it (1.14.5 00_code_on_actions.txt; vanilla moved recently_won_war
# here). Victory and Defeat are for war leaders after a year of war.
on_won_war = {
	on_actions = {
		gm_won_war_on_action
	}
}

gm_won_war_on_action = {
	trigger = {
		gm_system_enabled = yes
	}
	effect = {
		if = {
			limit = {
				exists = scope:benefitted_from_wargoal
				exists = scope:war
				scope:war = {
					is_warleader = root
					war_duration_months >= 12
				}
			}
			gm_com_war_victory = yes
		}
	}
}

on_lost_war = {
	on_actions = {
		gm_lost_war_on_action
	}
}

gm_lost_war_on_action = {
	trigger = {
		gm_system_enabled = yes
	}
	effect = {
		if = {
			limit = {
				exists = scope:victim_of_wargoal
				exists = scope:war
				scope:war = {
					is_warleader = root
					war_duration_months >= 12
				}
			}
			gm_com_war_defeat = yes
		}
	}
}

# ROOT = the character who died. A ruler who reigned 15 years or more
# (gm_ruler_months, kept by the world pass) brings A Reign Remembered: a
# mausoleum. At once, so it takes the slot ahead of the new reign the world
# pass sees later in the month.
on_character_death = {
	on_actions = {
		gm_character_death_on_action
	}
}

gm_character_death_on_action = {
	trigger = {
		gm_system_enabled = yes
		is_ruler_of_own_country = yes
	}
	effect = {
		save_scope_as = gm_late_ruler
		owner ?= {
			if = {
				limit = {
					gm_com_can_offer = yes
					has_variable = gm_ruler
					var:gm_ruler = scope:gm_late_ruler
					has_variable = gm_ruler_months
					var:gm_ruler_months >= 180
				}
				gm_com_ruler_death = yes
			}
		}
	}
}

# ROOT = the newly formed country: its founding is now (Germany in 1871 marks
# 1971, not its Prussian past).
on_country_formed = {
	on_actions = {
		gm_formed_on_action
	}
}

gm_formed_on_action = {
	trigger = {
		gm_system_enabled = yes
	}
	effect = {
		set_variable = { name = gm_founded_year value = year }
		gm_remove_state_var = { VAR = gm_com_centenary_100 }
		gm_remove_state_var = { VAR = gm_com_centenary_200 }
	}
}

# The landmarks standing in 1836 (phase 4, gm_landmark_effects.txt), then the
# trophies already abroad (loot spec 2026-10-12 §2, gm_trophy_effects.txt).
# After the lobby, because a game rule changed there is final only then; once
# per game. ROOT is none here, and the seeding never reads it.
on_game_started_after_lobby = {
	on_actions = {
		gm_landmarks_on_game_started_after_lobby
	}
}

gm_landmarks_on_game_started_after_lobby = {
	effect = {
		if = {
			limit = {
				gm_system_enabled = yes
				NOT = { has_global_variable = gm_landmarks_seeded }
			}
			set_global_variable = gm_landmarks_seeded
			gm_seed_landmarks = yes
			gm_seed_trophies = yes
			gm_monument_self_check = yes
		}
	}
}
