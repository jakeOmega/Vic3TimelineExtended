# ============================================================================
# GRAND MONUMENT ON-ACTIONS
# ============================================================================
# on_building_built is a vanilla on-action, extended here via the additive
# `events` list. NEVER add `effect = { }` to a vanilla on-action -- `events` and
# `on_actions` lists are additive and safe, `effect` is not.
# See docs/guides/scripting_best_practices.md.

on_building_built = {
	events = {
		monument_events.1
	}
}

on_monthly_pulse_country = {
	on_actions = {
		monument_events_on_action
	}
}

# Recurring monument flavour. Mirrors wonder_events_on_actions.txt: a large
# do-nothing weight with each entry gated on actually owning a monument of that
# dedication. Each event re-checks its own trigger, so the gates here are just
# to keep the roll cheap.
monument_events_on_action = {
	effect = {
		random_list = {
			800 = { } # overwhelmingly nothing

			3 = {
