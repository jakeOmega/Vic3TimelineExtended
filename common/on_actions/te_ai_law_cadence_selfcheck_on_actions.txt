# TEMPORARY self-check (2026-11-15): does the default strategy's INJECTed change_law_chance take?
#
# edited_default_strategy.txt raises ai_strategy_default's change_law_chance from vanilla's 1 to 40. No trigger or
# script value reads an AI strategy's value, and before this change the INJECT read 1 in normal times either way, so
# nothing has shown that an INJECT reaches this field (the one AI-strategy read, a subsidies block, kept vanilla's
# value: scripting_best_practices.md). This check reads the effect instead: the share of AI countries in the middle
# of a law enactment.
#
# ai_law_cadence, once, at the first global monthly pulse on or after 1850.1.1 that is also at least 36 months after
# the first pulse this build ran (te_alc_months counts them), over AI countries that are not decentralized nations
# (those never change a law: 121 of the 123 AI countries in an 1864 save that had changed none). The 36 months let a
# save begun on the build before settle at the new rate first: enactments under way then take a median 17 months to
# end, and the share read at once would be the old one. A new game gives its verdict in 1850.
# No line yet: the game hasn't reached 1850, or 36 months since this build was first loaded.
#   OK          at least 35% are enacting a law: the raised rate is in force (expected 35-65%).
#   BROKEN      under 30%: the rate is vanilla's. Eleven saves of four games on the build before (1864-2020) read
#               22-31%, 24% in 1864. The INJECT is ignored; see the fallback in the play-test list.
#   NO_VERDICT  30-35%, or fewer than 50 such AI countries (nothing to judge).
# With another mod that raises the agendas' own rates (kai_political_strategies.txt), OK can come from that mod alone.
#
# Play-test row: docs/testing/ai-economy-playtest.md § AI law cadence. Delete this file and
# te_ai_law_cadence_selfcheck_values.txt once it reads OK.
on_monthly_pulse = {
	on_actions = {
		te_ai_law_cadence_selfcheck_settle
		te_ai_law_cadence_selfcheck
	}
}

# The settling count: one a month from the first pulse this build runs, up to 36.
te_ai_law_cadence_selfcheck_settle = {
	trigger = {
		NOT = { has_global_variable = te_alc_checked }
	}
	effect = {
		if = {
			limit = { NOT = { has_global_variable = te_alc_months } }
			set_global_variable = { name = te_alc_months value = 0 }
		}
		if = {
			limit = { global_var:te_alc_months < 36 }
			change_global_variable = { name = te_alc_months add = 1 }
		}
	}
}

# The verdict, once (the latch is the first thing it sets).
te_ai_law_cadence_selfcheck = {
	trigger = {
		NOT = { has_global_variable = te_alc_checked }
		has_global_variable = te_alc_months
		global_var:te_alc_months >= 36
		game_date >= 1850.1.1
	}
	effect = {
		set_global_variable = te_alc_checked
		set_global_variable = { name = te_alc_chk_countries value = 0 }
		set_global_variable = { name = te_alc_chk_enacting value = 0 }
		every_country = {
			limit = {
				is_ai = yes
				NOT = { is_country_type = decentralized }
			}
			change_global_variable = { name = te_alc_chk_countries add = 1 }
			if = {
				limit = { enacting_any_law = yes }
				change_global_variable = { name = te_alc_chk_enacting add = 1 }
			}
		}
		random_country = {
			if = {
				limit = { te_alc_chk_countries_g < 50 }
				debug_log = "TE_SELF_CHECK NO_VERDICT ai_law_cadence countries=[SCOPE.ScriptValue('te_alc_chk_countries_g')|0] enacting=[SCOPE.ScriptValue('te_alc_chk_enacting_g')|0] (fewer than 50 AI countries that are not decentralized: nothing to judge) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else_if = {
				limit = { te_alc_chk_share >= 0.35 }
				debug_log = "TE_SELF_CHECK OK ai_law_cadence share=[SCOPE.ScriptValue('te_alc_chk_share')|2] enacting=[SCOPE.ScriptValue('te_alc_chk_enacting_g')|0] countries=[SCOPE.ScriptValue('te_alc_chk_countries_g')|0] (AI countries mid-enactment, against 0.22-0.31 on the build before: the default strategy's raised change_law_chance is in force) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else_if = {
				limit = { te_alc_chk_share < 0.30 }
				debug_log = "TE_SELF_CHECK BROKEN ai_law_cadence share=[SCOPE.ScriptValue('te_alc_chk_share')|2] enacting=[SCOPE.ScriptValue('te_alc_chk_enacting_g')|0] countries=[SCOPE.ScriptValue('te_alc_chk_countries_g')|0] (AI countries mid-enactment, within the 0.22-0.31 of the build before: the INJECT into ai_strategy_default's change_law_chance looks ignored; take the fallback in docs/testing/ai-economy-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK NO_VERDICT ai_law_cadence share=[SCOPE.ScriptValue('te_alc_chk_share')|2] enacting=[SCOPE.ScriptValue('te_alc_chk_enacting_g')|0] countries=[SCOPE.ScriptValue('te_alc_chk_countries_g')|0] (between 0.30 and 0.35: read the saves as docs/testing/ai-economy-playtest.md says) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
		}
	}
}
