# ======================================================================
#  AMENDMENT PETITIONS – ON ACTIONS
#
#  Once a year, a one-in-four chance that an interest group petitions for
#  one of the mod's amendments on a law already in force
#  (events/amendment_petition_events.txt). Each event's trigger carries
#  the law, the situation and te_ap_window_open, which holds back every
#  petition for five years after the last one; each event also has a
#  twenty-year cooldown of its own. A country that qualifies for none
#  sees nothing.
#  Docs: docs/systems/mod_systems.md § Amendment Petitions.
# ======================================================================

# Root = Country
on_yearly_pulse_country = {
	on_actions = {
		te_amendment_petition_yearly_events
	}
}
