# ============================================================================
# THE WORLD COURT — A CLAIM THE COURT STRUCK DOWN (redesign phase 7)
# ============================================================================
# docs/systems/un_redesign_design.md §0.12 item 1. When the World Court finds
# for the holder of a disputed state, the claimant's claim on the region is
# withdrawn, and for twenty years a war goal it adds against a state in that
# region costs it extra infamy, x E, with a notice (un_court_bar_war_goal,
# un_justice_effects.txt).
#
# The engine unions same-name on_action declarations across files (see
# un_mandate_on_actions.txt), so this entry joins the mandate's hook and the
# others on on_wargoal_added without disturbing them.
# ============================================================================

# Root = Diplomatic Play
# scope:actor = War Goal Owner
# (game/common/on_actions/00_code_on_actions.txt:6992-6999)
on_wargoal_added = {
	on_actions = {
		un_court_on_wargoal_added
	}
}

un_court_on_wargoal_added = {
	effect = {
		if = {
			limit = {
				exists = scope:actor
				scope:actor = { has_variable = un_court_barred_region }
			}
			un_court_bar_war_goal = yes
		}
	}
}
