# Space Race On-Actions

# Once a month per country (not per journal entry): the inactive-milestone
# cleanup, the revolution backstop, the post-race dividend, a player's Space
# Powers standings and, until 2026-12-31, the retired failure cooldown's
# old-save strip.
on_monthly_pulse_country = {
	on_actions = {
		space_race_on_action
	}
}

space_race_on_action = {
	effect = {
		# Extraplanetary Base room (sr_colony_module_effects.txt; colony modules
		# design § 9): its monthly refresh, for every country that can build
		# a base (In-Situ Resource Utilization, the building's technology),
		# colony or not; space_race_events.110 refreshes it too, the day after
		# a base is built or gains a level (UI pass design § 8).
		if = {
			limit = { has_technology_researched = in_situ_resource_utilization }
			sr_ebase_refresh = yes
		}
		# Fleets (solar colonization slice 6, sr_fleet_effects.txt): a fleet
		# power's month under Full Colonies, Space Command's construction first.
		if = {
			limit = { sr_fleet_power = yes }
			sr_fl_country_month = yes
		}
		# Its orbital fire support (6d) is refreshed in that month; a country no
		# longer a fleet power loses it here.
		else_if = {
			limit = { has_modifier = sr_orbital_superiority }
			remove_modifier = sr_orbital_superiority
		}
		# Earth orbit (slice 6d, sr_earth_orbit_effects.txt): a player told when
		# its Earth is blockaded, or its launch site lost and back; and of each
		# hostile fleet bound for Earth (the Earth alert, UI pass design § 9,
		# sr_fleets_generated.txt).
		if = {
			limit = {
				is_player = yes
				sr_colonies_full_enabled = yes
				OR = {
					sr_has_space_program = yes
					sr_fleet_power = yes
					has_variable = sr_launch_site_noted
				}
			}
			sr_earth_notice_step = yes
			sr_earth_alert_step = yes
		}
		# An old save's open Moon Base (lunar colonization design § 10): its
		# progress moves onto Solar System Colonization before the cleanup on
		# the next line deletes it (sr_launch_window_effects.txt).
		sr_lunar_carry_moon_base = yes
		sr_cleanup_inactive_space_race_milestones = yes
		# A choice to colonize the Moon lasts while a lunar world is free
		# (sr_site_effects.txt).
		sr_colonization_resumed_expire = yes
		sr_monthly_colony_discoveries = yes
		# The site_receiver_entry self-check (sr_site_effects.txt).
		sr_site_receiver_check_step = yes
		# What a revolution's winner keeps (space_race_effects.txt): the
		# backstop that puts the entries' modifiers back, and the probe result's
		# record, which also migrates a save from before the record existed.
		# Only for a country that has run a milestone (sr_space_race_participant),
		# which is every country with anything to put back.
		if = {
			limit = { sr_space_race_participant = yes }
			sr_sync_space_race_entries = yes
			sr_sync_probe_data = yes
			# After the race, the leftover progress pays Extraplanetary Base
			# throughput (space_race_effects.txt). Its one refresh site.
			sr_refresh_space_dividend = yes
			# The Rivals rows' estimates (space_race_effects.txt): players only.
			if = {
				limit = { is_player = yes }
				sr_rivals_estimate_refresh = yes
			}
			# An AI's approach costs stop outside the top three and resume when it
			# climbs back (sr_<m>_cost reads sr_ai_should_participate). Ranking
			# moves without any button press, so recalculate when the
			# participation the costs were last computed with is stale, or was
			# never recorded (saves from before PR #507).
			if = {
				limit = { NOT = { has_variable = sr_cost_participation } }
				sr_recalculate_cost = yes
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							sr_ai_should_participate = yes
							var:sr_cost_participation = 0
						}
						AND = {
							NOT = { sr_ai_should_participate = yes }
							var:sr_cost_participation = 1
						}
					}
				}
				sr_recalculate_cost = yes
			}
			# The same for solar colonization's billing, which stops while its
			# production method is off (sr_milestone_billed_solar_colonization):
			# the method changes without any button press.
			else_if = {
				limit = { NOT = { has_variable = sr_cost_solar_running } }
				sr_recalculate_cost = yes
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							sr_milestone_billed_solar_colonization = yes
							var:sr_cost_solar_running = 0
						}
						AND = {
							NOT = { sr_milestone_billed_solar_colonization = yes }
							var:sr_cost_solar_running = 1
						}
					}
				}
				sr_recalculate_cost = yes
			}
			# The same for the funding cap, which the pace multipliers read
			# (sr_funding_<m>_effective): technologies and Advanced Research V
			# move it with no button press. Recorded since 2026-10-29, so an
			# older save recalculates once.
			else_if = {
				limit = { NOT = { has_variable = sr_cost_funding_cap } }
				sr_recalculate_cost = yes
			}
			else_if = {
				limit = {
					OR = {
						var:sr_cost_funding_cap < sr_max_funding_level
						var:sr_cost_funding_cap > sr_max_funding_level
					}
				}
				sr_recalculate_cost = yes
			}
			# A revolution's winner whose inherited entry lost its cost and
			# funding modifiers (sr_cost_modifiers_missing): put them back. The
			# same test moves a save from before the funding split (2026-10-29)
			# onto sr_funding_throughput.
			else_if = {
				limit = { sr_cost_modifiers_missing = yes }
				sr_recalculate_cost = yes
			}
		}
		# The Space Powers standings on the Space Race tab
		# (space_race_effects.txt): players in the race, a month at a time.
		if = {
			limit = {
				is_player = yes
				sr_space_race_on = yes
				sr_in_space_race = yes
			}
			sr_powers_refresh = yes
		}
		else_if = {
			limit = { has_variable = sr_power_total }
			sr_powers_clear = yes
		}
		# Old saves: the post-setback cooldown is gone (2026-10-31; remove this
		# strip after 2026-12-31). Drop its counter and re-derive the statuses it
		# shielded (3 to 5), which the panel no longer draws.
		if = {
			limit = { has_variable = sr_failure_cooldown }
			remove_variable = sr_failure_cooldown
			sr_refresh_milestone_statuses = yes
		}
	}
}
