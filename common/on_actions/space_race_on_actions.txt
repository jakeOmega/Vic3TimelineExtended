# Space Race On-Actions

# Once a month per country (not per journal entry): the inactive-milestone
# cleanup, the revolution backstop and the global failure-cooldown tick.
on_monthly_pulse_country = {
	on_actions = {
		space_race_on_action
	}
}

space_race_on_action = {
	effect = {
		sr_cleanup_inactive_space_race_milestones = yes
		# What a revolution's winner keeps (space_race_effects.txt): the
		# backstop that puts the entries' modifiers back, and the probe result's
		# record, which also migrates a save from before the record existed.
		# Only for a country that has run a milestone (sr_space_race_participant),
		# which is every country with anything to put back.
		if = {
			limit = { sr_space_race_participant = yes }
			sr_sync_space_race_entries = yes
			sr_sync_probe_data = yes
		}
		if = {
			limit = { NOT = { has_variable = sr_failure_cooldown } }
			set_variable = { name = sr_failure_cooldown value = 0 }
		}
		if = {
			limit = { sr_has_failure_cooldown = yes }
			change_variable = { name = sr_failure_cooldown subtract = 1 }
		}
	}
}
