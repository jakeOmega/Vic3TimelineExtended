# Local Climate (docs/superpowers/specs/2026-11-01-local-climate-design.md § 5.1, § 5.4).
# Effects: common/scripted_effects/local_climate_effects.txt.
#
#   on_yearly_pulse_state     the yearly refresh of local_climate_discomfort, its one
#                             refresh site (a different day per state across the year)
#   on_monthly_pulse_state    the first application, only on a state with no
#                             te_climate_penalty_applied yet (a new game's first month,
#                             a save from before this, a newly colonized state)
#   on_yearly_pulse_country   once every ten years, from the first country to pulse,
#                             a debug.log line counting the mass migration targets
#
# Each pulse lists its own on-action, so the lists merge with vanilla's and the
# mod's other files' rather than replacing them.
on_yearly_pulse_state = {
	on_actions = {
		local_climate_yearly_on_action
	}
}

on_monthly_pulse_state = {
	on_actions = {
		local_climate_first_on_action
	}
}

on_yearly_pulse_country = {
	on_actions = {
		local_climate_log_on_action
	}
}

# ROOT = the state, so the multiplier reads the state's penalty.
local_climate_yearly_on_action = {
	effect = {
		te_update_local_climate_modifier = yes
	}
}

local_climate_first_on_action = {
	trigger = {
		NOT = { has_variable = te_climate_penalty_applied }
	}
	effect = {
		te_update_local_climate_modifier = yes
	}
}

# Informational, not a TE_SELF_CHECK: the define change (MASS_MIGRATION_TARGET_MIN_ARABLE_LAND
# = 0, common/defines/extra_defines.txt) lets a state with under 10 arable land be picked as a
# mass migration target, and this counts how many are. A decade with none proves nothing, so
# there is no OK or BROKEN. The global variable runs out after ten years, and the first
# country to pulse after that logs again. Play-test list: docs/testing/local-climate-playtest.md.
local_climate_log_on_action = {
	trigger = {
		NOT = { has_global_variable = te_climate_log_decade }
	}
	effect = {
		set_global_variable = { name = te_climate_log_decade value = yes days = 3650 }
		set_global_variable = { name = te_climate_log_targets value = 0 }
		set_global_variable = { name = te_climate_log_small_targets value = 0 }
		every_state = {
			limit = { is_mass_migration_target = yes }
			change_global_variable = { name = te_climate_log_targets add = 1 }
			if = {
				limit = { arable_land < 10 }
				change_global_variable = { name = te_climate_log_small_targets add = 1 }
			}
		}
		debug_log = "TE_LOCAL_CLIMATE mass_migration_targets=[GetGlobalVariable('te_climate_log_targets').GetValue|0] under_10_arable=[GetGlobalVariable('te_climate_log_small_targets').GetValue|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		remove_global_variable = te_climate_log_targets
		remove_global_variable = te_climate_log_small_targets
	}
}
