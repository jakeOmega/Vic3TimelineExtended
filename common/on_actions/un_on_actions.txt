# United Nations events - wired into monthly country pulse
un_events_on_action = {
	effect = {
		# Hygiene: drop an authorized-mandate index that no longer points at a
		# mandate container. un_mandate_close clears it on every ordinary
		# closure, so this only ever fires for a save carried across a version
		# that destroyed the container some other way. Guards are cheap and the
		# consequence of a dangling index would be a country that can never
		# propose a mandate again.
		if = {
			limit = {
				has_variable = un_mandate_current
				NOT = { var:un_mandate_current ?= { has_tag = un_mandate } }
			}
			remove_variable = un_mandate_current
		}

		# Recurring events - only for members, and only once the UN exists
		if = {
			limit = {
				je:je_united_nations ?= { has_modifier = un_member_modifier }
				has_global_variable = un_authority
			}
			random_list = {
				75 = { } # Nothing happens most months
				3 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = condemn } }
						any_country = {
							has_diplomatic_pact = { who = ROOT type = rivalry }
							is_at_war = yes
						}
						NOT = { has_modifier = un_ga_resolution_modifier }
					}
					trigger_event = { id = un_events.2 }
				}
				3 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = human_rights } }
						has_technology_researched = human_rights
						global_var:un_authority >= 40
						NOT = { je:je_united_nations ?= { has_modifier = un_human_rights_declaration_modifier } }
					}
					trigger_event = { id = un_events.3 }
				}
				3 = {
					trigger = {
						je:je_united_nations ?= { has_modifier = un_peacekeeping_contributor_modifier }
						country_rank >= rank_value:major_power
						any_country = {
							NOT = { this = ROOT }
							is_at_war = yes
							NOT = { has_war_with = ROOT }
							NOT = { has_modifier = un_peacekeeping_welcomed_modifier }
							NOT = { has_modifier = un_peacekeeping_resented_modifier }
						}
					}
					trigger_event = { id = un_events.4 }
				}
				3 = {
					trigger = {
						has_modifier = un_sanctions_enforcer_modifier
						NOT = { has_modifier = un_sanctions_enforcement_modifier }
						any_country = { has_modifier = un_sanctions_target_modifier }
					}
					trigger_event = { id = un_events.5 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = reform } }
						country_rank >= rank_value:great_power
						un_charter_reform_ripe = yes
						NOT = { has_modifier = un_reform_champion_modifier }
						NOT = { has_modifier = un_reform_opponent_modifier }
					}
					trigger_event = { id = un_events.6 }
				}
				3 = {
					trigger = {
						country_rank >= rank_value:major_power
						NOT = { has_modifier = un_humanitarian_aid_modifier }
						NOT = { has_modifier = un_humanitarian_token_modifier }
						any_country = {
							NOT = { this = ROOT }
							any_scope_state = {
								average_sol < 8
							}
						}
					}
					trigger_event = { id = un_events.7 }
				}
				2 = {
					trigger = {
						global_var:un_authority >= 30
						OR = {
							is_at_war = yes
							has_law = law_type:law_total_war
						}
						NOT = { has_modifier = un_court_compliance_modifier }
						NOT = { has_modifier = un_court_defiance_modifier }
					}
					trigger_event = { id = un_events.8 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = heritage } }
						global_var:un_authority >= 40
						NOT = { je:je_united_nations ?= { has_modifier = un_heritage_program_modifier } }
						NOT = { has_modifier = un_heritage_token_modifier }
					}
					trigger_event = { id = un_events.9 }
				}
				# un_events.10 (the authority crisis) is no longer rolled here.
				# Since redesign phase 2 it fires once, to every great power,
				# when the crisis opens (un_ladder_crisis_open).
				# ---- Expanded events (11-20) ----
				2 = {
					trigger = {
						country_rank >= rank_value:great_power
						global_var:un_authority >= 30
						NOT = { has_modifier = un_veto_crisis_modifier }
						NOT = { has_modifier = un_veto_abolition_modifier }
						any_country = {
							NOT = { this = ROOT }
							country_rank >= rank_value:great_power
							je:je_united_nations ?= { has_modifier = un_member_modifier }
						}
					}
					trigger_event = { id = un_events.11 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = decolonization } }
						global_var:un_authority >= 30
						NOT = { has_modifier = un_decolonization_champion_modifier }
						NOT = { has_modifier = un_colonial_defender_modifier }
						any_country = {
							any_subject_or_below = { count >= 1 }
						}
					}
					trigger_event = { id = un_events.12 }
				}
				2 = {
					trigger = {
						country_rank >= rank_value:great_power
						global_var:un_authority >= 20
						NOT = { has_modifier = un_ideological_champion_modifier }
						NOT = { has_modifier = un_coexistence_modifier }
						any_country = {
							NOT = { this = ROOT }
							country_rank >= rank_value:great_power
							je:je_united_nations ?= { has_modifier = un_member_modifier }
							has_diplomatic_pact = { who = ROOT type = rivalry }
						}
					}
					trigger_event = { id = un_events.13 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = npt } }
						global_var:un_authority >= 40
						has_technology_researched = nuclear_weapons
						NOT = { je:je_united_nations ?= { has_modifier = un_nonproliferation_modifier } }
						NOT = { has_modifier = un_nonproliferation_refusal_modifier }
					}
					trigger_event = { id = un_events.14 }
				}
				2 = {
					trigger = {
						global_var:un_authority >= 30
						NOT = { has_modifier = un_trade_arbitration_modifier }
						NOT = { has_modifier = un_unilateral_tariffs_modifier }
						any_country = {
							NOT = { this = ROOT }
							je:je_united_nations ?= { has_modifier = un_member_modifier }
							has_diplomatic_pact = { who = ROOT type = rivalry }
						}
					}
					trigger_event = { id = un_events.15 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = pandemic } }
						country_rank >= rank_value:major_power
						global_var:un_authority >= 20
						has_technology_researched = antibiotics
						NOT = { je:je_united_nations ?= { has_modifier = un_pandemic_cooperation_modifier } }
						NOT = { has_modifier = un_pandemic_isolationist_modifier }
					}
					trigger_event = { id = un_events.16 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = climate } }
						global_var:un_authority >= 30
						has_technology_researched = environmental_movement
						NOT = { je:je_united_nations ?= { has_modifier = un_climate_binding_modifier } }
						NOT = { has_modifier = un_climate_refusal_modifier }
					}
					trigger_event = { id = un_events.17 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = refugee } }
						global_var:un_authority >= 20
						NOT = { je:je_united_nations ?= { has_modifier = un_refugee_program_modifier } }
						NOT = { has_modifier = un_refugee_closed_modifier }
						any_country = {
							NOT = { this = ROOT }
							OR = {
								is_at_war = yes
								any_scope_state = { average_sol < 6 }
							}
						}
					}
					trigger_event = { id = un_events.18 }
				}
				2 = {
					trigger = {
						NOT = { has_global_variable = un_vote_active }
						NOT = { un_resolution_topic_on_cooldown = { TOPIC = space } }
						global_var:un_authority >= 40
						has_technology_researched = space_exploration
						country_rank >= rank_value:major_power
						NOT = { je:je_united_nations ?= { has_modifier = un_space_partnership_modifier } }
						NOT = { has_modifier = un_space_independent_modifier }
					}
					trigger_event = { id = un_events.19 }
				}
				2 = {
					trigger = {
						country_rank >= rank_value:great_power
						global_var:un_authority < 50
						NOT = { has_modifier = un_walkout_condemn_modifier }
						NOT = { has_modifier = un_walkout_join_modifier }
						any_country = {
							NOT = { this = ROOT }
							country_rank >= rank_value:major_power
							je:je_united_nations ?= { has_modifier = un_member_modifier }
							has_diplomatic_pact = { who = ROOT type = rivalry }

on_monthly_pulse_country = {
	on_actions = {

	}
}

# un_authority is a single GLOBAL value representing the UN's authority in the
# world. It must advance exactly ONCE per month regardless of member count or HQ
# host, so it runs from the global monthly pulse in the no-country game scope
# (same pattern as global_warming_events_on_action in extra_on_actions.txt). The
# engine unions same-name on_action declarations across files.
on_monthly_pulse = {
	on_actions = {
		un_global_authority_on_action
	}
}

un_global_authority_on_action = {
	effect = {
		if = {
			limit = { has_global_variable = un_founded }

			# One-time legacy-save migration: pre-refactor saves stored un_authority
			# per-country and have no global. Seed a neutral 50 and strip the stale
			# per-country copies so nothing reads them. Guarded so it fires exactly
			# once (the first monthly pulse after loading such a save).
			if = {
				limit = { NOT = { has_global_variable = un_authority } }
				set_global_variable = { name = un_authority value = 50 }
				every_country = {
					limit = { has_variable = un_authority }
					remove_variable = un_authority
				}
			}

			# The equilibrium model (un_authority_effects.txt): cache every
			# country's power share, fade the ledgers, snapshot the seven pillars
			# and the target, and move authority one step toward it. Replaces the
			# old drift toward 50.
			un_authority_monthly_update = yes

			# General Assembly upkeep: lapse a resolution whose results event never
			# fired, and free an orphaned vote lock (un_vote_effects.txt).
			un_resolutions_monthly_update = yes

			# Authorized-mandate upkeep: age live authorizations toward expiry,
			# re-validate them, bind or void them, and retire closed ones from
			# the register (un_mandate_effects.txt). No-op until the first
			# mandate is issued.
			un_mandates_monthly_update = yes

			# Headquarters upkeep: a live member hosts, and exactly one HQ
			# building stands, in the host's territory (un_hq_effects.txt).
			un_hq_monthly_update = yes

			# Last: at 0, during a crisis, the UN dissolves (un_ladder_effects.txt).
			# After the upkeep above on purpose — that upkeep assumes a UN that
			# still exists, and a remove_modifier made by the dissolution would not
			# be visible to it within this same effect block.
			un_ladder_dissolution_check = yes
		}

		# Founded or not (un_ladder_effects.txt): power blocs fill the vacuum
		# while the UN is Moribund or dissolved, and a founding conference for a
		# new UN ages toward its founding.
		un_ladder_bloc_vacuum_update = yes
		un_conference_monthly_update = yes
	}
}

# A state holding the UN Headquarters changed hands (conquest, cession,
# release, annexation of the host). Root = state. An HQ outside the host's
# territory goes at once rather than at the next monthly pulse, so the
# conqueror never runs one and the host can rebuild straight away.
on_state_owner_change = {
	on_actions = {
		un_hq_on_state_owner_change
	}
}

un_hq_on_state_owner_change = {
	effect = {
		if = {
			limit = { has_building = building_un_headquarters }
			un_hq_enforce_single_building = yes
			un_hq_refresh_state = yes
		}
	}
}
