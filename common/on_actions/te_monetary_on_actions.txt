# ============================================================================
# Monetary Policy — on-action wiring
# ============================================================================
# Spec: docs/systems/monetary_policy_design.md §16.1-16.2.
#
# Almost everything here feeds one effect, te_monetary_monthly_update
# (common/scripted_effects/te_monetary_effects.txt), which is the system's
# single owner: no other file writes te_policy_rate, te_rate_paid_pts or the
# interest modifier, and no step of it is ever called in isolation — see
# the contract comment on te_monetary_monthly_update. The one exception is the
# end of a revolution (on_civil_war_won, below), where the winner takes over
# the loser's te_policy_rate once, with the rest of its central bank; that
# advances nothing, so the update's non-idempotence is not in play.
#
# ROOT MATTERS AT EVERY ENTRY POINT. The update's last step applies its modifier
# with `multiplier = var:te_rate_paid_applied`, and the engine resolves a
# `multiplier =` expression against ROOT rather than against the scope the effect
# is running in. So a hook whose ROOT is the country being updated calls the
# effect directly; a hook whose ROOT is `none` (game start) or somebody else
# (the six country-creation hooks, where ROOT is the parent) dispatches
# `te_monetary_internal.1` instead — a hidden country event whose ROOT is the
# country it fires on (events/te_monetary_events.txt). Each entry point still
# causes exactly ONE update, so the effect's non-idempotent steps never run
# twice. Every claim about a hook's ROOT below cites the engine's own
# docs/on_actions.log.
#
# NOT GATED ON `banking_system_enabled` (§16.1). Vanilla's flat 20% base rate is
# cancelled unconditionally, in INJECT:base_values, and this is what replaces it
# with the mod's own number — so it has to run even with the banking system
# switched off, or a rule-off game would have every country borrowing free.
#
# THE GAME RULE IS APPLIED INSIDE THE UPDATE, NOT HERE. banking_system_simplified
# switches off the advanced half of the monthly update (te_mon_full_system,
# common/scripted_triggers/te_monetary_triggers.txt) while the passive rate stack
# runs on. The hooks below that serve ONLY the advanced half — the phase-4 world
# inflation, the phase-5 anchored-GDP scan, the yearly arrangement scan, the
# subject and default hooks — are gated on that trigger here, because skipping
# them under the simplified rule saves the scan rather than just its result.
# ============================================================================


# ----------------------------------------------------------------------------
# THE WORLD REFERENCE RATE — once a month, for everybody
# ----------------------------------------------------------------------------
# te_mon_era_base costs two any_country scans, so it is evaluated once a month
# into a global variable instead of once per country per month. Expected scope
# of on_monthly_pulse is `none`; the effect enters country scope itself.
on_monthly_pulse = {
	on_actions = {
		te_monetary_world_monthly_on_action
	}
}

#
# PHASE 3 ADDS THE WORLD RATE (§12.1) to the same once-a-month slot: the real,
# GDP-weighted rate of the great powers whose rate is discretionary, falling
# back on the reference rate just refreshed — hence the order. It reads whatever
# the countries' last pulse left (§17 check 7: the order of this pulse and
# on_monthly_pulse_country is unknown), and every consumer tolerates that.
te_monetary_world_monthly_on_action = {
	effect = {
		te_monetary_refresh_world_era_base = yes
		te_monetary_refresh_world_rate = yes
		if = {
			limit = { te_mon_full_system = yes }
			# [P4] Its sibling (§15.1): GDP-weighted inflation over all great powers.
			te_monetary_refresh_world_inflation = yes
			# [P5] World GDP, and the GDP pegged to each anchor (§15A.2's
			# reserve-currency cut).
			te_monetary_refresh_anchored_gdp = yes
		}
	}
}


# ----------------------------------------------------------------------------
# THE MONTHLY COUNTRY UPDATE
# ----------------------------------------------------------------------------
# Root = Country ("Expected Scope: country", docs/on_actions.log:1901), so step
# 9's `multiplier = var:...` resolves against the country being updated and the
# call is made directly, with no dispatch event. This is also the hook that
# heals anything applied from a root-less context — see the te_rate_paid_rooted
# marker in step 9.
#
# te_civil_war_retire_parent is the SHARED parent pointer's monthly retirement
# (the section at the foot of this file), not part of the monetary update.
on_monthly_pulse_country = {
	on_actions = {
		te_monetary_monthly_on_action
		te_civil_war_retire_parent
	}
}

te_monetary_monthly_on_action = {
	effect = {
		te_monetary_monthly_update = yes
		# The civil-war copy's once-per-win marker lives from a rebel win to the
		# winner's next monthly pulse, when te_civil_war_retire_parent retires
		# the pointer it guards (te_monetary_on_civil_war_won says why). Both go
		# on the same pulse, whichever of the two on-actions runs first.
		if = {
			limit = { has_variable = te_mon_cw_bank_taken }
			remove_variable = te_mon_cw_bank_taken
		}
	}
}

# ----------------------------------------------------------------------------
# GAME START
# ----------------------------------------------------------------------------
# The cancel-INJECTs apply on day one, but te_monetary_rate_paid would not exist
# until a country's first monthly pulse. Several tags start in debt, and with
# vanilla's flat base already cancelled they would borrow free for that first
# month instead of paying their own rate (§16.1).
#
# ROOT IS `none` HERE. The engine declares this on-action's expected scope as
# `none` (docs/on_actions.log:341; vanilla 00_code_on_actions.txt:3-5 spells it
# out: "No root scope set"), and `every_country` changes THIS, not ROOT. Step 9
# of the update writes its modifier with `multiplier = var:te_rate_paid_applied`,
# which the engine resolves against ROOT — so calling the update directly from
# here applied every country's rate at a multiplier of 1.0. Proven in-game
# 2026-09-19; see the header of events/te_monetary_events.txt and
# scripting_best_practices.md § "Scope Chain Issues with add_modifier + Script
# Values". The update therefore runs inside a country event, whose ROOT is the
# country it fires on.
#
# The world reference rate is refreshed first so every country's step 0 reads a
# populated global.
#
# Event timing at game start: the mod's proven precedent for fanning an event
# out to every country at game start is `GLOBAL = { every_country = {
# trigger_event = ... } }` in common/history/te_construction_market_global.txt,
# which runs with no delay and has its effect visible on day one. Vanilla ships
# no trigger_event from on_game_started to compare against (its own block is
# empty, and on_game_started_after_lobby uses only direct effects), so the
# no-delay form here is the reasonable reading rather than a proven one. If the
# day-one budget tooltip ever shows no rate at all, move this fan-out to a
# GLOBAL history file on that precedent — the self-healing marker in step 9
# bounds the damage to the first monthly pulse either way.
on_game_started = {
	on_actions = {
		te_monetary_on_game_started
	}
}

te_monetary_on_game_started = {
	effect = {
		te_monetary_refresh_world_era_base = yes
		# No country has pulsed yet, so no great power is discretionary and this
		# writes the reference rate — which is the 1836 answer anyway (§12.1).
		te_monetary_refresh_world_rate = yes
		if = {
			limit = { te_mon_full_system = yes }
			# [P4] Its sibling (§15.1): GDP-weighted inflation over all great powers.
			te_monetary_refresh_world_inflation = yes
		}
		every_country = {
			trigger_event = {
				id = te_monetary_internal.1
			}
		}
	}
}

# COUNTRY CREATION
# ============================================================================
# A newly created tag has no rate modifier at all, so until its first monthly
# pulse it borrows at 0% — exactly when the player is looking at it (§16.1),
# since the flat base is already cancelled in INJECT:base_values. Every hook
# below therefore runs the full update on the new country immediately.
#
# Scope note: the four `released_as` hooks and the two uprising hooks put the
# NEW country in scope:target and the parent in Root; on_country_formed puts
# the new country in Root. Idiom copied from
# common/on_actions/te_construction_market_on_actions.txt:80-164.

# Shared by all six. ONLY the new country is updated. Root is the parent — the
# releasing country, or the one the uprising is breaking away from (vanilla
# documents this at common/on_actions/00_code_on_actions.txt:3634, :5143, :5510)
# — and the parent already pulses monthly like everyone else. Running the update
# on it here would be a second non-idempotent call in the same month: it would
# drift the parent's policy rate an extra third of a point toward its target and
# draw both hidden random walks twice. The parent's GDP, debt load and rank do
# move when it loses states, but the very next monthly pulse prices that in; a
# one-month lag is the correct cost, a double pulse is not.
#
# THE UPDATE IS DISPATCHED, NOT CALLED. All six of these on-actions declare
# "Expected Scope: none" (docs/on_actions.log:503, :893, :1043, :1607, :1685,
# :2081) and leave the PARENT in ROOT. Step 9's
# `multiplier = var:te_rate_paid_applied` is resolved against ROOT, so a bare
# `scope:target = { te_monetary_monthly_update = yes }` would have stamped the
# parent's rate onto the newly created tag — the same class of bug as the
# game-start one above, with a wrong number instead of a 1.0. The country event
# re-roots on the country it fires on. See events/te_monetary_events.txt.
#
# AN UPRISING'S STATE IS ONLY ITS OWN UNTIL THE WAR ENDS. The rebels need a
# rate of their own to price their loans while they fight, so they get the full
# update — and a full monetary state — like any new tag. But when a revolution
# ends the winner annexes the loser, and the variable merge keeps the WINNER'S
# value wherever both hold one (scripting_best_practices.md § "What a Civil
# War's Winner Inherits"). A rebel win therefore used to throw the nation's
# whole central bank away and keep the rebels' fresh one: the vault, inflation,
# the peg, the exchange rate, every clock and the player's mandate (audit F3,
# #462). te_monetary_on_civil_war_won below now hands the winner the loser's
# bank at the end; a loyalist win needs nothing, because the original keeps its
# own values and the rebels' only fill gaps it never had. A seceder that wins
# is a new nation and simply keeps its own.
te_monetary_on_country_released = {
	effect = {
		scope:target ?= {
			trigger_event = {
				id = te_monetary_internal.1
			}
		}
	}
}

# Root = Country (the newly formed country — the engine declares "Expected
# Scope: country" for this one, docs/on_actions.log:1037, and vanilla's own
# block tests `c:LAO ?= ROOT`, 00_code_on_actions.txt:3898-3905). ROOT is the new
# tag here, not a parent, so the bare call is correct: step 9's multiplier
# resolves against the country being updated, there is nothing to double-pulse,
# and no dispatch event is needed.
on_country_formed = {
	on_actions = {
		te_monetary_on_country_formed
	}
}

te_monetary_on_country_formed = {
	effect = {
		te_monetary_monthly_update = yes
	}
}


# ============================================================================
# THE END OF A REVOLUTION — the winner takes over the loser's central bank
# ============================================================================
# Root = the winner, after the variable merge (vanilla 00_code_on_actions.txt:
# "This action is triggered after `on_revolution_end`"; the engine declares
# "Expected Scope: none", docs/on_actions.log:1415). Owner ruling 2026-09-26:
# the winner continues the nation. What is copied, and why each group is or is
# not, is the header of common/scripted_effects/te_monetary_civil_war_effects.txt.
#
# THE COPY RUNS ONLY FOR A REBEL WIN: the shared pointer te_cw_parent (the
# foot of this file) names a DIFFERENT object with no states left — the loser
# the winner has just annexed (is_country_alive is "at least one state on the
# map"). A loyalist winner inherits the rebels' pointer, which names itself; a
# seceder never gets one.
#
# CALLED DIRECTLY, NOT DISPATCHED. ROOT is the country being written, and the
# effect applies no `multiplier =` modifier, so the te_monetary_internal.1
# detour the creation hooks need buys nothing here. It does not run the monthly
# update either: the next pulse re-applies every modifier from the copied state.
#
# THE PROBE (TE_CW_PROBE). Whether a dead but not yet deleted country can be
# read through a stored scope variable at this hook has never been observed.
# The copy also needs the loser's te_rate_paid_pts to read as a real rate (the
# canary below), so an object that resolves but reads as zeros is a logged
# no-op rather than a copy of zeros over the nation's bank.
# Every branch logs to debug.log once per win — one line, or on a rebel win two,
# either side of the copy, which read the vault two ways: line 1 prints the
# loser's figures through the loc accessor, and line 2's winner vault equals
# line 1's winner vault plus the loser's only if SCRIPT could read the loser —
# the read everything in te_monetary_inherit_central_bank depends on. Delete the
# debug_log lines once one revolution has been read (monetary_policy_design.md
# §0.4 checklist item 32).
#
# ONCE PER LOSER. The pointer outlives this hook until the winner's next
# monthly pulse, so every reader in the hook sees it whatever order the files
# run in. That leaves a window: with two revolutions at once, the winner of the
# first can beat the second as the loyalists before that pulse, while the first
# loser still lingers — and the guard alone would take its bank a second time,
# adding its gold again. te_mon_cw_bank_taken records the object the bank was
# taken from, and the guard refuses that object. It is this reader's own
# marker, not a shared "spent" flag, so no other reader depends on whether
# this on-action ran before it.
on_civil_war_won = {
	on_actions = {
		te_monetary_on_civil_war_won
		te_civil_war_settle_parent
	}
}

te_monetary_on_civil_war_won = {
	effect = {
		var:te_cw_parent ?= { save_scope_as = te_cw_parent_now }
		if = {
			limit = {
				var:te_cw_parent ?= {
					NOT = { this = root }
					is_country_alive = no
					# THE CANARY. A loser that resolves but whose variables
					# read as nothing would have the copy write zeros over the
					# nation's rate, inflation, peg and mandate — worse than no
					# copy. Every country that has pulsed holds a rate paid
					# clamped to 0.5-60, so a real read is at least 0.5.
					has_variable = te_rate_paid_pts
					var:te_rate_paid_pts >= 0.5
				}
				NOT = { var:te_mon_cw_bank_taken ?= scope:te_cw_parent_now }
			}
			var:te_cw_parent = { save_scope_as = te_cw_loser }
			debug_log = "TE_CW_PROBE monetary 1/2 [ROOT.GetCountry.GetNameNoFormatting]: rebels won, the parent pointer resolves to a dead object. BEFORE the copy, winner: vault [ROOT.GetCountry.MakeScope.Var('te_bank_gold').GetValue|0] inflation [ROOT.GetCountry.MakeScope.Var('te_inflation').GetValue|1] mandate [ROOT.GetCountry.MakeScope.Var('te_mon_mandate').GetValue|0]. Loser, read by loc: vault [SCOPE.sCountry('te_cw_loser').MakeScope.Var('te_bank_gold').GetValue|0] inflation [SCOPE.sCountry('te_cw_loser').MakeScope.Var('te_inflation').GetValue|1] mandate [SCOPE.sCountry('te_cw_loser').MakeScope.Var('te_mon_mandate').GetValue|0]"
			te_monetary_inherit_central_bank = yes
			set_variable = { name = te_mon_cw_bank_taken value = scope:te_cw_loser }
			debug_log = "TE_CW_PROBE monetary 2/2 [ROOT.GetCountry.GetNameNoFormatting]: AFTER the copy, winner: vault [ROOT.GetCountry.MakeScope.Var('te_bank_gold').GetValue|0] inflation [ROOT.GetCountry.MakeScope.Var('te_inflation').GetValue|1] mandate [ROOT.GetCountry.MakeScope.Var('te_mon_mandate').GetValue|0]. Script read the loser if this vault is the winner's plus the loser's from line 1/2 and the inflation and mandate are the loser's"
		}
		else_if = {
			limit = { var:te_cw_parent ?= root }
			debug_log = "TE_CW_PROBE monetary [ROOT.GetCountry.GetNameNoFormatting]: loyalists won (the parent pointer names the winner itself); nothing copied"
		}
		else_if = {
			limit = {
				var:te_cw_parent ?= {
					NOT = { this = root }
					is_country_alive = no
				}
				var:te_mon_cw_bank_taken ?= scope:te_cw_parent_now
			}
			debug_log = "TE_CW_PROBE monetary [ROOT.GetCountry.GetNameNoFormatting]: this parent's bank was already taken at an earlier win, before the pointer was retired; nothing copied"
		}
		else_if = {
			limit = {
				var:te_cw_parent ?= {
					NOT = { this = root }
					is_country_alive = no
				}
			}
			debug_log = "TE_CW_PROBE monetary [ROOT.GetCountry.GetNameNoFormatting]: rebels won and the parent pointer resolves to a dead object, but its figures read as nothing (no rate paid of 0.5 or more); nothing copied"
		}
		else_if = {
			limit = {
				var:te_cw_parent ?= { is_country_alive = yes }
			}
			debug_log = "TE_CW_PROBE monetary [ROOT.GetCountry.GetNameNoFormatting]: the parent pointer names another object that is still alive; nothing copied"
		}
		else_if = {
			limit = { has_variable = te_cw_parent }
			debug_log = "TE_CW_PROBE monetary [ROOT.GetCountry.GetNameNoFormatting]: the parent pointer is set but does not resolve; nothing copied"
		}
		else = {
			debug_log = "TE_CW_PROBE monetary [ROOT.GetCountry.GetNameNoFormatting]: no parent pointer (a secession, or a revolution that began before this build); nothing copied"
		}
	}
}


# ============================================================================
# [P5] INTERNATIONAL MONETARY ARRANGEMENTS (§15A.1)
# ============================================================================
# Detection is three-way: a flagged country re-runs the discovery in step 1c of
# its own monthly update; the hooks below and the treaty articles' own hooks
# (common/treaty_articles/11*.txt) dispatch it when something changes; and the
# yearly pulse runs it for EVERYBODY, as the net under anything with no hook.
# The discovery is idempotent (te_monetary_internal.2's header), so the three
# may overlap freely.

# Root = Country: called directly. Guarded like te_monetary_internal.2.
te_monetary_arrangement_yearly_on_action = {
	effect = {
		if = {
			limit = {
				te_mon_full_system = yes
				has_variable = te_mon_anchor_kind
			}
			te_monetary_discover_arrangements = yes
			# [5c] An overlord's seigniorage, re-summed (§15A.4); step 9b applies it.
			te_monetary_update_board_seigniorage = yes
		}
	}
}

# [5c] A new subject gets its currency board at once rather than at its next
# flagged pulse. Root = Country (the NEW SUBJECT — vanilla's own block says so
# and scopes `top_overlord` from it, 00_code_on_actions.txt; the engine log
# declares "Expected Scope: none", §17 check 17). Dispatched a day later so the
# subject relationship has settled; the overlord is dispatched too, so it is
# flagged as somebody's anchor. The way back is not hooked here: vanilla's
# on_become_independent (Root = country, 00_code_on_actions.txt) exists, but
# it can also pulse on an overlord-to-overlord transfer (the note in
# decolonization_events.txt), so a board would have to re-test is_subject.
# The flagged monthly discovery handles it, within a month.
on_become_subject = {
	on_actions = {
		te_monetary_on_become_subject
	}
}

te_monetary_on_become_subject = {
	effect = {
		if = {
			limit = { te_mon_full_system = yes }
			trigger_event = { id = te_monetary_internal.2 days = 1 }
			if = {
				limit = { exists = overlord }
				overlord = { trigger_event = { id = te_monetary_internal.2 days = 1 } }
			}
		}
	}
}

# A guaranteed country defaults: the guarantee is called (§15A.2). Root =
# Country. The cooldown is set HERE, the te_peg.1 / te_inflation.1 shape, so an
# unanswered event is not handed out again.
on_country_default = {
	on_actions = {
		te_monetary_on_country_default
	}
}

te_monetary_on_country_default = {
	effect = {
		if = {
			limit = {
				te_mon_full_system = yes
				te_mon_has_effective_backstop = yes
				var:te_mon_lolr_default_cooldown = 0
			}
			# [6a] THE WARD'S RECORD, first (§15C.1, ruling H3). The counter is a
			# fact about the WARD, not about whoever is standing behind it this
			# decade, so it survives a change of guarantor: a ward cannot launder
			# its record by switching backers. Incrementing BEFORE everything else
			# is the file's convention: what prices THIS call reads
			# te_mon_lolr_calls_prior (the cooldown just below, te_lolr.1's honour
			# cost and its odds), what prices the ONGOING relationship reads the
			# counter itself (the relief taper, the signing score).
			change_variable = { name = te_mon_lolr_calls add = 1 }
			clamp_variable = {
				name = te_mon_lolr_calls
				min = 0
				max = te_mon_lolr_calls_cap
			}
			set_variable = { name = te_mon_lolr_call_forget value = te_mon_lolr_call_forget_months }
			set_variable = { name = te_mon_lolr_default_cooldown value = te_mon_lolr_default_cooldown_months }
			save_scope_as = te_lolr_ward
			# A treaty guarantor first; else [5b] the leader of a tier-3 monetary
			# union, which is lender of last resort to every adopter (§15A.3).
			if = {
				limit = { te_mon_lolr_is_effective = yes }
				var:te_mon_lolr_guarantor = {
					trigger_event = { id = te_lolr.1 popup = yes }
				}
			}
			else = {
				var:te_mon_anchor = {
					trigger_event = { id = te_lolr.1 popup = yes }
				}
			}
		}
	}
}


# ============================================================================
# SHARED — the civil-war parent pointer, te_cw_parent
# ============================================================================
# NOT MONETARY. It lives here because monetary policy is its first reader
# (te_monetary_on_civil_war_won above); nuclear custody (nuclear_crisis_design.md
# §0.10) and the history charts (docs/audits/civil_war_inheritance_audit.md F9)
# are the planned next ones. Move these four on-actions to a civil-war
# on-actions file of their own when a second reader arrives.
#
# THE CONTRACT. te_cw_parent is a country SCOPE variable meaning "the country
# object whose revolution this one rose against". Read it at on_civil_war_won
# (ROOT = the winner, after the merge), and only in this form:
#
#     var:te_cw_parent ?= { NOT = { this = root } is_country_alive = no }
#
# It holds exactly when the rebels won. ROOT is then the continuation of that
# dead object, and may read it while it lingers (the German save still held the
# loser, every variable and modifier intact, after this hook; it was gone two
# weeks later). Never test has_variable alone.
#
#   * SET on the rebels at on_revolution_start, to ROOT, the original.
#   * CLEARED on the original at on_revolution_start AND on_secession_start. A
#     country that once won a revolution as the rebels still carries a pointer
#     to the dead object it replaced. If a later war left it the LOYALIST
#     winner, the merge would keep that stale value (the winner's own value
#     wins) and a reader would take the win for a rebel one. Clearing at every
#     start means the pointer a winner holds always comes from this war.
#   * DROPPED at on_civil_war_won when it names the winner itself, which is
#     what a loyalist winner inherits from the rebels it defeated.
#   * RETIRED after any other win (a rebel win, as a rule), at the winner's
#     next monthly pulse — not in the hook, so every reader of on_civil_war_won
#     sees it whatever order the hook's files run in. te_civil_war_settle_parent
#     marks it (te_cw_parent_retire, which no reader tests) and
#     te_civil_war_retire_parent removes both. Until then it still names the
#     dead loser, and a second
#     civil war can end inside that window (two revolutions at once, the winner
#     of the first beating the second). A reader that must act ONCE per loser
#     therefore keeps its own marker of the object it acted on and refuses it
#     (te_monetary_on_civil_war_won's te_mon_cw_bank_taken) — its own, so no
#     reader depends on another having run first.
#   * RE-PARENTED at a rebel win: every other uprising whose pointer names the
#     annexed loser now fights the winner, its continuation, so its pointer is
#     set to ROOT. If that uprising later wins, it takes over the winner's
#     state rather than the long-dead loser's. One that ends without a win of
#     its own keeps a pointer to a LIVE country until it is the original at a
#     start hook; the guard's is_country_alive = no makes it inert.
#   * NEVER SET FOR A SECESSION. A seceder never continues the original: if it
#     wins it is a new nation and the original survives; if it loses, the
#     original is the winner. There is nothing for a reader to inherit, and a
#     pointer on a seceder would turn "dead parent" into a hazard — an original
#     annexed by a third party mid-war would read as a rebel win. A future
#     reader that needs a seceder's origin (nuclear custody's Budapest path)
#     should store it under a name of its own.
#
# The TE_CW_PROBE lines are the probe te_monetary_on_civil_war_won describes;
# delete them with it.

# Root = Country (the original), scope:target = Uprising country.
te_civil_war_record_parent = {
	effect = {
		if = {
			limit = { has_variable = te_cw_parent }
			remove_variable = te_cw_parent
		}
		if = {
			limit = { has_variable = te_cw_parent_retire }
			remove_variable = te_cw_parent_retire
		}
		scope:target ?= {
			set_variable = { name = te_cw_parent value = root }
			debug_log = "TE_CW_PROBE parent pointer recorded on the revolutionaries rising against [ROOT.GetCountry.GetNameNoFormatting]"
		}
	}
}

# Root = Country (the original), scope:target = Uprising country (a seceder).
te_civil_war_forget_parent = {
	effect = {
		if = {
			limit = { has_variable = te_cw_parent }
			remove_variable = te_cw_parent
		}
		if = {
			limit = { has_variable = te_cw_parent_retire }
			remove_variable = te_cw_parent_retire
		}
	}
}

# Root = the winner. Listed after te_monetary_on_civil_war_won, which logs the
# loyalist branch before this drops the pointer. Nothing here removes a pointer
# a reader could still need: the self-pointer means "nothing to read", and a
# rebel winner's pointer is only MARKED for te_civil_war_retire_parent. The
# re-parenting writes other countries' pointers, never ROOT's.
te_civil_war_settle_parent = {
	effect = {
		if = {
			limit = { var:te_cw_parent ?= root }
			remove_variable = te_cw_parent
			debug_log = "TE_CW_PROBE parent pointer named the winner itself (a loyalist win); dropped"
		}
		else_if = {
			# Any other pointer has served this war, whatever it resolves to.
			limit = { has_variable = te_cw_parent }
			set_variable = { name = te_cw_parent_retire value = yes }
			if = {
				limit = {
					var:te_cw_parent ?= {
						NOT = { this = root }
						is_country_alive = no
					}
				}
				var:te_cw_parent = { save_scope_as = te_cw_retired_parent }
				every_country = {
					limit = {
						NOT = { this = root }
						var:te_cw_parent ?= scope:te_cw_retired_parent
					}
					set_variable = { name = te_cw_parent value = root }
				}
			}
		}
	}
}

# Root = Country. The monthly pulse after a rebel win: the pointer has served
# every reader of that on_civil_war_won, so it and its mark go. One has_variable
# a month for every other country.
te_civil_war_retire_parent = {
	effect = {
		if = {
			limit = { has_variable = te_cw_parent_retire }
			if = {
				limit = { has_variable = te_cw_parent }
				remove_variable = te_cw_parent
			}
			remove_variable = te_cw_parent_retire
		}
	}
}
