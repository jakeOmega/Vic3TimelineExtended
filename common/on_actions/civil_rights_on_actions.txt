# ============================================================================
# CIVIL RIGHTS - revolution continuity (issue #463)
# je_civil_rights's own pulse runs only while the entry is active. These hooks
# carry what outlives it: the outcome a resolved struggle left behind, and what a
# revolution's winner needs, rebuilt or dropped depending on which side won.
# Design and variables:
# common/scripted_effects/civil_rights_effects.txt § REVOLUTION CONTINUITY.
# ============================================================================

on_monthly_pulse_country = {
	on_actions = {
		civil_rights_country_monthly_on_action
	}
}

# Counts the months since the outcome modifier was granted, so a winner gets back
# only what the loser had left, and migrates saves from before #463.
civil_rights_country_monthly_on_action = {
	effect = {
		cr_outcome_monthly_update = yes
	}
}

# Root = the original country, scope:target = the uprising country. The original
# records itself so that on_civil_war_won can tell which side won: the
# revolutionaries start with no mod variables, so if they win they inherit this
# pointer to the dead loser; if the loyalists win, it still names ROOT. The same
# for a secession, so loyalists who crush one also drop what the secessionists
# held (a seceded winner is a separate country without the pointer: a no-op).
on_revolution_start = {
	on_actions = {
		civil_rights_revolution_start_on_action
	}
}

on_secession_start = {
	on_actions = {
		civil_rights_revolution_start_on_action
	}
}

civil_rights_revolution_start_on_action = {
	effect = {
		set_variable = { name = cr_revolution_original value = ROOT }
	}
}

# Root = the winner, after the loser's variables have merged into it. Runs after
# on_revolution_end. The engine drops every modifier in the merge. If the
# revolutionaries won, re-add the outcome modifier (for the time it had left)
# and, for a struggle still in progress, the button policies. If the loyalists
# won, they keep their own modifiers; drop the policy mirrors and run marker
# they inherited from the rebels' own struggle.
on_civil_war_won = {
	on_actions = {
		civil_rights_civil_war_won_on_action
	}
}

civil_rights_civil_war_won_on_action = {
	effect = {
		cr_after_civil_war = yes
	}
}
