# ============================================================================
# COLONIAL GEOGRAPHY — the separation cache's two passes
#
# Every colony test reads whether a state is separated from its owner's
# capital from a monthly cache (ce_refresh_separation,
# common/scripted_effects/colonial_geography_effects.txt). It is filled at game
# start for every country and refreshed on each country's monthly pulse,
# whatever its technology: the naval laws' AI weights, the anti-colonialist
# ideology and the UN votes read the colony test at any date.
#
# The monthly pass then runs the Colonial Empire's re-entry pass
# (ce_reentry_monthly, common/scripted_effects/decolonization.txt), which gives
# a country whose entry has ended the entry back once it qualifies again. It
# runs here, after the step, so the colony test it asks reads this month's
# cache; another file's on_action may run before this one.
#
# Last, the self-checks of the civil-war reading the re-entry pass and the AI's
# hold rest on (colonial_civil_war_selfcheck, decolonization.txt; #1735).
# ============================================================================

on_game_started = {
	on_actions = {
		colonial_geography_game_start_on_action
	}
}

on_monthly_pulse_country = {
	on_actions = {
		colonial_geography_on_action
	}
}

colonial_geography_game_start_on_action = {
	effect = {
		every_country = {
			ce_separation_step = yes
		}
	}
}

colonial_geography_on_action = {
	effect = {
		ce_separation_step = yes
		ce_reentry_monthly = yes
		colonial_civil_war_selfcheck = yes
	}
}
