# Global Commons § 1.2: one once-a-game self-check, logistics_input_cut.
#
# The Extraplanetary Base's era-12 logistics upgrades (Tethered Launch, Fusion Tugs;
# common/production_methods/te_global_commons_logistics_pms.txt) cut its Launch Capacity input
# with goods_input_launch_capacity_mult in building_modifiers.unscaled. Vanilla registers that
# axis for six goods only; the mod registers this one (mod_entity_modifier_types.txt). The first
# yearly pulse of a country with an Extraplanetary Base of level 3 or more on either upgrade
# reads one such building's sum of the type:
#   OK      -0.65 to -0.25: one or two cuts of 30%, summed once per building
#   BROKEN  about 0: the type is unknown to the engine or the block isn't summed, so the
#           upgrades do nothing (debug.log then says "Unknown modifier type"); -0.9 or below:
#           unscaled is read per level, and a large base would buy no Launch Capacity at all.
# Fallback: for 0, fix the registration; for a per-level sum, apply the cut as a static
# modifier on the building from a yearly pulse instead (the Strategic Reserve hub's pattern).
# Play-test rows: docs/testing/global-commons-playtest-a.md.
on_yearly_pulse_country = {
	on_actions = {
		te_gc_logistics_input_cut_selfcheck
	}
}

te_gc_logistics_input_cut_selfcheck = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = te_gc_logistics_input_cut_checked }
				any_scope_building = {
					level >= 3
					OR = {
						has_active_production_method = pm_tethered_launch_building_space_mine
						has_active_production_method = pm_fusion_tugs_building_space_mine
					}
				}
			}
			set_global_variable = te_gc_logistics_input_cut_checked
			random_scope_building = {
				limit = {
					level >= 3
					OR = {
						has_active_production_method = pm_tethered_launch_building_space_mine
						has_active_production_method = pm_fusion_tugs_building_space_mine
					}
				}
				set_global_variable = { name = te_gc_logistics_chk_cut value = modifier:goods_input_launch_capacity_mult }
				set_global_variable = { name = te_gc_logistics_chk_level value = level }
				set_global_variable = { name = te_gc_logistics_chk_upgrades value = te_gc_logistics_active_upgrades }
				if = {
					limit = {
						modifier:goods_input_launch_capacity_mult >= -0.65
						modifier:goods_input_launch_capacity_mult <= -0.25
					}
					debug_log = "TE_SELF_CHECK OK logistics_input_cut (an Extraplanetary Base on upgrades=[SCOPE.ScriptValue('te_gc_logistics_chk_upgrades_g')|0] logistics upgrades reads goods_input_launch_capacity_mult=[SCOPE.ScriptValue('te_gc_logistics_chk_cut_g')|2] at level=[SCOPE.ScriptValue('te_gc_logistics_chk_level_g')|0]: the cut counts once per building)"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN logistics_input_cut (an Extraplanetary Base on upgrades=[SCOPE.ScriptValue('te_gc_logistics_chk_upgrades_g')|0] logistics upgrades reads goods_input_launch_capacity_mult=[SCOPE.ScriptValue('te_gc_logistics_chk_cut_g')|2] at level=[SCOPE.ScriptValue('te_gc_logistics_chk_level_g')|0], not -0.3 per upgrade: about 0 means the type is unregistered or not summed, about -0.3 x upgrades x level means unscaled is read per level; see docs/testing/global-commons-playtest-a.md)"
				}
			}
		}
	}
}
