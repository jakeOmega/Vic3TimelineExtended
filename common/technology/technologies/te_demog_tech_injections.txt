# Demographics: the base game's medical techs' treatment lines, the motor age's road deaths, and the transport techs'
# families among migrants.
# Spec: docs/superpowers/specs/2026-10-09-demographics-modifier-types-design.md (stage 1).
# Read only by the census (te_demog_mult_*); the engine never applies these types. The values
# are calibrated with `scripts/analysis/demographics_harness.py medicine`
# (docs/testing/demographics-fast-run-2026-10-09.md). The mod's own medical techs carry their
# lines in their own definitions (era_6.txt to era_11.txt). pharmaceuticals' line sits in its
# existing INJECT in modified.txt, so the tech keeps one INJECT.

INJECT:medical_degrees = {
	modifier = {
		state_infection_treatment_add = 0.03
	}
}

INJECT:modern_nursing = {
	modifier = {
		state_infection_treatment_add = 0.08
		state_maternal_treatment_add = 0.35
		state_medical_diffusion_add = 0.05
	}
}

# Medicine spreads faster with communication and medical training (the Causes of Death plan): the census's share of the
# gap closed each year, 10% everywhere plus 5% for each of Electric Telegraph, Modern Nursing, Radio (modified.txt),
# Intergovernmental Organizations, World Wide Web and Telemedicine (in their own definitions).
INJECT:electric_telegraph = {
	modifier = {
		state_medical_diffusion_add = 0.05
	}
}

INJECT:antibiotics = {
	modifier = {
		state_infection_treatment_add = 0.4
		state_maternal_treatment_add = 0.6
	}
}

# Families among migrants (modifier-types design, stage 3, 2026-10-19): cheaper travel lets migrants bring their families,
# the family kind's weight in the census's migrant profile on top of FAMILY_BASE. Combustion Engine's line sits in its
# block below; Commercial Aviation and Jet Engine Technology carry theirs in era_6.txt and era_7.txt.
INJECT:paddle_steamer = {
	modifier = {
		state_family_migration_add = 0.1
	}
}

INJECT:railways = {
	modifier = {
		state_family_migration_add = 0.1
	}
}

INJECT:combustion_engine = {
	modifier = {
		state_external_mortality_mult = 0.3
		state_family_migration_add = 0.15
	}
}

# Stage 2 (fertility): rubber condoms. The census adds each means tech's line to the
# traditional 0.4 (demographics_harness.py fertility). contraceptive_pill and
# modern_pharmaceuticals carry theirs in era_7.txt and era_8.txt.
INJECT:vulcanization = {
	modifier = {
		state_contraception_add = 0.2
	}
}

# Feminism: the will to use the methods, where vulcanization is the methods (owner,
# 2026-10-10: birth control was a feminist cause). Both are era 3; no anchor separates
# them, so the fit sets their sum and splits it.
INJECT:feminism = {
	modifier = {
		state_contraception_add = 0.15
	}
}

# Malaria Prevention ends the census's malaria factor (calibration step 3, the spec's §2.2): the people who lived with
# malaria died of it less only with 20th-century medicine. Quinine, which vanilla's malaria trait takes for its
# colonists, carries nothing: it protected colonists, not the people who lived there.
INJECT:malaria_prevention = {
	modifier = {
		state_malaria_control_add = 1
	}
}
