# ============================================================================
# WAR OF INDEPENDENCE — DIPLOMATIC PLAYS
#
# dp_te_independence_war: the play the rebels open against the metropole that
# chose to fight them (decolonization_events.70's Fight, or Blood in the
# Colonies'), the day after they form (.71, decol_iw_open_war in
# decolonization_war_effects.txt). Script-only, as vanilla dp_sepoy_mutiny is:
# never in the lens, created at escalation 80 as that one is, so the war breaks
# out in three weeks or so. The rebels' goal is te_colonial_independence; the
# metropole's, added at the creation as its primary demand (vanilla's Ezo
# republic shape), is te_colonial_reconquest. A side that backs down should
# concede the other's primary demand (unverified in this mod; the self-check
# reads it): the rebels, to be annexed back; the metropole, to let them go.
#
# dp_te_colonial_war: the first build's shape (2026-10-24), the metropole's
# play against the rebels with `war = yes` and escalation 100. In the owner's
# first game an AI empire's play never became a war (TE_SELF_CHECK BROKEN
# decol_iw_war_started; docs/testing/colonial-wars-playtest.md). The likeliest
# reading: it backed down, which cost it nothing, since the rebels held no
# demand. Nothing opens it now. It stays defined for a save that holds one.
#
# Neither goal self-enforces (no mirrored_wargoal): holding the territory does
# not end the war, only a capitulation or a peace does. A negotiated peace
# stays allowed: a white peace is independence. Zero infamy (owner ruling D7,
# 2026-10-24): the condemnation comes from the great powers and the bar.
#
# Design: docs/superpowers/archive/specs/2026-10-24-colonial-criteria-and-independence-wars-design.md § 2.4.
# ============================================================================

dp_te_independence_war = {
	war_goal = te_colonial_independence
	requires_interest_marker = no
	add_infamy_for_starting_initiator_wargoals = no
	add_infamy_for_starting_target_wargoals = no
	# The war is about the colony and nothing else (_diplomatic_plays.info).
	initiator_can_add_war_goals = no
	target_can_add_war_goals = no
	texture = "gfx/interface/icons/war_goals/independence.dds"
	selectable_in_lens = { always = no }

	# Only the rebels, against the metropole they name (decol_iw_rebels_formed).
	possible = {
		var:decol_iw_metropole ?= scope:target_country
	}

	ai_acceptance_max = 0

	on_weekly_pulse = { }
	on_war_begins = { }
	on_war_end = { }
}

dp_te_colonial_war = {
	war_goal = te_colonial_reconquest
	requires_interest_marker = no
	add_infamy_for_starting_initiator_wargoals = no
	# The war is about the colony and nothing else (_diplomatic_plays.info).
	initiator_can_add_war_goals = no
	target_can_add_war_goals = no
	texture = "gfx/interface/icons/war_goals/annex_country.dds"
	selectable_in_lens = { always = no }

	# Only against our own rebels: the country formed from our territory that
	# names us as the metropole it fights (decol_iw_rebels_formed).
	possible = {
		scope:target_country = {
			var:decol_iw_metropole ?= root
		}
	}

	# The rebels' AI never gives the territory back at the table.
	ai_acceptance_max = 0

	on_weekly_pulse = { }
	on_war_begins = { }
	on_war_end = { }
}

# dp_te_colonial_uprising: the play a colonial subject opens against its
# overlord when the overlord holds the line at decolonization_events.3 (1.15
# can_start_play_against_overlord gates it: no war of the overlord's targets
# the subject and none of its states is occupied; the overlord may be at war
# elsewhere). Its goal is vanilla's independence, held by the subject. A play
# of the mod's own, not vanilla dp_independence, whose possible asks for a
# refused autonomy request that a script-opened uprising never had. Once it is
# a war, the monthly settle gives it a record of kind 2
# (decol_iw_track_subject_wars), as any colonial subject's war against us.
dp_te_colonial_uprising = {
	war_goal = independence
	requires_interest_marker = no
	texture = "gfx/interface/icons/war_goals/independence.dds"
	confirmation_sound = "event:/SFX/UI/SubjectInteractions/subject_button_small_independence"
	selectable_in_lens = { always = no }

	# Only a direct subject, against its overlord.
	possible = {
		is_direct_subject_of = scope:target_country
	}

	ai_acceptance_max = 100

	on_weekly_pulse = { }
	on_war_begins = { }
	on_war_end = { }
}
