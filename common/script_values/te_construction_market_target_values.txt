# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

# The private sector's weekly construction spending: a 24th of the investment
# pool, so the pool settles at 24 weeks of gross income. Read from the pool
# alone so it holds whatever the sign of gross income. The earlier form,
# gross x pool / (24 x gross), is the same while gross income is positive, but
# fell back to a fixed 2.4M divisor at 0 or below, which made the target 0 or
# negative. Overinvestment (extra_on_actions.txt) cuts every pop's contribution,
# which pushes gross income there, so it switched off the private spending it
# exists to force.
te_construction_market_private_spending_target = {
	value = 0
	if = {
		limit = {
			investment_pool > 0
		}
		value = investment_pool
		divide = 24
	}
}


# The government purchase the player has set, as construction points a week:
#   Fixed Quantity (default): the number in the construction panel's +/- box
#   Fixed Budget:             the quantity the weekly budget is buying now
#                             (var:te_construction_market_budget_quantity,
#                             moved each week by
#                             te_construction_market_update_budget_quantity)
te_construction_market_public_setting = {
	value = 0
	if = {
		limit = {
			te_construction_market_budget_mode_on = yes
		}
		if = {
			limit = {
				has_variable = te_construction_market_budget_quantity
			}
			value = var:te_construction_market_budget_quantity
		}
	}
	else_if = {
		limit = {
			has_variable = te_construction_market_public_target
		}
		value = var:te_construction_market_public_target
	}
}

# What the government queue can absorb in a week: each queued level takes at
# most country_max_weekly_construction_progress_add.
te_construction_market_public_queue_cap = {
	value = construction_queue_num_queued_government_levels
	multiply = modifier:country_max_weekly_construction_progress_add
}

te_construction_market_public_use_target = {
	if = {
		limit = {
			has_variable = te_construction_market_public_target
		}
		if = {
			limit = {
				is_construction_paused = no
			}
			value = te_construction_market_public_setting
			# Cap at what the government queue can absorb this week, mirroring
			# the private-side cap in te_construction_market_private_use_target.
			# A player-set target above that would otherwise buy construction
			# the government queue can't use. Only the effective use is capped;
			# the variables (the number or budget shown in the construction
			# panel) keep the player's setting. The AI's target is already
			# capped the same way in te_construction_market_ai_purchase_target.
			max = te_construction_market_public_queue_cap
		}
		else = {
			value = 0
		}

	}
	else = {
		value = 1
	}
}

te_construction_market_investment_pool_x_1000 = {
	value = investment_pool
	multiply = 1000
}

te_construction_market_private_use_target = {
	value = te_construction_market_private_spending_target
	divide = var:te_construction_market_price_stabilizer
	if = {
		limit = {
			investment_pool_gross_income > 100000000
		}
		min = 1000
	}
	# Cap at what the private queue can absorb this week: each queued level
	# takes at most country_max_weekly_construction_progress_add. Without it,
	# an empty or short private queue still sizes site throughput (and the
	# government-funded construction-goods purchase) for the full
	# investment-pool target; the private sector can't use that slice, so the
	# government ends up spending past te_construction_market_public_target.
	# Worse, an unspendable pool keeps growing, which inflates this target.
	# Last clamp on purpose — it must override the `min = 1000` floor above.
	max = {
		value = construction_queue_num_queued_private_levels
		max = 1000
		multiply = modifier:country_max_weekly_construction_progress_add
	}
}


# What the country's construction sites together deliver this week.
#   market:              the government purchase + the private purchase; the
#                        per-site throughput modifier divides it among the sites
#   direct construction: the country's construction points, which the sites'
#                        production methods provide themselves (read by
#                        needed_private_levels; nothing sizes a site here)
te_construction_market_total_use_target = {
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		value = te_direct_construction_display_capacity
	}
	else = {
		value = te_construction_market_private_use_target
		add = te_construction_market_public_use_target
	}
}

# Multiplier of te_construction_market_private_share_modifier: the private
# share of construction, 0-100.
#   market:              follows the two purchases (below)
#   direct construction: the economic-system law's vanilla allocation
te_construction_market_private_share_target = {
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		value = te_direct_construction_law_private_share
	}
	else = {
		value = te_construction_market_market_private_share_target
	}
}

te_construction_market_market_private_share_target = {
	if = {
		limit = {
			te_construction_market_total_use_target > 0
		}
		value = te_construction_market_private_use_target
		divide = te_construction_market_total_use_target
	}
	else = {
		value = 1
	}
	multiply = 100
	# Keep a token private allocation while the private queue is empty (the
	# cap in te_construction_market_private_use_target then zeroes the share)
	# so the autonomous investment AI is never left at a 0% allocation, a
	# state vanilla never reaches. It adds no throughput (sites are still
	# sized to public + private use); worst case, if the engine wastes rather
	# than re-routes unused private allocation, ~1% of the government-bought
	# construction goes unused. Drop this block to trade that for a 0% share.
	if = {
		limit = {
			investment_pool_gross_income > 0
		}
		min = 1
	}
}

te_construction_market_price_stabilizer_next_value = {
	if = {
		limit = {
			NOT = {
				has_variable = te_construction_market_price_stabilizer
			}
		}
		value = TE_CONSTRUCTION_MARKET_SERVICES_PRICE
	}
	else = {
		if = {
			limit = {
				var:te_construction_market_price_stabilizer < te_construction_market_price
			}
			value = var:te_construction_market_price_stabilizer
			add = {
				value = te_construction_market_price
				subtract = var:te_construction_market_price_stabilizer
				divide = 10
			}
		}
		else = {
			value = var:te_construction_market_price_stabilizer
			subtract = {
				value = var:te_construction_market_price_stabilizer
				subtract = te_construction_market_price
				divide = 10
			}
		}
	}
}

# ----------------------------------------------------------------------------
# Fixed Budget mode (te_construction_market_budget_mode_on)
# ----------------------------------------------------------------------------
# The player sets what the government spends a week
# (var:te_construction_market_public_budget) instead of how much it buys. The
# quantity bought (var:te_construction_market_budget_quantity) moves each
# weekly pulse 1 / N of the way toward what the budget buys at that week's
# price (te_construction_market_update_budget_quantity), N being
# te_construction_market_budget_shift_weeks.
#
# Why only part of the way: the government's purchase is part of the demand in
# its market, so it moves the price it is divided by. Setting the quantity to
# budget / price outright chases that: buy more, the price rises, buy less next
# week, the price falls, and so on. If a 1% larger purchase raises the price by
# e%, closing 1/N of the gap each week leaves 1 - (1 + e) / N of it.
# For a small gap, N = 1 + e is the exact first-order step (Newton's method on
# quantity x price = budget): a purchase that moves the price little closes the
# gap at once, and a larger N approaches without overshooting. Across a large
# gap the price is not linear in the purchase (it stops moving at its floor and
# its ceiling), and a slope read at either end of the step misses that: a raise
# near the floor overshot by several times in testing. So N is taken from the
# chord between where the purchase is (Q) and where the budget wants it (T):
#   N = 1 + T / (T - Q) x (price at T / price now - 1)
# which is where the line through (Q, spending now - budget) and
# (T, spending at T - budget) crosses zero, and is 1 + e as the gap closes.
#
# The two prices come from the engine's price formula, so N follows the market.
# e is about 0.01 for a purchase that is a small part of a large market, and far
# larger when the purchase is a big part of it or the price is near its floor,
# where any fixed step is either too slow or overshoots. The formula, with the
# two defines it uses (common/defines/extra_defines.txt):
#   price = base * (1 + PRICE_RANGE * clamp(imbalance / (F - 1), -1, 1))
#   imbalance = (BUY - SELL) / min(BUY, SELL)
#   F = BUY_SELL_DIFF_AT_MAX_FACTOR: the price is at its ceiling when BUY = F x
#   SELL and at its floor when SELL = F x BUY (vanilla's F = 2 gives the
#   unnormalised clamp(imbalance, -1, 1); the mod's F = 4 makes it a third as steep)
# BUY and SELL are the construction good's market_goods_buy_orders and
# market_goods_sell_orders. The government's own share of BUY is the purchase in
# goods (points x te_construction_market_goods_per_point).
#
# The model sets the size of the step, not where it lands. Each week's target is
# still budget / the price the market actually shows, so the quantity settles
# where spending equals the budget whatever the model gets wrong (the site's
# local price against the market's, other buyers reacting, an engine lag); a
# wrong N only makes the approach slower or faster.
#
# The floor on N, when the budget is being raised, is insurance for those same
# blind spots: an overshoot on a raise spends past the budget. A cut has no
# floor: an overshoot there spends less than the budget, and a saving shouldn't
# be held back.

# The engine defines the price formula uses. Script can't read defines, so they
# are mirrored here; test_construction_market_budget_step.py fails when they
# differ from common/defines/extra_defines.txt.
TE_CONSTRUCTION_MARKET_PRICE_RANGE = 0.999
TE_CONSTRUCTION_MARKET_BUY_SELL_DIFF_AT_MAX_FACTOR = 4

# A raise in the budget is spread over at least this many weeks (N is never
# below it): a half-step at most.
TE_CONSTRUCTION_MARKET_BUDGET_MIN_RAISE_WEEKS = 2

# What the weekly budget buys at this week's price.
te_construction_market_budget_quantity_target = {
	value = 0
	if = {
		limit = {
			has_variable = te_construction_market_public_budget
		}
		value = var:te_construction_market_public_budget
		divide = {
			value = te_construction_market_price
			min = 1
		}
	}
}

# How far that is from the quantity being bought now.
te_construction_market_budget_quantity_gap = {
	value = te_construction_market_budget_quantity_target
	if = {
		limit = {
			has_variable = te_construction_market_budget_quantity
		}
		subtract = var:te_construction_market_budget_quantity
	}
}

# The step. Everything below is a read of the market as the last engine update
# left it; the chain is only read from the weekly pulse (and by the panel).
#
# Goods the construction sites use per construction point: the same factor
# te_construction_market_price scales the price by (construction cost scaling,
# at least 1).
te_construction_market_goods_per_point = {
	value = 1
	add = construction_cost_scaling_mult
	min = 1
}

# The construction good's buy and sell orders in the country's market
# (everyone in it, maintenance included). Block form, not a this.market.mg: dot
# chain (docs/guides/scripting_best_practices.md, "Is it signed below base
# price?"). Sell orders are floored at 1: they are divided by.
te_construction_market_budget_buy_orders = {
	value = 0
	market = {
		mg:construction = {
			add = market_goods_buy_orders
		}
	}
}

te_construction_market_budget_sell_orders = {
	value = 0
	market = {
		mg:construction = {
			add = market_goods_sell_orders
		}
	}
	min = 1
}

# The government's share of the buy orders, in goods: the purchase in force
# (the one those orders were placed for) ...
te_construction_market_budget_goods_now = {
	value = te_construction_market_public_use_target
	multiply = te_construction_market_goods_per_point
}

# ... and the purchase the budget wants at today's price, as far as the queue
# can use it (nothing while construction is paused).
te_construction_market_budget_goods_target = {
	value = te_construction_market_budget_quantity_target
	max = te_construction_market_public_queue_cap
	multiply = te_construction_market_goods_per_point
	if = {
		limit = {
			is_construction_paused = yes
		}
		value = 0
	}
}

# Everyone else's buy orders: the orders less the government's purchase in
# force (never below 0). The orders with the government at either size are
# these plus its goods.
te_construction_market_budget_other_buy_orders = {
	value = te_construction_market_budget_buy_orders
	subtract = te_construction_market_budget_goods_now
	min = 0
}

# The price relative to base with the government buying the purchase in force
# and, below, the purchase the budget wants. imbalance is
# (BUY - SELL) / min(BUY, SELL) / (F - 1), unclamped: -1 or below and 1 or above
# are the price's floor and ceiling. min(BUY, SELL) is `max = SELL` on BUY, and
# is floored at 1: it is divided by.
te_construction_market_budget_imbalance_now = {
	value = te_construction_market_budget_other_buy_orders
	add = te_construction_market_budget_goods_now
	subtract = te_construction_market_budget_sell_orders
	divide = {
		value = te_construction_market_budget_other_buy_orders
		add = te_construction_market_budget_goods_now
		max = te_construction_market_budget_sell_orders
		min = 1
	}
	divide = {
		value = TE_CONSTRUCTION_MARKET_BUY_SELL_DIFF_AT_MAX_FACTOR
		subtract = 1
	}
}

te_construction_market_budget_price_factor_now = {
	value = te_construction_market_budget_imbalance_now
	min = -1
	max = 1
	multiply = TE_CONSTRUCTION_MARKET_PRICE_RANGE
	add = 1
	min = 0.001
}

te_construction_market_budget_imbalance_target = {
	value = te_construction_market_budget_other_buy_orders
	add = te_construction_market_budget_goods_target
	subtract = te_construction_market_budget_sell_orders
	divide = {
		value = te_construction_market_budget_other_buy_orders
		add = te_construction_market_budget_goods_target
		max = te_construction_market_budget_sell_orders
		min = 1
	}
	divide = {
		value = TE_CONSTRUCTION_MARKET_BUY_SELL_DIFF_AT_MAX_FACTOR
		subtract = 1
	}
}

te_construction_market_budget_price_factor_target = {
	value = te_construction_market_budget_imbalance_target
	min = -1
	max = 1
	multiply = TE_CONSTRUCTION_MARKET_PRICE_RANGE
	add = 1
	min = 0.001
}

# N, in weeks: 1 + T / (T - Q) x (price at T / price now - 1), with T and Q the
# purchase the budget wants and the purchase in force, in goods (their ratio is
# the same in points). 1 when the two are equal: a purchase held at the queue's
# cap doesn't move the price. A cut can't fall below 1 (it never goes past the
# target); a raise is never spread over fewer than
# TE_CONSTRUCTION_MARKET_BUDGET_MIN_RAISE_WEEKS.
te_construction_market_budget_shift_weeks = {
	value = 1
	if = {
		limit = {
			OR = {
				te_construction_market_budget_goods_target > te_construction_market_budget_goods_now
				te_construction_market_budget_goods_target < te_construction_market_budget_goods_now
			}
		}
		value = te_construction_market_budget_price_factor_target
		divide = te_construction_market_budget_price_factor_now
		subtract = 1
		multiply = {
			value = te_construction_market_budget_goods_target
			divide = {
				value = te_construction_market_budget_goods_target
				subtract = te_construction_market_budget_goods_now
			}
		}
		add = 1
		min = 1
	}
	if = {
		limit = {
			te_construction_market_budget_quantity_gap > 0
		}
		min = TE_CONSTRUCTION_MARKET_BUDGET_MIN_RAISE_WEEKS
	}
}

# Next week's quantity: 1 / N of the gap closed. Within a point of the target
# it takes the target, so a budget cut to 0 reaches 0 instead of approaching it
# forever.
te_construction_market_budget_quantity_next_value = {
	value = te_construction_market_budget_quantity_target
	if = {
		limit = {
			has_variable = te_construction_market_budget_quantity
			OR = {
				te_construction_market_budget_quantity_gap > 1
				te_construction_market_budget_quantity_gap < -1
			}
		}
		value = te_construction_market_budget_quantity_gap
		divide = te_construction_market_budget_shift_weeks
		add = var:te_construction_market_budget_quantity
	}
}

# Switching from Fixed Quantity to Fixed Budget carries the purchase over, so
# nothing changes at the switch: the quantity to start from is the purchase in
# force this week, or, while none is (an empty government queue, or
# construction paused), the quantity set.
te_construction_market_budget_seed_quantity = {
	value = te_construction_market_public_use_target
	if = {
		limit = {
			te_construction_market_public_use_target <= 0
		}
		value = te_construction_market_public_setting
	}
}

# ...and the budget is what that costs at this week's price, to the nearest
# 1,000 (the smallest budget step in the panel).
te_construction_market_budget_seed = {
	value = te_construction_market_budget_seed_quantity
	multiply = te_construction_market_price
	divide = 1000
	round = yes
	multiply = 1000
}
