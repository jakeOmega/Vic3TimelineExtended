# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

te_construction_market_private_spending_target = {
	value = investment_pool_gross_income
	multiply = {
		value = investment_pool
		subtract = te_construction_market_investment_pool_size_target
		divide = te_construction_market_investment_pool_size_target
		add = 1
	}
}


te_construction_market_public_use_target = {
	if = {
		limit = {
			has_variable = te_construction_market_public_target
		}
		if = {
			limit = {
				is_construction_paused = no
			}
			value = var:te_construction_market_public_target
			# Cap at what the government queue can absorb this week, mirroring
			# the private-side cap in te_construction_market_private_use_target.
			# A player-set target above that would otherwise buy construction
			# the government queue can't use. Only the effective use is capped;
			# the variable (the number shown in the construction panel) keeps
			# the player's setting. The AI's target is already capped the same
			# way in te_construction_market_ai_purchase_target.
			max = {
				value = construction_queue_num_queued_government_levels
				multiply = modifier:country_max_weekly_construction_progress_add
			}
		}
		else = {
			value = 0
		}

	}
	else = {
		value = 1
	}
}

te_construction_market_investment_pool_size_target = {
	if = {
		limit = {
			investment_pool_gross_income > 0
		}
		value = investment_pool_gross_income
	}
	else = {
		value = 100000
	}
	multiply = 24
}

te_construction_market_investment_pool_x_1000 = {
	value = investment_pool
	multiply = 1000
}

te_construction_market_private_use_target = {
	value = te_construction_market_private_spending_target
	divide = var:te_construction_market_price_stabilizer
	if = {
		limit = {
			investment_pool_gross_income > 100000000
		}
		min = 1000
	}
	# Cap at what the private queue can absorb this week: each queued level
	# takes at most country_max_weekly_construction_progress_add. Without it,
	# an empty or short private queue still sizes site throughput (and the
	# government-funded construction-goods purchase) for the full
	# investment-pool target; the private sector can't use that slice, so the
	# government ends up spending past te_construction_market_public_target.
	# Worse, an unspendable pool keeps growing, which inflates this target.
	# Last clamp on purpose — it must override the `min = 1000` floor above.
	max = {
		value = construction_queue_num_queued_private_levels
		max = 1000
		multiply = modifier:country_max_weekly_construction_progress_add
	}
}


# What the country's construction sites together deliver this week; the
# per-site throughput modifier divides it among them.
#   market:              the government purchase + the private purchase
#   direct construction: what the Construction Sectors would deliver
#                        (te_direct_construction_refresh_capacity)
te_construction_market_total_use_target = {
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		value = te_direct_construction_display_capacity
	}
	else = {
		value = te_construction_market_private_use_target
		add = te_construction_market_public_use_target
	}
}

# Multiplier of te_construction_market_private_share_modifier: the private
# share of construction, 0-100.
#   market:              follows the two purchases (below)
#   direct construction: the economic-system law's vanilla allocation
te_construction_market_private_share_target = {
	if = {
		limit = {
			te_free_market_construction_off = yes
		}
		value = te_direct_construction_law_private_share
	}
	else = {
		value = te_construction_market_market_private_share_target
	}
}

te_construction_market_market_private_share_target = {
	if = {
		limit = {
			te_construction_market_total_use_target > 0
		}
		value = te_construction_market_private_use_target
		divide = te_construction_market_total_use_target
	}
	else = {
		value = 1
	}
	multiply = 100
	# Keep a token private allocation while the private queue is empty (the
	# cap in te_construction_market_private_use_target then zeroes the share)
	# so the autonomous investment AI is never left at a 0% allocation, a
	# state vanilla never reaches. It adds no throughput (sites are still
	# sized to public + private use); worst case, if the engine wastes rather
	# than re-routes unused private allocation, ~1% of the government-bought
	# construction goes unused. Drop this block to trade that for a 0% share.
	if = {
		limit = {
			investment_pool_gross_income > 0
		}
		min = 1
	}
}

te_construction_market_price_stabilizer_next_value = {
	if = {
		limit = {
			NOT = {
				has_variable = te_construction_market_price_stabilizer
			}
		}
		value = TE_CONSTRUCTION_MARKET_SERVICES_PRICE
	}
	else = {
		if = {
			limit = {
				var:te_construction_market_price_stabilizer < te_construction_market_price
			}
			value = var:te_construction_market_price_stabilizer
			add = {
				value = te_construction_market_price
				subtract = var:te_construction_market_price_stabilizer
				divide = 10
			}
		}
		else = {
			value = var:te_construction_market_price_stabilizer
			subtract = {
				value = var:te_construction_market_price_stabilizer
				subtract = te_construction_market_price
				divide = 10
			}
		}
	}
}
