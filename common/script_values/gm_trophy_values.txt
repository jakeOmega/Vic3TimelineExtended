# ---- What a taking carries off (spec §1), state scope ----------------------------
# The share of the monument's levels, by form: 1 for an object, 0.25 for a
# piece of a structure, 0.1 for a token (gm_trophy_triggers.txt).
gm_state_tr_share = {
	value = 0.1
	if = {
		limit = { gm_state_tr_form_is_piece = yes }
		value = 0.25
	}
	if = {
		limit = { gm_state_tr_form_is_object = yes }
		value = 1
	}
}

# The levels a taking carries off: the share, rounded up, at least one and at
# most every level.
gm_state_tr_levels = {
	value = gm_state_primary_levels
	multiply = gm_state_tr_share
	ceiling = yes
	min = 1
	max = gm_state_primary_levels
}

# The age a taking carries off with its levels (spec 2026-10-16 §5): all of
# it whole, else the levels' share of it.
gm_state_tr_age = {
	value = gm_state_primary_age
	if = {
		limit = { gm_state_tr_whole = 0 }
		multiply = gm_state_tr_levels
		divide = gm_state_primary_levels
	}
}

# The starting grandeur a taking carries off with its levels (sites spec
# 2026-10-18 §5): all of it whole, else the levels' share of it.
gm_state_tr_start = {
	value = gm_state_primary_start
	if = {
		limit = { gm_state_tr_whole = 0 }
		multiply = gm_state_tr_levels
		divide = gm_state_primary_levels
	}
}

# What they count for abroad: half their grandeur, starting grandeur and age,
# rounded up (spec §1).
gm_state_tr_value = {
	value = gm_state_tr_start
	add = gm_state_tr_age
	divide = 2
	ceiling = yes
}

# 1 when a taking carries off the whole monument.
gm_state_tr_whole = {
	value = 0
	if = {
		limit = { gm_state_tr_levels >= gm_state_primary_levels }
		value = 1
	}
}

# What is left standing after a piece is taken.
gm_state_tr_levels_left = {
	value = gm_state_primary_levels
	subtract = gm_state_tr_levels
	min = 0
}

# ---- Occupation (spec §2.1), state scope ------------------------------------------
# The engine registers the province iterators from a province, not a state
# (Supported Scopes: province): random_province in a state's effect logs "Wrong
# scope for effect: state, expected province" and does nothing (2026-10-16, once
# a month for every monument state), and any_province is registered the same
# way. A script value's every_province reads a state's provinces: the base game
# counts them so (occupation_values.txt, root = state). The self-check
# gm_occupation_read confirms it in game (gm_tr_occupation_read_self_check).

# How many provinces the state has.
gm_state_province_count = {
	value = 0
	every_province = {
		add = 1
	}
}

# The share of them scope:gm_tmp_tr_who controls, 0-1 (gm_state_tr_holds_half
# saves it).
gm_state_tr_who_province_share = {
	value = 0
	every_province = {
		limit = { controller = scope:gm_tmp_tr_who }
		add = 1
	}
	divide = { value = gm_state_province_count min = 1 }
}

# ---- Consequences (spec §3) --------------------------------------------------------
# Country scope, the taker, with the grandeur taken in var:gm_tr_tmp_grandeur
# (sites spec 2026-10-18 §3.6: its meaning, not its size): 2, plus 1 per 5
# grandeur, at most 8; doubled once UNESCO is in force.
gm_tr_infamy = {
	value = var:gm_tr_tmp_grandeur
	divide = 5
	add = 2
	max = 8
	if = {
		limit = { has_global_variable = un_agency_unesco }
		multiply = 2
	}
}

# tooltips.
gm_state_tr_infamy = {
	value = gm_state_tr_start
	add = gm_state_tr_age
	divide = 5
	add = 2
	max = 8
	if = {
		limit = { has_global_variable = un_agency_unesco }
		multiply = 2
	}
}

gm_tr_relations_taken = -30
gm_tr_relations_returned = 40
gm_tr_war_support = 10

# ---- The AI (spec §7) ----------------------------------------------------------------
# Country scope, the would-be looter, in percent a month: 2, plus 10 times the
# clout share of its governing groups that approve of Total War, Ethnostate or
# National Supremacy (law stances, so it follows the groups and not one
# hard-coded one), plus 5 for an unrecognized country.
gm_tr_ai_loot_base = {
	value = 2
	every_interest_group = {
		limit = { gm_tr_ig_looting_minded = yes }
		add = {
			value = ig_clout
			multiply = 10
		}
	}
	if = {
		limit = { is_country_type = unrecognized }
		add = 5
	}
}

# State scope, with the looter saved as scope:gm_tr_taker: plus 3 for a
# monument of 10 levels or more; a quarter once UNESCO is in force.
gm_tr_ai_loot_chance = {
	value = 0
	scope:gm_tr_taker = {
		add = gm_tr_ai_loot_base
	}
	if = {
		limit = { gm_state_grandeur >= 10 }
		add = 3
	}
	if = {
		limit = { has_global_variable = un_agency_unesco }
		multiply = 0.25
	}
}

# Container scope, a trophy on display, with the looter as scope:gm_tr_taker.
gm_tr_ai_loot_chance_trophy = {
	value = 0
	scope:gm_tr_taker = {
		add = gm_tr_ai_loot_base
	}
	if = {
		limit = { var:gm_tr_levels >= 10 }
		add = 3
	}
	if = {
		limit = { has_global_variable = un_agency_unesco }
		multiply = 0.25
	}
}

gm_tr_ai_return_chance = 2
