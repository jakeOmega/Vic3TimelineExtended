# Pop scope: literacy x size, SoL x size (the walk's sums).
te_demog_pop_literate = {
	value = literacy_rate
	multiply = total_size
}

te_demog_pop_sol = {
	value = standard_of_living
	multiply = total_size
}

# Building scope: levels in private, cooperative (self-owned) and state hands
# (bareword fractions read as values: docs/testing/demographics-probe-results-2026-10-08.md Q2).
te_demog_building_private_levels = {
	value = level
	multiply = private_ownership_fraction
}

te_demog_building_self_levels = {
	value = level
	multiply = self_ownership_fraction
}

te_demog_building_country_levels = {
	value = level
	multiply = country_ownership_fraction
}

# The census's year as a value (probe: add = year gives 1836). While fast mode's census clock
# runs (te_debug_demog.1, console only) it is the clock instead: a virtual year the global
# monthly pulse advances every te_demog_months_per_step months (te_demog_fast_effects.txt).
te_demog_year = {
	value = 0
	add = year
	if = {
		limit = { te_demog_clock_on = yes }
		value = global_var:te_demog_clock
	}
}

# The months one census step covers: a calendar year, or under the clock the months since its
# last step (te_demog_window_months, set as it advances). The walk's expected engine births and
# deaths are a month's curves times this.
te_demog_step_months = {
	value = 12
	if = {
		limit = { te_demog_clock_on = yes }
		value = global_var:te_demog_window_months
	}
}

# Fast mode's speed-up K: twelve months over the months a step (4, 6 or 12; 1 at normal
# speed and without the clock), and K - 1, the multiplier of the fast modifiers' +1 fields.
te_demog_fast_k = {
	value = 1
	if = {
		limit = { te_demog_clock_on = yes }
		value = 12
		divide = { value = global_var:te_demog_months_per_step min = 1 }
	}
}

te_demog_fast_k_minus_1 = {
	value = te_demog_fast_k
	subtract = 1
}

# State scope: the multipliers of te_demog_fast_births and te_demog_fast_deaths, the terms
# te_demog_fast_refresh_rates stored (state_birth_rate_mult and state_mortality_mult).
te_demog_fast_birth_term = {
	value = var:te_dg_fast_fb
}

te_demog_fast_death_term = {
	value = var:te_dg_fast_fd
}

# The open slot this year: year mod 150 (spec §1; demographics_model.step's open_k).
te_demog_open_slot = {
	value = te_demog_year
	modulo = 150
}

# State scope: desired fertility as a factor (spec §2.3; demographics_model.fertility):
# education x child survival x urban life. Reads this year's e0 (te_demog_life_table).
te_demog_desired = {
	value = 1
	multiply = te_demog_tfr_education
	multiply = te_demog_tfr_survival
	multiply = te_demog_tfr_urban
}

# State scope: desired fertility's three factors, each 1 at its reference and lower as
# the input grows. Named so the children-per-woman tooltips show them from the
# census's own formula: the capital's on the Population panel's Demographics tab
# (te_demog_ov_tfr_capital) and a state's on its Demographics subtab
# (te_demog_st_tfr_terms).
# Literacy:
te_demog_tfr_education = {
	value = var:te_dg_lit
	multiply = te_demog_k_education_weight
	multiply = -1
	add = 1
}

# Child survival, from life expectancy at birth between 30 and 80:
te_demog_tfr_survival = {
	value = var:te_dg_e0
	subtract = 30
	divide = 50
	min = 0
	max = 1
	multiply = te_demog_k_survival_weight
	multiply = -1
	add = 1
}

# Urban life, from the state's urban share:
te_demog_tfr_urban = {
	value = var:te_dg_urban_share
	multiply = te_demog_k_urban_weight
	multiply = -1
	add = 1
}

# State scope: children per woman = wealth x (1 - means x (1 - desired)) (spec §2.3;
# demographics_model.fertility).
te_demog_tfr = {
	value = 1
	subtract = var:te_dg_desired
	multiply = var:te_dg_means
	multiply = -1
	add = 1
	multiply = var:te_dg_wtfr
}

# War scope, on the owner's pulse: the owner's dead in this war so far (probe Q4).
te_demog_war_dead_root = {
	value = "num_country_dead(root)"
}

# State scope: how much of a migration is refugees (spec §2.5): devastation,
# turmoil and the owner being at war. A state's devastation reads as a 0-1 share,
# as turmoil does (vanilla compares devastation > 0.1, >= 0.5), though
# add_devastation takes points. The bareword reads are checked in game by
# te_debug_demog.1 option b's census line (Owner checks).
te_demog_crisis = {
	value = 0
	add = devastation
	add = turmoil
	if = {
		limit = { owner = { is_at_war = yes } }
		add = te_demog_k_crisis_at_war
	}
	min = 0
	max = 1
}

# State scope: the global city-size rank (1 = the biggest city, recomputed monthly into
# state_city_size_rank by city_rank_on_action) negated, so that ordered_scope_state, which takes
# the highest first, lists a country's biggest cities first (te_demog_country_census).
te_demog_neg_city_rank = {
	value = var:state_city_size_rank
	multiply = -1
}

# ============================================================================
# Wealth Concentration (spec 4.2; plan Task 11). Each state's score, te_dg_wc (0-100),
# drifts 3% a year toward its own target, a sum of terms around 50. The land term,
# te_demog_wc_land_term, is generated (te_demog_generated_values.txt). The proposals are
# the Decisions table's, starting points for the balance pass.
# ============================================================================

# State scope: the target. #822's law and amendments (0-100, centred on 50), land tenure
# by the state's agrarian share, who holds the state's capital, the state's income
# inequality, the country's taxes on wealth and its economic laws, clamped 0-100. Continuity of title
# (inheritance_events.6) pins it at the state's score: nobody's estate passes on.
te_demog_wc_target = {
	value = owner.te_inh_law_amendment_target
	add = te_demog_wc_land_term
	add = te_demog_wc_ownership_term
	add = te_demog_wc_inequality_term
	add = owner.te_demog_wc_tax_term
	add = owner.te_demog_wc_econ_term
	min = 0
	max = 100
	if = {
		limit = {
			owner = { has_variable = te_inh_title_continuity }
			has_variable = te_dg_wc
		}
		value = var:te_dg_wc
	}
}

# State scope: the walk's capital levels, in private, cooperative and state hands.
te_demog_wc_capital_levels = {
	value = var:te_dg_lv_priv
	add = var:te_dg_lv_self
	add = var:te_dg_lv_ctry
}

# State scope: the ownership term. The private share of the state's capital against
# cooperative and state ownership, x 40 around 0.65 and capped at 20 either way: all
# private +14, 15% private or less -20. Measured on the 1836 save (2026-10-09), large
# countries' capital is 55-65% private (self-owned farms, mines and workshops read
# cooperative), so the old centre of 0.9 held every country 10-17 points down for the
# whole game. Centred on 0.65 (owner's ruling) the term is near 0 in 1836 and falls as
# cooperative and state property grow. 0 for a state with under one level of capital.
te_demog_wc_ownership_term = {
	value = 0
	if = {
		limit = { te_demog_wc_capital_levels >= 1 }
		value = var:te_dg_lv_priv
		divide = { value = te_demog_wc_capital_levels min = 1 }
		subtract = 0.65
		multiply = 40
		min = -20
		max = 20
	}
}

# State scope: the inequality term. The rich save more, so fortunes grow from unequal
# incomes: +0.5 x (the state's Gini - 0.40) x 100, capped at 15 either way.
te_demog_wc_inequality_term = {
	value = var:te_dg_gini
	subtract = 0.4
	multiply = 50
	min = -15
	max = 15
}

# State scope: the people its last census counted, the States table's sort key and the
# figure each row shows (te_demog_country_lists).
te_demog_census_people = {
	value = 0
	if = {
		limit = { has_variable = te_dg_people }
		value = var:te_dg_people
	}
}

# Country scope: the law term, #822's law and amendments around 50 (the panel's bar and
# te_inh_concentration_target read it live; te_demog_wc_national stores the one it summed).
te_demog_wc_law_term = {
	value = te_inh_law_amendment_target
	subtract = 50
}

# Country scope: the tax term. Graduated Taxation -5; the legislated tax code's dividend
# rate x 20 when that rule is on (a 25% dividend tax -5); at most -10 in all.
te_demog_wc_tax_term = {
	value = 0
	if = {
		limit = { has_law = law_type:law_graduated_taxation }
		add = -5
	}
	if = {
		limit = { te_tax_code_full = yes }
		subtract = { value = te_tax_view_en_div_rate multiply = 20 }
	}
	min = -10
}

# Country scope: the economic laws term (owner, 2026-10-09). How freely capital compounds and
# combines: Laissez-Faire +10 and Extraction Economy +5 against Interventionism -5; Guilds and
# Chartered Monopolies and Freedom of Contract +5 each against Antitrust Enforcement
# (law_trust_busting) and Regulated Utilities -5. Traditionalism, Agrarianism, Industry Banned,
# Dirigisme and the command and cooperative laws are 0: every 1836 country holds a 0 or +5 law
# in each group, and state and cooperative property already count in the ownership term.
# National, like the tax term, so te_inh_concentration_target and the panel read it live. At
# most 15 either way.
te_demog_wc_econ_term = {
	value = 0
	if = {
		limit = { has_law = law_type:law_laissez_faire }
		add = 10
	}
	if = {
		limit = { has_law = law_type:law_extraction_economy }
		add = 5
	}
	if = {
		limit = { has_law = law_type:law_interventionism }
		add = -5
	}
	if = {
		limit = {
			OR = {
				has_law = law_type:law_guilds_chartered_monopolies
				has_law = law_type:law_freedom_of_contract
			}
		}
		add = 5
	}
	if = {
		limit = {
			OR = {
				has_law = law_type:law_trust_busting
				has_law = law_type:law_regulated_utilities
			}
		}
		add = -5
	}
	min = -15
	max = 15
}

# State scope: the score, and the score negated, so that ordered_scope_state (highest
# first) finds a country's most and least concentrated states (te_demog_wc_national).
te_demog_state_wc = {
	value = var:te_dg_wc
}

te_demog_neg_wc = {
	value = var:te_dg_wc
	multiply = -1
}
