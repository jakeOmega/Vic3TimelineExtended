# The census's year as a value (probe: add = year gives 1836). While fast mode's census clock
# runs (te_debug_demog.1, console only) it is the clock instead: a virtual year the global
# monthly pulse advances every te_demog_months_per_step months (te_demog_fast_effects.txt).
te_demog_year = {
	value = 0
	add = year
	if = {
		limit = { te_demog_clock_on = yes }
		value = global_var:te_demog_clock
	}
}

# The months one census step covers: a calendar year, or under the clock the months since its
# last step (te_demog_window_months, set as it advances). The walk's expected engine births and
# deaths are a month's curves times this.
te_demog_step_months = {
	value = 12
	if = {
		limit = { te_demog_clock_on = yes }
		value = global_var:te_demog_window_months
	}
}

# Fast mode's speed-up K: twelve months over the months a step (4, 6 or 12; 1 at normal
# speed and without the clock), and K - 1, the multiplier of the fast modifiers' +1 fields.
te_demog_fast_k = {
	value = 1
	if = {
		limit = { te_demog_clock_on = yes }
		value = 12
		divide = { value = global_var:te_demog_months_per_step min = 1 }
	}
}

te_demog_fast_k_minus_1 = {
	value = te_demog_fast_k
	subtract = 1
}

# State scope: the census's births and deaths terms in force (phase 2 step 4,
# te_demog_rate_effects.txt): M while its modifier is on, else 0. Read through has_modifier,
# so a state that lost its modifier (a change of owner, an old save) but still has te_dg_mb
# subtracts nothing that isn't there. Only for reads a tick or more after the refresh (the next
# refresh's prior term, the census line): inside the refresh's own effect has_modifier can't see
# its add or remove, so fast mode reads the refresh's locals instead.
te_demog_rate_births_applied = {
	value = 0
	if = {
		limit = {
			OR = {
				has_modifier = te_demog_census_births_up
				has_modifier = te_demog_census_births_down
			}
		}
		value = var:te_dg_mb
	}
}

te_demog_rate_deaths_applied = {
	value = 0
	if = {
		limit = {
			OR = {
				has_modifier = te_demog_census_deaths_up
				has_modifier = te_demog_census_deaths_down
			}
		}
		value = var:te_dg_md
	}
}

# State scope: the ownership term. The private share of the state's capital against
# cooperative and state ownership, x 40 around 0.65 and capped at 20 either way: all
# private +14, 15% private or less -20. Measured on the 1836 save (2026-10-09), large
# countries' capital is 55-65% private (self-owned farms, mines and workshops read
# cooperative), so the old centre of 0.9 held every country 10-17 points down for the
# whole game. Centred on 0.65 (owner's ruling) the term is near 0 in 1836 and falls as
# cooperative and state property grow. 0 for a state with under one level of capital.
te_demog_wc_ownership_term = {
	value = 0
	if = {
		limit = { te_demog_wc_capital_levels >= 1 }
		value = var:te_dg_lv_priv
		divide = { value = te_demog_wc_capital_levels min = 1 }
		subtract = 0.65
		multiply = 40
		min = -20
		max = 20
	}
}

# 2 x (the real policy rate - the economy's growth averaged over about five years), in percentage points, at most 5
# either way, and 0 until the country's banking cycle has started. Stored once a year with its inputs, at its yearly pulse
# before the national refresh (te_demog_wc_growth_update), so the panel's bar, the stored target and the states' targets
# all read the same figure until the next.
te_demog_wc_rg_term = {
	value = 0
	if = {
		limit = { has_variable = te_dg_wc_rg }
		value = var:te_dg_wc_rg
	}
}

# Country scope: 1 while the r > g term is in force (te_dg_wc_rg stored), else 0: the Return on Capital hovers give
# the rate and growth only then, and say the term adds nothing otherwise (the banking rule off, a cycle not started,
# or no yearly pulse since).
te_demog_wc_rg_on = {
	value = 0
	if = {
		limit = { has_variable = te_dg_wc_rg }
		value = 1
	}
}

# Country scope: the real policy rate, the policy rate less headline inflation, as the stance gap takes it
# (te_monetary_effects.txt step 8). The policy rate is nominal and the engine has no nominal prices, so GDP grows in
# real terms (monetary_policy_design.md §2, §12.1); a nominal rate against real growth would put most of a late-game,
# high-inflation world at +5. Under the simplified banking setting te_inflation stays 0 (te_mon_full_system).
te_demog_wc_real_rate = {
	value = var:te_policy_rate
	if = {
		limit = { has_variable = te_inflation }
		subtract = var:te_inflation
	}
}

# Country scope: the term from the stored inputs (te_dg_wc_r, te_dg_wc_g), for te_demog_wc_growth_update.
te_demog_wc_rg_now = {
	value = var:te_dg_wc_r
	subtract = var:te_dg_wc_g
	multiply = 2
	min = -5
	max = 5
}

# State scope: the state-owned share of the walk's capital levels, 0-1, the ownership shock's input
# (te_demog_wc_ownership_shock): the ownership term's walk and capital filter, so Manor Houses, Financial
# Districts, Subsistence Farms and government buildings are left out. 0 under one level of capital.
te_demog_wc_state_owned_share = {
	value = 0
	if = {
		limit = { te_demog_wc_capital_levels >= 1 }
		value = var:te_dg_lv_ctry
		divide = { value = te_demog_wc_capital_levels min = 1 }
	}
}

# Country scope: the land-tenure law's role, for the land-reform shock (te_demog_wc_land_reform_check): 1 Serfdom
# (Manorialism its variant), 2 Tenant Farmers (Latifundias and Expanded Latifundias), 3 Commercialized Agriculture,
# 4 Peasant Proprietorship (Homesteading), 5 Collectivized Agriculture, 0 none. has_law_or_variant gives a variant
# its parent's role (`parent =`).
te_demog_wc_land_code = {
	value = 0
	if = {
		limit = { has_law_or_variant = law_type:law_serfdom }
		value = 1
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_tenant_farmers }
		value = 2
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_commercialized_agriculture }
		value = 3
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_peasant_proprietorship }
		value = 4
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_collectivized_agriculture }
		value = 5
	}
}

# State scope: the score, and the score negated, so that ordered_scope_state (highest
# first) finds a country's most and least concentrated states (te_demog_wc_national).
te_demog_state_wc = {
	value = var:te_dg_wc
}

te_demog_neg_wc = {
	value = var:te_dg_wc
	multiply = -1
}
