# Campaign money (§ 8.3)

# What a firm spends: half a year's accrual, at least te_pa_cm_cost_floor pounds (a side's half spend, half that).
# What it buys is counted in units of a thousandth of its country's yearly investment-pool income, so it weighs the
# same in a small country as in a large one:
#   - momentum for a party in a campaign: te_pa_cm_momentum_per a unit, at most te_pa_cm_momentum_max a spend
#     (vanilla's momentum_medium, a medium election event, is 0.2);
#   - patronage for an interest group: the country's total for the group, in units, gives it political strength,
#     the full te_pa_patronage_cap (20%) at te_pa_pat_full_units; the total keeps te_pa_pat_keep of itself each
#     quarter, so it has faded about three years after the money stops.
# Its press makes it count te_pa_press_cm_extra more at strength 100 (§ 8.5).
te_pa_cm_cost_years = 0.5
te_pa_cm_cost_floor = 250
te_pa_cm_unit_share = 0.001
te_pa_cm_momentum_per = 0.05
te_pa_cm_momentum_max = 0.2
te_pa_pat_full_units = 10
te_pa_pat_keep = 0.8
te_pa_pat_floor = 0.05
te_pa_press_cm_extra = 0.25
# The Electoral Finance laws (§ 8.3's table): under Donation Limits a firm spends openly up to te_pa_cm_limit_share of
# its country's yearly investment-pool income a spend, and the rest in secret; under Publicly Funded Elections only in
# secret, at half effect. A firm goes on in secret only while its executive's Boldness (te_pa_bold_t, -1 to 1) is at
# least te_pa_cm_secret_bold.
te_pa_cm_limit_share = 0.01
te_pa_cm_secret_public_mult = 0.5
te_pa_cm_secret_bold = 0
# Secret money, exposed: the Standing and Connections it costs under each law (Task 4a's te_pa_expose_stand_loss is
# a holding's), and what it bought is taken back from the group's patronage. Secret money still at risk keeps
# te_pa_pat_keep of itself a quarter, like the patronage it bought, and is gone below te_pa_cm_secret_floor pounds.
te_pa_cm_expose_stand_limits = 10
te_pa_cm_expose_stand_public = 15
te_pa_cm_expose_conn_public = 10
te_pa_cm_secret_floor = 10
# The wager (§ 8.3): the side it backed in government after the election, or its patronized group entering
# government, +10 Connections; not, -5. A patronage wager on a group already in government is read
# te_pa_bet_months later.
te_pa_bet_win = 10
te_pa_bet_lose = 5
te_pa_bet_months = 36

# Company scope, inside te_pa_campaign_spend (te_pa_cm_want staged): the part it spends openly and the part in secret.
te_pa_cm_open_value = {
	value = var:te_pa_cm_want
	if = {
		limit = { owner = { te_pa_ef_limits = yes } }
		max = { value = 0 owner = { add = te_pa_cm_cap } }
	}
	if = { limit = { owner = { te_pa_ef_public = yes } } value = 0 }
	min = 0
}
te_pa_cm_hidden_value = {
	value = 0
	if = {
		limit = { te_pa_v_bold >= te_pa_cm_secret_bold }
		if = {
			limit = { owner = { te_pa_ef_limits = yes } }
			value = var:te_pa_cm_want
			subtract = { value = 0 owner = { add = te_pa_cm_cap } }
		}
		else_if = {
			limit = { owner = { te_pa_ef_public = yes } }
			value = var:te_pa_cm_want
		}
	}
	min = 0
}

# Interest group scope. How well the group stands on the firm's law at stake, for its side: its stance on the law
# (te_pa_gen_ig_stake_stance), or on Free Trade where the stake is the trade policy, the other way where the firm is
# against it (te_pa_stake_sign -1).
te_pa_ig_side_value = {
	value = 0
	if = {
		limit = { scope:te_pa_stake_firm ?= { var:te_pa_stake_trade ?= 1 } }
		add = te_pa_ig_free_trade_stance
	}
	else = {
		add = te_pa_gen_ig_stake_stance
	}
	if = {
		limit = { scope:te_pa_stake_firm ?= { var:te_pa_stake_sign ?= -1 } }
		multiply = -1
	}
}

# Interest group scope. Its stance on Free Trade, +2 to -2 (the trade policy is read from the market, not the law
# table, so its law has no number there).
te_pa_ig_free_trade_stance = {
	value = 0
	if = { limit = { law_stance = { law = law_type:law_free_trade value > approve } } value = 2 }
	else_if = { limit = { law_stance = { law = law_type:law_free_trade value > neutral } } value = 1 }
	else_if = { limit = { law_stance = { law = law_type:law_free_trade value < disapprove } } value = -2 }
	else_if = { limit = { law_stance = { law = law_type:law_free_trade value < neutral } } value = -1 }
}
