# ---- The doubling curve (§2.2) ---------------------------------------------
# Every effect a monument gives grows in steps of equal size, and each step
# costs twice the grandeur of the one before. With a first step of f, step n
# completes at f * (2^n - 1): for f = 5, at 5, 15, 35, 75, 155, 315, 635 and
# 1275. Inside a step the value grows linearly, so no level is wasted.
#
# Input: var:gm_curve_in on the current scope, which the caller sets first.
# Output: the fractional number of steps, 0-8. Each term is one step's share,
# clamped to 0-1 (min, then max: sequential clamps).
gm_curve_steps_f5 = {
	value = 0
	add = { value = var:gm_curve_in divide = 5 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 5 divide = 10 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 15 divide = 20 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 35 divide = 40 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 75 divide = 80 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 155 divide = 160 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 315 divide = 320 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 635 divide = 640 min = 0 max = 1 }
}

gm_curve_steps_f10 = {
	value = 0
	add = { value = var:gm_curve_in divide = 10 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 10 divide = 20 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 30 divide = 40 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 70 divide = 80 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 150 divide = 160 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 310 divide = 320 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 630 divide = 640 min = 0 max = 1 }
	add = { value = var:gm_curve_in subtract = 1270 divide = 1280 min = 0 max = 1 }
}

# The grandeur at which the step now being filled completes ("next step at"
# in the journal entry). Each later `if` overrides the earlier, so the result
# is the smallest step end above the input; the last end once the curve is
# complete.
gm_curve_next_f5 = {
	value = 1275
	if = { limit = { var:gm_curve_in < 635 } value = 635 }
	if = { limit = { var:gm_curve_in < 315 } value = 315 }
	if = { limit = { var:gm_curve_in < 155 } value = 155 }
	if = { limit = { var:gm_curve_in < 75 } value = 75 }
	if = { limit = { var:gm_curve_in < 35 } value = 35 }
	if = { limit = { var:gm_curve_in < 15 } value = 15 }
	if = { limit = { var:gm_curve_in < 5 } value = 5 }
}

# ---- The 1/n curve (owner, 2026-10-19) ---------------------------------------
# Local effects, prestige and cultural pull: step n takes n times the grandeur
# of the first, so each step is worth 1/n as much a point of grandeur as the
# first (1, 1/2, 1/3, ... where the doubling curve's are 1, 1/2, 1/4, ...) and
# the effect grows about as the square root of grandeur. With a first step of
# f, step n completes at f * n(n+1) / 2: for f = 5, at 5, 15, 30, 50, 75 ...
# and 2325 for the thirtieth. Grandeur being priced by world wealth (sites spec
# 2026-10-18), it no longer needs the doubling curve's steep taper there;
# approval and the other national effects keep it.
#
# The fixed form, for the local effects (state scope, gm_state_monthly): input
# var:gm_curve_in as above, output the fractional number of steps, 0-30. Thirty
# steps outlast a game: a monument ages by 2% of its starting grandeur a year,
# so a 50-level monument of the 1840s reaches about 1,700 by 2136. Prestige and cultural pull
# count every monument a country has and take gm_set_steps_lin_open instead,
# which has no end.
gm_curve_lin_steps = 30

# ---- Ageing (spec 2026-10-16 §2) ----------------------------------------------

# three times what it was dedicated with. Nothing is grander for being older

	multiply = gm_age_cap_years
	multiply = 2
	divide = 100

}

# State scope. What a month adds to the primary's age grandeur: 2% of its
# starting grandeur a year (owner, 2026-10-16: it doubles in fifty years;
# sites spec 2026-10-18 §3.2: of the starting grandeur, not compounding), as
# two whole numbers (no six-decimal literal, scripting_best_practices.md),
# until a century's worth is reached (gm_state_age_room).
gm_state_age_step = {
	value = gm_state_primary_start
	multiply = 2
	divide = 1200
	max = gm_state_age_room
}

# State scope. 1 while the primary is finished (gm_levels, ageing spec
# 2026-10-16 §3.1): the card's and the row's Finished line.
gm_state_disp_frozen = {
	value = 0
	if = {
		limit = {
			has_building = building_grand_monument
			has_variable = gm_levels
		}
		value = 1
	}
}

# State scope. What an anniversary's public option adds to the monument's age
# grandeur (ageing spec 2026-10-16 §5): a tenth of its starting grandeur, at
# least 1 (sites spec 2026-10-18 §3.6).
gm_evt_grandeur = {
	value = gm_state_primary_start
	multiply = 0.1
	min = 1
}

# Country scope, for the overview's policy line: months until the policy may
# change again (0 when it may).
gm_disp_policy_months_left = {
	value = 0
	if = {
		limit = {
			has_variable = gm_policy_cooldown
			has_variable = gm_policy_changed_month
		}
		value = 60
		add = var:gm_policy_changed_month
		subtract = te_history_month_index
		# The wait ends at the monthly tick after the 60th month: never 0
		# while the buttons are still greyed.
		min = 1
	}
}

# ---- National sums (§2.3, §2.4), country scope ---------------------------------
# Standing grandeur: every monument that is not contested (prestige, cultural pull),
# and the trophies this country holds (gm_trophy_values.txt).
gm_sum_standing = {
	value = 0
	every_scope_state = {
		limit = { gm_state_counts_standing = yes }
		add = gm_state_primary_grandeur
	}
	# Every secondary monument not contested stands whatever its state's
	# primary is (spec 2026-10-16 §4.2): a contested crown monument does not
	# take the marbles returned beside it.
	if = {
		limit = { has_variable_list = gm_monuments_held }
		every_in_list = {
			variable = gm_monuments_held
			limit = { gm_mon_counts_standing = yes }
			add = gm_mon_grandeur
		}
	}
	add = gm_tr_disp_held_grandeur
}

# Regime grandeur: regime and ruler monuments that fit (legitimacy), those
# standing beside another too (spec 2026-10-16 §4.2).
gm_sum_regime = {
	value = 0
	every_scope_state = {
		limit = {
			gm_state_status_fits = yes
			OR = {
				gm_state_kind_regime = yes
				gm_state_kind_ruler = yes
			}
		}
		add = gm_state_primary_grandeur
	}
	if = {
		limit = { has_variable_list = gm_monuments_held }
		every_in_list = {
			variable = gm_monuments_held
			limit = {
				gm_mon_status_fits = yes
				OR = {
					gm_mon_kind_regime = yes
					gm_mon_kind_ruler = yes
				}
			}
			add = gm_mon_grandeur
		}
	}
}

# The National Effects bars: how far the step now being filled has come, 0-1.
# On the doubling curve with first step f, the step that completes at N
# (gm_n_*, gm_curve_next_f*) began at (N - f) / 2, so at grandeur G (gm_g_*)
# it is (2G - N + f) / (N + f) full. Past the curve's last step the result
# exceeds 1 and the clamp holds the bar full. Each value's f must match its
# gm_set_steps call in gm_compute_totals (test_grand_monuments_layout.py).
# Prestige and cultural pull are on the 1/n curve (gm_set_steps_lin_open),
# which records the step's width W (gm_w_*): it began at N - W, so it is
# (G - N + W) / W full, and with no end the bar is never held full.
gm_disp_frac_prestige = {
	value = 0
	if = {
		limit = {
			has_variable = gm_g_standing
			has_variable = gm_n_standing
			has_variable = gm_w_standing
		}
		value = var:gm_g_standing
		subtract = var:gm_n_standing
		add = var:gm_w_standing
		divide = var:gm_w_standing
	}
	min = 0
	max = 1
}

# ---- World wealth (sites spec 2026-10-18 §3.2, Plan B) ------------------------------
# A level's starting grandeur is gm_level_grandeur_1836 times world GDP in 1836
# over world GDP when it is counted (gm_state_count_site_levels), so the same
# share of the world's wealth buys the same grandeur in any era. The 1836 figure
# is one constant for every save (spec §5): every country's latest GDP summed
# from a Timeline Extended save of 1836.1.20 came to 334,465,286
# (TE_SELF_CHECK gm_world_base compares a new game's first pass with it).
gm_world_gdp_1836 = {
	value = 334000000
}

# The grandeur a level adds in 1836 (owner, 2026-10-19: five, up from one): the
# world-wealth rate's scale, and what each level of an 1836 landmark, of a
# trophy abroad in 1836 and of a dug-up monument counts for. The records of
# monuments already standing keep the grandeur they were given.
gm_level_grandeur_1836 = {
	value = 5
}

# The world-wealth ratio now (no scope: world GDP, global_gdp_script_value), 1
# with none: the 1836 figure over today's.
gm_world_rate_now = {
	value = 1
	if = {
		limit = { global_gdp_script_value > 0 }
		value = gm_world_gdp_1836
		divide = global_gdp_script_value
		# The smallest a script number holds: gm_world_wealth_factor and the
		# opportunity's levels divide by it.
		min = 0.00001
	}
}

# The rate in force, the grandeur a level adds: the ratio written each January
# (gm_monument_monthly) and on any pass that finds none (gm_world_rate_ensure),
# 1 before the first write, times gm_level_grandeur_1836, to the nearest
# thousandth and at least 0.001 (owner, 2026-10-19). Scaled here, not where the
# ratio is written, so a save's ratio takes the scale without waiting for
# January.
gm_world_rate_value = {
	value = 1
	if = {
		limit = { has_global_variable = gm_world_rate }
		value = global_var:gm_world_rate
	}
	multiply = gm_level_grandeur_1836
	multiply = 1000
	round = yes
	divide = 1000
	min = 0.001
}

# How many times richer the world is than in 1836, for the hovers: the ratio
# in force, unscaled (gm_world_rate_now keeps it above 0).
gm_world_wealth_factor = {
	value = 1
	if = {
		limit = { has_global_variable = gm_world_rate }
		divide = global_var:gm_world_rate
	}
}
