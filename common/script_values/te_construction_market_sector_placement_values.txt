te_construction_market_buildings_under_construction = {
	value = 0
	every_scope_building = {
		limit = {
			is_under_construction = yes
		}
		add = 1
	}
}

# ---- Sizing the sites: how the week's purchase becomes each site's throughput ----
#
# Every site carries the same throughput: the week's purchase over the
# country's site levels. Staffing is not read (owner, 2026-11-13): a site is
# taken to be fully staffed. Its group asks for a hiring rate of 1.0 and its
# method makes it a very attractive job (building_job_attractiveness_mult on
# pm_te_construction_market_base), which was meant to fill a new site within
# the week. It does not (measured 2026-11-13, docs/testing/
# ai-economy-playtest.md, "First verdict"): a site hires one of its ten
# laborers a week, the base game's default rate, and most are removed before
# they fill, so a country whose construction moves between states gets a
# fraction of its purchase. A site delivers its throughput times its real
# staffing, so the sites together deliver the purchase when they are staffed
# and less while one is hiring; what is not delivered is not paid for. They never deliver more than the purchase, which
# is the failure to avoid: the government queue spends whatever the sites
# deliver, at the treasury's cost.
#
# Until 2026-11-13 each site had its own throughput, its share of the purchase
# divided by its staffing, with a site under 10% staffed sized at ten times
# its share. A site is placed unstaffed wherever construction starts, so a site
# sized before its first hiring (by the weekly pulse, or by the player's
# purchase buttons, which size the sites at once) was full days later and
# delivered ten times its share until the next pulse: the purchase bought ten
# times over, on the treasury (Workshop report: the cost of construction in the
# millions on a purchase of 50,000, the throughput 5,000%, 2,000%, then 15,000%
# in three weeks). A country whose one site had just moved to a new state got
# the full ten times.

# Country scope: the levels of the country's sites, summed. The market places
# its sites at level 1, but a site kept from a direct game can have more, and
# each level has its own ten laborers and delivers the throughput again.
te_construction_market_site_levels = {
	value = 0
	every_scope_building = {
		limit = {
			is_building_type = te_construction_market_site
		}
		add = level
	}
}

# Country scope: the multiplier of te_construction_market_capacity_modifier
# (building_throughput_add = 1) on every site. The purchase over the site
# levels, less the engine's own 1; 0 for a country with no site.
te_construction_market_site_throughput = {
	value = 0
	if = {
		limit = {
			te_construction_market_site_levels > 0
		}
		value = te_construction_market_total_use_target
		divide = te_construction_market_site_levels
		subtract = 1
	}
}
