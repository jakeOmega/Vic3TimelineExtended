# Local Climate: comfort (docs/superpowers/specs/2026-11-01-local-climate-design.md § 5, § 6, § 11, § 12).
# State scope throughout. Each state region's climate record (local_climate_generated.txt), the
# five technology-carried reliefs (local_climate_modifier_types.txt, te_local_climate_tech_injections.txt)
# and the world's felt warming become five terms, kept separate for the state panel, and their sum,
# te_climate_penalty: the percent of its migration attraction a state loses (local_climate_discomfort,
# state_migration_pull_mult). Four terms only cost; winter's mildness is signed, so a mild-wintered
# state with little else against it gains (te_climate_bonus, local_climate_amenity; § 12). A region
# with no row reads 0. The constants are the
# calibration's (§ 11.4); scripts/analysis/local_climate_calibration.py mirrors this file and prints
# its anchor table, and test_local_climate_comfort.py pins the anchors' targets. Change a constant
# here and rerun both.
#
#   summer warming = world's anomaly - the anomaly the record's normals hold at
#   winter warming = summer warming x amplification(latitude)
#   x       = max(0, heat index + summer warming x index per degree - H0)
#   heat    = (a x + b x^2) x (1 - heat relief)
#   winter' = winter + winter warming
#   cold    = max(0, C0 - winter') x cold weight x (1 - cold relief)
#   winter  = (max(0, N - winter') - max(0, min(winter', M) - N) x summer share) x per degree
#             x (1 - winter relief)                                    negative: a bonus
#   summer share = max(0, 1 - raw heat / K): a hot summer cancels a mild winter's draw
#   dry     = W x aridity x (1 - water relief)
#   disease = D x malaria x (1 - disease relief)
#   penalty = min(P_max, heat + cold + mildness + dry + disease)

# Tuning constants (§ 11.4), in percent of migration attraction.
te_climate_heat_start = 26 # H0: the heat index (0.7 x wet-bulb + 0.3 x air, on a typical year's worst day, °C) above which heat costs
te_climate_heat_linear = 4.1 # a: percent per degree of heat index at H0
te_climate_heat_quadratic = 0.36 # b: the curve, so the 8th degree costs a little over twice the 1st
te_climate_index_per_degree = 0.8 # the heat index's rise per degree of warming, with relative humidity about steady
te_climate_cold_start = 0 # C0: the coldest month's mean, °C, below which frost costs
te_climate_cold_weight = 0.94 # per degree of frost, on top of the winter's mildness below
te_climate_winter_neutral = 3 # N: a coldest-month mean like London's in 1836, which neither costs nor draws
te_climate_winter_mild = 15 # M: winters milder than this draw no more
te_climate_winter_per_degree = 0.6 # per degree of coldest-month mean from N, so the bonus is at most 7.2
te_climate_winter_heat_cancel = 20 # K: the heat cost before relief, percent, at which a hot summer cancels a mild winter's draw
te_climate_dry_per_class = 16 # W: per aridity class (1 limited water, 2 scarce)
te_climate_malaria_per_class = 5 # D: per malaria class (1 seasonal or local, 2 endemic)
te_climate_penalty_max = 80 # P_max: climate alone never takes all of a state's pull; a boom's jobs can still draw people
# Winter warming's amplification by latitude (§ 6, § 11.3): 1 up to the low latitude, rising in a straight
# line to the maximum at the Arctic (or Antarctic) Circle and beyond, as observed warming runs. Arctic
# amplification is a cold-season effect, so the hot days take the world's warming as it is.
te_climate_amp_low_lat = 30
te_climate_amp_high_lat = 66.5
te_climate_amp_max = 2.5
# The record holds today's climate normals (about 1991-2020), so a region reads it exactly when the
# world's anomaly is today's; 1836, at anomaly 0, is this much colder, times the amplification
# (owner, 2026-11-01).
te_climate_normals_warming = 1.1

# |latitude| of the region's reference place.
te_climate_abs_latitude = {
	value = te_climate_latitude
	if = {
		limit = { te_climate_latitude < 0 }
		multiply = -1
	}
}

# 1 at |latitude| <= te_climate_amp_low_lat, te_climate_amp_max at >= te_climate_amp_high_lat.
te_climate_amplification = {
	value = te_climate_abs_latitude
	subtract = te_climate_amp_low_lat
	min = 0
	divide = {
		value = te_climate_amp_high_lat
		subtract = te_climate_amp_low_lat
	}
	max = 1
	multiply = {
		value = te_climate_amp_max
		subtract = 1
	}
	add = 1
}

# The warming this region's hot days feel against the record's normals, °C: the felt anomaly less the
# anomaly the normals hold at. The felt anomaly, because an aerosol haze cools the air people live in
# (§ 6). Negative until the world reaches today's warming, and below that again if carbon removal takes
# the world past it. It reads the global anomaly, which works at state scope.
te_climate_summer_warming = {
	value = temperature_anomaly_display
	subtract = te_climate_normals_warming
}
# The same for its winters, times the latitude's amplification.
te_climate_winter_warming = {
	value = temperature_anomaly_display
	subtract = te_climate_normals_warming
	multiply = te_climate_amplification
}

# The record's hottest day and coldest month with warming added: warming raises the heat term and eases the
# cold one. The tile shows both.
te_climate_hot_day_eff = {
	value = te_climate_hot_day
	add = te_climate_summer_warming
}
te_climate_winter_eff = {
	value = te_climate_winter_mean
	add = te_climate_winter_warming
}

# Relief, 0 to 1 (§ 5.2): the sum of the carriers the owner holds, clamped, since carriers or a later
# law could pass 1.
te_climate_relief_heat = {
	value = modifier:state_climate_heat_relief_add
	min = 0
	max = 1
}
te_climate_relief_cold = {
	value = modifier:state_climate_cold_relief_add
	min = 0
	max = 1
}
# Winter's mildness shrinks with heating, insulation and warm clothing more slowly than frost does,
# and only the era 11+ technologies close it (owner, 2026-11-01; § 12).
te_climate_relief_winter = {
	value = modifier:state_climate_winter_relief_add
	min = 0
	max = 1
}
te_climate_relief_water = {
	value = modifier:state_climate_water_relief_add
	min = 0
	max = 1
}
# Malaria Prevention's share is the census's own Malaria Control (te_demog_tech_injections.txt),
# which eases a newcomer's fever risk as it eases the census's.
te_climate_relief_disease = {
	value = modifier:state_climate_disease_relief_add
	add = modifier:state_malaria_control_add
	min = 0
	max = 1
}

# Heat stress on a typical year's worst day with warming added. Air temperature rises with warming and
# relative humidity holds about steady, so the wet-bulb part rises a little slower than the air.
te_climate_heat_index_eff = {
	value = te_climate_heat_index
	add = {
		value = te_climate_summer_warming
		multiply = te_climate_index_per_degree
	}
}
# Degrees of heat index above the point where heat starts to cost.
te_climate_heat_excess = {
	value = te_climate_heat_index_eff
	subtract = te_climate_heat_start
	min = 0
}

# Each term before relief, in percent. Heat is (b x + a) x x = a x + b x^2.
te_climate_heat_raw = {
	value = te_climate_heat_excess
	multiply = te_climate_heat_quadratic
	add = te_climate_heat_linear
	multiply = te_climate_heat_excess
}
te_climate_cold_raw = {
	value = te_climate_cold_start
	subtract = te_climate_winter_eff
	min = 0
	multiply = te_climate_cold_weight
}
# A winter harsher than N turns settlers away and a milder one draws them, per degree; past M a
# milder winter draws no more. Negative is a bonus. The frost term adds below 0 °C.
te_climate_winter_harsh = {
	value = te_climate_winter_neutral
	subtract = te_climate_winter_eff
	min = 0
	multiply = te_climate_winter_per_degree
}
# A mild winter's draw shrinks with the summer's heat cost before relief and is gone at K: people who
# prize mild winters avoid hot, humid summers too, so Khartoum's or Singapore's warm winter draws
# nothing once air conditioning eases their heat (owner, 2026-11-01). Before relief, since a cooled
# house doesn't make the summer outside milder.
te_climate_summer_share = {
	value = te_climate_heat_raw
	divide = te_climate_winter_heat_cancel
	multiply = -1
	add = 1
	min = 0
}
te_climate_winter_mildness = {
	value = te_climate_winter_eff
	max = te_climate_winter_mild
	subtract = te_climate_winter_neutral
	min = 0
	multiply = te_climate_winter_per_degree
	multiply = te_climate_summer_share
}
te_climate_winter_raw = {
	value = te_climate_winter_harsh
	subtract = te_climate_winter_mildness
}
te_climate_dry_raw = {
	value = te_climate_aridity
	multiply = te_climate_dry_per_class
}
te_climate_disease_raw = {
	value = te_climate_malaria
	multiply = te_climate_malaria_per_class
}

# Each term after relief: what the state panel shows, term by term.
te_climate_heat_term = {
	value = te_climate_heat_raw
	multiply = {
		value = 1
		subtract = te_climate_relief_heat
	}
}
te_climate_cold_term = {
	value = te_climate_cold_raw
	multiply = {
		value = 1
		subtract = te_climate_relief_cold
	}
}
te_climate_winter_term = {
	value = te_climate_winter_raw
	multiply = {
		value = 1
		subtract = te_climate_relief_winter
	}
}
te_climate_dry_term = {
	value = te_climate_dry_raw
	multiply = {
		value = 1
		subtract = te_climate_relief_water
	}
}
te_climate_disease_term = {
	value = te_climate_disease_raw
	multiply = {
		value = 1
		subtract = te_climate_relief_disease
	}
}

# The comfort penalty, at most te_climate_penalty_max. Negative only through a mild winter, so never
# below minus te_climate_winter_per_degree x (te_climate_winter_mild - te_climate_winter_neutral). A
# region with no row reads 0 whatever the warming.
te_climate_penalty = {
	value = 0
	if = {
		limit = { te_climate_has_record > 0 }
		add = te_climate_heat_term
		add = te_climate_cold_term
		add = te_climate_winter_term
		add = te_climate_dry_term
		add = te_climate_disease_term
	}
	max = te_climate_penalty_max
}
# The bonus local_climate_amenity applies: the penalty's negative part, as a positive percent.
te_climate_bonus = {
	value = te_climate_penalty
	multiply = -1
	min = 0
}

# The state panel's Comfort row (Task 7). It shows what local_climate_discomfort applies, the penalty
# te_update_local_climate_modifier stored at the state's last yearly refresh, not the live value, which
# can drift between refreshes. 0 before the first application.
te_climate_penalty_now = {
	value = 0
	if = {
		limit = { has_variable = te_climate_penalty_applied }
		value = var:te_climate_penalty_applied
	}
}
# The bonus local_climate_amenity applies, from the same refresh; 0 on a save from before § 12 until then.
te_climate_bonus_now = {
	value = 0
	if = {
		limit = { has_variable = te_climate_bonus_applied }
		value = var:te_climate_bonus_applied
	}
}
# What the climate does to the state's pull, signed for the Comfort hover: negative a cost.
te_climate_effect_now = {
	value = te_climate_penalty_now
	multiply = -1
}
# Winter's mildness as what it draws, for the hovers: positive a bonus, negative a cost.
te_climate_winter_draw = {
	value = te_climate_winter_term
	multiply = -1
}
te_climate_winter_draw_raw = {
	value = te_climate_winter_raw
	multiply = -1
}
# 1 once the state's first monthly application has run; until then the row says Pending.
te_climate_penalty_shown = {
	value = 0
	if = {
		limit = { has_variable = te_climate_penalty_applied }
		value = 1
	}
}
