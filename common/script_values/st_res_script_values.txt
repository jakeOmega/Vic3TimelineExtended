# ============================================================================
# Strategic Reserve — Script Values
# ============================================================================
# Weekly signed-flow bookkeeping. Positive configured rates store goods,
# negative configured rates withdraw them. The hub applies the chosen base rate
# up to its own throughput cap (10 goods/level/week), and generic building
# throughput modifiers scale the active market flow from there.
# Scope contract: these values are evaluated in country scope. Live hub data is
# bridged in from building scope by st_res_refresh_hub_cache_effect, which writes
# st_res_hub_level_cached and st_res_hub_throughput_cached before these values are
# consumed by the JE, buttons, or hub-flow refresh helpers.
# ============================================================================

# Hub level — 0 if no hub exists.
st_res_hub_level = {
	value = 0
	add = var:st_res_hub_level_cached
}

st_res_weekly_base_rate_cap = {
	value = 0
	add = modifier:country_st_res_weekly_rate_cap_add
	min = 1
}

st_res_neg_weekly_base_rate_cap = {
	value = st_res_weekly_base_rate_cap
	multiply = -1
}

# Total throughput_add modifier on the hub, summing all contributions.
st_res_hub_throughput = {
	value = 0
	add = var:st_res_hub_throughput_cached
}

# Throughput scaling factor — clamped at 0.1x so a negative pile-on of
# debuffs can't flip the delta sign or zero out flow entirely.
st_res_throughput_factor = {
	value = 1
	add = st_res_hub_throughput
	min = 0.1
	max = 5
}

# Shared button step size. Tier 0 = 1, 1 = 10, 2 = 100, 3 = 1000, 4 = 10000.
# Tier var defaults to 1 (10-unit step) before the JE first runs init.
st_res_adjust_step_value = {
	value = 10
	if = {
		limit = {
			has_variable = st_res_adjust_step_tier
			var:st_res_adjust_step_tier = 0
		}
		value = 1
	}
	else_if = {
		limit = {
			has_variable = st_res_adjust_step_tier
			var:st_res_adjust_step_tier = 1
		}
		value = 10
	}
	else_if = {
		limit = {
			has_variable = st_res_adjust_step_tier
			var:st_res_adjust_step_tier = 2
		}
		value = 100
	}
	else_if = {
		limit = {
			has_variable = st_res_adjust_step_tier
			var:st_res_adjust_step_tier = 3
		}
		value = 1000
	}
	else_if = {
		limit = {
			has_variable = st_res_adjust_step_tier
			var:st_res_adjust_step_tier = 4
		}
		value = 10000
	}
}

# --- GRAIN ---
st_res_grain_decay_rate = {
	value = 0
	add = modifier:country_st_res_grain_decay_add
	divide = 52 # convert from per-year to per-week decay
	min = 0
	max = 1
}

st_res_grain_weekly_decay = {
	value = 0
	if = {
		limit = { has_variable = st_res_grain_stored }
		value = var:st_res_grain_stored
		multiply = st_res_grain_decay_rate
	}
}

# Actual gross flow the hub must apply to achieve the configured target net change
# (includes decay replenishment). Positive = hub buying, negative = hub selling.
# Decay-add must stay inside the workforce gate: an undermanned hub cannot
# auto-replenish decay loss. Pulling decay-add outside makes weekly_delta
# read 0 (= decay - decay) when the hub is actually doing nothing, hiding
# real stockpile loss from the player.
st_res_grain_actual_rate = {
	value = 0
	if = {
		limit = {
			has_variable = st_res_hub_workforce_cached
			var:st_res_hub_workforce_cached > 0.99
			has_variable = st_res_grain_rate
		}
		add = var:st_res_grain_rate
		add = st_res_grain_weekly_decay
	}
}

st_res_grain_actual_rate_base_applied = {
	value = st_res_grain_actual_rate
	min = {
		value = 0
		subtract = st_res_weekly_base_rate_cap
	}
	max = st_res_weekly_base_rate_cap
}

st_res_grain_weekly_delta = {
	value = 0
	add = st_res_grain_actual_rate_base_applied
	subtract = st_res_grain_weekly_decay
}

st_res_grain_capacity = {
	value = 0
	add = modifier:country_st_res_grain_capacity_add
}

st_res_grain_good_mult = {
	value = 0
	if = {
		limit = {
			has_variable = st_res_grain_stored
			st_res_grain_actual_rate_base_applied > 0
		}
		add = st_res_grain_actual_rate_base_applied
	}
	else_if = {
		limit = { has_variable = st_res_grain_stored }
		subtract = st_res_grain_actual_rate_base_applied
	}
	# Compensate for throughput modifiers, so total rate equals desired rate
	divide = st_res_throughput_factor
	# Subtract 1 to compensate for the PM's base 1-unit goods input/output:
	# applied as `goods_input_<good>_mult = good_mult`, the engine computes
	# final = base × (1 + good_mult) × throughput_factor, so good_mult must be
	# (R / throughput) - 1 for final to equal R. Without this, the hub overbuys
	# by `throughput_factor` units/week — most visible at maintenance buying
	# (rate=0+decay), where it'd otherwise consume `decay + throughput` instead
	# of just `decay`.
	subtract = 1
}

# Bounds for the rate clamp in st_res_clamp_stockpiles_effect — pure storage room.
# Decay is already netted out in weekly_delta; subtracting it again here would pull
# the rate variable below zero at full capacity and silently flip Status to "Withdrawing".
st_res_grain_max_withdrawable = {
	value = 0
	add = var:st_res_grain_stored
	multiply = -1
}

st_res_grain_max_storable = {
	value = 0
	add = st_res_grain_capacity
	subtract = var:st_res_grain_stored
}

st_res_grain_sale_profit = {
	value = 0
	if = {
		limit = {
			has_variable = st_res_grain_stored
			st_res_grain_actual_rate_base_applied < 0
		}
		subtract = st_res_grain_actual_rate_base_applied
		g:grain = {
			multiply = base_price
		}
		multiply = {
			value = 1
			add = this.market.mg:grain.market_goods_pricier
		}
	}
}

# ============================================================================
# Reserve policies — shared tuning constants
# ============================================================================
# EVERY tunable number for the automated reserve policies is defined once, here
# or in the three `st_res_apply_preset_*_base` effects in
# common/scripted_effects/st_res_effects.txt. Nothing below is duplicated in a
# trigger, a scripted GUI or a .gui expression: the sgui `is_valid` stepper
# gates and the weekly evaluator both read these values.
#
# Thresholds are expressed in PERCENTAGE POINTS against the good's base price,
# measured on the country's own market (`this.market.mg:<good>`), because that
# is the market the reserve hub's purchases and sales clear on — see
# st_res_<good>_price_rel below.

# Stepper increments, one per settings row in the policy panel.
st_res_policy_thr_step = { value = 5 }      # purchase / release threshold, pp
st_res_policy_pct_step = { value = 5 }      # protected floor / target ceiling, pp of capacity
st_res_policy_flow_step = { value = 50 }    # maximum weekly flow, units
st_res_policy_budget_step = { value = 1000 } # weekly purchase budget, GBP

# Hysteresis. A lane that is already engaged keeps going until the price
# crosses the *stop* threshold, which sits `band` points on the far side of the
# start threshold. Defaults: start buying below -10%, stop above -5%; start
# releasing above +20%, stop below +10%.
st_res_policy_buy_band = { value = 5 }
st_res_policy_sell_band = { value = 10 }

# The purchase and release bands may never overlap: with the stop thresholds at
# buy_thr + buy_band and sell_thr - sell_band, the two engaged regions stay
# disjoint exactly when sell_thr - buy_thr >= buy_band + sell_band.
st_res_policy_min_gap = {
	value = st_res_policy_buy_band
	add = st_res_policy_sell_band
}

# Absolute stepper bounds.
st_res_policy_thr_floor = { value = -50 }     # lowest purchase threshold, pp
st_res_policy_thr_ceiling = { value = 75 }    # highest release threshold, pp
st_res_policy_flow_min = { value = 50 }
st_res_policy_flow_max = { value = 2000 }
st_res_policy_budget_min = { value = 1000 }
st_res_policy_budget_max = { value = 100000 }
st_res_policy_stock_min = { value = 0 }       # lowest protected floor, % of capacity
st_res_policy_stock_max = { value = 100 }     # highest target ceiling, % of capacity
st_res_policy_stock_min_gap = { value = 10 }  # minimum ceiling - floor, pp

# ============================================================================
# Reserve policies — per-good price signal and validation helpers
# ============================================================================
# PRICE SIGNAL. `market_goods_pricier` / `market_goods_cheaper` are readable as
# script values inside `market_goods` scope (vanilla precedent:
# common/treaty_articles/13_goods_transfer.txt reads `value = market_goods_cheaper`).
# The `market = { mg:<good> = { add = ... } }` block form below is the shape
# vanilla itself uses to read a market_goods value from country scope
# (common/script_values/00_gfx_route_graphics_values.txt, gfx_infantry_mobilization_count).
# Do NOT rewrite it as a `this.market.mg:<good>.<value>` dot chain: nothing in
# vanilla reads through `mg:` that way, and a price read that silently returns
# zero would park every automated lane in "Waiting" forever with no error.
# Each is documented as "at least X percent pricier/cheaper than base price".
# The engine may return them clamped at zero or as signed mirrors of each other,
# and nothing in the docs settles which — so `_price_rel` is built as
# max(pricier, 0) - max(cheaper, 0), which yields the same signed premium under
# either behaviour. `min = 0` after the read is the max(x, 0) clamp.
#
# The market read is `this.market`, i.e. the country's OWN market. That is the
# market the Strategic Reserve Hub's goods input/output clears on, and it is the
# same market `st_res_<good>_sale_profit` already prices sales at. Player-facing
# localization calls it the "national market price" for exactly that reason.

# --- GRAIN ---
st_res_grain_price_up = {
	value = 0
	market = {
		mg:grain = {
			add = market_goods_pricier
		}
	}
	min = 0
}

st_res_grain_price_down = {
	value = 0
	market = {
		mg:grain = {
			add = market_goods_cheaper
		}
	}
	min = 0
}

# Signed premium against base price, as a fraction (+0.20 = 20% above base).
st_res_grain_price_rel = {
	value = st_res_grain_price_up
	subtract = st_res_grain_price_down
}

# Current market price of one unit, in GBP. Mirrors the shape of
# st_res_grain_sale_profit. `min = 1` keeps the budget division safe.
st_res_grain_unit_price = {
	value = 1
	g:grain = {
		multiply = base_price
	}
	multiply = {
		value = 1
		add = st_res_grain_price_rel
	}
	min = 1
}

# Stepper guard rails. A Paradox trigger needs a `var:` (or another real

# limit each variable may not step past, rather than as a gap compared to a
