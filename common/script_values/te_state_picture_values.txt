# Oil country: Oil Rigs hold at least this share of the state's building levels (subsistence

# most buildings, so 5% of levels stands for about a tenth of the state's GDP (owner, 2026-11-10).
te_state_picture_k_oil_share = { value = 0.05 }

# Growth: the state's people against a year before, as a rate, held within a tenth either way
# (a conquest or a migration wave is not a trend) and smoothed: each year's rate counts for
# te_state_picture_k_growth_new of the stored figure, so after n years of a steady rate r the
# stored figure is r x (1 - 0.8^n). It nears r from below and never reaches it: steady growth of
# exactly 5% a year never makes a boomtown. 6% a year makes one in its ninth year (4.99% after
# eight), 7% in its sixth, 8% in its fifth and the 10% cap in its fourth; a fall of 2% a year makes
# a declining city in its fourth. A boomtown is a city whose smoothed rate is at least
# te_state_picture_k_boom a year (the owner, 2026-11-14: "~5%/year"; it was 3%), a declining city
# one at te_state_picture_k_decline or below. In the owner's Singapore game (saves of 1947, 1949
# and 1957) 30 to 39 of 336 to 403 cities stood at a smoothed 3% or more, 18 to 24 at 4% and 14 or
# 15 at 5%. Declining is provisional: in the 1940 save 73 of 346 cities fell 1% in the one year.
te_state_picture_k_growth_new = { value = 0.2 }
te_state_picture_k_growth_cap = { value = 0.1 }
te_state_picture_k_boom = { value = 0.05 }
te_state_picture_k_boom_keep = { value = 0.045 }
te_state_picture_k_decline = { value = -0.01 }
te_state_picture_k_decline_keep = { value = -0.005 }
# A divided city: a metropolis with a Gini of at least this. Provisional: in the 1940 save the
# metropolises' median is 0.78 and a quarter are at 0.85 or more.
te_state_picture_k_divided = { value = 0.85 }
te_state_picture_k_divided_keep = { value = 0.83 }

# The rung of the city ladder the owner is on, 0 before its first city technology, when a state has
# no means and no upkeep tag.
te_state_picture_city_rung = {
	value = 0
	owner = {
		if = { limit = { has_technology_researched = art_deco_architecture } add = 1 }
		if = { limit = { has_technology_researched = modern_urban_planning } add = 1 }
		if = { limit = { has_technology_researched = computer_aided_design } add = 1 }
		if = { limit = { has_technology_researched = advanced_structural_engineering } add = 1 }
		if = { limit = { has_technology_researched = vactrains } add = 1 }
	}
}

# Means: the state's standard of living (average_sol) against the world's (the owner, 2026-11-14:
# "the world average +/- a percentage"; until then fixed floors by city rung, which in the owner's
# Singapore game tagged 88% of states poor, its AI owners being on rungs 3 and 4 by the 1940s). The
# world's figure is the plain average of every state's average_sol, each state counted once, measured
# once a calendar year (te_state_picture_world_measure); each state keeps the figure it was judged
# against (te_sp_world), so its hover gives the floors its picture stands on. Poor while under
# te_state_picture_k_poor_ratio of it, affluent from te_state_picture_k_affluent_ratio of it (the
# owner's pick, 2026-11-14, aiming at roughly 5% to 15% of the world's states on each side and 80% to
# 90% on neither). The average is not weighted by people: one rich, populous country shouldn't move
# every state's floors. In the 1957 save of the owner's Singapore game, Singapore's nine states
# (10.3% of the world's people, at a standard of living of 81) lifted a people-weighted figure from
# 16.5 to 23.1. Against the plain average, 0.5 tags 4% to 6% of states poor and 1.6 tags 4% to 8%
# affluent in saves of 1932 to 1978; in two later saves of other games on older builds (2020, 2054)
# 0.5 tags 20% to 29% poor and 1.6 tags 14% to 20% affluent (the table is in pull request #1799).
# Each ratio is one line.
te_state_picture_k_affluent_ratio = { value = 1.6 }
te_state_picture_k_poor_ratio = { value = 0.5 }
# The world's average the state was judged against, 0 before its first judgement.
te_state_picture_hover_world = {
	value = 0
	if = { limit = { has_variable = te_sp_world } value = var:te_sp_world }
}
te_state_picture_poor_floor = {
	value = te_state_picture_hover_world
	multiply = te_state_picture_k_poor_ratio
}
te_state_picture_affluent_floor = {
	value = te_state_picture_hover_world
	multiply = te_state_picture_k_affluent_ratio
}
# A state leaves poor, or affluent, only this far past the floor, so one at a floor doesn't change
# picture each year: half a point of standard of living at any floor, the owner's figure for the
# fixed floors. The floors now move with the world's figure too, which rose 0.3 to 0.8 points a
# year between saves of one game; the keep is not meant to absorb that: a state the world
# outgrows should change picture.
te_state_picture_k_means_keep = { value = 0.5 }
te_state_picture_poor_floor_keep = {
	value = te_state_picture_poor_floor
	add = te_state_picture_k_means_keep
}
te_state_picture_affluent_floor_keep = {
	value = te_state_picture_affluent_floor
	subtract = te_state_picture_k_means_keep
}
te_state_picture_sol = { value = average_sol }

te_state_picture_world_sol = {

# Upkeep: how far the standard of living sits under what the state's people expect
# (average_expected_sol, which the mod's Standard of Living Expectations system moves). Run-down
# from te_state_picture_k_rundown under, until within te_state_picture_k_rundown_keep.
# Provisional and this design's: the calibration game is to set them.
te_state_picture_k_rundown = { value = 2 }
te_state_picture_k_rundown_keep = { value = 1.5 }
te_state_picture_sol_shortfall = {
	value = average_expected_sol
	subtract = average_sol
}

# This year's growth rate (state scope; only where te_state_picture_growth_known holds) ...
te_state_picture_growth_rate = {
	value = var:te_dg_walk_pop
	divide = var:te_dg_walk_pop_prev
	subtract = 1
	min = { value = te_state_picture_k_growth_cap multiply = -1 }
	max = te_state_picture_k_growth_cap
}
# ... and the stored figure moved toward it.
te_state_picture_growth_smoothed = {
	value = te_state_picture_growth_rate
	subtract = var:te_sp_growth
	multiply = te_state_picture_k_growth_new
	add = var:te_sp_growth
}

# The largest city's share of the state's urban people where the census keeps none (Demographics
# disabled, or under 5,000 urban people): te_demog_k_mt_base as a fraction.
te_state_picture_k_share = { value = 0.4 }

# The people of the state's largest city: the walk's urban people (te_dg_urban, written every year
# whatever the Demographics rule) times the metropolitan share (te_dg_mt, 0-100, only while the
# census runs). -1 before the state's first walk.
te_state_picture_city_people = {
	value = -1
	if = {
		limit = { has_variable = te_dg_urban }
		value = var:te_dg_urban
		if = {
			limit = { has_variable = te_dg_mt }
			multiply = var:te_dg_mt
			divide = 100
		}
		else = {
			multiply = te_state_picture_k_share
		}
	}
}

# The size a city of that many people grows into (1 town, 2 city, 3 metropolis, 4 megacity) ...
te_state_picture_size_up = {
	value = 1
	if = {
		limit = { te_state_picture_city_people >= te_state_picture_k_megacity }
		value = 4
	}
	else_if = {
		limit = { te_state_picture_city_people >= te_state_picture_k_metropolis }
		value = 3
	}
	else_if = {
		limit = { te_state_picture_city_people >= te_state_picture_k_city }
		value = 2
	}
}

# ... and the size it has shrunk to: the same floors times te_state_picture_k_keep. Never below
# te_state_picture_size_up; between the two a state keeps the size it has.
te_state_picture_size_down = {
	value = 1
	if = {
		limit = { te_state_picture_city_people >= te_state_picture_k_megacity_keep }
		value = 4
	}
	else_if = {
		limit = { te_state_picture_city_people >= te_state_picture_k_metropolis_keep }
		value = 3
	}
	else_if = {
		limit = { te_state_picture_city_people >= te_state_picture_k_city_keep }
		value = 2
	}
}

# The state's whole population, for the self-check: its largest city can't hold more.
