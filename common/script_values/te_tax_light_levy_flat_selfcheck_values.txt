# TE_SELF_CHECK tax_light_levy_flat (common/on_actions/te_tax_light_levy_flat_selfcheck_on_actions.txt): the counts
# its window stores, read back for its verdict and its log line. Delete with that file. Each reads 0 until the first
# window opens (the counts are set together with te_tllf_window).
te_tllf_chk_n_g = {
	value = 0
	if = { limit = { has_global_variable = te_tllf_n } value = global_var:te_tllf_n }
}

	value = 0
	if = { limit = { has_global_variable = te_tllf_held } value = global_var:te_tllf_held }
}

# The shares the verdict reads: held, matched, at 0 or above over the levy group counted, control matched over the
# control counted (0 with none counted).
te_tllf_chk_held_share = {
	value = 0
	if = {
		limit = { te_tllf_chk_n_g > 0 }
		value = te_tllf_chk_held_g
		divide = te_tllf_chk_n_g
	}
}
te_tllf_chk_match_share = {
	value = 0
	if = {
		limit = { te_tllf_chk_n_g > 0 }
		value = te_tllf_chk_match_g
		divide = te_tllf_chk_n_g
	}
}

# Country scope: the country's dividend tax rate from every source less the code's rate (te_tax_en_div steps of
# te_tax_step_div), 0 without the variable.
te_tllf_chk_dev = {
	value = 0
	if = {
		limit = { has_variable = te_tax_en_div }
		value = modifier:tax_dividends_add
		subtract = {
			value = var:te_tax_en_div
			multiply = te_tax_step_div
		}
	}
}
