# ---- The workforce term W (state scope) ------------------------------------------------------------------------

# W's target: the census's target on the engine's scale less the define's base (calls 3 and 4; the pops' own bases'
# offset from it and the lines on top stay on top). demographics_model.workforce_term_target.
te_demog_wf_term_target = {
	value = te_demog_wf_target
	subtract = te_demog_k_working_adult_base
}

# W in force, read through has_modifier, so a state that lost its modifier (a change of owner, a revolution's winner,
# which keeps variables only) adds nothing that isn't there. Inside the refresh's own effect has_modifier can't see its
# add or remove, so the refresh reads this before it touches the pair.
te_demog_census_workforce_applied = {
	value = 0
	if = {
		limit = {
			OR = {
				has_modifier = te_demog_census_workforce_up
				has_modifier = te_demog_census_workforce_down
			}
		}
		value = var:te_dg_mw
	}
}

# The pair's multipliers: |W| on the side W is on (ROOT = the state).
te_demog_census_workforce_up_mult = { value = var:te_dg_mw min = 0 }
te_demog_census_workforce_down_mult = { value = var:te_dg_mw multiply = -1 min = 0 }

# The ratio the state's pops are set to now: their own bases, people-weighted (the walk's te_dg_wbase), plus the state's
# working-adult read (W, or outside Full the moved lines' pair, and whatever is on top). The engine pulls each pop's
# workforce toward it (the self-checks and the panel set the walk's te_dg_wf_engine beside it).
te_demog_wf_set = {
	value = modifier:state_working_adult_ratio_add
	if = {
		limit = { has_variable = te_dg_wbase }
		add = var:te_dg_wbase
	}
	else = {
		add = te_demog_k_working_adult_base
	}
}

# Where W heads, on the engine's scale (demographics_model.engine_total): the census's target, the pops' bases' offset
# from the define and the lines on top at the last refresh (te_dg_wf_rem; 0 before one).
te_demog_wf_engine_target = {
	value = te_demog_wf_target
	if = {
		limit = { has_variable = te_dg_wbase }
		add = var:te_dg_wbase
		subtract = te_demog_k_working_adult_base
	}
	if = {
		limit = { has_variable = te_dg_wf_rem }
		add = var:te_dg_wf_rem
	}
}

# Women's share of work deaths (§2.5; demographics_model.work_death_share): under Full the census's own count of women
# among the workers at the work deaths' ages at the last count (te_dg_fws_wk, te_demog_work_finish_step2), else the laws'
# share (te_demog_female_work_share, generated: 0.1 + state_female_work_share_add). te_demog_prepare reads it; the flows
# keep the laws' share.
te_demog_work_death_share = {
	value = te_demog_female_work_share
	if = {
		limit = {
			te_demog_effects_run = yes
			has_variable = te_dg_fws_wk
		}
		value = var:te_dg_fws_wk
	}
}

# ---- The pension setting (country scope) -----------------------------------------------------------------------

# The position (te_demog_k_pension_pos_*): the variable, Standard when it is absent.
te_demog_pension_pos = {
	value = te_demog_k_pension_pos_standard
	if = {
		limit = { has_variable = te_dg_pension_pos }
		value = var:te_dg_pension_pos
	}
}

# When the five-year term ends, as the year with its months (te_demog_ms_now's units: te_dg_pension_since holds the
# date the position was chosen, to the month): since + te_demog_k_pension_term_years; now when none was chosen (no term
# running).
te_demog_pension_term_end = {
	value = te_demog_ms_now
	if = {
		limit = { has_variable = te_dg_pension_since }
		value = var:te_dg_pension_since
		add = te_demog_k_pension_term_years
	}
}

# The years of the term still to run, to the month (0 when it is over or none runs).
te_demog_pension_term_left = {
	value = te_demog_pension_term_end
	subtract = te_demog_ms_now
	min = 0
}

# The panel's years: the one the position was chosen in and the one its term ends in.
te_demog_pension_since_year = {
	value = var:te_dg_pension_since
	floor = yes
}
te_demog_pension_term_end_year = {
	value = te_demog_pension_term_end
	floor = yes
}

# The age the linked positions head for (demographics_model.pension_position_age): the country's people-weighted
# old-age threshold less te_demog_k_pension_link_offset, at least te_demog_k_pension_link_floor; the floor before the
# country's first census under this build. Linked to Average Longevity follows the state's blend (te_dg_old_age); Linked
# to Common Longevity the untreated's own (te_dg_old_age_c, phase 3), which is the blend until anyone is treated.
te_demog_pension_link_target_average = {
	value = te_demog_k_pension_link_floor
	if = {
		limit = { has_variable = te_dg_old_age }
		value = var:te_dg_old_age
		subtract = te_demog_k_pension_link_offset
		min = te_demog_k_pension_link_floor
	}
}
te_demog_pension_link_target_common = {
	value = te_demog_pension_link_target_average
	if = {
		limit = { has_variable = te_dg_old_age_c }
		value = var:te_dg_old_age_c
		subtract = te_demog_k_pension_link_offset
		min = te_demog_k_pension_link_floor
	}
}
te_demog_pension_link_target = {
	value = te_demog_pension_link_target_average
	if = {
		limit = { te_demog_pension_pos = te_demog_k_pension_pos_linked_common }
		value = te_demog_pension_link_target_common
	}
}

# This year's step of a linked age toward its target, at most a year either way.
te_demog_pension_link_step = {
	value = te_demog_pension_link_target
	subtract = var:te_dg_pension_age
	min = te_demog_k_pension_link_step_neg
	max = te_demog_k_pension_link_step
}

# ---- The bills (country scope) ---------------------------------------------------------------------------------

# B: te_demog_k_pension_b_per_level x the Social Security level (late-game §5.1).
te_demog_pension_b = {
	value = "institution_investment_level(institution_social_security)"
	multiply = te_demog_k_pension_b_per_level
}

# The pension bill a week (§2.3; scripting_best_practices.md, "Expense Scaling with GDP"): GDP is yearly and the
# expense weekly, so GDP / 52 x the pensioners' share x B. A rich country at Standard with a fifth of its people
# pensioned and Social Security at 5 pays 6% of GDP a year (te_dg_pensioners, people-weighted over its census states).
te_demog_pension_bill_week = {
	value = gdp
	divide = 52
	multiply = var:te_dg_pensioners
	multiply = te_demog_pension_b
	min = 0
}

# The bill in force, a week, as the census set it (te_demog_bills_apply's multiplier; ROOT = the country).
te_demog_pension_bill_mult = { value = var:te_dg_pension_bill }

# The bill's share of the country's revenue (the panel; the AI's weight): 0 with no bill or no income.
te_demog_pension_bill_share = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_pension_bill
			income > 0
		}
		value = var:te_dg_pension_bill
		divide = { value = income min = 1 }
	}
}

# The health bill's term (§2.4; demographics_model.health_term): the need term, the people-weighted health need over
# 1836's less 1 times the weight, at most the cap (negative for a country younger than 1836's world), plus the
# chronic-treatment technologies and (phase 3, late-game §5.2) longevity care paid by the system.
te_demog_health_need_term = {
	value = var:te_dg_need
	subtract = 1
	multiply = te_demog_k_health_need_weight
	max = te_demog_k_health_need_cap
}
te_demog_health_term_now = {
	value = te_demog_health_need_term
	add = te_demog_health_tech_term
	add = te_demog_health_lx_term
}
# Longevity care paid by the system (late-game §5.2, call 13): te_demog_k_lx_public_health_weight x the share of people on
# the public route, at most the share in care (the country's te_dg_lx_pub and te_dg_elig; 0 before the late game).
te_demog_health_lx_term = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_lx_pub
			has_variable = te_dg_elig
		}
		value = var:te_dg_lx_pub
		max = var:te_dg_elig
		multiply = te_demog_k_lx_public_health_weight
	}
}
te_demog_health_bill_up_mult = { value = var:te_dg_health_term min = 0 }
# the health term below 0, as a saving (the hover's second wording); 0 with no term (the hover reads it unguarded)
te_demog_health_term_saving = {
	value = 0
	if = {
		limit = { has_variable = te_dg_health_term }
		value = var:te_dg_health_term
		multiply = -1
	}
}
te_demog_health_bill_down_mult = { value = var:te_dg_health_term multiply = -1 min = 0 }

# ---- The AI's yearly pulse (country scope; §2.2) ---------------------------------------------------------------

# Unemployment, people-weighted over the country's states: each state's at its last census (te_demog_unemployment,
# te_state_unemployment_share stored then: the share of the workforce with no work, 0-1).
te_demog_country_unemployment = {
	value = 0
	every_scope_state = {
		add = {
			value = te_demog_unemployment
			multiply = state_population
		}
	}
	divide = { value = total_population min = 1 }
}

# ---- The panel (te_demog_sec_work, te_state_demog_work, the overview's working age; gui/te_demographics_widgets.gui)
# THIS = a country or a state with the work figures (te_demog_work_ready_sgui, te_demog_state_work_sgui).

# The workforce share: the working-age share x the share at work (workers over people).
te_demog_wk_share = {
	value = var:te_dg_wa_share
	multiply = var:te_dg_at_work
}

# The share of the working ages at work as shown: their own workers over their people, 0 to 1. The stored te_dg_at_work
# also counts the workers past the end (te_dg_wk_past, of all people), so it passes 100% where they are many; the row
# adds them as a term of their own (te_demog_wk_past_pct), so working age x this + them = te_demog_wk_share.
te_demog_wk_at_work_shown = {
	value = var:te_dg_wa_share
	multiply = var:te_dg_at_work
	if = {
		limit = { has_variable = te_dg_wk_past }
		subtract = var:te_dg_wk_past
	}
	divide = { value = var:te_dg_wa_share min = 0.00001 }
	max = 1
	min = 0
}

# The workers past the end of the working ages over all people, in whole percent (rounded): the row shows them as a
# term of their own when this is above 0.
te_demog_wk_past_pct = {
	value = 0
	if = {
		limit = { has_variable = te_dg_wk_past }
		value = var:te_dg_wk_past
		multiply = 100
		add = 0.5
		floor = yes
	}
}

te_demog_wf_pop_base = {

	}
}

# Country scope: 1 under Old Age Pension, 2 under UBI or Post-Scarcity, else 0 (the hover's working-age bounds; the
# state's te_demog_pension_kind reads its owner the same way).
te_demog_country_pension_kind = {
	value = 0
	if = {
		limit = { OR = { has_law = law_type:law_universal_basic_income has_law = law_type:law_post-scarcity } }
		value = 2
	}
	if = {
		limit = { te_demog_has_pension_law = yes }
		value = 1
	}
}

# Country scope: the Social Security level that sets B (the bill's hover).
te_demog_social_security_level = {

}
