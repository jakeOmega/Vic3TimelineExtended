# ============================================================================
# SPACE RACE — display values
# ============================================================================
# Read by the milestone panels' overview and the programme row
# (gui/journal_entry_widgets/space_race_widget.gui). Rendered every frame an
# entry's panel is open, so these read the viewer's own variables and the
# space race's global flags only, and every variable read is guarded: a
# country that never ran a milestone, or a save from before the overview,
# reads 0 (test_space_race_layout.py). Display only: no effect, event or
# trigger reads a sr_disp_* value.
# Scope: country (the viewer)
# ============================================================================

# ---- A milestone's state, from its single derivation site -------------------
# 0 not running · 1 standard · 2 safe · 3 ambitious · 4 inside the
# post-setback review (any approach; the roll is skipped). Read from

# sr_set_milestone_status_base: absent -> 0, then 0 / 1 / 2 / 3-5 -> 1 / 2 / 3 / 4.

# ---- The programme row: one code per entry -----------------------------------
# 0 not begun · 1 under way · 2 another power got there first (and we have
# not finished it) · 3 achieved · 4 achieved, and we were first. Lit from 3 up,
# so the codes rise with what we hold. The same order of tests as the icon's
# tooltip, sr_prog_<m> in space_race_custom_loc.txt: first, then done, then
# running, then taken.

# The Interstellar Probe and the wait for its data share one cell (owner's
# play-test, round 3), with four states of its own: 0 not begun (or another
# power finished it first), 1 under way, 2 launched and awaiting data, 3 the
# data is in. sr_probe_launched is set when the probe entry completes, before
# the waiting entry opens, so state 2 has no gap. The cell's first-to-finish
# flag still reads sr_disp_prog_interstellar_probe above.
sr_disp_prog_interstellar = {
	value = 0
	if = { limit = { has_variable = sr_interstellar_results_received } value = 3 }
	else_if = { limit = { has_variable = sr_probe_launched } value = 2 }
	else_if = { limit = { has_variable = sr_active_interstellar_probe } value = 1 }
}

# ---- Rivals: the estimate rows ----------------------------------------------
# Read by the Rivals rows with the rival passed in: GuiScope.SetRoot(viewer)
# .AddScope('sr_rival', <the rival>).ScriptValue(...), the shape
# gui/market_panel.gui uses for a trade partner. ROOT is the viewer.
# The band: ±25% of the goal, ±10% / ±5% with a covert network of
# intelligence tier 1 / 2 in the rival (sr_has_intel_tier).
sr_disp_rival_band = {
	value = sr_rival_band_open
	if = {
		limit = { root = { sr_has_intel_tier = { TARGET = scope:sr_rival TIER = 2 } } }
		value = sr_rival_band_tier_2
	}
	else_if = {
		limit = { root = { sr_has_intel_tier = { TARGET = scope:sr_rival TIER = 1 } } }
		value = sr_rival_band_tier_1
	}
}
