# Country scope. Read-only; GUI passes current engine budget getters as value
# scopes (with private construction excluded). No monthly cache, game-rule gate
# or journal entry is required.

# Public total passed by the GUI after excluding Investment Pool Transfer.
te_budget_total = { value = scope:total }

# Ordinal passed by each sorted flow slot; no engine layout vector binding.
te_budget_row_slot = { value = scope:row_slot }

# The four building sums below run on every frame the tab is open. Each walks
# the country's states and looks up the one building of its type in each
# (`b:<type>`), never every building the country has (`every_scope_building`):
# a state has at most one building of a type, so the sum is the same, and the
# Imperial Union of America of December 2032 spent 1.76 s of a 13.9 s profile
# in the four building walks (docs/audits/late_game_performance.md § "The
# Budget Breakdown tab"). `exists = b:<type>` also finds a queued building of
# no level, as the building walk did. Each building type named here must be
# every member of its building group, vanilla's and the mod's:
# test_budget_breakdown.BuildingLookupTest holds that.
#
# Administration produces bureaucracy/tax capacity rather than saleable goods.
# Its negative weekly profit therefore measures its wages + input goods (and
# slave upkeep, if any). Do not use ALL government buildings' expenses: that
# would also allocate universities, ports and construction sites to ministries.
te_budget_administration_actual = {
	value = 0
	every_scope_state = {
		limit = { exists = b:building_government_administration }
		b:building_government_administration = {
			add = { value = weekly_profit multiply = -1 min = 0 }
		}
	}
}

# The overview predicts wages but uses last week's goods, while weekly_profit
# is last week's building balance. Limit allocation to the civil budget pool
# so a wage-level change cannot allocate more money than that pool contains.
#
# Every read of a scope the panel passes is behind `exists`: the engine has
# evaluated the institution-pool child once with none of the parent's AddScope
# values set (debug.log, 2026-11-01; see #1310), and an unguarded `scope:x`
# then logs "Value of wrong type ... 'none'" and "Undefined event target". The
# values read 0 for that frame.
te_budget_administration_cost = {
	value = 0
	if = {
		limit = {
			exists = scope:administration_actual
			exists = scope:civil_0
			exists = scope:civil_1
			exists = scope:civil_2
		}
		value = scope:administration_actual
		max = {
			value = scope:civil_0
			add = scope:civil_1
			add = scope:civil_2
			min = 0
		}
	}
}

# The Space Program's missions output country modifiers only, so like
# Administration its negative weekly profit is its wages + input goods (Launch
# Capacity). The engine pays it as a government building, under the civil
# getters; the Breakdown shows it under Programme Costs, Space Race instead.
te_budget_space_program_actual = {
	value = 0
	every_scope_state = {
		limit = { exists = b:building_space_program }
		b:building_space_program = {
			add = { value = weekly_profit multiply = -1 min = 0 }
		}
	}
}

# Capped like Administration, at what the civil pool holds after it, so the two
# moves together never take more than the civil getters contain.
te_budget_space_program_cost = {
	value = 0
	if = {
		limit = {
			exists = scope:space_program_actual
			exists = scope:civil_0
			exists = scope:civil_1
			exists = scope:civil_2
		}
		value = scope:space_program_actual
		max = {
			value = scope:civil_0
			add = scope:civil_1
			add = scope:civil_2
			subtract = te_budget_administration_cost
			min = 0
		}
	}
}

te_budget_produced_bureaucracy = { value = produced_bureaucracy min = 0 }

te_budget_institution_pool = {
	value = 0
	if = {
		limit = {
			exists = scope:institution_levels
			exists = scope:institution_usage
			te_budget_produced_bureaucracy > 0
			scope:institution_levels > 0
		}
		value = te_budget_administration_cost
		multiply = { value = scope:institution_usage min = 0 }
		divide = te_budget_produced_bureaucracy
		max = te_budget_administration_cost
	}
}

# Support-building deficits contain their own wages, goods and slave upkeep.
# Barracks/conscript costs and naval administration are already supplied by the
# native branch getters. In particular, barracks unit wages are NOT included in
# the logistics-centre deficit; subtracting the branch forecast here wrongly
# moves those wages into Other Military Costs. Select only support buildings and
# pass each sum once to the GUI, then add it directly to Materials and Support.
# Army: bg_army_logistics_center. Navy: bg_naval_fortification and
# bg_naval_logistics_center, one building type each.
te_budget_army_support_actual = {
	value = 0
	every_scope_state = {
		limit = { exists = b:building_army_logistics_center }
		b:building_army_logistics_center = {
			add = { value = weekly_profit multiply = -1 min = 0 }
		}
	}
}
