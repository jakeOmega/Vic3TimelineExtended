# Country scope. Read-only; GUI passes current engine budget getters as value
# scopes (with private construction excluded). No monthly cache, game-rule gate
# or journal entry is required.

# Public total passed by the GUI after excluding Investment Pool Transfer.
te_budget_total = { value = scope:total }

# Ordinal passed by each sorted flow slot; no engine layout vector binding.
te_budget_row_slot = { value = scope:row_slot }

# Administration produces bureaucracy/tax capacity rather than saleable goods.
# Its negative weekly profit therefore measures its wages + input goods (and
# slave upkeep, if any). Do not use ALL government buildings' expenses: that
# would also allocate universities, ports and construction sites to ministries.
te_budget_administration_actual = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_government_administration }
		add = { value = weekly_profit multiply = -1 min = 0 }
	}
}

# The overview predicts wages but uses last week's goods, while weekly_profit
# is last week's building balance. Limit allocation to the civil budget pool
# so a wage-level change cannot allocate more money than that pool contains.
te_budget_administration_cost = {
	value = scope:administration_actual
	max = {
		value = scope:civil_0
		add = scope:civil_1
		add = scope:civil_2
		min = 0
	}
}

te_budget_produced_bureaucracy = { value = produced_bureaucracy min = 0 }

te_budget_institution_pool = {
	value = 0
	if = {
		limit = {
			te_budget_produced_bureaucracy > 0
			scope:institution_levels > 0
		}
		value = te_budget_administration_cost
		multiply = { value = scope:institution_usage min = 0 }
		divide = te_budget_produced_bureaucracy
		max = te_budget_administration_cost
	}
}

# Modern logistics centres and naval fortifications are charged through the
# military budget, but the branch GUI getters can omit their operating costs.
# Read full operating deficits for each branch once. The GUI passes these sums
# into the chart scope; leaf values never repeat building scans. These buildings
# produce military services, not saleable output, so their deficit is upkeep.
te_budget_army_operating_actual = {
	value = 0
	every_scope_building = {
		limit = {
			OR = {
				is_building_group = bg_army
				is_building_group = bg_conscription
				is_building_group = bg_army_logistics_center
			}
		}
		add = { value = weekly_profit multiply = -1 min = 0 }
	}
}

te_budget_navy_operating_actual = {
	value = 0
	every_scope_building = {
		limit = {
			OR = {
				is_building_group = bg_navy
				is_building_group = bg_naval_administration
				is_building_group = bg_naval_fortification
				is_building_group = bg_naval_logistics_center
			}
		}
		add = { value = weekly_profit multiply = -1 min = 0 }
	}
}
