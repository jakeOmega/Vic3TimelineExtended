# ----------------------------------------------------------------------------
# Trend arrows. Each is 1 (rose), -1 (fell) or 0 (level) for a figure against
# last year's (te_dg_prev_<name>), and 2 (no arrow) unless te_dg_trend_ok says
# both years came from a step: a seed's figures are no year to compare with
# (te_demog_set_trend). The change divided by the step below, clamped and
# rounded, so a change of under half a step reads level. The panel tests the
# code for identity (gui_modding_guide.md gotcha #23). A state and a country
# keep the same names, so these work in either scope.
# ----------------------------------------------------------------------------

# Median age: a step of 0.1 years.
te_demog_trend_median = {
	value = 2
	if = {
		limit = {
			has_variable = te_dg_trend_ok
			var:te_dg_trend_ok > 0
			has_variable = te_dg_median
			has_variable = te_dg_prev_median
		}
		value = var:te_dg_median
		subtract = var:te_dg_prev_median
		divide = 0.1
		min = -1
		max = 1
		round = yes
	}
}
