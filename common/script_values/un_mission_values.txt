# ============================================================================
# UN MISSIONS — THE FIGURES (redesign phase 6)
# ============================================================================
# docs/systems/un_redesign_design.md §9; as shipped, §0.6. The UN happens in
# places: a mission sits in one state, its effects are that state's, and its
# outcome is the delivery the Assembly is judged on.
#
#   type            opened by                                 succeeds when
#   1 peacekeeping  a peacekeeping request carried in full     24 months of peace
#                   (un_vote.2, Established and above)         for the host
#   2 aid           an aid request carried (un_vote.2); a      the state has no famine
#                   donor answering a famine (un_events.7 A)   and its SoL, less the
#                                                              mission's own help, has
#                                                              risen past its goal
#   3 stabilisation a power answering a state collapse         the host is no longer a
#                   (un_events.4 A)                            failed state (then 12
#                                                              months to wind down)
#
# A mission FAILS when its host is attacked (at war and not the war's
# initiator) after it opened, or, for peacekeeping and stabilisation, when
# every contributor it had has gone. A host already at war when the mission
# opened is not "attacked" until it has known a month of peace; the war
# stalls a peacekeeping mission instead. A peacekeeping or stabilisation
# mission nobody has joined within un_mission_unstaffed_months LAPSES, as
# does any mission after un_mission_max_months. Success is a delivery entry
# and standing for the contributors; failure is a credibility debit.
#
# STRENGTH. Each month a mission's strength is
#     E x (1 - 0.5 x the share of the members' GDP withholding dues)
#       x (0.75 + 0.25 x its contributors, counted up to 3)
# so a Moribund UN's missions do nothing, withheld dues starve every mission
# (the budget is what the paying members pay), and a mission with three
# contributors is half as strong again as one with a single one. The state's
# modifier is multiplied by it.
#
# The writers are in common/scripted_effects/un_mission_effects.txt, the tests
# in common/scripted_triggers/un_mission_triggers.txt. Every figure is here,
# once; some are also written in words in the loc; change both.
#
# Scope: the strength values are evaluated on the mission container; the
# ordering values on a state.
# ============================================================================

# ---- Tuning ----------------------------------------------------------------

# Missions kept in the register, closed ones evicted oldest first. Active
# missions are never evicted. Also written as the literal 13 (cap + 1) in
# un_mission_register; change both.
un_mission_registry_cap = { value = 12 }

# A mission still open after this many months lapses.
un_mission_max_months = { value = 60 }

# Progress a month, out of 100: peacekeeping over 24 months of peace, a
# stabilisation over 12 months once the collapse is over (48 while it lasts).
un_mission_pk_progress = { value = 4.17 }
un_mission_stab_progress_recovered = { value = 8.34 }
un_mission_stab_progress_collapsing = { value = 2.09 }

# Aid succeeds once the state has no famine and its standard of living, less
# what the mission itself adds, has reached its GOAL: this target, or this
# much above what it was when the mission opened, whichever is higher; after
# at least un_mission_aid_min_months. So a mission cannot succeed by its own
# modifier, or in a state that never needed it.
un_mission_aid_sol_target = { value = 10 }
un_mission_aid_sol_gain = { value = 1 }
un_mission_aid_min_months = { value = 6 }

# The standard of living un_mission_aid_state_modifier adds at strength 1.
# Also written in the modifier (extra_modifiers.txt); change both.
un_mission_aid_sol_add = { value = 1.5 }

# A peacekeeping or stabilisation mission that no member has joined within
# this many months lapses: the Assembly voted for a force nobody sent.
un_mission_unstaffed_months = { value = 6 }

# What an outcome is worth. Delivery and credibility are institutional ledger
# entries; standing goes to each contributor.
un_mission_success_delivery = { value = 2 }
un_mission_failure_credibility = { value = -1 }
un_mission_success_standing = { value = 3 }

# The contributors' covert networks in a mission's host grow this much faster
# (§7.4): accepting help has a price.
un_mission_net_gain_mult = { value = 1.25 }

# ---- Joining at will -----------------------------------------------------------
# A major-power member may send a contingent to any mission in the field, and
# bring it home again (un_mission_volunteer / un_mission_withdraw). It pays
# for each mission it serves in this way; the members that pledged through the
# Assembly, or answered a collapse or a famine, pay through that path instead.
# The chamber shows the missions in eight display slots (0-7); the slot table
# is unrolled in un_mission_assign_slot, the three mission sguis and the
# widget's rows (test_un_chamber_mission_slots.py keeps them in step).

# A week's service in one mission is this share of what a programme costs
# (un_program_expense_value, half a percent of GDP a year): a quarter of a
# percent of GDP a year for each mission. The peacekeeping programme pays for
# its members' peacekeeping and stabilisation contingents (un_mission_charges).
un_mission_service_share = { value = 0.5 }

# Bringing a contingent home before the work is done: standing lost (a loss,
# so not scaled), and the host's regard (written as the literal -10 in
# un_mission_withdraw; change both).
un_mission_withdraw_standing = { value = 2 }

# The AI sends a contingent when a mission scores above this
# (un_mission_ai_join_score), to at most un_mission_ai_volunteer_cap missions.
un_mission_ai_join_threshold = { value = 10 }

# Scope: country
un_mission_ai_volunteer_cap = {
	value = 2
	if = {
		limit = { country_rank >= rank_value:great_power }
		add = 1
	}
}

# One mission's weekly service cost. Scope: country
un_mission_service_one_expense_value = {
	value = un_program_expense_value
	multiply = un_mission_service_share
}

# The missions the scoped country serves in at will and pays for. Scope: country
un_mission_service_count = {
	value = 0
	if = {
		limit = { has_global_variable_list = un_mission_registry }
		every_in_global_list = {
			variable = un_mission_registry
			limit = { un_mission_charges = { WHO = PREV } }
			add = 1
		}
	}
}

# Every active mission the scoped country serves in at will, paid for or not:
# what the AI's cap counts. Scope: country
un_mission_volunteer_count = {
	value = 0
	if = {
		limit = { has_global_variable_list = un_mission_registry }
		every_in_global_list = {
			variable = un_mission_registry
			limit = {
				has_tag = un_msn_active
				un_mission_served_by = { WHO = PREV }
				un_lobby_in_list = { LIST = un_msn_volunteers COUNTRY = PREV }
			}
			add = 1
		}
	}
}

# The week's cost of every mission the scoped country pays to serve in.
# Cached as var:un_msn_service_cached by un_mission_service_refresh, the
# multiplier of un_mission_service_cost. Scope: country
un_mission_service_expense_value = {
	value = un_mission_service_one_expense_value
	multiply = un_mission_service_count
}

# How much the AI member (ROOT) wants to serve in the scoped mission: the ties
# the lean reads (un_dossier_values.txt), what the mission lacks, and what the
# member stands for. Scope: mission container; ROOT: the AI member.
un_mission_ai_join_score = {
	value = -10
	# The mission needs hands: it is stronger with every contributor up to three.
	add = {
		value = 3
		subtract = un_mission_contributor_count
		min = 0
		multiply = 5
	}
	var:un_msn_host ?= {
		if = {
			limit = { has_treaty_alliance_with = { TARGET = ROOT } }
			add = 15
		}
		if = {
			limit = { is_in_same_power_bloc = ROOT }
			add = 15
		}
		if = {
			limit = { is_subject_of = ROOT }
			add = 15
		}
		if = {
			limit = { has_diplomatic_pact = { who = ROOT type = rivalry } }
			add = -20
		}
		if = {
			limit = { has_diplomatic_relevance = ROOT }
			add = 5
		}
		add = {
			value = relations:root
			divide = 10
			min = -10
			max = 10
		}
	}
	# The peacekeeping programme already pays for these.
	if = {
		limit = {
			OR = {
				has_tag = un_msn_peacekeeping
				has_tag = un_msn_stabilisation
			}
			ROOT = { un_standing_program_active_peacekeeping = yes }
		}
		add = 15
	}
	if = {
		limit = { ROOT = { has_law = law_type:law_humanitarian_regulations } }
		add = 5
	}
	if = {
		limit = { ROOT = { country_rank >= rank_value:great_power } }
		add = 5
	}
	if = {
		limit = {
			has_global_variable = un_authority
			global_var:un_authority >= 40
		}
		add = 5
	}
}

# ---- Strength ------------------------------------------------------------------

# Scope: mission container
un_mission_contributor_count = {
	value = 0
	if = {
		limit = { has_variable_list = un_msn_contributors }
		every_in_list = {
			variable = un_msn_contributors
			add = 1
		}
	}
}

# Scope: mission container
un_mission_strength_value = {
	value = un_mission_contributor_count
	max = 3
	multiply = 0.25
	add = 0.75
	multiply = un_mission_budget_factor
	multiply = un_enforcement
}

# The budget's share of the strength: withheld dues starve every mission.
# Globals only.
un_mission_budget_factor = {
	value = 1
	if = {
		limit = { has_global_variable = un_funding_arrears_share }
		subtract = {
			value = global_var:un_funding_arrears_share
			multiply = 0.5
		}
	}
}

# The state's standard of living without the mission's own help: its average
# SoL less what the aid modifier adds at the state's current strength.
# Scope: mission container
un_mission_underlying_sol = {
	value = 0
	var:un_msn_state ?= {
		add = average_sol
		if = {
			limit = {
				has_variable = un_msn_power
				has_modifier = un_mission_aid_state_modifier
			}
			subtract = {
				value = var:un_msn_power
				multiply = un_mission_aid_sol_add
			}
		}
	}
}

# An aid mission's goal (see un_mission_aid_sol_target), and how far it has
# to go from where it started (at least un_mission_aid_sol_gain).
# Scope: mission container
un_mission_aid_goal = {
	value = 0
	if = {
		limit = { has_variable = un_msn_sol_open }
		add = var:un_msn_sol_open
	}
	add = un_mission_aid_sol_gain
	min = un_mission_aid_sol_target
}

un_mission_aid_span = {
	value = un_mission_aid_goal
	if = {
		limit = { has_variable = un_msn_sol_open }
		subtract = var:un_msn_sol_open
	}
	min = un_mission_aid_sol_gain
}

# An aid mission's progress, for the display: how far the state's underlying
# standard of living has come from where it started toward the goal (99 at
# most; success is decided by un_mission_aid_recovered).
# Scope: mission container
un_mission_aid_progress_value = {
	value = un_mission_underlying_sol
	if = {
		limit = { has_variable = un_msn_sol_open }
		subtract = var:un_msn_sol_open
	}
	divide = un_mission_aid_span
	multiply = 100
	min = 0
	max = 99
}

# ---- Where a mission goes (Scope: state) ---------------------------------------------

# Aid: the state most in need, by standard of living, and by devastation,
# since an aid request can be carried on devastation alone.
un_mission_sol_need_order = {
	value = 100
	subtract = average_sol
	add = devastation
}

# ---- The state panel's UN Mission tile (Scope: state) ------------------------------
# Every value is 0 for a state with no active mission, so the tile can gate on
# the first and never read a missing variable.

# 0 none, 1 peacekeeping, 2 aid, 3 stabilisation.
un_state_mission_type = {
	value = 0
	if = {
		limit = { var:un_msn_here ?= { has_tag = un_msn_active } }
		var:un_msn_here = { add = var:un_msn_type }
	}
}
