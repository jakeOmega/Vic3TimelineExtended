# ============================================================================
# GLOBAL WARMING — how the AI chooses mitigation policies
# ============================================================================
# Read ONLY by the `ai_chance` blocks in
# common/scripted_buttons/global_warming_buttons.txt (and the console dump in
# common/scripted_effects/te_debug_gw_effects.txt). Nothing here gates a player
# control: `possible` stays in global_warming_triggers.txt, shared with the
# widget, and movement_environmental's radicalism factors mirror it.
#
# WHY A SCORE AND NOT PER-BUTTON TERMS. The buttons used to weigh temperature
# alone, and every repeal weight was zero above 0.3-0.5 C, so an adopted policy
# never came off. Adoption was a one-way ratchet: every AI eventually rolled
# every positive button, and a 2008 save (1.26 C) had 132 of 136 countries
# holding seven of the eight policies. The only brake was laissez-faire, which
# one country held -- the player.
#
# Now each policy has a climate will per country (0 to roughly 100) and a
# threshold on it. Adopt scores only at or above the threshold; repeal scores
# only below threshold - gw_ai_repeal_band. Between the two neither button
# scores, so a policy sits where the country's conditions put it, and how often
# the engine lets the AI click only sets how fast it gets there.
#
# ONE WILL PER POLICY, BUILT FROM THE SAME PARTS. Every will is the shared core
# (temperature, the environment ministry, an environmentalist in government)
# plus the same five signals, each 0..1 or -1..1, weighted by who the policy
# actually costs or helps. The weights are the table below; every will lists
# all five signals, zeros included, and test_gw_ai_policy_table.py checks the
# table against the code, so edit both together.
#
#   policy                   laissez_faire  industrialists  movement  wealth  fossil
#   climate_adaptation                 -10               0         5     -10       0
#   reforestation                      -10               0        10      -5       0
#   renewable_investment               -20              -5        10      10     -10
#   public_transit                     -10               0        10       5      -5
#   green_building_codes               -20              -5        10      10       0
#   emission_standards                 -20             -12        10      10      -5
#   carbon_tax                         -25             -12        10      10     -20
#   fossil_fuel_divestment             -25              -8        10      10     -25
#
# Why each row departs from the plain -20 / -8 / 10 / 10 / -10 of the first
# version, read off the policy's modifier in extra_modifiers.txt:
#   climate_adaptation  +0.5 SoL flat and -2.5% mortality are worth most where
#                       SoL is lowest, so wealth counts AGAINST it; it touches no
#                       industry or fuel; greens campaign for cuts, not coping;
#                       a state programme, not a market intervention.
#   reforestation       agriculture throughput and drought/flood cover pay most
#                       in farming economies, which run poorer.
#   renewable_investment  subsidised renewables, -10% conventional power plant
#                       throughput: costs money, cuts coal burn.
#   public_transit      -5% oil input, -25% automobile infrastructure use;
#                       infrastructure, which laissez-faire minds least.
#   green_building_codes  +5% construction input: a cost on builders.
#   emission_standards  -5% throughput in EVERY building: industry pays.
#   carbon_tax          oil and coal extraction taxed double, -10% fuel output,
#                       +10% manufacturing tax: exporters and industry pay.
#   fossil_fuel_divestment  +25% extraction tax: aimed at the fuel sector.
#
# THE BAND MUST STAY WIDER THAN ANY ONE POLITICAL SWING. Interest groups change
# at every election, and both IG terms are yes/no; with a band narrower than an
# IG weight, a country parked near a threshold would adopt and repeal a policy
# every four years. The band is 15; the largest IG weight is 12. Weights above
# 15 sit only on laws (a repeal on enacting laissez-faire is the point) or on
# continuous signals (fossil), never on a yes/no that flips on its own.
#
# Scope: country, throughout.
# ============================================================================

# ----------------------------------------------------------------------------
# The shared core: the same for every policy.
# ----------------------------------------------------------------------------
gw_ai_will_shared = {
	value = gw_ai_will_urgency
	if = {
		limit = { has_law_or_variant = law_type:law_ministry_of_the_environment }
		add = 15
	}
	if = {
		limit = {
			any_interest_group = {
				is_in_government = yes
				leader ?= { has_ideology = ideology:ideology_environmentalists }
			}
		}
		add = 8
	}
}

# How bad it has got: 20 per degree, 25 at 1.26 C, 40 at 2 C, 60 at 3 C. The
# per-policy thresholds already say how bad each policy needs it to be, so this
# is not weighted per policy.
gw_ai_will_urgency = {
	value = temperature_anomaly_display
	multiply = 20
	min = 0
	max = 70
}

# ----------------------------------------------------------------------------
# The five signals the table weights.
# ----------------------------------------------------------------------------

# 1 under laissez-faire.
gw_ai_sig_laissez_faire = {
	value = 0
	if = {
		limit = { has_law_or_variant = law_type:law_laissez_faire }
		add = 1
	}
}

# 1 while the industrialists are in government.
gw_ai_sig_industrialists = {
	value = 0
	if = {
		limit = {
			ig:ig_industrialists ?= { is_in_government = yes }
		}
		add = 1
	}
}

# The environmental movement's support: 0 at none, 1 at 20% of the population
# or more.
gw_ai_sig_movement = {
	value = 0
	every_political_movement = {
		limit = { is_political_movement_type = movement_environmental }
		add = this.political_movement_support
	}
	multiply = 5
	max = 1
}

# Standard of living against the world mean over countries with the journal
# entry (gw_refresh_global_counts_effect stores it monthly): -1 at 10 points
# below, 1 at 10 points above. 0 until the first monthly sweep.
gw_ai_sig_wealth = {
	value = 0
	if = {
		limit = { has_global_variable = gw_g_mean_sol }
		add = average_sol
		subtract = global_var:gw_g_mean_sol
		divide = 10
	}
	min = -1
	max = 1
}

# How far this country's market is a net fossil-fuel exporter: 0 while it
# burns at least the coal and oil it produces, 1 once it produces twice that.
# Continuous on purpose: the heaviest weights in the table sit here, and a
# yes/no cut-off would flip divestment every time a market wobbled across it.
gw_ai_sig_fossil = {
	value = gw_ai_market_fossil_output_ratio
	subtract = 1
	min = 0
	max = 1
}

# Coal and oil production over consumption for this country's market.
gw_ai_market_fossil_output_ratio = {
	value = 0
	market = {
		mg:coal = { add = market_goods_production }
		mg:oil = { add = market_goods_production }
	}
	divide = {
		value = 0
		market = {
			mg:coal = { add = market_goods_consumption }
			mg:oil = { add = market_goods_consumption }
		}
		min = 1
	}
}

# ----------------------------------------------------------------------------
# Thresholds. A policy's threshold appears ONCE, in its adopt margin; the repeal
# margin is derived from it, so the two can never drift apart
# (test_gw_ai_policy_table.py checks the pairing).
#
# What the thresholds mean for a country whose five signals are all 0 -- no
# laissez-faire, no industrialists or environmentalists in government, no
# movement, average wealth, no fuel exports -- where every policy's will is the
# shared core:
#
#   will            no ministry   with ministry (+15)
#   1.26 C          25            40
#   2.0 C           40            55
#   3.0 C           60            75
#
#   threshold   policy                   first adopted by
#   25          Climate Adaptation       everyone at 1.26 C, if authority allows
#   30          Reforestation Subsidies  a ministry, or a movement, at 1.26 C
#   35          Renewable Investment     a ministry at 1.26 C
#   40          Public Transit           a ministry at 1.26 C; everyone at 2 C
#   45          Green Building Codes     ministry + movement at 1.26 C
#   45          Emission Standards       ministry + movement at 1.26 C
#   55          Carbon Tax               ministry + movement + green IG at 1.26 C
#   70          Fossil-Fuel Divestment   rare before 3 C
#
# The signals then move each country off that line by its row in the table:
# a poor country reaches adaptation up to 10 points (half a degree) early and a
# rich one that much late; a full exporter reaches divestment 25 points late.
# ----------------------------------------------------------------------------

# Repeal waits until the will is this far BELOW the threshold.
gw_ai_repeal_band = 15

# Adopt weight = gw_ai_adopt_base_weight + margin, capped. The cap keeps one
# policy from swamping the other buttons in the same roll.
gw_ai_adopt_base_weight = 10
gw_ai_max_weight = 30

# Extra repeal weight on a policy that costs authority, while the country's
# available authority is negative.
gw_ai_authority_deficit_repeal_weight = 15

gw_ai_adopt_margin_climate_adaptation = {
	value = gw_ai_will_climate_adaptation
	subtract = 25
}
gw_ai_repeal_margin_climate_adaptation = {
	value = 0
	subtract = gw_ai_adopt_margin_climate_adaptation
	subtract = gw_ai_repeal_band
}
