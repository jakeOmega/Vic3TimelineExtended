# ============================================================================
# Monetary policy, phase 5 — international monetary arrangements: values
# ============================================================================
# Spec: docs/systems/monetary_policy_design.md §15A, as shipped in §0.8.
# Effects: common/scripted_effects/te_monetary_arrangement_effects.txt.
#
# All country scope unless a comment says otherwise. Units are percentage
# POINTS, like the rest of the monetary files.
#
# VARIABLE CONTRACT (phase 5). Same two rules as every other phase: initialised
# behind has_variable (te_monetary_init_arrangement_variables), never removed.
#
#   te_mon_anchor           scope       the anchor country. A SCOPE variable:
#                                       cannot hold 0, read as
#                                       `var:te_mon_anchor = { ... }` behind
#                                       has_variable, never as a number, never
#                                       through GetValue. Only meaningful while
#                                       te_mon_anchor_kind > 0. CAN BE ABSENT
#                                       after init (a country that was never
#                                       anchored has none) — the
#                                       te_basket_market_owner contract
#   te_mon_anchor_kind      0 to 3      0 none, 1 treaty peg (5a), 2 bloc
#                                       currency (5b), 3 currency board (5c).
#                                       Highest wins
#   te_mon_anchor_rate      pp          \  this pulse's COPIES of the anchor's
#   te_mon_anchor_fx_index  50 to 150    | te_policy_rate, te_fx_index, own
#   te_mon_anchor_c         0 to 1       | credibility c and inflation anchor,
#   te_mon_anchor_target    0 or 2      /  taken by step 1c so that every later
#                                       read — script values, the dashboard — is
#                                       a plain local var: and never a
#                                       cross-country chain. They hold last
#                                       pulse's figures when nobody is anchored
#   te_mon_peg_parity_offset 0 to 50    index points a kind-1 pegger's parity
#                                       sits BELOW its anchor's index (§0.8
#                                       ruling G4: "re-peg lower"). 0 on every
#                                       change of anchor
#   te_mon_arr_scan         0 or 1      1 = this country holds some role in an
#                                       arrangement, so step 1c runs the FULL
#                                       discovery every month for it; 0 = it is
#                                       discovered by the hooks and the yearly
#                                       scan only (§0.8 ruling G2)
#   te_mon_swap_recipient   0 or 1      receives a swap line from somebody
#   te_mon_swap_provider    scope       ...the ONLY provider (111's can_ratify
#                                       allows one at a time, §6a/H1). Same
#                                       scope-variable contract as te_mon_anchor
#   te_mon_swap_drawn       money >= 0  [6a] what the recipient has DRAWN on
#                                       that line and not yet repaid. Grows in a
#                                       financial crisis to te_mon_swap_line_limit,
#                                       shrinks out of one, settled in a lump
#                                       when the role ends. A LOAN, not a gift
#   te_mon_lolr_ward        0 or 1      has a lender of last resort
#   te_mon_lolr_guarantor   scope       ...the (first) guarantor. Same contract
#   te_mon_lolr_calls       0 to 5      [6a] how many times THIS WARD has had
#                                       its guarantee called, this one included.
#                                       A fact about the ward, so it survives a
#                                       change of guarantor. Everything pricing
#                                       THIS call reads te_mon_lolr_calls_prior
#                                       (= calls - 1); everything pricing the
#                                       ONGOING relationship reads the counter
#   te_mon_lolr_call_forget 0 to 120    [6a] ...and the months left before one
#                                       call is forgotten. Re-armed on each call
#   te_mon_lolr_suspended_months 0-60   a GUARANTOR'S clock: it reneged, so its
#                                       other wards' benefit is off this long
#   te_mon_lolr_default_cooldown 0-60   a WARD'S clock: months before its next
#                                       default may call the guarantee again
#   te_mon_arr_recipient_pts  >= 0      \  what the three arrangement modifiers
#   te_mon_arr_provider_pts   >= 0       | SHOULD be at, in pp, recomputed by
#   te_mon_arr_standing_pts   >= 0      /  the discovery
#   te_mon_arr_recipient_applied >= 0   \  ...and what they were last applied
#   te_mon_arr_provider_applied  >= 0    | at. BACK MULTIPLIERS — never remove
#   te_mon_arr_standing_applied  >= 0   /
#   te_mon_anchored_gdp_m   >= 0        GDP (millions) of everyone anchored to
#                                       this country. Written by the GLOBAL
#                                       monthly pass, like te_mon_world_own_den
#   te_mon_board_wrong_months 0 to 24   [5c] consecutive months a currency-board
#                                       subject's DISPLAYED band has sat at an
#                                       extreme (1 or 5). Never the true gap
#   te_mon_board_lever_applied 0 or 1   [5c] is te_mon_board_wrong_stance on?
#   te_mon_board_seigniorage_pts >= 0   [5c] an OVERLORD'S weekly minting gain:
#                                       half its board subjects' minting,
#                                       re-summed yearly
#   te_mon_board_seigniorage_applied    ...and what was last applied. BACKS A
#                                       MULTIPLIER — never remove it
#   te_mon_emergency_controls_months 0-12  the kind-1 crisis's Defend option:
#                                       capital controls forced on. Read by
#                                       te_mon_capital_controls_in_force
#   te_mon_receivership     0 or 1      [6c] this country's finances are run by
#                                       a commission somebody else appointed
#   te_mon_receiver         scope       ...and whose. Same scope-variable
#                                       contract as te_mon_anchor
#   te_mon_receivership_clear_months 0-12  [6c] consecutive months the debtor has
#                                       been under scaled_debt 0.25 and out of
#                                       default. At 12 the receivership ends
#                                       itself (114's non_fulfillment): a
#                                       sentence, not a permanent tribute
# ============================================================================


# --- §15A.1 the imported rate's spread over the anchor's --------------------
te_mon_anchor_spread_peg = { value = 0.5 }
te_mon_anchor_spread_bloc = { value = 0 }
te_mon_anchor_spread_board = { value = 0.25 }
# [6c] A peg imposed at gunpoint is a penalty rate, not a borrowed one: twice
# the chosen peg's spread (§15C.3, ruling C1). The currency-board-at-gunpoint
# shape, without the board's subject relationship to soften it.
te_mon_anchor_spread_peg_imposed = { value = 1 }

te_mon_anchor_spread = {
	value = 0
	if = {
		limit = { has_variable = te_mon_anchor_kind }
		if = {
			limit = {
				var:te_mon_anchor_kind = 1
				te_mon_is_imposed_pegger = yes
			}
			value = te_mon_anchor_spread_peg_imposed
		}
		else_if = {
			limit = { var:te_mon_anchor_kind = 1 }
			value = te_mon_anchor_spread_peg
		}
		else_if = {
			limit = { var:te_mon_anchor_kind = 3 }
			value = te_mon_anchor_spread_board
		}
		else = {
			value = te_mon_anchor_spread_bloc
		}
	}
}

# The share of the anchor's credibility an anchored country imports:
# c = max( own c , 0.8 x anchor's c ).
te_mon_anchor_credibility_share = { value = 0.8 }


# --- §15A.2 the three articles ----------------------------------------------

# currency_peg, pegger: structural premium cut, pp.
te_mon_peg_standing_cut = { value = 0.5 }

# currency_peg, anchor ("reserve currency"): -0.1pp of structural premium per
# 5% of world GDP pegged to it, capped at -0.5.
te_mon_reserve_currency_per_step = { value = 0.1 }
te_mon_reserve_currency_step_share = { value = 0.05 }
te_mon_reserve_currency_cap = { value = 0.5 }

te_mon_reserve_currency_cut = {
	value = 0
	if = {
		limit = {
			has_variable = te_mon_anchored_gdp_m
			var:te_mon_anchored_gdp_m > 0
			has_global_variable = te_mon_world_gdp_m
			global_var:te_mon_world_gdp_m > 0
		}
		value = var:te_mon_anchored_gdp_m
		divide = global_var:te_mon_world_gdp_m
		divide = te_mon_reserve_currency_step_share
		multiply = te_mon_reserve_currency_per_step
		min = 0
		max = te_mon_reserve_currency_cap
	}
}

# swap_line, recipient: peg confidence a month and the cyclical premium cut.
te_mon_swap_confidence = { value = 2 }
te_mon_swap_premium_cut = { value = 1 }
# ...and the provider's treasury draw while the recipient is in a FINANCIAL
# crisis: 0.1% of the RECIPIENT'S GDP a month. Recipient scope.
#
# [6a] THE LINE IS A LOAN (§15C.1). War left the draw trigger with ruling H2 —
# te_mon_in_financial_crisis, not te_mon_in_external_crisis — because a war is
# the one leg a player can switch on at will, and a swap line is a central
# bank's liquidity instrument, not war finance. The draw stops at the limit, it
# is repaid at the same rate once the crisis clears, and whatever is still
# outstanding when the role ends is settled in one lump. No interest: the
# recipient's own rate paid is already the price of its crisis.
te_mon_swap_crisis_draw = {
	value = gdp
	multiply = 0.001
}

# The ceiling on the balance: twenty months of draw. Recipient scope.
te_mon_swap_line_limit = {
	value = gdp
	multiply = 0.02
}

# How much of the limit is still unused. Recipient scope.
te_mon_swap_draw_room = {
	value = te_mon_swap_line_limit
	subtract = var:te_mon_swap_drawn
	min = 0
}

# This month's draw: the monthly rate, or whatever room is left. Recipient scope.
te_mon_swap_draw_this_month = {
	value = te_mon_swap_crisis_draw
	max = te_mon_swap_draw_room
	min = 0
}

# ...and this month's repayment: the same rate, or whatever is still owed.
te_mon_swap_repay = {
	value = te_mon_swap_crisis_draw
	max = var:te_mon_swap_drawn
	min = 0
}

# lender_of_last_resort, ward: this share of its §7.6 debt-load premium is
# taken off. Ward scope.
#
# [6a] THE GUARANTEE REMEMBERS (§15C.1, ruling H3). Four things now read the
# ward's own call counter, and this is one of them: the relief tapers
# 0.5 / 0.35 / 0.2 / 0 as the ward is bailed out again and again, so the
# moral-hazard loop closes on the third default. A guarantee that shrank with
# the ward's DEBT would not be a guarantee; one that shrinks with its RECORD is
# the difference between an unlucky ward and a serial one.
te_mon_lolr_relief_share_base = { value = 0.5 }

te_mon_lolr_relief_share = {
	value = te_mon_lolr_relief_share_base
	if = {
		limit = {
			has_variable = te_mon_lolr_calls
			var:te_mon_lolr_calls = 1
		}
		value = 0.35
	}
	else_if = {
		limit = {
			has_variable = te_mon_lolr_calls
			var:te_mon_lolr_calls = 2
		}
		value = 0.2
	}
	else_if = {
		limit = {
			has_variable = te_mon_lolr_calls
			var:te_mon_lolr_calls >= 3
		}
		value = 0
	}
}

te_mon_lolr_relief = {
	value = te_mon_scaled_debt_premium
	multiply = te_mon_lolr_relief_share
}

# The calls made BEFORE the one being priced right now. te_monetary_on_country_
# default increments the counter and then dispatches te_lolr.1, so everything
# that prices THIS call — the honour cost, te_lolr.1's odds, the cooldown the
# on-action itself sets — reads this, and everything that prices the ONGOING
# relationship (the relief above, the signing score) reads the counter. Ward
# scope.
te_mon_lolr_calls_prior = {
	value = 0
	if = {
		limit = { has_variable = te_mon_lolr_calls }
		value = var:te_mon_lolr_calls
		subtract = 1
		min = 0
	}
}

# ...and what honouring the guarantee costs: 5% of the WARD'S GDP, half as much
# again for every call before this one, to 15%. Ward scope.
te_mon_lolr_honour_cost_base_share = { value = 0.05 }
te_mon_lolr_honour_cost_per_call = { value = 0.5 }
te_mon_lolr_honour_cost_cap_multiple = { value = 3 }

te_mon_lolr_honour_multiplier = {
	value = 1
	add = {
		value = te_mon_lolr_calls_prior
		multiply = te_mon_lolr_honour_cost_per_call
	}
	max = te_mon_lolr_honour_cost_cap_multiple
}

te_mon_lolr_honour_cost = {
	value = gdp
	multiply = te_mon_lolr_honour_cost_base_share
	multiply = te_mon_lolr_honour_multiplier
}

te_mon_lolr_suspension_months = { value = 60 }

# The WARD'S cooldown before its next default may call the guarantee again:
# five years, and five more for every call before this one. Ward scope, read by
# te_monetary_on_country_default in the same effect that incremented the counter.
te_mon_lolr_default_cooldown_base = { value = 60 }

te_mon_lolr_default_cooldown_months = {
	value = te_mon_lolr_calls_prior
	add = 1
	multiply = te_mon_lolr_default_cooldown_base
}

# How long a quiet ward keeps a call on its record: ten years without a new one
# forgets one (step 1c's clocks), and the clock is re-armed on every call.
te_mon_lolr_call_forget_months = { value = 120 }
te_mon_lolr_calls_cap = { value = 5 }


# --- [6c] debt_receivership (§15C.3, ruling C2) -----------------------------
# The coercive counterpart to the guarantee is not a guarantee at gunpoint; it
# is externally supervised finances — Egypt's Caisse de la Dette 1876, the
# Ottoman Public Debt Administration 1881, the Dominican customs receivership
# 1905. The debtor gets the guarantee's relief, because creditors ARE reassured
# by a receiver, and pays for it every month out of the customs receipts.
#
# THE RELIEF TAPERS BY THE DEBTOR'S OWN RECORD, like a guarantee's: it is
# te_mon_lolr_relief, so a country that has already burned two guarantors is
# worth less to a receiver too. Its default under a receivership pays it
# nothing — te_monetary_on_country_default tests the guarantee, not this — so
# the counter never moves while the commission sits.
te_mon_receivership_take_share = { value = 0.001 }

# Debtor scope: what the commission remits to the receiver, a month.
te_mon_receivership_take = {
	value = gdp
	multiply = te_mon_receivership_take_share
}

# The exit ramp: twelve consecutive months worked down this far ends it.
te_mon_receivership_exit_debt = { value = 0.25 }
te_mon_receivership_exit_months = { value = 12 }


# --- "Asymmetric magnitudes, shared mechanic" (§15A.1) ----------------------
#   provider cost = recipient benefit x clamp( recipient gdp / provider gdp , 0 , 1 )
#
# Evaluated in the RECIPIENT'S scope from inside the provider's discovery, where
# ROOT is the provider and root.var:te_mon_work_gdp holds ITS GDP (cached first:
# a bare `root.gdp` is a trigger read, which a divide slot does not take).
te_mon_arr_gdp_ratio_to_root = {
	value = gdp
	divide = {
		value = root.var:te_mon_work_gdp
		min = 1
	}
	min = 0
	max = 1
}

# What one swap line costs ROOT. Recipient scope.
te_mon_arr_swap_cost_to_root = {
	value = te_mon_swap_premium_cut
	multiply = te_mon_arr_gdp_ratio_to_root
}

# What one guarantee costs ROOT. Ward scope.
te_mon_arr_lolr_cost_to_root = {
	value = te_mon_lolr_relief
	multiply = te_mon_arr_gdp_ratio_to_root
}

# [6c] ...and what one receivership costs ROOT. Debtor scope. A receiver does
# carry a share of the relief its name buys the debtor, exactly as a guarantor
# does — the difference is that it is paid for it (te_mon_receivership_take).
te_mon_arr_receivership_cost_to_root = {
	value = te_mon_lolr_relief
	multiply = te_mon_arr_gdp_ratio_to_root
}


# [5b] What a bloc's monetary-union principle hands this member (§15A.3): tier
# 1+'s cooperation — half a swap line from the leader — and, for an adopter at
# tier 3, the leader as lender of last resort. The second only where no TREATY
# guarantee already pays the same relief: two guarantors of one debt are not
# two rescues.
te_mon_union_recipient_value = {
	value = 0
	if = {
		limit = { te_mon_union_gets_cooperation = yes }
		add = te_mon_union_cooperation_cut
	}
	if = {
		limit = {
			te_mon_union_backstop_is_effective = yes
			# [6c] ...and no receivership either: two names behind one debt are
			# not two reliefs, whichever way round they were arrived at.
			te_mon_has_treaty_debt_relief = no
		}
		add = te_mon_lolr_relief
	}
}


# --- What this country RECEIVES, in pp of cyclical premium ------------------
# A reneging guarantor's wards get nothing from it while its clock runs.
te_mon_arr_recipient_value = {
	value = 0
	if = {
		limit = { te_mon_is_swap_recipient = yes }
		add = te_mon_swap_premium_cut
	}
	# [6c] A guarantee, or a receivership — never both: 112 and 114 exclude each
	# other by type, and the else_if says so anyway.
	if = {
		limit = { te_mon_lolr_is_effective = yes }
		add = te_mon_lolr_relief
	}
	else_if = {
		limit = { te_mon_is_under_receivership = yes }
		add = te_mon_lolr_relief
	}
	# [5b] tier 1+ cooperation and tier 3's backstop add here.
	add = te_mon_union_recipient_value
}

# ...and in pp of STRUCTURAL premium: the pegger's cut, the reserve currency's.
te_mon_arr_standing_value = {
	value = te_mon_reserve_currency_cut
	# [6c] The -0.5pp is the market's reward for a CHOSEN discipline, so an
	# imposed pegger does not get it (§15C.3, ruling C1). Lenders are not
	# reassured by a promise somebody else extracted.
	if = {
		limit = {
			te_mon_is_anchored = yes
			var:te_mon_anchor_kind = 1
			te_mon_is_imposed_pegger = no
		}
		add = te_mon_peg_standing_cut
	}
}

# --- §15A.4 subject currency boards (5c) ------------------------------------

# The share of a board subject's minting that goes to its overlord. The
# subject's half is te_mon_board_subject's country_minting_mult = -0.5, which
# must be restated there (a static modifier cannot read a script value).
te_mon_board_seigniorage_share = { value = 0.5 }

# What ONE board subject is worth to its overlord, a week: that share of the
# subject's flat minting adds. Subject scope. §17 check 21 is whether this
# `modifier:` read works at all from script; if it reads 0 the overlord simply
# gains nothing, and the fallback is to count gold-mine levels.
#
# BE HONEST ABOUT THE SIZE (§15A.4): vanilla's base minting is a flat 500 and
# the real money is gold-mine production methods, so for a colony without gold
# both sides of this are rounding errors. It is a GOLD-COLONY lever.
te_mon_board_seigniorage_from_subject = {
	value = modifier:country_minting_add
	multiply = te_mon_board_seigniorage_share
	min = 0
}

# The wrong-stance lever: months at an extreme band before liberty desire moves.
te_mon_board_wrong_months_threshold = { value = 6 }
te_mon_board_wrong_months_cap = { value = 24 }
