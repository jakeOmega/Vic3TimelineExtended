# ============================================================================
# COVERT WARFARE — Script Values
# ============================================================================
# These script values drive the Covert Warfare system.
# Operations are tracked as diplomatic pacts (togglable like increase_relations).
# Active operations are detected via any_scope_diplomatic_pact checks.
#
# INTELLIGENCE CAPACITY uses modifier hooks applied directly to techs, laws,
# and institutions via country_intelligence_capacity_add.
# ============================================================================

# ============================================================================
# SECTION 1: TUNING CONSTANTS
# ============================================================================

# Intelligence Capacity component scaling
intelligence_capacity_literacy_max = 50
intelligence_capacity_gdp_max = 25

# Operation slots: base/rank via INJECT:base_values and INJECT:country_ranks
covert_ops_max_slot_cap = 10

# Detection probability ranges (base % chance per month)
covert_ops_detection_base = 10

# Funding level detection reductions (cumulative: -3 at level 2, -8 at level 3, -13 at level 4, -19 at level 5)
covert_ops_detection_funding_2_reduction = 3
covert_ops_detection_funding_3_reduction = 5
covert_ops_detection_funding_4_reduction = 5
covert_ops_detection_funding_5_reduction = 6

# Per-operation roll scale, applied inside covert_operation_detection_chance
# (so it shows in the widget). Every running operation rolls its own iw_detect
# each month, so with N operations the chance that at least one burns is
# 1 - prod(1 - c_i). 1 = honest per-operation odds; lower it (0.6 puts three
# 10% operations near the old single-roll 18%) if N-operation risk lands hot.
covert_ops_detection_multi_op_scale = 1

# The lowest monthly detection risk any operation can carry. Full funding
# subtracts more than the base risk on its own, so a well-funded agency
# against a target with no counterintelligence sits exactly here: 0.1%/month
# is 1.2% a year, which is "almost with impunity" without ever being "safe".
# Fractions are real rolls, not a silent zero — vanilla's Montenegro raiding
# journal entry passes computed chances of 0.15 and 0.45 to `random`.
covert_ops_detection_floor = 0.1

# Counterintelligence IC bonus per funding level above 1 (applied via iw_funding_defense modifier)
covert_ops_funding_defense_ic_per_level = 5

# Target counterintelligence: detection penalty from target IC advantage
covert_ops_target_ic_factor = 15
covert_ops_target_ic_max = 20

# ---- Exposure blowback ----
# What being caught costs, before the phase factor and the option's split.
# Which tier a type is in lives in covert_code_tier_* (scripted triggers).
# Anchored on the old flat costs at the moderate tier's establishing phase:
# Acknowledge (2 infamy / -10 relations) lands close to the old flat
# 2 infamy / -15 relations; Deny (1 infamy / -20 relations) is a materially
# softer relations hit than the old flat 1 infamy / -30, not "about there."
covert_exposure_infamy_mild = 1
covert_exposure_infamy_moderate = 2
covert_exposure_infamy_severe = 4
covert_exposure_infamy_war = 0

# Caught in the run-up instead of during the war: sabotage before the shooting
# reads as manufacturing the war, so it is charged at the severe tier's infamy
# and the moderate tier's relations. Named separately from those tiers so this
# case can be retuned without moving either of them.
covert_exposure_infamy_war_prewar = 4
covert_exposure_relations_war_prewar = -20

covert_exposure_relations_mild = -10
covert_exposure_relations_moderate = -20
covert_exposure_relations_severe = -40

# How far along the operation was. A network barely laid costs half; one that
# has been running a year costs half again as much.
covert_exposure_phase_mult_preparatory = 0.5
covert_exposure_phase_mult_establishing = 1
covert_exposure_phase_mult_operational = 1.5

# The two options' split. Owning it costs the infamy but keeps the bilateral
# damage down; denying it halves the infamy and takes the full relations hit.
covert_exposure_acknowledge_relations_mult = 0.5
covert_exposure_deny_infamy_mult = 0.5

# What a severe exposure costs with everyone tied to the target. The
# tooltip prints it through covert_exposure_third_party_relations_display
# below rather than retyping the number, so retuning this line retunes the
# text with it.
covert_exposure_third_party_relations = -10

# The same number as a positive magnitude, for covert_exposure_third_party_tt
# ("...loses N relations with us"). Derived, not a second constant: the loc
# string would otherwise have to render "loses -10".
covert_exposure_third_party_relations_display = {
	value = covert_exposure_third_party_relations
	multiply = -1
	round = yes
}

# Guardrail, matching te_force_become_subject's own infamy cap.
covert_exposure_infamy_max = 25

# Maximum funding level (0 = dormant, max = deep state)
iw_funding_level_max = 5

# ---- Per-operation priority (slice 3) ----
# An operation's priority (iw_priority on its container, 1 when missing)
# multiplies what it does, what it costs and how exposed it is. Cost outruns
# effect at every step: that is the diminishing return that stops "everything
# at 3" from being the answer.
covert_op_priority_max = 3
covert_op_priority_2_effect_mult = 1.35
covert_op_priority_3_effect_mult = 1.6
# Each operation's share of the covert budget (priority 1 = one share).
covert_op_priority_2_cost_mult = 1.6
covert_op_priority_3_cost_mult = 2.4
# Raw detection points per priority level above 1, added before covert
# efficiency (so late-game tradecraft still hides most of it).
covert_op_priority_detect_add = 2
# A fully operational operation works at this multiple of an establishing one.
# Replaces the literal `multiplier = 2` the apply sites used to carry.
covert_op_phase_full_mult = 2

# ---- Per-target networks (slice 4) ----
# An operator's standing presence inside one target country, 0 to
# covert_net_max. It grows while operations run against the target and
# decays when none do. Gains shrink above half strength (the UN-standing
# idiom); losses do not.
covert_net_max = 100
covert_net_dr_divisor = 50
covert_net_dr_floor = 0.1
# Monthly gain with one operation running; each extra operation against the
# same target adds covert_net_extra_op_factor of it (2 ops x1.5, 3 ops x2,
# counted up to 3).
covert_net_gain_base = 2
covert_net_extra_op_factor = 0.5
# Monthly decay with no operations running. Funding at or above
# covert_net_maintain_funding_level keeps the handlers paid, halving it.
covert_net_decay_base = 1.5
covert_net_maintain_funding_level = 3
covert_net_maintain_decay_mult = 0.5
# Being caught rolls up part of the network (about a year of single-op work).
covert_net_burn_loss = 25
# Retaliation in kind (covert_warfare.2 option b, #430): the defender turns
# the exposure into a network inside the perpetrator's country, worth what the
# burn cost the perpetrator's own network — one country's loss, the other's
# foothold. Through covert_net_gain, so it adds less above half strength.
covert_net_retaliation_seed = 25
# Raw detection points removed per point of strength (-5 at full strength),
# before covert efficiency like the rest of the raw score.
covert_net_detect_factor = 0.05
# Months of head start a new operation gets per point of strength, rounded.
# The cap stays below the 6-month establishing threshold, so phase 1 is never
# skipped: a full network puts a new operation one month from establishing.
covert_net_head_start_per_point = 0.05
covert_net_head_start_max = 5
# ---- Agency experience: Tradecraft (slice 5) ----
# A country's covert service's accumulated experience, 0 to
# covert_tradecraft_max, in the country variable iw_tradecraft. It grows while
# operations run past the preparatory phase, falls when one is burned and
# drifts down while the agency sits idle. Gains shrink above half (the
# UN-standing idiom); losses do not. Writers: covert_tradecraft_gain /
# covert_tradecraft_loss (covert_warfare_effects.txt, § TRADECRAFT).
covert_tradecraft_max = 100
# Monthly gain per operation at establishing phase or better, counting at most
# covert_tradecraft_ops_counted operations' worth.
covert_tradecraft_gain_per_op = 1
covert_tradecraft_ops_counted = 4
# A preparatory operation counts as this fraction of one. Without it a burned
# operation, relaunched preparatory, taught nothing for six months while still
# rolling detection, and at 10% detection one moderate operation averaged ~31
# instead of the ~57 it does with 0.5 (§ TRADECRAFT table).
covert_tradecraft_prep_gain_fraction = 0.5
covert_tradecraft_dr_divisor = 50
covert_tradecraft_dr_floor = 0.1
# What a burn costs, by the exposed operation's tier. Unscaled. Net of these,
# at 10% monthly detection a moderate operation still earns about +0.4 a
# month, a severe one +0.1 and a mild one +0.7 (spec § Slice 5).
covert_tradecraft_loss_mild = 3
covert_tradecraft_loss_moderate = 6
covert_tradecraft_loss_severe = 9
covert_tradecraft_loss_war = 3
# Monthly drift while no operation runs at all.
covert_tradecraft_decay = 0.25
# Tier floors: Untested < 20 <= Fledgling < 40 <= Established < 60 <= Seasoned
# < 80 <= Veteran. Unlocks: covert_tradecraft_unlocks_* (scripted triggers).
covert_tradecraft_tier_1_floor = 20
covert_tradecraft_tier_2_floor = 40
covert_tradecraft_tier_3_floor = 60
covert_tradecraft_tier_4_floor = 80
# Networks grow this much faster per tier (x1.6 at Veteran). The intelligence
# capacity bonus per tier is the iw_tradecraft_bonus static modifier itself.
covert_tradecraft_net_gain_per_tier = 0.15
# ---- New operations (slice 6) ----
# Regime change: once fully operational, a monthly push on any coup already
# under way in the target (je_ip4_coup's bar runs 10 -> 120 and moves weekly
# by commander strength minus coup resistance). Vanilla's coup events nudge it
# by 15; this is a steady third of that. Most of the operation's pull on a
# coup comes through coup resistance, not this push (plan table:
# docs/superpowers/plans/2026-09-23-covert-new-operations.md).
covert_regime_change_coup_push = 5
# Cultivate assets: a network with one running against its target grows this
# much faster. Stacks with the extra-operation factor and with Tradecraft.
# Alone against a target: strength 50 in ~17 months (25 for any other
# operation), 75 in ~28 (42).
covert_net_cultivate_mult = 1.5
# The AI proposes cultivate assets only where its network is weaker than this.
covert_cultivate_ai_net_ceiling = 50
# ---- Network intelligence (slice 7) ----
# A network this strong reports on the service it has penetrated: from tier 1
# the target's intelligence capacity and technology count, from tier 2 also how
# many covert operations the target runs against the operator. Display only;
# nothing in script reads the report back. One operation reaches tier 1 in ~25
# months and tier 2 in ~42; cultivate assets alone in ~17 and ~28. A burn (-25)
# from tier 2 drops the network straight back to tier 1.
covert_net_intel_tier_1_strength = 50
covert_net_intel_tier_2_strength = 75

	min = 0

# Operation slots: entirely from modifier country_covert_operation_slot_add.
# Base (1 via INJECT:base_values), rank (GP +2, Major +1 via country_ranks),
# and tech/law bonuses all feed into a single modifier breakdown.
covert_operation_max_slots = {
	value = modifier:country_covert_operation_slot_add
	min = 0
	max = covert_ops_max_slot_cap
}

# Maximum operations of the same type that can run simultaneously
# 1 base + 1 with mainframe_computers + 1 with cyber_warfare + 1 for a Veteran
# agency (slice 5). Only gates launching: an operation over the cap after the
# score falls keeps running.
covert_ops_max_per_type = {
	value = 1
	if = { limit = { has_technology_researched = mainframe_computers } add = 1 }
	if = { limit = { has_technology_researched = cyber_warfare } add = 1 }
	if = { limit = { covert_tradecraft_unlocks_extra_per_type = yes } add = 1 }
}

# Currently active operations — exact pact count using step pattern
# (each "count >= N" check adds 1, giving the precise total)
# IMPORTANT: first_country filter ensures we only count ops WE initiated, not ops targeting us
covert_operations_active = {
	value = 0
	if = { limit = { any_scope_diplomatic_pact = { count >= 1 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 2 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 3 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 4 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 5 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 6 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 7 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 8 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 9 is_covert_operation_pact = yes first_country = prev } } add = 1 }
	if = { limit = { any_scope_diplomatic_pact = { count >= 10 is_covert_operation_pact = yes first_country = prev } } add = 1 }
}

covert_operations_available_slots = {
	value = 0
	owner = {
		add = covert_operation_max_slots
		subtract = covert_operations_active
	}
	min = 0
}

# Operations as the upkeep counts them: the sum of every operation's
# priority-weighted share (iw_priority_cost_sum, covert_refresh_priority_cost).
# A save made before slice 3 has no sum until the first refresh, and gets the
# live pact count — exactly the old formula — in the meantime.
covert_priority_cost_units = {
	value = 0
	if = {
		limit = { has_variable = iw_priority_cost_sum }
		add = var:iw_priority_cost_sum
	}
	else = {
		add = covert_operations_active
	}
}

covert_operations_cost_scale = {
	value = gdp
	multiply = 0.00005 # small, but multiplied by funding level and number of ops, so it adds up for big economies
}

# Funding cost: multiplier for the covert_operation_funding_cost modifier
# Scales with the priority-weighted operation count + 1 (base maintenance) and funding level
covert_operation_cost_mult = {
	value = 0
	owner = {
		add = covert_priority_cost_units
		add = 1 # base maintenance — you pay for defensive benefits even with no ops
		# Scale by funding level (0-5); 0 means no cost
		multiply = {
			value = 0
			if = {
				limit = { has_variable = iw_funding_level }
				add = var:iw_funding_level
			}
		}
		# Scale with GDP so big economies pay proportionally
		multiply = covert_operations_cost_scale
	}
	min = 0
}

# Detection chance per month for active operations
# Factors: base risk, funding reduction, target counterintelligence (IC + type defense), operation priority, attacker efficiency
# Formula: raw = base - funding_reduction - network + target_defense_penalty + priority_penalty, final = raw × (1 - efficiency), clamped [covert_ops_detection_floor, 50]
# var:target_max_ic and var:target_type_defense must be set before evaluating.
covert_operation_detection_chance = {
	value = covert_ops_detection_base
	# Higher funding reduces detection risk (better tradecraft)
	if = {
		limit = {
			has_variable = iw_funding_level
			var:iw_funding_level >= 2
		}
		subtract = covert_ops_detection_funding_2_reduction
	}
	if = {
		limit = {
			has_variable = iw_funding_level
			var:iw_funding_level >= 3
		}
		subtract = covert_ops_detection_funding_3_reduction
	}
	if = {
		limit = {
			has_variable = iw_funding_level
			var:iw_funding_level >= 4
		}
		subtract = covert_ops_detection_funding_4_reduction
	}
	if = {
		limit = {
			has_variable = iw_funding_level
			var:iw_funding_level >= 5
		}
		subtract = covert_ops_detection_funding_5_reduction
	}
	# A network inside the target hides the operation (slice 4). Staged per
	# operation by covert_op_refresh_detection; 0 without a network.
	subtract = covert_net_detect_reduction
	# Target's counterintelligence increases detection risk (IC + type defense vs your IC)
	add = target_counterintelligence_penalty
	# Priority: a busier operation is more exposed (before efficiency, so
	# better tradecraft still hides most of it)
	add = covert_op_priority_detect_penalty
	# Attacker's efficiency reduces detection: multiply by (1 - eff_mult), floor 0.2
	multiply = {
		value = 1
		subtract = modifier:country_covert_operation_efficiency_mult
		min = 0.2
	}
	# Per-operation roll scale (Section 1) applied here so the widget row and
	# the roll (covert_op_roll_chance) read the exact same number.
	multiply = covert_ops_detection_multi_op_scale
	min = covert_ops_detection_floor
	max = 50
}

# Target counterintelligence: penalty based on target's total defense vs yours.
# Total defense = target IC + target type-specific defense modifier.
# var:target_max_ic and var:target_type_defense are staged per operation by covert_op_refresh_detection.
target_counterintelligence_penalty = {
	value = 0
	if = {
		limit = { has_variable = target_max_ic }
		add = var:target_max_ic
		# Add type-specific defense
		if = {
			limit = { has_variable = target_type_defense }
			add = var:target_type_defense
		}
		divide = {
			value = intelligence_capacity_total
			min = 1
		}
		subtract = 1
		multiply = covert_ops_target_ic_factor
		min = 0
		max = covert_ops_target_ic_max
	}
}

# A higher-priority operation is busier and more exposed: covert_op_priority_detect_add
# raw points per level above 1. var:iw_priority_staging is staged per operation
# by covert_op_refresh_detection; missing reads as priority 1.
# Scope: operator country
covert_op_priority_detect_penalty = {
	value = 0
	if = {
		limit = { has_variable = iw_priority_staging }
		add = var:iw_priority_staging
		subtract = 1
		multiply = covert_op_priority_detect_add
		min = 0
	}
}

# The chance one operation rolls against this month. iw_detect is written on
# the container by covert_op_refresh_detection at the top of the same pulse
# (covert_ops_sync_all), so it is never a month stale here, and the
# covert_ops_detection_multi_op_scale is already folded into it by
# covert_operation_detection_chance — the widget row and the roll agree by
# construction, so the harness's pin to iw_detect = 100 is a true 100% roll.
# A named script value is the proven form for `random = { chance = ... }`
# (space_race_effects.txt); `chance = var:` is not.
# Scope: covert operation container
covert_op_roll_chance = {
	value = var:iw_detect
	min = 0
	max = 100
}

# Equilibrium: gains diminish and losses do not, so the score levels off.

#   per-op detection   ops              average        months to 40
#   10%                1 moderate       57 / 69        131 / 81

#   10%                4 moderate       67 / 74         31 / 18
#   10%                1 severe         25 / 44        208 / 138
#    5%                1 moderate       82 / 85         62 / 51
#    5%                1 severe         73 / 77         74 / 57
#    2%                1 moderate       94 / 95         47 / 45

# Severe operations stay hard to learn from at high detection, on purpose.
