nd_exchange_quiet_days = {
	value = 0
	if = {
		limit = { has_variable = nd_exchange_last_launch }
		value = game_date
		subtract = var:nd_exchange_last_launch
	}
}

# AI weights for the stand-down (nuclear_incident.70). This side's struck
# states: each one a reason to refuse.
nd_exchange_states_struck = {
	value = 0
	every_scope_state = {
		limit = {
			OR = {
				has_modifier = nuclear_strike_aftermath
				has_modifier = tactical_nuke_aftermath
			}
		}
		add = 1
	}
	max = 5
}

# What a TE_NUCLEAR observer record prints for its IDs and crisis figures
# (nuclear_observer_effects.txt). nd_log_context copies them into these global
# variables just before the record: a ScriptValue read inside debug_log can't
# see a saved scope and evaluates against ROOT, while a global reads the same
# from any scope (scripting_best_practices.md § "`debug_log` Loc-String
# Templating Limitations").
nd_log_exchange_id = {
	value = 0
	if = {
		limit = { has_global_variable = nd_log_exchange_id_now }
		value = global_var:nd_log_exchange_id_now
	}
}
