# AI weights for the stand-down (nuclear_incident.70). This side's struck
# states: each one a reason to refuse.
nd_exchange_states_struck = {
	value = 0
	every_scope_state = {
		limit = {
			OR = {
				has_modifier = nuclear_strike_aftermath
				has_modifier = tactical_nuke_aftermath
			}
		}
		add = 1
	}
	max = 5
}

# What the arsenal watch's TE_NUCLEAR lines print about the country building or
# holding warheads (type=build_burst, build_cap, ai_stockpile_cap,
# arsenal_milestone; 2026-11-07, after France reached some 1,200 warheads in an
# observer game with nothing logged). Read with THIS.ScriptValue from the
# entry's weekly pulse, where the country is ROOT. Country scope.
nd_log_is_ai = {
	value = 0
	if = {
		limit = { is_ai = yes }
		value = 1
	}
}
nd_log_at_war = {
	value = 0
	if = {
		limit = { is_at_war = yes }
		value = 1
	}
}
nd_log_progress_mult = {
	value = modifier:country_nuclear_program_progress_mult
}
# Weekly innovation as the funding buttons' ai_chance reads it, in their bands
# (country_innovation is a trigger): the band's floor.
nd_log_innovation = {
	value = 0
	if = {
		limit = { country_innovation >= 2000 }
		value = 2000
	}
	else_if = {
		limit = { country_innovation >= 1000 }
		value = 1000
	}
	else_if = {
		limit = { country_innovation >= 500 }
		value = 500
	}
	else_if = {
		limit = { country_innovation >= 300 }
		value = 300
	}
	else_if = {
		limit = { country_innovation >= 200 }
		value = 200
	}
	else_if = {
		limit = { country_innovation >= 150 }
		value = 150
	}
	else_if = {
		limit = { country_innovation > 100 }
		value = 100
	}
}
# 1 while a rival with warheads is believed to hold more than our true count:
# the funding buttons' race term (+20 to the increase, -20 to the decrease).
nd_log_rival_bigger = {
	value = 0
	if = {
		limit = {
			has_variable = nuclear_weapon_stockpile
			any_rival_country = {
				has_modifier = nuclear_power
				has_variable = nd_public_estimate
				var:nd_public_estimate > ROOT.var:nuclear_weapon_stockpile
			}
		}
		value = 1
	}
}
