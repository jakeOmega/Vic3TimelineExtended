# ============================================================================
# GLOBAL WARMING — display-only script values
# ============================================================================
# Read by gui/journal_entry_widgets/global_warming_widget.gui, by
# common/customizable_localization/global_warming_custom_loc.txt and by
# je_global_warming's status/reason loc. Nothing here has a gameplay effect.
#
# WHY THIS FILE EXISTS: every value below is O(1). The figures the journal entry
# used to show were computed live, in loc, every frame the panel was open --
# live market/world values now include state and household pop sweeps. All UI
# readers, including the emissions map and treaty text, use stored snapshots.
# The heavy work happens once a year per market leader and once a month for
# the policy adoption counts; these
# readers only fetch what was stored.
#
# EVERY READER IS GUARDED. The journal entry has `is_shown_when_inactive`, so
# these are evaluated in the scope of countries that have never run a pulse, and
# on saves upgraded mid-game. An unguarded `var:`/`global_var:` read there is the
# "Failed to fetch variable" spam that gotcha #14 describes.
#
# FRESHNESS: the emissions figures are re-based each January (see
# gw_rebase_annual_emissions_effect). `gw_has_yearly_figures` is what the widget
# uses to show "figures update each January" instead of a misleading 0.
# ============================================================================

# ----------------------------------------------------------------------------
# Emissions — snapshots retain internal units; formatted amounts scale ×1000
# ----------------------------------------------------------------------------

# This country's MARKET's annual emissions. Stored on the market leader, because
# the sweep that produces it is market-wide: a member reads its leader's copy.
# There is no per-country emissions figure in this system -- emissions are a
# property of net building emissions plus household heating and atmospheric removal.
# Scope: country
gw_market_emis_raw = {
	value = 0
	market_capital ?= {
		owner ?= {
			if = {
				limit = { has_variable = gw_disp_market_emis }
				add = var:gw_disp_market_emis
			}
		}
	}
}

# Formatted emission amount only; raw readers above retain internal units.
gw_market_emis_display = {
	value = gw_market_emis_raw
	multiply = gw_emission_display_scale
}

# The negative part of the above: atmospheric carbon removed by Direct Air Capture.
# Scope: country
gw_capture_display = {
	value = 0
	market_capital ?= {
		owner ?= {
			if = {
				limit = { has_variable = gw_disp_capture }
				add = var:gw_disp_capture
			}
		}
	}
}

# World annual emissions, measured as the change in the cumulative global
# variable over the last calendar year rather than re-summed from every market.
# Scope: any
gw_global_emis_raw = {
	value = 0
	if = {
		limit = { has_global_variable = gw_g_global_emis }
		add = global_var:gw_g_global_emis
	}
}

# Formatted emission amount only; raw readers above retain internal units.
gw_global_emis_display = {
	value = gw_global_emis_raw
	multiply = gw_emission_display_scale
}

# This market's share of world annual emissions, in per cent. Clamped to 0-100
# so the unsigned history chart has a fixed, honest scale; a market that is a
# net carbon sink reads 0 here and shows its negative figure in the emissions
# row instead.
# Scope: country
gw_share_pct_display = {
	value = 0
	if = {
		limit = {
			has_global_variable = gw_g_global_emis
			global_var:gw_g_global_emis > 0
		}
		add = gw_market_emis_raw
		divide = global_var:gw_g_global_emis
		multiply = 100
	}
	min = 0
	max = 100
}

# Temperature change implied by the last completed year of world emissions.
# temperature_anomaly_display is cumulative/10000, so one year of emissions
# moves it by that year's emissions / 10000.
# Scope: any
gw_yearly_change_display = {
	value = gw_global_emis_raw
	divide = 10000
}

# Unsigned magnitude, for loc that supplies its own sign and colour through
# gw_trend_prefix (customizable localization).
# Scope: any
gw_yearly_change_abs_display = {
	value = gw_yearly_change_display
	if = {
		limit = { gw_yearly_change_display < 0 }
		multiply = -1
	}
}

# ----------------------------------------------------------------------------
# The overview (gw_disp_*): codes and 0-1 fractions for the widget's icons,
# bar and pies (gui/journal_entry_widgets/global_warming_widget.gui,
# te_gw_overview_panel). Display only, O(1), no variable of their own.
# Scope: country (read as JournalEntry.GetCountry.MakeScope.ScriptValue)
# ----------------------------------------------------------------------------

# THE WARMING LADDER'S ONE DEFINITION. The tier as a code, 0 (negligible) to 6
# (apocalyptic). gw_severity_text / gw_severity_short
# (common/customizable_localization/global_warming_custom_loc.txt) compare this
# code, and the overview picks its tier icon by it, so the word, the sentence
# and the icon cannot drift. The thresholds also match the threshold events in
# global_warming_events_on_action (0.5, 1, 2, 3, 4 C) and the journal entry's
# 4 C goal: change them there too, or not at all.
gw_disp_tier_code = {
	value = 0
	if = {
		limit = { temperature_anomaly_display >= 4.0 }
		value = 6
	}
	else_if = {
		limit = { temperature_anomaly_display >= 3.0 }
		value = 5
	}
	else_if = {
		limit = { temperature_anomaly_display >= 2.0 }
		value = 4
	}
	else_if = {
		limit = { temperature_anomaly_display >= 1.0 }
		value = 3
	}
	else_if = {
		limit = { temperature_anomaly_display >= 0.5 }
		value = 2
	}
	else_if = {
		limit = { temperature_anomaly_display >= 0.1 }
		value = 1
	}
}

# The temperature at which the next tier begins: the marker on the overview's
# bar. The ladder above, one step up (test_global_warming_layout.py holds the
# two tables equal). At the top tier there is no next one; it reads 4 and the
# marker is hidden. The codes are compared with < and >, the forms the
# file's other script-value triggers use, rather than with =.
gw_disp_next_threshold = {
	value = 4
	if = {
		limit = { gw_disp_tier_code < 1 }
		value = 0.1
	}
	else_if = {
		limit = { gw_disp_tier_code < 2 }
		value = 0.5
	}
	else_if = {
		limit = { gw_disp_tier_code < 3 }
		value = 1.0
	}
	else_if = {
		limit = { gw_disp_tier_code < 4 }
		value = 2.0
	}
	else_if = {
		limit = { gw_disp_tier_code < 5 }
		value = 3.0
	}
}

# The bar's scale is the journal entry's own: 0 to its 4 C goal.
gw_disp_temp_frac = {
	value = temperature_anomaly_display
	divide = 4
	min = 0
	max = 1
}

gw_disp_next_frac = {
	value = gw_disp_next_threshold
	divide = 4
	min = 0
	max = 1
}

# Direction of last year's warming: 1 warming, -1 cooling, 0 steady (or no
# yearly figure yet). The same test gw_trend_prefix colours the rate with.
gw_disp_temp_trend = {
	value = 0
	if = {
		limit = { gw_yearly_change_display > 0 }
		value = 1
	}
	else_if = {
		limit = { gw_yearly_change_display < 0 }
		value = -1
	}
}

# How far ahead the overview's headline projects: "1.61 -> 1.81 C" is the
# reading now and in this many years, if every year warms as much as the last
# (the owner, 2026-09-29: temperature has no resting point to head towards, so
# a projection stands in for the UN's "-> target"). Ten years rather than one:
# at a typical +0.02 C a year, one year's step would round away in the
# headline's two decimals. The tooltip says what the projection assumes.
gw_disp_projection_years = {
	value = 10
}

# The reading gw_disp_projection_years from now at last year's rate. Equal to
# the reading until the first yearly figure exists (the rate reads 0), which is
# when the headline shows "(rate in January)" instead of an arrow.
gw_disp_temp_projected = {
	value = gw_yearly_change_display
	multiply = gw_disp_projection_years
	add = temperature_anomaly_display
}

# The overview bar's layers (play-test round 3; the layer list is in
# te_gw_overview_panel's comment). The projection on the bar's 0-4 C scale,
# then the lower and the higher of reading and projection. The solid fill runs
# to the lower and a translucent layer under it to the higher. Rising, the
# solid part ends at the reading and a red stretch (warming is bad) runs on to
# the projection; falling, the solid part ends at the projection and a green
# stretch (cooling is good) runs on to the reading. Falling is gw_disp_temp_trend < 0, which is exactly when the
# projection (the reading plus ten years of last year's change) is lower.
gw_disp_proj_frac = {
	value = gw_disp_temp_projected
	divide = 4
	min = 0
	max = 1
}

gw_disp_bar_low_frac = {
	value = gw_disp_temp_frac
	if = {
		limit = { gw_disp_temp_trend < 0 }
		value = gw_disp_proj_frac
	}
}

gw_disp_bar_high_frac = {
	value = gw_disp_proj_frac
	if = {
		limit = { gw_disp_temp_trend < 0 }
		value = gw_disp_temp_frac
	}
}

# 1 while the next threshold is being approached: warming, and below the top
# tier. Picks the bar tooltip's "N years away" line; otherwise it says no
# threshold is being approached at last year's rate.
gw_disp_heading_code = {
	value = 0
	if = {
		limit = {
			gw_disp_temp_trend > 0
			gw_disp_tier_code < 6
		}
		value = 1
	}
}

# Years to the next threshold at last year's rate, for the bar's tooltip. 0
# unless gw_disp_heading_code is 1, where the rate is positive. Capped, so a
# year of near-zero emissions cannot print a five-digit wait.
gw_disp_years_to_next = {
	value = 0
	if = {
		limit = { gw_disp_heading_code > 0 }
		value = gw_disp_next_threshold
		subtract = temperature_anomaly_display
		divide = gw_yearly_change_display
	}
	min = 0
	max = 999
}
