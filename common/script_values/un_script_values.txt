# 2/3 supermajority threshold of currently-eligible voting members.
# Used for expulsion and charter-reform votes: the vote passes when
# un_res_support * 3 >= count * 2.
un_vote_eligible_member_count = {
	value = 0
	every_country = {
		limit = {
			je:je_united_nations ?= { has_modifier = un_member_modifier }
			# Article 19 (redesign phase 5): a member that has lost its vote
			# for arrears is not counted.
			NOT = { un_dues_vote_suspended = yes }
		}
		add = 1
	}
}

# Supermajority pass check (expulsion, charter reform): positive when
# un_res_support * 3 >= count * 2, i.e., 2/3 supermajority achieved. Trigger
# with: un_vote_supermajority_passed >= 0.
# Scope: resolution container (read by un_resolution_decide)
un_vote_supermajority_passed = {
	value = var:un_res_support
	multiply = 3
	# Subtracted twice on purpose — that is the `- count * 2` of the 2/3 test,
	# written as two steps rather than a nested `subtract = { multiply = 2 }`.
	subtract = un_vote_eligible_member_count
	subtract = un_vote_eligible_member_count
}

# Vote margin of a resolution: yes votes minus no votes. A simple-majority
# resolution passes at >= 1.
# Scope: resolution container
un_res_margin = {
	value = var:un_res_support
	subtract = var:un_res_oppose
}

# Closed resolutions kept in un_resolution_history. Past this, the oldest
# entry whose topic cooldown has run out is destroyed.
# Also written as the literal 26 (cap + 1) in un_resolution_archive — change both.
un_resolution_history_cap = 25

# ---- Resolution lobbying ----

# How many vote commitments one resolution's proposer may secure, ever. Counted
# against un_res_commit_count on the resolution container, which only ever goes
# up — a commitment that has already been settled still occupies a slot, so the
# cap cannot be recycled by lobbying somebody who then votes early.
#
# Deliberately tiny. Two commitments cannot carry a resolution on their own in
# an Assembly of any size; they are there to swing a vote the proposer has
# already nearly won, at the price of two standing diplomatic obligations.
un_lobby_commitment_cap = 2

# Eviction order for un_resolution_history: highest for the oldest resolution.
# Scope: resolution container
un_resolution_age_order = {
	value = 0
	subtract = var:un_res_seq
}

# Display order for the UN chamber's history list: highest for the newest
# resolution, so `ordered_in_global_list = { order_by = un_res_recency_order }`
# walks un_resolution_history newest-first. The inverse of
# un_resolution_age_order; read-only, used by un_chamber_history_sgui.
# Scope: resolution container
un_res_recency_order = {
	value = var:un_res_seq
}

# ---- Authorized Military Mandates ----
# See common/scripted_effects/un_mandate_effects.txt for the container model.

# How long an authorization stays usable before it lapses, in months of the
# global UN pulse. Five years: long enough to raise the play and fight it,
# short enough that a mandate cannot be banked indefinitely.
un_mandate_expiry_months = 60

# Closed mandates kept in un_mandate_registry for the chamber's record. Past
# this, the oldest closed entry is destroyed.
# Also written as the literal 13 (cap + 1) in un_mandate_prune_registry — change both.
un_mandate_registry_cap = 12

# How long a closed mandate stays on the record before it is destroyed, even
# under the cap.
un_mandate_closed_retention_months = 60

# Months of authorization left. Clamped at 0 so a mandate the sweep has not
# reached yet never displays a negative figure.
# Scope: mandate container
un_mandate_months_left = {
	value = un_mandate_expiry_months
	subtract = var:un_mnd_months
	min = 0
}

# Display order for the chamber's mandate list: highest for the newest
# mandate, so `ordered_in_global_list = { order_by = un_mandate_recency_order }`
# walks un_mandate_registry newest-first.
# Scope: mandate container
un_mandate_recency_order = {
	value = var:un_mnd_seq
}

# Eviction order for un_mandate_registry: highest for the oldest mandate.
# The inverse of un_mandate_recency_order.
# Scope: mandate container
un_mandate_age_order = {
	value = 0
	subtract = var:un_mnd_seq
}

# Which candidate target the propose button tables a mandate against, when
# more than one qualifies. Prefers the state the Assembly has already censured
# over one merely notorious, and a rival over a stranger, and breaks ties
# toward the weaker state — the Assembly authorizes the winnable restoration,
# not the biggest war available.
# Scope: candidate target country; ROOT = the prospective proposer
un_mandate_case_score = {
	value = 0
	if = { limit = { has_modifier = un_condemned_modifier } add = 40 }
	if = { limit = { has_modifier = un_sanctions_target_modifier } add = 40 }
	if = { limit = { has_modifier = un_sanctions_voluntary_target_modifier } add = 20 }
	if = { limit = { has_modifier = un_non_binding_rebuke_modifier } add = 20 }
	if = {
		limit = { has_diplomatic_pact = { who = ROOT type = rivalry } }
		add = 20
	}
	subtract = country_rank
}
