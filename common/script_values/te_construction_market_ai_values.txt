te_construction_market_ai_purchase_target = {
	if = {
		limit = {
			is_construction_paused = no
			construction_queue_num_queued_government_levels > 0
		}
		value = net_fixed_income
		min = 0
		divide = var:te_construction_market_price_stabilizer
		multiply = {
			add = scaled_gold_reserves
			subtract = scaled_debt
			add = 1

			# Bankruptcy-avoidance brake: when reserves are thin under shock
			# (war or financial-cycle stress), cut spending PROPORTIONALLY
			# rather than waiting for debt to spike. Linear sliding scale —
			# avoids cliff effects that would cascade economic shocks.
			#
			# scaled_gold_reserves runs ~0 (empty) to ~1 (well-stocked).
			# Subtracting (target - reserves) when reserves < target eats
			# into the multiplier; the higher the target (war/banking stress),
			# the deeper the cut while reserves are thin.

			# Wartime: target adequate reserves of ~0.5 scaled
			if = {
				limit = {
					is_at_war = yes
					scaled_gold_reserves < 0.5
				}
				subtract = {
					value = 0.5
					subtract = scaled_gold_reserves
				}
			}

			# Banking cycle stress (high bubble pressure or active crash):
			# target ~0.3 scaled — modest extra caution
			if = {
				limit = {
					has_variable = bubble_pressure
					var:bubble_pressure >= 50
					scaled_gold_reserves < 0.3
				}
				subtract = {
					value = 0.3
					subtract = scaled_gold_reserves
				}
			}

			# Floor at 0.1× — never fully halt; AI should keep some momentum
			min = 0.1
		}
		min = {
			value = 10
			if = {
				limit = {
					bureaucracy < 0
				}
				add = {
					value = income
					divide = {
						value = te_construction_market_price
						min = 50
						max = 150
					}
					divide = 5
				}
			}
		}
		round_to = 1
		max = {
			value = construction_queue_num_queued_government_levels
			multiply = modifier:country_max_weekly_construction_progress_add
		}
	}
	else = {
		value = 0
	}
}

# ai_value of the Construction Sector in a market game (building_construction_sector,
# common/buildings/te_construction_market_site.txt). Scope: the state the AI is
# considering building in.
#
# A production building scores its ai_value plus profit terms, and in play the AI
# underbuilt the sector: its scoring doesn't seem to credit how cheap a level is.
# A level costs construction_cost_very_low (100 points) against 600 to 800 for the
# base game's factories and 2,000 to 5,000 for the mod's heavy industry, so while
# its margin is positive it returns far more per point spent than anything else
# the AI could build. TE_CONSTRUCTION_SECTOR_AI_VALUE_BASE makes up for that; an
# ordinary factory scores 1,000 and Tooling Workshops 2,000.
#
# The need for more sectors is read from the construction good's price in the
# state's market, not from the queue. The government's purchase and the
# investors' are both divided by the price, so a queue lengthens as the price
# rises and already follows it, and a count of queued levels also follows the
# country's size. Linear in the price premium: the base at base price, up
# TE_CONSTRUCTION_SECTOR_AI_VALUE_PER_PRICIER per 100% above it (about 15,000 at
# the ceiling, above the base game's monuments at 10,000), down
# TE_CONSTRUCTION_SECTOR_AI_VALUE_PER_CHEAPER per 100% below it (about 1,000, an
# ordinary factory, at the floor). Which of market_goods_pricier and
# market_goods_cheaper is clamped at 0 is undocumented, so each is floored at 0
# (scripting_best_practices.md, "Is it signed below base price?").
TE_CONSTRUCTION_SECTOR_AI_VALUE_BASE = 5000
TE_CONSTRUCTION_SECTOR_AI_VALUE_PER_PRICIER = 10000
TE_CONSTRUCTION_SECTOR_AI_VALUE_PER_CHEAPER = 4000

te_construction_sector_ai_value = {
	value = TE_CONSTRUCTION_SECTOR_AI_VALUE_BASE
	add = {
		value = 0
		market = {
			mg:construction = {
				add = market_goods_pricier
			}
		}
		min = 0
		multiply = TE_CONSTRUCTION_SECTOR_AI_VALUE_PER_PRICIER
	}
	subtract = {
		value = 0
		market = {
			mg:construction = {
				add = market_goods_cheaper
			}
		}
		min = 0
		multiply = TE_CONSTRUCTION_SECTOR_AI_VALUE_PER_CHEAPER
	}
	# At war: deprioritize construction expansion (focus existing capacity on war effort)
	if = {
		limit = {
			owner ?= {
				is_at_war = yes
			}
		}
		subtract = 500
	}
	# Active banking-cycle stress: don't add construction during a financial crunch
	if = {
		limit = {
			owner ?= {
				has_variable = bubble_pressure
				var:bubble_pressure >= 60
			}
		}
		subtract = 500
	}
}
