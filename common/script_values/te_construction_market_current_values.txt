# ----------------------------------------------------------------------------
# Railway train method, for the Transport-principle retooling waiver
# (te_railway_note_train_method, common/scripted_effects/extra_effects.txt).
# ----------------------------------------------------------------------------
# The state's railway train method as a code: steam 1, electric 2, diesel 3,
# plus 10 for the Transport principle tier 3 variant of the same method
# (pm_*_trains_principle_transport_3); anything else 0. A state holds at most
# one railway, which is why this can be read, and recorded, at state scope:
# buildings do not reliably hold variables (scripting_best_practices.md).
# Scope: state
te_state_railway_train_code = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_railway }
		if = {
			limit = { has_active_production_method = pm_steam_trains }
			add = 1
		}
		else_if = {
			limit = { has_active_production_method = pm_electric_trains }
			add = 2
		}
		else_if = {
			limit = { has_active_production_method = pm_diesel_trains }
			add = 3
		}
		else_if = {
			limit = { has_active_production_method = pm_steam_trains_principle_transport_3 }
			add = 11
		}
		else_if = {
			limit = { has_active_production_method = pm_electric_trains_principle_transport_3 }
			add = 12
		}
		else_if = {
			limit = { has_active_production_method = pm_diesel_trains_principle_transport_3 }
			add = 13
		}
	}
}

# Current code minus the recorded one (the code itself when nothing is
# recorded yet). +10 / -10 means the same train method moved into / out of its
# tier 3 variant.
# Scope: state
te_state_railway_train_code_change = {
	value = te_state_railway_train_code
	if = {
		limit = { has_variable = te_rail_train_code }
		subtract = var:te_rail_train_code
	}
}
