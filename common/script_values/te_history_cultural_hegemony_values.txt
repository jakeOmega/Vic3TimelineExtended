# Global annual Cultural Hegemony history display. No country-local state.
ch_model_hist_has_data = {
	value = 0
	if = {
		limit = { has_global_variable = ch_model_hist_last_y }
		value = 1
	}
}
ch_model_hist_since_year = {
	value = 0
	if = {
		limit = { has_global_variable = ch_model_hist_since_y }
		value = global_var:ch_model_hist_since_y
	}
}

	value = 0
	if = {
		limit = { has_global_variable = ch_model_hist_last_y }
		value = global_var:ch_model_hist_last_y
	}
}
