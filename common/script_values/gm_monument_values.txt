# ============================================================================
# GRAND MONUMENTS — several monuments in one state: script values
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-13-multiple-monuments-per-state-design.md.
# One building_grand_monument stands in a state, at the level of everything
# standing there (the engine allows one). The PRIMARY monument is the one the
# state's own variables describe (gm_skin, gm_form, gm_name, ...); every other
# monument is a SECONDARY, a gm_monument container (gm_monument_effects.txt)
# in the state's gm_monuments_here, with its levels in gm_tr_levels.
# gm_state_grandeur (gm_values.txt) is the building's level, so the state's
# whole; gm_state_primary_grandeur is the primary's own part of it.

# Container scope. The levels a secondary monument stands for in the building.
gm_mon_levels = {
	value = 0
	if = {
		limit = { has_variable = gm_tr_levels }
		value = var:gm_tr_levels
	}
}

# State scope. The levels of the secondaries standing here (the monthly pass
# writes the list), 0 with none: the rebuilds' arithmetic.
gm_state_secondary_levels = {
	value = 0
	if = {
		limit = { has_variable_list = gm_monuments_here }
		every_in_list = {
			variable = gm_monuments_here
			add = gm_mon_levels
		}
	}
}

# State scope. What the building holds for the primary: its level less the
# secondaries', at least 1; 0 with no building. Read only where the building
# itself is the question (an undedicated primary's size, building beside, the
# merge arithmetic); sites spec 2026-10-18 §3.3.
gm_state_built_levels = {
	value = gm_state_level
	subtract = gm_state_secondary_levels
	min = 1
	if = {
		limit = { NOT = { has_building = building_grand_monument } }
		value = 0
	}
}

# State scope. The primary's own levels: its record (gm_levels) once it has
# one, else what the building holds for it; 0 with no building. Every reader
# that means "what was built" (rebuild heights, trophy shares, Rededicate's
# cost, the ceremony's level) reads this, never gm_state_primary_grandeur.
gm_state_primary_levels = {
	value = gm_state_built_levels
	if = {
		limit = { has_variable = gm_levels }
		value = var:gm_levels
	}
	if = {
		limit = { NOT = { has_building = building_grand_monument } }
		value = 0
	}
}

# State scope. The primary's starting grandeur (sites spec 2026-10-18 §3.2):
# its record, else its levels, one a level (an older save's monument, which
# was counted so); 0 with no building.
gm_state_primary_start = {
	value = gm_state_primary_levels
	if = {
		limit = { has_variable = gm_start_grandeur }
		value = var:gm_start_grandeur
	}
	if = {
		limit = { NOT = { has_building = building_grand_monument } }
		value = 0
	}
}

# Container scope. The grandeur its years, events and gifts have added to its
# levels (spec 2026-10-16 §1), or 0: an older save's monument has none.
gm_mon_age = {
	value = 0
	if = {
		limit = { has_variable = gm_tr_age }
		value = var:gm_tr_age
	}
}

# Container scope. Its starting grandeur: the record, else its levels, one a
# level (an older save's container, which was counted so).
gm_mon_start = {
	value = gm_mon_levels
	if = {
		limit = { has_variable = gm_tr_start_grandeur }
		value = var:gm_tr_start_grandeur
	}
}

# Container scope. What a month adds to a secondary's age grandeur: 2% of its
# starting grandeur a year, until a century's worth is reached
# (gm_state_age_step). No monument policy changes it (policy spec 2026-10-23).
gm_mon_age_step = {
	value = gm_mon_start
	multiply = 2
	divide = 1200
	max = gm_mon_age_room
}

# Container scope. The age grandeur a secondary may still gain from its years
# (gm_state_age_room).
gm_mon_age_room = {
	value = gm_mon_start
	multiply = gm_age_cap_years
	multiply = 2
	divide = 100
	subtract = gm_mon_age
	min = 0
}

# Container scope. The grandeur a secondary monument stands for: its starting
# grandeur and its age.
gm_mon_grandeur = {
	value = gm_mon_start
	add = gm_mon_age
}

# Container scope. What each of a contested secondary's three choices does
# now (GM-4), for their tooltips: the figures gm_preview_choices stored at the
# last refresh (gm_effects.txt), guarded; and Rededicate's halved grandeur.
gm_mon_disp_prev_td_legit = {
	value = 0
	if = {
		limit = { has_variable = gm_prev_td_legit }
		value = var:gm_prev_td_legit
	}
}

gm_mon_disp_prev_td_base = {
	value = 0
	if = {
		limit = { has_variable = gm_prev_td_base }
		value = var:gm_prev_td_base
	}
}

gm_mon_disp_prev_td_sup = {
	value = 0
	if = {
		limit = { has_variable = gm_prev_td_sup }
		value = var:gm_prev_td_sup
	}
}

gm_mon_disp_prev_rd_sup = {
	value = 0
	if = {
		limit = { has_variable = gm_prev_rd_sup }
		value = var:gm_prev_rd_sup
	}
}

gm_mon_disp_prev_rd_grandeur = {
	value = gm_mon_grandeur
	divide = 2
}

# State scope. The grandeur standing here in secondary monuments, 0 with none.
gm_state_secondary_grandeur = {
	value = 0
	if = {
		limit = { has_variable_list = gm_monuments_here }
		every_in_list = {
			variable = gm_monuments_here
			add = gm_mon_grandeur
		}
	}
}

# State scope. The primary's age grandeur (gm_age), 0 with none or no building.
gm_state_primary_age = {
	value = 0
	if = {
		limit = {
			has_building = building_grand_monument
			has_variable = gm_age
		}
		value = var:gm_age
	}
}

# State scope. What the primary monument stands for: its starting grandeur and
# its age; 0 with no building. Every reader that means "this monument's own
# grandeur" (the dedication's national sums, the IG nets, the card) reads
# this, so a returned artifact adds to the state's standing grandeur without
# being counted as the primary's dedication. Level arithmetic reads
# gm_state_primary_levels instead.
gm_state_primary_grandeur = {
	value = gm_state_primary_start
	add = gm_state_primary_age
}

# State scope. How many monuments stand here: the primary, its secondaries and
# a construction site (sites spec 2026-10-18 §5: a site counts toward six).
gm_state_monument_count = {
	value = 0
	if = {
		limit = { has_building = building_grand_monument }
		add = 1
	}
	if = {
		limit = { has_building = building_grand_monument_site }
		add = 1
	}
	if = {
		limit = { has_variable_list = gm_monuments_here }
		every_in_list = {
			variable = gm_monuments_here
			add = 1
		}
	}
}

# ---- Looting one monument of a state (spec §4.2) -----------------------------------
# Container scope, a secondary monument. The share of its levels an occupier
# carries off, by form: 1 for an object, 0.25 for a piece of a structure, 0.1
# for a token (gm_state_tr_share, gm_trophy_values.txt, on gm_tr_form).
gm_mon_tr_share = {
	value = 0.1
	if = {
		limit = { gm_mon_form_is_piece = yes }
		value = 0.25
	}
	if = {
		limit = { gm_mon_form_is_object = yes }
		value = 1
	}
}

# The levels a taking carries off: the share, rounded up, at least one and at
# most every level.
gm_mon_tr_levels = {
	value = gm_mon_levels
	multiply = gm_mon_tr_share
	ceiling = yes
	min = 1
	max = gm_mon_levels
}

# Container scope. The starting grandeur a taking carries off with its levels:
# all of it whole, else the levels' share of it.
gm_mon_tr_start = {
	value = gm_mon_start
	if = {
		limit = { gm_mon_tr_whole = 0 }
		multiply = gm_mon_tr_levels
		divide = gm_mon_levels
	}
}

# Container scope. The age a taking carries off with its levels: all of it
# whole, else the levels' share of it.
gm_mon_tr_age = {
	value = gm_mon_age
	if = {
		limit = { gm_mon_tr_whole = 0 }
		multiply = gm_mon_tr_levels
		divide = gm_mon_levels
	}
}

# 1 when a taking carries off the whole monument.
gm_mon_tr_whole = {
	value = 0
	if = {
		limit = { gm_mon_tr_levels >= gm_mon_levels }
		value = 1
	}
}

# State scope. In percent: the chance that the AI's one roll at a state falls on
# a secondary monument, every monument standing here being as likely.
gm_state_tr_ai_secondary_chance = {
	value = gm_state_monument_count
	subtract = 1
	multiply = 100
	divide = gm_state_monument_count
}

# Container scope. A secondary's status for its row (ageing spec 2026-10-16
# §4.3), gm_status_code's codes: 2 upheld, 3 heritage, 4 contested.
gm_mon_status_code = {
	value = 2
	if = {
		limit = { gm_mon_is_heritage = yes }
		value = 3
	}
	if = {
		limit = { gm_mon_is_contested = yes }
		value = 4
	}
}

# Container scope. What an anniversary's public option adds to a secondary's
# age (gm_evt_grandeur's twin): a tenth of its starting grandeur, at least 1.
gm_mon_evt_grandeur = {
	value = gm_mon_start
	multiply = 0.1
	min = 1
}

# ---- The state view (spec §5.1) ---------------------------------------------------
# State scope. Secondary monuments standing here (the list the monthly pass and
# every return or taking keep), 0 with none: the Also Standing Here card shows
# while it is above 0.
gm_state_mon_disp_here = {
	value = 0
	if = {
		limit = { gm_state_has_secondary = yes }
		every_in_list = {
			variable = gm_monuments_here
			add = 1
		}
	}
}

# What the merge's log lines and self-check print (gm_sh_log_numbers): copied
# into global variables first, as nd_log_context does, because a ScriptValue
# read inside debug_log evaluates against ROOT and cannot see a saved scope
# (scripting_best_practices.md § "`debug_log` Loc-String Templating
# Limitations"). The orphan's recorded levels, the most levels any building in
# its region holds unaccounted for, and how many buildings stand there.
gm_sh_log_levels = {
	value = 0
	if = {
		limit = { has_global_variable = gm_sh_log_levels_now }
		value = global_var:gm_sh_log_levels_now
	}
}

gm_sh_log_buildings = {
	value = 0
	if = {
		limit = { has_global_variable = gm_sh_log_buildings_now }
		value = global_var:gm_sh_log_buildings_now
	}
}


# ---- Clear Away (owner, 2026-10-31) -------------------------------------------------
# Once six monuments stand in a state (gm_state_monument_count, the gate on a new
# construction site), its owner may clear away the least grand of them to make
# room for a grander one. Contested monuments have their own three choices and
# are left out.

# Container scope. The order ordered_in_list takes the least grand first by
# (it sorts the highest first).
gm_mon_grandeur_neg = {
	value = gm_mon_grandeur
	multiply = -1
}

# State scope. The least grandeur among the monuments here that may be cleared
# away (gm_state_primary_clear_eligible, gm_mon_clear_eligible); 0 with none.
gm_state_least_grandeur = {
	value = 0
	if = {
		limit = { gm_state_has_clear_eligible_secondary = yes }
		ordered_in_list = {
			variable = gm_monuments_here
			limit = { gm_mon_clear_eligible = yes }
			order_by = gm_mon_grandeur_neg
			position = 0
			check_range_bounds = no
			add = gm_mon_grandeur
		}
		if = {
			limit = { gm_state_primary_clear_eligible = yes }
			max = gm_state_primary_grandeur
		}
	}
	else_if = {
		limit = { gm_state_primary_clear_eligible = yes }
		value = gm_state_primary_grandeur
	}
}

# Container scope. How far a monument beside another stands above the least
# grand in its state: 0 for the least.
gm_mon_least_gap = {
	value = gm_mon_grandeur
	if = {
		limit = { exists = var:gm_tr_origin }
		var:gm_tr_origin = { subtract = gm_state_least_grandeur }
	}
}

# State scope, for the state card's and the journal entry's Clear Away button on
# the state's own monument: 1 while six stand here and it is the least grand.
gm_state_disp_clearable = {
	value = 0
	if = {
		limit = {
			gm_state_monument_count >= 6
			gm_state_primary_is_least = yes
		}
		value = 1
	}
}

# Container scope, for the same button on a monument beside it.
gm_mon_disp_clearable = {
	value = 0
	if = {
		limit = {
			gm_mon_is_least = yes
			var:gm_tr_origin ?= { gm_state_monument_count >= 6 }
		}
		value = 1
	}
}
