# The sample: how many law starts the cohort must make before the verdict, and how many the control must make for
# its share to count.
te_alw_chk_sample = 40
te_alw_chk_control_sample = 20

te_alw_chk_starts_g = { value = global_var:te_alw_starts }
te_alw_chk_hits_g = { value = global_var:te_alw_hits }
te_alw_chk_ps_g = { value = global_var:te_alw_hits_ps }
te_alw_chk_cps_g = { value = global_var:te_alw_hits_cps }
te_alw_chk_rcl_g = { value = global_var:te_alw_hits_rcl }
te_alw_chk_pro_starts_g = { value = global_var:te_alw_pro_starts }
te_alw_chk_pro_hits_g = { value = global_var:te_alw_pro_hits }
te_alw_chk_ctrl_starts_g = { value = global_var:te_alw_ctrl_starts }
te_alw_chk_ctrl_hits_g = { value = global_var:te_alw_ctrl_hits }

# The share of each group's law starts that went to one of the three laws (0 with none counted).
te_alw_chk_share = {
	value = 0
	if = {
		limit = { global_var:te_alw_starts > 0 }
		value = global_var:te_alw_hits
		divide = global_var:te_alw_starts
	}
}
te_alw_chk_ctrl_share = {
	value = 0
	if = {
		limit = { global_var:te_alw_ctrl_starts > 0 }
		value = global_var:te_alw_ctrl_hits
		divide = global_var:te_alw_ctrl_starts
	}
}
# Twice the control's share, which OK must reach when the control has its sample.
te_alw_chk_ctrl_share_twice = {
	value = te_alw_chk_ctrl_share
	multiply = 2
}
