# Percentage (0-100) of this country's population living in pops at wealth 25
# or more: the pops whose Healthcare need (popneed_healthcare) buys a real
# amount of Drugs. Gates drugs_shortage.1, so a Drugs shortage in a country of
# poor pops is not told as a medicine shortage. Country scope.
drugs_healthcare_population_share = {
	value = 0
	every_scope_pop = {
		limit = { wealth >= 25 }
		add = total_size
	}
	multiply = 100
	divide = {
		value = total_population
		min = 1
	}
}
