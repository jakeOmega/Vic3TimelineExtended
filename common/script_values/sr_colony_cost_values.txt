# ============================================================================
# SOLAR SYSTEM COLONIZATION — the colonies' costs on the Space Program
# ============================================================================
# Country scope. Five costs of holding and serving colonies, each a flat amount
# of Launch Capacity a week a unit, taken as more Launch Capacity input on the
# Space Program and nothing else (owner, 2026-11-25). Until then each was a
# share of the programme's Launch Capacity at base (its active methods' Launch
# Capacity at full staff, sr_emig_lc_base), so a garrison cost ten times as
# much while the programme ran the Interstellar Mission as while it ran Solar
# Colonization (owner, 2026-11-25: "why does it cost more if I am sending an
# interstellar probe out?"). Each amount is its old share of Solar
# Colonization's 10,000. Before that each was Space Program throughput, which
# raised the programme's outputs with its inputs: a module level's upkeep raised
# fabrication, which built more modules, which raised the upkeep.
#
# The modifiers (space_race_modifiers.txt, "the colonies' costs") are on the
# Space Program building, each a goods_input_launch_capacity_mult of 0.01 a
# unit, added at multiplier = 100 x its Launch Capacity a week / the building's
# Launch Capacity as the engine bills it (sr_emig_bill_divisor,
# sr_colony_population_values.txt), from the one refresh site sr_ccost_refresh
# (sr_colony_cost_effects.txt). The engine bills a building's Launch Capacity as
# base x (1 + throughput) x (1 + the Launch Capacity input mults), so a unit
# takes its amount whatever the mission and the funding, as the settlers' bill
# does (sr_settler_launches). test_colony_costs.py holds the amounts to the
# modifiers.
# ============================================================================

# The Space Program's throughput as the engine multiplies a Launch Capacity
# input mult by: 1 plus the country's Space Program throughput (funding, the
# milestone rewards). State-scope throughput (devastation, riots) and the
# global building_throughput_add are not divided out, as in the settlers' bill
# (sr_emig_bill_divisor, which reads this). Floored at 0.1: a factor of 0 or
# below would divide by nothing.
sr_program_throughput_factor = {
	value = modifier:building_space_program_throughput_add
	add = 1
	min = 0.1
}

# ---- Each cost's Launch Capacity a week a unit ------------------------------------
# For the modifiers' multipliers, the AI's budget and the panel.
sr_ccost_upkeep_lc = { value = 100 }
sr_ccost_supply_lc = { value = 300 }
sr_ccost_garrison_lc = { value = 200 }
sr_ccost_strike_lc = { value = 500 }
sr_ccost_breach_lc = { value = 1000 }

# The strike's and the breach's terms, in months (space_race_events.96, .97).
sr_strike_settlement_term = 36
sr_dome_breach_term = 12

# ---- Each cost's units ---------------------------------------------------------
# The three the entry declared until 2026-11-25, with its gates: the rule at
# simple or full colonies, and the entry active, which a colony holder's always
# is (its possible) and a supplier's must be (sr_free_can_supply).
# The module levels past the first at the colonies held (colony modules design
# § 8).
sr_ccost_upkeep_units = {
	value = 0
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_journal_entry = je_space_race_solar_colonization
			sr_mod_upkeep_levels >= 1
		}
		add = sr_mod_upkeep_levels
	}
}
# A unit a free colony supplied (slice 5 design § 3).
sr_ccost_supply_units = {
	value = 0
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_journal_entry = je_space_race_solar_colonization
			sr_free_supply_count >= 1
		}
		add = sr_free_supply_count
	}
}
# A unit a garrisoned colony (slice 5 design § 2).
sr_ccost_garrison_units = {
	value = 0
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_journal_entry = je_space_race_solar_colonization
			sr_garrison_count >= 1
		}
		add = sr_garrison_count
	}
}
# A unit while the strike's or the breach's term runs. The counter is removed
# at 0 (sr_ccost_refresh), so having it is the term.
sr_ccost_strike_units = {
	value = 0
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_variable = sr_strike_settlement_months
		}
		add = 1
	}
}
sr_ccost_breach_units = {
	value = 0
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_variable = sr_dome_breach_months
		}
		add = 1
	}
}

# ---- Each modifier's multiplier on the building --------------------------------
# 100 x its Launch Capacity a week / the building's Launch Capacity as billed: a
# unit of the modifier is 1% of that (sr_emig_bill_units_value, the same).
sr_ccost_upkeep_mult = {
	value = sr_ccost_upkeep_units
	multiply = sr_ccost_upkeep_lc
	multiply = 100
	divide = sr_emig_bill_divisor
}
sr_ccost_supply_mult = {
	value = sr_ccost_supply_units
	multiply = sr_ccost_supply_lc
	multiply = 100
	divide = sr_emig_bill_divisor
}
sr_ccost_garrison_mult = {
	value = sr_ccost_garrison_units
	multiply = sr_ccost_garrison_lc
	multiply = 100
	divide = sr_emig_bill_divisor
}
sr_ccost_strike_mult = {
	value = sr_ccost_strike_units
	multiply = sr_ccost_strike_lc
	multiply = 100
	divide = sr_emig_bill_divisor
}
sr_ccost_breach_mult = {
	value = sr_ccost_breach_units
	multiply = sr_ccost_breach_lc
	multiply = 100
	divide = sr_emig_bill_divisor
}

# ---- The Launch Capacity in all ---------------------------------------------------
# The Launch Capacity a week the five take together. sr_ccost_refresh stores it
# as var:sr_ccost_lc, which the AI's mission budget adds to every mission's bill
# (sr_ai_mission_bill_*): whatever the mission, the program buys it too.
sr_ccost_lc_value = {
	value = 0
	add = {
		value = sr_ccost_upkeep_units
		multiply = sr_ccost_upkeep_lc
	}
	add = {
		value = sr_ccost_supply_units
		multiply = sr_ccost_supply_lc
	}
	add = {
		value = sr_ccost_garrison_units
		multiply = sr_ccost_garrison_lc
	}
	add = {
		value = sr_ccost_strike_units
		multiply = sr_ccost_strike_lc
	}
	add = {
		value = sr_ccost_breach_units
		multiply = sr_ccost_breach_lc
	}
}
# The Launch Capacity as last billed: 0 before the first refresh, and with
# nothing to pay.
sr_ccost_lc_now = {
	value = 0
	if = {
		limit = { has_variable = sr_ccost_lc }
		value = var:sr_ccost_lc
	}
}
