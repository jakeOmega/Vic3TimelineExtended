# ============================================================================
# GRAND MONUMENTS — unique dedications (signatures): script values
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-16-grand-monument-signatures-design.md.
# The sums are written by gm_sig_compute_sums (gm_signature_effects.txt); these
# read them for the panels, every read guarded (a save older than signatures
# has none of these variables until its first refresh).

# ---- Per-step values (§2, §12) ----------------------------------------------------
# Mirrors of the static modifiers in gm_modifiers.txt, each of which carries one
# step (the multiplier is the step count). The loc reads these; the registry
# test pins every pair equal (SIGNATURES).
gm_step_sig_antiquity = 1
gm_step_sig_archive = 2
gm_step_sig_archive_spread = 0.02
gm_step_sig_throne = 0.03
gm_step_sig_royal_pantheon = 0.05
gm_step_sig_martial_glory = 0.05
gm_step_sig_liberty = 0.03
gm_step_sig_grande_armee = 0.03
gm_step_sig_mandate_of_heaven = 1
gm_step_sig_glory_of_athens = 0.02
gm_step_sig_to_victory = 0.02
gm_step_sig_unknown_soldier = -0.05
gm_step_sig_concert_of_nations = 0.02
gm_step_sig_conquerors_of_space = 0.02
gm_step_sig_olympic = -0.01
gm_step_sig_heroes = 0.05

gm_step_sig_local_antiquity = 0.05
gm_step_sig_local_archive = 0.1
gm_step_sig_local_first_emperor = 0.10

# ---- National Effects (country scope) ----------------------------------------------
# Per signature with a national effect: what it gives (steps x step), the
# grandeur behind it, where its next step completes and how many monuments
# carry it, then the row's bar: (2G - N + 5) / (N + 5), as gm_disp_frac_* in
# gm_values.txt on the curve with first step 5. Ancestral Antiquity's is on the
# 1/n curve with no end (gm_sig_set_steps_lin_open): (G - N + W) / W.
gm_display_sig_antiquity = {
	value = 0
	if = {
		limit = { has_variable = gm_s_sig_antiquity }
		value = var:gm_s_sig_antiquity
		multiply = gm_step_sig_antiquity
	}
}

gm_display_g_sig_antiquity = {
	value = 0
	if = {
		limit = { has_variable = gm_g_sig_antiquity }
		value = var:gm_g_sig_antiquity
	}
}

gm_display_c_sig_antiquity = {

		limit = { has_variable = gm_c_sig_antiquity }
		value = var:gm_c_sig_antiquity
	}
}

gm_disp_frac_sig_antiquity = {
	value = 0
	if = {
		limit = {
			has_variable = gm_g_sig_antiquity
			has_variable = gm_n_sig_antiquity
			has_variable = gm_w_sig_antiquity
		}
		value = var:gm_g_sig_antiquity
		subtract = var:gm_n_sig_antiquity
		add = var:gm_w_sig_antiquity
		divide = var:gm_w_sig_antiquity
	}
	min = 0
	max = 1
}
