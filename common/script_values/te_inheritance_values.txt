# ----------------------------------------------------------------------
#  The concentration score (country scope)
#  te_inh_concentration, 0-100: how much of the nation's property belongs
#  to a few great fortunes. Each state keeps its own score (te_dg_wc),
#  closing 3% of the gap to its own target a year (half in about 23 years,
#  one generation); the national figure is their average weighted by where
#  property is held (te_demog_wc_national, te_demog_wealth_effects.txt). The
#  law and its amendments are one term of a state's target
#  (te_demog_wc_target, te_demog_values.txt). A country with no scored
#  state yet keeps this file's own drift (te_inh_yearly_update).
# ----------------------------------------------------------------------

# The law's own target, before amendments. Customary Inheritance (and a
# country with no inheritance law) is the 50 baseline.
te_inh_law_target = {
	value = 50
	if = {
		limit = { has_law = law_type:law_primogeniture }
		value = 80
	}
	else_if = {
		limit = { has_law = law_type:law_free_testation }
		value = 60
	}
	else_if = {
		limit = { has_law = law_type:law_equal_inheritance }
		value = 25
	}
	else_if = {
		limit = {
			OR = {
				has_law = law_type:law_state_universal_heir
				has_law = law_type:law_non_inheritable_usage_rights
			}
		}
		value = 0
	}
}

# The law's target with amendments, 0-100: the law and amendment term of
# each state's target (te_demog_wc_target), centred on 50.
te_inh_law_amendment_target = {
	value = te_inh_law_target
	if = {
		limit = {
			active_law:lawgroup_inheritance ?= {
				has_amendment = amendment_type:amendment_estate_duties
			}
		}
		add = -25
	}
	if = {
		limit = {
			active_law:lawgroup_inheritance ?= {
				has_amendment = amendment_type:amendment_birthright_endowment
			}
		}
		add = -30
	}
	if = {
		limit = {
			active_law:lawgroup_inheritance ?= {
				has_amendment = amendment_type:amendment_perpetual_trusts
			}
		}
		add = 30
	}
	if = {
		limit = {
			active_law:lawgroup_inheritance ?= {
				has_amendment = amendment_type:amendment_undivided_farm_succession
			}
		}
		add = 10
	}
	min = 0
	max = 100
}

# The target the law tooltip and the concept show, and the drift's for a
# country with no scored state: the states' ownership-weighted target once
# they have scores (te_demog_wc_national); until then the law's own.
# The law (with its amendments) and the taxes on wealth move every state's
# target alike, so a change since the last national refresh shows at once:
# the stored target moves by the difference between the term now and the
# term te_demog_wc_national summed (the states' own drift reads the live
# target at their pulses). Recognised continuity of title
# (inheritance_events.6) pins it at the current score: nobody's estate
# passes on.
te_inh_concentration_target = {
	value = te_inh_law_amendment_target
	if = {
		limit = { has_variable = te_dg_wc_target }
		value = var:te_dg_wc_target
		if = {
			limit = { has_variable = te_dg_wc_t_law }
			add = te_demog_wc_law_term
			subtract = var:te_dg_wc_t_law
		}
		if = {
			limit = { has_variable = te_dg_wc_t_tax }
			add = te_demog_wc_tax_term
			subtract = var:te_dg_wc_t_tax
		}
	}
	min = 0
	max = 100
	if = {
		limit = {
			has_variable = te_inh_title_continuity
			has_variable = te_inh_concentration
		}
		value = var:te_inh_concentration
	}
}

# Next year's score, for a country with no scored state: 3% of the gap to
# the target closes.
te_inh_concentration_next = {
	value = te_inh_concentration_target
	subtract = var:te_inh_concentration
	multiply = 0.03
	add = var:te_inh_concentration
	min = 0
	max = 100
}

# inh_great_fortunes' multiplier: (score - 50) / 50 above 50, else 0.
te_inh_great_fortunes_mult_value = {
	value = var:te_inh_concentration
	subtract = 50
	divide = 50
	min = 0
	max = 1
}

# inh_dispersed_wealth's multiplier: (50 - score) / 50 below 50, else 0.
te_inh_dispersed_mult_value = {
	value = 50
	subtract = var:te_inh_concentration
	divide = 50
	min = 0
	max = 1
}

# Display (law tooltips, the concept, the modifiers' descriptions). Safe
# before the first yearly pulse has set the score: the target stands in.
te_inh_concentration_display = {
	value = te_inh_concentration_target
	if = {
		limit = { has_variable = te_inh_concentration }
		value = var:te_inh_concentration
	}
}
