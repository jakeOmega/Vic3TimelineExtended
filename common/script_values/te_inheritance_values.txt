# ----------------------------------------------------------------------
#  The concentration score (country scope)
#  te_inh_concentration, 0-100: how much of the nation's property belongs
#  to a few great fortunes. Each state keeps its own score (te_dg_wc),
#  closing 3% of the gap to its own target a year (half in about 23 years,
#  one generation); the national figure is their average weighted by where
#  property is held (te_demog_wc_national, te_demog_wealth_effects.txt). The
#  law and its amendments are one term of a state's target
#  (te_demog_wc_target, te_demog_values.txt). A country with no scored
#  state yet keeps this file's own drift (te_inh_yearly_update).
# ----------------------------------------------------------------------

# The law's target with amendments, 0-100: the law and amendment term of
# each state's target (te_demog_wc_target), centred on 50. Each law and
# amendment carries its line (country_wealth_concentration_inheritance_add,
# 2026-10-21, so its tooltip says it): Primogeniture & Entail +30, Freedom of
# Testation +10, Customary Inheritance (and a country with no inheritance
# law) none, Forced Heirship -25, State as Universal Heir and Possession by
# Use -50; Estate Duties -30, the Birthright Endowment -20, Perpetual
# Dynastic Trusts +30, Undivided Farm Succession +10. The clamp holds Forced
# Heirship with Estate Duties at 0.
te_inh_law_amendment_target = {
	value = 50
	add = modifier:country_wealth_concentration_inheritance_add
	min = 0
	max = 100
}

# The target the law tooltip and the concept show, and the drift's for a
# country with no scored state: the states' ownership-weighted target once
# they have scores (te_demog_wc_national); until then the law's own.
# The law (with its amendments), the taxes on wealth, the economic laws, the public goods laws and the slavery law
# move every state's target alike, so a change since the last national
# refresh shows at once: the stored target moves by the difference between
# the term now and the term te_demog_wc_national summed (the states' own drift reads the live
# target at their pulses). Recognised continuity of title
# (inheritance_events.6) pins it at the current score: nobody's estate
# passes on.
te_inh_concentration_target = {
	value = te_inh_law_amendment_target
	if = {
		limit = { has_variable = te_dg_wc_target }
		value = var:te_dg_wc_target
		if = {
			limit = { has_variable = te_dg_wc_t_law }
			add = te_demog_wc_law_term
			subtract = var:te_dg_wc_t_law
		}
		if = {
			limit = { has_variable = te_dg_wc_t_tax }
			add = te_demog_wc_tax_term
			subtract = var:te_dg_wc_t_tax
		}
		if = {
			limit = { has_variable = te_dg_wc_t_econ }
			add = te_demog_wc_econ_term
			subtract = var:te_dg_wc_t_econ
		}
		if = {
			limit = { has_variable = te_dg_wc_t_public }
			add = te_demog_wc_public_term
			subtract = var:te_dg_wc_t_public
		}
		if = {
			limit = { has_variable = te_dg_wc_t_slavery }
			add = te_demog_wc_slavery_term
			subtract = var:te_dg_wc_t_slavery
		}
	}
	min = 0
	max = 100
	if = {
		limit = {
			has_variable = te_inh_title_continuity
			has_variable = te_inh_concentration
		}
		value = var:te_inh_concentration
	}
}

# Next year's score, for a country with no scored state: 3% of the gap to
# the target closes.
te_inh_concentration_next = {
	value = te_inh_concentration_target
	subtract = var:te_inh_concentration
	multiply = 0.03
	add = var:te_inh_concentration
	min = 0
	max = 100
}

# The modifiers' strengths (docs/superpowers/specs/2026-10-22-demographic-effects-impact-design.md
# § The curve): full strength (1) at a score of 80 or 20, a reference, not a
# ceiling. The high side is a straight line on to 5/3 at 100; the low side
# keeps half the slope past 20, to 4/3 at 0: (x + min(x, 1)) / 2, where x is
# the reach past 50 in te_inh_k_wc_span points. Python twin:
# scripts/analysis/demographics_effects.py (wc_high, wc_low).
te_inh_k_wc_span = { value = 30 }

# The country modifiers' multipliers, from the national figure.
# inh_great_fortunes': (score - 50) / 30 above 50, else 0.
te_inh_great_fortunes_mult_value = {
	value = var:te_inh_concentration
	subtract = 50
	divide = te_inh_k_wc_span
	min = 0
}

# inh_dispersed_wealth's: the reach below 50, then half the slope past 20.
te_inh_dispersed_reach = {
	value = 50
	subtract = var:te_inh_concentration
	divide = te_inh_k_wc_span
	min = 0
}

te_inh_dispersed_mult_value = {
	value = te_inh_dispersed_reach
	add = {
		value = te_inh_dispersed_reach
		max = 1
	}
	divide = 2
}

# State scope (te_inh_refresh_wc_state_effects): the state modifiers'
# multipliers, from the state's own score, on the same curve.
# inh_land_hunger: the high side times the state's agrarian share
# (te_dg_agr_share, Peasants and Farmers over its people).
te_inh_wc_high_mult_value = {
	value = var:te_dg_wc
	subtract = 50
	divide = te_inh_k_wc_span
	min = 0
}

te_inh_wc_low_reach = {
	value = 50
	subtract = var:te_dg_wc
	divide = te_inh_k_wc_span
	min = 0
}

te_inh_wc_low_mult_value = {
	value = te_inh_wc_low_reach
	add = {
		value = te_inh_wc_low_reach
		max = 1
	}
	divide = 2
}

te_inh_land_hunger_mult_value = {
	value = te_inh_wc_high_mult_value
	multiply = var:te_dg_agr_share
	min = 0
}

# Display (law tooltips, the concept, the modifiers' descriptions). Safe
# before the first yearly pulse has set the score: the target stands in.
te_inh_concentration_display = {
	value = te_inh_concentration_target
	if = {
		limit = { has_variable = te_inh_concentration }
		value = var:te_inh_concentration
	}
}
