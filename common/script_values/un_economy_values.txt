# ---- The World Development Fund: tuning ------------------------------------------
# The Fund's share of the budget is un_dev_fund_budget_share
# (un_charter_values.txt: none under the founding charter, 5% after Reform I,
# a quarter after Reform II, while it stands), shared with the Standing UN
# Force and the lending facility. Its form follows the charter level every
# month: a reform strengthens a standing Fund with no new vote.

# A member below this share of the represented members' GDP per head receives
# grants (owner, 2026-10-02), by charter level: a tenth of the average under
# the founding charter, a quarter after Reform I, half after Reform II.
un_dev_fund_line_share_founding = { value = 0.1 }
un_dev_fund_line_share_reform_1 = { value = 0.25 }
un_dev_fund_line_share_reform_2 = { value = 0.5 }

# The line's share at the charter's current level. Globals only.
un_dev_fund_line_share = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_charter_level
			global_var:un_charter_level >= 2
		}
		add = un_dev_fund_line_share_reform_2
	}
	else_if = {
		limit = {
			has_global_variable = un_charter_level
			global_var:un_charter_level >= 1
		}
		add = un_dev_fund_line_share_reform_1
	}
	else = {
		add = un_dev_fund_line_share_founding
	}
}

# At or above this multiple of the average, a member is among the richest:
# it leans hard against the Fund, which it pays for and never draws on.
un_dev_fund_rich_share = { value = 2 }

# ---- The World Development Fund: the line ----------------------------------------

# GDP per head. Snapshotted monthly as var:un_dev_fund_gdp_ph on every
# represented member. Scope: country
un_dev_fund_gdp_per_head = {
	value = 0
	if = {
		limit = { total_population > 0 }
		add = gdp
		divide = total_population
	}
}

# The represented members' GDP and people, summed: their GDP per head is the
# one over the other (the average income of a member's citizen, not the
# average of the members' figures, so a microstate counts for its people).
un_dev_fund_members_gdp_value = {
	value = 0
	every_country = {
		limit = { un_member_represented = yes }
		add = gdp
	}
}

un_dev_fund_members_pop_value = {
	value = 0
	every_country = {
		limit = { un_member_represented = yes }
		add = total_population
	}
}

# The members' average GDP per head, from the sums just snapshotted.

un_dev_fund_avg_value = {

		}
		add = global_var:un_dev_fund_members_gdp
		divide = global_var:un_dev_fund_members_pop

# The level's share of the members' average: the line before its floor.

		add = global_var:un_dev_fund_avg
		multiply = un_dev_fund_line_share
	}

# The line a recipient's GDP per head must be at or under: the level's share
# of the average, never below the poorest member in good standing's GDP per
# head (global_var:un_dev_fund_line_floor, snapshotted just before; owner,

un_dev_fund_line_value = {

	if = {
		limit = { has_global_variable = un_dev_fund_line_floor }

# ---- The World Development Fund: the grants -------------------------------------

# What the Fund takes of the budget a week: its share at the charter's level
# of what the paying members' dues make (un_budget_weekly, snapshotted by
# un_authority_monthly_update). 0 while it does not stand, and under the
# founding charter. Globals only.
un_dev_fund_budget_weekly_value = {
	value = 0
	if = {
		limit = { has_global_variable = un_budget_weekly }
		add = global_var:un_budget_weekly
		multiply = un_dev_fund_budget_share
	}
}

# What the Fund pays out a week, all recipients together: its share of the
# budget, and the Development Programs contributions
# (global_var:un_dev_fund_donations, summed by this month's update before the
# grants are worked out), so the grants pay out exactly what the budget and
# the contributors put in. 0 while it does not stand. Globals only.
un_dev_fund_pot_weekly_value = {
	value = un_dev_fund_budget_weekly_value
	if = {
		limit = {
			has_global_variable = un_inst_development_fund
			has_global_variable = un_dev_fund_donations
		}
		add = global_var:un_dev_fund_donations
	}
}

# What it would pay out a week if it stood today (the chamber's preview): its
# share of the budget at the current level, and the contributions being paid
# now (under the founding charter, the contributions alone). Globals only.
un_dev_fund_pot_preview_value = {
	value = 0
	if = {
		limit = { has_global_variable = un_budget_weekly }
		add = global_var:un_budget_weekly
		multiply = un_dev_fund_budget_share_at_level
	}
	if = {
		limit = { has_global_variable = un_dev_fund_donations }
		add = global_var:un_dev_fund_donations
	}
}

# This member's share of the recipients' people, 0..1: a recipient is paid in
# proportion to its population. The division comes first so that the product
# with the pot never multiplies two large numbers together.
# Scope: country
un_dev_fund_pop_share_value = {
	value = 0
	if = {
		limit = {
			un_dev_fund_eligible = yes
			has_global_variable = un_dev_fund_eligible_pop
			global_var:un_dev_fund_eligible_pop > 0
		}
		add = total_population
		divide = global_var:un_dev_fund_eligible_pop
		max = 1
	}
}

# This member's weekly grant: the multiplier of the grant modifier, stored as
# var:un_dev_fund_grant by un_dev_fund_monthly_update. Scope: country
un_dev_fund_grant_value = {
	value = un_dev_fund_pop_share_value
	multiply = un_dev_fund_pot_weekly_value
	min = 0
}

# What it would be paid a week if the Fund stood today. Scope: country
un_dev_fund_grant_preview_value = {
	value = un_dev_fund_pop_share_value
	multiply = un_dev_fund_pot_preview_value
	min = 0
}

# ---- The World Development Fund: the AI's prospect ---------------------------------
# What the Fund means to a country deciding whether to join, pay its dues or
# vote for it (common/script_values/un_ai_values.txt). Read by the AI only.

# The weekly grant the scoped country would draw this month as a represented
# member paying its dues: its own grant while the Fund pays it; otherwise, if
# its GDP per head is at or under the line, its people's share of the pot,
# counted beside the people the Fund pays now. 0 unless the Fund stands. A
# non-member or a member withholding its dues is not in un_dev_fund_eligible_pop,
# so its people are added to it here. A joiner would also shift the members'
# average a little; that is ignored. Scope: country
un_dev_fund_prospect_value = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_inst_development_fund
			has_global_variable = un_dev_fund_line
		}
		if = {
			limit = { un_dev_fund_eligible = yes }
			add = un_dev_fund_grant_value
		}
		else_if = {
			limit = {
				total_population > 0
				un_dev_fund_gdp_per_head <= global_var:un_dev_fund_line
			}
			add = total_population
			divide = un_dev_fund_prospect_pop_value
			multiply = un_dev_fund_pot_weekly_value
			min = 0
		}
	}
}

# The people the Fund would pay with us among them: those it pays now, and
# ours. Scope: country (one not in un_dev_fund_eligible_pop)
un_dev_fund_prospect_pop_value = {
	value = total_population
	if = {
		limit = { has_global_variable = un_dev_fund_eligible_pop }
		add = global_var:un_dev_fund_eligible_pop
	}
	min = 1
}

# The prospect as a share of our GDP a year: 0.01 is 1%. Scope: country
un_dev_fund_prospect_share = {
	value = un_dev_fund_prospect_value
	multiply = 52
	divide = {
		value = gdp
		min = 1
	}
	min = 0
}

# What it would be paid a week if the Fund stood today, as a share of our GDP a
# year (the vote's lean: the Fund is not yet founded, or its form is about to
# be read at a new level). Scope: country
un_dev_fund_grant_preview_share = {
	value = un_dev_fund_grant_preview_value
	multiply = 52
	divide = {
		value = gdp
		min = 1
	}
	min = 0
}

# ---- The World Food Reserve: tuning ---------------------------------------------

# One draw takes at most this share of a pledger's grain stockpile...
un_fr_pledge_share = { value = 0.25 }
# ...and at most this many units from any one pledger.
un_fr_draw_cap = { value = 2500 }
# The grain one new aid mission needs for the full relief. When the pledgers
# could give more, each gives the same fraction of what it offers.
un_fr_draw_target = { value = 5000 }
# A pledger holding less than this is not drawn on (the Reserve does not
# scrape a nearly empty store, and a token stock earns no standing).
un_fr_min_stock = { value = 500 }
# The relief lasts this many monthly state pulses (a year).
un_fr_relief_duration = { value = 12 }
# The delivery entry for a full relief (reason 34; less when less was drawn).
un_fr_delivery_points = { value = 1 }
# Standing for each pledger drawn on (reason 42; before diminishing returns).
un_fr_pledger_standing_gain = { value = 2 }
# The AI pledges once it holds this much grain.
un_fr_ai_spare_grain = { value = 2000 }

# The hunger term of the delivery pillar (owner, 2026-10-02: "reduce UN
# authority if there is food insecurity in members, weighted by people food
# insecure, and raise it if food insecurity drops sufficiently low"). While the
# Reserve stands: un_fr_hunger_floor x the share of the members' people in
# famine-struck states over un_fr_hunger_full_share, at most a full
# un_fr_hunger_floor; and +un_fr_hunger_bonus while that share is below
# un_fr_hunger_bonus_share. Range -10..+3.
un_fr_hunger_floor = { value = -10 }
un_fr_hunger_full_share = { value = 0.05 }
un_fr_hunger_bonus = { value = 3 }
un_fr_hunger_bonus_share = { value = 0.005 }

# ---- The World Food Reserve: hunger ----------------------------------------------

# The members' people living in states with a famine (has_famine). Every
# member counts, suspended or not: its people are the members' people.
un_fr_members_famine_pop_value = {
	value = 0
	every_country = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		every_scope_state = {
			limit = { has_famine = yes }
			add = state_population
		}
	}
}

# Their share of the members' people, 0..1, from the snapshot sums.
# Globals only.
un_fr_famine_share_value = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_fr_members_pop
			global_var:un_fr_members_pop > 0
		}
		add = global_var:un_fr_famine_pop
		divide = global_var:un_fr_members_pop
		max = 1
	}
}

# What hunger would do to the delivery pillar if the Reserve stood: -10..+3.
# Globals only.
un_fr_hunger_term_potential_value = {
	value = 0
	if = {
		limit = { has_global_variable = un_fr_famine_share }
		if = {
			limit = { global_var:un_fr_famine_share < un_fr_hunger_bonus_share }
			add = un_fr_hunger_bonus
		}
		else = {
			add = global_var:un_fr_famine_share
			divide = un_fr_hunger_full_share
			max = 1
			multiply = un_fr_hunger_floor
		}
	}
}

# ---- The lean ------------------------------------------------------------------
# un_lean_interests_<key> (un_lean_charter_interests, un_charter_values.txt):
# the topic's own interests term. Scope: the voting country; expects
# scope:un_resolution. Sizes as in un_lean_interests (un_dossier_values.txt).

# The World Development Fund: the members it would pay are strongly for, and
# the more so the larger the grant it would bring them; once a reformed charter
# gives it a share of the budget, the members above the average against (they
# pay its share of the dues and get nothing back), the richest more so, and
# laissez-faire governments against. Under the founding charter it costs them
# nothing, and they lean against it only a little. "Below the line" reads
# the line at the charter's current level (the monthly snapshot,
# un_dev_fund_below_line). The grant is today's preview as a share of our GDP
# (un_dev_fund_grant_preview_share), at un_ai_points_per_gdp_share points: 20
# for a grant of 1% of GDP a year, at most un_lean_dev_fund_grant_cap. Under
# the founding charter the Fund would pay only contributions, and none can run
# before it stands (owner, 2026-10-03), so the preview there is 0 and a poor
# member votes on the base alone: the hope of contributions.
un_lean_dev_fund_below_line_base = { value = 30 }
un_lean_dev_fund_grant_cap = { value = 30 }

un_lean_interests_development_fund = {
	value = 0
	if = {
		limit = { un_dev_fund_below_line = yes }
		add = un_lean_dev_fund_below_line_base
		add = {
			value = un_dev_fund_grant_preview_share
			multiply = un_ai_points_per_gdp_share
			max = un_lean_dev_fund_grant_cap
		}
	}
	# What the richer members give up. Under the founding charter the Fund
	# takes nothing from the budget and pays out the voluntary contributions
	# alone, so they lose nothing by it and it should pass easily (owner,
	# 2026-10-03); only the richest and laissez-faire governments grumble. From
	# Charter Reform I it takes a share of the dues they pay.
	else_if = {
		limit = { un_dev_fund_budget_share_at_level > 0 }
		if = {
			limit = { un_dev_fund_rich = yes }
			add = -25
		}
		else_if = {
			limit = { un_dev_fund_above_average = yes }
			add = -10
		}
	}
	else_if = {
		limit = { un_dev_fund_rich = yes }
		add = -5
	}
	if = {
		limit = { has_law_or_variant = law_type:law_laissez_faire }
		if = {
			limit = { un_dev_fund_budget_share_at_level > 0 }
			add = -15
		}
		else = {
			add = -5
		}
	}
	if = {
		limit = { has_law = law_type:law_humanitarian_regulations }
		add = 10
	}
}

# The World Food Reserve: members with a famine at home strongly for, members
