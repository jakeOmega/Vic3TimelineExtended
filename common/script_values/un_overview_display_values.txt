# ---- Membership and tier -----------------------------------------------------

# 0 no UN · 1 cannot join · 2 can join · 3 member · 4 permanent member ·
# 5 suspended, the seat carried · 6 suspended. Same order of tests as
# un_chamber_status_sgui, which is the icon's tooltip.
un_disp_member_code = {
	value = 0
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		value = 3
		if = {
			limit = { is_un_permanent_member = yes }
			value = 4
		}
		if = {
			limit = { un_representation_suspended = yes }
			value = 6
			if = {
				limit = { un_seat_carried = yes }
				value = 5
			}
		}
	}
	else_if = {
		limit = { has_global_variable = un_founded }
		value = 1
		if = {
			limit = { un_membership_eligible = yes }
			value = 2
		}
	}
}

# -1 while there is no tier (no UN, or before its first monthly update), else
# global_var:un_tier: 0 moribund, 1 contested, 2 established, 3 strong,
# 4 supranational (un_ladder_triggers.txt).
un_disp_tier_code = {
	value = -1
	if = {
		limit = {
			has_global_variable = un_founded
			has_global_variable = un_tier
		}
		value = global_var:un_tier
	}
}

# ---- The overview's cells: what needs us (the 2026-10-27 audit, UN-1, UN-2) ----
# Each cell of the overview (un_overview_widget.gui) picks its icon and word by
# one of these codes, so the panel decides nothing itself.

# Our standing, for the Standing cell. -1 none (we have never been a member:
# there is no score); 0 frozen (we have left: the score waits for our return,
# un_standing_effects.txt never resets it); 1 not yet recorded (a member before
# its first monthly pulse); then the tier: 2 Disgraced, 3 Poor, 4 Neutral,
# 5 Respected, 6 Exemplary (un_standing_triggers.txt, in the same order).
un_disp_standing_code = {
	value = -1
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		value = 1
		if = {
			limit = { has_variable = un_standing }
			value = 4
			if = { limit = { un_standing_tier_disgraced = yes } value = 2 }
			if = { limit = { un_standing_tier_poor = yes } value = 3 }
			if = { limit = { un_standing_tier_respected = yes } value = 5 }
			if = { limit = { un_standing_tier_exemplary = yes } value = 6 }
		}
	}
	else_if = {
		limit = { has_variable = un_standing }
		value = 0
	}
}

# Our ballot on the resolution in session, for the Session cell and the row
# under the tally (UN-2). -1 nothing in session; 0 not cast, so a vote is due;
# 1 cast for; 2 cast against; 3 vetoed; 4 we tabled it, so our vote is the
# motion; 5 no vote (not a member, our representation or credentials
# suspended, Article 19, or the ballot closed). The same tests in the same
# order as un_chamber_vote_sgui's is_valid, which refuses a second ballot by
# naming the first; a veto books a no vote too, so it is asked first.
un_disp_our_vote_code = {
	value = -1
	if = {
		limit = {
			has_global_variable = un_active_resolution
			un_session_open = yes
		}
		value = 5
		if = {
			limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
			if = {
				limit = { un_chamber_is_proposer = yes }
				value = 4
			}
			else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						un_lobby_in_list = { LIST = un_res_yes COUNTRY = ROOT }
					}
				}
				value = 1
			}
			else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						has_tag = un_res_vetoed
						has_variable = un_res_vetoer
						var:un_res_vetoer ?= ROOT
					}
				}
				value = 3
			}
			else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						un_lobby_in_list = { LIST = un_res_no COUNTRY = ROOT }
					}
				}
				value = 2
			}
			else_if = {
				limit = { un_chamber_ballot_open = yes }
				value = 0
			}
		}
	}
}

# Our dues, for the Dues cell. 0 not a member (the cell hides); 1 our
# overlord pays for our suspended seat; 2 none assessed (Moribund); 3 paid;
# 4 withheld; 5 withheld long enough to lose our vote (Article 19). The same
# tests in the same order as un_chamber_dues_lines, its hover.
un_disp_dues_code = {
	value = 0
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		value = 3
		if = {
			limit = { NOT = { un_dues_self_assessed = yes } }
			value = 1
		}
		else_if = {
			limit = { un_dues_vote_suspended = yes }
			value = 5
		}
		else_if = {
			limit = { un_dues_is_withholding = yes }
			value = 4
		}
		else_if = {
			limit = { NOT = { un_dues_assessed = yes } }
			value = 2
		}
	}
}

# Our record, for the Our Record cell: the gravest resolution the case
# against us supports today. 0 none; 1 a condemnation; 2 sanctions; 3 a
# military mandate. The thresholds Our Record's "Would support" rows read.
un_disp_record_code = {
	value = 0
	if = {
		limit = { un_case_strength >= un_case_threshold_condemn }
		value = 1
	}
	if = {
		limit = { un_case_strength >= un_case_threshold_sanctions }
		value = 2
	}
	if = {
		limit = { un_case_strength >= un_case_threshold_mandate }
		value = 3
	}
}

# Our motions, for the Motions cell. 0 not a member (the cell hides); 1 a
# resolution holds the floor; 2 we wait to table again (our own tabling marks,
# un_ga_resolution_modifier and un_request_cooldown); 3 the floor is ours:
# un_disp_propose_ready_count topics have a case. The count is the monthly
# snapshot (un_propose_ready_snapshot), not a live sweep of the thirty-four
# topics' triggers every frame the overview is open.
un_disp_floor_code = {
	value = 0
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		value = 3
		if = {
			limit = { has_global_variable = un_vote_active }
			value = 1
		}
		else_if = {
			limit = {
				OR = {
					has_modifier = un_ga_resolution_modifier
					has_modifier = un_request_cooldown
				}
			}
			value = 2
		}
	}
}

# ---- The resolution in session (the General Assembly's tally and time bars) -----
# The tally bar spans the members with a vote (global_var:un_vote_eligible_count,
# snapshotted monthly from un_vote_eligible_member_count: the denominator the
# supermajority rule uses). Three stacked bars draw it: red to 1, grey to
# not_no_frac, green to yes_frac.

un_disp_res_eligible = {
	# Before the first monthly snapshot (a save loaded mid-vote, or a
	# resolution tabled in the UN's first month) the bar spans the ballots cast,
	# so it never reads "of 1".
	value = un_disp_res_yes
	add = un_disp_res_no
	min = 1
	if = {
		limit = {
			has_global_variable = un_vote_eligible_count
			global_var:un_vote_eligible_count > 0
		}
		value = global_var:un_vote_eligible_count
	}
}
un_disp_res_yes = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_active_resolution
			un_session_open = yes
		}
		global_var:un_active_resolution = {
