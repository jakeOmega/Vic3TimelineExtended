# ---- Membership and tier -----------------------------------------------------

# 0 no UN · 1 cannot join · 2 can join · 3 member · 4 permanent member ·
# 5 suspended, the seat carried · 6 suspended. Same order of tests as
# un_chamber_status_sgui, which is the icon's tooltip.
un_disp_member_code = {
	value = 0
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		value = 3
		if = {
			limit = { is_un_permanent_member = yes }
			value = 4
		}
		if = {
			limit = { un_representation_suspended = yes }
			value = 6
			if = {
				limit = { un_seat_carried = yes }
				value = 5
			}
		}
	}
	else_if = {
		limit = { has_global_variable = un_founded }
		value = 1
		if = {
			limit = { un_membership_eligible = yes }
			value = 2
		}
	}
}

# The viewer's own line in the Policy tooltip (je_un_auth_bar_policy_own):
# our UN Authority Target modifier, and what it adds to the pillar after the
# good-standing rule and our weight (un_policy_contribution). Live rather than
# snapshotted, since each reads one country, not a world sweep; the pillar
# itself is the monthly snapshot, so the two can differ until the next update.
un_disp_policy_own_modifier = {
	value = modifier:country_un_authority_target_add
}
un_disp_policy_own_contribution = {
	value = un_policy_contribution
}
# 1 while our positive total counts for nothing, because we are outside the UN
# or undermining it (un_policy_counts_in_full); 0 otherwise.
un_disp_policy_own_held = {
	value = 0
	if = {
		limit = {
			NOT = { un_policy_counts_in_full = yes }
			modifier:country_un_authority_target_add > 0
		}
		value = 1
	}
}

# ---- The resolution in session (the General Assembly's tally and time bars) -----
# The tally bar spans the members with a vote (global_var:un_vote_eligible_count,
# snapshotted monthly from un_vote_eligible_member_count: the denominator the
# supermajority rule uses). Three stacked bars draw it: red to 1, grey to
# not_no_frac, green to yes_frac.

un_disp_res_eligible = {
	# Before the first monthly snapshot (a save loaded mid-vote, or a
	# resolution tabled in the UN's first month) the bar spans the ballots cast,
	# so it never reads "of 1".
	value = un_disp_res_yes
	add = un_disp_res_no
	min = 1
	if = {
		limit = {
			has_global_variable = un_vote_eligible_count
			global_var:un_vote_eligible_count > 0
		}
		value = global_var:un_vote_eligible_count
	}
}
un_disp_res_yes = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_active_resolution
			un_session_open = yes
		}
		global_var:un_active_resolution = {
			if = {
				limit = { has_variable = un_res_support }
				add = var:un_res_support
			}
		}
	}
}
un_disp_res_no = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_active_resolution
			un_session_open = yes
		}
		global_var:un_active_resolution = {
			if = {
				limit = { has_variable = un_res_oppose }
				add = var:un_res_oppose
			}
		}
	}
}
un_disp_res_yes_frac = {
	value = un_disp_res_yes
	divide = un_disp_res_eligible

	max = 1
}
un_disp_res_no_frac = {
	value = un_disp_res_no

	max = 1
}
un_disp_res_not_no_frac = {
