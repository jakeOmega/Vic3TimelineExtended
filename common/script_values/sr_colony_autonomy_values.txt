# ---- Free colonies (design § 3) -----------------------------------------------------
# A world needs this many people to stand as a free colony: below it a dead
# holder's world is released, and an outpost can only be closed.
sr_free_colony_minimum = { value = 1000 }

# Ties, 0-100, per supplier: gained each month of supply, lost each month
# without; at full ties the colony asks to join (space_race_events.99). A colony

# declined petition falls back to it.
sr_free_ties_gain_month = { value = 2 }
sr_free_ties_decay_month = { value = 1 }
sr_free_ties_full = { value = 100 }

sr_free_trade_max_units = { value = 5 }

# The AI supplies at most this many free colonies (sr_free_ai_review).

# ---- Colony life (design § 7) -------------------------------------------------------
# The people a colony needs for each event, the monthly chances (whole percents,
# for a holder of such a colony, outside the ten-year cooldown a strike or a
# breach starts; sr_life_month rolls the breach's chance and then 50%, so it is
# half a percent) and the share of its people a dome breach kills. Outside the
# cooldown a strike comes about once in eight years and a breach once in
# sixteen; with ten quiet years after each, a nation sees one about every
# fifteen years, four or so in a late game. The first child fires once per
# nation.
sr_life_first_child_people = { value = 2000 }
sr_life_strike_people = { value = 10000 }
sr_life_breach_people = { value = 5000 }
sr_life_strike_chance = { value = 1 }
sr_life_breach_chance = { value = 1 }
sr_life_breach_share = { value = 0.1 }

# ---- Evacuation (design § 4) --------------------------------------------------------
# create_pop takes a constant size, so evacuees come home in batches of this
# many (the remainder of a batch stays behind and is lost with the world).
sr_evac_batch_people = { value = 1000 }

# The colony_evacuation self-check's reading (sr_colony_evacuate): how much the
# capital state grew, and 90% of what was sent. Any scope; the globals are set
# just before.
sr_evac_log_rose = {
	value = 0
	if = {
		limit = {
			has_global_variable = sr_evac_log_before
			has_global_variable = sr_evac_log_after
		}
		value = global_var:sr_evac_log_after
		subtract = global_var:sr_evac_log_before
	}
}
sr_evac_log_rose_needed = {
	value = 0
	if = {
		limit = { has_global_variable = sr_evac_log_sent }
		value = global_var:sr_evac_log_sent
		multiply = 0.9
	}
}
# ... and the placeholder (promethean) people still in the receiving state once
# both conversions ran: 0 when change_pop_religion and change_pop_culture took
# effect at once. State scope.
sr_evac_placeholder_left = {
	value = 0
	every_scope_pop = {
		limit = { culture = cu:promethean }
		add = total_size
	}
}
