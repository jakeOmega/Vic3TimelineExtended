# What the AI construction pass reads (common/scripted_effects/te_ai_build_pass_effects.txt).
# Country scope unless marked. Its numbers and their reasons:
# te_construction_market_ai_values.txt, THE PASS.

# The debt a government may hold and still have the pass queue for it, as a
# share of the debt ceiling: TE_AI_BUILD_PASS_DEBT_MAX on the rate curve's low
# side (none at the middle, at war or in a bubble), never more than a third of
# the surplus carries at the rate the curve's high end asks. Apart from the
# purchase's credit allowance, so that the credit's switch (CREDIT_MAX) at 0
# leaves the pass reaching a government in debt at a low rate.
te_ai_build_pass_debt_tolerance = {
	value = te_construction_market_ai_rate_ease
	multiply = TE_AI_BUILD_PASS_DEBT_MAX
	max = te_construction_market_ai_credit_service_cap
}

# What the government may spend on a new level, as a share of a limit: its reserves
# above the reserve the rate asks for, plus the debt it may hold at its rate, less
# its debt. Reserves and debt are never both above nothing, so a government in
# debt has the means while its debt is within the tolerance, and one with reserves
# while they reach the reserve asked for. Negative is short.
te_ai_build_pass_means = {
	value = scaled_gold_reserves
	subtract = te_construction_market_ai_required_reserves
	add = te_ai_build_pass_debt_tolerance
	subtract = scaled_debt
}

te_ai_build_pass_surplus_share = {
	value = net_fixed_income
	divide = {
		value = fixed_income
		min = 1
	}
}

# Weeks the surplus takes to pay for a level at the smoothed price: a university's
# points (vanilla's construction_cost_medium) and an administration's
# (construction_cost_very_low). te_ai_build_pass_may_queue holds the price
# variable and a surplus above 0.
te_ai_build_pass_university_weeks = {
	value = construction_cost_medium
	multiply = var:te_construction_market_price_stabilizer
	divide = {
		value = net_fixed_income
		min = 1
	}
}
te_ai_build_pass_administration_weeks = {
	value = construction_cost_very_low
	multiply = var:te_construction_market_price_stabilizer
	divide = {
		value = net_fixed_income
		min = 1
	}
}

# Weekly innovation over its cap. The cap is the modifier's sum: 50 base and 150 at
# full literacy (vanilla's code static modifiers), with every other source. 1 where
# the cap reads nothing, so no university is queued on a failed read.
te_ai_build_pass_innovation_cap = { value = modifier:country_weekly_innovation_max_add }
te_ai_build_pass_innovation_share = {
	value = 1
	if = {
		limit = {
			te_ai_build_pass_innovation_cap > 0
		}
		value = country_innovation
		divide = te_ai_build_pass_innovation_cap
	}
}

# The net bureaucracy under which the pass queues an administration: BUREAUCRACY_MARGIN of the bureaucracy the
# country uses.
te_ai_build_pass_bureaucracy_margin = {
	value = bureaucracy_usage
	multiply = TE_AI_BUILD_PASS_BUREAUCRACY_MARGIN
}

# State scope: how far its taxes outrun its tax capacity (negative where capacity is to spare).
te_ai_build_pass_tax_shortfall = {
	value = tax_capacity_usage
	subtract = tax_capacity
}

# The country's levels of the kind the pass last queued (te_ai_build_pass_pending_kind:
# 1 a university, 2 a Government Administration), built, and still in a queue.
te_ai_build_pass_tracked_levels = {
	value = 0
	if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 1
		}
		every_scope_building = {
			limit = {
				is_building_type = building_university
			}
			add = level
		}
	}
	else = {
		every_scope_building = {
			limit = {
				is_building_type = building_government_administration
			}
			add = level
		}
	}
}
te_ai_build_pass_tracked_queued = {
	value = 0
	if = {
		limit = {
			var:te_ai_build_pass_pending_kind = 1
		}
		every_scope_building = {
			limit = {
				is_building_type = building_university
			}
			add = te_construction_market_building_queued_levels
		}
	}
	else = {
		every_scope_building = {
			limit = {
				is_building_type = building_government_administration
			}
			add = te_construction_market_building_queued_levels
		}
	}
}

# The self-check's reads (TE_SELF_CHECK ai_build_pass).
te_ai_build_pass_built_g = { value = global_var:te_ai_build_pass_built }
te_ai_build_pass_waiting_g = { value = global_var:te_ai_build_pass_waiting }
te_ai_build_pass_lost_g = { value = global_var:te_ai_build_pass_lost }
# Built plus lost: the levels whose fate is known.
te_ai_build_pass_decided_g = {
	value = global_var:te_ai_build_pass_built
	add = global_var:te_ai_build_pass_lost
}
te_ai_build_pass_sampled_g = {
	value = te_ai_build_pass_decided_g
	add = global_var:te_ai_build_pass_waiting
}
te_ai_build_pass_lost_share = {
	value = global_var:te_ai_build_pass_lost
	divide = {
		value = te_ai_build_pass_decided_g
		min = 1
	}
}
