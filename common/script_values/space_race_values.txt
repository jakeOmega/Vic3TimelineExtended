# Progress a milestone drifts forward each month while NO approach is selected.
# The monthly pulse's `else` branch and the journal-entry widget's "drifting"
# line both read this, so the rate the panel quotes is the rate the pulse adds.
# Selecting an approach switches both over to sr_progress below.
sr_progress_drift = {
	value = 0.5
}

# Monthly progress value (unified for both approaches)
# Base progress from space program PM, techs, buildings, companies, partnerships, AND funding
# All contribute via country_space_race_progress_add modifier (visible in tooltip breakdown)
# Funding: sr_funding_progress modifier on each JE (0.5 per funding level and +25% building_space_program throughput per funding level, stacks across JEs)
# Approach: ambitious gives country_space_race_progress_mult = 0.5 per JE
sr_progress = {
	value = modifier:country_space_race_progress_add
	# Approach multiplier (ambitious JEs contribute +50% each via modifier)
	multiply = {
		value = 1
		add = modifier:country_space_race_progress_mult
	}
	min = 0.5
}
