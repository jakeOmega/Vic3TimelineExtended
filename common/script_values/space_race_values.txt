# Goals scale with later PMs providing more base progress
# PM progression: earth_orbit=1, lunar/probes=2, stations=3, mars=4, interstellar=5

	value = 50
}

	value = 100
}

	value = 200
}

	value = 150

	value = 300
}
sr_mars_landing_goal = {
	value = 450
}
sr_interstellar_probe_goal = {
	value = 600
}
sr_solar_colonization_goal = {
	# Base goal scales with stage - further out = harder: 300 for the Moon (stage
	# 0), then 450, 525, 600, 700, 800 (sr_colonization_goal_stage_<n>,
	# gen_solar_sites.STAGE_GOALS; lunar colonization design § 3.2). Under simple
	# and full colonies, with a target, the target's own goal (the stages are no
	# locks there; slices 2-3 design § 9), the landing site's discount included.
	value = sr_colonization_goal_stage_0
	if = {
		limit = {
			sr_colonies_contested_enabled = yes
			has_variable = sr_cf_target
		}
		value = sr_cf_target_goal
	}
	else_if = {
		limit = {
			has_global_variable = sr_colonization_stage
			global_var:sr_colonization_stage >= 5
		}
		value = sr_colonization_goal_stage_5
	}
	else_if = {
		limit = {
			has_global_variable = sr_colonization_stage
			global_var:sr_colonization_stage >= 4
		}
		value = sr_colonization_goal_stage_4
	}
	else_if = {
		limit = {
			has_global_variable = sr_colonization_stage
			global_var:sr_colonization_stage >= 3
		}
		value = sr_colonization_goal_stage_3
	}
	else_if = {
		limit = {
			has_global_variable = sr_colonization_stage
			global_var:sr_colonization_stage >= 2
		}
		value = sr_colonization_goal_stage_2
	}
	else_if = {
		limit = {
			has_global_variable = sr_colonization_stage
			global_var:sr_colonization_stage >= 1
		}
		value = sr_colonization_goal_stage_1
	}
}

# Monthly progress value, the same for every running milestone whatever its
# approach (Standard, Safe or Ambitious; the approach modifiers act through
# the country-wide multipliers below). Until 2026-09-29 a milestone with no
# approach chosen drifted at a flat 0.5 instead; the Standard approach
# replaced that state (owner's play-test, round 2).
# Base progress from space program PM, techs, buildings, companies, partnerships, AND funding
# All contribute via country_space_race_progress_add modifier (visible in tooltip breakdown)
# Funding: sr_funding_progress modifier on each JE (0.5 per effective funding level, stacks across JEs: a level up to the cap counts in full, the levels past it 1/2, 1/3, ... 1/7, sr_funding_<m>_effective); the +25% building_space_program throughput per level is sr_funding_throughput's
# Approach: ambitious gives country_space_race_progress_mult = 0.5 per JE
sr_progress = {
	value = modifier:country_space_race_progress_add
	# Approach multiplier (ambitious JEs contribute +50% each via modifier)
	multiply = {
		value = 1
		add = modifier:country_space_race_progress_mult
	}
	min = 0.5
}

# ======= After the race: Commercial Space Industry =======
# The points of sr_commercial_space_dividend (space_race_modifiers.txt), each
# worth +5% Extraplanetary Base throughput: the flat progress the country's
# sources still give once it has nothing left to race for
# (sr_space_race_concluded). Applied only then, by sr_refresh_space_dividend.
# - The flat sum, not sr_progress: its 0.5 floor would pay every finished
#   country for nothing, and the progress bonuses (Antimatter Engines alone
#   reach +250%) would let the dividend run away.
# - Funding is already out of it: sr_funding_progress (and its throughput twin,
#   sr_funding_throughput) sits only on a billed entry

#   progress-granting colony specialisations give together (9.25). At 5% a

#   of a mining one (+6% Extraplanetary Base throughput on average), so mining
#   stays the long-term pick.

# ---- Funding past the cap (lunar colonization design § 5.1) ----------------
# Per milestone, in country scope:
# - sr_funding_<m>_within_cap: the level, up to the cap.
# - sr_funding_<m>_past_cap: the levels past the cap (0 at or below it). Not
#   capped at the ladder's length: a cap that falls below a milestone's level
#   (a power bloc principle lost) leaves the level standing, and the excess
#   simply counts past the cap. Nothing clamps the level down to the cap.
# - sr_funding_<m>_past_cap_effective: what those levels are worth in levels of
#   pace, the ladder 1/2 + 1/3 + ... + 1/7 (0.50, 0.83, 1.08, 1.28, 1.45,
#   1.59). A level past the sixth step, reachable only by a falling cap, adds
#   nothing.
# And in journal-entry scope, sr_funding_<m>_effective: the two added up, the
# multiplier of the entry's sr_funding_progress. Every country read sits inside
# owner = { }, as sr_funding_<m>_level's does (je_multiplier_scope_audit).
sr_funding_suborbital_within_cap = {
	value = 0
	if = { limit = { has_variable = sr_funding_suborbital } add = var:sr_funding_suborbital }
	max = sr_max_funding_level
	min = 0
}
sr_funding_suborbital_past_cap = {
	value = 0
	if = { limit = { has_variable = sr_funding_suborbital } add = var:sr_funding_suborbital }
	subtract = sr_max_funding_level
	min = 0
}
sr_funding_suborbital_past_cap_effective = {
	value = 0
	if = { limit = { sr_funding_suborbital_past_cap >= 1 } add = sr_funding_past_cap_step_1 }
	if = { limit = { sr_funding_suborbital_past_cap >= 2 } add = sr_funding_past_cap_step_2 }
	if = { limit = { sr_funding_suborbital_past_cap >= 3 } add = sr_funding_past_cap_step_3 }
	if = { limit = { sr_funding_suborbital_past_cap >= 4 } add = sr_funding_past_cap_step_4 }
	if = { limit = { sr_funding_suborbital_past_cap >= 5 } add = sr_funding_past_cap_step_5 }
	if = { limit = { sr_funding_suborbital_past_cap >= 6 } add = sr_funding_past_cap_step_6 }
}
sr_funding_suborbital_effective = {
	value = 0
	owner = {
		add = sr_funding_suborbital_within_cap

sr_mission_units_earth_orbit = 500
sr_mission_units_moon = 1500
sr_mission_units_mars = 5000
sr_mission_units_solar = 10000
sr_mission_units_deep_space = 50000
sr_mission_units_interstellar = 100000

# Our market's Launch Capacity price, never below the base price: the AI
# should not plan on a glut lasting (st_res_grain_price_up's shape).
sr_launch_capacity_price_up = {
	value = 0
	market = {
		mg:launch_capacity = {
			add = market_goods_pricier
		}
	}
	min = 0
}

sr_launch_capacity_unit_price = {
	value = 1
	g:launch_capacity = {
		multiply = base_price
	}
	multiply = {
		value = 1
		add = sr_launch_capacity_price_up
	}
}

# The mission the programme runs now, as its weekly units (0 without one).
# Each mission carries every lower mission's bool, so the highest bool held
# names it.
sr_mission_units_current = {
	value = 0
	if = {
		limit = { modifier:country_sr_interstellar_mission_program_bool = yes }
		add = sr_mission_units_interstellar
	}
	else_if = {
		limit = { modifier:country_sr_deep_space_exploration_program_bool = yes }
		add = sr_mission_units_deep_space
	}
	else_if = {
		limit = { modifier:country_sr_solar_colonization_program_bool = yes }
		add = sr_mission_units_solar
	}
	else_if = {
		limit = { modifier:country_sr_mars_mission_program_bool = yes }
		add = sr_mission_units_mars
	}
	else_if = {
		limit = { modifier:country_sr_moon_mission_program_bool = yes }
		add = sr_mission_units_moon
	}
	else_if = {
		limit = { modifier:country_sr_earth_orbit_program_bool = yes }
		add = sr_mission_units_earth_orbit
	}
}

# Founding: the whole Earth Orbit bill.
sr_mission_bill_earth_orbit = {
	value = sr_mission_units_earth_orbit
	multiply = sr_launch_capacity_unit_price
}

# Switching up: only the increase, since net_fixed_income already pays the
# mission the programme runs now.
sr_mission_bill_increase_moon = {
	value = sr_mission_units_moon
	subtract = sr_mission_units_current
	multiply = sr_launch_capacity_unit_price
}
sr_mission_bill_increase_mars = {
	value = sr_mission_units_mars
	subtract = sr_mission_units_current
	multiply = sr_launch_capacity_unit_price
}
sr_mission_bill_increase_solar = {
	value = sr_mission_units_solar
	subtract = sr_mission_units_current
	multiply = sr_launch_capacity_unit_price
}
sr_mission_bill_increase_deep_space = {
	value = sr_mission_units_deep_space
	subtract = sr_mission_units_current
	multiply = sr_launch_capacity_unit_price
}
sr_mission_bill_increase_interstellar = {
	value = sr_mission_units_interstellar
	subtract = sr_mission_units_current
	multiply = sr_launch_capacity_unit_price
}
