# Script values for GUI chart visualizations
# These are read by GUI property expressions via MakeScope.ScriptValue()

# ============================================================================
# TRADE VALUE (country scope)
# ============================================================================

# Total world market trade value for a country
# Used by market panel bar chart: Country.MakeScope.ScriptValue('gui_country_total_trade_value')
gui_country_total_trade_value = {
	value = total_trade_value
}

# ============================================================================
# TOP TRADE PARTNERS (market / country scope)
# Used by update_market_trade_partners in trade_partner_effects.txt
# ============================================================================

# Sort value for ordered_market: mutual trade between partner market's owner and our market owner
# Evaluated in partner market scope (iterated). scope:tp_market_owner must be saved.
gui_trade_with_market_owner = {
	value = 0
	trade_center = {
		owner = {
			add = "mutual_trade_value_with_country(scope:tp_market_owner)"
		}
	}
}

# Annual bilateral trade value (imports + exports) between our market owner and
# the partner market owner. mutual_trade_value_with_country is the *weekly* base
# value of goods traded. Vanilla prints no unit, but every market flow is per week
# and the engine describes the related piracy value as an "average weekly base
# value", so x52 gives the per-year figure the Top Trade Partners table shows.
# Evaluated in our market owner's country scope. scope:tp_partner_country = the
# partner market's owner (saved in store_trade_partner_data).
gui_tp_annual_trade = {
	value = "mutual_trade_value_with_country(scope:tp_partner_country)"
	multiply = 52
}

# Bilateral trade as a percentage of our market owner's GDP (0-100+).
# YEARLY trade over `gdp`: the `gdp` script value is yearly ("the scoped country's
# yearly predicted GDP", docs/engine/event_targets_summary.txt; the monetary
# system annualises its weekly flows the same way), while the trade value is
# weekly, so dividing it unscaled came out 52 times too low. Trade is at base
# prices and GDP at market prices; the table's tooltip says so.
# Evaluated in our market owner's country scope. scope:tp_partner_country as above.
gui_tp_trade_pct_gdp = {
	value = gui_tp_annual_trade
	divide = {
		value = gdp
		min = 1
	}
	multiply = 100
}

# Count of trade partners found (for GUI visibility checks)
# Evaluated in country scope: Market.GetOwner.MakeScope.ScriptValue('gui_tp_count')
gui_tp_count = {
	value = 0
	if = {
		limit = { has_variable = tp_count }
		add = var:tp_count
	}
}

# ============================================================================
# DIRECTIONAL BILATERAL TRADE VOLUME (market scope, GUI-only)
# Used by the Import/Export Partners charts in market_panel.gui. The chart
# evaluates these via a GuiScope where root = the panel's market and
# scope:base_market = the partner market (the iterated chart item):
#   GuiScope.SetRoot( <panel market>.MakeScope )
#           .AddScope('base_market', <partner market>.MakeScope)
#           .ScriptValue('gui_imports_from_target')
# market_imports/market_exports return total *units* of goods (not value).
# ============================================================================

# Units the root market imports from scope:base_market.
gui_imports_from_target = {
	value = "market_imports(scope:base_market)"
}

# Units the root market exports to scope:base_market.
gui_exports_to_target = {
	value = "market_exports(scope:base_market)"
}

# ----------------------------------------------------------------------------
# Ordering helpers for the partner-market lists built in trade_partner_effects.txt.
# Evaluated in an *iterated* market scope; scope:tp_current_market = the market
# whose partner lists are being built (the panel's market).
#   market_exports(scope:tp_current_market) in iterated scope
#       = iterated exports to current = current's IMPORTS from iterated.
#   market_imports(scope:tp_current_market) in iterated scope
#       = iterated imports from current = current's EXPORTS to iterated.
# ----------------------------------------------------------------------------

# Volume the current market imports from the iterated market (import-partner sort).
te_tp_exports_to_current = {
	value = "market_exports(scope:tp_current_market)"
}

# Volume the current market exports to the iterated market (export-partner sort).
te_tp_imports_from_current = {
	value = "market_imports(scope:tp_current_market)"
}

# ============================================================================
# HIDDEN STATE MODIFIERS (state scope, GUI-only)
# ============================================================================

# How many of the state's modifiers the state panel hides. gui/states_panel.gui
# leaves out of both modifier rows every modifier with the mod's blank icon
# (gfx/interface/icons/timed_modifier_icons/modifier_system.dds), and fades the
# condensed view's box when the state carries no other: when its modifier count
# is not above this. A GUI can tell a modifier only by its icon, so the box needs
# the hidden ones named here: one term per static modifier with that icon.
# test_hidden_state_modifiers.py fails when such a modifier has no term.
# Used by: State.MakeScope.ScriptValue('gui_state_hidden_modifier_count')
gui_state_hidden_modifier_count = {
	value = 0
	if = {
		limit = { has_modifier = pollution_generation_mult_effect }
		add = 1
	}
	if = {
		limit = { has_modifier = tourism_output }
		add = 1
	}
	if = {
		limit = { has_modifier = tourism_throughput }
		add = 1
	}
	if = {
		limit = { has_modifier = migration_crowding }
		add = 1
	}
	if = {
		limit = { has_modifier = free_port_state_modifier }
		add = 1
	}
	if = {
		limit = { has_modifier = free_port_tariff_import_modifier }
		add = 1
	}
	if = {
		limit = { has_modifier = free_port_tariff_export_modifier }
		add = 1
	}
	if = {
		limit = { has_modifier = te_map_mode_overlay }
		add = 1
	}
	if = {
		limit = { has_modifier = te_demog_census_births_up }
		add = 1
	}
	if = {
		limit = { has_modifier = te_demog_census_births_down }
		add = 1
	}
	if = {
		limit = { has_modifier = te_demog_census_deaths_up }
		add = 1
	}
	if = {
		limit = { has_modifier = te_demog_census_deaths_down }
		add = 1
	}
}
