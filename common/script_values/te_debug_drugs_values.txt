# Drugs price in this country's market, as a multiple of the base price.
# Callers check that the market trades Drugs first.

	value = 1
	market = {
		mg:opium = {
			add = market_goods_pricier
		}
	}
}

# ---- World Drugs supply by production method (any scope) -------------------
# Levels, not output: a plantation makes 20-55 Drugs a level and Pharmaceutical
# Industries 30-345, so the results table weights each method by its output.
te_dbg_drugs_world_levels_plantation_default = {
	value = 0
	every_country = {
		every_scope_building = {
			limit = {
				is_building_type = building_opium_plantation
				has_active_production_method = default_building_opium_plantation
			}
			add = level
		}
	}
}

# ---- debug.log readout (evaluated against ROOT) -----------------------------
# A debug_log `[SCOPE.ScriptValue('x')|N]` evaluates against ROOT, and it cannot
# read variables directly (scripting_best_practices.md § "`debug_log` Loc-String
# Templating Limitations"). te_debug_drugs_measure copies each country's numbers
# onto ROOT as te_dbg_drugs_cur_* before logging them through these wrappers.
# -1 means the variable was missing.
te_dbg_drugs_log_cur_pop = {
	if = {
		limit = { has_variable = te_dbg_drugs_cur_pop }
		value = var:te_dbg_drugs_cur_pop
	}
	else = {
		value = -1
	}
}
