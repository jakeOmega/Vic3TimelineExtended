# ---- World Drugs supply by production method (any scope) -------------------
# Levels, not output: a plantation makes 20-55 Drugs a level and Pharmaceutical
# Industries 30-345, so the results table weights each method by its output.
te_dbg_drugs_world_levels_plantation_default = {
	value = 0
	every_country = {
		every_scope_building = {
			limit = {
				is_building_type = building_opium_plantation
				has_active_production_method = default_building_opium_plantation
			}
			add = level
		}
	}
