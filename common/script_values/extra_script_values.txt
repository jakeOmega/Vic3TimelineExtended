	# Authority-cost multiplier on forcing a law, scaled by override capacity. Two segments:
	#   capacity 2–4 → modest relief  1 - 0.03*(cap-1)  (cap2=0.97, cap3=0.94, cap4=0.91) — the
	#       penalty-buffer (extra_modifiers.txt) and tier access are the main reward for these points.
	#   capacity ≥5 → deep discount  4/capacity  (cap5=0.80, cap8=0.50, cap16=0.25) — approaches 0,
	#       never reaches it. The drop at 4→5 rewards crossing into deep-override territory.
	# Reads the RAW float sum — no floor/round, so fractional institution sources and the decaying
	# penalty both count proportionally.
	value = 1
	if = {
		limit = {
			"modifier:country_legislative_override_capacity_add" >= 2
			"modifier:country_legislative_override_capacity_add" < 5
		}
		subtract = {
			value = modifier:country_legislative_override_capacity_add
			subtract = 1
			multiply = 0.03
		}
	}
	else_if = {
		limit = { "modifier:country_legislative_override_capacity_add" >= 5 }
		multiply = 4
		divide = {
			value = modifier:country_legislative_override_capacity_add
		}
	}
}

# ========== SoL Expectations System ==========

# Half-life in years: how long it takes for the gap between expected and actual SoL
# to close by half. 10 years means ~50% adapted after 10y, ~75% after 20y, ~94% after 40y.
# Change this single value to tune convergence speed.
sol_expectations_half_life_years = 10

# Monthly adaptation rate derived from half-life: rate = ln(2) / (H * 12)
# ln(2) ≈ 0.6931. At default 5y: 0.6931 / 60 ≈ 0.01155/month.
# The country_sol_expectation_adaptation_rate_mult modifier scales this.
sol_expectations_adaptation_rate_value = {
	value = 0.6931
	divide = {
		value = sol_expectations_half_life_years
		multiply = 12
	}
	multiply = {
		value = 1
		add = modifier:country_sol_expectation_adaptation_rate_mult
	}
	min = 0.001
	max = 0.1
}

# Gap between target SoL and current expected SoL (including our modifier).
# Target = average_sol + country_sol_expectations_target_add.
# Positive = expectations below target (needs upward correction).
# Negative = expectations above target (needs downward correction).
# Must be cached in a variable BEFORE removing the modifier, since
# average_expected_sol won't recalculate mid-effect after removal.
sol_expectations_gap_value = {

# Fiscal policy effect based on spending relative to GDP: the multiplier on
# financial_cycle_government_fiscal_policy_effect (country_finance_value_monthly_add = 1),
# so this IS the monthly push on finance_cycle_value.
#
# `income` and `total_expenses` are WEEKLY and `gdp` is ANNUAL
# (docs/engine/event_targets_summary.txt), so the flows are annualised through
# the same te_mon_deficit_annualise_factor that te_mon_deficit_pct_value uses —
# the one hypothesis constant, so if a running game ever shows the flows to be
# annual already (monetary_policy_design.md ruling P4) both channels flip
# together. Before 2026-09-22 this value divided the weekly flow by annual GDP
# and multiplied by 100, i.e. it was 1/52 of what its own comment claimed ("1%
# of GDP a year adds 1 point a month"): a 3% deficit moved the cycle 0.06 a
# month, which is why the channel read as dormant
# (docs/audits/banking_cycle_simulation.md F8).
#
# 0.1 cycle points a month per 1% of GDP of deficit, clamped to +-1: it takes
# a 10%-of-GDP deficit to push the cycle a full point a month, about one phase
# modifier's worth, and the clamp keeps anything larger from pinning the cycle
# in frenzy on its own. A surplus cools the cycle the same way. The unclamped
# "1 point per 1%" the old comment intended was measured: it runs the cycle to
# a mean of 53-57 and 20-26 crashes a century (§6). 0.25 (+1 at a 4% deficit)
# shipped first and was cut to 0.1 on 2026-09-22: an ordinary wartime deficit
# should not be worth a phase modifier by itself.
financial_cycle_government_fiscal_policy_effect_size = {
	value = total_expenses
	subtract = income
	multiply = te_mon_deficit_annualise_factor
	divide = { value = gdp min = 1 }
	multiply = 10
	min = -1
	max = 1
}

# linear up to +0.25 at 50 bubble pressure, then a gentle curve to +0.45 at
# max pressure. Until 2026-09-22 the upper half was quadratic to +1.25: at
# bubble 80+ that was a standing +7-12 cycle points a month, which no tool
# set could turn, so a frenzy was never escapable. Flattened with frenzy's
# bubble 20 -> 8 (banking_cycle_simulation.md §10); crash frequency moved
# ~1%, and maxed tools now escape a few percent of frenzies.
bubble_inertia_multiplier_script_value = {
	value = 0
	owner = {
		if = {
			limit = {
				var:bubble_pressure <= 15
			}
			add = 0
		}
		else_if = {
			limit = {
				var:bubble_pressure < 50
			}
			add = {
				value = var:bubble_pressure
				subtract = 15
				multiply = 0.25
				divide = 35 # rises by 0.25 over 35 points of pressure
			}
		}
		else = {
			add = {
				# f(x) = 0.25 + 0.005 (x - 50) - 0.00002 (x - 50)^2
				# f(50) = 0.25, f'(50) = 1/200 (continuous with the branch above),
				# f(100) = 0.45
				value = 0.25
				add = {
					value = var:bubble_pressure
					subtract = 50
					multiply = 0.005
				}
				add = {
					value = var:bubble_pressure
					subtract = 50
					multiply = {
						value = var:bubble_pressure
						subtract = 50
					}
					multiply = -0.00002
				}
			}
		}
	}
}

# ====================================================================================
# CONSTRUCTION COST SCALING
# ====================================================================================
# Makes construction more expensive for richer countries (GDP per capita based).
# Prevents exponential growth and provides catchup for poorer nations.
# Applied via on_yearly_pulse_country → construction_cost_scaling_on_action.
# Modifier: construction_cost_scaling (country_construction_goods_cost_mult = 1, scaled by multiplier).
#
# TUNING GUIDE:
# 1. Adjust construction_cost_gdppc_reference to match a "developed 1836 nation" GDP/c.
#    Check in-game with debug_log or console: a developed 1836 nation (UK, Netherlands)
#    should have GDP/c roughly equal to this value.
# 2. floor_ratio and ceiling_ratio are multiples of the reference:
#    - floor_ratio = 1 means no penalty at the reference level (a rich 1836 nation).
#    - ceiling_ratio = 14 means max penalty at 14x the reference (late-game AI economy).
# 3. max_mult is the maximum construction cost multiplier (10 = +1000% cost at ceiling).
#
# GDP is in yearly terms. GDP/c = gdp / total_population (country scope).
# ====================================================================================

# Reference GDP/c for a "developed 1836 nation" (UK, Netherlands).
# If in-game GDP/c for UK 1836 ≈ 50, set this to 50. If ≈ 0.5, set to 0.5.
construction_cost_gdppc_reference = 1

# No construction penalty below this multiple of reference GDP/c
construction_cost_floor_ratio = 1  # 1x reference = rich 1836 nation baseline

# Maximum penalty reached at this multiple of reference GDP/c
construction_cost_ceiling_ratio = 100  # 100x reference = very rich late-game economy

# Maximum construction cost multiplier at ceiling (10 = construction costs up to +1000%)
construction_cost_max_mult = 10

# Derived thresholds (do not edit - change the ratios above instead)
construction_cost_gdppc_floor = {
	value = construction_cost_gdppc_reference
	multiply = construction_cost_floor_ratio
}
construction_cost_gdppc_ceiling = {
	value = construction_cost_gdppc_reference
	multiply = construction_cost_ceiling_ratio
}

# Country-scope GDP per capita
country_gdp_per_capita = {
	value = gdp
	divide = {
		value = total_population
		min = 1
	}
}

# Computed construction cost penalty: 0 at floor, max_mult at ceiling, linear interpolation.
# Formula: clamp(0, max_mult, (gdppc - floor) / (ceiling - floor) * max_mult)
construction_cost_scaling_mult = {
	value = country_gdp_per_capita
	subtract = construction_cost_gdppc_floor
	min = 0  # poor countries: no penalty; also the lower half of the clamp documented above (everything downstream is non-negative)
	divide = {
		value = construction_cost_gdppc_ceiling
		subtract = construction_cost_gdppc_floor
		min = 1  # prevent division by zero
	}
	multiply = construction_cost_max_mult
	max = construction_cost_max_mult
}

# Adjusted construction cost scaling that combines MULTIPLICATIVELY with other
# goods_input_construction_mult modifiers (ministry of environment, laws, etc.).
# Without this, other modifiers are just added to the scaling, making their
# marginal effect negligible when scaling is high (e.g. +20% ministry on top
# of +500% scaling = only 3% increase in final cost, instead of 20%).
#
# Formula: adjusted = base_scaling * (1 + other_construction_mults)
# where other_mults = total modifier:goods_input_construction_mult minus our
# own prior contribution (stored in var:prior_construction_scaling).
# Result: total = adjusted + other_mults = (1 + base_scaling) * (1 + other_mults) - 1
# which is the correct multiplicative combination.
construction_cost_scaling_adjusted_mult = {
	value = construction_cost_scaling_mult
	multiply = {
		value = 1
		add = modifier:goods_input_construction_mult
		if = {
			limit = { has_variable = prior_construction_scaling }
			subtract = var:prior_construction_scaling
		}
	}
	min = 0
}

# ====================================================================================
# MIGRATION CROWDING
# ====================================================================================
# Purpose: reduces migration pull to heavily populated states using population
# density (population per arable land) as a proxy for geographic capacity.
# Applied via `on_yearly_pulse_state` → `migration_crowding_on_action`.
# Modifier: `migration_crowding` (base: state_migration_pull_mult = -0.1,
# multiplier computed below).
#
# Tuning:
# - `migration_crowding_density_reference`: population per unit arable land
#   for a "normally full" state (e.g., 100 arable & 5M people → density 50000).
# - `floor_ratio`: multiple of reference below which no penalty applies (1 = no penalty).
# - `ceiling_ratio`: multiple of reference where linear phase begins (default 10).
# - Arable land ranges: small ~20–50, medium ~80–150, large ~200–500+.
#
# Behavior / formula (normalized ratio r):
# - Compute `state_population_density` = state_population /
#   (arable_land - same_owner_state_region_arable_land_added) × density_mult.
# - Normalize: r = (density - floor) / (ceiling - floor). r = 0 at floor, r = 1 at ceiling.
# - Multiplier M (continuous piecewise):
#     * r <= 0:         M = 0
#     * 0 < r < 1:       M = 4.5 * r^2        (gradual quadratic; M=4.5 at r=1 → 45%)
#     * r >= 1:          M = 9 * r - 4.5      (linear continuation; slope 9)
# - Actual migration pull reduction percent = M * 0.1
#   (since modifier base is -0.1, M=4.5 → 45% reduction; M=14.5 → 145% reduction).
#
# Notes:
# - Uses `same_owner_state_region_arable_land_added` to isolate base arable land
#   so split states don't undercount geographic base.
# - Ratio capping is intentionally removed to allow linear scaling above the ceiling.
# ====================================================================================
migration_crowding_density_reference = 10000
migration_crowding_floor_ratio = 1
migration_crowding_ceiling_ratio = 10 # Linear growth kicks in here

migration_crowding_density_floor = {
	value = migration_crowding_density_reference
	multiply = migration_crowding_floor_ratio
}
migration_crowding_density_ceiling = {
	value = migration_crowding_density_reference
	multiply = migration_crowding_ceiling_ratio
}

# State population density calculation.
# Use the same-owner regional arable-land cache so split states don't undercount base land.
state_population_density = {
	value = state_population
	divide = {
		value = arable_land_base
		min = 1
	}
	divide = {
		value = 1
		add = modifier:state_migration_crowding_density_mult
		min = 0.1
	}
}

# Normalized ratio: 0 at floor (1x), 1.0 at ceiling (10x).
# Used for localization tooltips and intermediate math.
migration_crowding_ratio = {
	value = state_population_density
	subtract = migration_crowding_density_floor
	min = 0
	divide = {
		value = migration_crowding_density_ceiling
		subtract = migration_crowding_density_floor
		min = 1
	}
}

# Final Multiplier (Modifier base is -0.1)
migration_crowding_mult = {
	if = {
		limit = { migration_crowding_ratio <= 0 }
		value = 0
	}
	# Gradual Phase (1x to 10x): 4.5 * r^2
	# At ratio 1.0 (10x density): hits 4.5 mult (45% penalty).
	# Slope at ratio 1.0 is 9.0.
	else_if = {
		limit = { migration_crowding_ratio < 1 }
		value = migration_crowding_ratio
		multiply = migration_crowding_ratio # ratio^2
		multiply = 4.5
	}
	# Linear Phase (10x+): 9 * r - 4.5
	# At ratio 1.0: 9 - 4.5 = 4.5 (Continuous).
	# At ratio 2.11 (20x density): 9 * (19/9) - 4.5 = 14.5 (145% penalty).
	# Slope is constant 9.0 (Continuous with gradual phase).
	else = {
		value = migration_crowding_ratio
		multiply = 9
		subtract = 4.5
	}

	min = 0
}

# Actual migration pull reduction percentage
migration_crowding_pull_pct = {
	value = migration_crowding_mult
	multiply = 0.1
}

# Country scope. Subjugation strength: read by force_become_subject's possible
# clause. Score >= 1.0 lets the actor subjugate scope:target_country (which
# must be a fellow bloc member; the action's potential clause guarantees that).
#
# Tuning intent (vanilla baseline was 5x prestige — too generous given how
# easily prestige is amassed):
#   - Base: prestige / target.prestige / 20  →  1.0 at 20x prestige (no bonuses)
#   - Sovereign Empire: +0.5 universal       →  10x prestige anywhere
#   - Per-identity matched: +0.4 in domain   →  ~12x prestige (axis-aligned)
#   - identity_diplomatic: no bonus on purpose
te_subjugation_strength = {
	value = 0
	add = {
		value = prestige
		divide = { value = scope:target_country.prestige min = 1 }
		divide = 20
	}
	add = modifier:country_subjugation_strength_add

	# Sovereign Empire — universal (no axis match needed)
	if = {
		limit = { power_bloc ?= { has_identity = identity:identity_sovereign_empire } }
		add = modifier:country_subjugation_sovereign_strength_add
	}

	# Religious — same state religion
	if = {
		limit = {
			power_bloc ?= { has_identity = identity:identity_religious }
			religion = scope:target_country.religion
		}
		add = modifier:country_subjugation_religious_strength_add
	}

	# Ideological Union — same governance principle (chiefdom/monarchy/republic/etc.)
	if = {
		limit = {
			power_bloc ?= { has_identity = identity:identity_ideological_union }
			te_same_governance_principle = { TARGET = scope:target_country }
		}
		add = modifier:country_subjugation_ideological_strength_add
	}

	# Cultural — actor and target share at least one primary culture
	if = {
		limit = {
			power_bloc ?= { has_identity = identity:identity_cultural }
			any_primary_culture = {
				save_temporary_scope_as = te_subj_pc
				scope:target_country = { country_has_primary_culture = scope:te_subj_pc }
			}
		}
		add = modifier:country_subjugation_cultural_strength_add
	}

	# Trade League — target is at least "Reliant" (economic_dependence >= 3) on actor
	if = {
		limit = {
			power_bloc ?= { has_identity = identity:identity_trade_league }
			te_target_economically_subordinate = { TARGET = scope:target_country }
		}
		add = modifier:country_subjugation_trade_strength_add
	}

	# Military Treaty Organization — actor's army at least 5x target's
	if = {
		limit = {
			power_bloc ?= { has_identity = identity:identity_military_treaty_organization }
			te_target_militarily_weaker = { TARGET = scope:target_country }
		}
		add = modifier:country_subjugation_military_strength_add
	}
}

# === Tiered scaler script values for events ============================
# Used by add_modifier { multiplier = sv_<resource>_event_<tier> } so the
# resulting modifier scales with the country's current resource value
# instead of being a hardcoded delta that goes invisible at late-game scale.
#
# See docs/engine/event_magnitude_report.md for the audit that flags missing usage,
# and docs/guides/scripting_best_practices.md § "Fast-scaling resources require
# scaled effects" for the rationale.
#
# Tiers: tiny=0.5%, small=2%, medium=5%, large=15%, huge=30%

sv_prestige_event_tiny   = { value = prestige  multiply = 0.005 }
sv_prestige_event_small  = { value = prestige  multiply = 0.02  }
sv_prestige_event_medium = { value = prestige  multiply = 0.05  }
sv_prestige_event_large  = { value = prestige  multiply = 0.15  }
sv_prestige_event_huge   = { value = prestige  multiply = 0.30  }

sv_treasury_event_large  = { value = gold_reserves  multiply = 0.15  }

# Money-flow (weekly): pair with a static modifier carrying
# country_expenses_add = 1, country_tax_income_add = 1, or
# country_minting_add = 1, applied as
#   add_modifier = { name = X multiplier = sv_money_flow_event_<tier> }
# Same shape as banking_event_expense_<tier> (which stays for banking
# events; sv_money_flow_event_<tier> is the generic name to prefer for
# new content). Tiers are weekly fractions of GDP.
sv_money_flow_event_small  = { value = gdp  multiply = 0.0001 }   # 0.01% GDP / week
sv_money_flow_event_medium = { value = gdp  multiply = 0.0002 }   # 0.02% GDP / week
sv_money_flow_event_large  = { value = gdp  multiply = 0.0004 }   # 0.04% GDP / week
sv_money_flow_event_huge   = { value = gdp  multiply = 0.0006 }   # 0.06% GDP / week

# Population-transfer disruption strength: var:pop_transfer_count / total_population,
# clamped. Must be a top-level named script value — `divide = total_population` as a
# bareword silently resolves to 'none' inside an inline anonymous script-value block
# (the value = { … } form on set_variable), but works in a named definition. Vanilla
# precedent: country_urbanization_rate. Caller: country scope with pop_transfer_count
# already set on the same scope.
te_pop_transfer_disruption_strength = {
	value = var:pop_transfer_count
	if = {
		limit = { total_population > 0 }
		divide = total_population
	}
	min = 0.25
	max = 2
}
