# ---- Fabrication (design § 5) ------------------------------------------------------
# Every module's points; at most this many modules a month.
sr_module_points = { value = 100 }
sr_mod_fab_per_month = { value = 2 }
# A module for a world with launch windows is built only once its window is open
# or this many months off (sr_mq_head_ready, the generated ready test; the owner,
# 2026-11-23): until then it waits in the queue, its points in the stock, and a
# module further down that can go is built ahead of it (sr_mq_promote_ready), so
# a Mercury module never waits behind a Mars one. A window is three months and a
# module built in its last month misses it, so at sr_mod_fab_per_month a world
# takes about (lead + 2) x 2 modules a window: 6 at a lead of 1, 8 at 2.
sr_mod_build_lead_months = { value = 1 }
# Country scope. The months until the queue's first module is built for its
# window (0 once its window is that near, or for a world with none).
sr_mq_head_window_months = {
	value = sr_mq_head_wait_months
	subtract = sr_mod_build_lead_months
	min = 0
}

# Country scope. The module stock, the points kept while fabrication stalls (the
# owner, 2026-11-15): a base that holds the fourth module past a world's slots
# (5 x sr_module_points, colony management design § 5), a year of this month's
# fabrication (the methods and Orbital Yards pay for their own storage), and the
# Space Program's Module Storage method (country_sr_module_storage_add,
# sr_colony_module_pms.txt). A module past a world's slots is queued only while
# the stock can hold its price (sr_mod_can_queue_<type>_<key>). sr_mod_month adds
# no points past it (or past the queue's first module's price, where that is
# more: sr_mod_fab_stock_fill_cap), but never takes away a stock above it.
sr_mod_fab_stock_base = { value = 500 }
sr_mod_fab_stock_months = { value = 12 }
sr_mod_fab_stock_cap = {
	value = sr_mod_fab_stock_base
	add = sr_mod_fab_stock_year
	add = sr_mod_fab_storage
	floor = yes
}
sr_mod_fab_stock_year = {
	value = sr_mod_fab_rate
	multiply = sr_mod_fab_stock_months
}
sr_mod_fab_storage = {
	value = modifier:country_sr_module_storage_add
}
# Country scope. What sr_mod_month fills the stock to: its cap, or the queue's
# first module's price where that is more, so a module queued while the cap was
# higher (more fabrication then, or a storage method since switched off) is still
# built rather than stopping the queue (review of #1847).
sr_mod_fab_stock_fill_cap = {
	value = sr_mod_fab_stock_cap
	if = {
		limit = { has_variable = sr_mq_1_site }
		min = sr_mq_head_cost
	}
}

# Dismantling (owner, 2026-10-31): the months a module level takes to come
# down, one level at a time a world, so a colony can't be stripped bare when a
# rival moves on it. A blockade holds the count; losing the world ends it.
sr_mod_dismantle_months = { value = 12 }

# Country scope. The month's points: the Space Program's fabrication method
# (te_colony_module_pms.txt), workforce-scaled, and the colonies' Orbital Yards
# (colony management design § 7).
sr_mod_fab_rate_method = {
	value = modifier:country_sr_module_fabrication_add
}
sr_mod_fab_rate = {
	value = sr_mod_fab_rate_method
	add = { value = sr_mod_units_yard multiply = sr_mod_yard_points }
}

# ---- The AI's fabrication budget (the owner, 2026-11-20) ---------------------------
# An AI runs the dearest fabrication method whose bill is within this share of
# its government's weekly fixed income, half the share a mission may climb to
# (sr_ai_mission_share_up, sr_ai_mission_budget_values.txt): the two are paid
# together. Each later method costs about 10% less a point than the one before
# (sr_colony_module_pms.txt), so a step down slows the modules more than it
# saves on each. Its cheapest method runs within any budget, as every method
# did before (sr_mod_activate_best_fabrication).
sr_mod_ai_fab_share = { value = 0.05 }
sr_mod_ai_fab_budget = {
	value = fixed_income
	multiply = sr_mod_ai_fab_share
}

# The methods' weekly bills at full staff and base prices, their goods times
# their base prices (sr_colony_module_pms.txt; test_solar_sites.py holds them
# equal). The ladder below gates only the two dearer; the AI's settlers read all
# three (sr_mod_ai_fab_bill_now).
sr_mod_fab_bill_base_orbital_assembly = { value = 340000 }
sr_mod_fab_bill_base_lunar_foundries = { value = 732000 }
sr_mod_fab_bill_base_fusion_shipyards = { value = 1368000 }

# What the bill costs now, as the mission budget prices a mission's: the
# program's throughput with its funding (its goods scale with it), times our
# market's Launch Capacity premium, never below base (sr_launch_capacity_price_up,
# space_race_values.txt). Launch Capacity is 70% to 76% of every method's bill;
# the premium stands in for the rest. An estimate: an understaffed programme
# buys less.
sr_mod_fab_bill_price_factor = {
	value = sr_ai_program_throughput_funded
	multiply = {
		value = 1
		add = sr_launch_capacity_price_up
	}
}

# The bill of the method the Space Program runs now, at today's prices; 0 with
# none. What it leaves of the budget is the AI's settlers' (sr_emig_ai_room,
# colony people design § 9; the owner: "Share fabrication's 5%").
sr_mod_ai_fab_bill_now = {
	value = 0
	if = { limit = { any_scope_building = { is_building_type = building_space_program has_active_production_method = pm_sr_fabrication_fusion_shipyards } } value = sr_mod_fab_bill_base_fusion_shipyards }
	else_if = { limit = { any_scope_building = { is_building_type = building_space_program has_active_production_method = pm_sr_fabrication_lunar_foundries } } value = sr_mod_fab_bill_base_lunar_foundries }
	else_if = { limit = { any_scope_building = { is_building_type = building_space_program has_active_production_method = pm_sr_fabrication_orbital_assembly } } value = sr_mod_fab_bill_base_orbital_assembly }
	multiply = sr_mod_fab_bill_price_factor
}

# What each method's bill leaves of the budget: 0 or more, the AI may run it.
sr_mod_ai_fab_room_lunar_foundries = {
	value = sr_mod_ai_fab_budget
	subtract = {
		value = sr_mod_fab_bill_base_lunar_foundries
		multiply = sr_mod_fab_bill_price_factor
	}
}
sr_mod_ai_fab_room_fusion_shipyards = {
	value = sr_mod_ai_fab_budget
	subtract = {
		value = sr_mod_fab_bill_base_fusion_shipyards
		multiply = sr_mod_fab_bill_price_factor
	}
}

# ---- Extraplanetary Base room (design § 9) --------------------------------------------
# Country scope. Its Extraplanetary Base levels, built, and built and queued.
# Each state's base is the engine's b: lookup, which also finds one whose
# levels are all queued, never a walk of the state's buildings: the base's
# can_build_private reads the queued count, and the private investment pass
# reads that in every state (scripting_best_practices.md § "A Building's
# `possible` Runs in Every State for Private Investment, Before
# `can_build_private`"). The Resort Colony's below do the same.
sr_ebase_levels = {
	value = 0
	every_scope_state = {
		limit = { exists = b:building_space_mine }
		add = b:building_space_mine.level
	}
}

sr_ebase_queued_levels = {
	value = 0
	every_scope_state = { # REVIEWED 2026-11-22: investors must obey the room, so it stays in can_build_private; one b: lookup a state, 36 reads in a week of the 2024 USA
		limit = { exists = b:building_space_mine }
		add = b:building_space_mine.level_after_queued_constructions
	}
}
