# The weekly innovation cap is the modifier's sum: 50 base and 150 at full literacy (vanilla's code static
# modifiers), China's 134.5 at a literacy of 0.56 in the AI-only run of 2026-11-16.
te_rgc_innovation_cap = { value = modifier:country_weekly_innovation_max_add }
te_rgc_innovation_share = {
	value = 1
	if = {
		limit = { te_rgc_innovation_cap > 0 }
		value = country_innovation
		divide = te_rgc_innovation_cap
	}
}

	}
