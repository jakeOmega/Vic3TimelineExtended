# The Nation tab's rows (gui/te_nation_tab.gui). Country scope, read through
# GetPlayer.MakeScope. Each is safe before its system has first run. The
# lists' counts are generated (te_nation_generated_values.txt).

# Legislative Override Capacity, the raw modifier sum (its sources:
# concept_legislative_override_capacity).
te_nation_override_capacity = {
	value = modifier:country_legislative_override_capacity_add
}

# Its bar: 0 to +30 as 0 to 1. Every value below 0 plays the same: the
# Forceful Legislation event needs 0 (minor_events_timelineextended.2), its
# options and the tax code's Force Through need 2 or 4, and the authority
# discount (forced_law_through_event_capacity_discount) starts at 2. So the
# bar starts at 0, empty there and below, and the row's figure keeps the true
# number (owner, 2026-10-17: the sub-zero stretch was wasted space). 30 is
# about the most a country holds at once: the strongest law of each of the 15
# groups that carry it (20), the three institutions at full investment (3.5),
# Creative Legislature V (4) and the two surveillance technologies (2).
te_nation_override_bar = {
	value = te_nation_override_capacity
	divide = 30
	min = 0
	max = 1
}

# The strongest Forceful Legislation option the capacity alone opens
# (minor_events_timelineextended.2 in events/minor_events.txt): 0 none (below
# 0), 1 expedite, 2 push decisively (at 2), 3 force it through (at 4).
te_nation_override_cap_tier = {
	value = 0
	if = {
		limit = { te_nation_override_capacity >= 4 }
		value = 3
	}
	else_if = {
		limit = { te_nation_override_capacity >= 2 }
		value = 2
	}
	else_if = {
		limit = { te_nation_override_capacity >= 0 }
		value = 1
	}
}

# The strongest option open now: the event also needs more authority
# produced than each option costs (its trigger and options' triggers).
te_nation_override_tier = {
	value = 0
	if = {
		limit = {
			te_nation_override_capacity >= 4
			produced_authority > forced_law_through_event_authority_cost_large
		}
		value = 3
	}
	else_if = {
		limit = {
			te_nation_override_capacity >= 2
			produced_authority > forced_law_through_event_authority_cost_medium
		}
		value = 2
	}
	else_if = {
		limit = {
			te_nation_override_capacity >= 0
			produced_authority > forced_law_through_event_authority_cost_small
		}
		value = 1
	}
}

# The authority the next option needs, while authority, not capacity, holds
# the tier below what the capacity opens; 0 otherwise.
te_nation_override_next_cost = {
	value = 0
	if = {
		limit = { te_nation_override_cap_tier > te_nation_override_tier }
		if = {
			limit = { te_nation_override_tier = 0 }
			value = forced_law_through_event_authority_cost_small
		}
		else_if = {
			limit = { te_nation_override_tier = 1 }
			value = forced_law_through_event_authority_cost_medium
		}
		else = {
			value = forced_law_through_event_authority_cost_large
		}
	}
}

# Global Technological Development: the technology spread the yearly pulse
# last applied (tech_spread_on_action), 0 before it. Never the sweep itself
# (global_tech_spread_script_value), which the GUI would rerun every frame.
te_nation_tech_spread = {
	value = 0
	if = {
		limit = { has_variable = te_global_tech_spread_applied }
		value = var:te_global_tech_spread_applied
	}
}

# Judicial Independence (te_judiciary_values.txt): 1 once the score is
# seeded (the first monthly pulse; an old save's second), whose row shows
# te_jud_display heading for te_jud_ceiling.
te_nation_judiciary_on = {
	value = 0
	if = {
		limit = { has_variable = te_judicial_independence }
		value = 1
	}
}

# 1 while a paper court's line (a government that can ignore its courts)
# cuts the ceiling, so the row names both of the ceiling's sources.
te_nation_jud_paper = {
	value = 0
	if = {
		limit = { "modifier:country_judicial_independence_ceiling_mult" < 0 }
		value = 1
	}
}

# 1 under Algorithmic Governance, whose mandate the row shows.
te_nation_mandate_on = {
	value = 0
	if = {
		limit = { has_law = law_type:law_algorithmic_governance }
		value = 1
	}
}

# Construction cost scaling: what it adds to a point's cost on its own, as the yearly pulse last set it (the GDP
# part x the share that applies, te_cc_scaling_factor - 1), 0 before it. Not the modifier's own value
# (var:prior_construction_scaling), which is raised so that it multiplies with the other construction-cost
# modifiers; the row's tooltip shows both factors.
te_nation_construction_cost = {
	value = te_cc_scaling_factor
	subtract = 1
}

# 1 while excess private construction or Overinvestment is in force.
te_nation_private_construction = {
	value = 0
	if = {
		limit = {
			OR = {
				has_modifier = too_much_private_construction
				has_modifier = overinvestment_modifier
			}
		}
		value = 1
	}
}
