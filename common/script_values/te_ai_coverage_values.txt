# AI coverage of the mod's own content (#1822): the sizes no vanilla value forces, named so they can be tuned after
# an observer run. docs/systems/mod_systems/ai.md § AI Coverage of the Mod's Content has the reasoning and the
# vanilla anchors; docs/testing/ai-economy-playtest.md § AI coverage of the mod's content what a save should show.

# ---- Buildings (ai_value is read in state scope: `owner` is the country deciding) ----

# A power plant the mod adds: hydro, nuclear, renewable, fusion (common/buildings/06_zz_te_clean_power.txt).
# Vanilla's building_power_plant scores 1000, plus 4000 in a state with no power plant built or under way, because
# Electricity is a local good: a state with none can't run a method that uses it. The mod's plants had no ai_value,
# so only the coal and oil plant earned the 4000. They now score the same way, and "a power plant" means any of
# them (te_state_has_a_power_plant), for vanilla's plant too (its restated ai_value, extra_buildings.txt), so the
# first plant of any kind takes the bonus and a second kind doesn't earn it again.
te_ai_power_plant_base_value = 1000
te_ai_first_power_plant_bonus = 4000
te_clean_power_plant_ai_value = {
	value = te_ai_power_plant_base_value
	if = {
		limit = { te_state_has_a_power_plant = no }
		add = te_ai_first_power_plant_bonus
	}
}

# Highway, Airport, Network Infrastructure (common/buildings/11_zz_te_infrastructure.txt): vanilla's railway base,
# 2000, twice a plain building's 1000. Highways and airports make what a railway makes (infrastructure and
# Transportation); Network Infrastructure makes the local good Digital Access. Parity, not a cure: in a 2058 save
# the AI world already held 528 highways, 464 airports and 522 network buildings beside 580 railways.
te_infrastructure_ai_value = 2000

# The mod's wonders, the UN Headquarters and a Power Bloc Headquarters (common/buildings/08_zz_te_wonders.txt):
# vanilla's value for every monument a country can build (the Eiffel Tower, Big Ben, the Statue of Liberty).
# Without one they scored a government building's base, 1000; 16 of the 37 wonders stood in that 2058 save.
te_wonder_ai_value = 10000

# A megaproject's construction site and the Antimatter Warhead Plant (common/buildings/extra_buildings.txt). The
# engine's base for a government building, which is what they scored with no ai_value, so a solvent country
# behaves as before; nothing for a country in default or under a declared bankruptcy, since a site's inputs run
# to hundreds of millions a week and the government pays for them.
te_ai_megaproject_base_value = 1000
te_megaproject_ai_value = {
	value = te_ai_megaproject_base_value
	if = {
		limit = {
			owner = {
				OR = {
					in_default = yes
					has_modifier = declared_bankruptcy
				}
			}
		}
		multiply = 0
	}
}
