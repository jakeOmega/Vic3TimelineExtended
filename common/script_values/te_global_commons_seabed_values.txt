# ---- Pools (§ 2.2), in Deep-Sea Mine levels: first estimates ----
te_seabed_pool_pacific = 300
te_seabed_pool_atlantic = 250
te_seabed_pool_indian = 200
te_seabed_pool_arctic = 100
te_seabed_pool_southern = 150

# The reserved half of each pool under the Regulated regime (§ 3.2), rounded
# down; the open half is the rest.
te_seabed_pool_reserved_pacific = {
	value = te_seabed_pool_pacific
	divide = 2
	floor = yes
}
te_seabed_pool_reserved_atlantic = {
	value = te_seabed_pool_atlantic
	divide = 2
	floor = yes
}
te_seabed_pool_reserved_indian = {

	divide = 2

te_seabed_pool_reserved_arctic = {

	divide = 2
	floor = yes
}
te_seabed_pool_reserved_southern = {
	value = te_seabed_pool_southern
	divide = 2
	floor = yes
}

# ---- Claim weight (§ 2.3): first estimates ----
# Home basin: (coast weight × incorporated coastal states there + port levels
# there + navy × its share of the country's incorporated coastal states)
# × policy factor. Distant basin: distant navy share × navy × policy factor.
te_seabed_coast_weight = 2
# An unincorporated coastal state's count in the coast term and the navy's
# spread (seabed claims spec § 5.1); its ports count in full.
te_seabed_uninc_coast = 0.5
te_seabed_distant_navy_share = 0.05
# Added to Southern weight by an Antarctic research station (§ 4), after the
# policy factor.
te_seabed_station_weight = 5
# Unilateral Licensing, or the Scramble regime for everyone: navy × this.
te_seabed_assertive_navy_mult = 2
# The navy term is naval power projection ÷ this (seabed claims spec § 6.1), so
# a typical fleet's term stays near its old count of ships (navy_size) beside
# the coast and port terms. Each ship projects its ship_power_projection_add:
# 35-465 for vanilla's hulls, 75-1,350 for the mod's era-10 to 12 ones
# (common/ship_types/extra_ship_types.txt), so an era-11 fleet, when Abyssal
# Plain Mining comes, averages about 300 a ship. A first estimate: set it from
# the median ratio the seabed_power_projection self-check logs
# (docs/testing/global-commons-playtest.md).
te_seabed_power_projection_scale = 300
# Policy factors (§ 2.5).
te_seabed_policy_factor_survey = 0.75
te_seabed_policy_factor_sponsor = 1
te_seabed_policy_factor_unilateral = 1
# A coastal country without Abyssal Plain Mining is weighed at this in place
# of a policy factor, and its share waits for it (model: LATENT_FACTOR, the
# owner 2026-10-27): every sharing weighs every coastal country, so the first
# claimant wins only what its weight does against everyone's, at most
# 1 / this times its share once all have the technology on one policy. The
# race's dial, with Survey's factor beside it: 0 hands the first claimant
# every pool, 1 leaves no head start. 0.75, not the first 0.25 (owner,
# 2026-10-27: Sponsor Contractors' 4x over a waiting country was too large).
# Survey's factor, so researching the technology moves nothing until a
# policy is chosen.
te_seabed_latent_factor = 0.75

# ---- Policy (§ 2.5; seabed claims spec §§ 2.3-2.4) ----
# Influence held while the policy stands: unit modifiers
# (te_seabed_policy_sponsor_cost / te_seabed_policy_unilateral_cost) applied
# with the cost as multiplier (te_seabed_refresh_policy_cost). Sponsor
# Contractors: the base plus so much per claim it protects, uncapped
# ("claiming the whole basin should be expensive"). Unilateral Licensing:
# Sponsor's figure × the multiplier, and at least its own base. First
# estimates, to tune in the play-test.
te_seabed_influence_sponsor = 50
te_seabed_influence_per_claim = 0.5
te_seabed_influence_unilateral = 150
te_seabed_unilateral_cost_mult = 1.5
# Any change of policy holds it this many months (the AI's too); the first
# choice is free.
te_seabed_policy_lock_months = 60
# Relations a year with each other Unilateral claimant sharing a home basin.
# Applied as half from each side (te_seabed_unilateral_relations), since
# relations are one figure for the pair.
te_seabed_unilateral_relations = -10
te_seabed_unilateral_relations_half = {
	value = te_seabed_unilateral_relations
	divide = 2
}

# ---- Sponsor Contractors' floors (seabed claims spec § 2.2) ----
# At the year's first sharing a claimant on Sponsor Contractors keeps at least
# this much of its claims in each basin (model: FLOOR_KEEP, set_floors).
te_seabed_floor_keep = 0.75

# ---- The holding fee (seabed claims spec § 2.1) ----
# A week, on each licence whose level has stayed unbuilt since the year's
# first sharing before last: this share of a level's base output value
# (te_seabed_level_output_value), half the royalty (model: FEE_RATE).
te_seabed_fee_rate = 0.02

# ---- Long-haul (§ 2.4) ----
# Display figure: te_seabed_long_haul (te_global_commons_seabed_modifiers.txt)
# carries building_throughput_add = -0.25 at multiplier 1, on the Deep-Sea Mine.
te_seabed_long_haul_penalty_pct = 25

# The state's holding-fee base (model: fee_base_state): licences on levels
# already unbuilt at the last two first-sharings of a year, the smallest of
# the count now and its two snapshots (te_seabed_take_snapshots). A state
# without both snapshots (an old save, a split, a revolution's first year)
# pays nothing yet. The snapshots are state variables, so they pass with the
# state.
te_seabed_state_fee_base = {
	value = 0
	if = {
		limit = {
			has_variable = te_seabed_unb_snap1
			has_variable = te_seabed_unb_snap2
		}
		value = te_seabed_state_unbuilt_lic
		max = var:te_seabed_unb_snap1
		max = var:te_seabed_unb_snap2
		min = 0
	}
}
