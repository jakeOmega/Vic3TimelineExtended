# External banking tools. Country scope. Money is pounds; GDP is annual.
# Trade volumes are engine comparison triggers, so export reliance is banded.
# These are policy abstractions, not a second foreign-exchange balance sheet.
banking_external_inflow_factor = { value = 0.5 }
banking_external_sterilized_factor = { value = 0.25 }
# 0.5 / 24, rounded up so the cap is reached at month 24. Keep literals to
# five decimals: 0.020833 logged "Badly read script value" at load.
banking_external_fx_protection_step = { value = 0.02084 }
banking_external_fx_protection_cap = { value = 0.5 }
banking_external_sterilization_share = { value = 0.0001 }
banking_external_surrender_share = { value = 0.0005 }
banking_external_import_share = { value = 0.0005 }
banking_external_import_fatigue_cap = { value = 12 }

# No current policy = no new flow restraint; positive side only. Eligibility
# matters immediately after a law change, before monthly cleanup removes a tool.
banking_external_inflow_damp = {
	value = 1
	if = {
		limit = { banking_external_eligible_restrict_inflows = yes banking_tool_restrict_inflows_active = yes }
		value = banking_external_inflow_factor
	}
}
banking_external_gold_pressure_factor = {
	value = 1
	if = {
		limit = { banking_external_eligible_sterilize_inflows = yes banking_tool_sterilize_inflows_active = yes has_variable = banking_external_sterilization_funded var:banking_external_sterilization_funded = 1 }
		value = banking_external_sterilized_factor
	}
}
# The stock of safer debt survives a repeal, revolution or temporary law change;
# it unwinds at the same speed as it accumulated. Ineligible regimes get no benefit.
banking_external_fx_exposure = {
	value = 1
	if = {
		limit = { banking_external_available = yes has_variable = banking_external_fx_protection }
		subtract = var:banking_external_fx_protection
	}
	min = 0.5
	max = 1
}
banking_external_fx_protection_pct = {
	value = 1
	subtract = banking_external_fx_exposure
	multiply = 100
}
banking_external_export_share = {
	value = 0
	if = {
		limit = { exists = market market = { market_exports_reliance = { value > 0.3 } } }
		value = 1
	}
	else_if = {
		limit = { exists = market market = { market_exports_reliance = { value > 0.1 } } }
		value = 0.5
	}
	else_if = {
		limit = { exists = market market = { market_exports_reliance = { value > 0 } } }
		value = 0.25
	}
}
banking_external_sterilization_cost = {
	value = gdp
	multiply = banking_external_sterilization_share
	min = 0
}
# A month's sterilization bill, charged a quarter of it a week as a government
# expense (banking_external_charge_sterilization, owner 2026-10-27). The
# multiplier is evaluated in the entry's scope, so it reads the owner.
banking_external_sterilization_weekly_charge = {
	value = banking_external_sterilization_cost
	divide = 4
}
banking_external_sterilization_weekly_charge_for_je = {
	value = 0
	owner = { add = banking_external_sterilization_weekly_charge }
}
banking_external_surrender_purchase = {
	value = gdp
	multiply = banking_external_surrender_share
	multiply = banking_external_export_share
	max = te_mon_gold_inflow_room
	max = banking_external_purchase_cash
	min = 0
}
# Debit the snapshot, not a recomputed amount after the vault has changed.
banking_external_surrender_payment = {
	value = 0
	if = {
		limit = { has_variable = banking_external_purchase }
		subtract = var:banking_external_purchase
	}
}
banking_external_import_fatigue = {
	value = 0
	if = {
		limit = { has_variable = banking_external_import_months }
		add = var:banking_external_import_months
	}
	min = 0
	max = banking_external_import_fatigue_cap
}
banking_external_import_cost = {
	value = banking_external_import_fatigue
	divide = banking_external_import_fatigue_cap
	add = 1
	multiply = banking_external_import_share
	multiply = gdp
	min = 0
}
# ...and import financing's, at this month's escalated cost.
banking_external_import_weekly_charge = {
	value = banking_external_import_cost
	divide = 4
}
banking_external_import_weekly_charge_for_je = {
	value = 0
	owner = { add = banking_external_import_weekly_charge }
}

banking_external_real_gap = {
	value = 0
	if = {
		limit = { has_variable = te_policy_rate has_variable = te_mon_world_rate_now }
		add = var:te_policy_rate
		subtract = te_mon_fx_inflation_read
		subtract = var:te_mon_world_rate_now
	}
}

banking_external_combined_import_cost = {
	value = banking_external_import_cost
	if = {
		limit = { banking_tool_sterilize_inflows_active = yes banking_external_eligible_sterilize_inflows = yes }
		add = banking_external_sterilization_cost
	}
}

# The cash FX Surrender may spend: what the treasury holds less a month of the
# other two tools' bills, which their weekly charges will draw.
banking_external_purchase_cash = {
	value = gold_reserves
	if = {
		limit = { banking_tool_sterilize_inflows_active = yes }
		subtract = banking_external_sterilization_cost
	}
	if = {
		limit = { banking_tool_emergency_import_financing_active = yes }
		subtract = banking_external_import_cost
	}
	min = 0
}

# What FX Surrender pays from the treasury this month for the bank's gold, once
# a month and outside the weekly budget (banking_external_purchase_reserves):
# the Breakdown's Banking row names it (GUI audit BNK-14). The other two
# tools' bills are weekly expenses, in the budget. Display only.
banking_external_surrender_purchase_display = {
	value = 0
	if = {
		limit = { banking_tool_fx_surrender_active = yes }
		value = banking_external_surrender_purchase
	}
}
