# ============================================================================
# GLOBAL COMMONS — SEABED LICENCES: how the AI chooses its seabed policy
# ============================================================================
# Read ONLY by te_seabed_ai_choose_policy (te_global_commons_seabed_effects.txt),
# yearly from on_yearly_pulse_country for AI claimants: the AI holds the policy
# with the highest score (§ 2.8). Nothing here gates a player's choice; the
# panel's buttons have their own conditions (te_global_commons_seabed_sguis.txt).
# Building needs no AI code: the engine's builder obeys the gate (§ 2.6).
#
# Scope: country, with ROOT = the same country (the yearly country pulse).
#
#   policy        when it scores
#   Survey        always te_seabed_ai_will_survey: the default
#   Sponsor       the policy would bring at least one licence AND Tech-Critical
#                 Metals sell above base price at home: 30, +40 per 100% the
#                 price stands above base, at most +20
#   Unilateral    the same licence test, then by rank (great +25, major +10,
#                 others -20), UN standing (poor or disgraced +15, respected
#                 -15, exemplary -60: a large weight, not a ban), a rival that
#                 also claims +10, Tech-Critical Metals above 1.25x base +15,
#                 an ITLOS party under the Regulated regime -15 (defection
#                 costs, § 3.3)
#
# The policy held now gets te_seabed_ai_stay_bonus, a band wider than any one
# term that flips on its own (the rivalry, the standing tiers' edges), so a
# country near a boundary doesn't change policy every year. A policy that
# holds influence scores only while the AI can spare it: the cost plus
# te_seabed_ai_influence_spare to take it up, and it is dropped once influence
# runs into deficit (te_seabed_ai_influence_deficit_pull).
# ============================================================================

te_seabed_ai_will_survey = 20
te_seabed_ai_stay_bonus = 10
te_seabed_ai_influence_spare = 100
te_seabed_ai_influence_deficit_pull = 50
# Sponsor Contractors' pull per unused claim it would protect, and its cap.
te_seabed_ai_protect_per_claim = 0.1
te_seabed_ai_protect_max = 20

# ---- The licence test: what Sponsor Contractors would bring ----
# Σ over basins of pool × w' ÷ (W − w + w'), w' = w × r: the country's weight
# at the Sponsor factor against last year's world weight, r = the Sponsor
# factor over the current one. An estimate, for the AI only.
te_seabed_ai_ratio = {
	value = te_seabed_policy_factor_sponsor
	divide = te_seabed_policy_factor
}

te_seabed_ai_projected_pacific = {
	value = 0
	if = {
		limit = {
			has_variable = te_seabed_w_pacific
			var:te_seabed_w_pacific > 0
			has_global_variable = te_seabed_world_w_pacific
		}
		value = te_seabed_pool_pacific
		multiply = var:te_seabed_w_pacific
		multiply = te_seabed_ai_ratio
		divide = {
			value = te_seabed_ai_ratio
			subtract = 1
			multiply = var:te_seabed_w_pacific
			add = global_var:te_seabed_world_w_pacific
			min = 1
		}
	}
}

	add = te_seabed_ai_projected_arctic

# How far above base Tech-Critical Metals sell in the country's market, as a
# fraction (0 at or below base). The block form, as vanilla reads market
# goods from country scope (scripting_best_practices.md, market_goods_pricier).
te_seabed_ai_tech_metals_premium = {
	value = 0
	market = {
		mg:tech_metals = { add = market_goods_pricier }
	}
	min = 0
}

# ---- The wills ----

te_seabed_ai_will_sponsor = {
	value = 0
	if = {
		limit = {
			te_seabed_ai_projected >= 1
			te_seabed_ai_tech_metals_premium > 0
		}
		value = 30
		add = {
			value = te_seabed_ai_tech_metals_premium
			multiply = 40
			max = 20
		}
	}
	# Protection (seabed claims spec § 2.2): the more unused claims of its own,
	# the more Sponsor Contractors keeps of them each January.
	if = {
		limit = { te_seabed_ai_projected >= 1 }
		add = {
			value = te_seabed_protected_claims
			multiply = te_seabed_ai_protect_per_claim
			max = te_seabed_ai_protect_max
		}
	}
}

te_seabed_ai_will_unilateral = {
	value = 0
	if = {
		limit = { te_seabed_ai_projected >= 1 }
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = 25
		}
		else_if = {
			limit = { country_rank >= rank_value:major_power }
			add = 10
		}
		else = {
			add = -20
		}
		if = {
			limit = {
				OR = {
					un_standing_tier_poor = yes
					un_standing_tier_disgraced = yes
				}
			}
			add = 15
		}
		else_if = {
			limit = { un_standing_tier_respected = yes }
			add = -15
		}
		else_if = {
			limit = { un_standing_tier_exemplary = yes }
			add = -60
		}
		if = {
			limit = {
				any_country = {
					has_variable = te_seabed_shared
					has_diplomatic_pact = { who = ROOT type = rivalry }
				}
			}
			add = 10
		}
		if = {
			limit = { te_seabed_ai_tech_metals_premium > 0.25 }
			add = 15
		}
		if = {
			limit = {
				te_seabed_regime_regulated = yes
				te_seabed_itlos_party = yes
			}
			add = -15
		}
	}
}

# ---- The scores the choice compares ----

# Influence an AI must have to take up a policy that holds it.
te_seabed_ai_sponsor_influence_start = {
	value = te_seabed_sponsor_cost_value
	add = te_seabed_ai_influence_spare
}

te_seabed_ai_unilateral_influence_start = {
	value = te_seabed_unilateral_cost_value
	add = te_seabed_ai_influence_spare
}

te_seabed_ai_score_survey = {
	value = te_seabed_ai_will_survey
	if = {
		limit = { te_seabed_policy_survey = yes }
		add = te_seabed_ai_stay_bonus
	}
}

te_seabed_ai_score_sponsor = {
	value = te_seabed_ai_will_sponsor
	if = {
		limit = { te_seabed_policy_sponsor = yes }
		add = te_seabed_ai_stay_bonus
		if = {
			limit = { influence < 0 }
			subtract = te_seabed_ai_influence_deficit_pull
		}
	}
	else_if = {
		limit = {
			influence < te_seabed_ai_sponsor_influence_start
		}
		value = 0
	}
}

te_seabed_ai_score_unilateral = {
	value = te_seabed_ai_will_unilateral
	if = {
		limit = { te_seabed_policy_unilateral = yes }
		add = te_seabed_ai_stay_bonus
		if = {
			limit = { influence < 0 }
			subtract = te_seabed_ai_influence_deficit_pull
		}
	}
	else_if = {
		limit = {
			influence < te_seabed_ai_unilateral_influence_start
		}
		value = 0
	}
}

# ============================================================================
# Leases (seabed claims spec §§ 4, 8): Lease <Basin> Seabed Claims' AI
# ============================================================================
# Read by the articles' ai blocks (122_seabed_lease.txt). Scope: a country.
#   spare     own claims in the basin above what its building will want: the
#             levels it has queued now, as a year's pace (a level takes about
#             te_seabed_ai_lease_build_months to build, so the queue turns over
#             twice a year), plus a margin; with nothing queued, all but the
#             margin (spec § 4: no levels queued, or claims well above its pace)
#   offer     the spare, while the country isn't short of claims itself
#   request   10, +40 per 100% Tech-Critical Metals stand above base (at most
#             +20), in a home basin, while its room is at most the margin
# ============================================================================

te_seabed_ai_lease_pace_months = 12
te_seabed_ai_lease_build_months = 6
te_seabed_ai_lease_margin = 5

te_seabed_ai_lease_spare_pacific = {
	value = te_seabed_own_claims_pacific
	subtract = {
		value = te_seabed_queued_total_disp
		multiply = te_seabed_ai_lease_pace_months
		divide = te_seabed_ai_lease_build_months
		add = te_seabed_ai_lease_margin
	}
	min = 0
	floor = yes
}

te_seabed_ai_lease_offer_value_pacific = {
	value = 0
	if = {
		limit = { te_seabed_room > te_seabed_ai_lease_margin }
		value = te_seabed_ai_lease_spare_pacific
	}
}

te_seabed_ai_lease_request_value_pacific = {
	value = 0
	if = {
		limit = {
			te_seabed_is_claimant = yes
			has_variable = te_seabed_home_pacific
			NOT = { te_seabed_room > te_seabed_ai_lease_margin }
			te_seabed_ai_tech_metals_premium > 0
		}
		value = 10
		add = {
			value = te_seabed_ai_tech_metals_premium
			multiply = 40
			max = 20
		}
	}
}
