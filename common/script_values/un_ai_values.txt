# ============================================================================
# THE UNITED NATIONS — HOW THE AI DECIDES TO JOIN, PAY AND CONTRIBUTE
# ============================================================================
# Read ONLY by the `ai_chance` blocks of fourteen buttons in
# common/scripted_buttons/un_buttons.txt: Join / Leave, Withhold / Pay dues,
# Fund / Stop funding Development Programs and Raise / Reduce its size,
# Contribute to / End peacekeeping, Champion / Stop championing and
# Undermine / Stop undermining the order.
# Nothing here gates a player control. docs/systems/un_redesign_design.md §0.13
# has the review behind it, with every other AI decision point and its verdict.
#
# WHY A SCORE AND NOT PER-BUTTON TERMS. Each pair used to weigh a few laws and
# the UN's authority, and nothing a member gains or pays:
#   - Join scored 50 for nearly everyone, whatever the dues or the World
#     Development Fund were worth to it. An isolationist great power scored 15
#     to join and 20 to leave, so it joined, walked out, sat out the five-year
#     lockout and joined again. Every member scored 10 to leave below authority
#     20.
#   - At Strong and Supranational every member scored 5 / 15 to withhold its
#     dues from the tier alone, against 20 to pay again when healthy, so healthy
#     members toggled. Withholding ignored the Fund: a member it paid gave up a
#     grant larger than its dues to save them. Two of the owner's saves (2042
#     at Established, a later one at Supranational) had 55 of 95 and 67 of 88
#     members withholding.
#   - Development Programs and peacekeeping scored for any great power once
#     authority passed 40 and stopped only on laissez-faire, war or a collapsing
#     UN: ratchets (scripting_best_practices.md, "An Adopt Weight With No Repeal
#     Counterweight Is a Ratchet"), blind to a treasury in default.
#   - Any great power scored +20 to champion the order below authority 30 and
#     +15 to undermine it at 60, and stopped only at 80 and below 20, so a power
#     with every reason to back the UN undermined it once it grew strong. The
#     owner (2026-10-03): a mild pull from authority is fine, but a power with
#     good reasons either way should not switch on authority alone.
#
# Now each pair reads ONE score per country, and a line on it: the "on" button
# scores only at or above the line, the "off" button only a band below it.
# Between the two neither scores, so a country sits where its conditions put
# it, and how often the engine lets the AI click only sets how fast it gets
# there. The band is wider than any yes/no term that flips on its own (debt
# crossing 0.5 or 0.75, a state's devastation, rank slipping a step): a law, a
# default or the Fund's founding is meant to move a country across. One
# exception is deliberate: war ends peacekeeping (-30 against a band of 25), as
# it did before, and the programme's ten-year cooldown, which both its buttons
# respect, holds that to one round a decade.
#
# MONEY IN ONE UNIT. Every money term is a share of GDP a year, at
# un_ai_points_per_gdp_share points: 20 points for 1% of GDP. The levy is
# 0.1% / 0.2% / 0.4% / 1% of GDP at Contested / Established / Strong /
# Supranational (un_dues_values.txt), so -2 / -4 / -8 / -20. The Fund's grant
# (un_dev_fund_prospect_share, un_economy_values.txt) counts the same way, so a
# grant the size of the dues cancels them, and the term stops at +40: a grant of
# 2% of GDP a year is already a large part of a poor government's income, and
# past it the score would only drown every other reason. In the Supranational
# save nearly every member sits under the Fund's line, because one member holds
# almost all of the members' GDP; at Reform II the Fund there would pay several
# members more than their GDP, which is why the term is capped.
#
#   pair               score                     on at   off below   on weight      off weight
#   join / leave       un_ai_membership_will         0         -25   will + 5 (100)  -25 - will (50)
#   withhold / pay     un_ai_dues_will               0         -25   pay: will + 5 (60)  withhold: -25 - will (50)
#   development        un_ai_development_will       20          -5   will - 10 (50)  -5 - will (50)
#   peacekeeping       un_ai_peacekeeping_will      20          -5   will - 10 (50)  -5 - will (50)
#   champion           un_ai_order_will             20          -5   will - 10 (50)  -5 - will (50)
#   undermine          un_ai_dissent_will           20          -5   will - 10 (50)  -5 - will (50)
#
# The size of a Development Programs contribution is a seventh pair on its own
# score, un_ai_development_generosity, with a line per size held (§ Development
# Programs below).
#
# The order and dissent scores are one stance, signed: un_ai_dissent_will is
# minus un_ai_order_will. A power champions at +20 or more, undermines at -20
# or less, and does neither in between; each stops 25 past its own line.
#
# test_un_ai_values.py checks that each pair reads its own score and line, that
# the off line sits the band below the on line, and that the band is wider than
# every term listed in un_ai_self_flipping_terms below.
#
# Scope: country, throughout (the country the button belongs to; no term reads
# ROOT, so the debug dump can evaluate them for every country).
# ============================================================================

# ---- Shared tuning ----------------------------------------------------------

# 20 points for each 1% of GDP a year (a share of 0.01).
un_ai_points_per_gdp_share = { value = 2000 }

# The money term stops here: a grant of 2% of GDP a year.
un_ai_money_cap = { value = 40 }

# The largest yes/no term that flips without anyone deciding: debt crossing a
# line (10 and 15), devastation or poverty (10), famine (10), GDP per head
# crossing the great powers' average or half of it (15 each), rank slipping a
# step (20 and 15), a climate emitter's share (20), UN sanctions on us (20), a
# record crossing the condemnation line (10), authority's whole pull on the
# order stance (15). Every band below must be
# wider (test_un_ai_values.py reads this figure and each band). War on the
# peacekeeping score is the one exception (see the header).
un_ai_self_flipping_terms = { value = 20 }

# The UN's authority, 50 when there is none to read. Globals only.
un_ai_authority = {
	value = 0
	if = {
		limit = { has_global_variable = un_authority }
		add = global_var:un_authority
	}
	else = {
		add = 50
	}
}

# ---- Money: the levy and the Fund ------------------------------------------------

# What membership's money does for us: the Fund's grant (our own, or what we
# would draw as a paying member) less the levy, both as a share of GDP a year,
# at un_ai_points_per_gdp_share. At most un_ai_money_cap; at least the
# Supranational levy (-20).
un_ai_money_points = {
	value = un_dev_fund_prospect_share
	subtract = un_dues_rate
	multiply = un_ai_points_per_gdp_share
	max = un_ai_money_cap
}

# How much of the treasury's case against paying dues still stands once the
# Fund is counted: 1 with no grant, 0 when the grant we would keep by paying
# covers the dues. A member in default whose grant exceeds its dues makes money
# by paying, so its debt is no reason to withhold. Scope: country
un_ai_dues_net_cost_factor = {
	value = 1
	if = {
		limit = { un_dues_weekly_value > 0 }
		subtract = {
			value = un_dev_fund_prospect_value
			divide = un_dues_weekly_value
		}
	}
	min = 0
	max = 1
}

# A treasury in trouble, as a penalty: default, then debt against the ceiling.
# Shared by the dues, Development Programs and peacekeeping scores; the dues
# score scales it by un_ai_dues_net_cost_factor.
un_ai_treasury_penalty = {
	value = 0
	if = {
		limit = { in_default = yes }
		add = -60
	}
	else_if = {
		limit = { scaled_debt >= 0.75 }
		add = -25
	}
	else_if = {
		limit = { scaled_debt >= 0.5 }
		add = -10
	}
}

# ---- Membership ---------------------------------------------------------------------

un_ai_membership_join_line = { value = 0 }
un_ai_membership_band = { value = 25 }

# The crisis term: where it starts, and how much each point of authority
# below it costs. A typical small member (a score of about 40) leaves near
# authority 8, one with weak reasons (about 25) near 13, a great power or a
# humanitarian member (60 and more) only as the UN dies.
un_ai_membership_crisis_start = { value = 30 }
un_ai_membership_crisis_per_point = { value = 3 }

# What a seat is worth to us, member or not: the same reasons on both sides of
# the door, so joining and leaving read one figure.
un_ai_membership_will = {
	# The habit of membership: a seat, its privileges, a voice in the Assembly.
	value = 30
	# A great power stands for a permanent seat; a major power is heard. The
	# old Join weight gave a great power +30 here; rank can slip and return on
	# its own, so the step down from great power stays inside the band.
	if = {
		limit = { country_rank >= rank_value:great_power }
		add = 20
	}
	if = {
		limit = { country_rank >= rank_value:major_power }
		add = 15
	}
	if = {
		limit = { has_law_or_variant = law_type:law_isolationism }
		add = -80
	}
	if = {
		limit = { has_law = law_type:law_humanitarian_regulations }
		add = 20
	}
	if = {
		limit = { has_law = law_type:law_limited_war }
		add = 20
	}
	if = {
		limit = { has_law = law_type:law_total_war }
		add = -40
	}
	# The levy, and the Fund's grant against it.
	add = un_ai_money_points
	# What staying out costs: the pariah modifiers from Established and from
	# Strong, and at Supranational a standing sanctions case against a
	# non-member of major-power rank or with a nuclear programme or arsenal
	# (un_case_standing_applies, read here as "would apply if we were out").
	if = {
		limit = { un_tier_at_least_established = yes }
		add = 5
	}
	if = {
		limit = { un_tier_at_least_strong = yes }
		add = 5
	}
	if = {
		limit = {
			un_tier_is_supranational = yes
			OR = {
				country_rank >= rank_value:major_power
				has_modifier = nuclear_power
				nuclear_program_has_programme = yes
			}
		}
		add = 30
	}
	# A failing UN: from authority un_ai_membership_crisis_start down,
	# un_ai_membership_crisis_per_point off for every point it falls (-30 at
	# 20, -60 at 10, -90 at 0). Members leave as the crisis deepens, those
	# with the weakest reasons first, rather than all at once at a cliff
	# (owner, 2026-10-03).
	if = {
		limit = { un_ai_authority < un_ai_membership_crisis_start }
		subtract = {
			value = un_ai_membership_crisis_start
			subtract = un_ai_authority
			multiply = un_ai_membership_crisis_per_point
		}
	}
	# Help we could ask for: only a member requests aid or peacekeepers
	# (un_propose_aid_available, un_propose_peacekeepers_available), and the
	# World Food Reserve relieves members' famines.
	if = {
		limit = {
			any_scope_state = {
				OR = {
					average_sol < 8
					devastation > 0.05
				}
			}
		}
		add = 10
	}
	if = {
		limit = {
			has_global_variable = un_inst_food_reserve
			has_famine = yes
		}
		add = 10
	}
	# The conventions in force bind every member, a joiner at once
	# (un_apply_ratified_conventions).
	add = un_ai_convention_terms
	# A member's stance on the order, and an outsider's rejection of it.
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
		add = 20
	}
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_undermine_order_cost } }
		add = -30
	}
	if = {
		limit = { has_modifier = un_rejectionist_modifier }
		add = -20
	}
}

# ---- Development Programs (the Fund's voluntary part) ------------------------------------
# A contribution pays the Fund's grants (un_dev_fund_contributor). It opens
# only while the Fund stands, to any member the Fund does not pay that pays its
# dues (un_fund_development_button), and starts at Token, 0.0625% of GDP a
# year.
# The owner (2026-10-03): most of the time the AI should not give money
# without hope of larger gains, with humanitarian leaders the exception. And
# (2026-10-05): a contribution now buys infamy decay, prestige, standing and
# goodwill with the recipients, so rich members are much more likely to give,
# but typically only a token. So two scores:
#   - WHETHER to give (un_ai_development_will, the usual pair): a humanitarian
#     law, championing the order, recipients among our subjects, bloc
#     partners and allies, wealth (GDP per head at or above the members'
#     average, more at twice it) and a record to launder (infamy) start a
#     Token contribution; rank, a state-led economy and a working UN only add
#     to one of those. Laissez-faire weighs less than it did (-10, was -25):
#     a token now pays its way.
#   - HOW MUCH (un_ai_development_generosity): only the reasons to give beyond
#     a token, the humanitarian law and championing (a size each) and
#     recipients among our own (half a size), less what we cannot afford
#     (owner, 2026-10-06: being poor should hold a contribution back): GDP per
#     head below the great powers' average, or half of it, and debt
#     (un_ai_development_poverty). A contributor with none of them moves up to
#     Small (un_ai_development_base_size); above that it raises a size at
#     un_ai_development_size_step per size it already gives (20 for Modest,
#     40 for Substantial, 60 for Generous, which these reasons, 50 together,
#     do not reach) and cuts back a band below the line of the size it holds,
#     so a rich power with none of them stays at Small, and a poor or
#     borrowing one at Token.
#
#   size pair (un_raise / un_reduce_development_button), by the size we hold:
#   size held     raise at   reduce below
#   Token (1)     0          -
#   Small (2)     20         -25
#   Modest (3)    40         -5
#   Substantial   60         15
#   Generous (5)  -          35

un_ai_development_line = { value = 20 }
un_ai_development_band = { value = 25 }

un_ai_development_will = {
	value = 0
	if = {
		limit = { has_law = law_type:law_humanitarian_regulations }
		add = 20
	}
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
		add = 20
	}
	# un_economy_triggers.txt
	if = {
		limit = { un_ai_development_recipients_among_ours = yes }
		add = 15
	}
	# Wealth: each step flips on its own as GDP per head moves around the
	# average, by 15 and then 10, inside the band.
	if = {
		limit = { un_dev_fund_rich = yes }
		add = 25
	}
	else_if = {
		limit = { un_dev_fund_above_average = yes }
		add = 15
	}
	# A record to launder: a contribution speeds infamy's decay.
	if = {
		limit = { infamy >= infamy_threshold:notorious }
		add = 10
	}
	if = {
		limit = { country_rank >= rank_value:great_power }
		add = 5
	}
	if = {
		limit = {
			OR = {
				has_law = law_type:law_interventionism
				has_law = law_type:law_command_economy
			}
		}
		add = 5
	}
	if = {
		limit = { un_tier_at_least_established = yes }
		add = 5
	}
	if = {
		limit = { has_law_or_variant = law_type:law_laissez_faire }
		add = -10
	}
	if = {
		limit = { has_law_or_variant = law_type:law_isolationism }
		add = -30
	}
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_undermine_order_cost } }
		add = -40
	}
	# Paying into a fund that pays us is half a circle.
	if = {
		limit = { un_dev_fund_receiving = yes }
		add = -15
	}
	add = un_ai_treasury_penalty
}

# How much beyond a token: only the reasons to give generously.
un_ai_development_size_step = { value = 20 }

un_ai_development_generosity = {
	value = 0
	if = {
		limit = { has_law = law_type:law_humanitarian_regulations }
		add = 20
	}
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
		add = 20
	}
	# Half a size, since a great power almost always has a recipient among its

# What we cannot afford to give, as a penalty on the generosity score (owner,
# 2026-10-06: "being poor, in treasury terms or GDP per head relative to other
# great powers, should count as a pretty strong reason to downgrade or not
# upgrade"):
#   - GDP per head below our peers' (un_ai_development_peer_ratio): -15, and
#     below half of it -30. Each step flips on its own by 15, inside the band.
#   - Debt: the shared treasury penalty (-10 at half the ceiling, -25 at three
#     quarters, -60 in default) and -15 more from a quarter of the ceiling, so
#     borrowing at all holds a contribution back.
# Scope: country
un_ai_development_poverty = {
	value = 0
	if = {
		limit = { un_ai_development_peer_ratio < 0.5 }
		add = -30
	}
	else_if = {
		limit = { un_ai_development_peer_ratio < 1 }
		add = -15
	}
	if = {
		limit = { scaled_debt >= 0.25 }
		add = -15
	}
	add = un_ai_treasury_penalty
}

# Our GDP per head as a share of our peers': the represented great powers'
# average (global_var:un_dev_fund_gp_avg), whatever our own rank, or the
# members' average while no great power sits in the Assembly. 1 with no figure
# to compare, so no figure is no penalty. A single bar for every rank keeps a
# rank change from moving it. Scope: country
un_ai_development_peer_gdp_ph = {
	value = 0
	if = {
		limit = {
			has_global_variable = un_dev_fund_gp_avg
			global_var:un_dev_fund_gp_avg > 0
		}
		add = global_var:un_dev_fund_gp_avg
	}
	else_if = {
		limit = { has_global_variable = un_dev_fund_avg }
		add = global_var:un_dev_fund_avg
	}
}

un_ai_development_peer_ratio = {
	value = 0
	if = {
		limit = {
			has_variable = un_dev_fund_gdp_ph
			un_ai_development_peer_gdp_ph > 0
		}
		add = var:un_dev_fund_gdp_ph
		divide = un_ai_development_peer_gdp_ph
	}
	else = {
		add = 1
	}
}

# The size a contributor settles at with no reason to give more and nothing
# holding it back (generosity 0): Small, the money Token gave before the five
# sizes (2026-10-06). A member poorer than its peers or in debt (a negative
# generosity) stays at Token.
un_ai_development_base_size = { value = 2 }

# The generosity a contributor needs to raise from the size it holds: the
# step per size held above the base, 0 to reach it.
un_ai_development_raise_line = {
	value = un_dev_fund_size_value
	subtract = un_ai_development_base_size
	add = 1
	multiply = un_ai_development_size_step
}

# Below this it cuts back: the band under the line it raised to the size it
# holds at.
un_ai_development_reduce_line = {
	value = un_dev_fund_size_value
	subtract = un_ai_development_base_size
	multiply = un_ai_development_size_step
	subtract = un_ai_development_band
}

un_raise_development_ai_chance = {
	value = 0
	if = {
		limit = {
			un_dev_fund_size_value >= 1
			un_dev_fund_size_value < un_dev_fund_size_max
			un_ai_development_generosity >= un_ai_development_raise_line
		}
		add = un_ai_development_generosity
		subtract = un_ai_development_raise_line
		add = 10
		max = 50
	}
}

un_reduce_development_ai_chance = {
	value = 0
	if = {
		limit = {
			un_dev_fund_size_value > 1
			un_ai_development_generosity < un_ai_development_reduce_line
		}
		add = un_ai_development_reduce_line
		subtract = un_ai_development_generosity
		max = 50
	}
}

# ---- Peacekeeping programme ---------------------------------------------------------------
# 0.5% of GDP a year, bureaucracy and dearer military goods, for leverage,
# prestige and the missions it staffs (un_peacekeeping_mission_button; a
# major power). Ending it waits out the ten-year cooldown its start set.

un_ai_peacekeeping_line = { value = 20 }
un_ai_peacekeeping_band = { value = 25 }

un_ai_peacekeeping_will = {
	value = 0
	if = {
		limit = { has_law = law_type:law_humanitarian_regulations }
		add = 20
	}
	if = {
		limit = { has_law = law_type:law_limited_war }
		add = 15
	}
	if = {
		limit = { country_rank >= rank_value:great_power }
		add = 15
	}
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_champion_order_cost } }
		add = 15
	}
	if = {
		limit = { un_tier_at_least_established = yes }
		add = 5
	}
	# Blue helmets in the field to staff.
	if = {
		limit = {
			has_global_variable_list = un_mission_registry
			any_in_global_list = {
				variable = un_mission_registry
				has_tag = un_msn_active
				OR = {
					has_tag = un_msn_peacekeeping
					has_tag = un_msn_stabilisation
				}
			}
		}
		add = 10
	}
	if = {
		limit = { is_at_war = yes }
		add = -30
	}
	if = {
		limit = { has_law = law_type:law_total_war }
		add = -20
	}
	if = {
		limit = { has_law_or_variant = law_type:law_isolationism }
		add = -30
	}
	if = {
		limit = { je:je_united_nations ?= { has_modifier = un_undermine_order_cost } }
		add = -40
	}
	add = un_ai_treasury_penalty
}

un_ai_peacekeeping_stop_line = {
	value = un_ai_peacekeeping_line
	subtract = un_ai_peacekeeping_band
}

un_peacekeeping_ai_chance = {
	value = 0
	if = {
		limit = { un_ai_peacekeeping_will >= un_ai_peacekeeping_line }
		add = un_ai_peacekeeping_will
		subtract = un_ai_peacekeeping_line
		add = 10
		max = 50
	}
}

un_end_peacekeeping_ai_chance = {
	value = 0
	if = {
		limit = { un_ai_peacekeeping_will < un_ai_peacekeeping_stop_line }
		add = un_ai_peacekeeping_stop_line
		subtract = un_ai_peacekeeping_will
		max = 50
	}
}

# ---- Champion / undermine the order ------------------------------------------------------
# A great power's stance on the order (un_champion_order_button and the
# undermine button: the commitment pillar's +1 / -1, at a cost in Authority and
# influence). Its reasons are its laws, its seat, its bloc's commitment to the
# institution, what the conventions ask of it and whether the Assembly has
# turned on it. Authority adds a mild pull, at most 7.5 either way: a weak UN
# draws a would-be champion in, a strong one makes a sovereign power wary. It is
# smaller than the band, so authority alone never moves a power across
# (owner, 2026-10-03).

un_ai_order_line = { value = 20 }
un_ai_order_band = { value = 25 }

# The mild pull: 0.15 a point from 50, so +7.5 at 0 and -7.5 at 100.
un_ai_order_authority_weight = { value = 0.15 }

un_ai_order_will = {
	value = 0
	if = {
		limit = { has_law = law_type:law_humanitarian_regulations }
		add = 25
	}
	if = {
		limit = { has_law = law_type:law_limited_war }
		add = 15
	}
	if = {
		limit = { has_law_or_variant = law_type:law_isolationism }
		add = -30
	}
	if = {
		limit = { has_law = law_type:law_total_war }
		add = -25
	}
	# The order's architecture seats us at its top table.
	if = {
		limit = { is_un_permanent_member = yes }
		add = 10
	}
	# The Multilateral Institutions principle (0.25 to 0.5): +10 to +20.
	add = {
		value = modifier:country_un_institutional_alignment
		multiply = 40
	}
	# Half of what the conventions in force ask of us (-20 to +7.5).
	add = {
		value = un_ai_convention_terms
		multiply = 0.5
	}
	# The Assembly against us: its sanctions, or a record it could act on.
	if = {
		limit = { has_modifier = un_sanctions_target_modifier }
		add = -20
	}
	if = {
		limit = { un_case_strength >= un_case_threshold_condemn }
		add = -10
	}
	add = {
		value = 50
		subtract = un_ai_authority
		multiply = un_ai_order_authority_weight
	}
}

# The same stance, read from the other side: undermining scores where
# championing would score below zero.
un_ai_dissent_will = {
	value = 0
	subtract = un_ai_order_will
}

un_ai_order_stop_line = {
	value = un_ai_order_line
	subtract = un_ai_order_band
}

un_champion_order_ai_chance = {
	value = 0
	if = {
		limit = { un_ai_order_will >= un_ai_order_line }
		add = un_ai_order_will
		subtract = un_ai_order_line
		add = 10
		max = 50
	}
}

un_stop_championing_ai_chance = {

		limit = { un_ai_order_will < un_ai_order_stop_line }
		add = un_ai_order_stop_line
		subtract = un_ai_order_will
		max = 50
	}
}

un_undermine_order_ai_chance = {
	value = 0
	if = {
		limit = { un_ai_dissent_will >= un_ai_order_line }
		add = un_ai_dissent_will
		subtract = un_ai_order_line
		add = 10
		max = 50
	}
}

un_stop_undermining_ai_chance = {
	value = 0
	if = {
		limit = { un_ai_dissent_will < un_ai_order_stop_line }
		add = un_ai_order_stop_line
		subtract = un_ai_dissent_will
		max = 50
	}
}
