# ---------------------------------------------------------------------------
# The cartel (§§ 6.1, 6.2)
# ---------------------------------------------------------------------------

# A firm whose goal is Dominance or A Protected Market forms one at this share of its sector...
te_pa_cartel_share = 0.5
# ...or at this one under Strict IP Protection with a laboratory at te_pa_gen_patent_pool_lab (§ 10.3's patent pool;
# the laboratory is Task 7's, so the branch waits for it).
te_pa_cartel_share_ip = 0.4
# It ends when the share falls this far below the share it forms at, so a firm on the line doesn't form and end a
# cartel at every turn (the share is read at each turn).
te_pa_cartel_end_margin = 0.05
# Strength at formation, and its growth a year: 15 under Freedom of Contract, 10 otherwise, up to the share x 100.
te_pa_cartel_start = 5
te_pa_cartel_growth_year = 10
te_pa_cartel_growth_free_year = 15
# Under Regulated Utilities, outside infrastructure and utilities, it is capped here; a captured regulator raises the
# cap toward te_pa_cartel_ru_cap_max with its strength.
te_pa_cartel_ru_cap = 40
te_pa_cartel_ru_cap_max = 80
# Rents: at strength 100 the firm's funds accrue this much faster (§ 6.1 stage 4: up to double).
te_pa_cartel_funds_extra = 1
# The strengths the ledger marks (§ 6.5's charter needs 80).
te_pa_cartel_mark_1 = 50
te_pa_cartel_mark_2 = 80
te_pa_cartel_mark_3 = 100

# Company scope. The share it forms a cartel at.
te_pa_cartel_threshold_value = {
	value = te_pa_cartel_share
	if = {
		limit = {
			owner = { has_law = law_type:law_strict_ip_protection }
			te_pa_v_lab >= te_pa_gen_patent_pool_lab
		}
		value = te_pa_cartel_share_ip
	}
}

# Company scope. The share below which its cartel ends.
te_pa_cartel_end_value = {
	value = te_pa_cartel_threshold_value
	subtract = te_pa_cartel_end_margin
}

# Company scope. A quarter's growth.
te_pa_cartel_growth_quarter = {
	value = te_pa_cartel_growth_year
	if = { limit = { owner = { has_law = law_type:law_freedom_of_contract } } value = te_pa_cartel_growth_free_year }
	divide = te_pa_quarters_a_year
}

# Company scope. Regulated Utilities' cap, raised by the captured regulator: 40 at none, 80 at strength 100.
te_pa_cartel_ru_cap_now = {
	value = te_pa_cartel_ru_cap_max
	subtract = te_pa_cartel_ru_cap
	multiply = te_pa_v_reg
	divide = 100
	add = te_pa_cartel_ru_cap
}

# Company scope. How strong its cartel can be: its share x 100, under Regulated Utilities outside infrastructure and
# utilities no more than the cap, never past 100.
te_pa_cartel_cap_value = {
	value = te_pa_v_share
	multiply = 100
	if = { limit = { te_pa_cartel_ru_capped = yes } max = te_pa_cartel_ru_cap_now }
	max = 100
	min = 0
}

# Company scope. The strength after a quarter's growth, held to the cap (a share that falls takes it down with it).
te_pa_cartel_after_quarter = {
	value = te_pa_v_cartel
	add = te_pa_cartel_growth_quarter
	max = te_pa_cartel_cap_value
	min = 0
}

# Company scope. The cartel modifier's multiplier: its strength over 100 (the generated modifiers hold strength 100).
te_pa_cartel_mult_value = {
	value = te_pa_v_cartel
	divide = 100
	min = 0
	max = 1
}

# Company scope. Rents: the funds' accrual is this many times its usual rate, 1 with no cartel, 2 at strength 100
# (te_pa_funds_year, te_pa_core_values.txt).
te_pa_cartel_funds_factor = {
	value = te_pa_v_cartel
	divide = 100
	multiply = te_pa_cartel_funds_extra
	add = 1
}

# Company scope. Its cartel's strength as the state knows it, for the tab (te_pa_row_sync): 0 while it is secret
# and unexposed, so its row shows no chip and its detail no line (§ 4.6: secrecy is against the state).
te_pa_cartel_shown_value = {
	value = 0
	if = { limit = { var:te_pa_cartel_known ?= 1 } value = te_pa_v_cartel }
}

# Company scope. 1 while it holds a cartel the antitrust law in force allows (the chip's hover: open, or known and
# illegal).
te_pa_cartel_open_value = {
	value = 0
	if = {
		limit = {
			te_pa_v_cartel > 0
			te_pa_cartel_lawful = yes
		}
		value = 1
	}
}

te_pa_expose_base = 10
# The speech laws: the freer the press, the likelier.
te_pa_expose_outlawed_dissent = -6
te_pa_expose_censorship = -3
te_pa_expose_protected_speech = 5
# The policing and Home Affairs institutions, a level of investment each.
te_pa_expose_police_per_level = 1
te_pa_expose_home_affairs_per_level = 2
# The courts: Judicial Independence (docs/systems/mod_systems/judiciary.md, 0-100) above or below the middle, a
# tenth of a point each; the middle while the country has none.
te_pa_expose_courts_neutral = 50
te_pa_expose_courts_per_point = 0.1
te_pa_expose_max = 50
# Exposed: the firm's Standing falls this much (and its foundation is tainted, te_pa_found_taint).
te_pa_expose_stand_loss = 15

# Country scope. Judicial Independence, the middle while unset.
te_pa_v_judicial_independence = {
	value = te_pa_expose_courts_neutral
	if = { limit = { has_variable = te_judicial_independence } value = var:te_judicial_independence }
}

# Country scope. What the state brings to a roll: its speech law, its policing and Home Affairs, its courts.
te_pa_expose_state_value = {
	value = 0
	if = { limit = { has_law = law_type:law_outlawed_dissent } add = te_pa_expose_outlawed_dissent }
	else_if = { limit = { has_law = law_type:law_censorship } add = te_pa_expose_censorship }
	else_if = { limit = { has_law = law_type:law_protected_speech } add = te_pa_expose_protected_speech }
	add = { value = "institution_investment_level(institution_police)" multiply = te_pa_expose_police_per_level }
	add = { value = "institution_investment_level(institution_home_affairs)" multiply = te_pa_expose_home_affairs_per_level }
	add = {
		value = te_pa_v_judicial_independence
		subtract = te_pa_expose_courts_neutral
		multiply = te_pa_expose_courts_per_point
	}
}

# Company scope. The hook later tasks add their terms to (0 now): Task 4b's Inquiry (§ 6.6, +30% for a year) and
# Task 5's press (§ 8.5: the firm's own press lowers its chance, a rival's raises it). Each is a term in this one
# value, so te_pa_expose_chance_value reads them all.
te_pa_expose_hook_value = {
	value = 0
}

# Company scope. A secret holding's yearly chance of exposure, before a captured regulator's cut: the base, the state,
# what buyers have paid for (te_pa_expose_push, § 6.3) and the hook, 0 to te_pa_expose_max.
te_pa_expose_chance_value = {
	value = te_pa_expose_base
	add = { value = 0 owner = { add = te_pa_expose_state_value } }
	add = te_pa_v_expose_push
	add = te_pa_expose_hook_value
	min = 0
	max = te_pa_expose_max
}

# Company scope. The cartel's roll: cut by the captured regulator's strength (§ 6.4: at 100, never exposed), rounded,
# as random's chance takes a whole percentage (scripting_best_practices.md, "A Per-Mille Chance").
te_pa_expose_cartel_value = {
	value = 100
	subtract = te_pa_v_reg

	divide = 100
	round = yes
	min = 0
}

# Company scope. The captured regulator's own roll.
te_pa_expose_reg_value = {
	value = te_pa_expose_chance_value
	round = yes
	min = 0
}

te_pa_stand_after_exposure = {
	value = te_pa_v_stand
	subtract = te_pa_expose_stand_loss
	min = 0
}
