# The stockpile, 0 if never created.
nd_stockpile = {
	value = 0
	if = {
		limit = { has_variable = nuclear_weapon_stockpile }
		add = var:nuclear_weapon_stockpile
	}
	min = 0
}

# Arsenal size as it bears on costs: warheads beyond fifty add nothing (§4 —
# diminishing returns, real upkeep).
nd_stockpile_capped = {
	value = nd_stockpile
	max = 50
}

# 0.4 for a token arsenal up to 1.0 for fifty warheads: how much of the
# readiness bill an arsenal of this size runs up.
nd_upkeep_size_factor = {
	value = nd_stockpile_capped
	divide = 50
	multiply = 0.6
	add = 0.4
}

# ----------------------------------------------------------------------------
# UPKEEP — weekly, GDP-scaled (§3.2)
# ----------------------------------------------------------------------------
# Worked in units of 1/100,000 of GDP per week, so no constant is small enough
# to fall below fixed-point precision; one unit is ≈ 0.052 % of GDP a year.
#   custody        0.2 × warheads (capped at 50)          ≤ 10 units
#                  (0.1 at Recessed: warheads in storage)
#   heightened     4 × size factor                         ≤  4 units
#   high alert     12 × size factor                        ≤ 12 units
#   safeguards     2 per level (0–3)                       ≤  6 units
#   hardening      3 per level (0–3)                       ≤  9 units
#   automatic retaliation (authority 4)                      3 units
#   cap            40 units ≈ 2 % of GDP a year
nd_upkeep_units_custody = {
	value = nd_stockpile_capped
	multiply = 0.2
}

# Custody at Recessed: warheads in storage cost half to keep.
nd_upkeep_units_custody_recessed = {
	value = nd_upkeep_units_custody
	multiply = 0.5
}

# What does not depend on readiness: safeguards, hardening, and an Automatic
# Retaliation system (3 units, as one hardening level).
nd_upkeep_units_fixed = {
	value = 0
	if = {
		limit = { nd_authority_automatic = yes }
		add = 3
	}
	if = {
		limit = { has_variable = nd_safeguards }
		add = {
			value = var:nd_safeguards
			multiply = 2
		}
	}
	if = {
		limit = { has_variable = nd_hardening }
		add = {
			value = var:nd_hardening
			multiply = 3
		}
	}
}

# Custody at the readiness in force, plus the fixed items. The per-level
# previews below build from the same parts, so a button's cost is exact.
nd_upkeep_units_base = {
	value = nd_upkeep_units_fixed
	if = {
		limit = { nd_readiness_recessed = yes }
		add = nd_upkeep_units_custody_recessed
	}
	else = {
		add = nd_upkeep_units_custody
	}
}

nd_upkeep_units_readiness_2 = {
	value = nd_upkeep_size_factor
	multiply = 4
}

nd_upkeep_units_readiness_3 = {
	value = nd_upkeep_size_factor
	multiply = 12
}

nd_upkeep_units = {
	value = nd_upkeep_units_base
	if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 2
		}
		add = nd_upkeep_units_readiness_2
	}
	else_if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 3
		}
		add = nd_upkeep_units_readiness_3
	}
	# Clamp the rate before it is scaled by GDP: the order is load-bearing
	# (scripting_best_practices.md § Script-Value Keys Apply in Written Order).
	min = 0
	max = 40
}

# The weekly expense, in money. Cached into nd_upkeep_weekly_cached by
# nd_apply_posture_modifiers; the JE-scoped modifier reads the cache, never
# this, because a JE-scope re-evaluation cannot reach `gdp`.
nd_upkeep_weekly = {
	value = gdp
	divide = 100000
	multiply = nd_upkeep_units
	min = 0
}

# The highest survivability investment can reach with this technology.
nd_survivability_cap = {
	value = 25                                  # dispersed bombers
	if = {
		limit = { has_technology_researched = radar }
		add = 15                                # warning time to get bombers up
	}
	if = {
		limit = { has_technology_researched = ICBMs }
		add = 20                                # hardened silos
	}
	if = {
		limit = { has_technology_researched = advanced_submarine_technology }
		add = 25                                # missile submarines at sea
	}
	if = {
		limit = { has_technology_researched = missile_defense_systems }
		add = 10
	}
	# The last stretch toward full security has to come from a fast launch
	# authority (nd_launch_credit), not from technology (rework spec §3.1).
	max = 80
}

# Without investment, survivability decays toward this floor.
nd_survivability_floor = {
	value = nd_survivability_cap
	multiply = 0.3
}

# This month's survivability step: toward the cap at 2 % of the gap per
# hardening level, or 1 % of the excess back toward the floor without it.
nd_survivability_step = {
	value = 0
	if = {
		limit = {
			has_variable = nd_survivability
			has_variable = nd_hardening
			var:nd_hardening >= 1
		}
		# No floor: above the ceiling (a save from before it fell from 95 to
		# 80, rework spec §3.1) the step is negative and settles to it.
		add = nd_survivability_cap
		subtract = var:nd_survivability
		multiply = {
			value = var:nd_hardening
			multiply = 0.02
		}
	}
	else_if = {
		limit = {
			has_variable = nd_survivability
			var:nd_survivability > nd_survivability_floor
		}
		add = var:nd_survivability
		subtract = nd_survivability_floor
		multiply = -0.01
	}
}

# This country's crisis danger band, 0–3, whichever side it is on.
nd_own_danger_band = {
	value = 0
	if = {
		limit = {
			nd_is_crisis_owner = yes
			has_variable = nd_crisis_danger
		}
		add = var:nd_crisis_danger
	}
	else_if = {
		limit = {
			nd_is_crisis_target = yes
			exists = var:nd_crisis_opponent
		}
		var:nd_crisis_opponent = {
			if = {
				limit = { has_variable = nd_crisis_danger }
				add = var:nd_crisis_danger
			}
		}
	}
	divide = 25
	floor = yes
	max = 3
}

# Meaningful incident exposure, per mille PER WEEK. Routine mishaps use their
# own monthly roll and never dilute this pool. Peacetime bases below are monthly
# equivalents divided by 52/12; the crisis surcharge is already weekly.
# Benchmark: two high-alert powers, danger >=75, strain 0, reliability 60,
# Existential doctrine: ~28% chance of an incident across two weekly pulses each
# (calibrated 2026-10-28 to the rework spec's §2.3 launch bands; ~50% before).
#
# Two parts, one roll. The base part draws from every family this country is
# eligible for; the crisis part draws only from the misreadings a crisis
# drives (a warning, an exercise, a commander cut off), never a weapons
# accident in a depot, and is zero when none of them is eligible
# (nd_fire_incident).
nd_incident_base_monthly_permille = {
	value = 1
	if = { limit = { nd_readiness_recessed = yes } value = 0.5 }
	else_if = { limit = { nd_readiness_high_alert = yes } value = 10 }
	else_if = { limit = { nd_readiness_heightened_or_higher = yes } value = 4 }
}

# A war with an enemy believed armed (rework spec §5.1): band 2, or band 3 when
# the war threatens either side (goals on incorporated states, a side losing,
# a nuclear weapon landed). Alongside a crisis the higher band applies.
nd_war_danger_band = {
	value = 0
	if = {
		limit = { nd_war_band_3_against_any = yes }
		value = 3
	}
	else_if = {
		limit = {
			any_enemy_in_war = { nd_believed_armed = yes }
		}
		value = 2
	}
}

nd_incident_band = {
	value = nd_own_danger_band
	if = {
		limit = { nd_war_danger_band > nd_own_danger_band }
		value = nd_war_danger_band
	}
}

# Calibrated 2026-10-28 to the rework spec's §2.3 Acute-crisis bands and its
# §2.1 war-year rates (scripts/analysis/nuclear_incident_rates.py,
# test_nuclear_incident_risk.TestSpecBands): band 3 was 175, band 2 50.
nd_incident_crisis_weekly_permille = {
	value = 0
	if = { limit = { nd_incident_band >= 3 } value = 85 }
	else_if = { limit = { nd_incident_band >= 2 } value = 35 }
	else_if = { limit = { nd_incident_band >= 1 } value = 5 }
	# A force below operational alert is harder to misread.
	if = { limit = { nd_readiness_heightened_or_higher = no } multiply = 0.25 }
	else_if = { limit = { nd_readiness_high_alert = no } multiply = 0.6 }
	if = { limit = { nd_incident_misreading_eligible = no } value = 0 }
}

nd_incident_doctrine_multiplier = {
	value = 1
	if = { limit = { nd_doctrine_warfighting = yes } value = 1.6 }
	else_if = { limit = { nd_doctrine_compellence = yes } value = 1.35 }
	else_if = { limit = { nd_doctrine_flexible = yes } value = 1.15 }
}

# Strain, reliability and doctrine scale both parts alike.
nd_incident_rate_multiplier = {
	value = 1
	if = {
		limit = { has_variable = nd_strain }
		multiply = { value = var:nd_strain divide = 100 add = 1 }
	}
	if = {
		limit = { has_variable = nd_reliability }
		multiply = { value = 100 subtract = var:nd_reliability divide = 100 add = 0.5 }
	}
	multiply = nd_incident_doctrine_multiplier
}

# Chance (percent) that a launch sequence started under delegated authority or
# launch on warning is halted by someone in the chain — the Petrov or
# Arkhipov who refuses. There is no player veto on these branches (§8.1); this
# number is the whole of the protection, which is why the panel shows it.
nd_hold_chance = {
	value = 60
	if = {
		limit = { has_variable = nd_reliability }
		value = var:nd_reliability
	}
	# Half of it, on a floor of 53: someone in the chain doubts a warning
	# however poorly the force is run, and safeguards already raise
	# reliability. Calibrated 2026-10-28 so that Launch on Warning with no
	# safeguards is reckless (a quarter over an Acute crisis), not suicidal
	# (two thirds): rework spec §2.3.
	multiply = 0.5
	add = 53
	if = {
		limit = { has_variable = nd_safeguards }
		add = {
			value = var:nd_safeguards
			multiply = 5
		}
	}
	if = {
		limit = { has_variable = nd_punished_skeptic }
		subtract = 15
	}
	if = {
		limit = { nd_authority_launch_on_warning = yes }
		subtract = 10
	}
	# The moment the warning arrives in (the Petrov judgement): at peace with
	# no crisis, a real first strike makes no sense and a few missiles on a
	# screen look like a fault; in a Warning or a Confrontation it is
	# thinkable; in an Acute crisis or a war nothing argues against it.
	if = {
		limit = {
			is_at_war = no
			nd_in_crisis = no
		}
		add = 25
	}
	else_if = {
		limit = {
			is_at_war = no
			NOT = { nd_crisis_stage_at_least = { STAGE = 3 } }
		}
		add = 10
	}
	# The suspect's posture (rework spec §4.3): forces in storage cannot be
	# launching, so a warning about them is easy to doubt; forces at High
	# Alert make it easy to believe. The caller saves the suspect as
	# scope:nd_hold_suspect; the panel's figure has none.
	if = {
		limit = { exists = scope:nd_hold_suspect }
		scope:nd_hold_suspect = {
			if = {
				limit = { nd_readiness_recessed = yes }
				add = 20
			}
			else_if = {
				limit = { nd_readiness_high_alert = yes }
				add = -5
			}
		}
	}
	# A hotline: the crisis's talks are open, so someone can be asked.
	if = {
		limit = { nd_crisis_talks_open_either = yes }
		add = 10
	}
	# Our own vulnerability: commanders who fear losing the force are quicker
	# to use it. The shape of the crisis danger's vulnerability part.
	subtract = nd_own_vulnerability_value
	min = 20
	# No investment takes the risk away (owner, 2026-10-28).
	max = nd_hold_ceiling
}

# The highest chance of a halt in this moment: 97 at peace with no crisis, 92
# in a Warning or a Confrontation, 85 in an Acute crisis or a war.
nd_hold_ceiling = {
	value = 85
	if = {
		limit = {
			is_at_war = no
			nd_in_crisis = no
		}
		value = 97
	}
	else_if = {
		limit = {
			is_at_war = no
			NOT = { nd_crisis_stage_at_least = { STAGE = 3 } }
		}
		value = 92
	}
}

# How exposed our forces are, as a pressure on the people holding them:
# (50 − effective survivability) ÷ 5, between 0 and 10. The chance of a halt
# subtracts it, and the crisis danger counts the more vulnerable side's
# (nd_cd_vulnerability_value).
nd_own_vulnerability_value = {
	value = 50
	subtract = nd_effective_survivability
	divide = 5
	min = 0
	max = 10
}

# The survivability a crisis counts: the stored figure scaled by readiness.
# Bombers on strip alert, submarines surged to sea and launchers dispersed
# ride out a first strike that would catch them on the ground or in port, and
# warheads in storage are the easiest target of all. It sets how many warheads
# are exposed (nd_exposed_warheads), so survivors and every reader of them;
# below 50 it also raises the crisis danger and lowers the chance of a halt
# (nd_own_vulnerability_value). The AI's choice of launch authority reads
# survivors at Heightened with no credit instead (nd_ai_review_posture):
# standing down must not push it towards Launch on Warning.
nd_effective_survivability = {
	value = nd_survivability_or_zero
	if = {
		limit = { nd_readiness_recessed = yes }
		multiply = nd_survivability_factor_recessed
	}
	else_if = {
		limit = { nd_readiness_heightened_or_higher = no }
		multiply = nd_survivability_factor_routine
	}
	else_if = {
		limit = { nd_readiness_high_alert = no }
		multiply = nd_survivability_factor_heightened
	}
}

# SECOND-STRIKE SURVIVORS (rework spec §3.2) — how many of our warheads would
# be left after the other side spent its whole arsenal on them. Our own

# authority or standing down does not feed back into its own choice); and at
# the technology ceiling, Heightened, no credit ("hardening won't close it").
# ----------------------------------------------------------------------------

# What is at stake (rework spec §8.2): the demands of the side the target
# would concede to (nd_stakes_demander: the issuer, or for a guarantee dispute
# the country it covers) on the target. Annexation or reunification −30 and
# nothing else counted; a subjugation type or regime change −20; −10 for each
# type that takes a state or a subject from us; −4 for each lesser type.
# At most −40. No trigger tells a primary demand from another, so every goal
# counts (owner call 10). Target scope; scope:nd_target and
# scope:nd_stakes_demander saved.
nd_yp_stakes_value = {
	value = 0
	if = {
		limit = { exists = scope:nd_stakes_demander }
		if = {
			limit = {
				scope:nd_stakes_demander = {
					OR = {
						nd_side_holds_goal_against = { TYPE = annex_country TARGET = scope:nd_target }
						nd_side_holds_goal_against = { TYPE = te_reunify_country TARGET = scope:nd_target }
						nd_side_holds_goal_against = { TYPE = unification TARGET = scope:nd_target }
						nd_side_holds_goal_against = { TYPE = restore_union TARGET = scope:nd_target }
						nd_side_holds_goal_against = { TYPE = te_colonial_reconquest TARGET = scope:nd_target }
					}
				}
			}
			add = -30
		}
		else = {
			if = {
				limit = {
					scope:nd_stakes_demander = {
						OR = {
							nd_side_holds_goal_against = { TYPE = make_protectorate TARGET = scope:nd_target }
							nd_side_holds_goal_against = { TYPE = make_dominion TARGET = scope:nd_target }
							nd_side_holds_goal_against = { TYPE = make_tributary TARGET = scope:nd_target }
							nd_side_holds_goal_against = { TYPE = make_personal_union TARGET = scope:nd_target }
							nd_side_holds_goal_against = { TYPE = make_crown_land TARGET = scope:nd_target }
							nd_side_holds_goal_against = { TYPE = make_chartered_company TARGET = scope:nd_target }
							nd_side_holds_goal_against = { TYPE = regime_change TARGET = scope:nd_target }
						}
					}
				}
				add = -20
			}
			add = nd_yp_stakes_territory_value
			add = nd_yp_stakes_minor_value
		}
	}
	# In a war with the issuer the existence threat already weighs on us
	# through nd_yp_war_value (−15); count only the rest here, so an
	# annexation still totals −30.
	if = {
		limit = {
			nd_yp_stakes_value_before_war_overlap < 0
			nd_enemy_threatens_existence = { ENEMY = scope:nd_issuer }
		}
		add = 15
	}
	min = -40
	round = yes
}

					nd_side_holds_goal_against = { TYPE = make_dominion TARGET = scope:nd_target }
					nd_side_holds_goal_against = { TYPE = make_tributary TARGET = scope:nd_target }
					nd_side_holds_goal_against = { TYPE = make_personal_union TARGET = scope:nd_target }

# The AI's chance (percent) of claiming a deliberate first strike was

}

# ---- Trades (rework spec §8.6, phase 4c) --------------------------------------
# Parties saved (scope:nd_issuer, scope:nd_target), in a crisis over a dispute
# that trades. What the target would give up less what the issuer would, on
# §8.2's scale (ruling R6): a basing or guarantee treaty 10, like a state; a
# missile-defense state 4, a lesser goal; a claimed incorporated state 10; a
# caught covert operation 4; an active battlestation site 10. Within disputes
# 8–10 only the counts matter. Each half counts what the dispute's trigger and
# concession read (nd_dispute_*_by, nd_crisis_concede_*). The AI gives no more
# than it gets: a target accepts at 0 or less, an issuer at 0 or more.
nd_trade_balance_value = {
	value = 0
	if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 7 } }
		scope:nd_target = {
			every_scope_treaty = {
				limit = {
					any_scope_article = {
						OR = {
							has_type = nuclear_basing
							has_type = nuclear_guarantee
						}
						source_country = scope:nd_target
						target_country = {
							NOT = { THIS = scope:nd_issuer }
							is_adjacent_to_country = scope:nd_issuer
						}
					}
				}
				add = 10
			}
			every_scope_state = {
				limit = {
					nd_state_has_missile_defense = yes
					any_neighbouring_state = { owner = scope:nd_issuer }
				}
				add = 4
			}
		}
		scope:nd_issuer = {
			every_scope_treaty = {
				limit = {
					any_scope_article = {
						OR = {
							has_type = nuclear_basing
							has_type = nuclear_guarantee
						}
						source_country = scope:nd_issuer
						target_country = {
							NOT = { THIS = scope:nd_target }
							is_adjacent_to_country = scope:nd_target
						}
					}
				}
				subtract = 10
			}
			every_scope_state = {
				limit = {
					nd_state_has_missile_defense = yes
					any_neighbouring_state = { owner = scope:nd_target }
				}
				subtract = 4
			}
		}
	}
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 8 } }
		scope:nd_issuer = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
					state_region = { scope:nd_target = { has_claim = PREV } }
				}
				add = 10
			}
		}
		scope:nd_target = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
					state_region = { scope:nd_issuer = { has_claim = PREV } }
				}
				subtract = 10
			}
		}
	}
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 9 } }
		scope:nd_issuer = {
			if = {
				limit = { has_variable_list = iw_known_ops }
				every_in_list = {
					variable = iw_known_ops
					limit = {
						var:iw_attacker ?= scope:nd_target
						NOT = { has_tag = iw_known_counterintel_support }
					}
					add = 4
				}
			}
		}
		scope:nd_target = {
			if = {
				limit = { has_variable_list = iw_known_ops }
				every_in_list = {
					variable = iw_known_ops
					limit = {
						var:iw_attacker ?= scope:nd_issuer
						NOT = { has_tag = iw_known_counterintel_support }
					}
					subtract = 4
				}
			}
		}
	}
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 10 } }
		scope:nd_target = {
			every_scope_state = {
				limit = {
					any_scope_building = {
						is_building_type = building_orbital_battlestation_construction_site
						NOT = { has_active_production_method = pm_orbital_battlestation_construction_paused }
