# The stockpile, 0 if never created.
nd_stockpile = {
	value = 0
	if = {
		limit = { has_variable = nuclear_weapon_stockpile }
		add = var:nuclear_weapon_stockpile
	}
	min = 0
}

# Arsenal size as it bears on costs: warheads beyond fifty add nothing (§4 —
# diminishing returns, real upkeep).
nd_stockpile_capped = {
	value = nd_stockpile
	max = 50
}

# 0.4 for a token arsenal up to 1.0 for fifty warheads: how much of the
# readiness bill an arsenal of this size runs up.
nd_upkeep_size_factor = {
	value = nd_stockpile_capped
	divide = 50
	multiply = 0.6
	add = 0.4
}

# ----------------------------------------------------------------------------
# UPKEEP — weekly, GDP-scaled (§3.2)
# ----------------------------------------------------------------------------
# Worked in units of 1/100,000 of GDP per week, so no constant is small enough
# to fall below fixed-point precision; one unit is ≈ 0.052 % of GDP a year.
#   custody        0.2 × warheads (capped at 50)          ≤ 10 units
#   heightened     4 × size factor                         ≤  4 units
#   high alert     12 × size factor                        ≤ 12 units
#   safeguards     2 per level (0–3)                       ≤  6 units
#   hardening      3 per level (0–3)                       ≤  9 units
#   cap            40 units ≈ 2 % of GDP a year
nd_upkeep_units_base = {
	value = nd_stockpile_capped
	multiply = 0.2
	if = {
		limit = { has_variable = nd_safeguards }
		add = {
			value = var:nd_safeguards
			multiply = 2
		}
	}
	if = {
		limit = { has_variable = nd_hardening }
		add = {
			value = var:nd_hardening
			multiply = 3
		}
	}
}

nd_upkeep_units_readiness_2 = {
	value = nd_upkeep_size_factor
	multiply = 4
}

nd_upkeep_units_readiness_3 = {
	value = nd_upkeep_size_factor
	multiply = 12
}

nd_upkeep_units = {
	value = nd_upkeep_units_base
	if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 2
		}
		add = nd_upkeep_units_readiness_2
	}
	else_if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 3
		}
		add = nd_upkeep_units_readiness_3
	}
	# Clamp the rate before it is scaled by GDP: the order is load-bearing
	# (scripting_best_practices.md § Script-Value Keys Apply in Written Order).
	min = 0
	max = 40
}

# The weekly expense, in money. Cached into nd_upkeep_weekly_cached by
# nd_apply_posture_modifiers; the JE-scoped modifier reads the cache, never
# this, because a JE-scope re-evaluation cannot reach `gdp`.
nd_upkeep_weekly = {
	value = gdp
	divide = 100000
	multiply = nd_upkeep_units
	min = 0
}

# The highest survivability investment can reach with this technology.
nd_survivability_cap = {
	value = 25                                  # dispersed bombers
	if = {
		limit = { has_technology_researched = radar }
		add = 15                                # warning time to get bombers up
	}
	if = {
		limit = { has_technology_researched = ICBMs }
		add = 20                                # hardened silos
	}
	if = {
		limit = { has_technology_researched = advanced_submarine_technology }
		add = 25                                # missile submarines at sea
	}
	if = {
		limit = { has_technology_researched = missile_defense_systems }
		add = 10
	}
	max = 95
}

# Without investment, survivability decays toward this floor.
nd_survivability_floor = {
	value = nd_survivability_cap
	multiply = 0.3
}

# This month's survivability step: toward the cap at 2 % of the gap per
# hardening level, or 1 % of the excess back toward the floor without it.
nd_survivability_step = {
	value = 0
	if = {
		limit = {
			has_variable = nd_survivability
			has_variable = nd_hardening
			var:nd_hardening >= 1
		}
		add = nd_survivability_cap
		subtract = var:nd_survivability
		multiply = {
			value = var:nd_hardening
			multiply = 0.02
		}
		min = 0
	}
	else_if = {
		limit = {
			has_variable = nd_survivability
			var:nd_survivability > nd_survivability_floor
		}
		add = var:nd_survivability
		subtract = nd_survivability_floor
		multiply = -0.01
	}
}

# This country's crisis danger band, 0–3, whichever side it is on.
nd_own_danger_band = {
	value = 0
	if = {
		limit = {
			nd_is_crisis_owner = yes
			has_variable = nd_crisis_danger
		}
		add = var:nd_crisis_danger
	}
	else_if = {
		limit = {
			nd_is_crisis_target = yes
			exists = var:nd_crisis_opponent
		}
		var:nd_crisis_opponent = {
			if = {
				limit = { has_variable = nd_crisis_danger }
				add = var:nd_crisis_danger
			}
		}
	}
	divide = 25
	floor = yes
	max = 3
}

# Monthly incident probability, per mille. One roll per country per month,
# shared across any crisis (a weekly crisis tick never rolls).
#   routine 1, heightened 4, high alert 10
#   × (1 + strain/100) × (0.5 + (100 − reliability)/100) × (1 + 0.5 × danger band)
#   capped at 30 ‰
# nd_debug_incident_odds (console harness only) forces 1000.
nd_incident_permille = {
	value = 1
	if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 2
		}
		value = 4
	}
	else_if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 3
		}
		value = 10
	}
	if = {
		limit = { has_variable = nd_strain }
		multiply = {
			value = var:nd_strain
			divide = 100
			add = 1
		}
	}
	if = {
		limit = { has_variable = nd_reliability }
		multiply = {
			value = 100
			subtract = var:nd_reliability
			divide = 100
			add = 0.5
		}
	}
	multiply = {
		value = nd_own_danger_band
		multiply = 0.5
		add = 1
	}
	max = 30
	if = {
		limit = { has_variable = nd_debug_incident_odds }
		value = 1000
	}
}

# Chance (percent) that a launch sequence started under delegated authority or
# launch on warning is halted by someone in the chain — the Petrov or
# Arkhipov who refuses. There is no player veto on these branches (§8.1); this
# number is the whole of the protection, which is why the panel shows it.
nd_hold_chance = {
	value = 60
	if = {
		limit = { has_variable = nd_reliability }
		value = var:nd_reliability
	}
	if = {
		limit = { has_variable = nd_safeguards }
		add = {
			value = var:nd_safeguards
			multiply = 5
		}
	}
	if = {
		limit = { has_variable = nd_punished_skeptic }
		subtract = 15
	}
	if = {
		limit = { nd_authority_launch_on_warning = yes }
		subtract = 10
	}
	min = 20
	max = 97
}

# The doctrine tables a group can judge with.
nd_ig_doctrine_militarist_value = {
	value = 0
	owner = {
		if = {
			limit = { nd_doctrine_nfu = yes }
			add = -2
		}
		else_if = {
			limit = { nd_doctrine_existential = yes }
			add = -1
		}
		else_if = {
			limit = { nd_doctrine_compellence = yes }
			add = 1
		}
		else_if = {
			limit = { nd_doctrine_warfighting = yes }
			add = 2
		}
	}
}

nd_ig_doctrine_restraint_value = {
	value = 0
	owner = {
		if = {
			limit = { nd_doctrine_nfu = yes }
			add = 2
		}
		else_if = {
			limit = { nd_doctrine_existential = yes }
			add = 1
		}
		else_if = {
			limit = { nd_doctrine_flexible = yes }
			add = -1
		}
		else = {
			add = -2
		}
	}
}

nd_ig_doctrine_professional_value = {
	value = 0
	owner = {
		if = {
			limit = {
				OR = {
					nd_doctrine_nfu = yes
					nd_doctrine_warfighting = yes
				}
			}
			add = -1
		}
		else_if = {
			limit = { nd_doctrine_flexible = yes }
			add = 1
		}
	}
}

# A militarist or restraint stance picks its table, for any group (a class,
# or the Armed Forces' and Industrialists' lean); the Armed Forces with no
# lean judge as officers; the Industrialists with none have no doctrine view.
# A mild stance is capped at ±1.
nd_ig_term_doctrine_value = {
	value = 0
	if = {
		limit = { nd_ig_posture_judged = yes }
		if = {
			limit = { nd_ig_stance_militarist = yes }
			add = nd_ig_doctrine_militarist_value
		}
		else_if = {
			limit = { nd_ig_stance_restraint = yes }
			add = nd_ig_doctrine_restraint_value
		}
		else_if = {
			limit = { nd_ig_class_professional = yes }
			add = nd_ig_doctrine_professional_value
		}
		if = {
			limit = {
				OR = {
					nd_ig_stance_militarist = yes
					nd_ig_stance_restraint = yes
				}
				nd_ig_stance_full = no
			}
			min = -1
			max = 1
		}
	}
}

nd_ig_term_readiness_value = {
	value = 0
	if = {
		limit = { nd_ig_posture_judged = yes }
		if = {
			limit = { nd_ig_class_militarist = yes }
			owner = {
				if = {
					limit = { var:nd_readiness = 1 }
					add = -1
				}
				else_if = {
					limit = { var:nd_readiness = 3 }
					add = 1
				}
			}
		}
		else_if = {
			limit = { nd_ig_class_restraint = yes }
			owner = {
				if = {
					limit = { var:nd_readiness = 3 }
					add = -1
				}
			}
		}
		else_if = {
			limit = { nd_ig_class_professional = yes }
			owner = {
				if = {
					limit = {
						var:nd_readiness = 1
						nd_has_plausible_attacker = yes
					}
					add = -1
				}
				else_if = {
					limit = { var:nd_readiness = 2 }
					add = 1
				}
			}
		}
	}
}

nd_ig_term_authority_value = {
	value = 0
	if = {
		limit = {
			nd_ig_posture_judged = yes
			nd_ig_class_restraint = yes
			owner = { nd_authority_launch_on_warning = yes }
		}
		add = -1
	}
}

nd_ig_term_strain_value = {
	value = 0
	if = {
		limit = {
			nd_ig_posture_judged = yes
			nd_ig_class_professional = yes
			owner = {
				var:nd_readiness = 3
				has_variable = nd_strain
				var:nd_strain >= 50
			}
		}
		add = -1
	}
}

# Prolonged alerts and open crises are bad for business. Not tenure-gated.
nd_ig_term_business_value = {
	value = 0
	if = {
		limit = { nd_ig_class_business = yes }
		owner = {
			if = {
				limit = {
					nd_is_initialized = yes
					var:nd_readiness = 3
					has_variable = nd_readiness_months
					var:nd_readiness_months >= 3
				}
				add = -1
			}
			if = {
				limit = { nd_crisis_stage_at_least = { STAGE = 2 } }
				add = -1
			}
		}
	}
}

			nd_ig_stance_full = no
		}

		max = 1
	}
	min = -2
	max = 2
}
