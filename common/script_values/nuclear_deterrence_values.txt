# The stockpile, 0 if never created.
nd_stockpile = {
	value = 0
	if = {
		limit = { has_variable = nuclear_weapon_stockpile }
		add = var:nuclear_weapon_stockpile
	}
	min = 0
}

# Arsenal size as it bears on costs: warheads beyond fifty add nothing (§4 —
# diminishing returns, real upkeep).
nd_stockpile_capped = {
	value = nd_stockpile
	max = 50
}

# 0.4 for a token arsenal up to 1.0 for fifty warheads: how much of the
# readiness bill an arsenal of this size runs up.
nd_upkeep_size_factor = {
	value = nd_stockpile_capped
	divide = 50
	multiply = 0.6
	add = 0.4
}

# ----------------------------------------------------------------------------
# UPKEEP — weekly, GDP-scaled (§3.2)
# ----------------------------------------------------------------------------
# Worked in units of 1/100,000 of GDP per week, so no constant is small enough
# to fall below fixed-point precision; one unit is ≈ 0.052 % of GDP a year.
#   custody        0.2 × warheads (capped at 50)          ≤ 10 units
#                  (0.1 at Recessed: warheads in storage)
#   heightened     4 × size factor                         ≤  4 units
#   high alert     12 × size factor                        ≤ 12 units
#   safeguards     2 per level (0–3)                       ≤  6 units
#   hardening      3 per level (0–3)                       ≤  9 units
#   automatic retaliation (authority 4)                      3 units
#   missile submarines  6 × size factor (fielded, a coast)  ≤  6 units
#   cap            46 units ≈ 2.4 % of GDP a year: the sum of the parts, so
#                  no part is ever free (phase 2 plan D9)
nd_upkeep_units_custody = {
	value = nd_stockpile_capped
	multiply = 0.2
}

# Custody at Recessed: warheads in storage cost half to keep.
nd_upkeep_units_custody_recessed = {
	value = nd_upkeep_units_custody
	multiply = 0.5
}

# The upkeep's ceiling, in units: the sum of every part at its dearest. It
# rose from 40 with the submarine leg (phase 2 plan D9): at 40, the dearest
# posture's submarines would have cost nothing.
nd_upkeep_units_cap = 46

# The missile submarine leg, the dearest to keep: crews at sea and their boats
# on patrol (2026-11-20 delivery spec §4.3; owner call 11: a first figure, to
# tune in play-testing). It scales with the arsenal, as alert does.
nd_upkeep_units_submarines = {
	value = 0
	if = {
		limit = { nd_fields_missile_submarines = yes }
		add = nd_upkeep_units_submarines_fielded
	}
}

# What the leg costs while at sea, fielded or not yet: the Focus row's figure
# for the project (phase 2 plan D16) reads it too.
nd_upkeep_units_submarines_fielded = {
	value = nd_upkeep_size_factor
	multiply = 6
}

# What does not depend on readiness: safeguards, hardening, an Automatic
# Retaliation system (3 units, as one hardening level) and the missile
# submarines.
nd_upkeep_units_fixed = {
	value = 0
	add = nd_upkeep_units_submarines
	if = {
		limit = { nd_authority_automatic = yes }
		add = 3
	}
	if = {
		limit = { has_variable = nd_safeguards }
		add = {
			value = var:nd_safeguards
			multiply = 2
		}
	}
	if = {
		limit = { has_variable = nd_hardening }
		add = {
			value = var:nd_hardening
			multiply = 3
		}
	}
}

# Custody at the readiness in force, plus the fixed items. The per-level
# previews below build from the same parts, so a button's cost is exact.
nd_upkeep_units_base = {
	value = nd_upkeep_units_fixed
	if = {
		limit = { nd_readiness_recessed = yes }
		add = nd_upkeep_units_custody_recessed
	}
	else = {
		add = nd_upkeep_units_custody
	}
}

nd_upkeep_units_readiness_2 = {
	value = nd_upkeep_size_factor
	multiply = 4
}

nd_upkeep_units_readiness_3 = {
	value = nd_upkeep_size_factor
	multiply = 12
}

nd_upkeep_units = {
	value = nd_upkeep_units_base
	if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 2
		}
		add = nd_upkeep_units_readiness_2

# Where hardening takes survivability (2026-11-20 delivery spec §3.1): each
# part of our forces removes a share of what is still exposed to a first
# strike, so every part counts, by less the more is already protected, and
# nothing reaches 100. One factor a part, 1 while the part isn't held, so the
# panel can print the chain and scripts/analysis/nuclear_survivors.py can
# mirror it. Since phase 1 this replaces the flat ceiling (25 + 15 + 20 + 25 +
# 10, at most 80), so "ceiling" in a stored variable's name means this target.
nd_exposure_factor_bombers = {
	value = 0.75                                # dispersed bombers, always
}

nd_exposure_factor_radar = {
	value = 1
	if = {
		limit = { has_technology_researched = radar }
		value = 0.80                            # warning time to get bombers up
	}
}

# The two legs' factors while fielded, named so the Focus rows' terms and the
# target with a leg read the same figures (phase 2 plan D16).
nd_exposure_factor_icbms_held = 0.70            # hardened silos

nd_exposure_factor_submarines_held = 0.50       # missile submarines at sea

nd_exposure_factor_icbms = {
	value = 1
	if = {
		limit = { nd_fields_icbms = yes }
		value = nd_exposure_factor_icbms_held
	}
}

nd_exposure_factor_submarines = {
	value = 1
	if = {
		limit = { nd_fields_missile_submarines = yes }
		value = nd_exposure_factor_submarines_held
	}
}

nd_exposure_factor_missile_defense = {
	value = 1
	if = {
		limit = { has_technology_researched = missile_defense_systems }
		value = 0.85
	}
}

# The share of the force a first strike could still reach at the target.
nd_survivability_exposed_share = {
	value = nd_exposure_factor_bombers
	multiply = nd_exposure_factor_radar
	multiply = nd_exposure_factor_icbms
	multiply = nd_exposure_factor_submarines
	multiply = nd_exposure_factor_missile_defense

nd_survivability_target = {
	value = 1
	subtract = nd_survivability_exposed_share
	multiply = 100
}

# The target with a leg fielded, for that project's Focus row (phase 2 plan

	multiply = nd_exposure_factor_radar
	multiply = nd_exposure_factor_icbms_held
	multiply = nd_exposure_factor_submarines

	multiply = nd_exposure_factor_submarines_held
	multiply = nd_exposure_factor_missile_defense
	multiply = -1

	multiply = 100

# Without investment, survivability decays toward this floor.
nd_survivability_floor = {
	value = nd_survivability_target
	multiply = 0.3
}

# This month's survivability step: toward the target at 2 % of the gap per
# hardening level, or 1 % of the excess back toward the floor without it.
nd_survivability_step = {
	value = 0
	if = {
		limit = {
			has_variable = nd_survivability
			has_variable = nd_hardening
			var:nd_hardening >= 1
		}
		# No floor: above the target (a save from before it fell, or a part
		# lost) the step is negative and settles to it.
		add = nd_survivability_target
		subtract = var:nd_survivability
		multiply = {
			value = var:nd_hardening
			multiply = nd_hardening_rate_per_level
		}
	}
	else_if = {
		limit = {
			has_variable = nd_survivability
			var:nd_survivability > nd_survivability_floor
		}
		add = nd_survivability_floor
		subtract = var:nd_survivability
		multiply = nd_survivability_erosion_rate
	}
}

# The share of the gap to the target each hardening level closes a month, and
# the share of the lead over the floor lost a month without hardening.
nd_hardening_rate_per_level = 0.02

nd_survivability_erosion_rate = 0.01

# The step at one hardening level more or less, for the stepper's buttons
# (2026-11-20 delivery spec D12): nd_survivability_step at that level.
nd_survivability_step_at_hardening_up = {
	value = 0
	if = {
		limit = { has_variable = nd_survivability }
		add = nd_survivability_target
		subtract = var:nd_survivability
		multiply = {
			value = nd_display_hardening
			add = 1
			multiply = nd_hardening_rate_per_level
		}
	}
}

nd_survivability_step_at_hardening_down = {
	value = 0
	if = {
		limit = {
			has_variable = nd_survivability
			has_variable = nd_hardening
			var:nd_hardening >= 2
		}
		add = nd_survivability_target
		subtract = var:nd_survivability
		multiply = {
			value = var:nd_hardening
			subtract = 1
			multiply = nd_hardening_rate_per_level
		}
	}
	else_if = {
		limit = {
			has_variable = nd_survivability
			var:nd_survivability > nd_survivability_floor
		}
		add = nd_survivability_floor
		subtract = var:nd_survivability
		multiply = nd_survivability_erosion_rate
	}
}

# This country's crisis danger band, 0–3, whichever side it is on.
nd_own_danger_band = {
	value = 0
	if = {
		limit = {
			nd_is_crisis_owner = yes
			has_variable = nd_crisis_danger
		}
		add = var:nd_crisis_danger
	}
	else_if = {
		limit = {
			nd_is_crisis_target = yes
			exists = var:nd_crisis_opponent
		}
		var:nd_crisis_opponent = {
			if = {
				limit = { has_variable = nd_crisis_danger }
				add = var:nd_crisis_danger
			}
		}
	}
	divide = 25
	floor = yes
	max = 3
}

# Meaningful incident exposure, per mille PER WEEK. Routine mishaps use their
# own monthly roll and never dilute this pool. Peacetime bases below are monthly
# equivalents divided by 52/12; the crisis surcharge is already weekly.
# Benchmark: two high-alert powers, danger >=75, strain 0, reliability 60,
# Existential doctrine: ~28% chance of an incident across two weekly pulses each
# (calibrated 2026-10-28 to the rework spec's §2.3 launch bands; ~50% before).
#
# Two parts, one roll. The base part draws from every family this country is
# eligible for; the crisis part draws only from the misreadings a crisis
# drives (a warning, an exercise, a commander cut off), never a weapons
# accident in a depot, and is zero when none of them is eligible
# (nd_fire_incident).
nd_incident_base_monthly_permille = {
	value = 1
	if = { limit = { nd_readiness_recessed = yes } value = 0.5 }
	else_if = { limit = { nd_readiness_high_alert = yes } value = 10 }
	else_if = { limit = { nd_readiness_heightened_or_higher = yes } value = 4 }
}

# A war with an enemy believed armed (rework spec §5.1): band 2, or band 3 when
# the war threatens either side (goals on incorporated states, a side losing,
# a nuclear weapon landed). Alongside a crisis the higher band applies.
nd_war_danger_band = {
	value = 0
	if = {
		limit = { nd_war_band_3_against_any = yes }
		value = 3
	}
	else_if = {
		limit = {
			any_enemy_in_war = { nd_believed_armed = yes }
		}
		value = 2
	}
}

nd_incident_band = {
	value = nd_own_danger_band
	if = {
		limit = { nd_war_danger_band > nd_own_danger_band }
		value = nd_war_danger_band
	}
}

# Calibrated 2026-10-28 to the rework spec's §2.3 Acute-crisis bands and its
# §2.1 war-year rates (scripts/analysis/nuclear_incident_rates.py,
# test_nuclear_incident_risk.TestSpecBands): band 3 was 175, band 2 50.
nd_incident_crisis_weekly_permille = {
	value = 0
	if = { limit = { nd_incident_band >= 3 } value = 85 }
	else_if = { limit = { nd_incident_band >= 2 } value = 35 }
	else_if = { limit = { nd_incident_band >= 1 } value = 5 }
	# A force below operational alert is harder to misread.
	if = { limit = { nd_readiness_heightened_or_higher = no } multiply = 0.25 }
	else_if = { limit = { nd_readiness_high_alert = no } multiply = 0.6 }
	if = { limit = { nd_incident_misreading_eligible = no } value = 0 }
}

nd_incident_doctrine_multiplier = {
	value = 1
	if = { limit = { nd_doctrine_warfighting = yes } value = 1.6 }
	else_if = { limit = { nd_doctrine_compellence = yes } value = 1.35 }
	else_if = { limit = { nd_doctrine_flexible = yes } value = 1.15 }
}

# Strain, reliability and doctrine scale both parts alike.
nd_incident_rate_multiplier = {
	value = 1
	if = {
		limit = { has_variable = nd_strain }
		multiply = { value = var:nd_strain divide = 100 add = 1 }
	}
	if = {
		limit = { has_variable = nd_reliability }
		multiply = { value = 100 subtract = var:nd_reliability divide = 100 add = 0.5 }
	}
	multiply = nd_incident_doctrine_multiplier
}

# Halved 2026-10-16 (it fired every two years or so at High Alert and read as

# What Launch on Warning takes from the chance of a halt, and what a
# safeguards level adds to it (nd_authority_terms_3 and
# nd_safeguards_level_terms print them; tests hold the figures to these).
nd_hold_launch_on_warning_cut = 10

nd_safeguards_halt_per_level = 5

nd_hold_chance_unclamped = {
	value = 60
	if = {
		limit = { has_variable = nd_reliability }
		value = var:nd_reliability
	}
	# Half of it, on a floor of 53: someone in the chain doubts a warning
	# however poorly the force is run, and safeguards already raise
	# reliability. Calibrated 2026-10-28 so that Launch on Warning with no
	# safeguards is reckless (a quarter over an Acute crisis), not suicidal
	# (two thirds): rework spec §2.3.
	multiply = 0.5
	add = 53
	if = {
		limit = { has_variable = nd_safeguards }
		add = {
			value = var:nd_safeguards
			multiply = nd_safeguards_halt_per_level
		}
	}
	if = {
		limit = { has_variable = nd_punished_skeptic }
		subtract = 15
	}
	if = {
		limit = { nd_authority_launch_on_warning = yes }
		subtract = nd_hold_launch_on_warning_cut
	}
	# The moment the warning arrives in (the Petrov judgement): at peace with
	# no crisis, a real first strike makes no sense and a few missiles on a
	# screen look like a fault; in a Warning or a Confrontation it is
	# thinkable; in an Acute crisis or a war nothing argues against it.
	if = {
		limit = {
			is_at_war = no
			nd_in_crisis = no
		}
		add = 25
	}
	else_if = {
		limit = {
			is_at_war = no
			NOT = { nd_crisis_stage_at_least = { STAGE = 3 } }
		}
		add = 10
	}
	# The suspect's posture (rework spec §4.3): forces in storage cannot be
	# launching, so a warning about them is easy to doubt; forces at High
	# Alert make it easy to believe. The caller saves the suspect as
	# scope:nd_hold_suspect; the panel's figure has none.
	if = {
		limit = { exists = scope:nd_hold_suspect }
		scope:nd_hold_suspect = {
			if = {
				limit = { nd_readiness_recessed = yes }
				add = 20
			}
			else_if = {
				limit = { nd_readiness_high_alert = yes }
				add = -5
			}
		}
	}
	# A hotline: the crisis's talks are open, so someone can be asked.
	if = {
		limit = { nd_crisis_talks_open_either = yes }
		add = 10
	}
	# Our own vulnerability: commanders who fear losing the force are quicker
	# to use it. The shape of the crisis danger's vulnerability part.
	subtract = nd_own_vulnerability_value
}

# Percent chance that the warheads landed on a Central Authorization
# country's command cut it: max(0, 1 − (R/L)²) (owner, 2026-11-19). L counts
# the capital's and, for a Leadership salvo, the command's other sites
# (nd_salvo_hit; owner, 2026-11-19: half the warheads go elsewhere, and all
# count).
# Nothing below L = R, as before (owner call 7); above it the chance climbs
# fast, so a full salvo of 25 on an arsenal the AI builds nearly always cuts
# it. Until 2026-11-19 it was (L − R)/(L + R), at most 72% from 25 warheads
# on any power with both technologies.

	value = 0
	scope:nd_sv_enemy = {
		if = {

				value = var:nd_salvo_victim_r
				divide = var:nd_salvo_command_landed
				multiply = {
					value = var:nd_salvo_victim_r
					divide = var:nd_salvo_command_landed
				}
			}
		}
	}
	multiply = 100
	min = 0
}

# What yielding costs the target itself (#1893). In a war, conceding drops the
# war goals it holds against the other side, and those to free or take one of
# that side's direct subjects, whether it leads its side or not
# (nd_crisis_concede_war; the set nd_holds_any_goal_against reads). Weighed by
# type on the stakes' scale at half, since giving up what it hopes to win
# weighs less than facing a loss: an annexation or reunification −15 and
# nothing else counted; a subjugation type or regime change −10; −5 for each
# type that takes a state or a subject; −2 for each lesser type; at most −20,
# then half while it is losing the war to the other side, whose goals it is
# unlikely to enforce. Only where the yield is the war concession: dispute 2,
# or 3 while the target fights the protégé (in a play it backs down and keeps
# its goals, and these reads find no war goal there). The AI's stake gate
# (nd_ai_stakes_block) asks before any crisis, with no dispute stored, where
# a war between the two is dispute 2. Target scope; scope:nd_issuer and
# scope:nd_forfeit_to (the country the yield names) saved.
nd_yp_forfeit_value = {
	value = 0
	if = {
		limit = {
			exists = scope:nd_forfeit_to
			scope:nd_issuer = {
				OR = {
					NOT = { has_variable = nd_crisis_dispute }
					var:nd_crisis_dispute ?= 2
					var:nd_crisis_dispute ?= 3
				}
			}
		}
		if = {
			limit = {
				OR = {
					war_participant_has_war_goal_of_type_against = { type = annex_country target = scope:nd_forfeit_to }
					war_participant_has_war_goal_of_type_against = { type = te_reunify_country target = scope:nd_forfeit_to }
					war_participant_has_war_goal_of_type_against = { type = unification target = scope:nd_forfeit_to }
					war_participant_has_war_goal_of_type_against = { type = restore_union target = scope:nd_forfeit_to }
					war_participant_has_war_goal_of_type_against = { type = te_colonial_reconquest target = scope:nd_forfeit_to }
				}
			}
			add = -15
		}
		else = {
			if = {
				limit = {
					OR = {
						war_participant_has_war_goal_of_type_against = { type = make_protectorate target = scope:nd_forfeit_to }
						war_participant_has_war_goal_of_type_against = { type = make_dominion target = scope:nd_forfeit_to }
						war_participant_has_war_goal_of_type_against = { type = make_tributary target = scope:nd_forfeit_to }
						war_participant_has_war_goal_of_type_against = { type = make_personal_union target = scope:nd_forfeit_to }
						war_participant_has_war_goal_of_type_against = { type = make_crown_land target = scope:nd_forfeit_to }
						war_participant_has_war_goal_of_type_against = { type = make_chartered_company target = scope:nd_forfeit_to }
						war_participant_has_war_goal_of_type_against = { type = regime_change target = scope:nd_forfeit_to }
					}
				}
				add = -10
			}
			add = nd_yp_forfeit_territory_value
			add = nd_yp_forfeit_minor_value
		}
		min = -20
		if = {
			limit = { nd_is_losing_war_to = { ENEMY = scope:nd_forfeit_to } }
			multiply = 0.5
		}
	}
	round = yes
}

# The doctrine tables a group can judge with.
nd_ig_doctrine_militarist_value = {
	value = 0
	owner = {
		if = {
			limit = { nd_doctrine_nfu = yes }
			add = -2
		}
		else_if = {
			limit = { nd_doctrine_existential = yes }
			add = -1
		}
		else_if = {
			limit = { nd_doctrine_compellence = yes }
			add = 1
		}
		else_if = {
			limit = { nd_doctrine_warfighting = yes }
			add = 2
		}
	}
}

nd_ig_doctrine_restraint_value = {
	value = 0
	owner = {
		if = {
			limit = { nd_doctrine_nfu = yes }
			add = 2
		}
		else_if = {
			limit = { nd_doctrine_existential = yes }
			add = 1
		}
		else_if = {
			limit = { nd_doctrine_flexible = yes }
			add = -1
		}
		else = {
			add = -2
		}
	}
}

nd_ig_doctrine_professional_value = {
	value = 0
	owner = {
		if = {
			limit = {
				OR = {
					nd_doctrine_nfu = yes
					nd_doctrine_warfighting = yes
				}
			}
			add = -1
		}
		else_if = {
			limit = { nd_doctrine_flexible = yes }
			add = 1
		}
	}
}

# A militarist or restraint view picks its table (a class, or the Armed
# Forces' and Industrialists' lean from their leader); the Armed Forces with
# no lean judge as officers; the Industrialists with none have no doctrine
# view. A mild view is capped at ±1.
nd_ig_term_doctrine_value = {
	value = 0
	if = {
		limit = { nd_ig_posture_judged = yes }
		if = {
			limit = { nd_ig_is_militarist = yes }
			add = nd_ig_doctrine_militarist_value
		}
		else_if = {
			limit = { nd_ig_is_restraint = yes }
			add = nd_ig_doctrine_restraint_value
		}
		else_if = {
			limit = { nd_ig_class_professional = yes }
			add = nd_ig_doctrine_professional_value
		}
		if = {
			limit = {
				OR = {
					nd_ig_is_militarist = yes
					nd_ig_is_restraint = yes
				}
				nd_ig_view_full = no
			}
			min = -1
			max = 1
		}
	}
}

nd_ig_term_readiness_value = {
	value = 0
	if = {
		limit = { nd_ig_posture_judged = yes }
		if = {
			limit = { nd_ig_class_militarist = yes }
			owner = {
				if = {
					limit = { var:nd_readiness <= 1 }
					add = -1
				}
				else_if = {
					limit = { var:nd_readiness = 3 }
					add = 1
				}
			}
		}
		else_if = {
			limit = { nd_ig_class_restraint = yes }
			owner = {
				if = {
					limit = { var:nd_readiness = 3 }
					add = -1
				}
				# Warheads in storage: the restraint they ask for.
				else_if = {
					limit = { var:nd_readiness = 0 }
					add = 1
				}
			}
		}
		else_if = {
			limit = { nd_ig_class_professional = yes }
			owner = {
				# Officers never like a force that cannot fire for weeks.
				if = {
					limit = { var:nd_readiness = 0 }
					add = -1
				}
				else_if = {
					limit = {
						var:nd_readiness = 1
						nd_has_plausible_attacker = yes
					}
					add = -1
				}
				else_if = {
					limit = { var:nd_readiness = 2 }
					add = 1
				}
			}
		}
	}
}

nd_ig_term_authority_value = {
	value = 0
	if = {
		limit = {
			nd_ig_posture_judged = yes
			nd_ig_class_restraint = yes
			owner = {
				OR = {
					nd_authority_launch_on_warning = yes
					nd_authority_automatic = yes
				}
			}
		}
		add = -1
	}
}

nd_ig_term_strain_value = {
	value = 0
	if = {
		limit = {
			nd_ig_posture_judged = yes
			nd_ig_class_professional = yes
			owner = {
				var:nd_readiness = 3
				has_variable = nd_strain
				var:nd_strain >= 50
			}
		}
		add = -1
	}
}

# Prolonged alerts and open crises are bad for business. Not tenure-gated.
nd_ig_term_business_value = {
	value = 0
	if = {
		limit = { nd_ig_class_business = yes }
		owner = {
			if = {
				limit = {
					nd_is_initialized = yes
					var:nd_readiness = 3
					has_variable = nd_readiness_months
					var:nd_readiness_months >= 3
				}
				add = -1
			}
			if = {
				limit = { nd_crisis_stage_at_least = { STAGE = 2 } }
				add = -1
			}
		}
	}
}

# The nuclear taboo (nuclear_taboo_values.txt §4.4): restraint groups object to
# the arsenal itself once the world holds the bomb in horror, and more to a big
# one. Not tenure-gated — holding warheads is not a posture choice.
nd_ig_term_possession_value = {
	value = 0
	if = {
		limit = { nd_ig_class_restraint = yes }
		owner = {
			if = {
				limit = { nd_taboo_burden_value >= nd_taboo_burden_step_full }
				add = -2
			}
			else_if = {
				limit = { nd_taboo_burden_value >= nd_taboo_burden_step_mild }
				add = -1
			}
		}
	}
}

		limit = { nd_fields_missile_submarines = yes }

		}

			has_variable = nd_readiness

		}

			value = nd_ceiling_lowered_capped
			divide = 50
			multiply = 0.6
			add = 0.4
			multiply = 4
		}
	}
	else_if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 3
		}
		add = {
			value = nd_ceiling_lowered_capped
			divide = 50
			multiply = 0.6
			add = 0.4
			multiply = 12
		}

	multiply = {
		value = gdp

		value = nd_ceiling_raised_capped
		multiply = 0.2
		if = {
			limit = { nd_readiness_recessed = yes }
			multiply = 0.5
		}
	}
	if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 2
		}
		add = {
			value = nd_ceiling_raised_capped
			divide = 50
			multiply = 0.6
			add = 0.4
			multiply = 4
		}
	}
	else_if = {
		limit = {
			has_variable = nd_readiness
			var:nd_readiness = 3
		}
		add = {
			value = nd_ceiling_raised_capped
			divide = 50
			multiply = 0.6
			add = 0.4
			multiply = 12
		}
	}
	min = 0
	max = nd_upkeep_units_cap
	multiply = {
		value = gdp
		divide = 100000
	}
}

# ---- Trades (rework spec §8.6, phase 4c) --------------------------------------
# Parties saved (scope:nd_issuer, scope:nd_target), in a crisis over a dispute
# that trades. What the target would give up less what the issuer would, on
# §8.2's scale (ruling R6): a basing or guarantee treaty 10, like a state; a
# missile-defense state 4, a lesser goal; a claimed incorporated state 10; a
# caught covert operation 4; an active battlestation site 10. Within disputes
# 8–10 only the counts matter. Each half counts what the dispute's trigger and
# concession read (nd_dispute_*_by, nd_crisis_concede_*). The AI gives no more
# than it gets: a target accepts at 0 or less, an issuer at 0 or more.
nd_trade_balance_value = {
	value = 0
	if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 7 } }
		scope:nd_target = {
			every_scope_treaty = {
				limit = {
					any_scope_article = {
						OR = {
							has_type = nuclear_basing
							has_type = nuclear_guarantee
						}
						source_country = scope:nd_target
						target_country = {
							NOT = { THIS = scope:nd_issuer }
							is_adjacent_to_country = scope:nd_issuer
						}
					}
				}
				add = 10
			}
			every_scope_state = {
				limit = {
					nd_state_has_missile_defense = yes
					any_neighbouring_state = { owner = scope:nd_issuer }
				}
				add = 4
			}
		}
		scope:nd_issuer = {
			every_scope_treaty = {
				limit = {
					any_scope_article = {
						OR = {
							has_type = nuclear_basing
							has_type = nuclear_guarantee
						}
						source_country = scope:nd_issuer
						target_country = {
							NOT = { THIS = scope:nd_target }
							is_adjacent_to_country = scope:nd_target
						}
					}
				}
				subtract = 10
			}
			every_scope_state = {
				limit = {
					nd_state_has_missile_defense = yes
					any_neighbouring_state = { owner = scope:nd_target }
				}
				subtract = 4
			}
		}
	}
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 8 } }
		scope:nd_issuer = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
					state_region = { scope:nd_target = { has_claim = PREV } }
				}
				add = 10
			}
		}
		scope:nd_target = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
					state_region = { scope:nd_issuer = { has_claim = PREV } }
				}
				subtract = 10
			}
		}
	}
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 9 } }
		scope:nd_issuer = {
			if = {
				limit = { has_variable_list = iw_known_ops }
				every_in_list = {
					variable = iw_known_ops
					limit = {
						var:iw_attacker ?= scope:nd_target
						NOT = { has_tag = iw_known_counterintel_support }
					}
					add = 4
				}
			}
		}
		scope:nd_target = {
			if = {
				limit = { has_variable_list = iw_known_ops }
				every_in_list = {
					variable = iw_known_ops
					limit = {
						var:iw_attacker ?= scope:nd_issuer
						NOT = { has_tag = iw_known_counterintel_support }
					}
					subtract = 4
				}
			}
		}
	}
	else_if = {
		limit = { scope:nd_issuer = { var:nd_crisis_dispute = 10 } }
		scope:nd_target = {
			every_scope_state = {
				limit = {
					any_scope_building = {
						is_building_type = building_orbital_battlestation_construction_site
						NOT = { has_active_production_method = pm_orbital_battlestation_construction_paused }
					}
				}
				add = 10
			}
		}
		scope:nd_issuer = {
			every_scope_state = {
				limit = {
					any_scope_building = {
						is_building_type = building_orbital_battlestation_construction_site
						NOT = { has_active_production_method = pm_orbital_battlestation_construction_paused }
					}
				}
				subtract = 10
			}
		}
	}
}
