# Batched private investment (docs/systems/mod_systems/construction.md § Batched Private Investment). Effects:
# common/scripted_effects/te_batch_investment_effects.txt. Hooks: common/on_actions/te_batch_investment_on_actions.txt.
# test_batch_investment.py runs these values and the effects against a model of the engine's weekly pass.

# ---- Tuning ---------------------------------------------------------------------------------------------------------

# Where the engine stops queueing private levels in the largest economies: the 2050 report's queue stood at exactly
# 1,000 after every weekly pass. The likely source is NAI's CONSTRUCTION_MAX_NUM_PRODUCTION_BUILDING_CONSTRUCTIONS
# (00_ai.txt): 1 + 0.05 for each point of construction the country makes a week, the scaled part at most 999. If so
# the limit is each country's own and reaches 1,000 at about 20,000 construction a week; a country under that is held
# to fewer and never batches, since te_bi_month_end asks for a queue that reached te_bi_engine_queue_near. Script
# cannot read a define, so a change to those three would need this changed with it.
te_bi_engine_queue = 1000

# "At the limit": a few levels short of it still counts. The last picks of a pass may be abroad, which the hook does
# not see at home, and a few levels can finish between the pass and the monthly pulse.
te_bi_engine_queue_near = {
	value = te_bi_engine_queue
	subtract = 10
}

# The last slots under the engine's limit are left for it to fill a level at a time: script can only add levels at
# home, so the investors' picks abroad and their first level of a new building stay the engine's, and these slots are
# where it places them once the batches have filled the rest. Each costs one engine decision a week (about 22 ms in the
# USA of 2050). They are also how script sees that the engine still fills the queue to its limit (var:te_bi_maxq is
# read at the engine's picks), so the reserve must stay above the ten of te_bi_engine_queue_near.
te_bi_reserve = 100

# A batch is never queued past this.
te_bi_ceiling = {
	value = te_bi_engine_queue
	subtract = te_bi_reserve
}

# One pick of the engine's stands for at most a hundred levels, its own included ("pick 100 levels at a time instead
# of 1", owner, 2026-11-26), and a building never has more than a tenth of its built levels queued.
te_bi_batch_max = 100
te_bi_share = 0.1

# A building that makes a good no market carries (te_bi_local_goods_building): ten levels a pick, and a fiftieth of
# its size queued at most. Too many levels of these at once cannot be sold anywhere else (owner, 2026-11-26).
te_bi_local_batch_max = 10
te_bi_local_share = 0.02

# What a batch adds beside the engine's own level, at most.
te_bi_batch_extra_max = {
	value = te_bi_batch_max
	subtract = 1
}

te_bi_local_batch_extra_max = {
	value = te_bi_local_batch_max
	subtract = 1
}

# A country is watched (its investors' picks counted) from this many private levels queued at a monthly pulse, and
# until it has fewer than te_bi_unwatch_queue while not batching. A country that places 200 levels a week holds at
# least 200 in its queue, since a level takes a week or more.
te_bi_watch_queue = 200
te_bi_unwatch_queue = 100

# Batching starts when the investors placed this many levels at home in a month, about 200 a week ("once we're
# building hundreds of levels a week", owner, 2026-11-26), and stops under te_bi_off_levels.
te_bi_on_levels = 800
te_bi_off_levels = 400

# ---- Sizing ---------------------------------------------------------------------------------------------------------

# Building scope: the levels a batch adds. The room is a share of the building's built levels less what is queued for
# it already, in any queue, the level the engine has just queued included; a batch takes that room, up to a hundred
# levels a pick with the engine's own. So a building never has more than a tenth of its size queued, however long its
# levels take; each further pick of it in the same pass adds another hundred until the tenth is reached; and a new
# building or one under twenty levels (a hundred, for a local good's) gets no batch.
#
# Until the first run in game (2026-11-27) the hundred also capped what a building could have queued, so every
# further pick of a building that held its hundred was a single level: 315 of the 597 picks of the USA's first two
# batching passes were such picks, of buildings of several thousand levels, at 22 ms each.
te_bi_batch_extra = {
	value = level
	if = {
		limit = { te_bi_local_goods_building = yes }
		multiply = te_bi_local_share
	}
	else = {
		multiply = te_bi_share
	}
	floor = yes
	add = level
	subtract = level_after_queued_constructions
	if = {
		limit = { te_bi_local_goods_building = yes }
		max = te_bi_local_batch_extra_max
	}
	else = {
		max = te_bi_batch_extra_max
	}
	min = 0
}

# Country scope, a watched country: the levels its investors placed at home since the last monthly pulse. The picks the
# hook saw and the levels the batches added; not var:te_bi_other, the queue's growth it did not see at home, which is
# mostly picks abroad.
te_bi_placed = {
	value = var:te_bi_home
	add = var:te_bi_added
}

# Country scope, a watched country: the most unseen growth a hook that runs at each pick would leave (te_bi_selfcheck_hook).
# Twenty times the picks seen at home is investors placing 95% of their levels abroad.
te_bi_other_seen_limit = {
	value = var:te_bi_home
	multiply = 20
}

# Country scope: what the investment pool paid out this week, which is what it spent on construction.
te_bi_pool_spending = {
	value = investment_pool_gross_income
	subtract = investment_pool_net_income
}
