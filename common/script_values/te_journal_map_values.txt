# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The Map Room's selector. Each
# map's color scale, its factor 0-1, is generated: te_journal_map_scales_generated.txt, from the table in
# scripts/generators/gen_journal_map_scales.py (one shape per family, so a change to a shape changes every map using it).

# ---------------------------------------------------------------------------
# The selector: which of the Map Room's maps the player picked
# ---------------------------------------------------------------------------
# 1 turmoil, 3 life expectancy, 4 fertility, 5 median age, 6 devastation, 7 natural change, 8 death rate, 9 birth
# rate, 10 infant mortality, 11 leading cause of death, 12 sex balance, 13 net migration, 14 aged 65 and over, 15 urban
# share, 16 Gini coefficient, 17 Dynastic Wealth; the systems' (each also on its own entry): 18 emissions, 19 nuclear
# weapons, 20 covert networks, 21 the United Nations, 22 political models, 23 ocean basins, 24 grandeur, 25 colonies;
# the Map Room's own again: 26 tourism's Base Appeal, 27 Base Comfort (Local Climate before any technology), 28 Current
# Comfort (after the owner's), 29 GDP per capita, 30 Migration Crowding; 31 power grids (the Power Grids tab's map,
# 2026-11-07); 32 city rank (tourism's), 33 armed forces (Soldiers and Officers, the yearly walk's), 34 company
# headquarters (their weekly profit), 35 war dead (the census's, last year; 2026-11-08); 36 ownership (the walk's), 37
# population growth (the walk's), 38 family planning, 39 health care, 40 fighting-age men (the census's; 2026-11-08,
# second round); 41 capitalists, 42 aristocrats, 43 clergy, 44 academics (each class's share of the people, the
# walk's), 45 trade capacity (the state's own, its Trade Center's; 2026-11-11), 46 investment (what the state's owners
# pay into the Investment Pool, estimated; 2026-11-11); 0 none. 2 was literacy, retired because
# vanilla's own literacy map mode shows it (a save holding 2 shows the unpainted map and "Pick a map" until the next
# pick); 6 was the scope probe until the fourth session. A global variable (te_jmap_set_selector), since a state's map_color reaches neither the
# entry nor a country (the scope probe painted blue, 2026-11-05): one player's pick is every player's in multiplayer.
# Root-free, so the Map Room's state roots and the pick buttons' country roots read the same number.
te_jmap_selector = {
	value = 0
	if = {
		limit = { has_global_variable = te_jmap_sel }
		value = global_var:te_jmap_sel
	}
}

# ---------------------------------------------------------------------------
# The census's figures (state scope: the state being painted or hovered)
# ---------------------------------------------------------------------------
# The runner-up cause of death's deaths over the leader's, 0-1 (te_demog_dc_second_share over te_demog_dc_lead_share):
# the Leading Cause map stripes the runner-up's color over the leader's from 0.5, half the leader's deaths (owner,
# 2026-11-06). 0 for a state with no deaths, or whose last count came before te_dg_dc_second was stored.
te_jmap_cause_second_ratio = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_dc_total
			has_variable = te_dg_dc_lead
			has_variable = te_dg_dc_second
		}
		value = te_demog_dc_second_share
		divide = {
			value = te_demog_dc_lead_share
			min = 0.001
		}
	}
	min = 0
	max = 1
}

# The state's share of the world's greenhouse gas emissions last year, 0-1 (negative for a state that takes more out of
# the air than it puts in): its yearly snapshot (gw_state_emis, gw_snapshot_country_emissions_effect) over the world's
# annual total (gw_g_global_emis, re-based each January), both in internal units. A share, not an amount (owner,
# 2026-11-06), so a color means the same thing in 1850 and 2000, whatever the world emits.
te_jmap_gw_share = {
	value = 0
	if = {
		limit = {
			has_variable = gw_state_emis
			te_jmap_gw_world > 0
		}
		value = var:gw_state_emis
		divide = te_jmap_gw_world
	}
	min = -1
	max = 1
}

# The viewer's network in the state's owner, 0-100 (covert warfare's iw_nets: one container per target, on the
# operator, iw_net_strength; nd_loose_network_tier reads it the same way). 0 with no network, no viewer, or the viewer
# itself. root is the state: the map's, or the hover's (MakeScope).
te_jmap_net_strength = {
	value = 0
	if = {
		limit = {
			exists = owner
			global_var:te_jmap_viewer ?= { has_variable_list = iw_nets }
		}
		global_var:te_jmap_viewer = {
			every_in_list = {
				variable = iw_nets
				limit = {
					var:iw_target ?= root.owner
					has_variable = iw_net_strength
				}
				add = var:iw_net_strength
			}
		}
	}
	min = 0
	max = 100
}

# Company Headquarters (34): the weekly profit of the state's company headquarters, summed. Script can't read a
# company's profit (te_company_effects.txt), but each company has one headquarters building (its production method
# follows the company's category), and an ownership building's profit is what the levels it owns pay it, less its own
# wages: the company's earnings coming home. Whether it books them so is TE_SELF_CHECK hq_profit (te_jmap_hq_selfcheck).
#
# A headquarters is found by its building group, never its type. The engine makes a building type for each company type,
# building_<company type> (building_company_basic_fabrics, building_regional_company_hbc ...), from the two
# headquarters templates, so has_building = building_company_headquarter and is_building_type = building_company_
# headquarter match no state: a 1890 save held 61 headquarters and no building of either template type, and this map
# painted every state grey (2026-11-08). Vanilla counts them by group too (08_tsarpoints_buttons.txt).
#
# The map reads state variables the global monthly pulse writes (te_jmap_store_hqs, te_journal_map_effects.txt), as
# Armed Forces reads the yearly walk's: the paint walks no buildings, and a game's first month after a load is grey.

# The walk itself, state scope: the store's input only.
te_jmap_hq_walk_count = {
	value = 0
	every_scope_building = {
		limit = { is_building_group = bg_company_headquarter }
		add = 1
	}
}

te_jmap_hq_walk_profit = {
	value = 0
	every_scope_building = {
		limit = { is_building_group = bg_company_headquarter }
		add = weekly_profit
	}
}

# Ownership (36): who holds the state's capital, the yearly walk's building levels (te_demog_effects.txt: privately,
# by their own workers, by the state; ownership buildings, subsistence farms and government buildings left out). The
# leader's code (1 private, 2 their workers, 3 the state; a tie to the first of them), 0 for a state without a walk or
# with under one level of capital (te_demog_wc_capital_levels, Dynastic Wealth's own test).
te_jmap_own_lead = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_lv_priv
			has_variable = te_dg_lv_self
			has_variable = te_dg_lv_ctry
		}
		if = {
			limit = { te_demog_wc_capital_levels >= 1 }
			if = {
				limit = {
					var:te_dg_lv_priv >= var:te_dg_lv_self
					var:te_dg_lv_priv >= var:te_dg_lv_ctry
				}
				value = 1
			}
			else_if = {
				limit = { var:te_dg_lv_self >= var:te_dg_lv_ctry }
				value = 2
			}
			else = {
				value = 3
			}
		}
	}
}

# The leader's share of the state's capital levels, 1/3-1 (0 with no leader): the map fades the leader's color by it.
te_jmap_own_lead_share = {
	value = 0
	if = {
		limit = { te_jmap_own_lead = 1 }
		value = var:te_dg_lv_priv
		divide = { value = te_demog_wc_capital_levels min = 1 }
	}
	else_if = {
		limit = { te_jmap_own_lead = 2 }
		value = var:te_dg_lv_self
		divide = { value = te_demog_wc_capital_levels min = 1 }
	}
	else_if = {
		limit = { te_jmap_own_lead = 3 }
		value = var:te_dg_lv_ctry
		divide = { value = te_demog_wc_capital_levels min = 1 }
	}
}

# Population Growth (37): the change in the state's people from one yearly walk to the next, per 1,000 of last year's
# (te_dg_walk_pop against te_dg_walk_pop_prev, which te_demog_walk_keep_prev keeps every game, whatever the rule). It
# takes in births, deaths and migration, and also land gained or lost: a state that took or gave up provinces since
# its last walk jumps. 0 for a state without two walks.
te_jmap_pop_growth = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_walk_pop
			has_variable = te_dg_walk_pop_prev
		}
		if = {
			limit = { var:te_dg_walk_pop_prev > 0 }
			# the ratio first, then per 1,000: a state of tens of millions times 1,000 needn't fit the fixed point
			value = var:te_dg_walk_pop
			divide = var:te_dg_walk_pop_prev
			subtract = 1
			multiply = 1000
		}
	}
}

# Trade Capacity (45; Trade Volume until 2026-11-11, when it painted the owner's trade and every state of a country
# alike): the state's own Trade Capacity, what its Trade Center gives it (pm_trade_center's workforce-scaled
# state_trade_capacity_add, 10 a level fully staffed, with a company flagship's or a treaty port's on top) times its
# multipliers (state_trade_capacity_mult: the trade laws, principles and companies; a country's lines flow down to its
# states). Read live: two modifier reads, no walk. Whether the read carries what the building's throughput adds is
# unverified (docs/testing/journal-map-modes-playtest.md, check 28).
te_jmap_trade_capacity = {
	value = modifier:state_trade_capacity_add
	multiply = {
		value = 1
		add = modifier:state_trade_capacity_mult
		min = 0
	}
	min = 0
}

# The owner's trade with the world market, the total value of what it buys and sells there (total_trade_value, the
# figure vanilla's Free Trade entry je_liberalism_2 counts): the hover's second line, for scale. Script has no state's
# own share of it. 0 for a state with no owner.
te_jmap_trade_value = {
	value = 0
	if = {
		limit = { exists = owner }
		owner = {
			add = total_trade_value
		}
	}
	min = 0
}

# Investment (46, 2026-11-11): what the state's owners pay into their country's Investment Pool each week, estimated.
# The engine takes a share of the dividends a state's owners are paid (state_<pop type>_investment_pool_contribution_
# add, raised by _efficiency_mult), and script reads neither a pop's dividends nor a building's shares, so the estimate
# is the weekly profit of the buildings the owners sit in, which is what the levels they own pay them less the
# building's own wages (the Company Headquarters map's read, te_jmap_hq_selfcheck), times the share their leading
# owners reinvest: capitalists' for Financial Districts and company headquarters, aristocrats' for Manor Houses. Left
# out: levels their own workers own and the government's reinvested dividends, whose shares script can't split, the
# clergy's and bureaucrats' Manor Houses (counted at the aristocrats' share) and a regional headquarters, which pays
# its company's home. A building at a loss pays nothing.
#
# The map reads state variables a store writes (te_jmap_store_invest, te_journal_map_effects.txt) at a pick of the map
# and each month while it is the pick, as Company Headquarters reads its store's: the paint walks no buildings.

# The walks, state scope: the store's input only.
te_jmap_invest_walk_cap_profit = {
	value = 0
	every_scope_building = {
		limit = {
			OR = {
				is_building_group = bg_financial_districts
				is_building_group = bg_company_headquarter
			}
			weekly_profit > 0
		}
		add = weekly_profit
	}
}

te_jmap_invest_walk_ari_profit = {
	value = 0
	every_scope_building = {
		limit = {
			is_building_group = bg_manor_houses
			weekly_profit > 0
		}
		add = weekly_profit
	}
}
