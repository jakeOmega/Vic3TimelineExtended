# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The Map Room's selector. Each
# map's color scale, its factor 0-1, is generated: te_journal_map_scales_generated.txt, from the table in
# scripts/generators/gen_journal_map_scales.py (one shape per family, so a change to a shape changes every map using it).

# ---------------------------------------------------------------------------
# The selector: which of the Map Room's maps the player picked
# ---------------------------------------------------------------------------
# 1 turmoil, 3 life expectancy, 4 fertility, 5 median age, 6 devastation, 7 natural change, 8 death rate, 9 birth
# rate, 10 infant mortality, 11 leading cause of death, 12 sex balance, 13 net migration, 14 aged 65 and over, 15 urban
# share, 16 Gini coefficient, 17 Dynastic Wealth; the systems' (each also on its own entry): 18 emissions, 19 nuclear
# weapons, 20 covert networks, 21 the United Nations, 22 political models, 23 ocean basins, 24 grandeur, 25 colonies;
# the Map Room's own again: 26 tourism's Base Appeal, 27 Base Comfort (Local Climate before any technology), 28 Current
# Comfort (after the owner's), 29 GDP per capita, 30 Migration Crowding; 0 none. 2 was literacy, retired because
# vanilla's own literacy map mode shows it (a save holding 2 shows the unpainted map and "Pick a map" until the next
# pick); 6 was the scope probe until the fourth session. A global variable (te_jmap_set_selector), since a state's map_color reaches neither the
# entry nor a country (the scope probe painted blue, 2026-11-05): one player's pick is every player's in multiplayer.
# Root-free, so the Map Room's state roots and the pick buttons' country roots read the same number.
te_jmap_selector = {
	value = 0
	if = {
		limit = { has_global_variable = te_jmap_sel }
		value = global_var:te_jmap_sel
	}
}

# ---------------------------------------------------------------------------
# The systems' figures (state scope: the state being painted or hovered)
# ---------------------------------------------------------------------------
# The state's share of the world's greenhouse gas emissions last year, 0-1 (negative for a state that takes more out of
# the air than it puts in): its yearly snapshot (gw_state_emis, gw_snapshot_country_emissions_effect) over the world's
# annual total (gw_g_global_emis, re-based each January), both in internal units. A share, not an amount (owner,
# 2026-11-06), so a color means the same thing in 1850 and 2000, whatever the world emits.
te_jmap_gw_share = {
	value = 0
	if = {
		limit = {
			has_variable = gw_state_emis
			te_jmap_gw_world > 0
		}
		value = var:gw_state_emis
		divide = te_jmap_gw_world
	}
	min = -1
	max = 1
}

# The world's emissions last year (gw_g_global_emis), 0 before the first January; any scope.
te_jmap_gw_world = {
	value = 0
	if = {
		limit = { has_global_variable = gw_g_global_emis }
		value = global_var:gw_g_global_emis
	}
}

# The state's emissions last year in internal units (gw_state_emis), 0 before its first snapshot.
te_jmap_gw_state_emis = {
	value = 0
	if = {
		limit = { has_variable = gw_state_emis }
		value = var:gw_state_emis
	}
}

# The state's emissions last year in display units (the panel's K/M figures: internal x 1000, gw_disp_own_emis).
te_jmap_gw_disp_emis = {
	value = 0
	if = {
		limit = { has_variable = gw_state_emis }
		value = var:gw_state_emis
		multiply = 1000
	}
}

# The viewer's network in the state's owner, 0-100 (covert warfare's iw_nets: one container per target, on the
# operator, iw_net_strength; nd_loose_network_tier reads it the same way). 0 with no network, no viewer, or the viewer
# itself. root is the state: the map's, or the hover's (MakeScope).
te_jmap_net_strength = {
	value = 0
	if = {
		limit = {
			exists = owner
			global_var:te_jmap_viewer ?= { has_variable_list = iw_nets }
		}
		global_var:te_jmap_viewer = {
			every_in_list = {
				variable = iw_nets
				limit = {
					var:iw_target ?= root.owner
					has_variable = iw_net_strength
				}
				add = var:iw_net_strength
			}
		}
	}
	min = 0
	max = 100
}

# ---------------------------------------------------------------------------
# Hover figures (map_tooltip's loc, state scope)
# ---------------------------------------------------------------------------
# Base Comfort's figure as what it does to migration attraction, signed as te_climate_effect_now is (negative a cost,
# positive a mild winter's draw): the penalty before relief that te_update_local_climate_modifier stored.
te_jmap_comfort_base_effect = {
	value = 0
	if = {
		limit = { has_variable = te_climate_base_penalty }
		value = var:te_climate_base_penalty
		multiply = -1
	}
}
