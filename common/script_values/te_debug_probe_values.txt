# TEMPORARY PROBE, read by common/on_actions/te_debug_probe_on_actions.txt:
# delete both files together once a launch has logged the TE_PROBE_LOC lines.
# The scope's te_probe_loc_value (the probe sets 7 on the country and 9 on
# its capital), or -1 without it.
te_probe_loc_value_sv = {
	value = -1
	if = {
		limit = { has_variable = te_probe_loc_value }
		value = var:te_probe_loc_value
	}
}

# Line 11: the saved capital's te_probe_loc_value (9), read through a saved
# scope rather than the scope the value is evaluated in; -1 without it.
te_probe_loc_saved_scope_sv = {
	value = -1
	if = {
		limit = { exists = scope:te_probe_capital }
		scope:te_probe_capital = {
			if = {
				limit = { has_variable = te_probe_loc_value }
				add = 1
				add = var:te_probe_loc_value
			}
		}
	}
}
