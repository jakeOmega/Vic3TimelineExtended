# ---- Binding Ceasefire: the order ------------------------------------------------

# How long a carried ceasefire order stands, in months.
un_cf_order_length = { value = 24 }

# War support the aggressor loses each month, in every war it fights on the
# attacking side against a represented member, at enforcement 1. Times E. The
# war then ends through the engine's own capitulation: there is no forced
# white peace (the nuclear crisis's concession works the same way,
# nd_crisis_concede_war).
un_cf_drain_base = { value = 2.5 }

# The vetoed form, a call for a ceasefire: one hit, at enforcement 1. Times E.
un_cf_call_hit_base = { value = 10 }

# Months an ordered aggressor may go on fighting before its standing pays for
# it (the time a peace takes to negotiate). It pays once.
un_cf_grace_months = { value = 3 }

# Standing an aggressor loses for fighting on past the grace (code 39), and
# the lighter loss when it voted for its own censure.
un_cf_fight_on_loss = { value = 5 }
un_cf_fight_on_accepted_loss = { value = 2 }

# The aggression record an aggressor takes each month it fights on under the
# order (the dossier decays every record by a five-year half-life).
un_cf_dossier_monthly = { value = 1 }

# The UN's ledgers. Institutional entries, booked as written.
# Each month the war goes on under the order: peace and order (code 30).
un_cf_order_debit = { value = -0.25 }
# The debit accrues on the aggressor and is booked every this many months of
# fighting (and when the order ends), so a 24-month order books at most five
# log entries rather than twenty-four.
un_cf_order_debit_batch_months = { value = 6 }
# What the accrued months come to. Scope: the aggressor
un_cf_order_debit_accrued = {
	value = 0
	if = {
		limit = { has_variable = un_cf_debit_months }
		add = var:un_cf_debit_months
		multiply = un_cf_order_debit
	}
}
# The war ends inside the order, and the aggressor gained nothing from it:
# delivery (code 31).
un_cf_delivery_credit = { value = 1.5 }
# The order runs out with the war still on, or the war ends with the aggressor
# holding gains from the victim: credibility (code 32).
un_cf_defied_debit = { value = -1 }

# ---- Binding Ceasefire: derived ----------------------------------------------------

# Signed, for add_war_war_support. un_enforcement reads only globals, so these
# are sound in any scope (the monthly update has none).
un_cf_war_support_drain = {
	value = un_cf_drain_base
	multiply = un_enforcement
	multiply = -1
}

un_cf_call_war_support_hit = {
	value = un_cf_call_hit_base
	multiply = un_enforcement
	multiply = -1
}

# The states the scoped country holds. An order stores its victim's at
# imposition (un_cf_victim_states); a victim that holds fewer once the war is
# over lost ground under the order (un_order_ceasefire_aggressor_gained).
# Scope: country
un_order_state_count_value = {
	value = 0
	every_scope_state = {
		add = 1
	}
}

# Display: the months an order against us has left. Scope: country
un_disp_cf_months_left = {
	value = 0
	if = {
		limit = { has_variable = un_cf_order_months }
		add = var:un_cf_order_months
	}
}

# Display: the war support a month of fighting under an order costs, as a
# positive number. Globals only.
un_disp_cf_drain = {
	value = un_cf_drain_base
	multiply = un_enforcement
}

# ---- Supervised Self-Determination Referendum ------------------------------------

# A subject qualifies at this liberty desire or more.
un_ref_liberty_desire_floor = { value = 50 }

# The chance a held referendum ends in independence: the subject's liberty
# desire (to the nearest 5 below), held between these.
un_ref_chance_min = { value = 20 }
un_ref_chance_max = { value = 90 }

# Liberty desire the subject gains or loses (add_liberty_desire).
# Held, and the subject votes to stay:
un_ref_stay_liberty_desire = { value = -20 }
# Refused by the overlord:
un_ref_refused_liberty_desire = { value = 15 }

# Standing for the overlord: holding the vote (code 40, nominal: diminishing
# returns apply), refusing it (code 41).
un_ref_held_gain = { value = 4 }
un_ref_refused_loss = { value = 5 }

# The UN's ledgers, institutional.
# Held: delivery (code 33).
un_ref_delivery_credit = { value = 1 }
# Refused: credibility (code 32).
un_ref_refused_debit = { value = -1 }

# ---- Supervised Self-Determination Referendum: derived ---------------------------

# The subject's liberty desire, to the nearest 5 below. liberty_desire is a
# compare trigger with no script-value form, so the value is read as a ladder.
# Scope: the subject
un_ref_liberty_desire_band = {
	value = 0
	if = { limit = { liberty_desire >= 100 } value = 100 }
	else_if = { limit = { liberty_desire >= 95 } value = 95 }
	else_if = { limit = { liberty_desire >= 90 } value = 90 }
	else_if = { limit = { liberty_desire >= 85 } value = 85 }
	else_if = { limit = { liberty_desire >= 80 } value = 80 }
	else_if = { limit = { liberty_desire >= 75 } value = 75 }
	else_if = { limit = { liberty_desire >= 70 } value = 70 }
	else_if = { limit = { liberty_desire >= 65 } value = 65 }
	else_if = { limit = { liberty_desire >= 60 } value = 60 }
	else_if = { limit = { liberty_desire >= 55 } value = 55 }
	else_if = { limit = { liberty_desire >= 50 } value = 50 }
	else_if = { limit = { liberty_desire >= 45 } value = 45 }
	else_if = { limit = { liberty_desire >= 40 } value = 40 }
	else_if = { limit = { liberty_desire >= 35 } value = 35 }
	else_if = { limit = { liberty_desire >= 30 } value = 30 }
	else_if = { limit = { liberty_desire >= 25 } value = 25 }
	else_if = { limit = { liberty_desire >= 20 } value = 20 }
	else_if = { limit = { liberty_desire >= 15 } value = 15 }
	else_if = { limit = { liberty_desire >= 10 } value = 10 }
	else_if = { limit = { liberty_desire >= 5 } value = 5 }
}

# The chance, in percent, that a held referendum ends in independence.
# Scope: the subject
un_ref_independence_chance = {
	value = un_ref_liberty_desire_band
	min = un_ref_chance_min
	max = un_ref_chance_max
}

# The roll un_referendum.1 makes: the chance stored on the overlord when the
# event opened (un_ref_chance), so the option shows the odds it rolls. A named
# script value is the proven form for `random = { chance = ... }`
# (covert_op_roll_chance); `chance = var:` is not.
# Scope: the overlord
un_ref_roll_chance = {
	value = 0
	if = {
		limit = { has_variable = un_ref_chance }
		add = var:un_ref_chance
	}
	min = 0
	max = 100
}

# ---- The lean (voter scope; scope:un_resolution, scope:un_vote_target,
#      scope:un_vote_proposer) ---------------------------------------------------
# The generic terms (the target, the target's allies and bloc, the grounds,
# the proposer, glass houses) already apply through the classification:
# a ceasefire is punitive, a referendum accuses its overlord.

# Binding Ceasefire. Members at war with the aggressor lean hard for; those
# fighting beside it without an alliance (an ally is the target term's)
# against; the humanitarian law for, total war against.
un_lean_interests_ceasefire = {
	value = 0
	if = {
		limit = {
			exists = scope:un_vote_target
			NOT = { scope:un_vote_target ?= this }
		}
		if = {
			limit = { has_war_with = scope:un_vote_target }
			add = 50
		}
		else_if = {
			limit = {
				NOT = { has_treaty_alliance_with = { TARGET = scope:un_vote_target } }
				scope:un_vote_target = { any_cobelligerent_in_war = { this = ROOT } }
			}
			add = -30
		}
	}
	if = { limit = { has_law = law_type:law_humanitarian_regulations } add = 15 }
	if = { limit = { has_law = law_type:law_total_war } add = -20 }
}

# Supervised Self-Determination Referendum. Overlords fear the precedent, as
# on the decolonisation declaration; subjects, the named subject most of all,
# and council republics are for it. (`any_direct_subject = { is_subject = yes }`
# is "holds a direct subject": every country it iterates is one.)
un_lean_interests_referendum = {
	value = 0
	if = {
		limit = { any_direct_subject = { is_subject = yes } }
		add = -30
	}
	else = {
		add = 10
	}
	if = {
		limit = { is_subject = yes }
		add = 20
	}
	if = {
		limit = { scope:un_resolution ?= { var:un_res_referendum_subject ?= root } }
		add = 30
	}
	if = { limit = { has_law = law_type:law_council_republic } add = 20 }
	if = { limit = { has_variable = former_overlord } add = 10 }
}

# ---- The AI's buttons (proposer scope) ---------------------------------------------

# Binding Ceasefire: a member fights for one most of all when it is the
# victim, and readily against a rival; a strong Assembly is worth asking.
un_ceasefire_ai_chance = {
	value = 0
	if = {
		limit = {
			save_temporary_scope_as = un_order_proposer
			any_country = {
				has_war_with = scope:un_order_proposer
				un_ceasefire_target_eligible = yes
			}
		}
		add = 40
	}
	if = {
		limit = {
			save_temporary_scope_as = un_order_proposer
			any_country = {
				has_diplomatic_pact = { who = scope:un_order_proposer type = rivalry }
				un_ceasefire_target_eligible = yes
			}
		}
		add = 20
	}
	if = { limit = { has_law = law_type:law_humanitarian_regulations } add = 10 }
	if = { limit = { country_rank >= rank_value:great_power } add = 10 }
	if = {
		limit = {
			has_global_variable = un_authority
			global_var:un_authority >= 50
		}
		add = 10
	}
	if = { limit = { has_law = law_type:law_total_war } add = -20 }
	if = { limit = { has_law = law_type:law_isolationism } add = -30 }
	min = 0
}

# Supervised Self-Determination Referendum: a subject asking for its own
# vote, a council republic, a former colony or a country without subjects;
# an overlord seldom calls one on another.
un_referendum_ai_chance = {
	value = 0
	if = {
		limit = {
			save_temporary_scope_as = un_order_proposer
			un_referendum_subject_eligible = yes
		}
		add = 30
	}
	if = {
		limit = {
			save_temporary_scope_as = un_order_proposer
			any_country = {
				un_referendum_subject_eligible = yes
				overlord = { has_diplomatic_pact = { who = scope:un_order_proposer type = rivalry } }
			}
		}
		add = 15
	}
	if = { limit = { has_law = law_type:law_council_republic } add = 20 }
	if = { limit = { has_variable = former_overlord } add = 10 }
	if = {
		limit = { any_direct_subject = { is_subject = yes } }
		add = -40
	}
	else = {
		add = 10
	}
	if = {
		limit = {
			has_global_variable = un_authority
			global_var:un_authority >= 50
		}
		add = 10
	}
	min = 0
}
