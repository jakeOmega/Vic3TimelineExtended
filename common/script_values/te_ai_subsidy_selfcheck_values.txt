# The should_remove self-check (#933, te_ai_subsidy_should_remove_selfcheck):
# AI-owned power plants and railways, and the share of each that is subsidized,
# in countries running one of the strategies te_ai_subsidy_clean_strategy names
# ("clean") and in the other AI countries ("other"). Railways stay must_have
# everywhere, so their share is the control: it shows is_subsidized reads.
# Any scope.
te_ai_pp_count_clean = {
	value = 0
	every_country = {
		limit = {
			is_ai = yes
			te_ai_subsidy_clean_strategy = yes
		}
		every_scope_state = {
			every_scope_building = {
				limit = { is_building_type = building_power_plant }
				add = 1
			}
		}
	}
}

te_ai_pp_subsidized_share_clean = {
	value = 0
	every_country = {
		limit = {
			is_ai = yes
			te_ai_subsidy_clean_strategy = yes
		}
		every_scope_state = {
			every_scope_building = {
				limit = {
					is_building_type = building_power_plant
					is_subsidized = yes
				}
				add = 1
			}
		}
	}
	divide = { value = te_ai_pp_count_clean min = 1 }
}
