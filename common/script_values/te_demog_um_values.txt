# Country scope: the share of last year's eligible deaths that became uploads (0 with none).
te_demog_um_hosted_share = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_um_dem
			var:te_dg_um_dem > 0
		}
		value = var:te_dg_um_new
		divide = { value = var:te_dg_um_dem min = 0.00001 }
		max = 1
	}
}

# Country or state scope: last year's uploads per 1,000 people (bodies), for the deaths' hovers: the country's stored
# figure (te_dg_um_rate, te_demog_um_country), a state's from its last step's uploads.
te_demog_um_upload_rate = {
	value = 0
	if = {
		limit = { has_variable = te_dg_um_rate }
		value = var:te_dg_um_rate
	}
	else_if = {
		limit = {
			has_variable = te_dg_um_new
			has_variable = te_dg_people
		}
		value = var:te_dg_um_new
		divide = { value = var:te_dg_people min = 1 }
		multiply = 1000
	}
}

# Country or state scope: every body's death per 1,000 people last year, the uploads' too: deaths per 1,000 (true deaths,
# te_dg_cdr) plus the uploads. The Added on Top figures compare it with the census's own rate (te_dg_cdr_model), which
# counts the uploads, as the engine does.
te_demog_cdr_all = {
	value = var:te_dg_cdr
	add = te_demog_um_upload_rate
}
