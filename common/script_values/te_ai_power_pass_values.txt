# The AI electrification pass (common/scripted_effects/te_ai_power_pass_effects.txt; #1877 trap 2): the numbers it
# sizes a state's plant by, and the reasons for them. docs/systems/mod_systems/ai.md § AI Electrification Pass.

# Electricity a level of vanilla's coal and oil plant (building_power_plant) makes at full staff, at the best method
# the country has unlocked. The plant's group is most_productive (extra_pm_groups.txt), so the AI runs the method
# with the highest output: pm_early_power_plant 25 (Electrical Generation), pm_coal-fired_plant 50 (Steam Turbine),
# pm_modern_coal-fired_plant 90 (Modern Urban Planning). The oil methods make more (80 from Oil Turbine, 140 modern)
# but need oil, which a state may not get; reading coal's figure sizes a little large where oil is used, never
# short. Country scope; te_ai_power_pass_effect stores it as the country variable te_ai_power_level_output, which
# the state values below read through owner.
te_ai_power_level_output_value = {
	value = 25
	if = {
		limit = { has_technology_researched = steam_turbine }
		value = 50
	}
	if = {
		limit = { has_technology_researched = modern_urban_planning }
		value = 90
	}
}

# State scope. Levels of the state's urban centers.
te_ai_power_uc_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_urban_center }
		add = level
	}
}

# State scope. Levels of vanilla's coal and oil plant in the state, built (what the pass follows its batches by).
te_ai_power_plant_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_power_plant }
		add = level
	}
}

# Electricity a level of urban center takes on electric street lights at full staff: pm_electric_streetlights'
# goods_input_electricity_add (vanilla 3, workforce_scaled).
TE_AI_POWER_LIGHT_INPUT = 3

# State scope. What the state's urban centers would buy if they took electric street lights, while they are on no
# lighting or gas lights: their levels times TE_AI_POWER_LIGHT_INPUT. Zero once they are on electric lights or
# better, since what they buy is then in the state's consumption.
te_ai_power_light_need = {
	value = 0
	if = {
		limit = { te_ai_power_lights_unswitched = yes }
		add = te_ai_power_uc_levels
		multiply = TE_AI_POWER_LIGHT_INPUT
	}
}

# State scope. What the state's coal and oil plants make at full staff at the country's best method. A plant hires
# only as it finds buyers, so what it makes lags this (the 1925 test run of 2026-11-28: the 20 largest AI
# economies' plants 12-91% staffed, two years after the pass began); sizing and switching by what is made would
# queue again for plants that only lack workers, and never switch a state whose new plant waits for its buyers.
# 0 for a country the pass has not reached (no Electrical Generation yet).
te_ai_power_plant_capacity = {
	value = 0
	if = {
		limit = { owner = { has_variable = te_ai_power_level_output } }
		value = te_ai_power_plant_levels
		multiply = owner.var:te_ai_power_level_output
	}
}

# State scope. What the state can count on: its plants' capacity, or what is made in it if more (another kind of
# plant, a Grid Transfer Station's delivery).
te_ai_power_supply = {
	value = te_grid_electricity_production
	if = {
		limit = { te_ai_power_plant_capacity > te_grid_electricity_production }
		value = te_ai_power_plant_capacity
	}
}

# State scope. How much Electricity the state is short of: what it buys, plus what its urban centers would buy on
# electric lights, less what it can count on (te_ai_power_supply). Not what it buys alone: a state with no power
# can't run the methods that use it, so it would read as needing none (scripting_best_practices.md § "AI and
# Cost-Only Buildings"). Last week's figures; never below zero.
te_ai_power_short = {
	value = te_grid_electricity_consumption
	add = te_ai_power_light_need
	subtract = te_ai_power_supply
	min = 0
}

# The most levels the pass queues in a state at once. A level is 400 points (construction_cost_medium), so a batch
# of 10 is 4,000. A large city's lights take more than that (a 200-level urban center takes 600 Electricity, 24
# levels of the early plant); a state that needs more gets its next batch once this one is decided
# (te_ai_power_follow). The 1925 test run's batches of at most 5 left the 20 largest buying twice what they made.
TE_AI_POWER_MAX_BATCH = 10

# The most batches of the pass's a country waits on at once. The government queue's length in weeks
# (construction_queue_government_duration, "very roughly approximated" in the engine's own words) never held the
# pass back in the 1925 test run: France had 28 batches waiting at the end, the USA 29, and a third of the levels
# decided were lost to the 30-month limit. Three keeps the queue for the country's other government building.
TE_AI_POWER_MAX_PENDING = 3

# Country scope. The batches of the pass's the country waits on.
te_ai_power_pending_count = {
	value = 0
	every_scope_state = {
		limit = { has_variable = te_ai_power_pending }
		add = 1
	}
}

# State scope. Levels of the coal and oil plant that would cover te_ai_power_short at the country's best method,
# rounded to the nearest, at most TE_AI_POWER_MAX_BATCH: a state short of under half a level's output gets none.
te_ai_power_levels_wanted = {
	value = te_ai_power_short
	divide = owner.var:te_ai_power_level_output
	round = yes
	max = TE_AI_POWER_MAX_BATCH
}

# A state's batch is given up on after this many monthly follows with some of it unbuilt: 36 months. A 10-level batch
# is 4,000 points, about 80 weeks of a large 1920s government's own construction, and it may wait behind two more of
# the pass's and up to a year of other building. Its unbuilt levels count as lost for the self-check, so a BROKEN can
# mean slow as well as dropped: the self-check's line prints the cohort's built and lost.
TE_AI_POWER_FOLLOW_MONTHS = 36

# State scope, for a state with a batch followed (te_ai_power_follow): the plant levels that would mean the batch is
# built, the levels of it built so far (never below 0 or above the batch: a level taken down there, or an investor's,
# moves the count), and the levels still missing.
te_ai_power_follow_target = {
	value = var:te_ai_power_base
	add = var:te_ai_power_want
}
te_ai_power_follow_got = {
	value = te_ai_power_plant_levels
	subtract = var:te_ai_power_base
	min = 0
	max = var:te_ai_power_want
}
te_ai_power_follow_missing = {
	value = var:te_ai_power_want
	subtract = te_ai_power_follow_got
}

# The share of the light need the state's spare must cover before its urban centers switch to electric lights:
# four-fifths. The 1925 test run switched at half (or just after a batch, whatever its size) and the AI put 47% of
# the switches back on gas or no lighting, short of Electricity; the plants' capacity, not what they make, is read,
# so a new plant with no buyers yet still counts.
TE_AI_POWER_SWITCH_SHARE = 0.8

# State scope. What the state can count on and does not already buy.
te_ai_power_spare = {
	value = te_ai_power_supply
	subtract = te_grid_electricity_consumption
}

# State scope. The share of the light need the spare must cover (TE_AI_POWER_SWITCH_SHARE times the need).
te_ai_power_switch_threshold = {
	value = te_ai_power_light_need
	multiply = TE_AI_POWER_SWITCH_SHARE
}

# The self-checks' thresholds (te_ai_power_pass_on_actions.txt). The build check follows a closed cohort, the first
# TE_AI_POWER_COHORT batches queued in a game, and reads once every one is decided: the first batches decided are the
# quick builds (losses show only at the follow's limit), so a verdict on them reads OK whatever the loss rate (the
# 1925 test run: OK on the first 32 decided, none lost; 35% of levels lost by the end).
TE_AI_POWER_COHORT = 40
# Read once this many of the cohort are decided: a batch whose state passes to a country that does not follow it (a
# player, or one without Electrical Generation) would otherwise hold the verdict for good.
TE_AI_POWER_COHORT_READ = 36
TE_AI_POWER_CHECK_LOOKS = 20

# Of the cohort's levels, the share lost; and of the switched states looked at a month on, the share the AI had put
# back.
te_ai_power_lost_share = {
	value = te_ai_power_cohort_lost_g
	divide = {
		value = te_ai_power_cohort_built_g
		add = te_ai_power_cohort_lost_g
		min = 1
	}
}
te_ai_power_revert_share = {
	value = te_ai_power_first_reverts_g
	divide = {
		value = te_ai_power_looked_g
		min = 1
	}
