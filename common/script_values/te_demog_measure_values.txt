# ---- The money upkeep (§3.4; scripting_best_practices.md, "Expense Scaling with GDP") ---------------------------
# Each is a _cost modifier's multiplier on country_expenses_add = 1, so it is the weekly cost: gdp is a year's, so a
# share of GDP a year is divided by 52. Refreshed when the measure goes on and by the country's yearly census
# (te_demog_ms_yearly). The census's figures exist under Full; without them the factor is the reference.
# Child Allowance: 0.5% of GDP a year x (the 0-14 share / 0.3).
te_demog_ms_cost_child_allowance = {
	value = gdp
	multiply = 0.005
	if = {
		limit = { has_variable = te_dg_young_share }
		multiply = var:te_dg_young_share
		divide = 0.3
	}
	divide = 52
	min = 0
}
# Family Housing: 0.25% of GDP a year x the urban share (0.5 without the census). Step 3b's fit (§3.4: 0.4%): the full
# pro-natalist package's upkeep within 1-3% of a rich country's revenue at the top of the gate's band.
te_demog_ms_cost_family_housing = {
	value = gdp
	multiply = 0.0025
	if = {
		limit = { has_variable = te_dg_urban_share }
		multiply = var:te_dg_urban_share
	}
	else = {
		multiply = 0.5
	}
	divide = 52
	min = 0
}
# Parental Leave: 0.3% of GDP a year x the share of women at work.
te_demog_ms_cost_parental_leave = {
	value = gdp
	multiply = 0.003
	multiply = te_demog_ms_women_at_work
	divide = 52
	min = 0
}
# Free Contraception: 0.1% of GDP a year.
te_demog_ms_cost_free_contraception = {
	value = gdp
	multiply = 0.001
	divide = 52
	min = 0
}

# The share of women at work: phase 2 step 1's women's rate (te_demog_women_rate: the women's-work family's sum, with
# the lower strata's floor) in the capital, every game, over the most a rate reaches (te_demog_k_rate_max): the rates
# are in the game's units since 2026-10-20, and this keeps the upkeep where it was. 0 without a capital.
te_demog_ms_women_at_work = {
	value = 0
	if = {
		limit = { exists = capital }
		value = capital.te_demog_women_rate
		divide = { value = te_demog_k_rate_max min = 0.01 }
	}
}

# ---- The AI's figures (§3.6) -----------------------------------------------------------------------------------------
# The population's change over the last year, as a share: each state's people at its yearly walk against the walk
# before (te_dg_walk_pop_prev, kept every game), summed over the states that hold both, so a state gained or lost
# moves neither sum. 0 without a year to compare.
te_demog_ms_pop_now_sum = {
	value = 0
	every_scope_state = {
		limit = {
			has_variable = te_dg_walk_pop
			has_variable = te_dg_walk_pop_prev
		}
		add = var:te_dg_walk_pop
	}
}
te_demog_ms_pop_prev_sum = {
	value = 0
	every_scope_state = {
		limit = {
			has_variable = te_dg_walk_pop
			has_variable = te_dg_walk_pop_prev
		}
		add = var:te_dg_walk_pop_prev
	}
}
te_demog_ms_pop_trend = {
	value = te_demog_ms_pop_now_sum
	subtract = te_demog_ms_pop_prev_sum
	divide = { value = te_demog_ms_pop_prev_sum min = 1 }
}

# Crowding for the AI: the share of the country's people in states Migration Crowding presses at its ceiling or past it
# (migration_crowding_mult_applied 4.5 or more: ten times the reference density, migration pull -45%). Any crowding at
# all is too common to mean anything (73% of countries in an 1837 save, 88% in 1902), and 4.5 none before 1902; projected
# from 1902's densities at twice and three times its people, half a country's people at the ceiling is about 4% and 20%
# of countries: China, East India, Japan, Egypt, Java (the review of 3a; numbers in the PR).
te_demog_ms_crowded_share = {
	value = 0
	every_scope_state = {
		limit = {
			has_variable = migration_crowding_mult_applied
			var:migration_crowding_mult_applied >= 4.5
		}
		add = state_population
	}
	divide = { value = te_demog_ms_population min = 1 }
}
te_demog_ms_population = {
	value = 0
	every_scope_state = {
		add = state_population
	}
}

# The census's share aged 20 to 29 (§4.3's youth bulge), from its five-year bands: the 20-29s over the 0-14s, times
# the 0-14 share. 0 where the census doesn't run. Every band exists once the country has a census year.
te_demog_ms_youth_share = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_census_year
			has_variable = te_dg_young_share
		}
		value = var:te_dg_cbf4
		add = var:te_dg_cbm4
		add = var:te_dg_cbf5
		add = var:te_dg_cbm5
		multiply = var:te_dg_young_share
		divide = {
			value = var:te_dg_cbf0
			add = var:te_dg_cbm0
			add = var:te_dg_cbf1
			add = var:te_dg_cbm1
			add = var:te_dg_cbf2
			add = var:te_dg_cbm2
			min = 1
		}
	}
}
