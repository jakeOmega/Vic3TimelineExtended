# Legislated tax code: display values for the code's schema-level tokens
# (docs/systems/tax_code_schema.md, "Display values"). The per-instrument and
# per-good ones, and the panels' per-record, per-slot, last-change and history
# views, are generated into te_tax_generated_values.txt.
#
# Every te_tax_view_* reads only variables it has just tested with has_variable
# and returns the schema sentinel when one is absent (0 for a counter, an index
# or a flag, -1 for a month). They write nothing and iterate nothing, so a GUI
# can read them on every frame.

# The day a bill introduced today could pass (game_date counts days in 365-day
# years from day 0, 1 January: te_tax_support_values.txt, "Debate clock"), and
# that day's month as a te_history_month_index value.
te_tax_dr_pass_day = {
	value = game_date
	add = te_tax_view_dr_debate_days
}

te_tax_dr_pass_year = {
	value = te_tax_dr_pass_day
	divide = 365
	floor = yes
}

te_tax_dr_pass_day_of_year = {
	value = te_tax_dr_pass_year
	multiply = -365
	add = te_tax_dr_pass_day
}

te_tax_dr_pass_month = {
	value = te_tax_dr_pass_year
	multiply = 12
	# The first day of each month after January, counted from 1 January.
	if = { limit = { te_tax_dr_pass_day_of_year >= 334 } add = 11 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 304 } add = 10 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 273 } add = 9 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 243 } add = 8 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 212 } add = 7 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 181 } add = 6 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 151 } add = 5 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 120 } add = 4 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 90 } add = 3 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 59 } add = 2 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 31 } add = 1 }
}

# State scope, for the workbench's and the review's state rows: 1 if the
# owner's open draft names the state for regional relief. Where the draft
# leaves regional relief alone, existing law in its commencement month
# (te_tax_view_dr_relief_base_state). Inside owner = { }, PREV is the state.
te_tax_view_dr_relief_state = {
	value = 0
	if = {
		limit = {
			owner ?= {
				has_variable = te_tax_dr_on
				var:te_tax_dr_on = 1
				has_variable = te_tax_dr_regrel
				var:te_tax_dr_regrel >= 0
			}
		}
		if = {
			limit = {
				owner = {
					has_variable_list = te_tax_dr_relief_states
					is_target_in_variable_list = { name = te_tax_dr_relief_states target = PREV }
				}
			}
			value = 1
		}
	}
	else = {
		value = te_tax_view_dr_relief_base_state
	}
}

# State scope: 1 if existing law in the owner's draft's commencement month
# names the state for regional relief: the list of an awaiting package that
# commences before it and restates the states (the last in commencement order,
# te_tax_p<slot>_relief_states), else the owner's enacted list. 0 without an
# open draft. Inside owner = { }, PREV is the state.
te_tax_view_dr_relief_base_state = {
	value = 0
	if = {
		limit = {
			owner ?= {
				has_variable = te_tax_dr_on
				var:te_tax_dr_on = 1
				has_variable = te_tax_dr_due
			}
		}
		if = {
			limit = { owner = { te_tax_relief_base_from_a = { DUE = te_tax_dr_due } } }
			if = {
				limit = {
					owner = {
						has_variable_list = te_tax_pa_relief_states
						is_target_in_variable_list = { name = te_tax_pa_relief_states target = PREV }
					}
				}
				value = 1
			}
		}
		else_if = {
			limit = { owner = { te_tax_relief_base_from_b = { DUE = te_tax_dr_due } } }
			if = {
				limit = {
					owner = {
						has_variable_list = te_tax_pb_relief_states
						is_target_in_variable_list = { name = te_tax_pb_relief_states target = PREV }
					}
				}
				value = 1
			}
		}
		else_if = {
			limit = {
				owner = {
					has_variable_list = te_tax_en_relief_states
					is_target_in_variable_list = { name = te_tax_en_relief_states target = PREV }
				}
			}
			value = 1
		}
	}
}

# 1 while the snapshot is current: taken this calendar month (the processor
# retakes it on the 1st, so an older one means a month was missed), under the
# code in force (every writer of the code is followed by a snapshot, a migration
# by te_tax.4's the next day; the test catches the gap and an inherited one),
# and, while a bill is under debate, for its current revision. 0 otherwise: the
# panels mark the estimate out of date.
te_tax_view_snap_current = {
	value = 0
	if = {
		limit = {
			has_variable = te_tax_snap_month
			var:te_tax_snap_month = te_history_month_index
			has_variable = te_tax_snap_code_version
			has_variable = te_tax_code_version
			var:te_tax_snap_code_version = var:te_tax_code_version
			has_variable = te_tax_snap_bill_rev
			OR = {
				NOT = { AND = { has_variable = te_tax_bl_on var:te_tax_bl_on = 1 } }
				AND = { has_variable = te_tax_bl_rev var:te_tax_snap_bill_rev = var:te_tax_bl_rev }
			}
		}
		value = 1
	}
}

# State scope, the review's regional relief rows: the change in the state's taxes
# the draft makes against existing law in its month, as a multiple of what the
# state collects now (State.GetTaxationRevenue in the GUI). A band n is
# state_tax_collection_mult -0.25 n (te_tax_relief_region_<n>), so a state keeps
# (4 - n) / 4 of its taxes; other collection modifiers are taken as unchanged.
# 0 without a draft that changes regional relief.
te_tax_view_est_regrel_state = {
	value = 0
	if = {
		limit = {
			owner ?= {
				has_variable = te_tax_dr_on
				var:te_tax_dr_on = 1
				has_variable = te_tax_dr_due
				has_variable = te_tax_dr_regrel
				var:te_tax_dr_regrel >= 0
			}
		}
		value = te_tax_est_regrel_keep_bill
		subtract = te_tax_est_regrel_keep_law
		divide = te_tax_est_regrel_keep_now
	}
}

# State scope: 4 - the band the state has now (its mark, te_tax_relief_state,
# with the owner's enacted band); at least 1, so never a zero divisor.
te_tax_est_regrel_keep_now = {
	value = 4
	if = {
		limit = {
			has_variable = te_tax_relief_state
			var:te_tax_relief_state = 1
			owner ?= { has_variable = te_tax_en_regrel }
		}
		subtract = owner.var:te_tax_en_regrel
	}
	min = 1
