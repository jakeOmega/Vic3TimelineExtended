# Legislated tax code: display values for the code's schema-level tokens
# (docs/systems/tax_code_schema.md, "Display values"). The per-instrument and
# per-good ones, and the panels' per-record, per-slot, last-change and history
# views, are generated into te_tax_generated_values.txt.
#
# Every te_tax_view_* reads only variables it has just tested with has_variable
# and returns the schema sentinel when one is absent (0 for a counter, an index
# or a flag, -1 for a month). They write nothing and iterate nothing, so a GUI
# can read them on every frame.

# Percentage points per relief band: band 1 = 25%, band 2 = 50%
# (te_tax_relief_* in common/static_modifiers/te_tax_modifiers.txt).
te_tax_relief_pct_step = 25

te_tax_view_schema = {
	value = 0
	if = { limit = { has_variable = te_tax_schema } value = var:te_tax_schema }
}

te_tax_view_migrated = {
	value = 0
	if = { limit = { has_variable = te_tax_migrated } value = var:te_tax_migrated }
}

te_tax_view_code_version = {
	value = 0
	if = { limit = { has_variable = te_tax_code_version } value = var:te_tax_code_version }
}

te_tax_view_last_month = {
	value = -1
	if = { limit = { has_variable = te_tax_last_month } value = var:te_tax_last_month }
}

te_tax_view_next_month = {
	value = -1
	if = { limit = { has_variable = te_tax_next_month } value = var:te_tax_next_month }
}

# Agricultural wage relief band (0 none, 1, 2) and its percentage.

		limit = { has_variable = te_tax_en_agrel }
		value = var:te_tax_en_agrel
		multiply = te_tax_relief_pct_step
	}
}

# State scope: 1 if the state is named in the enacted regional relief.

# The next scheduled change's year and 1-based month (-1 when none is scheduled).
te_tax_view_next_month_y = {
	value = -1
	if = {
		limit = { has_variable = te_tax_next_month var:te_tax_next_month >= 0 }
		value = var:te_tax_next_month
		divide = 12
		floor = yes
	}
}

te_tax_view_next_month_mo = {
	value = -1
	if = {
		limit = { has_variable = te_tax_next_month var:te_tax_next_month >= 0 }
		value = var:te_tax_next_month
		divide = 12
		floor = yes
		multiply = -12
		add = var:te_tax_next_month
		add = 1
	}
}

# The bill under debate's revision (0 without a bill): read only while the
# bill is open, as its payload is removed when it closes.
te_tax_view_bl_rev = {
	value = 0
	if = {
		limit = {
			has_variable = te_tax_bl_on
			var:te_tax_bl_on = 1
			has_variable = te_tax_bl_rev
		}
		value = var:te_tax_bl_rev
	}
}

# Days of debate the bill still needs (te_tax_support_values.txt), 0 without a
# bill. The wrapper repeats the bill's guard so the panels read one value.
te_tax_view_debate_days_left = {
	value = 0
	if = {
		limit = {
			has_variable = te_tax_bl_on
			var:te_tax_bl_on = 1
			has_variable = te_tax_bl_day
		}
		value = te_tax_debate_days_left
	}
}

# ---------------------------------------------------------------------------
# The workbench, review and passage panels (plan Task 8). The per-instrument,
# per-good, per-slot and interest-group views are generated
# (te_tax_generated_values.txt). A draft view reads the draft only while
# te_tax_dr_on is 1, a bill view the bill only while te_tax_bl_on is 1: the
# support snapshot (te_tax_sup_*, te_tax_com_*) outlives the bill it was for.
# ---------------------------------------------------------------------------

# 1 if the open draft would make a minor bill (te_tax_draft_is_minor), else 0.
te_tax_view_dr_minor = {
	value = 0
	if = {
		limit = { has_variable = te_tax_dr_on var:te_tax_dr_on = 1 has_variable = te_tax_dr_due }
		if = {
			limit = { te_tax_draft_is_minor = yes }
			value = 1
		}
	}
}

# Days of debate a bill made from the open draft would need.
te_tax_view_dr_debate_days = {
	value = 0
	if = {
		limit = { has_variable = te_tax_dr_on var:te_tax_dr_on = 1 has_variable = te_tax_dr_due }
		value = te_tax_debate_days_major
		if = {
			limit = { te_tax_draft_is_minor = yes }
			value = te_tax_debate_days_minor
		}
	}
}

# The day a bill introduced today could pass (game_date counts days in 365-day
# years from day 0, 1 January: te_tax_support_values.txt, "Debate clock"), and
# that day's month as a te_history_month_index value.
te_tax_dr_pass_day = {
	value = game_date
	add = te_tax_view_dr_debate_days
}

te_tax_dr_pass_year = {
	value = te_tax_dr_pass_day
	divide = 365
	floor = yes
}

te_tax_dr_pass_day_of_year = {
	value = te_tax_dr_pass_year
	multiply = -365
	add = te_tax_dr_pass_day
}

te_tax_dr_pass_month = {
	value = te_tax_dr_pass_year
	multiply = 12
	# The first day of each month after January, counted from 1 January.
	if = { limit = { te_tax_dr_pass_day_of_year >= 334 } add = 11 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 304 } add = 10 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 273 } add = 9 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 243 } add = 8 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 212 } add = 7 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 181 } add = 6 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 151 } add = 5 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 120 } add = 4 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 90 } add = 3 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 59 } add = 2 }
	else_if = { limit = { te_tax_dr_pass_day_of_year >= 31 } add = 1 }
}

# The earliest commencement month the open draft can have and still pass in
# time if it is introduced today: the month after the one its debate ends in
# (a bill passes only before its commencement month, te_tax_can_pass). -1
# without a draft.
te_tax_view_dr_earliest_month = {
	value = -1
	if = {
		limit = { has_variable = te_tax_dr_on var:te_tax_dr_on = 1 has_variable = te_tax_dr_due }
		value = te_tax_dr_pass_month
		add = 1
	}
}

te_tax_view_dr_earliest_month_y = {
	value = -1
	if = {
		limit = { has_variable = te_tax_dr_on var:te_tax_dr_on = 1 te_tax_view_dr_earliest_month >= 0 }
		value = te_tax_view_dr_earliest_month
		divide = 12
		floor = yes
	}
}

te_tax_view_dr_earliest_month_mo = {
	value = -1
	if = {
		limit = { has_variable = te_tax_dr_on var:te_tax_dr_on = 1 te_tax_view_dr_earliest_month >= 0 }
		value = te_tax_view_dr_earliest_month
		divide = 12
		floor = yes
		multiply = -12
		add = te_tax_view_dr_earliest_month

	}
}

# The passage panel, while a bill is under debate: the clout share committed to
# its current revision (te_tax_committed_share), that share plus the
# persuadable groups' (te_tax_open_clout), the share passage needs, and the
# country's legitimacy.
te_tax_view_committed_share = {
	value = 0
	if = {
		limit = { has_variable = te_tax_bl_on var:te_tax_bl_on = 1 has_variable = te_tax_bl_due }
		value = te_tax_committed_share
	}
}

te_tax_view_open_share = {
	value = 0
	if = {
		limit = { has_variable = te_tax_bl_on var:te_tax_bl_on = 1 has_variable = te_tax_bl_due }
		if = {
			limit = { te_tax_eligible_clout > 0 }
			value = te_tax_committed_clout
			add = te_tax_open_clout
			divide = te_tax_eligible_clout
		}
	}
}

te_tax_view_passage_share = {
	value = 0
	if = {
		limit = { has_variable = te_tax_bl_on var:te_tax_bl_on = 1 }
		value = te_tax_passage_share
	}
}

te_tax_view_legitimacy = {
	value = 0
	if = {
		limit = { has_variable = te_tax_bl_on var:te_tax_bl_on = 1 }
		value = legitimacy
	}
}
