# Legislated tax code: display values for the code's schema-level tokens
# (docs/systems/tax_code_schema.md, "Display values"). The per-instrument and
# per-good ones, and the panels' per-record, per-slot, last-change and history
# views, are generated into te_tax_generated_values.txt.
#
# Every te_tax_view_* reads only variables it has just tested with has_variable
# and returns the schema sentinel when one is absent (0 for a counter, an index
# or a flag, -1 for a month). They write nothing and iterate nothing, so a GUI
# can read them on every frame.

# Percentage points per relief band: band 1 = 25%, band 2 = 50%
# (te_tax_relief_* in common/static_modifiers/te_tax_modifiers.txt).
te_tax_relief_pct_step = 25

te_tax_view_schema = {
	value = 0
	if = { limit = { has_variable = te_tax_schema } value = var:te_tax_schema }
}

te_tax_view_migrated = {
	value = 0
	if = { limit = { has_variable = te_tax_migrated } value = var:te_tax_migrated }
}

te_tax_view_code_version = {
	value = 0
	if = { limit = { has_variable = te_tax_code_version } value = var:te_tax_code_version }
}

te_tax_view_last_month = {
	value = -1
	if = { limit = { has_variable = te_tax_last_month } value = var:te_tax_last_month }
}

te_tax_view_next_month = {
	value = -1
	if = { limit = { has_variable = te_tax_next_month } value = var:te_tax_next_month }
}

# Agricultural wage relief band (0 none, 1, 2) and its percentage.
te_tax_view_agrel = {
	value = 0
	if = { limit = { has_variable = te_tax_en_agrel } value = var:te_tax_en_agrel }
}

te_tax_view_agrel_pct = {
	value = 0
	if = {
		limit = { has_variable = te_tax_en_agrel }
		value = var:te_tax_en_agrel
		multiply = te_tax_relief_pct_step
	}
}

# Regional relief band (0 none, 1, 2) and its percentage.
te_tax_view_regrel = {
	value = 0
	if = { limit = { has_variable = te_tax_en_regrel } value = var:te_tax_en_regrel }
}

te_tax_view_regrel_pct = {
	value = 0
	if = {
		limit = { has_variable = te_tax_en_regrel }
		value = var:te_tax_en_regrel
		multiply = te_tax_relief_pct_step
	}
}

# State scope: 1 if the state is named in the enacted regional relief.
te_tax_view_relief_state = {
	value = 0
	if = { limit = { has_variable = te_tax_relief_state } value = var:te_tax_relief_state }
}

# The next scheduled change's year and 1-based month (-1 when none is scheduled).
te_tax_view_next_month_y = {
	value = -1
	if = {
		limit = { has_variable = te_tax_next_month var:te_tax_next_month >= 0 }
		value = var:te_tax_next_month
		divide = 12
		floor = yes
	}
}

te_tax_view_next_month_mo = {
	value = -1
	if = {
		limit = { has_variable = te_tax_next_month var:te_tax_next_month >= 0 }
		value = var:te_tax_next_month
		divide = 12
		floor = yes
		multiply = -12
		add = var:te_tax_next_month
		add = 1
	}
}

# The bill under debate's revision (0 without a bill): read only while the
# bill is open, as its payload is removed when it closes.
te_tax_view_bl_rev = {
	value = 0
	if = {
		limit = {
			has_variable = te_tax_bl_on
			var:te_tax_bl_on = 1
			has_variable = te_tax_bl_rev
		}
		value = var:te_tax_bl_rev
	}
}

# Days of debate the bill still needs (te_tax_support_values.txt), 0 without a
# bill. The wrapper repeats the bill's guard so the panels read one value.
te_tax_view_debate_days_left = {
	value = 0
	if = {
		limit = {
			has_variable = te_tax_bl_on
			var:te_tax_bl_on = 1
			has_variable = te_tax_bl_day
		}
		value = te_tax_debate_days_left
	}
}
