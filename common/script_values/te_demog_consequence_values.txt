# ============================================================================
# DEMOGRAPHICS — conscription, the youth bulge and sex balance (phase 2 step 4)
# docs/superpowers/archive/specs/2026-10-12-demographics-phase2-consequences-design.md §4; demographics_model's
# sex_births, sex_women_add, sex_fewer_men, sex_more_men, conscription_term and youth_bulge are the twins.
# The constants and band sums are generated (te_demog_k_*, te_demog_youth_share, te_demog_state_men and the country's;
# gen_demographics.consequence_values); the effects are in te_demog_consequence_effects.txt.

# ---- Sex balance (§4.4): the census's own terms, under Full and Display only ---------------------------------------

# State scope: children per woman's sex-balance term (te_demog_tfr reads it): the balance the step read (te_dg_sxb,
# stored only where men are short) over te_demog_k_sex_fewer_from, at most 1 and at least te_demog_k_sex_births_floor.
# 1 at a seed, which stores no balance.
te_demog_tfr_sex = {
	value = 1
	if = {
		limit = { has_variable = te_dg_sxb }
		value = var:te_dg_sxb
		divide = { value = te_demog_k_sex_fewer_from min = 1 }
		min = te_demog_k_sex_births_floor
		max = 1
	}
}

# State scope: what the women's rate gains where men are short (te_demog_women_rate adds it): up to
# te_demog_k_sex_women_rate, full te_demog_k_sex_fewer_span men below the threshold. 0 at a seed.
te_demog_sex_women_add = {
	value = 0
	if = {
		limit = { has_variable = te_dg_sxb }
		value = te_demog_k_sex_fewer_from
		subtract = var:te_dg_sxb
		divide = { value = te_demog_k_sex_fewer_span min = 1 }
		min = 0
		max = 1
		multiply = te_demog_k_sex_women_rate
	}
}

# ---- Sex balance: the state modifiers' multipliers (Full only), and the panel's ------------------------------------
# Both multipliers are 0 while te_demog_k_sex_effects is 0 (held back: demographics_params.SEX_BALANCE_EFFECTS), so
# te_demog_consequences_apply adds neither modifier; the panel's hover then says the effects aren't applied yet.

# State or country scope: how far men are short, 0 to 1, from the census's latest balance (te_dg_sex_balance, men per
# 100 women aged 20-59): 0 at te_demog_k_sex_fewer_from or more, 1 a span below. te_demog_census_fewer_men's multiplier.
te_demog_sex_fewer_men = {
	value = 0
	if = {
		limit = {
			te_demog_k_sex_effects > 0
			has_variable = te_dg_sex_balance
		}
		value = te_demog_k_sex_fewer_from
		subtract = var:te_dg_sex_balance
		divide = { value = te_demog_k_sex_fewer_span min = 1 }
		min = 0
		max = 1
	}
}

# State or country scope: how far men are in excess, 0 to 1: 0 at te_demog_k_sex_more_from or fewer, 1 a span above.
# te_demog_census_more_men's multiplier.
te_demog_sex_more_men = {
	value = 0
	if = {
		limit = {
			te_demog_k_sex_effects > 0
			has_variable = te_dg_sex_balance
		}
		value = var:te_dg_sex_balance
		subtract = te_demog_k_sex_more_from
		divide = { value = te_demog_k_sex_more_span min = 1 }
		min = 0
		max = 1
	}
}

# The modifiers' strengths (the 2026-10-22 effects design): each ramp without its ceiling (the reach), then
# (x + min(x, 1)) / 2, at most 2 (demographics_params.EFFECT_CAP): 70 or 150 men give 1.5, 60 or 170 give 2. The
# ramps above stay 0-1 for the women's rate and the panel's model figures (demographics_model.sex_fewer_men_strength,
# sex_more_men_strength).
te_demog_sex_fewer_men_reach = {
	value = 0
	if = {
		limit = {
			te_demog_k_sex_effects > 0
			has_variable = te_dg_sex_balance
		}
		value = te_demog_k_sex_fewer_from
		subtract = var:te_dg_sex_balance
		divide = { value = te_demog_k_sex_fewer_span min = 1 }
		min = 0
	}
}
te_demog_sex_fewer_men_strength = {
	value = te_demog_sex_fewer_men_reach
	add = {
		value = te_demog_sex_fewer_men_reach
		max = 1
	}
	divide = 2
	max = 2
}
te_demog_sex_more_men_reach = {
	value = 0
	if = {
		limit = {
			te_demog_k_sex_effects > 0
			has_variable = te_dg_sex_balance
		}
		value = var:te_dg_sex_balance
		subtract = te_demog_k_sex_more_from
		divide = { value = te_demog_k_sex_more_span min = 1 }
		min = 0
	}
}
te_demog_sex_more_men_strength = {
	value = te_demog_sex_more_men_reach
	add = {
		value = te_demog_sex_more_men_reach
		max = 1
	}
	divide = 2
	max = 2
}

# State or country scope: the children-per-woman term the latest balance would give (the panel's hover; the census reads
# te_demog_tfr_sex, the balance its step began with).
te_demog_sex_births_shown = {
	value = 1
	if = {
		limit = { has_variable = te_dg_sex_balance }
		value = var:te_dg_sex_balance
		divide = { value = te_demog_k_sex_fewer_from min = 1 }
		min = te_demog_k_sex_births_floor
		max = 1
	}
}

# State or country scope: what the women's rate gains at the latest balance (the panel's hover).
te_demog_sex_women_shown = {
	value = te_demog_sex_fewer_men
	multiply = te_demog_k_sex_women_rate
}

# ---- Conscription (§4.1) ----------------------------------------------------------------------------------------------

# State scope: the pool (men of calendar age 20 or more and biological age 39 or less, te_dg_fam) over the census's
# people, 0 before a census under phase 2 step 4.
te_demog_fam_share = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_fam
			has_variable = te_dg_people
		}
		value = var:te_dg_fam
		divide = { value = var:te_dg_people min = 1 }
	}
}

# State scope: the pool over the census's men (the panel's figure).
te_demog_fam_men_share = {
	value = 0
	if = {
		limit = { has_variable = te_dg_fam }
		value = var:te_dg_fam
		divide = { value = te_demog_state_men min = 1 }
	}
}

# State scope: T, the conscription term (demographics_model.conscription_term): the pool's share against the 1836
# world's, less 1, down to te_demog_k_conscription_min (-1: no conscripts) and with no ceiling (owner, 2026-10-16). Under
# Full the census applies it (te_dg_mc); under Display only the panel shows what it would be.
te_demog_conscription_term = {
	value = te_demog_fam_share
	divide = { value = te_demog_k_conscription_ref min = 0.001 }
	subtract = 1
	min = te_demog_k_conscription_min
}

# Country scope: the same for the country's census states together (te_dg_fam_share, its country census).
te_demog_country_conscription = {
	value = 0
	if = {
		limit = { has_variable = te_dg_fam_share }
		value = var:te_dg_fam_share
		divide = { value = te_demog_k_conscription_ref min = 0.001 }
		subtract = 1
		min = te_demog_k_conscription_min
	}
}

# State scope: |te_dg_mc|, the conscription term's size (0 with none stored).
te_demog_conscription_size = {
	value = 0
	if = {
		limit = { has_variable = te_dg_mc }
		value = var:te_dg_mc
		if = {
			limit = { var:te_dg_mc < 0 }
			multiply = -1
		}
	}
}

# State scope: the fixed-sign pair's multipliers (te_demog_census_conscription_up/_down): |te_dg_mc| on its side, else 0.
# Both are 0 while te_demog_k_conscription_effects is 0 (held back: demographics_params.CONSCRIPTION_EFFECTS) and for a
# term smaller than te_demog_k_conscription_cut (demographics_model.conscription_applied).
te_demog_conscription_up = {
	value = 0
	if = {
		limit = {
			te_demog_k_conscription_effects > 0
			has_variable = te_dg_mc
			var:te_dg_mc > 0
			te_demog_conscription_size >= te_demog_k_conscription_cut
		}
		value = var:te_dg_mc
	}
}
te_demog_conscription_down = {
	value = 0
	if = {
		limit = {
			te_demog_k_conscription_effects > 0
			has_variable = te_dg_mc
			var:te_dg_mc < 0
			te_demog_conscription_size >= te_demog_k_conscription_cut
		}
		value = var:te_dg_mc
		multiply = -1
	}
}

# ---- The youth bulge (§4.3) -------------------------------------------------------------------------------------------

# State scope: the unemployment the census stored at the state's last census walk (te_dg_unemp, te_demog_work_walk_store):
# te_state_unemployment_share then, the share of the workforce with no work that every mod reader takes
# (te_unemployment_values.txt), 0-1, 0 before one. The youth bulge, the event wave, the pension AI and the panel read
# this. The harness counts the same share from a save's pops (demographics_save_inputs.state_unemployment); the
# conversion equals it where the engine's figure is per person, as the self-checks unemployment_rate_read and
# unemployment_share_read test. Not the engine's state_unemployment_rate bare: it read 0.048 where the pops' share was
# 0.195 (2026-10-17), the idle workforce over all the state's people. And a scripted GUI reads it as none, so the
# figure is stored from an effect and read from the variable.
te_demog_unemployment = {
	value = 0
	if = {
		limit = { has_variable = te_dg_unemp }
		value = var:te_dg_unemp
	}
}

# State scope: the same, for the panel's youth bulge (the Youth row's gate and figures read this name).
te_demog_unemployment_census = {
	value = te_demog_unemployment
}

# State scope: the youth bulge's multiplier now, y = the share factor x the unemployment factor, at most 2
# (demographics_model.youth_bulge; the 2026-10-22 effects design): each factor is its ramp, (the share aged 20-29 -
# te_demog_k_youth_from) / the span and unemployment / te_demog_k_youth_unemployment_full, extended past 1 at half the
# slope (te_demog_youth_share_part, te_demog_youth_unemployment_part, below).
te_demog_youth_bulge_now = {
	value = te_demog_youth_share_part
	multiply = te_demog_youth_unemployment_part
	max = 2
}

# State scope: the youth-bulge modifier's multiplier (te_demog_consequences_apply): te_dg_yb, 0 while
# te_demog_k_youth_effects is 0 (held back: demographics_params.YOUTH_BULGE_EFFECTS) or below te_demog_k_youth_cut
# (demographics_model.youth_applied).
te_demog_youth_applied = {
	value = 0
	if = {
		limit = {
			te_demog_k_youth_effects > 0
			has_variable = te_dg_yb
			var:te_dg_yb >= te_demog_k_youth_cut
		}
		value = var:te_dg_yb
	}
}

# State scope: the youth bulge the panel shows: the one in force under Full (te_dg_yb, set at the yearly census), else
# what it would be at the census's own reads (te_demog_youth_bulge_now's formula, with the unemployment it stored). No
# live engine read: the Youth row's gate (te_demog_state_youth_sgui) evaluates it from a scripted GUI.
te_demog_youth_bulge_shown = {
	value = te_demog_youth_share_part
	multiply = te_demog_youth_unemployment_part
	max = 2
	if = {
		limit = { has_variable = te_dg_yb }
		value = var:te_dg_yb
	}
}

# State scope: the youth bulge's two factors, from the last census: each ramp's reach, then (x + min(x, 1)) / 2.
te_demog_youth_share_reach = {
	value = te_demog_youth_share
	subtract = te_demog_k_youth_from
	divide = { value = te_demog_k_youth_span min = 0.001 }
	min = 0
}
te_demog_youth_share_part = {
	value = te_demog_youth_share_reach
	add = {
		value = te_demog_youth_share_reach
		max = 1
	}
	divide = 2
}
te_demog_youth_unemployment_reach = {
	value = te_demog_unemployment_census
	divide = { value = te_demog_k_youth_unemployment_full min = 0.001 }
	min = 0
}
te_demog_youth_unemployment_part = {
	value = te_demog_youth_unemployment_reach
	add = {
		value = te_demog_youth_unemployment_reach
		max = 1
	}
	divide = 2
}
