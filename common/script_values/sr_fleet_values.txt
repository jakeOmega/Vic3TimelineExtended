# Country scope. Fleet strength Space Command builds a month: the methods'

# (design § 5.1: nothing is built in an occupied state). Staffing is in the
# number already.

	value = 0
	every_scope_state = {
		limit = { sr_state_held_by_owner = yes }
		add = modifier:state_sr_ship_construction_add
	}
}

# of fleet strength in service, at full (design § 5).

# applied to each Space Command building (sr_fl_upkeep_refresh).

# The hostile strength each posture engages (design § 7.1): stationed, every
# hostile fleet; standing off, the hostiles stationed or standing off; closing
# in, the hostiles stationed or closing in. sr_cb_hs_calc: the hostiles
# stationed, which draw a closing-in fleet's defence fire and bar a seizure.
sr_cb_e1_value = {
	value = 0
	every_in_global_list = {
		variable = sr_fleet_powers
		limit = { has_variable = sr_cb_present has_war_with = prev }
		add = sr_cb_all_value
	}
}

# The hostiles' fire per point engaged that reaches each of our postures:
# stationed, all three of theirs; standing off, their stationed and standing

# what it engages, so the losses sum to the fire).

		limit = { has_variable = sr_cb_present has_war_with = prev }

# ---- The Orbital Battlestation (design § 7.5), country scope ------------------------------------
# Full strength: 25 warship strength a level, in the states the owner controls.
sr_battlestation_full = {
	value = 0
	every_scope_state = {
		limit = { sr_state_held_by_owner = yes }
		every_scope_building = {
			limit = { is_building_type = building_orbital_battlestation }
			add = { value = level multiply = sr_battlestation_per_level }
		}
	}

		limit = { has_variable = sr_battlestation_strength }
		value = var:sr_battlestation_strength

	}
