# ---- The fleets' clock (2026-11-25 design § 1.2) ------------------------------------
# Fleets move on four ticks a month: the 1st (the global monthly step), the
# 8th, 15th and 22nd (space_race_events.120-.122, queued from the 1st). A
# tick's index is te_history_month_index x 4 + its quarter (0 to 3), kept in
# global_var:sr_fl_tick by each tick (sr_fl_tick_set). The month counter is
# the floor: read on a 1st before its tick runs, or if a month's ticks stopped,
# the index is still the month's quarter 0. Every departure, landing and
# weeks-left reads this. Any scope.
# docs/superpowers/specs/2026-11-25-space-fleet-clock-and-battles-design.md
sr_fl_tick_now = {
	value = te_history_month_index
	multiply = 4
	if = {
		limit = {
			has_global_variable = sr_fl_tick
			global_var:sr_fl_tick > { value = te_history_month_index multiply = 4 }
		}
		value = global_var:sr_fl_tick
	}
}
# A tick index is over 88,000 from 1836 (1836 x 12 x 4) and a month index
# under 30,000 to the game's end: below this, a fleet's _arrive or _depart is
# an old save's month (sr_fl_ticks_migrate_once).
sr_fl_tick_floor = { value = 50000 }

# billed on Space Command and they fire at half without it, so yards building
# for a country with none would give it warships with no bill; and a Space

# Command's. sr_sc_rate_buildings, Space Command's alone, is the self-check's.

	value = 0
	every_scope_state = {
		limit = { sr_state_held_by_owner = yes }
		add = modifier:state_sr_ship_construction_add

# Country scope. Its Space Command buildings of a level or more, wherever they
# stand. A nation runs one (owner, 2026-10-31): with two or more (a second
# comes with an annexed or formed country's land) the month's sweep keeps one
# and dismantles the rest (sr_sc_one_per_nation, before the upkeep reaches
# them), so the fleets' upkeep has one building to sit on.
sr_sc_buildings = {
	value = 0
	every_scope_state = {
		every_scope_building = {
			limit = {
				is_building_type = building_space_command
				level >= 1
			}
			add = 1
		}
	}
}

# State scope, the sweep's order: the Space Command kept is the one with the
# most levels, and on a tie the capital's, where the Space Program stands
# (sr_space_program_site; the market capital's until 2026-11-03, when the
# program moved to the capital). Only a tie-break: a rule keeping the capital's
# alone would demolish a player's Space Command each time the capital moved.
sr_sc_keeper_score = {
	value = 0
	if = {
		limit = { has_building = building_space_command }
		value = b:building_space_command.level
		multiply = 10
	}
	if = {
		limit = { owner.capital ?= this }
		add = 1
	}
}

# ---- Combat (design § 7) ----------------------------------------------------------------
# Lanchester's square law: a month, each side loses k x the other's fire. Since
# the 2026-11-25 design § 2 it is the siege's (the Defences' fire on fleets
# closing in) and, at a quarter a tick, the ground-to-orbit batteries'.
sr_fleet_k = { value = 0.1 }
# The battle (§ 2.4): the square law in steps of sr_bt_k, repeated within a
# tick until no fighting pair is left or sr_bt_max_steps is reached (a fight
# of fleets with almost no fire, which goes on at the next tick). An even
# fight breaks in about 14 steps. A fleet breaks, and drops out, once it has
# lost sr_bt_break_share of the strength it brought to the battle (§ 3's first
# guesses: the owner, 2026-11-25, asked for those to be set and tuned later).
sr_bt_k = { value = 0.05 }
sr_bt_max_steps = { value = 60 }
sr_bt_break_share = { value = 0.5 }
# The batteries fire once a tick (§ 2.5): a quarter of a month's, so a month
# of ticks deals what a month's round dealt.
sr_battery_tick_share = { value = 0.25 }
# The AI engages where its win test holds (§ 2.7): our strength there x our
# fire against the same product for every power at war with us there, times
# this margin. Its defence sends a fleet on the same test.
sr_fl_ai_engage_margin = { value = 1 }
# Fleet quality: +25% for each of Directed Energy Weapons, Orbital Weapon
# Platforms and Antimatter Production. Country scope.
sr_fleet_quality_step = { value = 0.25 }
sr_fleet_quality = {
	value = 1
	if = { limit = { has_technology_researched = directed_energy_weapons } add = sr_fleet_quality_step }
	if = { limit = { has_technology_researched = orbital_weapon_platforms } add = sr_fleet_quality_step }
	if = { limit = { has_technology_researched = antimatter_production } add = sr_fleet_quality_step }
}

# Orbital fire support (design § 15): the margin by which our fleets at Earth
# outmatch the hostile fleets there, min(1, (ours - theirs) / max(theirs,
# sr_orbital_superiority_floor)), ours at their readiness (sr_fl_at_earth_ready:
# half without a Space Command); 0 with no fleet of ours there. Every fleet at
# Earth counts, whatever its posture: sr_fl_at_<loc> is posture-blind, so one
# closing in on a lunar world counts as at Earth (lunar colonization § 8.5). The
# multiplier of sr_orbital_superiority, so a full margin gives +10% offense
# and +5% defense.
sr_orbital_superiority_floor = { value = 10 }
sr_orbital_superiority_margin = {
	value = 0
	if = {
		limit = { has_variable = sr_fl_at_earth }
		value = sr_fl_at_earth_ready
		subtract = sr_loc_ai_enemy_fleets_earth
		divide = {
			value = sr_loc_ai_enemy_fleets_earth
			min = sr_orbital_superiority_floor
		}
		max = 1
		min = 0
	}
}
