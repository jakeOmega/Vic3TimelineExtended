# Country scope. Fleet strength Space Command builds a month: the methods'
# workforce-scaled state modifier, summed over the states the country controls
# (design § 5.1: nothing is built in an occupied state). Staffing is in the
# number already. Plus the colonies' Orbital Yards (colony management design
# § 7); sr_sc_rate_buildings, Space Command's alone, is the self-check's.
sr_sc_rate_buildings = {
	value = 0
	every_scope_state = {
		limit = { sr_state_held_by_owner = yes }
		add = modifier:state_sr_ship_construction_add
	}
}
sr_sc_rate = {
	value = sr_sc_rate_buildings
	add = {
		value = sr_mod_units_yard
		multiply = sr_mod_yard_ships
	}
}

# The upkeep: Space Command's maintenance line rises +900% for every this many
# points of fleet strength in service, at full (Space Race UI pass design § 10).
sr_fleet_upkeep_points = { value = 10 }

# Country scope. The upkeep's units, set on the country before they are
# applied (sr_fl_upkeep_refresh). The ground reserve (6d) counts: hulls kept on
# the ground cost the same as in orbit, so holding them back saves nothing.
sr_fleet_upkeep_units_value = {
	value = sr_fl_strength_full_total
	add = sr_reserve_total
	divide = sr_fleet_upkeep_points
}

# Country scope. Its Space Command buildings of a level or more, wherever they
# stand (a second comes with an annexed or formed space power): each has the
# maintenance line, and the units are split between them.
sr_sc_buildings = {
	value = 0
	every_scope_state = {
		every_scope_building = {
			limit = {
				is_building_type = building_space_command
				level >= 1
			}
			add = 1
		}
	}
}

# Country scope. The multiplier on each of them: the fleets pay the upkeep
# once, however many buildings carry it.
sr_fleet_upkeep_per_building_value = {
	value = sr_fleet_upkeep_units_value
	if = {
		limit = { sr_sc_buildings > 1 }
		divide = sr_sc_buildings
	}
}

# Orbital fire support (design § 15): the margin by which our fleets at Earth
# outmatch the hostile fleets there, min(1, (ours - theirs) / max(theirs,
# sr_orbital_superiority_floor)); 0 with no fleet of ours there. Every fleet at
# Earth counts, whatever its posture: sr_fl_at_<loc> is posture-blind, so one
# closing in on a lunar world counts as at Earth (lunar colonization § 8.5). The
# multiplier of sr_orbital_superiority, so a full margin gives +10% offense
# and +5% defense.
sr_orbital_superiority_floor = { value = 10 }
sr_orbital_superiority_margin = {
	value = 0
	if = {
		limit = { has_variable = sr_fl_at_earth }
		value = var:sr_fl_at_earth
		subtract = sr_loc_ai_enemy_fleets_earth
		divide = {
			value = sr_loc_ai_enemy_fleets_earth
			min = sr_orbital_superiority_floor
		}
		max = 1
		min = 0
	}
}
