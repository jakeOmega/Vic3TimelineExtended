# Momentum band (banking_dash_momentum_band): 1 Collapsing · 2 Falling ·
# 3 Steady · 4 Rising · 5 Surging.
banking_disp_momentum_band_code = {
	value = 0
	if = {
		limit = { has_variable = finance_cycle_momentum }
		if = {
			limit = { var:finance_cycle_momentum >= 3 }
			value = 5
		}
		else_if = {
			limit = { var:finance_cycle_momentum >= 1 }
			value = 4
		}
		else_if = {
			limit = { var:finance_cycle_momentum > -1 }
			value = 3
		}
		else_if = {
			limit = { var:finance_cycle_momentum > -3 }
			value = 2
		}
		else = {
			value = 1
		}
	}
}

# Price band (te_mon_inflation_band_name): 1 Deflation · 2 Stable prices ·
# 3 Elevated · 4 High · 5 Very high · 6 Hyperinflation · 7 Foreign money
# (dollarised) · 8 Set by plan (a command economy). The two states with no
# band come first, as in the word's own tests.
banking_disp_price_band_code = {
	value = 0
	if = {
		limit = { has_variable = te_inflation_band }
		if = {
			limit = { te_mon_is_dollarised = yes }
			value = 7
		}
		else_if = {
			limit = { te_mon_has_inflation = no }
			value = 8
		}
		else_if = {
			limit = { var:te_inflation_band <= 1 }
			value = 1
		}
		else_if = {
			limit = { var:te_inflation_band <= 2 }
			value = 2
		}
		else_if = {
			limit = { var:te_inflation_band <= 3 }
			value = 3
		}
		else_if = {
			limit = { var:te_inflation_band <= 4 }
			value = 4
		}
		else_if = {
			limit = { var:te_inflation_band <= 5 }
			value = 5
		}
		else = {
			value = 6
		}
	}
}
