# Construction-market subsystem — values for direct construction, the model
# games run with free_market_construction_rule disabled
# (te_free_market_construction_off). See docs/systems/mod_systems.md,
# "Free Market Construction off".
#
# The rule's production methods do the construction itself (the Construction
# Site is the buildable Construction Sector, te_construction_market_pms.txt), so
# script only restores the law-set private share and keeps the AI's sites
# (te_direct_construction_keep_ai_sites).

# Private share of construction (0-100) under direct construction: the
# economic-system law's vanilla allocation. These are the values
# common/laws/construction_system_law_injections.txt cancels so that, under
# the market, only te_construction_market_private_share_modifier sets the
# share; this puts them back. Keep the two files in step.
te_direct_construction_law_private_share = {
	value = 0
	if = {
		limit = { has_law = law_type:law_traditionalism }
		value = 25
	}
	else_if = {
		limit = { has_law = law_type:law_interventionism }
		value = 50
	}
	else_if = {
		limit = { has_law = law_type:law_agrarianism }
		value = 50
	}
	else_if = {
		limit = { has_law = law_type:law_industry_banned }
		value = 50
	}
	else_if = {
		limit = { has_law = law_type:law_extraction_economy }
		value = 50
	}
	else_if = {
		limit = { has_law = law_type:law_laissez_faire }
		value = 75
	}
	else_if = {
		limit = { has_law = law_type:law_cooperative_ownership }
		value = 35
	}
	else_if = {
		limit = { has_law = law_type:law_command_economy }
		value = 10
	}
}

# Display: construction points per week from every source (the Construction
# Sites' levels, te_direct_construction_base_modifier, Engineering & Logistics
# barracks, ...).
te_direct_construction_display_capacity = {
	value = modifier:country_construction_add
}

# State scope. Level of this state's Construction Site, 0 without one.
te_direct_construction_state_site_level = {
	value = 0
	if = {
		limit = {
			has_building = te_construction_market_site
		}
		value = b:te_construction_market_site.level
	}
}
