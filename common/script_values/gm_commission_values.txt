# ============================================================================
# GRAND MONUMENTS — opportunities: script values
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-12-grand-monument-opportunities-design.md
# (v2's commissions, reworked; the script keeps their gm_com_* names).
# Country scope throughout. Every var: read is guarded: the widget renders these
# for countries that have never had an opportunity.

# ---- Progress (spec §2) ---------------------------------------------------------
# The levels of the monuments that count toward the open opportunity today
# (gm_state_com_counts, gm_mon_com_counts): fitting, of its dedication, in its
# state if it has one. Levels, never age (owner, 2026-10-16): only building,
# a new monument or one beside another, completes an opportunity. The name
# stays for saves (the gm_com_* records).
gm_com_fit_grandeur = {
	value = 0
	if = {
		limit = { has_variable = gm_com_dedication }
		every_scope_state = {
			limit = { gm_state_com_counts = yes }
			add = gm_state_primary_start
		}
		if = {
			limit = { has_variable_list = gm_monuments_held }
			every_in_list = {
				variable = gm_monuments_held
				limit = { gm_mon_com_counts = yes }
				add = gm_mon_start
			}
		}
	}
}

# Grandeur added since acceptance, floored at 0: raising an old monument counts
# as much as founding a new one, and a counted monument that is contested later
# drops out without driving the count below zero.
gm_com_progress = {
	value = 0
	if = {
		limit = {
			has_variable = gm_com_open
			has_variable = gm_com_baseline
		}
		value = gm_com_fit_grandeur
		subtract = var:gm_com_baseline
		min = 0
	}
}

# ---- Time (spec §2) ----------------------------------------------------------------
# The months an opportunity runs, fixed on seizing: 60 for a great power, 120
# for a major power (owner, 2026-10-05: about half the construction for the same
# five levels), 180 for any other (owner, 2026-10-12). A rank won or lost later
# does not change a running one.
gm_com_time_months = {
	value = 180
	if = {
		limit = { country_rank >= rank_value:great_power }
		value = 60
	}
	else_if = {
		limit = { country_rank >= rank_value:major_power }
		value = 120
	}
	# A historical opportunity of ten levels has twice as long (signatures spec
	# §6.1): the rank's months times its target over five.
	if = {
		limit = {
			has_variable = gm_com_target
			var:gm_com_target > 5
		}
		multiply = var:gm_com_target
		divide = 5
	}
}

# ---- Building it (usability §1, §4) ---------------------------------------------------
# The grandeur still missing: the target less the progress, at least 0.
gm_com_grandeur_to_go = {
	value = 0
	if = {
		limit = {
			has_variable = gm_com_open
			has_variable = gm_com_target
		}
		value = var:gm_com_target
		subtract = gm_com_progress
		min = 0
	}
}

# The levels the grandeur still missing takes at today's world-wealth rate
# (sites spec 2026-10-18 §3.2, Plan B), rounded up, at most 100 a press: one
# `while` should queue no more.
gm_com_levels_to_go = {
	value = gm_com_grandeur_to_go
	divide = gm_world_rate_value
	ceiling = yes
	max = 100
}

# The levels the whole target takes at today's rate (the offer's cost line).

	value = 0
	if = {

		value = var:gm_com_target
		divide = gm_world_rate_value
		ceiling = yes
		max = 100
	}
}
