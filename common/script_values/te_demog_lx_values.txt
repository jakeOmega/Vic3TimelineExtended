# State scope: r, the biological years a calendar year of those in longevity care (§1.2; demographics_model.ageing_rate).
te_demog_lx_rate = {
	value = 1
	add = modifier:state_ageing_rate_add
	min = 0
	max = 1
}

# State scope: Biological Immortality's pull, the share of a treated body's gap to 35 closed a year (§1.4).
te_demog_lx_pull = {
	value = modifier:state_rejuvenation_add
	min = 0
	max = 1
}

# State scope: 1 where longevity care exists for the owner, an ageing carrier or the pull (demographics_model.lx_carrier).
te_demog_lx_carrier_on = {
	value = 0
	if = {
		limit = {
			OR = {
				te_demog_lx_rate < 1
				te_demog_lx_pull > 0
			}
		}
		value = 1
	}
}

# State scope, the owner's laws (§4.2), read from the census-only types they carry (demographics_modifier_types.txt;
# since 2026-10-17, so each law's tooltip says what it does): Human Purity's ban; the health system's cover (Public
# Health Insurance 1, Charitable 0.25, Mandatory Augmentation 1); the floor on the public route (Mandatory Augmentation
# 0.9); the market's reach below the owner's threshold (Unrestricted Augmentation 20).
te_demog_lx_banned = {
	value = 0
	if = {
		limit = { owner = { modifier:country_longevity_care_banned_bool = yes } }
		value = 1
	}
}
te_demog_lx_cover = {
	value = modifier:state_longevity_care_cover_add
	min = 0
	max = 1
}
te_demog_lx_floor = {
	value = modifier:state_longevity_care_floor_add
	min = 0
	max = 1
}
te_demog_lx_market_reach = {
	value = modifier:state_longevity_care_market_reach_add
}

# State scope: the public route's reach (demographics_model.lx_public): the drifted health-care access x the health
# system's cover, at least the floor.
te_demog_lx_public = {
	value = var:te_dg_acc
	multiply = te_demog_lx_cover
	min = te_demog_lx_floor
}

# State scope: the owner's market threshold (te_dg_lx_w, the country's yearly census; te_demog_k_lx_market_start before
# its first), less the market's reach (demographics_model.lx_threshold).
te_demog_lx_owner_w = {
	value = te_demog_k_lx_market_start
	if = {
		limit = { owner ?= { has_variable = te_dg_lx_w } }
		value = owner.var:te_dg_lx_w
	}
}
te_demog_lx_threshold = {
	value = te_demog_lx_owner_w
	subtract = te_demog_lx_market_reach
}

# Country scope: the laws' factor on the market threshold's fall, 1 + their price fall (Private Health Insurance's
# insurers pool the price, +100%; the Regulated Augmentation Market +50%).
te_demog_lx_market_laws = {
	value = 1
	add = modifier:country_longevity_care_price_fall_mult
	min = 0
}

# Country scope: the market threshold's fall this year (demographics_model.lx_market_next): te_demog_k_lx_market_fall x the
# laws' factor x (1 + te_demog_k_lx_market_diffusion x the diffusion technologies' lines).
te_demog_lx_market_fall = {
	value = te_demog_lx_market_diffusion
	multiply = te_demog_k_lx_market_diffusion
	add = 1
	multiply = te_demog_lx_market_laws
	multiply = te_demog_k_lx_market_fall
}

# State scope: the old-age threshold the untreated have (Linked to Common Longevity follows it, five years below): the
# untreated's own table in a state with treated people (te_dg_old_age_c), else the state's (te_dg_old_age).
te_demog_old_age_untreated = {
	value = var:te_dg_old_age
	if = {
		limit = { has_variable = te_dg_old_age_c }
		value = var:te_dg_old_age_c
	}
}

# Country or state scope: the longevity gap (§4.3), Living to 100 for the treated less the untreated, in points of 1; 0
# before the late game.
te_demog_lx_gap = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_lx_l100t
			has_variable = te_dg_lx_l100u
		}
		value = var:te_dg_lx_l100t
		subtract = var:te_dg_lx_l100u
	}
}

# Country scope: the longevity gap's politics' multiplier (§4.4, call 15): the gap x the share of people not in care,
# 0 to 1 (te_demog_lx_gap_politics; on since 2026-10-16).
te_demog_lx_gap_politics_x = {
	value = 0
	if = {
		limit = { has_variable = te_dg_elig }
		value = te_demog_lx_gap
		multiply = { value = 1 subtract = var:te_dg_elig }
		min = 0
		max = 1
	}
}

# Country or state scope: how much younger people's bodies are than their ages, the mean calendar age less the mean
# biological one (te_dg_lx_age, te_dg_lx_bio), in years, over everyone (children and the untreated at 0; a country's
# over its late-game states); 0 before the late game. Past half a year the chronic illness hovers say their rate is by
# the body's age and give it (te_demog_cd_chr_lx, te_demog_st_chr_lx).
te_demog_lx_body_gap = {
	value = 0
	if = {
		limit = {
			has_variable = te_dg_lx_age
			has_variable = te_dg_lx_bio
		}
		value = var:te_dg_lx_age
		subtract = var:te_dg_lx_bio
		min = 0
	}
}
