# ============================================================================
# COLONIAL EMPIRE SCRIPT VALUES
# Used by colonial_stability_bar monthly_progress and decolonization events
# ============================================================================

# ---- Thresholds of the colonial tests, named so the panel can print them ----

# The fewest people a state may hold and still count as a colonial
# territory. is_overseas_colonial_state, is_decolonization_eligible_state and
# is_unintegrated_culturally_distinct_state (colonial_empire_triggers.txt) and
# colonial_high_acceptance_count below all read it; it filters tiny outposts
# (Falklands, Wallis & Futuna). The eligible-territories list's hover prints it.
colonial_territory_min_population = 100000

# The share of our national average standard of living below which a
# territory's is poor enough for the living-standards signal
# (colonial_state_sol_qualifying_floor). Named so the eligible-territories
# list's hover can print it.
colonial_state_sol_floor_share = 0.8

# The indigenous-population signal's share: a non-primary culture counts when
# it is more than this fraction of the state (culture_percent_state in
# colonial_state_indigenous_signal and is_unintegrated_culturally_distinct_state,
# which take a script value as homeland_triggers.txt's do). The
# eligible-territories list's hover prints it.
colonial_state_indigenous_min_share = 0.1

# Count of overseas colonial states (non-homeland, overseas, 100k+ pop, has unintegrated pops)
colonial_state_count = {
	value = 0
	every_scope_state = {
		limit = {
			is_overseas_colonial_state = yes
		}
		add = 1
	}
}

# Count of overseas colonial states with very low acceptance pops (< acceptance_status_2, i.e. 20)
colonial_low_acceptance_count = {
	value = 0
	every_scope_state = {
		limit = {
			is_overseas_colonial_state = yes
			any_scope_pop = {
				# acceptance_status_2 = 20 (vanilla); matches colonial_empire_buttons.txt's threshold
				pop_acceptance < acceptance_status_2
			}
		}
		add = 1
	}
}

# Count of overseas colonial states where ALL pops have high acceptance (>= 60)
colonial_high_acceptance_count = {
	value = 0
	every_scope_state = {
		limit = {
			NOT = { is_homeland_of_country_cultures = owner }
			state_population >= colonial_territory_min_population
			is_overseas_from_capital = yes
			NOT = {
				any_scope_pop = {
					pop_acceptance < acceptance_status_4
				}
			}
		}
		add = 1
	}
}

# Count of states the three decolonization decisions would consider. Snapshotted
# into colonial_empire_eligible_count by colonial_empire_refresh_display so the
# widget's decision rows can say how many territories are on the table without
# walking every state per frame. Same `is_decolonization_eligible_state` trigger
# the buttons use, so the count cannot disagree with what the buttons offer.
colonial_eligible_state_count = {
	value = 0
	every_scope_state = {
		limit = { is_decolonization_eligible_state = yes }
		add = 1
	}
}

# Subset of the above that the Round Table Conference accepts: no pop below
# acceptance_status_2. Mirrors colonial_empire_possible_round_table's gate.
colonial_round_table_state_count = {
	value = 0
	every_scope_state = {
		limit = {
			is_decolonization_eligible_state = yes
			NOT = {
				any_scope_pop = {
					pop_acceptance < acceptance_status_2
				}
			}
		}
		add = 1
	}
}

# Count of great powers with anti-colonial stance (set via event 14).
# Self-exclusion is `this = prev`, not `this = ROOT`: the bar reads these
# through `owner.` from journal-entry scope, where ROOT is not guaranteed to be
# the empire, while `prev` inside the iterator is the empire for every caller.
colonial_gp_condemners_count = {
	value = 0
	every_country = {
		limit = {
			NOT = { this = prev }
			country_rank >= rank_value:great_power
			has_variable = gp_anti_colonial_stance
		}
		add = 1
	}
}

# Count of great powers with pro-colonial stance (set via event 14)
colonial_gp_supporters_count = {
	value = 0
	every_country = {
		limit = {
			NOT = { this = prev }
			country_rank >= rank_value:great_power
			has_variable = gp_pro_colonial_stance
		}
		add = 1
	}
}

# Prestige-weighted versions of the two counts above. Each great power adds
# its own prestige divided by the empire's, clamped to [0.25, 2.0]: a peer
# great power weighs 1.0, a great power with double (or more) the empire's
# prestige weighs 2.0, and one with a quarter (or less) weighs 0.25. `prev`
# inside the iterator is the empire; its prestige is floored at 1 so a
# zero-prestige empire cannot divide by zero.
#
# These drive the bar's per-power condemnation and support terms; the plain
# counts above still feed the events' and buttons' AI weights.
colonial_gp_condemnation_weight = {
	value = 0
	every_country = {
		limit = {
			NOT = { this = prev }
			country_rank >= rank_value:great_power
			has_variable = gp_anti_colonial_stance
		}
		add = {
			value = prestige
			divide = { value = prev.prestige min = 1 }
			min = 0.25
			max = 2
		}
	}
}

colonial_gp_support_weight = {
	value = 0
	every_country = {
		limit = {
			NOT = { this = prev }
			country_rank >= rank_value:great_power
			has_variable = gp_pro_colonial_stance
		}
		add = {
			value = prestige
			divide = { value = prev.prestige min = 1 }
			min = 0.25
			max = 2
		}
	}
}

# Prestige held by every great power plus the empire itself (the empire is
# counted even when it is not a great power). This is the pool the two tier
# escalations measure the condemners against. It is not world prestige
# because the condemners are a subset of the great powers that excludes the
# empire, so a two-thirds share of world prestige would almost never be
# reachable. Floored at 1 for the division below.
colonial_gp_prestige_pool = {
	value = 0
	every_country = {
		limit = {
			OR = {
				this = prev
				country_rank >= rank_value:great_power
			}
		}
		add = prestige
	}
	min = 1
}

# Combined prestige of the condemning great powers as a share of that pool
# (0-1). Gates the tier escalations: a third -> diplomatic isolation,
# two-thirds -> international consensus. Because the empire's own prestige is
# in the pool, a dominant empire is much harder to isolate than a small one.
# Snapshotted into colonial_empire_condemner_share for the widget.
colonial_gp_condemner_prestige_share = {
	value = 0
	every_country = {
		limit = {
			NOT = { this = prev }
			country_rank >= rank_value:great_power
			has_variable = gp_anti_colonial_stance
		}
		add = prestige
	}
	divide = colonial_gp_prestige_pool
}

# The supporting great powers' share of the same pool (0-1). Display only: the
# overview's pie shows it beside the condemners' share, but neither escalation
# reads it; support acts through the per-power colonial_stability_term_gp_support.
# Snapshotted into colonial_empire_supporter_share for the widget.
colonial_gp_supporter_prestige_share = {
	value = 0
	every_country = {
		limit = {
			NOT = { this = prev }
			country_rank >= rank_value:great_power
			has_variable = gp_pro_colonial_stance
		}
		add = prestige
	}
	divide = colonial_gp_prestige_pool
}

# Bureaucracy cost of colonial maintenance - scaled by number of colonial states
colonial_maintenance_bureaucracy_cost = {
	value = colonial_state_count
	multiply = -0.01
	min = -0.25
}

# Metropole state count: total states minus overseas colonial states.
# Floored at 1 to avoid divide-by-zero when an empire is wholly colonial.
metropole_state_count = {
	value = 0
	every_scope_state = {
		limit = {
			NOT = { is_overseas_colonial_state = yes }
		}
		add = 1
	}
	min = 1
}

# Overreach ratio: how much your colonial holdings exceed ~1.5x your metropole.
# 0 (or below) means no penalty; positive values feed the bar's overreach drain.
# A larger metropole tolerates a larger empire proportionally — France can hold
# more before overreach than Belgium, Belgium can hold its small empire freely.
colonial_overreach_ratio = {
	value = colonial_state_count
	divide = metropole_state_count
	subtract = 1.5
	min = 0
}

# ============================================================================
# PROGRAMME LEVELS
#
# Each of the three colonial programmes runs at a level from 0 (off) to
# colonial_programme_max_level, held in a country variable: colonial_invest_level,
# colonial_garrison_level, colonial_assim_level. The top level is the full
# programme and each level is an equal share of it, linearly: the bar term
# (colonial_stability_term_*), the programme's static modifier (applied by
# colonial_programmes_refresh_modifiers at a third, two thirds or all of its
# values) and its costs all scale with the level.
#
# The level variables are also the record a revolution's winner inherits:
# variables merge, modifiers do not, so the refresh rebuilds the modifiers from
# the levels (decol_sync_programmes, decolonization.txt).
#
# Read the levels through the *_level_value wrappers: before a country first
# steps a programme the variable does not exist, and the widget and the bar
# read these every frame. A save from before the levels holds a running
# programme's modifier and no level until the entry's next monthly pulse
# migrates it (decol_migrate_programmes); until then the wrapper reads the
# modifier as the top level, so a step, the refresh and the bar all see the
# programme it was running, not 0.
#
# SCOPE: country.
# ============================================================================

# The top level. Literal 3s that mean this value: the `< 3` gates in
# colonial_empire_possible_*_up (colonial_empire_triggers.txt), the multiplier
# ladder in colonial_programme_apply_modifier and the "/ 3" in the widget's
# level labels (decolonization.txt, te_journal_entries_l_english.yml).
colonial_programme_max_level = 3

colonial_invest_level_value = {
	value = 0
	if = {
		limit = { has_variable = colonial_invest_level }
		value = var:colonial_invest_level
	}
	else_if = {
		limit = { has_modifier = colonial_development_investment_modifier }
		value = colonial_programme_max_level
	}
}

colonial_garrison_level_value = {
	value = 0
	if = {
		limit = { has_variable = colonial_garrison_level }
		value = var:colonial_garrison_level
	}
	else_if = {
		limit = { has_modifier = colonial_military_garrison_modifier }
		value = colonial_programme_max_level
	}
}

colonial_assim_level_value = {
	value = 0
	if = {
		limit = { has_variable = colonial_assim_level }
		value = var:colonial_assim_level
	}
	else_if = {
		limit = { has_modifier = colonial_cultural_assimilation_modifier }
		value = colonial_programme_max_level
	}
}

# The share of the full programme now running, 0-1. Scales the bar term and the
# programme-month counters that choose the ending (a month at the top level
# counts one, as a month of the old on/off programme did).
colonial_invest_strength = {
	value = colonial_invest_level_value
	divide = colonial_programme_max_level
}

colonial_garrison_strength = {
	value = colonial_garrison_level_value
	divide = colonial_programme_max_level
}

colonial_assim_strength = {
	value = colonial_assim_level_value
	divide = colonial_programme_max_level
}

# What one level adds to the bar's monthly change: the full-strength figure (the
# law-adjusted effectiveness aggregate) divided by the levels. Read by the AI
# review and by the widget's rows and step tooltips.
colonial_invest_step_term = {
	value = modifier:country_colonial_invest_effectiveness_add
	divide = colonial_programme_max_level
}

colonial_garrison_step_term = {
	value = modifier:country_colonial_garrison_effectiveness_add
	divide = colonial_programme_max_level
}

colonial_assim_step_term = {
	value = modifier:country_colonial_assim_effectiveness_add
	divide = colonial_programme_max_level
}

# ============================================================================
# PROGRAMME COSTS
#
# UNITS. country_expenses_add is a WEEKLY amount (the budget panel's lines are
# weekly) and `gdp` is a YEARLY figure (the engine's "yearly predicted GDP",
# in country and state scope alike), so every money value here divides a
# yearly figure by colonial_weeks_per_year. Until 2026-10 the running cost was
# gdp x 0.005, commented as a monthly charge but paid weekly: 0.5% of yearly
# GDP every week, 26% of GDP a year, whatever the size of the empire. A
# play-test AI with GDP £2.314B and two small colonies paid £11,570,607 a week,
# over half its spending.
#
# INVESTMENT's running cost scales with what is being developed: the yearly
# GDP of our overseas colonial states, never counted as less than
# colonial_dev_base_floor_share of our own GDP. The floor is the programme's
# fixed overhead, and it gives an empire of colonial subjects alone, which has
# no colonial states of its own, a cost. Each level costs
# colonial_invest_rate_per_level of that base a year.
#
# Target, as a share of our GDP a year per level:
#   colonies make 10% of our GDP (a typical empire)   0.5%   (1.5% at level 3)
#   colonies make 25%                                 1.25%  (3.75%)
#   colonies under 1% (the floor)                     0.05%  (0.15%)
# For scale, one Cultural Hegemony funding level costs about 0.26% a year.
# The play-test case above is on the floor: £22.3k a week per level, £66.8k at
# level 3, against the old £11.57M.
#
# ONE-OFFS are charged on every step up, in weeks of one Investment level's
# running cost, so they scale with the colonies too (they were 2% and 1% of
# yearly GDP, commented as monthly). A typical empire pays 0.25% of GDP a step
# for Investment and 0.125% for Assimilation; the play-test case £0.58M and
# £0.29M.
#
# Garrison costs no money: its authority upkeep is in its static modifier and
# scales with the level like the rest of it (100 a level, 300 at the top).
# ============================================================================

colonial_weeks_per_year = 52

colonial_invest_rate_per_level = 0.05

colonial_dev_base_floor_share = 0.01

colonial_invest_startup_weeks = 26

colonial_assim_startup_weeks = 13

# Garrison's authority upkeep per level: colonial_military_garrison_modifier's
# country_authority_cost_add (300) over the three levels. The raise gate
# (colonial_empire_possible_garrison_up) and the widget's cost line read it.
colonial_garrison_authority_per_level = 100

# Assimilation's authority upkeep per level: colonial_cultural_assimilation_modifier's
# 100 over the three levels, rounded up. The AI's raise check reads it.
colonial_assim_authority_per_level = 34

# The yearly GDP of our overseas colonial states. Iterates every state, so it is
# cached once a month (and when the entry starts) into var:colonial_dev_colonial_gdp by
# colonial_programmes_refresh_cost_base; everything else reads the cache.
colonial_development_gdp = {
	value = 0
	every_scope_state = {
		limit = { is_overseas_colonial_state = yes }
		add = gdp
	}
}

# The base each Investment level costs a share of: the cached colonial GDP,
# never less than the floor share of our own GDP. O(1), so the widget can read
# it every frame. Before the cache exists it is the floor.
colonial_dev_cost_base = {
	value = 0
	if = {
		limit = { has_variable = colonial_dev_colonial_gdp }
		value = var:colonial_dev_colonial_gdp
	}
	min = {
		value = gdp
		multiply = colonial_dev_base_floor_share
	}
}

# The weekly cost of ONE Investment level. Positive.
colonial_invest_level_cost = {
	value = colonial_dev_cost_base
	multiply = colonial_invest_rate_per_level
	divide = colonial_weeks_per_year
}

# The weekly running cost at the current level. colonial_programmes_refresh_modifiers
# copies it into var:colonial_invest_cost_week, which is the expense modifier's
# multiplier (a flat variable, so the stored multiplier never re-runs the sum).
colonial_invest_weekly_cost = {
	value = colonial_invest_level_cost
	multiply = colonial_invest_level_value
}

# One-off payment for each Investment step up. Negative, for add_treasury.
colonial_invest_step_cost = {
	value = colonial_invest_level_cost
	multiply = colonial_invest_startup_weeks
	multiply = -1
}

# What one Assimilation level takes off our bureaucracy: the modifier's -15%
# over the three levels, as a share of what we produce. The AI raises it only
# with a surplus larger than this (colonial_ai_can_raise_assim).
colonial_assim_bureaucracy_per_level = {
	value = produced_bureaucracy
	multiply = 0.05
	min = 0
}

# One-off payment for each Assimilation step up. Negative, for add_treasury.
colonial_assim_step_cost = {
	value = colonial_invest_level_cost
	multiply = colonial_assim_startup_weeks
	multiply = -1
}

# The two one-offs as positive amounts, for the widget's cost lines.
colonial_invest_step_cost_display = {
	value = colonial_invest_step_cost
	multiply = -1
}

colonial_assim_step_cost_display = {
	value = colonial_assim_step_cost
	multiply = -1
}

# Garrison's authority upkeep at the current level, for the widget.
colonial_garrison_authority_display = {
	value = colonial_garrison_level_value
	multiply = colonial_garrison_authority_per_level
}

# ============================================================================
# AI PROGRAMME REVIEW (colonial_ai_review_programmes, decolonization.txt)
#
# The monthly change above which another programme level adds nothing: the
# monthly cap while the bar is below 100, and colonial_ai_full_bar_margin once
# it is at 100. At 100 the bar cannot rise, so a level is useful only while it
# keeps the change safely positive: a dip below 100 restarts the 60-month
# completion count (colonial_at_100_months), so "the bar is full" alone is no
# reason to cut. Trimmed to 0, the change would leave the bar at 100 only
# until the first war or bout of turmoil, and the review would then pay to
# raise the levels again below 100.
# ============================================================================

# The buffer the AI keeps at a full bar: the size of the war term
# (colonial_stability_term_war), so a new war alone does not knock the bar off 100.
colonial_ai_full_bar_margin = 0.5

colonial_ai_programme_ceiling = {
	value = colonial_stability_drift_cap
	if = {
		limit = { colonial_stability_bar_at_least = { N = 100 } }
		value = colonial_ai_full_bar_margin
	}
}

# The monthly change, before the cap, with one level of a programme taken
# away. The AI lowers that programme when this still reaches the ceiling: the
# level is wasted. O(1), like the display value it starts from.
colonial_ai_drift_without_invest_step = {
	value = colonial_stability_drift_uncapped_display
	subtract = colonial_invest_step_term
}

colonial_ai_drift_without_garrison_step = {
	value = colonial_stability_drift_uncapped_display
	subtract = colonial_garrison_step_term
}

colonial_ai_drift_without_assim_step = {
	value = colonial_stability_drift_uncapped_display
	subtract = colonial_assim_step_term
}

# The weights the review picks a programme to raise by, 0 when it cannot be
# raised (colonial_ai_can_raise_*, colonial_empire_triggers.txt). They are the
# weights the six AI-only scripted buttons carried before the levels:
# Investment for any empire, more with restless colonies and for a great
# power; Garrison only with two or more restless colonies; Assimilation for
# any empire, more with an accepted colony.
colonial_ai_invest_raise_weight = {
	value = 0
	if = {
		limit = { colonial_ai_can_raise_invest = yes }
		add = 30
		if = {
			limit = { colonial_low_acceptance_count >= 2 }
			add = 20
		}
		if = {
			limit = { country_rank >= rank_value:great_power }
			add = 10
		}
	}
}

colonial_ai_garrison_raise_weight = {
	value = 0
	if = {
		limit = {
			colonial_ai_can_raise_garrison = yes
			colonial_low_acceptance_count >= 2
		}
		add = 40
		if = {
			limit = { colonial_low_acceptance_count >= 4 }
			add = 20
		}
	}
}

colonial_ai_assim_raise_weight = {
	value = 0
	if = {
		limit = { colonial_ai_can_raise_assim = yes }
		add = 20
		if = {
			limit = { colonial_high_acceptance_count >= 1 }
			add = 15
		}
	}
}

# Scope: state. SoL threshold below which the state qualifies as colonial-
# feeling regardless of cultural acceptance: 0.8 x owner's national average.
# Multiplicative scaling means a poorer metropole has a lower absolute floor
# (its colonies need less SoL to "feel poor"); a richer metropole has higher
# absolute tolerance but the same proportional sensitivity. Captures the
# Portugal-with-assimilatory-citizenship case: high acceptance but the
# colonies are still markedly poorer than the mainland.
colonial_state_sol_qualifying_floor = {
	value = owner.average_sol
	multiply = colonial_state_sol_floor_share
}

# ============================================================================
# COLONIAL STABILITY DRIFT — LEAF TERMS
#
# Single source of truth for every term of colonial_stability_bar's
# monthly_progress. The bar consumes these leaves one-for-one (each still
# carrying its own `desc` key, so the bar's hover breakdown keeps full
# per-term granularity); the widget and the history chart consume the group
# sums and colonial_stability_drift_total below. Nothing is typed twice.
#
# SCOPE: country. The bar's monthly_progress runs in journal_entry scope and
# reaches these as `owner.colonial_stability_term_X` -- the same idiom the bar
# already used for colonial_overreach_ratio and the acceptance counts.
#
# EDITING RULE: to retune a term, change ONLY the leaf here. Do not add a
# number to the bar, the widget, the charts or any loc string. To add a term:
# add a leaf, add it to its group below, and add one `add` line to the bar
# with a matching `desc` key. The full old-term -> leaf mapping is tabulated
# in docs/systems/mod_systems.md § Stability Bar Formula.
#
# The bar's 22nd and last line, colonial_stability_term_cap, is not one of
# these leaves: it caps the sum of all 21 and lives in the MONTHLY CAP
# section after the group sums.
#
# Every leaf returns 0 when its gate is false (the gate moved from the bar's
# enclosing `if` into the leaf), so the bar's `add` lines are unconditional.
# ============================================================================

# --- Base drift ---
colonial_stability_term_base = {

# --- Great-power pressure (linear + two tier escalations) and support ---
# The per-power terms scale with each great power's prestige relative to ours
# (colonial_gp_condemnation_weight / _support_weight above), so their figures
# are what a peer great power contributes; both weights are 0 when nobody
# holds the stance, so no gate is needed. The two escalations are flat and
# fire on the condemners' share of great-power prestige
# (colonial_gp_condemner_prestige_share), not on how many of them there are.
# They stack: at two-thirds both apply.
colonial_stability_term_gp_condemnation = {
	value = colonial_gp_condemnation_weight
	multiply = -0.6
}

colonial_stability_term_gp_high_pressure = {
	value = 0
	if = {
		limit = {
			# A share of a third needs at least one condemner; this stops at the
			# first and spares the share's two walks over every country when
			# nobody condemns (2026-10-03 script profile).

# ============================================================================
# MONTHLY CAP
#
# However large the sum of the leaves, the bar moves at most this far in one
# month, in either direction: 100 / 60, so crossing the whole 0-100 bar takes
# at least five years. The cap is applied to the TOTAL, after everything is
# summed, not to any single term. A sudden spike (a superpower condemning a
# small empire) therefore cannot empty the bar in a few months, but a large
# enough sum still outweighs the empire's own efforts, and the empire has to
# change the inputs (win the great power over, integrate the colony, release
# it) to turn the trend around.
#
# The bar applies it as one last leaf, colonial_stability_term_cap: the
# difference between the capped and the uncapped sum, 0 while the sum is
# inside the cap. Its hover line therefore shows how much the cap absorbed.
# ============================================================================

colonial_stability_drift_cap = 1.667

# Projected monthly change after the cap, identical by construction to what
# the bar applies (its 21 leaves plus colonial_stability_term_cap). Snapshotted
# into colonial_empire_d_total for the signed history series, which is sampled
# from the monthly pulse and can afford the iteration.
colonial_stability_drift_total = {
	value = colonial_stability_drift_uncapped
	max = colonial_stability_drift_cap
	min = {
		value = colonial_stability_drift_cap
		multiply = -1
	}
}

# The bar's last line. Capped minus uncapped, so the clamp above finishes
# inside its own script value before the subtraction runs.
colonial_stability_term_cap = {
	value = colonial_stability_drift_total
	subtract = colonial_stability_drift_uncapped
}

# The uncapped total, but safe to read from the widget every frame: the six O(1)
# groups are evaluated live and the three iterating ones come from the snapshot
# variables. That mix is deliberate rather than a compromise -- the six live
# groups are exactly the ones a button press changes, so the headline moves the
# frame after a click, which a pure snapshot could never do (add_modifier and
# remove_modifier results are invisible inside the effect that wrote them).
# The three snapshotted groups cannot be changed by any button here.

	value = colonial_stability_drift_base
	add = colonial_stability_drift_laws
	add = colonial_stability_drift_igs
	add = colonial_stability_drift_rank
	add = colonial_stability_drift_policies
	add = colonial_stability_drift_domestic
	if = {
		limit = { has_variable = colonial_empire_d_overreach }
		add = var:colonial_empire_d_overreach
	}
	if = {
		limit = { has_variable = colonial_empire_d_gp }
		add = var:colonial_empire_d_gp
	}
	if = {
		limit = { has_variable = colonial_empire_d_acceptance }
		add = var:colonial_empire_d_acceptance
	}
}

# The widget's headline: the same, capped. O(1), like the uncapped version.

	value = colonial_stability_drift_uncapped_display
	max = colonial_stability_drift_cap
	min = {
		value = colonial_stability_drift_cap
		multiply = -1
	}

# How much the cap is absorbing this month (0 inside the cap), shown as the

colonial_stability_drift_cap_display = {

			value = 2
		}
