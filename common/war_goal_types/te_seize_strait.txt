# ============================================================================
# SEIZE STRAIT — take only a state's strait shore, not the whole state
#
# Base 1.15 names two shore provinces for each of its 12 straits
# (common/strait_definitions/strait_definitions.txt, `first_land_endpoint` /
# `second_land_endpoint`), and holding one of them makes a country that
# strait's controller (concept_strait_control_desc). This goal takes the shore
# provinces the target state holds and leaves the rest of the state with its
# owner, as the base game's canal purchase events do (canal_events.3 / .7 hand
# over the two canal provinces with `set_owner_of_provinces`).
#
#   1. One goal per state. Where a strait's two shores lie in two states
#      (Malacca: Singapore in Malaya, the far shore in North Sumatra), each
#      shore is its own goal, and the two can belong to different countries.
#      Panama, Suez, Kiel and Lombok have both shores in one state, and one
#      goal takes both. Owner's call, 2026-11-07.
#   2. A canal zone (Suez, Panama, Kiel) is offered only once its canal is
#      built (owner's call, 2026-11-07): the canal building needs the whole
#      state region or a treaty port there, so a seized zone could never
#      build it (common/buildings/10_canals.txt).
#   3. kind = custom: no engine effect, the whole transfer is
#      te_seize_strait_shores in on_enforced. The province list lives in
#      te_seize_strait_triggers.txt and te_seize_strait_effects.txt and is
#      pinned against vanilla's strait definitions by test_seize_strait_wargoal.py.
#   4. Conflicts like take_treaty_port: `conflicts_with_annex_state` and
#      `validate_conflicts_conquer_state` keep it off a state another goal in
#      the play already conquers. No `annexes_entire_state`. Like
#      conquer_state, never a treaty port (`validate_target_not_treaty_port`):
#      a treaty article holds that state, and moving its province would
#      break it.
#   5. No mirrored_wargoal block. `method = territorial` builds its retake goal
#      from the goal's annexation, which a custom goal doesn't declare, so this
#      goal is enforced only by capitulation or a peace deal, like the mod's
#      other custom goal (te_colonial_independence).
#   6. Infamy is vanilla take_treaty_port's (the target state's
#      state_infamy_value and the two subject multipliers), times the share of
#      the state's provinces taken, never below a quarter
#      (te_seize_strait_infamy_share). A state that is nothing but the shore
#      costs what conquering it costs. infamy_severity is left out: the base
#      game doesn't document it, and conquer_state doesn't set it.
#   7. Maneuvers are take_treaty_port's (base 5, scaled by the state's
#      population), half of conquer_state's base.
#
# Play: common/diplomatic_plays/te_seize_strait_play.txt
# Docs: docs/systems/mod_systems/war.md § Seize Strait War Goal
# ============================================================================

te_seize_strait = {
	icon = "gfx/interface/icons/war_goals/take_treaty_port.dds"

	kind = custom

	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		requires_interest

		# Validation
		validate_target_not_treaty_port
		validate_conflicts_conquer_state
		conflicts_with_annex_state
	}

	# Just below conquer_state (69).
	execution_priority = 68

	contestion_type = control_target_state

	target_type = state

	possible = {
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
	}

	valid = {
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		scope:target_state = {
			owner = scope:target_country
			te_state_has_seizable_strait_shore = yes
		}
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 5
		}

		multiply = {
			desc = "MANEUVERS_TARGET_STATE_POPULATION_FACTOR"
			value = scope:target_state.state_population
			divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
			multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
			max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_PER_STATE
		}
	}

	infamy = {
		value = 0

		scope:target_state = {
			add = state_infamy_value
		}

		multiply = {
			desc = "TE_INFAMY_STRAIT_SHORE_SHARE"
			value = 1
			scope:target_state = {
				multiply = te_seize_strait_infamy_share
			}
		}

		multiply = {
			desc = "INFAMY_FROM_LOYAL_SUBJECT"
			value = 1

			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						liberty_desire < define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_THRESHOLD
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_SUBJECT_MULT
			}
		}

		multiply = {
			desc = "INFAMY_FROM_REBELLIOUS_SUBJECT" # Rebellious as in they are the aggressor in the play
			value = 1

			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						is_diplomatic_play_enemy_of = root
						scope:diplomatic_play ?= {
							initiator_is = scope:target_country
						}
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_REBELLING_SUBJECT_MULT
			}
		}

		min = define:NDiplomacy|WAR_GOAL_INFAMY_MINIMUM
	}

	on_enforced = {
		# scopes: root = holder (the country that receives the shore),
		# creator_country, target_country, target_state.
		scope:target_state = {
			state_region = {
				save_scope_as = te_ss_region
			}
		}
		te_seize_strait_selfcheck_before = yes
		scope:target_state = {
			te_seize_strait_shores = yes
		}
		te_seize_strait_selfcheck_after = yes
	}

	ai = {
		is_significant_demand = no
	}
}
