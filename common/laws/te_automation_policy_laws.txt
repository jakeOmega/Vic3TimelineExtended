# ============================================================================
# AUTOMATION POLICY (era 13 spec § 7.1)
# ============================================================================
# Four laws in lawgroup_automation_policy (Economy), from Humanoid Robotics:
#
#   law_unrestricted_automation   the default; nothing changes
#   law_robot_tax                 the machines stay and are taxed
#   law_human_labor_quotas        the robotic and AI-managed methods are withdrawn
#   law_automation_dividend       the machines' output is paid out as living standards
#
# ORDER MATTERS. law_unrestricted_automation comes first in this file and has
# no gate: the engine gives any country history leaves without a law in a group
# the group's first law in file order, which is also what an old save loading
# the group for the first time gets (scripting_best_practices.md, "First Law
# in a Lawgroup File Is the Unseeded Default"). history/extra_history.txt seeds
# it for 1836 too. A country created later (released, formed, a revolution's)
# has no history: it gets the first law or its creator's, which is unread in
# game (docs/testing/era-13-playtest.md). Either is safe: the law has no gate,
# so it is also the consistency walk's terminal fallback.
#
# WHAT THE LAWS DO.
# - Displacement (owner, 2026-11-04): each law is an answer to the workers the
#   machines put out of work. Under Unrestricted Automation a rise in the
#   automated share costs Trade Unions approval and raises radicals for a year
#   (te_automation_displacement, scaled by the year's rise, so automating
#   slowly stays cheap); Robot Tax halves it, Automation Dividend and Human
#   Labor Quotas end it. The cost of changing fast, not of being automated.
# - Human Labor Quotas works through the production methods: the 40 methods
#   marked robotic or AI-managed (`building_robotic_or_ai_managed_bool = yes`,
#   which te_ap_runs_automated_method reads; common/scripted_triggers/
#   te_automation_policy_triggers.txt explains the choice) also carry
#   `disallowing_laws = { law_human_labor_quotas }`, as vanilla's slavery and
#   Industry Banned methods are withdrawn. Every group keeps an ungated method.
#   Since the government picks a building's methods, the withdrawal alone
#   only takes options away, so the law's own block carries what the
#   guarantee buys politically: the Trade Unions' strength, fewer radicals.
# - Robot Tax and Automation Dividend work through two scaled country
#   modifiers, te_automation_robot_tax and te_automation_dividend
#   (common/static_modifiers/te_automation_policy_modifiers.txt), refreshed
#   monthly from the automated share by te_ap_monthly_update, their only
#   writer (common/scripted_effects/te_automation_policy_effects.txt), as is
#   te_automation_displacement. So those laws' own effects lists are empty,
#   and their on_enact lines say what the modifiers will do
#   (scripting_best_practices.md, "Law enactment preview").
#
# The interest groups' stances: AUTOMATION_STANCES in ideology_modifications.py.
# ============================================================================

law_unrestricted_automation = {
	group = lawgroup_automation_policy

	icon = "gfx/interface/icons/law_icons/law_unrestricted_automation.dds"

	progressiveness = 0

	on_enact = {
		custom_tooltip = te_ap_law_displacement_tt
	}
}

law_robot_tax = {
	group = lawgroup_automation_policy

	icon = "gfx/interface/icons/law_icons/law_robot_tax.dds"

	progressiveness = 25

	unlocking_technologies = {
		humanoid_robotics
	}

	on_enact = {
		custom_tooltip = te_ap_law_robot_tax_tt
		custom_tooltip = te_ap_law_share_tt
		custom_tooltip = te_ap_law_displacement_half_tt
	}
}

law_human_labor_quotas = {
	group = lawgroup_automation_policy

	icon = "gfx/interface/icons/law_icons/law_human_labor_quotas.dds"

	progressiveness = 50

	unlocking_technologies = {
		humanoid_robotics
	}

	modifier = {
		# The unions keep their members' jobs, and so their members.
		interest_group_ig_trade_unions_pol_str_mult = 0.1
		# Work kept for people: fewer turn to movements for it.
		state_radicals_from_political_movements_mult = -0.1
	}

	on_enact = {
		custom_tooltip = te_ap_law_quotas_tt
		custom_tooltip = te_ap_law_displacement_none_tt
	}
}

law_automation_dividend = {
	group = lawgroup_automation_policy

	icon = "gfx/interface/icons/law_icons/law_automation_dividend.dds"

	progressiveness = 75

	unlocking_technologies = {
		humanoid_robotics
	}

	on_enact = {
		custom_tooltip = te_ap_law_dividend_tt
		custom_tooltip = te_ap_law_share_tt
		custom_tooltip = te_ap_law_displacement_none_tt
	}
}
