# ============================================================================
# BODILY AUTONOMY (era 13 spec § 7.1; docs/systems/mod_systems/social_movements.md)
# ============================================================================
# Five laws in lawgroup_bodily_autonomy (Human Rights). The group is about how
# a body looks, chosen for self-expression: appearance, sex characteristics,
# fur, tails, and at the end a body plan of one's own. What a body can do is
# Human Augmentation's, a separate group, so the interest groups split
# differently on the two (ideology_modifications.py, BODILY_AUTONOMY_STANCES).
#
# WHAT EACH LAW ALLOWS (owner, 2026-11-24). Four destinations, not a ladder:
# each leads on its own axis and pays for it.
#
#                             an adult may     a change may be    the state builds
#                             change their     inherited, or      for more than one
#                             own body         chosen for a child kind of body
#   law_natural_form          no               no                 no
#   law_licensed_modification with a board's   no                 no
#                             approval
#   law_morphological_freedom yes              no                 no
#   law_clade_citizenship     yes              yes                yes
#   law_prescribed_form       the state's design, required of everyone
#
#   Licensed Modification  ORDER. The boards decide, so the state keeps its
#       hand on the question (authority) and the argument cools (radicals from
#       movements). Held ten years, or to the debate's end, it settles the debate.
#   Morphological Freedom  THE PERSON. A liberty: anything that ends with the
#       person who chose it, while public life stays built for the standard
#       human. People move to where they can be themselves and a trade grows up
#       around them (migration pull, services, art). The cost is cohesion
#       (assimilation) and the backlash (the Devout's attraction: it grows the
#       one group that strongly opposes the law, morphological_freedom_events.5).
#   Clade Citizenship      A PEOPLE. Recognition: a lineage can be founded, its
#       children are born citizens, and schools, workplaces, housing and medicine
#       are built for them too. People born to a body, in workplaces rebuilt
#       around them, die less at work and by accident (the census's two safety
#       shares), and settle worlds the standard body can't (space colony
#       growth). The costs are the accommodation (the bureaucracy each head of
#       population costs) and a people that doesn't assimilate. It carries no
#       art line: that is Morphological Freedom's.
#   Prescribed Form        CONTROL. Below.
#
# SIZES (owner, 2026-11-24, from a 2039 game: New York's assimilation stood at
# +690%, its migration pull at about +194%, the country's authority at 3,960).
# A late-game law's percentage adds to a sum of hundreds of percent, so a line
# sized like an 1836 law's does nothing: +10% assimilation on +690% is 1.3%.
# These are sized to move such a total by a real fraction. In that game:
# Morphological Freedom's migration pull +75% is about a quarter more people
# drawn, its assimilation -100% an eighth less; Clade Citizenship's -300% is
# three eighths less, and ends assimilation in a country with under +200% of
# its own. The two safety shares are factors of their own with a cap, so they
# are not diluted.
#
# THE TECHNOLOGY CARRIES NOTHING NOW. Self-Directed Evolution held the work
# safety and the colony growth until 2026-11-24; both moved here (owner), so the
# law is what turns the science into a population.
#
# CLADE CITIZENSHIP AND THE BAN ON GENETIC MODIFICATION bar each other
# (disallowing_laws on both; owner, 2026-11-24): a heritable change is genetic
# modification. A country that comes to hold both (a revolution, a regime
# change) is moved off one by extra_law_consistency_generated.txt.
#
# ORDER MATTERS. law_natural_form comes first in this file: the engine gives a
# country with no law in a group the group's first law in file order, which is
# also what an old save loading this group for the first time gets, and what a
# country created later gets (scripting_best_practices.md, "First Law in a
# Lawgroup File Is the Unseeded Default"). common/history/extra_history.txt
# seeds it too, as it does law_no_augmentation.
#
# NATURAL FORM CARRIES NOTHING. Every country holds it from 1836 until era 12,
# so a modifier on it would apply for two centuries with no law to answer it.
# For the same reason every ideology is neutral on it: the groups that want it
# (the Devout, the Rural Folk) show it by opposing the other laws.
#
# THE ACCEPTANCE HOOK. When appearance is chosen, the heritage trait, the
# nearest thing the game has to how people look, should matter less. The two
# middle laws add to country_acceptance_no_shared_heritage_trait_add on top of
# the citizenship law in force (+5, +10). The mod's citizenship laws
# (common/laws/modified.txt) are not vanilla's: no shared heritage reads -20 to
# 0, and Civil Rights Movement and Social Justice Movements add +10 each by era
# 9, so under four of the six citizenship laws a pop sharing no heritage trait
# already gets +20, against +20 to +25 for one sharing a heritage trait or
# group, and the bonus lifts it above them. Read in game whether the engine
# then gives a pop that shares the trait the higher no-shared value
# (docs/testing/era-13-playtest.md; scripting_best_practices.md, Citizenship
# Acceptance: Read the Mod's Laws, Not Vanilla's). By era 12 most countries
# accept or assimilate nearly everyone, so these lines are flavor (owner,
# 2026-11-24); the laws stand on their other lines.
#
# CLADE CITIZENSHIP TAKES HERITAGE OUT OF ACCEPTANCE (spec § 7.3). A static
# bonus can't do it in this engine. Under the mod's citizenship laws, with Civil
# Rights Movement and Social Justice Movements, a shared heritage trait, a
# shared trait group and none read 20 / 20-25 / 20 under four of the six
# (Sovereign Subjecthood; Cultural, Assimilatory and Universal Citizenship),
# 40 / 30 / 10 under Ancestral and 60 / 60 / 0 under Racialized Citizenship:
# a flat +20 to no shared heritage would rank those pops 20 points above a
# shared trait under the first four, and still leave 30 to 60 points of
# heritage under the other two. So the law carries Morphological Freedom's +10
# (no pop is worse off than under it), and te_clade_parity_refresh
# (common/scripted_effects/te_bodily_autonomy_effects.txt, the monthly country
# pulse) raises each of the three heritage figures to the highest of them,
# with te_clade_parity_trait, _group and _none at multiplier = the gap. Every
# pop then reads the same heritage figure, and none ranks above another by it.
#
# ASSIMILATION ACROSS HERITAGES (owner, 2026-11-25; assimilation_cultures.md).
# The engine assimilates a pop only within its heritage group; the mod's yearly
# pass (te_assimilation_culture_effects.txt) does the rest, under the
# assimilation_cultures_rule game rule. country_cross_heritage_assimilation_add
# is the share of the usual rate at which a pop joins a culture of another
# heritage group: nothing under Natural Form, half under Licensed Modification
# (a board approves each change), the whole of it under the other three. Each
# law's own assimilation line then moves that step with the rest, since the
# pass reads the state's assimilation sum: Prescribed Form's +200% makes it the
# fastest, and Clade Citizenship's -300% ends it in a country with under +200%
# of its own. Under the rule's Unrestricted setting the line does nothing.
#
# THE DEBATE. common/journal_entries/je_morphological_freedom.txt succeeds on
# law_morphological_freedom or law_clade_citizenship, is settled by
# law_licensed_modification held ten years or to its end, and fails on
# law_natural_form held to its end.
#
# PRESCRIBED FORM, the fifth law (owner, 2026-11-04), is the dystopian end
# of the group: the state sets a standard body and requires it. Nobody can be
# told apart by looks, so it takes Clade Citizenship's heritage parity
# (te_clade_parity_refresh) without its tolerance line: every heritage figure
# rises to the highest in force and no further. Under the inclusive
# citizenship laws that changes nothing; under Racialized Citizenship it is the
# fastest end to heritage discrimination there is, equality by erasure. The
# government chooses the form (authority) and its bureaucracy enforces it in
# every state (the bureaucracy's cost per head of population). Its other lines
# were resized with the rest on 2026-11-24.
#
# A SIXTH LAW goes last in this file, with a stance in each
# BODILY_AUTONOMY_LISTS row in ideology_modifications.py and in the
# hand-written blocks of common/ideologies/extra_ideologies.txt
# (ideology_lawgroup_audit names any it misses), then
# scripts/generators/regen_ideologies_offline.py.
# ============================================================================

	}
}

law_clade_citizenship = {
	group = lawgroup_bodily_autonomy

	icon = "gfx/interface/icons/law_icons/law_clade_citizenship.dds"

	unlocking_technologies = {
		self_directed_evolution
	}

	# A heritable change is genetic modification (owner, 2026-11-24); the ban
	# names this law in turn (common/laws/extra_laws.txt).
	disallowing_laws = {
		law_ban_on_genetic_modification
	}

	progressiveness = 80

	# What the parity does, on the law's tooltip: its lines are scaled
	# modifiers a pulse applies, so the law's own block can't show them.
	on_enact = {
		custom_tooltip = te_clade_law_parity_tt
	}

	modifier = {
		# People born to a body, in workplaces and streets rebuilt around them.
		# The census's modern safety (demographics_params.SAFETY_TYPES): work
		# reaches its cap, SAFETY_MAX 0.95, from the era 7-9 carriers' 0.90, and
		# accidents and violence 0.65 from the technologies' 0.50. Both were
		# Self-Directed Evolution's or nobody's until 2026-11-24.
		state_work_safety_add = 0.05
		state_external_safety_add = 0.15
		# Colonists adapted to low gravity and radiation: every space colony we
		# hold grows by 0.3% a month, not 0.2% (each world's sr_site_growth_<key>,
		# gen_solar_sites.GROWTH_MODIFIER).
		country_sr_colony_growth_add = 0.001
		state_migration_pull_mult = 0.5
		# Morphological Freedom's line, so no pop is worse off than under it; the
		# heritage parity on top (te_clade_parity_refresh) lifts every heritage
		# figure to the highest of the three
		country_acceptance_no_shared_heritage_trait_add = 10
		country_cross_heritage_assimilation_add = 1
		# A people of its own doesn't take up the majority's ways.
		state_assimilation_mult = -3
		# Schools, housing, workplaces and medicine for more than one kind of
		# body, per head of population (Prescribed Form's line, for the opposite
		# reason: about 9% at era 12).
		state_bureaucracy_population_base_cost_factor_mult = 1
	}

	ai_enact_weight_modifier = {
		value = 0

		# Half while the country holds an unreformed social law (#1877 trap 4, te_ai_law_weight_values.txt).
		multiply = te_ai_mod_law_group_factor
		add = te_ai_mod_law_group_offset
	}
}
