# Legislated tax code: the carrier law (docs/superpowers/specs/2026-09-29-legislated-tax-code-design.md).
#
# Under te_tax_code_rule (common/game_rules/extra_game_rules.txt) every country's
# taxation law is this one. It collects nothing by itself: all five native tax
# levels carry all five rate keys at zero. The rates come from the generated
# amendments in common/amendments/te_tax_amendments_generated.txt
# (amendment_te_tax_<instrument>_<index>, at most one per instrument, rate =
# index x step), which each carry the same value in all five tax_modifier_*
# blocks. The native tax level (Budget panel, AI, history) then moves only the
# political static modifier tax_modifier_<level> and never a collected rate.
#
# Nobody enacts this law. can_enact is always no and the AI weight is
# prohibitive; the migration installs it with activate_law, which skips can_enact
# and the tech gates. is_visible hides it unless the rule is on. The five vanilla
# taxation laws are the mirror image (common/laws/te_tax_vanilla_law_gates.txt).
# `possible` is NOT a law field (the engine rejects it); can_enact and is_visible
# are the law gates. The generated amendments use `possible`, which is valid there.
#
# gen_law_consistency.py never picks this law as a replacement
# (CARRIER_LAW_DENYLIST); without that it would become the taxation group's
# terminal fallback.

law_te_tax_code = {
	group = lawgroup_taxation
	icon = "gfx/interface/icons/law_icons/law_te_tax_code.dds"
	progressiveness = 0
	is_visible = { te_tax_code_full = yes }
	can_enact = { always = no }
	ai_will_do = { always = no }
	ai_enact_weight_modifier = { value = -100000 }
