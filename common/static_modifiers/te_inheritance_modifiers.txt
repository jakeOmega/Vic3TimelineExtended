# ----------------------------------------------------------------------
#  Wealth Concentration (demographics spec §4.2). Sizes at full strength
#  (a score of 80 or 20; docs/superpowers/specs/2026-10-22-demographic-
#  effects-impact-design.md): the multiplier, read from a variable, keeps
#  rising past it, to 5/3 at 100 and 4/3 at 0 (te_inheritance_values.txt).
#  Concentration grows faster but crashes more often; dispersal grows
#  slower but steadier. Pinned by demographics_effects.SIZES.
#  Country pair: the national figure, refreshed by te_demog_wc_national
#  (te_inh_apply_concentration_modifiers), multipliers
#  te_inh_great_fortunes_mult and te_inh_dispersed_mult: the strata's clout
#  (a few great fortunes buy influence beyond their own state) and the
#  financial cycle, which is national.
#  State three: the state's own score (te_dg_wc), refreshed on the state's
#  yearly pulse (te_inh_refresh_wc_state_effects), multipliers
#  te_inh_wc_high_mult, te_inh_wc_low_mult and te_inh_land_hunger_mult.
#  Saving, wealth's local clout, mobility, schooling and the rural exodus
#  happen where the property is.
# ----------------------------------------------------------------------

# National score above 50. A few great fortunes buy influence, and their
# extra saving is lent on and bids up assets: sharp rises in top incomes
# came before 1929 and 2008 (Kumhof, Rancière and Winant 2015). Bubble
# pressure is read by the banking cycle once it runs (+0.3 a month at full
# strength; banking_cycle_simulation.md §27).
inh_great_fortunes = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_positive.dds
	country_aristocrats_pol_str_mult = 0.2
	country_capitalists_pol_str_mult = 0.2
	country_bubble_pressure_monthly_add = 0.3
}

# National score below 50. Few great houses remain to buy influence, and
# many owners deciding apart don't all pile into one bond issue (Barings and
# Argentina, 1890). -0.4 a month at full strength, more than the other side's
# +0.3: bubble pressure stops at 0, so a push down moves crashes about half as
# far as the same push up (banking_cycle_simulation.md §27).
inh_dispersed_wealth = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_negative.dds
	country_aristocrats_pol_str_mult = -0.2
	country_capitalists_pol_str_mult = -0.2
	country_bubble_pressure_monthly_add = -0.4
}

# State score above 50. The great houses save and invest, and their wealth
# buys clout; the rest find the ladder short, and landowners hold back
# public schooling (Galor, Moav and Vollrath 2009).
inh_concentrated_property = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_positive.dds
	state_aristocrats_investment_pool_efficiency_mult = 0.2
	state_capitalists_investment_pool_efficiency_mult = 0.2
	state_political_strength_from_wealth_mult = 0.2
	state_pop_qualifications_mult = -0.2
	state_education_access_add = -0.1
}

# State score below 50. Small owners save too, so no investment line; the
# same wealth buys less influence, the ladder and the schools open, and a
# wide propertied middle resists revolutionary movements.
inh_dispersed_property = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_coins_negative.dds
	state_political_strength_from_wealth_mult = -0.1
	state_pop_qualifications_mult = 0.1
	state_education_access_add = 0.1
	state_radicals_from_political_movements_mult = -0.1
}

# State score above 50 in farming country, times the agrarian share. Great
# estates leave the young no land of their own: they leave (southern
# Italy's emigrants) or rise (Andalusia, Russia 1905, Mexico 1910), and the
# Land Reform movement shows where the base game creates it.

	state_pop_support_movement_land_reform_mult = 0.25

}

# ----------------------------------------------------------------------
#  Rural-scaled birth and migration (state). The state's yearly pulse

# Forced Heirship: smallholders have fewer children so the plot is not
# split again, and stay on it. The birth line sits on the moved lines'
# absorbed family, as Primogeniture's does (phase 3; the parent spec's
# section 13: the two "should be folded together"): outside the
# Demographics rule's Full setting the moved lines' pair re-applies it (Laws
# and Institutions in a state's birth rates), and under Full the census's
# Family Limitation takes its place, since Forced Heirship x the farm share
# raises that practice's target (te_demog_fl_target). The engine fact it
# rests on is primogeniture_scaled's (te_debug_demog_pr_selfcheck_start).
inh_forced_heirship_rural = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_documents_negative.dds
	state_demog_absorbed_birth_rate_mult = -0.15
	state_migration_quota_mult = -0.2
}

# Primogeniture & Entail: large families, and the younger sons leave the
# land. The birth line sits on the moved lines' absorbed family
# (docs/superpowers/archive/plans/2026-10-11-demographics-late-marriage.md, Task 9;
# the late-marriage spec 4.5, call 4): outside the Demographics rule's Full
# setting the moved lines' pair re-applies it (Laws and Institutions in a
# state's birth rates), and under Full the census's Late Marriage takes its
# place, since impartible inheritance raises it. Forced Heirship's line
# moved the same way in phase 3, to Family Limitation.
# Unverified: that the engine sums a script-only type on a state modifier
# scaled by its multiplier. te_debug_demog_pr_selfcheck_start logs TE_SELF_CHECK
# once a game; BROKEN means take the plan's Plan B.
inh_primogeniture_rural = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_documents_positive.dds
	state_demog_absorbed_birth_rate_mult = 0.1
	state_migration_quota_mult = 0.2
}

# Undivided Farm Succession: one heir takes the farm and the siblings
# leave, as under Primogeniture. Under Forced Heirship it replaces
# inh_forced_heirship_rural.
inh_undivided_farm_rural = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_documents_positive.dds
	state_migration_quota_mult = 0.2
}
