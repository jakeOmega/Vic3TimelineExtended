# ---- Convention on Cultural Diversity (un_agency_ccd) ----

# A party. The ideology resistance halves the cultural hegemony system's
# Foreign Cultural Benchmark at x1 (ch_ideology_resistance_factor, which floors
# the benchmark at nothing). The removal speed slows the decay of minority
# homelands in our states (te_homeland_removal_monthly_effect). Minority
# cultures also assimilate more slowly: the tradeoff.
un_cultural_diversity_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_flag_positive.dds
	country_prestige_mult = 0.02
	country_ideology_resistance_mult = 0.5
	state_assimilation_mult = -0.2
	state_homeland_removal_speed_mult = -0.3
}

	icon = gfx/interface/icons/timed_modifier_icons/modifier_flag_negative.dds
