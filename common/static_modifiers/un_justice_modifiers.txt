# The arms embargo is a trade penalty on every military good: Victoria 3 has
# one trade advantage per good, so it bites on imports and exports alike
# (owner, 2026-10-02: it replaces the Strategic Reserve block, which was
# dropped). The military goods are the base game's seven (category = military
# in common/goods/): small arms, ammunition, artillery, tanks, aeroplanes,
# man-o-wars and ironclads. The mod adds none.
# ============================================================================

un_arms_embargo_target_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_negative.dds
	goods_trade_advantage_small_arms_mult = -0.3
	goods_trade_advantage_ammunition_mult = -0.3
	goods_trade_advantage_artillery_mult = -0.3
	goods_trade_advantage_tanks_mult = -0.3
	goods_trade_advantage_aeroplanes_mult = -0.3
	goods_trade_advantage_manowars_mult = -0.3
	goods_trade_advantage_ironclads_mult = -0.3
	country_military_goods_cost_mult = 0.05
}
