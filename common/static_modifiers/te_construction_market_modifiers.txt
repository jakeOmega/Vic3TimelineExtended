# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

# Per-site throughput multiplier. Re-applied each weekly pulse with
# `multiplier = te_construction_market_per_site` so each construction site
# absorbs its share of the country's total construction-use target.
te_construction_market_capacity_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	building_throughput_add = 1
}

# Country-scope private/public allocation modifier. Re-applied each weekly
# pulse with `multiplier = te_construction_market_private_share_target`
# (a 0..100 percentage) which is then * 0.01 to land in [0, 1].
te_construction_market_private_share_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	country_private_construction_allocation_mult = 0.01
}

# ---- Direct construction (free_market_construction_rule disabled) ----
# Kept on every country while the rule is off, re-applied weekly by
# te_direct_construction_apply_modifiers. See docs/systems/mod_systems.md,
# "Free Market Construction off".

# Unscaled. Construction Sectors - which in this mode only employ - are paid
# for by the government, as vanilla's are.
te_direct_construction_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	country_building_construction_sector_require_subsidies_bool = yes
}

# Multiplied by var:te_dc_output_cancel = 1 + every OTHER country-level
# goods_output_construction_mult (e.g. nuclear_crash_program_strain's -5%), so
# the country-level total is exactly -100% and nothing produces the
# construction good: not the idled sectors, not Engineering & Logistics
# barracks. Exact for the same reason as the input cancel below.
te_direct_construction_no_output_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	goods_output_construction_mult = -1
}

# Multiplied by var:te_dc_input_cancel = 1 + every OTHER country-level
# goods_input_construction_mult, so the country-level total is exactly -100%
# and no building consumes the construction good: no maintenance, no company
# inputs, no retooling tax. An exact cancel rather than a large negative,
# because nothing establishes that the engine floors a goods input at zero.
te_direct_construction_no_maintenance_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	goods_input_construction_mult = -1
}

# Multiplied by var:te_dc_throughput_cancel = the Construction Sector's
# throughput from country-level sources (te_direct_construction_throughput_factor),
# which idles its production method: in this mode the Construction Site buys
# the goods for each point actually spent, so the sector must not buy them too.
te_direct_construction_sector_idle_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	building_construction_sector_throughput_add = -1
}
