# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

# The Construction Sites' throughput. Re-applied to every site each weekly
# pulse (te_construction_market_apply_per_site_modifier) at one multiplier,
# te_construction_market_site_throughput, so the country's sites together
# deliver the week's purchase.
te_construction_market_capacity_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	building_throughput_add = 1
}

# A Construction Site's crew: state scope, market games. A site is placed with
# ten laborer jobs a level, and a building hires a tenth of its jobs a week
# whatever its group's hiring rates say (measured 2026-11-13), so in its first
# week it takes one laborer a level. This then cuts the group's jobs in the
# state by 90%, to that one laborer, and the site is fully staffed from its
# first hire instead of its tenth week. The Free Market Construction mod's
# device; this copy lost it on 2025-09-02 and has it back since 2026-11-13.
# te_construction_market_site_crew adds and removes it. Never in a direct
# game, whose site is a sector in the same group and would run at full output
# on a tenth of its workers (te_construction_market_clear_crew).
te_construction_market_crew_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	building_group_te_construction_market_bg_employee_mult = -0.9
}

# Country-scope private/public allocation modifier. Re-applied each weekly
# pulse with `multiplier = te_construction_market_private_share_target`
# (a 0..100 percentage) which is then * 0.01 to land in [0, 1].
te_construction_market_private_share_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	country_private_construction_allocation_mult = 0.01
}

# Direct construction (free_market_construction_rule disabled): a flat base of
# construction for every country, so one without Construction Sites can still
# build them. Script adds it and takes it away with the rule
# (te_direct_construction_ensure_base, te_construction_market_direct_effects.txt);
# until 2026-11-13 the rule setting applied it itself, which did not follow
# the save; it seems to follow the last lobby (#1767). 10 points a week, the
# base game's maximum (vanilla country_gdp_construction, which the mod empties in
# extra_modifiers.txt): what a player gets there under two of the three
# settings of its Base Construction Capacity rule, flat for every country here
# (#1768; 5 until 2026-11-13). Ten wooden site levels' worth
# (pm_te_direct_construction_wooden, 1 point a level): small beside a working
# construction sector. The points carry no wages.
te_direct_construction_base_modifier = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	country_construction_add = 10
}
