# ============================================================================
# NUCLEAR DETERRENCE AND CRISIS DIPLOMACY — static modifiers
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0 is what shipped).
#
# Three families are *managed*: exactly one member is on at a time and a
# tracking variable records which, because add_modifier / remove_modifier are
# invisible to has_modifier inside the block that ran them
# (scripting_best_practices.md § Worked case — swapping one member of a banded
# modifier family). Their single writers:
#   nd_doctrine_mod_*        nd_apply_posture_modifiers  (je_nuclear_program scope)
#   nd_readiness_mod_*       nd_apply_posture_modifiers  (je_nuclear_program scope)
#   nd_upkeep_cost           nd_apply_posture_modifiers  (je_nuclear_program scope,
#                            multiplier = root.var:nd_upkeep_weekly_cached)
#   nd_posture_approval_*    nd_refresh_domestic_stance  (interest-group scope)
# Everything else below is a timed one-off applied by an event or a crisis
# outcome.
# ============================================================================

# ---- Doctrine: when we say we will use nuclear weapons --------------------
# The international half of each doctrine (§3.1). The domestic half is the
# nd_posture_approval_* family; the launch permission is
# nd_doctrine_permits_strike in nuclear_deterrence_triggers.txt.

# 1 — No first use / assured retaliation: reassures, weak for coercion.

# ---- Readiness: how quickly forces can respond ----------------------------
# Routine carries no modifier. The benefit is response preparedness (the same
# attack-success knob the strike resolution reads) — never a general combat
# bonus (§3.2). The cost is nd_upkeep_cost, strain and incident exposure.
nd_readiness_mod_2 = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_positive.dds
	country_nuclear_weapon_attack_success_add = 0.05
}

	country_nuclear_weapon_attack_success_add = 0.1
	country_diplomatic_play_maneuvers_add = 5
}
