# A metropolis: one city holds most of the state's urban people. (Its crowding is te_demog_mt_crowding, below.)
# - Infrastructure: a compact city is cheaper to serve, so its people give more infrastructure.
# - Agglomeration: its services work at a larger scale. (Not the Financial District's: it produces nothing, so
#   throughput has nothing to scale, #1096.)
# - Unrest: one city concentrates its radicals (the Paris of 1848), a little.
te_demog_mt_metropolis = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_positive.dds
	state_infrastructure_from_population_mult = 0.2
	building_urban_center_throughput_add = 0.1
	state_turmoil_effects_mult = 0.1
}

# A metropolis's crowding: it is denser than the state's average, so the state crowds sooner (a negative value: the
# type is the state's capacity), and congestion then lowers the share's target: the loop that limits the metropolis.
# A model term, so it keeps the 0-1 strength (te_demog_mt_e_pos) and the share's balance points stay where they were;
# bookkeeping, so the mod's blank icon keeps it out of the state's rows (it shows in Migration Crowding's breakdown).
te_demog_mt_crowding = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_system.dds
	state_migration_crowding_density_mult = -0.1
}

# Market towns: the urban people spread over many small towns, which are dearer to serve, so they give less
# infrastructure; a shortfall slows further dispersal until railways catch up.
te_demog_mt_dispersed = {
	icon = gfx/interface/icons/timed_modifier_icons/modifier_gear_negative.dds
	state_infrastructure_from_population_mult = -0.2
}
