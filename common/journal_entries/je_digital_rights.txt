# The Digital Rights debate. It opens in era 9 with Automated Surveillance or
# Cybersecurity, and succeeds on Strong Privacy Rights, which Cybersecurity unlocks
# (Universal Digital Identity, era 11, until 2026-11-04: two eras past the opening,
# so the debate nearly always timed out). The law also needs Guaranteed Liberties,
# the slow half; the status line names whichever of the two is missing. It fails on
# Intrusive Surveillance with the Ministry of Intelligence, and a timeout under
# Moderate Data Privacy costs half the penalty and no radicals.
je_digital_rights = {
	icon = "gfx/interface/icons/event_icons/je_digital_rights.dds"
	group = je_group_internal_affairs

	modifiers_while_active = {
		digital_rights_debate_active_modifier
	}

	is_shown_when_inactive = {
		has_game_rule = social_movements_enabled
		OR = {
			has_technology_researched = automated_surveillance
			has_technology_researched = cybersecurity
		}
		NOT = { is_country_type = decentralized }
		# Rebels do not start their own debate mid-war (and, winning, keep a fresh
		# outcome), as in je_morphological_freedom. Here, not in possible, so it can
		# never touch an active record a revolution's winner inherits.
		is_revolutionary = no
		# Set by the success, failure and timeout outcomes. A revolution's winner gets a
		# fresh record of an entry the loser had finished, and without this it would
		# re-activate: a failed debate would fail again at once and replay its
		# outcome event, and a timed-out one would run its twenty years again.
		NOT = { has_variable = digital_rights_resolved }
		NOT = {
			has_law = law_type:law_strong_privacy_rights
		}
	}

	possible = {
		NOT = { has_variable = digital_rights_resolved }
		OR = {
			has_technology_researched = automated_surveillance
			has_technology_researched = cybersecurity
		}
		NOT = {
			has_law = law_type:law_strong_privacy_rights
		}
	}

	immediate = {
	}

	on_monthly_pulse = {
		random_events = {
			400 = 0
			10 = surveillance_events.1
			10 = surveillance_events.2
			10 = surveillance_events.3
			10 = surveillance_events.4
			10 = surveillance_events.5
		}
	}

	status_desc = {
		first_valid = {
			# What stands between the country and Strong Privacy Rights comes first:
			# the technology (an entry opened by Automated Surveillance may not have
			# it), then the law it needs. The failure outcome's own tooltip carries the
			# Intrusive Surveillance warning while either is missing.
			triggered_desc = {
				desc = je_digital_rights_status_needs_tech
				trigger = {
					NOT = { has_technology_researched = cybersecurity }
				}
			}
			triggered_desc = {
				desc = je_digital_rights_status_needs_liberties
				trigger = {
					NOT = { has_law = law_type:law_guaranteed_liberties }
				}
			}
			triggered_desc = {
				desc = je_digital_rights_status_surveillance
				trigger = {
					has_law = law_type:law_intrusive_surveillance
				}
			}
			triggered_desc = {
				desc = je_digital_rights_status_minimal
				trigger = {
					has_law = law_type:law_minimal_privacy_protection
				}
			}
			triggered_desc = {
				desc = je_digital_rights_status_moderate
				trigger = {
					has_law = law_type:law_moderate_data_privacy
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_digital_rights_complete_conclusion_desc
		trigger = {
			has_law = law_type:law_strong_privacy_rights
		}
		effect = {
			set_variable = digital_rights_resolved
			trigger_event = { id = surveillance_events.200 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_digital_rights_fail_conclusion_desc
		trigger = {
			has_law = law_type:law_intrusive_surveillance
			has_law = law_type:law_ministry_of_intelligence_and_security
			# Full surveillance state established
		}
		effect = {
			set_variable = digital_rights_resolved
			trigger_event = { id = surveillance_events.100 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_digital_rights_invalid_conclusion_desc
		trigger = {
			is_country_type = decentralized
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = je_digital_rights_timeout_conclusion_desc
		effect = {
			set_variable = digital_rights_resolved
			if = {
				# Halfway there: a data protection regime, if not the constitutional right.
				# Half the stagnation and no radicals.
				limit = {
					has_law = law_type:law_moderate_data_privacy
				}
				add_modifier = {
					name = digital_rights_stagnation_modifier
					days = long_modifier_time
					multiplier = 0.5
					is_decaying = yes
				}
			}
			else = {
				add_radicals = {
					value = small_radicals
					pop_type = academics
				}
				add_modifier = {
					name = digital_rights_stagnation_modifier
					days = long_modifier_time
					is_decaying = yes
				}
			}
		}
	}

	# 20 years, as the other debates settled by law have (15 until 2026-11-04).
	timeout = 7300

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
