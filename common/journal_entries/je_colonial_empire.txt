		# - colonial_empire_completed: permanent block. All success paths

		#   any successful end commits the player out of further colonial JE.
		# - colonial_empire_collapsed_recently: 10-year cooldown after a fail
		#   that involved bar=0 (negotiated withdrawal or crumbling collapse).
		#   The voluntary fail path (bar>0, no colonies) sets NO cooldown.

		# country holds before its first run. It deletes a finished record and
		# never activates the type again, so after an end the monthly re-entry

		# a month after the country is due it (ce_reentry_due,
		# colonial_empire_triggers.txt, which restates this block word for
		# word; test_colonial_empire_reentry.py holds the two together).
		# So reacquiring a colony after a voluntary end, or holding one when
		# the cooldown runs out, starts the entry within two months.
		NOT = { has_variable = colonial_empire_completed }

	# --- The nine endings (1.15 outcomes, #1432) ----------------------------
	# One outcome per ending, so the journal panel lists every ending with its
	# condition before it happens. Two outcomes true on one update: the first
	# declared wins and only its effect runs (probed, scripting_best_practices.md

	# triggers are written to exclude each other anyway, so the panel ticks one

	# level 1 counts a third); ties go to Garrison, then Investment. The

	# which are 0 until the first monthly pulse creates the variables: the panel

	# read of a counter that does not exist yet logged an error per frame.
	#
	# The endings read only variables and the bar, so a revolution's winner,
	# which inherits the loser's variables (the Act flags and the counters),
	# continues on the same path.

	on_monthly_pulse = {
		effect = {
			# --- Path-dependence tracking ---
			# Count months each programme runs, weighted by its level: a month
			# at the top level counts 1 (as a month of the old on/off programme
			# did), a month at level 1 a third. Read by the success and failure effects
			# to choose the appropriate resolution event, so a cheap low level
			# cannot steer the ending as fast as the full programme.
			if = {
				limit = { NOT = { has_variable = colonial_invest_months } }
				set_variable = { name = colonial_invest_months value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = colonial_garrison_months } }
				set_variable = { name = colonial_garrison_months value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = colonial_assimilate_months } }
				set_variable = { name = colonial_assimilate_months value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = colonial_solidified_months } }
				set_variable = { name = colonial_solidified_months value = 0 }
			}
			# JE total duration counter — read by long-duration home-front events
			# (e.g. "The Universities Speak" at >10 years).
			if = {
				limit = { NOT = { has_variable = colonial_je_total_months } }
				set_variable = { name = colonial_je_total_months value = 0 }
			}
			change_variable = { name = colonial_je_total_months add = 1 }
			if = {
				limit = { colonial_invest_level_value >= 1 }
				change_variable = { name = colonial_invest_months add = colonial_invest_strength }
			}
			if = {
				limit = { colonial_garrison_level_value >= 1 }
				change_variable = { name = colonial_garrison_months add = colonial_garrison_strength }
			}
			if = {
				limit = { colonial_assim_level_value >= 1 }
				change_variable = { name = colonial_assimilate_months add = colonial_assim_strength }
			}

			# --- Solidified-band sustained-time counter ---
			# Increments while bar is in Solidified band; resets if it drops.
			# The Imperial Federation Act capstone requires 36 sustained months.
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" >= 90 }
				}
				change_variable = { name = colonial_solidified_months add = 1 }
			}
			else = {
				set_variable = { name = colonial_solidified_months value = 0 }
			}

			# --- Bar-at-100 sustained-time counter ---
			# Increments while bar is at the cap; resets if it drops. The basic
			# completion path requires 60 months at 100 (longer than the IF
			# Act's 36-mo ≥90 gate) so reaching the cap isn't an instant win
			# and leaves the IF Act decision as the meaningfully faster route.
			if = {
				limit = { NOT = { has_variable = colonial_at_100_months } }
				set_variable = { name = colonial_at_100_months value = 0 }
			}
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" >= 100 }
				}
				change_variable = { name = colonial_at_100_months add = 1 }
			}
			else = {
				set_variable = { name = colonial_at_100_months value = 0 }
			}

			# --- Programme costs, the AI's review ---
			# The colonies' GDP that Investment's weekly cost is a share of, cached
			# once a month (colonial_programmes_refresh_cost_base). Then an AI
			# reviews its three programmes' levels against the bar and its budget,
			# reading the widget's projected change (its three iterating groups
			# from last month's snapshot). The programmes' modifiers and
			# Investment's expense line follow the levels and the cache on their
			# own (the scaling_modifiers_while_active blocks above).
			colonial_programmes_refresh_cost_base = yes
			if = {
				limit = { is_ai = yes }
				colonial_ai_review_programmes = yes
			}

			# The pressuring-great-powers leaderboard used to be built here, into
			# seven colonial_condemner_rank_N slot variables read by the
			# status_desc. It is gone: the widget's International Pressure
			# section is a table of script containers built in
			# colonial_empire_refresh_display (colonial_empire_refresh_gp_rows,
			# every empire's), each reaching its country through a stored capital
			# rather than the `Var('x').GetCountry.GetName` chain that
			# gui_modding_guide gotcha #11 records as rendering blank in this mod.

			# --- State-side assimilation disruption reapplication ---
			# While Cultural Assimilation runs, every overseas colonial state,
			# and every unintegrated colonial state of a qualifying subject,
			# takes a recurring goods-output / qualifications hit at the
			# programme's level (a third of it a level). Forced cultural
			# conversion stalls the local economy. 2-month duration; reapplied
			# each pulse so it sticks while the programme runs, runs out when it
			# ends.
			if = {
				limit = { colonial_assim_level_value >= 3 }
				colonial_assim_disrupt_colonies = { MULT = 1 }
			}
			else_if = {
				limit = { colonial_assim_level_value >= 2 }
				colonial_assim_disrupt_colonies = { MULT = 0.667 }
			}
			else_if = {
				limit = { colonial_assim_level_value >= 1 }
				colonial_assim_disrupt_colonies = { MULT = 0.333 }
			}

			# Phase modifiers based on current bar value (five status bands)
			je:je_colonial_empire = { remove_modifier = colonial_empire_solidified_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_stable_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_strained_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_under_pressure_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_crumbling_modifier }

			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 20 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_crumbling_modifier } }
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 40 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_under_pressure_modifier } }
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 65 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_strained_modifier } }
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 90 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_stable_modifier } }
			}
			else = {
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_solidified_modifier } }
			}

			# --- Subject-side LD pressure ---
			# Bar status pushes Liberty Desire onto qualifying colonial
			# subjects. Strained band -> mild LD bump; Crumbling -> strong;
			# Collapsing -> severe. Stable / Solidified -> no LD pressure.
			# Pressure is one-way: subjects' LD does not feed back into the
			# bar (the bar is driven by GP condemnation, governance, and
			# overreach terms in the bar's monthly_progress script).
			every_subject_or_below = {
				limit = { is_qualifying_colonial_subject = yes }
				remove_modifier = colonial_overlord_strained_subject_ld_modifier
				remove_modifier = colonial_overlord_crumbling_subject_ld_modifier
				remove_modifier = colonial_overlord_collapsing_subject_ld_modifier
			}
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 20 }
				}
				every_subject_or_below = {
					limit = { is_qualifying_colonial_subject = yes }
					add_modifier = { name = colonial_overlord_collapsing_subject_ld_modifier }
				}
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 40 }
				}
				every_subject_or_below = {
					limit = { is_qualifying_colonial_subject = yes }
					add_modifier = { name = colonial_overlord_crumbling_subject_ld_modifier }
				}
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 65 }
				}
				every_subject_or_below = {
					limit = { is_qualifying_colonial_subject = yes }
					add_modifier = { name = colonial_overlord_strained_subject_ld_modifier }
				}
			}
			# Stable (>=65) and Solidified (>=90): no LD pressure (modifiers
			# already removed at the top of this block).

			# Apply radicalism to civil rights movement specifically (not all movements)
			every_political_movement = {
				limit = { is_political_movement_type = movement_civil_rights }
				remove_modifier = colonial_pressure_movement_radicalism
				remove_modifier = colonial_crumbling_movement_radicalism
			}
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 20 }
				}
				every_political_movement = {
					limit = { is_political_movement_type = movement_civil_rights }
					add_modifier = { name = colonial_crumbling_movement_radicalism }
				}
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 40 }
				}
				every_political_movement = {
					limit = { is_political_movement_type = movement_civil_rights }
					add_modifier = { name = colonial_pressure_movement_radicalism }
				}
			}

			# --- Widget display state -------------------------------------
			# Last, so the snapshot reflects this month's settled phase
			# modifiers and this month's bar value.
			colonial_empire_refresh_display = yes

			# Sample the month for the history charts. Runs after the refresh
			# so the stability bucket and the drift total are this month's.
			te_history_record_colonial_samples = yes
		}
	}

	weight = 500
	can_revolution_inherit = yes
	can_deactivate = no
