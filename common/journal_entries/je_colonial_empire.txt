je_colonial_empire = {
	icon = "gfx/interface/icons/event_icons/je_colonial_empire.dds"

	group = je_group_foreign_affairs

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	# 1.15: with the rule off, drop the inactive record for good (a rule never changes mid-game).
	disable_evaluation = {
		NOT = { has_game_rule = decolonization_enabled }
	}
	is_shown_when_inactive = {
		has_game_rule = decolonization_enabled
		has_technology_researched = decolonization
	}

	possible = {
		# Holders of a colony (is_overseas_colonial_state: separated, distinct,
		# subordinated) OR of a qualifying colonial subject (Colonial
		# Administration Colonies, Chartered Companies, and separated
		# Protectorates sharing no heritage trait with the metropole). The
		# subject branch catches modern-era colonial relations where the
		# metropole owns no colonies directly -- e.g. UK over India, Netherlands
		# over the DEIC, France over Tunisia/Morocco.
		OR = {
			any_scope_state = {
				is_overseas_colonial_state = yes
			}
			country_has_qualifying_colonial_subject = yes
		}
		# Re-entry gates. Set at JE end:
		# - colonial_empire_completed: permanent block. All success paths
		#   (60-mo at bar 100, IF Act capstone) satisfy the user's definition
		#   of "successful completion" (no colony loss, bar not at zero), so
		#   any successful end commits the player out of further colonial JE.
		# - colonial_empire_collapsed_recently: 10-year cooldown after a fail
		#   that involved bar=0 (negotiated withdrawal or crumbling collapse).
		#   The voluntary fail path (bar>0, no colonies) sets NO cooldown.
		# The engine reads this block only to activate the inactive record a
		# country holds before its first run. It deletes a finished record and
		# never activates the type again, so after an end the monthly re-entry
		# pass (ce_reentry_monthly, decolonization.txt) adds the entry back,
		# a month after the country is due it (ce_reentry_due,
		# colonial_empire_triggers.txt, which restates this block word for
		# word; test_colonial_empire_reentry.py holds the two together).
		# So reacquiring a colony after a voluntary end, or holding one when
		# the cooldown runs out, starts the entry within two months.
		NOT = { has_variable = colonial_empire_completed }
		NOT = { has_variable = colonial_empire_collapsed_recently }
	}

	should_be_pinned_by_default_uninvolved_or_context = yes

	scripted_progress_bar = colonial_stability_bar

	# --- Player-facing surface --------------------------------------------
	# The panels (gui/journal_entry_widgets/colonial_empire_widget.gui, which
	# sets the order of the sections): the "?" to the Help page and the
	# overview above the status line, the live sections above the button grid,
	# and the charts below it.
	widget = {
		gui = "gui/journal_entry_widgets/colonial_empire_widget.gui"
		name = "widget_je_colonial_empire_overview"
		container = "custom_widget_container_1"
	}
	# MARKER, not content: the overview draws the stability bar, so the journal
	# panel drops its own bar block at the bottom (gui/journal_entry.gui; see
	# je_banking.txt for the mechanism).
	widget = {
		gui = "gui/journal_entry_widgets/te_je_bars_on_top_marker.gui"
		name = "widget_te_je_bars_on_top_marker"
		container = "custom_widget_container_7"
	}
	# container_2 (under the status line, where the button grid used to be):
	# why stability is moving, international pressure, the three programmes and
	# the three decolonization decisions.
	widget = {
		gui = "gui/journal_entry_widgets/colonial_empire_widget.gui"
		name = "widget_je_colonial_empire_status"
		container = "custom_widget_container_2"
	}
	# container_3: the history charts.
	widget = {
		gui = "gui/journal_entry_widgets/colonial_empire_widget.gui"
		name = "widget_je_colonial_empire_reference"
		container = "custom_widget_container_3"
	}

	# Scripted buttons - the three decolonization decisions.
	# Every one of these carries `is_ai = yes` in its `visible`, so the vanilla
	# grid shows a human nothing; they remain the AI's only path into the
	# decisions (its `ai_chance` blocks). See the header comment in
	# common/scripted_buttons/colonial_empire_buttons.txt before touching them.
	# The three programmes have no buttons: the AI steps them in the monthly
	# pulse (colonial_ai_review_programmes), a player through the widget.
	scripted_button = ce_release_colonial_territory
	scripted_button = ce_round_table_conference
	scripted_button = ce_planned_decolonization

	immediate = {
		# Populate the widget's display state on the first frame, so a freshly
		# activated entry shows real numbers instead of dashes for a month.
		colonial_empire_refresh_display = yes
		# And the colonies' GDP that Investment's costs are a share of, so the
		# first month's rows and steps charge the real figure, not the floor. A
		# cache recomputed from live state, so re-running it is harmless.
		colonial_programmes_refresh_cost_base = yes
	}

	progressbar = no

	# The status line says only what the panel's overview cannot. The five tier
	# phrases, the five great-power-pressure phrases and the seven-slot
	# condemner roster that used to live here (18 triggered_desc in total) are
	# the widget's overview (the band and its tooltip, which carries the status
	# sentence) and "International Pressure" section. So an ACTIVE entry has no
	# status line (no triggered_desc below is valid; the owner's rule, play-test
	# round 2: status text that repeats the overview goes). The entry is shown
	# while inactive (`is_shown_when_inactive`), where the widgets are hidden,
	# and there the line says why it is not running: the three ways `possible`
	# fails. That inactive record exists only before a country's first run:
	# the engine deletes a finished record (the re-entry pass adds a new one,
	# ce_reentry_monthly), so the _completed and _cooldown lines can show only
	# on a record that has never run, held by a country with the variable: in
	# practice a revolution's winner (fresh records, and the loser's
	# variables). After an end, the Timeline Extended window's Colonial Empire
	# tab says why (te_ce_readout, colonial_empire_widget.gui).
	#
	# `first_valid` + `triggered_desc`, the form the journal-entry script docs
	# name and this mod's other entries use (je_banking.txt, je_civil_rights.txt).
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_colonial_empire_status_completed
				trigger = {
					NOT = { has_journal_entry = je_colonial_empire }
					has_variable = colonial_empire_completed
				}
			}
			triggered_desc = {
				desc = je_colonial_empire_status_cooldown
				trigger = {
					NOT = { has_journal_entry = je_colonial_empire }
					has_variable = colonial_empire_collapsed_recently
				}
			}
			triggered_desc = {
				desc = je_colonial_empire_status_no_colonies
				trigger = {
					NOT = { has_journal_entry = je_colonial_empire }
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_colonial_empire_complete_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_colonial_empire_complete_tt
				OR = {
					# Basic completion: colonial_empire_completion_months (60, five
					# years) sustained at the cap. Reaching 100 is not enough on its
					# own — the bar must hold. Longer than the IF Act's
					# colonial_empire_federation_months (36) at ≥90 so the decision
					# path stays meaningfully faster for countries that qualify. Both
					# live in colonial_empire_values.txt, which the overview reads too.
					root = {
						has_variable = colonial_at_100_months
						var:colonial_at_100_months >= colonial_empire_completion_months
					}
					# IF Act capstone — set by the Imperial Federation Act decision
					root = {
						has_variable = imperial_federation_taken
					}
				}
			}
		}
		# Path-dependent resolution: which policy did the empire lean on?
		# Months tracked in vars colonial_invest_months / colonial_garrison_months /
		# colonial_assimilate_months by the monthly pulse below. The dispatch
		# picks the dominant path and triggers a flavor-matched event. It reads them
		# through colonial_{invest,garrison,assimilate}_months_value, which are 0
		# until the first pulse creates the variables: the journal panel renders
		# the outcomes' effects as tooltips every frame, and a bare var: read of a
		# counter that does not exist yet logged an error per frame.
		#
		# IF Act capstone routes (300/301) take priority — reaching the Solidified
		# capstone is the explicit "victory" path; it bypasses the bar-100 routing.
		effect = {
			if = {
				limit = { has_variable = imperial_federation_iron_fist }
				trigger_event = { id = decolonization_events.300 }   # Iron Fist Federation capstone
			}
			else_if = {
				limit = { has_variable = imperial_federation_civilizing }
				trigger_event = { id = decolonization_events.301 }   # Civilizing Mission capstone
			}
			else_if = {
				limit = {
					colonial_garrison_months_value >= colonial_invest_months_value
					colonial_garrison_months_value >= colonial_assimilate_months_value
					colonial_garrison_months_value > 24
				}
				trigger_event = { id = decolonization_events.203 }   # Iron Fist
			}
			else_if = {
				limit = {
					colonial_invest_months_value >= colonial_assimilate_months_value
					colonial_invest_months_value > 24
				}
				trigger_event = { id = decolonization_events.202 }   # Commonwealth Path
			}
			else_if = {
				limit = { colonial_assimilate_months_value > 24 }
				trigger_event = { id = decolonization_events.205 }   # Quiet Assimilation
			}
			else = {
				trigger_event = { id = decolonization_events.200 }   # generic Empire Endures
			}
			# Chosen-release count (decol_record_chosen_release) is read only by
			# the voluntary-fail event .206; drop it on every other ending.
			remove_variable = decol_chosen_releases

			# Clean up persistent state from this JE run. Path-tracking variables,
			# country-scope policy modifiers, civil-rights movement modifiers, and
			# subject-side LD modifiers all need explicit removal -- they are
			# attached to entities that survive the JE's end and would otherwise
			# stick around indefinitely.
			colonial_empire_je_cleanup_effect = yes
			# No success event forms a country, so the programme-month counters
			# can go now (the failure outcome keeps them for .201 / .204).
			colonial_empire_clear_path_months = yes

			# Permanent re-entry block. Any successful completion (bar 100 hold or
			# IF Act capstone) commits the player out of further je_colonial_empire
			# runs even if colonies are reacquired later.
			set_variable = { name = colonial_empire_completed value = yes }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_colonial_empire_fail_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = je_colonial_empire_fail_tt
					scope:journal_entry = {
						"scripted_bar_progress(colonial_stability_bar)" <= 0
					}
				}

				custom_tooltip = {
					text = je_colonial_empire_voluntary_fail_tt
					# The colony test reads the monthly separation cache
					# (colonial_geography_effects.txt). A save from before the
					# cache holds none until our first pass, and would read "no
					# colonies" here and end the entry.
					has_variable = ce_separation_cached
					NOT = {
						OR = {
							any_scope_state = { is_overseas_colonial_state = yes }
							country_has_qualifying_colonial_subject = yes
						}
					}
					scope:journal_entry = {
						"scripted_bar_progress(colonial_stability_bar)" > 0
					}
				}
			}
		}
		# Path-dependent failure: did the empire bleed out (garrison-only),
		# negotiate its way out (invest + assimilate), or voluntarily decolonize?
		effect = {
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(colonial_stability_bar)" > 0
					}
				}
				# .206 reads and then clears var:decol_chosen_releases itself.
				trigger_event = { id = decolonization_events.206 }   # Voluntary Decolonization
				# .206 forms no country. .204 and .201 below start the collapse's
				# wave of declarations (decol_wave_start), whose successors'
				# legacy (apply_decolonization_path) reads these counters; the
				# wave clears them after its last answer (decol_wave_end).
				colonial_empire_clear_path_months = yes
				# Not a cooldown: the owner's rule is that a voluntary end sets
				# none, and possible does not read this. It is a guard against a
				# flickering colony test. The re-entry pass (ce_reentry_due) waits
				# for it, so a country whose one colony passes the test on two
				# monthly passes and fails it a few days later (this fail is
				# checked every 4 days) does not loop through .206, a fresh entry
				# and .206 again every two months.
				set_variable = { name = ce_reentry_voluntary_hold value = yes days = 365 }
			}
			else_if = {
				limit = {
					NAND = {
						colonial_garrison_months_value >= colonial_invest_months_value
						colonial_garrison_months_value >= colonial_assimilate_months_value
					}
					OR = {
						colonial_invest_months_value > 24
						colonial_assimilate_months_value > 24
					}
				}
				trigger_event = { id = decolonization_events.204 }   # Negotiated Withdrawal
				remove_variable = decol_chosen_releases   # read only by .206
				# Bar reached zero -- 10-year cooldown blocks re-entry even if
				# colonies are reacquired. Voluntary fail (bar>0) above does not
				# set this flag, so peaceful decolonizers can re-engage freely:
				# the re-entry pass (ce_reentry_monthly) adds the entry back a
				# month after the country is due it, here once the ten years
				# have run out.
				set_variable = { name = colonial_empire_collapsed_recently value = yes days = 3650 }
			}
			else = {
				trigger_event = { id = decolonization_events.201 }   # The Empire Crumbles (bloody)
				remove_variable = decol_chosen_releases   # read only by .206
				set_variable = { name = colonial_empire_collapsed_recently value = yes days = 3650 }
			}

			# Subject-holders: instead of forced release, every qualifying
			# colonial subject takes a strong LD spike + relations hit. Subjects
			# don't auto-secede but become much harder to hold; the player has to
			# negotiate releases or weather the LD-driven independence pressure.
			# Direct-state release (where applicable) happens through the wave of
			# declarations .204 / .201 start (decol_wave_start, .77 -> .70).
			# Removed first: the entry can now run again after a collapse (the
			# re-entry pass), a second collapse comes about 13 years after the
			# first (ten years, up to two months, some 34 months at the drift
			# cap), and a same-name add_modifier stacks, so a subject through two
			# collapses would carry the 20-year aftershock twice.
			every_subject_or_below = {
				limit = { is_qualifying_colonial_subject = yes }
				remove_modifier_if_exists_effect = { MODIFIER = colonial_collapse_subject_unrest_modifier }
				add_modifier = {
					name = colonial_collapse_subject_unrest_modifier
					days = very_long_modifier_time
					is_decaying = no
				}
				change_relations = { country = ROOT value = -50 }
			}

			# Clean up persistent JE state. The collapse_subject_unrest_modifier
			# applied just above is intentionally preserved (failure consequence)
			# -- the cleanup helper removes the bar-band LD modifiers but not the
			# failure aftershock. It leaves the programme-month counters, which
			# the wave .201 / .204 start still needs (decol_wave_end clears them).
			colonial_empire_je_cleanup_effect = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			# --- Programmes from a save made before the levels ---
			# A save from before the programme levels (2026-10) holds the
			# modifier of each programme it ran but no level; it becomes the top
			# level (decol_migrate_programmes, decolonization.txt). Every
			# programme test below reads the level (colonial_*_level_value),
			# which is visible at once where a modifier added in this block
			# would not be. A revolution's winner inherits the levels and gets
			# the modifiers back from the one refresh further down (and at once
			# from te_civil_war_on_won, through decol_sync_programmes).
			decol_migrate_programmes = yes

			# --- Path-dependence tracking ---
			# Count months each programme runs, weighted by its level: a month
			# at the top level counts 1 (as a month of the old on/off programme
			# did), a month at level 1 a third. Read by the success and failure effects
			# to choose the appropriate resolution event, so a cheap low level
			# cannot steer the ending as fast as the full programme.
			if = {
				limit = { NOT = { has_variable = colonial_invest_months } }
				set_variable = { name = colonial_invest_months value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = colonial_garrison_months } }
				set_variable = { name = colonial_garrison_months value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = colonial_assimilate_months } }
				set_variable = { name = colonial_assimilate_months value = 0 }
			}
			if = {
				limit = { NOT = { has_variable = colonial_solidified_months } }
				set_variable = { name = colonial_solidified_months value = 0 }
			}
			# JE total duration counter — read by long-duration home-front events
			# (e.g. "The Universities Speak" at >10 years).
			if = {
				limit = { NOT = { has_variable = colonial_je_total_months } }
				set_variable = { name = colonial_je_total_months value = 0 }
			}
			change_variable = { name = colonial_je_total_months add = 1 }
			if = {
				limit = { colonial_invest_level_value >= 1 }
				change_variable = { name = colonial_invest_months add = colonial_invest_strength }
			}
			if = {
				limit = { colonial_garrison_level_value >= 1 }
				change_variable = { name = colonial_garrison_months add = colonial_garrison_strength }
			}
			if = {
				limit = { colonial_assim_level_value >= 1 }
				change_variable = { name = colonial_assimilate_months add = colonial_assim_strength }
			}

			# --- Solidified-band sustained-time counter ---
			# Increments while bar is in Solidified band; resets if it drops.
			# The Imperial Federation Act capstone requires 36 sustained months.
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" >= 90 }
				}
				change_variable = { name = colonial_solidified_months add = 1 }
			}
			else = {
				set_variable = { name = colonial_solidified_months value = 0 }
			}

			# --- Bar-at-100 sustained-time counter ---
			# Increments while bar is at the cap; resets if it drops. The basic
			# completion path requires 60 months at 100 (longer than the IF
			# Act's 36-mo ≥90 gate) so reaching the cap isn't an instant win
			# and leaves the IF Act decision as the meaningfully faster route.
			if = {
				limit = { NOT = { has_variable = colonial_at_100_months } }
				set_variable = { name = colonial_at_100_months value = 0 }
			}
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" >= 100 }
				}
				change_variable = { name = colonial_at_100_months add = 1 }
			}
			else = {
				set_variable = { name = colonial_at_100_months value = 0 }
			}

			# --- Programme costs, the AI's review, the modifiers ---
			# The colonies' GDP that Investment's weekly cost is a share of, cached
			# once a month (colonial_programmes_refresh_cost_base). Then an AI
			# reviews its three programmes' levels against the bar and its budget,
			# reading the widget's projected change (its three iterating groups
			# from last month's snapshot). Then the one refresh re-applies each
			# programme's modifier at its level and Investment's expense line at
			# this week's cost, so the line follows the colonies' GDP; the steps
			# refresh it too, at once.
			colonial_programmes_refresh_cost_base = yes
			if = {
				limit = { is_ai = yes }
				colonial_ai_review_programmes = yes
			}
			colonial_programmes_refresh_modifiers = yes

			# The pressuring-great-powers leaderboard used to be built here, into
			# seven colonial_condemner_rank_N slot variables read by the
			# status_desc. It is gone: the widget's International Pressure
			# section is a table of script containers built in
			# colonial_empire_refresh_display (colonial_empire_refresh_gp_rows,
			# every empire's), each reaching its country through a stored capital
			# rather than the `Var('x').GetCountry.GetName` chain that
			# gui_modding_guide gotcha #11 records as rendering blank in this mod.

			# --- State-side assimilation disruption reapplication ---
			# While Cultural Assimilation runs, every overseas colonial state,
			# and every unintegrated colonial state of a qualifying subject,
			# takes a recurring goods-output / qualifications hit at the
			# programme's level (a third of it a level). Forced cultural
			# conversion stalls the local economy. 2-month duration; reapplied
			# each pulse so it sticks while the programme runs, runs out when it
			# ends.
			if = {
				limit = { colonial_assim_level_value >= 3 }
				colonial_assim_disrupt_colonies = { MULT = 1 }
			}
			else_if = {
				limit = { colonial_assim_level_value >= 2 }
				colonial_assim_disrupt_colonies = { MULT = 0.667 }
			}
			else_if = {
				limit = { colonial_assim_level_value >= 1 }
				colonial_assim_disrupt_colonies = { MULT = 0.333 }
			}

			# Phase modifiers based on current bar value (five status bands)
			je:je_colonial_empire = { remove_modifier = colonial_empire_solidified_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_stable_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_strained_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_under_pressure_modifier }
			je:je_colonial_empire = { remove_modifier = colonial_empire_crumbling_modifier }

			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 20 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_crumbling_modifier } }
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 40 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_under_pressure_modifier } }
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 65 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_strained_modifier } }
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 90 }
				}
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_stable_modifier } }
			}
			else = {
				je:je_colonial_empire = { add_modifier = { name = colonial_empire_solidified_modifier } }
			}

			# --- Subject-side LD pressure ---
			# Bar status pushes Liberty Desire onto qualifying colonial
			# subjects. Strained band -> mild LD bump; Crumbling -> strong;
			# Collapsing -> severe. Stable / Solidified -> no LD pressure.
			# Pressure is one-way: subjects' LD does not feed back into the
			# bar (the bar is driven by GP condemnation, governance, and
			# overreach terms in the bar's monthly_progress script).
			every_subject_or_below = {
				limit = { is_qualifying_colonial_subject = yes }
				remove_modifier = colonial_overlord_strained_subject_ld_modifier
				remove_modifier = colonial_overlord_crumbling_subject_ld_modifier
				remove_modifier = colonial_overlord_collapsing_subject_ld_modifier
			}
			if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 20 }
				}
				every_subject_or_below = {
					limit = { is_qualifying_colonial_subject = yes }
					add_modifier = { name = colonial_overlord_collapsing_subject_ld_modifier }
				}
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 40 }
				}
				every_subject_or_below = {
					limit = { is_qualifying_colonial_subject = yes }
					add_modifier = { name = colonial_overlord_crumbling_subject_ld_modifier }
				}
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 65 }
				}
				every_subject_or_below = {
					limit = { is_qualifying_colonial_subject = yes }
					add_modifier = { name = colonial_overlord_strained_subject_ld_modifier }
				}
			}
			# Stable (>=65) and Solidified (>=90): no LD pressure (modifiers
			# already removed at the top of this block).

			# Apply radicalism to civil rights movement specifically (not all movements)
			every_political_movement = {
				limit = { is_political_movement_type = movement_civil_rights }
				remove_modifier = colonial_pressure_movement_radicalism

				}
				every_political_movement = {
					limit = { is_political_movement_type = movement_civil_rights }
					add_modifier = { name = colonial_crumbling_movement_radicalism }
				}
			}
			else_if = {
				limit = {
					scope:journal_entry = { "scripted_bar_progress(colonial_stability_bar)" < 40 }
				}
				every_political_movement = {
					limit = { is_political_movement_type = movement_civil_rights }
					add_modifier = { name = colonial_pressure_movement_radicalism }
				}
			}

			# --- Widget display state -------------------------------------
			# Last, so the snapshot reflects this month's settled phase
			# modifiers and this month's bar value.
			colonial_empire_refresh_display = yes

			# Sample the month for the history charts. Runs after the refresh
			# so the stability bucket and the drift total are this month's.
			te_history_record_colonial_samples = yes
		}
