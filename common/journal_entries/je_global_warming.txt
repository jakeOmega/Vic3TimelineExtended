je_global_warming = {
	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		has_game_rule = global_warming_enabled
	}

	possible = {
		temperature_anomaly_display >= 0.1
	}

	# One line. The six-tier ladder that used to be six triggered_desc blocks
	# here now lives in gw_severity_text
	# (common/customizable_localization/global_warming_custom_loc.txt), because
	# the widget's conditions table needs the same tier and two copies of a
	# threshold table drift. The six sentence keys themselves are unchanged.
	status_desc = je_global_warming_status

	# Shown in the journal list and near the foot of the panel, so the current
	# reading is legible without opening the entry.
	progress_desc = je_global_warming_progress

	# Progress bar: current temperature vs the 4 deg C critical threshold.
	#
	# DO NOT "SIMPLIFY" `4 - temperature_anomaly_display` TO `4`. Per vanilla's
	# common/journal_entries/journal_entries.md, current_value and goal_add_value
	# are evaluated once, AT ACTIVATION, and added together to form the goal.
	# Activation happens when `possible` first passes, i.e. at ~0.1 C, so the goal
	# is 0.1 + (4 - 0.1) = 4.0 and is frozen there for the rest of the game. A
	# flat `value = 4` would instead make the goal 4.1, or worse on a save that
	# activated the entry late.
	current_value = {
		value = temperature_anomaly_display
	}

	goal_add_value = {
		value = {
			value = 4
			subtract = temperature_anomaly_display
		}
	}

	progressbar = yes

	# ---- PLAYER-FACING SURFACE: the climate dashboard ---------------------
	# The sixteen scripted buttons below stay declared because their `ai_chance`
	# is the AI's only route into this system, but each carries `is_ai = yes` in
	# its `visible`, so a human sees these three widgets instead of the grid.
	# Every action the grid offered has a control here — see the mapping in
	# docs/systems/journal_entry_systems.md.
	widget = {
		gui = "gui/journal_entry_widgets/global_warming_widget.gui"
		name = "widget_je_gw_conditions"
		container = "custom_widget_container_1"
	}

	widget = {
		gui = "gui/journal_entry_widgets/global_warming_widget.gui"
		name = "widget_je_gw_policies"
		container = "custom_widget_container_2"
	}

	widget = {
		gui = "gui/journal_entry_widgets/global_warming_widget.gui"
		name = "widget_je_gw_history"
		container = "custom_widget_container_3"
	}

	scripted_button = gw_carbon_tax_button
	scripted_button = gw_remove_carbon_tax_button
	scripted_button = gw_renewable_investment_button
	scripted_button = gw_remove_renewable_investment_button
	scripted_button = gw_climate_adaptation_button
	scripted_button = gw_remove_climate_adaptation_button
	scripted_button = gw_emission_standards_button
	scripted_button = gw_remove_emission_standards_button
	scripted_button = gw_reforestation_button
	scripted_button = gw_remove_reforestation_button
	scripted_button = gw_public_transit_button
	scripted_button = gw_remove_public_transit_button
	scripted_button = gw_fossil_fuel_divestment_button
	scripted_button = gw_remove_fossil_fuel_divestment_button
	scripted_button = gw_green_building_codes_button
	scripted_button = gw_remove_green_building_codes_button

	immediate = {
	}

	complete = {
		always = no
	}

	on_monthly_pulse = {
		effect = {
			# ---- Update warming modifier: temperature anomaly used directly as multiplier ----
			if = {
				limit = { temperature_anomaly_display >= 0.01 }
				je:je_global_warming = { remove_modifier = global_warming }
				je:je_global_warming = { add_modifier = { name = global_warming multiplier = temperature_anomaly_display } }
			}
			else_if = {
				limit = { je:je_global_warming = { has_modifier = global_warming } }
				je:je_global_warming = { remove_modifier = global_warming }
			}

			# History samples for the widget's two charts. Gated on the game rule
			# and on te_history_country_is_tracked inside the helper, and wrapped
			# in hidden_effect there, so this costs nothing for an untracked
			# country and never writes state during a tooltip render.
			te_history_record_global_warming_samples = yes
		}
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = no
}
