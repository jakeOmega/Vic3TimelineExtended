je_global_warming = {
	icon = "gfx/interface/icons/event_icons/je_global_warming.dds"

	group = je_group_internal_affairs

	# Activation needs both of these (vanilla journal_entries.md). The entry
	# opens at 0.1 C of warming under the Global Warming game rule, or for any
	# country holding a restrictive Resource Transition law (#660), with the
	# rule on or off: its Fossil Transition section carries the retirement
	# programmes. With the rule off the entry shows that section alone
	# (gw_rule_enabled_sgui), and nothing climate runs: the temperature stays 0
	# (global_warming_events_on_action), so no warming modifier, threshold
	# event or policy follows. gw_entry_unlocked repeats `possible` for the
	# Market panel's Climate tab, word for word.
	is_shown_when_inactive = {
		OR = {
			has_game_rule = global_warming_enabled
			rt_restrictive_transition_law = yes
		}
	}

	possible = {
		OR = {
			temperature_anomaly_display >= 0.1
			rt_restrictive_transition_law = yes
		}
	}

	# No status line while the entry is active: the overview above it shows the
	# tier (its icon's tooltip is the tier's sentence, gw_severity_text), the
	# temperature and where it is heading, and status text that repeats the
	# overview goes (the owner's rule for every panel, 2026-09-29). An empty
	# status string hides vanilla's status box (StringIsEmpty in
	# gui/journal_entry.gui). The greyed entry, before the world has warmed
	# 0.1 C, keeps one line saying when it opens: its widgets draw nothing
	# then. `has_journal_entry` is true only for an active entry.
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_global_warming_status_inactive
				trigger = {
					NOT = { has_journal_entry = je_global_warming }
				}
			}
			triggered_desc = {
				desc = je_global_warming_status_none
				trigger = { always = yes }
			}
		}
	}

	# Shown in the journal list and near the foot of the panel, so the current
	# reading is legible without opening the entry.
	progress_desc = je_global_warming_progress

	# Progress bar: current temperature vs the 4 deg C critical threshold.
	#
	# DO NOT "SIMPLIFY" `4 - temperature_anomaly_display` TO `4`. Per vanilla's
	# common/journal_entries/journal_entries.md, current_value and goal_add_value
	# are evaluated once, AT ACTIVATION, and added together to form the goal.
	# Activation happens when `possible` first passes, i.e. at ~0.1 C, so the goal
	# is 0.1 + (4 - 0.1) = 4.0 and is frozen there for the rest of the game. A
	# flat `value = 4` would instead make the goal 4.1, or worse on a save that
	# activated the entry late.
	current_value = {
		value = temperature_anomaly_display
	}

	goal_add_value = {
		value = {
			value = 4
			subtract = temperature_anomaly_display
		}
	}

	progressbar = yes

	# ---- PLAYER-FACING SURFACE: the climate panels -------------------------
	# The twenty climate scripted buttons below stay declared because their `ai_chance`
	# is the AI's only route into this system, but each carries `is_ai = yes` in
	# its `visible`, so a human sees these three widgets instead of the grid.
	# Every action the grid offered has a control here — see the mapping in
	# docs/systems/journal_entry_systems.md. The order of the sections lives in
	# the widget file's composers (te_gw_status_sections,
	# te_gw_reference_sections): the overview above the status line, the live
	# sections above the (hidden) grid, the charts and "How Global Warming
	# Works" below it.
	widget = {
		gui = "gui/journal_entry_widgets/global_warming_widget.gui"
		name = "widget_je_gw_overview"
		container = "custom_widget_container_1"
	}

	widget = {
		gui = "gui/journal_entry_widgets/global_warming_widget.gui"
		name = "widget_je_gw_status"
		container = "custom_widget_container_2"
	}

	widget = {
		gui = "gui/journal_entry_widgets/global_warming_widget.gui"
		name = "widget_je_gw_reference"
		container = "custom_widget_container_3"
	}

	# MARKER, not content: the overview draws the temperature bar, so the
	# journal panel drops its own bar near the foot of the entry. The mod's
	# gui/journal_entry.gui hides the bar block of any entry with a widget in
	# custom_widget_container_7; with #582 that covers the goal bar
	# (`progressbar = yes`) as well as scripted bars, of which this entry has
	# none. `progressbar = yes` itself is unchanged. See je_banking.txt for the
	# mechanism.
	widget = {
		gui = "gui/journal_entry_widgets/te_je_bars_on_top_marker.gui"
		name = "widget_te_je_bars_on_top_marker"
		container = "custom_widget_container_7"
	}

	scripted_button = gw_fossil_fuel_tariffs_button
	scripted_button = gw_remove_fossil_fuel_tariffs_button
	scripted_button = gw_carbon_tax_button
	scripted_button = gw_remove_carbon_tax_button
	scripted_button = gw_renewable_investment_button
	scripted_button = gw_remove_renewable_investment_button
	scripted_button = gw_climate_adaptation_button
	scripted_button = gw_remove_climate_adaptation_button
	scripted_button = gw_emission_standards_button
	scripted_button = gw_remove_emission_standards_button
	scripted_button = gw_reforestation_button
	scripted_button = gw_remove_reforestation_button
	scripted_button = gw_carbon_removal_button
	scripted_button = gw_remove_carbon_removal_button
	scripted_button = gw_public_transit_button
	scripted_button = gw_remove_public_transit_button
	scripted_button = gw_fossil_fuel_divestment_button
	scripted_button = gw_remove_fossil_fuel_divestment_button
	scripted_button = gw_green_building_codes_button
	scripted_button = gw_remove_green_building_codes_button

	# Resource Transition retirement programmes (#660): the AI's route, as the
	# climate buttons above are (is_ai = yes; a human uses the Fossil
	# Transition section). common/scripted_buttons/resource_transition_buttons.txt.
	scripted_button = rt_start_coal_button
	scripted_button = rt_stop_coal_button
	scripted_button = rt_start_oil_button
	scripted_button = rt_stop_oil_button
	scripted_button = rt_start_power_button
	scripted_button = rt_stop_power_button

	immediate = {
	}

	complete = {
		always = no
	}

	on_monthly_pulse = {
		effect = {
			gw_restore_carbon_removal_policy_effect = yes
			# ---- Update warming modifier: temperature anomaly used directly as multiplier ----
			if = {
				limit = { temperature_anomaly_display >= 0.01 }
				je:je_global_warming = { remove_modifier = global_warming }
				je:je_global_warming = { add_modifier = { name = global_warming multiplier = temperature_anomaly_display } }
			}
			else_if = {
				limit = { je:je_global_warming = { has_modifier = global_warming } }
				je:je_global_warming = { remove_modifier = global_warming }
			}

			# History samples for the widget's two charts. Gated on the game rule
			# and on te_history_country_is_tracked inside the helper, and wrapped
			# in hidden_effect there, so this costs nothing for an untracked
			# country and never writes state during a tooltip render.
			te_history_record_global_warming_samples = yes

			# The panel's Top Emitters rows (global_warming_effects.txt):
			# players only, since nobody else looks. Display only.
			if = {
				limit = { is_player = yes }
				gw_top_emitters_refresh = yes
			}
		}
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = no
}
