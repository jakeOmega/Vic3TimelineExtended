# ============================================================================
# COVERT WARFARE — Journal Entry (Command Center)
# ============================================================================
# Two custom widgets carry the interface: widget_je_covert_command_centre shows
# intelligence capacity, slots, the funding ladder and its stepper, detection
# factors and covert defence; widget_je_covert_operations lists each running
# operation. status_desc is left with the one thing that is genuinely prose —
# the intelligence-standing verdict — plus the empty-state pointer.
#
# Operations are toggled via diplomatic actions (pact-based); each running
# operation's state lives in a script container (covert_warfare_effects.txt).
# ============================================================================

je_covert_warfare = {
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	group = je_group_covert_warfare

	# The command centre carries everything status_desc used to say in prose.
	# Its funding stepper calls the very shared trigger and effect the scripted
	# buttons below call, so the two surfaces cannot diverge — which is why
	# those buttons are hidden from humans with `visible = { is_ai = yes }`
	# rather than deleted.
	widget = {
		gui = "gui/journal_entry_widgets/covert_operations_widget.gui"
		name = "widget_je_covert_command_centre"
		container = "custom_widget_container_1"
	}
	widget = {
		gui = "gui/journal_entry_widgets/covert_operations_widget.gui"
		name = "widget_je_covert_operations"
		container = "custom_widget_container_2"
	}

	is_shown_when_inactive = {
		has_game_rule = covert_warfare_enabled
	}

	# Activate only when slots exceed the base (1) + rank bonus (GP +2, Major +1):
	# tech / law / institution has granted at least one extra slot beyond what
	# every country of this rank gets for free.
	possible = {
		OR = {
			AND = {
				country_rank >= rank_value:great_power
				covert_operation_max_slots >= 4
			}
			AND = {
				country_rank = rank_value:major_power
				covert_operation_max_slots >= 3
			}
			AND = {
				NOT = { country_rank >= rank_value:major_power }
				covert_operation_max_slots >= 2
			}
		}
	}

	status_desc = {
		# ---- Intelligence Status Tier (relative to global best) ----
		first_valid = {
			triggered_desc = {
				desc = je_iw_status_fortress
				trigger = { intelligence_capacity_relative >= 80 }
			}
			triggered_desc = {
				desc = je_iw_status_hardened
				trigger = { intelligence_capacity_relative >= 60 }
			}
			triggered_desc = {
				desc = je_iw_status_defended
				trigger = { intelligence_capacity_relative >= 40 }
			}
			triggered_desc = {
				desc = je_iw_status_exposed
				trigger = { intelligence_capacity_relative >= 20 }
			}
			triggered_desc = {
				desc = je_iw_status_vulnerable
				trigger = { always = yes }
			}
		}

		# Everything else that used to live here — the capacity breakdown, the
		# slot counts, the funding level and its cost, the detection factors and
		# the defensive side — is drawn by widget_je_covert_command_centre
		# (custom_widget_container_1, above this text). Per-operation rows come
		# from widget_je_covert_operations (custom_widget_container_2, below).
		# Only the empty state stays, because it tells the player where
		# operations are actually launched.
		triggered_desc = {
			desc = je_iw_no_operations
			trigger = {
				covert_operations_active < 1
			}
		}
	}

	# Funding management buttons (operations launched via diplomatic actions)
	scripted_button = iw_increase_funding_button
	scripted_button = iw_decrease_funding_button

	immediate = {
		# Initialize funding variable if missing
		if = {
			limit = { NOT = { has_variable = iw_funding_level } }
			set_variable = { name = iw_funding_level value = 0 }
		}
	}

	complete = {
		always = no
	}

	on_monthly_pulse = {
		effect = {
			# ---- AI Funding Management ----
			# AI automatically manages funding level based on financial health,
			# strategic need, active operations, and cost awareness.
			# Player uses scripted buttons instead.
			#
			# Cost formula: (priority-weighted ops + 1) × funding_level × covert_operations_cost_scale
			# So more ops = higher cost per funding level. AI should scale funding
			# to match its ambitions and avoid overpaying when it has many ops.
			if = {
				limit = {
					is_player = no
				}
				# In default or severe financial trouble: cut funding
				# But if we have active ops, drop to 1 instead of 0 to avoid
				# auto-cancelling all operations (which wastes the setup investment)
				if = {
					limit = {
						OR = {
							in_default = yes
							has_modifier = declared_bankruptcy
						}
					}
					if = {
						limit = { covert_operations_active >= 1 }
						set_variable = { name = iw_funding_level value = 1 }
					}
					else = {
						set_variable = { name = iw_funding_level value = 0 }
					}
				}
				# Has rivals/enemies and is financially healthy: fund operations
				else_if = {
					limit = {
						country_rank >= rank_value:major_power
						OR = {
							any_country = {
								has_diplomatic_pact = { who = ROOT type = rivalry }
							}
							is_at_war = yes
						}
					}
					# At war: prioritize high funding for wartime ops
					if = {
						limit = { is_at_war = yes }
						# Great powers at war: max funding
						if = {
							limit = { country_rank >= rank_value:great_power }
							set_variable = { name = iw_funding_level value = iw_funding_level_max }
						}
						else = {
							set_variable = { name = iw_funding_level value = 3 }
						}
					}
					# Peacetime with many active ops (3+): keep funding moderate
					# to avoid runaway costs (cost scales with ops × funding)
					else_if = {
						limit = {
							covert_operations_active >= 3
							country_rank >= rank_value:great_power
						}
						set_variable = { name = iw_funding_level value = 3 }
					}
					# Great powers with few ops: high funding (better tradecraft)
					else_if = {
						limit = {
							country_rank >= rank_value:great_power
							NOT = { in_default = yes }
						}
						set_variable = { name = iw_funding_level value = 3 }
					}
					# Major powers or cautious rulers: moderate funding
					else = {
						set_variable = { name = iw_funding_level value = 1 }
					}
				}
				# No rivals and not at war but has active operations:
				# maintain minimal funding to keep ops running
				else_if = {
					limit = { covert_operations_active >= 1 }
					set_variable = { name = iw_funding_level value = 1 }
				}
				# No rivals, no war, no active ops: no need to fund
				else = {
					set_variable = { name = iw_funding_level value = 0 }
				}
			}

			# ---- Track global max IC for relative tier display ----
			ordered_country = {
				limit = { intelligence_capacity_total >= 1 }
				order_by = intelligence_capacity_total
				max = 1
				set_global_variable = { name = global_max_ic value = intelligence_capacity_total }
			}

			# ---- Update intelligence capacity defense modifier ----
			# NOTE: Script values use owner={} internally to access country data
			# from JE scope (owner is a no-op from country scope).
			if = {
				limit = { intelligence_capacity_total >= 1 }
				je:je_covert_warfare = { remove_modifier = intelligence_capacity_defense }
				je:je_covert_warfare = { add_modifier = { name = intelligence_capacity_defense multiplier = intelligence_capacity_modifier_mult } }
			}
			else_if = {
				limit = { je:je_covert_warfare = { has_modifier = intelligence_capacity_defense } }
				je:je_covert_warfare = { remove_modifier = intelligence_capacity_defense }
			}

			# ---- Domestic defense from unused operation slots ----
			if = {
				limit = { covert_operations_available_slots >= 1 }
				je:je_covert_warfare = { remove_modifier = iw_domestic_defense }
				je:je_covert_warfare = { add_modifier = { name = iw_domestic_defense multiplier = covert_operations_available_slots } }
			}
			else_if = {
				limit = { je:je_covert_warfare = { has_modifier = iw_domestic_defense } }
				je:je_covert_warfare = { remove_modifier = iw_domestic_defense }
			}

			# ---- Counterintelligence bonus from funding level ----
			if = {
				limit = { covert_ops_funding_ci_mult >= 1 }
				je:je_covert_warfare = { remove_modifier = iw_funding_defense }
				je:je_covert_warfare = { add_modifier = { name = iw_funding_defense multiplier = covert_ops_funding_ci_mult } }
			}
			else_if = {
				limit = { je:je_covert_warfare = { has_modifier = iw_funding_defense } }
				je:je_covert_warfare = { remove_modifier = iw_funding_defense }
			}

			# ---- Apply operation funding cost ----
			if = {
				limit = {
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
				je:je_covert_warfare = { remove_modifier = covert_operation_funding_cost }
				je:je_covert_warfare = { add_modifier = { name = covert_operation_funding_cost multiplier = covert_operation_cost_mult } }
			}
			else_if = {
				limit = { je:je_covert_warfare = { has_modifier = covert_operation_funding_cost } }
				je:je_covert_warfare = { remove_modifier = covert_operation_funding_cost }
			}

			# ---- Age the "last exposed a foreign operation" record ----
			# covert_op_burn sets iw_last_exposed_age to 0 on the target whose
			# counterintelligence caught an operation. It only ever counts up;
			# the command centre stops showing the line once it passes ten years
			# (covert_last_exposed_known_sgui).
			if = {
				limit = { has_variable = iw_last_exposed_age }
				change_variable = { name = iw_last_exposed_age add = 1 }
			}

			# ---- Decay defender-event cooldown (24-month window) ----
			# covert_warfare.2 sets iw_defender_event_cooldown when it fires;
			# without decay, the defender can only ever see the detection event
			# once per game. iw_defender_event_cooldown_age tracks elapsed
			# months and clears both vars after 24 months.
			if = {
				limit = { has_variable = iw_defender_event_cooldown_age }
				change_variable = { name = iw_defender_event_cooldown_age add = 1 }
				if = {
					limit = { var:iw_defender_event_cooldown_age >= 24 }
					remove_variable = iw_defender_event_cooldown
					remove_variable = iw_defender_event_cooldown_age
				}
			}

			# ---- AI priorities ----
			# Before the sync, so the sync's detection refresh and the upkeep
			# weights see the new levels. A human's priorities come from the
			# operations widget and are never touched here.
			if = {
				limit = { is_player = no }
				covert_ai_manage_priorities = yes
			}

			# ---- Reconcile operation containers with pacts ----
			# Creates containers for pacts that lack one, destroys containers whose
			# pact has ended, and refreshes each operation's detection display.
			# Must run before aging and phase effects, which read the containers.
			covert_ops_sync_all = yes

			# ---- Age operation durations ----
			covert_ops_age_all_durations = yes

			# ---- Apply phase-based operation effects (only if funding > 0) ----
			if = {
				limit = {
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
				covert_ops_apply_all_phase_effects = yes
			}

			# ---- Detection: one roll per operation, at most one burn ----
			# Each container's iw_detect was refreshed by covert_ops_sync_all at
			# the top of this pulse. Funding 0 has no operations to roll (every
			# pact's requirement_to_maintain lapses), so the gate is belt and braces.
			if = {
				limit = {
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
				covert_ops_roll_detection_all = yes
			}
		}
	}

	weight = 800
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = no
	should_update_on_player_command = yes
}
