# ============================================================================
# COVERT WARFARE — Journal Entry (Command Center)
# ============================================================================
# Two custom widgets carry the interface, both from
# gui/journal_entry_widgets/covert_operations_widget.gui, whose types a future
# system tab can compose too (docs/guides/gui_style_guide.md rule 9):
# widget_je_covert_overview (the "?" to the Help page, then standing, funding,
# Tradecraft, slots, capacity) and widget_je_covert_status (Operations, Agent
# Networks, Funding, Detection Risk, Counterintelligence). The entry has no
# status_desc: the intelligence-standing verdict it
# printed is the overview's standing icon's tooltip, and its empty-state
# pointer to the diplomatic actions is the Operations section's.
#
# Operations are toggled via diplomatic actions (pact-based); each running
# operation's state lives in a script container (covert_warfare_effects.txt).
# ============================================================================

je_covert_warfare = {
	icon = "gfx/interface/icons/event_icons/je_covert_warfare.dds"

	group = je_group_covert_warfare

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	# The overview, the live sections and the explanations, in that order
	# (gui/journal_entry_widgets/covert_operations_widget.gui). Funding's stepper
	# calls the very shared trigger and effect the scripted buttons below call,
	# so the two surfaces cannot diverge — which is why those buttons are hidden
	# from humans with `visible = { is_ai = yes }` rather than deleted.
	widget = {
		gui = "gui/journal_entry_widgets/covert_operations_widget.gui"
		name = "widget_je_covert_overview"
		container = "custom_widget_container_1"
	}
	widget = {
		gui = "gui/journal_entry_widgets/covert_operations_widget.gui"
		name = "widget_je_covert_status"
		container = "custom_widget_container_2"
	}

	# 1.15: with the rule off, drop the inactive record for good. A rule can be changed
	# mid-game, from the screen that changes country; an active entry ends through its
	# invalid outcome on the rule.
	disable_evaluation = {
		NOT = { te_covert_warfare_on = yes }
	}
	is_shown_when_inactive = {
		te_covert_warfare_on = yes
	}

	# Activate once the setting's gate is met (covert_warfare_entry_unlocked,
	# covert_warfare_triggers.txt): under the late unlock a named source of an
	# operation slot beyond the base slot and the rank bonus, a technology or the
	# Ministry of Intelligence and Security's law (covert_warfare_slot_source,
	# which says why each source is named rather than counted); under the early
	# unlock also Central Archives, with the budget capped at Professional until
	# the late gate (covert_agency_limited). One trigger, so the Military panel's
	# Covert tab shows the same checklist as one line.
	possible = {
		covert_warfare_entry_unlocked = yes
	}

	# Funding management buttons (operations launched via diplomatic actions)
	scripted_button = iw_increase_funding_button
	scripted_button = iw_decrease_funding_button

	# ---- The agency's five standing modifiers ----
	# Held on this entry while it is active (1.15's scaling modifiers): the
	# engine re-reads each value from the owning country about every four days,
	# so nothing adds, removes or refreshes them. Each value is 0 wherever the
	# old monthly pulse removed its modifier. has_modifier can't see them: test
	# the value's inputs instead.
	# Defense from intelligence capacity (off below 1 total capacity).
	scaling_modifiers_while_active = {
		modifier = intelligence_capacity_defense
		value = { value = intelligence_capacity_modifier_mult }
	}
	# Each unused operation slot redirected to domestic intelligence.
	scaling_modifiers_while_active = {
		modifier = iw_domestic_defense
		value = { value = covert_operations_available_slots }
	}
	# Counterintelligence from funding: funding level − 1, from level 2.
	scaling_modifiers_while_active = {
		modifier = iw_funding_defense
		value = { value = covert_ops_funding_ci_mult }
	}
	# The weekly expense, GDP-scaled; 0 at funding 0.
	scaling_modifiers_while_active = {
		modifier = covert_operation_funding_cost
		value = { value = covert_operation_cost_mult }
	}
	# Tradecraft: one stack per tier above Untested.
	scaling_modifiers_while_active = {
		modifier = iw_tradecraft_bonus
		value = { value = covert_tradecraft_bonus_mult }
	}

	immediate = {
		# Initialize funding variable if missing
		if = {
			limit = { NOT = { has_variable = iw_funding_level } }
			set_variable = { name = iw_funding_level value = 0 }
		}
		# Agency experience starts at 0 (slice 5). Also seeded by the monthly
		# pass, for saves whose entry was already active.
		covert_tradecraft_init = yes
		# Opened under the early unlock, before the late gate: marked, so that
		# a switch to the late unlock closes the entry until the late gate (the
		# invalid outcome below). Cleared by the pulse once the late gate is
		# met. Set only while the condition holds, so a re-run (a revolution's
		# inherited entry) writes the same.
		if = {
			limit = { covert_agency_limited = yes }
			set_variable = iw_opened_early
		}
	}

	on_monthly_pulse = {
		effect = {
			te_scaling_inherited_selfcheck_rebel_entry = yes		# TEMPORARY (#1404): te_scaling_selfcheck_effects.txt
			# ---- AI Funding Management ----
			# AI automatically manages funding level based on financial health,
			# strategic need, active operations, and cost awareness.
			# Player uses scripted buttons instead.
			#
			# Cost formula: (priority-weighted ops + 1) × funding_level × covert_operations_cost_scale
			# So more ops = higher cost per funding level. AI should scale funding
			# to match its ambitions and avoid overpaying when it has many ops.
			if = {
				limit = {
					is_player = no
				}
				# In default or severe financial trouble: cut funding
				# But if we have active ops, drop to 1 instead of 0 to avoid
				# auto-cancelling all operations (which wastes the setup investment)
				if = {
					limit = {
						OR = {
							in_default = yes
							has_modifier = declared_bankruptcy
						}
					}
					if = {
						limit = { covert_operations_active >= 1 }
						set_variable = { name = iw_funding_level value = 1 }
					}
					else = {
						set_variable = { name = iw_funding_level value = 0 }
					}
				}
				# Borrowing toward bankruptcy: step down to the floor before the
				# default, keeping running operations (and a war's) at 1.
				else_if = {
					limit = {
						taking_loans = yes
						weeks_until_bankruptcy < iw_ai_funding_bankruptcy_weeks
					}
					if = {
						limit = {
							OR = {
								covert_operations_active >= 1
								is_at_war = yes
							}
						}
						set_variable = { name = iw_funding_level value = 1 }
					}
					else = {
						set_variable = { name = iw_funding_level value = 0 }
					}
				}
				# Has rivals/enemies and is financially healthy: fund operations
				else_if = {
					limit = {
						country_rank >= rank_value:major_power
						OR = {
							any_country = {
								has_diplomatic_pact = { who = ROOT type = rivalry }
							}
							is_at_war = yes
						}
					}
					# At war: prioritize high funding for wartime ops
					if = {
						limit = { is_at_war = yes }
						# Great powers at war: max funding
						if = {
							limit = { country_rank >= rank_value:great_power }
							set_variable = { name = iw_funding_level value = iw_funding_level_max }
						}
						else = {
							set_variable = { name = iw_funding_level value = 3 }
						}
					}
					# Peacetime with many active ops (3+): keep funding moderate
					# to avoid runaway costs (cost scales with ops × funding)
					else_if = {
						limit = {
							covert_operations_active >= 3
							country_rank >= rank_value:great_power
						}
						set_variable = { name = iw_funding_level value = 3 }
					}
					# Great powers with few ops: high funding (better tradecraft)
					else_if = {
						limit = {
							country_rank >= rank_value:great_power
							NOT = { in_default = yes }
						}
						set_variable = { name = iw_funding_level value = 3 }
					}
					# Major powers or cautious rulers: moderate funding
					else = {
						set_variable = { name = iw_funding_level value = 1 }
					}
				}
				# No rivals and not at war but has active operations:
				# maintain minimal funding to keep ops running
				else_if = {
					limit = { covert_operations_active >= 1 }
					set_variable = { name = iw_funding_level value = 1 }
				}
				# No rivals, no war, no active ops: no need to fund
				else = {
					set_variable = { name = iw_funding_level value = 0 }
				}
			}

			# ---- The early unlock's budget cap ----
			# Before the late gate an early agency's funding stops at
			# Professional (covert_agency_funding_cap): the AI's levels above
			# are cut to it, and so is a level set before a switch to the early
			# unlock. The steppers stop there on their own
			# (covert_possible_increase_funding).
			if = {
				limit = {
					covert_agency_limited = yes
					has_variable = iw_funding_level
					var:iw_funding_level > covert_agency_funding_cap
				}
				set_variable = { name = iw_funding_level value = covert_agency_funding_cap }
			}
			if = {
				limit = {
					has_variable = iw_opened_early
					covert_warfare_slot_source = yes
				}
				remove_variable = iw_opened_early
			}

			# ---- The world's best intelligence capacity, for the panel ----
			# Display only: intelligence_capacity_relative turns it into the
			# standing tier, the capacity bar and the overview tooltip. It is a
			# pass over every country, the same for every holder, so the first
			# holder to pulse takes it and the lock spares the rest for 25
			# days: one census each 1st, when the entries pulse together. Every
			# holder took it before, in the same frame, the last one's figure
			# kept: about 250 passes a month, the pulse's largest part in a 1931
			# game (docs/audits/late_game_performance.md).
			if = {
				limit = { NOT = { has_global_variable = iw_max_ic_lock } }
				set_global_variable = { name = iw_max_ic_lock value = yes days = 25 }
				ordered_country = {
					limit = { intelligence_capacity_total >= 1 }
					order_by = intelligence_capacity_total
					max = 1
					set_global_variable = { name = global_max_ic value = intelligence_capacity_total }
				}
			}

			# The agency's standing modifiers are the entry's
			# scaling_modifiers_while_active (above): nothing to refresh here.

			# The exposure record's age (iw_last_exposed_age) and the defender
			# event's cooldown are aged by covert_warfare_monthly_country
			# (covert_warfare_on_actions.txt): the countries operations are run
			# against need not hold this entry.

			# ---- AI priorities ----
			# Before the sync, so the sync's detection refresh and the upkeep
			# weights see the new levels. A human's priorities come from the
			# operations widget and are never touched here.
			if = {
				limit = { is_player = no }
				covert_ai_manage_priorities = yes
			}

			# ---- Reconcile operation containers with pacts ----
			# Creates containers for pacts that lack one, destroys containers whose
			# pact has ended, and refreshes each operation's detection display.
			# Must run before aging and phase effects, which read the containers.
			covert_ops_sync_all = yes

			# ---- Per-target networks (slice 4) ----
			# Once a month, here and only here: covert_ops_sync_all also runs on
			# every funding / priority click, so the tick cannot live inside it.
			# Ensures a network per target, grows or decays each, reaps the dead.
			covert_nets_sync = yes

			# ---- Operations we watch (the target's side) ----
			# Straight after the network pass, which has just counted them:
			# reap records whose operation is gone, age the rest, retry an
			# exposure that waited, and let an AI expose by chance.
			covert_watch_monthly = yes

			# ---- Age operation durations ----
			covert_ops_age_all_durations = yes

			# ---- Our caught operations (the attacker's side) ----
			# Before the detection roll: frees an exposure lock or a pending
			# decision whose event never ran, and makes an operation whose
			# target lost its record catchable again. Runs at any funding.
			covert_watch_attacker_monthly = yes

			# ---- Tradecraft (slice 5) ----
			# Once a month, here and only here (the steppers re-run the sync).
			# After aging, so an operation that just reached establishing
			# counts this month. Runs at any funding: idle decay needs it.
			covert_tradecraft_monthly = yes

			# ---- Apply phase-based operation effects (only if funding > 0) ----
			if = {
				limit = {
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
				covert_ops_apply_all_phase_effects = yes
			}

			# ---- Detection: one roll per operation, at most one burn ----
			# Each container's iw_detect was refreshed by covert_ops_sync_all at
			# the top of this pulse. Funding 0 has no operations to roll (every
			# pact's requirement_to_maintain lapses), so the gate is belt and braces.
			if = {
				limit = {
					has_variable = iw_funding_level
					var:iw_funding_level >= 1
				}
				covert_ops_roll_detection_all = yes
			}
		}
	}

	# The rule switched off mid-game, from the screen that changes country
	# (2026-11-08). Nothing else ends this entry, so its declared modifiers, its
	# pulse and every operation ran on. It ends on its next update and takes the
	# modifiers along; covert_stop_for_rule ends the operations and the agent
	# networks, which other systems read (Space Race intelligence, loose
	# material, the map).
	# Also the early unlock switched to the late one while an entry opened
	# early (iw_opened_early) is still short of the late gate: the same
	# teardown, and covert_warfare_monthly_country adds the entry back once the
	# late gate is met (iw_reopen_at_gate).
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_covert_warfare_invalid_conclusion_desc
		trigger = {
			OR = {
				NOT = { te_covert_warfare_on = yes }
				AND = {
					has_variable = iw_opened_early
					NOT = { te_covert_warfare_early = yes }
					NOT = { covert_warfare_slot_source = yes }
				}
			}
		}
		effect = {
			covert_stop_for_rule = yes
			if = {
				limit = { te_covert_warfare_on = yes }
				set_variable = iw_reopen_at_gate
			}
			if = {
				limit = { has_variable = iw_opened_early }
				remove_variable = iw_opened_early
			}
		}
	}

	weight = 800
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = no
	should_update_on_player_command = yes

	# The map (vanilla 1.15 journal entry map mode; docs/systems/mod_systems/journal_map_modes.md): the viewer's covert
	# network in each country. Root: the state being painted, every state in the world. The Map Room draws the same map as
	# its selector 20, from the same triggers in the same order (common/scripted_triggers/te_journal_map_triggers.txt; the
	# parity test in test_journal_map_modes.py).
	map_color = {
		if = {
			limit = { te_jmap_owner_is_viewer = yes }
			value = "te_jmap_net_you"
		}
		else_if = {
			limit = { te_jmap_net_some = yes }
			lerp = {
				min_color = "te_jmap_net_low"
				max_color = "te_jmap_net_high"
				factor = te_jmap_net_factor
			}
		}
		else = {
			value = "te_jmap_none"
		}
	}

	map_tooltip = {
		if = {
			limit = { te_jmap_owner_is_viewer = yes }
			custom_tooltip_no_bullet = te_jmap_tt_net_you
		}
		else_if = {
			limit = { te_jmap_net_some = yes }
			custom_tooltip_no_bullet = te_jmap_tt_net
		}
		else = {
			custom_tooltip_no_bullet = te_jmap_tt_net_none
		}
	}
}
