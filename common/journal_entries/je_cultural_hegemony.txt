# ============================================================================
# CULTURAL HEGEMONY — Journal Entry (Display Only)
# ============================================================================
# This JE is the player's window into the Cultural Hegemony system.
# It shows cultural pull breakdown, global leaderboard, and tier status.
# All computation runs in on_actions (cultural_hegemony_on_actions.txt) via
# scripted effects (cultural_hegemony_effects.txt) from game start. The entry
# declares the system's standing modifiers (scaling_modifiers_while_active,
# below), each from a country variable those effects or its buttons write.
# The JE activates once any country has researched Mass Media and this
# country has Romanticism (`possible`), so scores are stable when first shown.
# ============================================================================

je_cultural_hegemony = {
	icon = "gfx/interface/icons/event_icons/je_cultural_hegemony.dds"

	group = je_group_soft_power

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	# 1.15: with the rule off, drop the inactive record for good (a rule never changes mid-game).
	disable_evaluation = {
		NOT = { has_game_rule = cultural_hegemony_enabled }
	}
	is_shown_when_inactive = {
		has_game_rule = cultural_hegemony_enabled
	}

	possible = {
		ch_entry_unlocked = yes
	}

	scripted_button = ch_increase_program_funding_button
	scripted_button = ch_decrease_program_funding_button
	scripted_button = ch_world_exposition_button
	scripted_button = ch_disable_world_exposition_button
	scripted_button = ch_cultural_institutes_button
	scripted_button = ch_disable_cultural_institutes_button
	scripted_button = ch_global_media_campaign_button
	scripted_button = ch_disable_global_media_campaign_button
	scripted_button = ch_cultural_protectionism_button
	scripted_button = ch_disable_cultural_protectionism_button

	# The human-facing surface. Every scripted button above carries `is_ai = yes`,
	# so the vanilla button grid renders nothing for a player; these panels
	# replace it. Their controls call ch_policy_sgui, which delegates to the very
	# triggers and effects the buttons use. Three named roots, in the order
	# cultural_hegemony_widget.gui composes for both hosts: the overview above the
	# status text, the live sections below it, and the history and "How Cultural
	# Hegemony Works" below the (empty for a human) button grid.
	widget = {
		gui = "gui/journal_entry_widgets/cultural_hegemony_widget.gui"
		name = "widget_je_ch_overview"
		container = "custom_widget_container_1"
	}
	widget = {
		gui = "gui/journal_entry_widgets/cultural_hegemony_widget.gui"
		name = "widget_je_ch_status"
		container = "custom_widget_container_2"
	}
	widget = {
		gui = "gui/journal_entry_widgets/cultural_hegemony_widget.gui"
		name = "widget_je_ch_reference"
		container = "custom_widget_container_3"
	}

	status_desc = {
		# No status text while the entry is active: the panels above and below
		# show everything it said (the overview's tier and share, the programmes'
		# funding row), and status text that repeats the overview goes (owner,
		# 2026-09-29). The entry is also shown while inactive
		# (is_shown_when_inactive), when the panels are hidden and the scores
		# already run, so it keeps one line there: tier and share.
		# has_journal_entry counts active entries only.
		triggered_desc = {
			desc = je_cultural_hegemony_status
			trigger = { NOT = { has_journal_entry = je_cultural_hegemony } }
		}
	}

	# ==== THE ENTRY'S MODIFIERS (vanilla 1.15) ================================
	# Re-read on the entry's own update, about every four days, with the owning
	# country as root, and on only while the entry is active (can_deactivate =
	# no, and nothing invalidates it, so once active it stays). Nothing adds,
	# removes or tests these by name: has_modifier cannot see a declared
	# modifier, and an add_modifier beside a declaration would double it. Each
	# value is 0 wherever the old add-and-remove code took the modifier off.
	#
	# The hegemon's bonus: the share of world cultural pull in percent, from 1%,
	# at most 100, from the monthly cache (var:ch_total).
	scaling_modifiers_while_active = {
		modifier = cultural_hegemony_effect
		value = {
			value = cultural_hegemony_modifier_mult
		}
	}
	# The Foreign Cultural Benchmark, the legitimacy a country trailing the
	# hegemon loses: the strength the monthly pulse below writes. It sat on the
	# country before; only this entry's pulse ever added it, so every country
	# that carried it holds the entry active.
	scaling_modifiers_while_active = {
		modifier = cultural_hegemony_foreign_benchmark
		value = {
			value = 0
			if = {
				limit = { ch_trails_hegemon = yes }
				add = var:ch_benchmark_applied
			}
		}
	}
	# Programme funding: its pull per step and its weekly cost, one step's cost
	# per step, while ch_program_funding_in_force (the Ministry of Culture and
	# at least one step).
	scaling_modifiers_while_active = {
		modifier = ch_cultural_program_funding
		value = {
			value = 0
			if = {
				limit = { ch_program_funding_in_force = yes }
				add = ch_program_funding_level_value
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ch_cultural_program_funding_cost
		value = {
			value = 0
			if = {
				limit = { ch_program_funding_in_force = yes }
				add = ch_program_funding_total_cost_mult
			}
		}
	}
	# The four programmes, each on while its country variable is set
	# (ch_active_<programme>, cultural_hegemony_triggers.txt), and International
	# Cultural Outreach's cost, one funding step's (GDP-scaled) worth while it
	# runs.
	scaling_modifiers_while_active = {
		modifier = ch_world_exposition
		value = {
			value = 0
			if = {
				limit = { ch_active_world_exposition = yes }
				add = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ch_world_exposition_cost
		value = {
			value = 0
			if = {
				limit = { ch_active_world_exposition = yes }
				add = ch_program_funding_cost_mult
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ch_cultural_institutes_program
		value = {
			value = 0
			if = {
				limit = { ch_active_cultural_institutes = yes }
				add = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ch_global_media_campaign
		value = {
			value = 0
			if = {
				limit = { ch_active_global_media_campaign = yes }
				add = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ch_cultural_protectionism_program
		value = {
			value = 0
			if = {
				limit = { ch_active_cultural_protectionism = yes }
				add = 1
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = ch_program_funding_level } }
			set_variable = { name = ch_program_funding_level value = 0 }
		}

		# Variables are already populated by on_actions before JE activates.
		# Initialize any that might not exist (safety fallback).
		if = {
			limit = { NOT = { has_variable = ch_total } }
			set_variable = { name = ch_total value = 0 }
			set_variable = { name = ch_art value = 0 }
			set_variable = { name = ch_sol value = 0 }
			set_variable = { name = ch_monuments value = 0 }
			set_variable = { name = ch_megaprojects value = 0 }
		}
	}

	on_weekly_pulse = {
		effect = {
			legacy_scaling_strip_cultural_hegemony = yes		# PENDING REMOVAL after 2026-12-31 (#1404): an older save's added copies, legacy_modifier_cleanup.txt
		}
	}

	on_monthly_pulse = {
		effect = {
			# Losing the Ministry of Culture shuts every cultural programme down:
			# funding to 0 (which takes the funding pair to 0 too) and the four
			# programmes' variables gone, so their declared modifiers, and the
			# outreach's cost, come off.
			if = {
				limit = { NOT = { has_law = law_type:law_ministry_of_culture } }
				set_variable = { name = ch_program_funding_level value = 0 }
				remove_variable = ch_world_exposition_on
				remove_variable = ch_cultural_institutes_on
				remove_variable = ch_global_media_campaign_on
				remove_variable = ch_cultural_protectionism_on
			}

			# The Foreign Cultural Benchmark's strength, which the entry declares
			# the modifier at: this month's cultural_hegemony_benchmark_mult, or 0
			# below 0.1, written every month so a stale value (an inherited one,
			# say) never lasts. Cached here, as before, because the live value
			# computes the world's cultural pull; the declaration and
			# ch_trails_hegemon read the variable.
			set_variable = { name = ch_benchmark_applied value = cultural_hegemony_benchmark_mult }
			if = {
				limit = { var:ch_benchmark_applied < 0.1 }
				set_variable = { name = ch_benchmark_applied value = 0 }
			}

			# Milestone events — self-gated by ch_milestone_* variables, fire
			# at most once per country when cultural_pull_total crosses 10 / 25
			# (.9 at 25 also needs the world's #1 rank).
			trigger_event = { id = cultural_hegemony.8 }
			trigger_event = { id = cultural_hegemony.9 }
		}
	}

	weight = 900
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = no

	# The map (vanilla 1.15 journal entry map mode; docs/systems/mod_systems/journal_map_modes.md): each country's
	# political model at the last census of models, in the model pie's colors. Root: the state being painted, every state
	# in the world. The Map Room draws the same map as its selector 22, from the same triggers in the same order
	# (common/scripted_triggers/te_journal_map_triggers.txt; the parity test in test_journal_map_modes.py).
	map_color = {

			value = "te_jmap_ch_communist"

		}
		else_if = {

		else_if = {

		}
		else_if = {

		}
		else_if = {

		else_if = {

		else_if = {
			limit = { te_jmap_ch_model = { MODEL = 12 } }

		}
		else_if = {

		}
	}

	map_tooltip = {
		if = {
			limit = { te_jmap_ch_no_data = yes }
			custom_tooltip_no_bullet = te_jmap_tt_ch_no_data
		}
		else_if = {
			limit = { te_jmap_ch_hegemon = yes }
			custom_tooltip_no_bullet = te_jmap_tt_ch_hegemon
		}
		else = {
			custom_tooltip_no_bullet = te_jmap_tt_ch_model
		}
	}
}
