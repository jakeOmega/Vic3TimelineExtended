# ============================================================================
# NUCLEAR WEAPONS — the programme, the posture and the crisis in one entry
# ============================================================================
# Shown to the player as "Nuclear Weapons". The key is still je_nuclear_program
# because the programme half predates the rest: saves carry the entry under
# that key, and an auto-activating entry under a new key never appears in a
# save started before it existed (scripting_best_practices.md § JE
# Auto-Activation Requires BOTH `is_shown_when_inactive` AND `possible`).
#
# Until 2026-09-25 posture and crises lived on their own entry,
# je_nuclear_deterrence. They were folded in here to spend one journal slot
# instead of two. The design's reason for keeping them apart
# (docs/systems/nuclear_crisis_design.md §2) was about GATING, not about the
# number of entries: a demoted power must keep its posture, its upkeep and its
# accidents, and losing rank must stop the programme without deleting the
# arsenal. That now lives in two places:
#   * `possible` is the union of both old gates (nuclear_program_entry_applies)
#     — a country that can run a programme, one that holds warheads, or one
#     party to a nuclear crisis;
#   * the programme half of the weekly pulse runs only while
#     nuclear_program_has_programme holds, and funding is zeroed the week it
#     stops holding. Every other "does this country have a programme?" test in
#     the mod asks nuclear_program_has_programme too, never has_journal_entry.
#
# Panels, top to bottom (gui/journal_entry.gui draws status desc(188),
# container_2(207), the button grid(215), container_3(236), scripted bars(246),
# the native bar(670), container_4(728), reason(742)):
#   container_3  the programme — only while there is one; sits directly above
#                the native bar, which is progress toward the next warhead
#   native bar   progress toward the next warhead
#   container_4  posture (only while armed), then the crisis and reputation,
#                then delivery and defence and the world's nuclear powers
#
# Deterrence logic lives in scripted effects (nuclear_deterrence_effects.txt,
# nuclear_crisis_effects.txt); the pulses below are their only callers.
# ============================================================================
je_nuclear_program = {
	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	group = je_group_foreign_affairs

	# Auto-activation needs this and `possible` together (journal_entries.md).
	# The tech keeps the entry in view, inactive, for a country that has the
	# science but not the standing, whose status line then says what it lacks;
	# the second clause lets a country without the tech activate when it is
	# armed or party to a crisis.
	is_shown_when_inactive = {
		has_game_rule = nuclear_weapons_enabled
		OR = {
			has_technology_researched = nuclear_weapons
			nuclear_program_entry_applies = yes
		}
	}

	# Kept, and kept visible to the AI: these two declarations are the AI's only
	# path into the programme. `is_ai = yes` on their `visible` hides the grid
	# from a human, who uses the programme panel instead. See the header of
	# common/scripted_buttons/nuclear_program_buttons.txt.
	scripted_button = increase_funding_nuclear_program
	scripted_button = decrease_funding_nuclear_program

	widget = {
		gui = "gui/journal_entry_widgets/nuclear_program_widget.gui"
		name = "widget_je_nuclear_programme"
		container = "custom_widget_container_3"
	}

	# Three widgets stacked in container_4, drawn in the order declared.
	widget = {
		gui = "gui/journal_entry_widgets/nuclear_deterrence_widget.gui"
		name = "widget_je_nuclear_posture"
		container = "custom_widget_container_4"
	}

	widget = {
		gui = "gui/journal_entry_widgets/nuclear_deterrence_widget.gui"
		name = "widget_je_nuclear_crisis"
		container = "custom_widget_container_4"
	}

	widget = {
		gui = "gui/journal_entry_widgets/nuclear_program_widget.gui"
		name = "widget_je_nuclear_balance"
		container = "custom_widget_container_4"
	}

	current_value = {
		value = root.var:nuclear_weapon_program_progress
	}

	goal_add_value = {
		value = nuclear_weapon_program_goal_value
	}

	progressbar = yes

	# Body single-sourced in common/scripted_triggers/nuke_triggers.txt. With
	# can_deactivate = yes the entry drops back to inactive the moment this goes
	# false — a disarmament settlement or the UN's NPT modifier landing on a
	# country that is not in a crisis, before the weekly pulse can run — and the
	# status line, which asks the same triggers, explains why.
	possible = {
		nuclear_program_entry_applies = yes
	}

	immediate = {
		# Runs on *every* activation, not just the first. The stockpile and the
		# first-device flag are therefore created only if they are missing: a
		# great power that drops below `possible` (losing the entry) and is later
		# promoted again used to have its entire arsenal silently deleted here.
		# Progress and funding are still zeroed unconditionally — the goal value
		# is fixed at activation from current_value + goal_add_value, so a
		# re-activation carrying leftover progress would give the native bar a
		# non-zero baseline and a goal of 100 + that leftover, and the bar would
		# read negative after the next warhead wrapped it.
		#
		# The four programme variables are created for every country the entry
		# activates for, programme or not, because the AI's funding buttons read
		# them unguarded. That is why their existence no longer means "has a
		# programme" — ask nuclear_program_has_programme.
		if = {
			limit = { NOT = { has_variable = nuclear_weapon_stockpile } }
			set_variable = {
				name = nuclear_weapon_stockpile
				value = 0
			}
		}
		if = {
			limit = { NOT = { has_variable = nuclear_weapons_program_first_nuke_done } }
			set_variable = {
				name = nuclear_weapons_program_first_nuke_done
				value = 0
			}
		}
		set_variable = {
			name = nuclear_weapon_program_progress
			value = 0
		}
		set_variable = { # REVIEWED 2026-09-26: a new record has no nuclear_weapon_program_funding modifier, so funding restarts at 0 to match it
			name = nuclear_weapons_program_funding
			value = 0
		}
		nuclear_program_refresh_state_effect = yes

		# Posture. A freshly activated entry is a new object: none of the
		# managed modifiers are on it, whatever the trackers from a previous
		# activation say. They are put on by nd_rebuild_posture_modifiers from
		# the first weekly pulse, not here — see its header.
		set_variable = { name = nd_doctrine_mod_on value = 0 } # REVIEWED 2026-09-26: tracker of a modifier a new record lacks; the first weekly pulse re-adds it
		set_variable = { name = nd_readiness_mod_on value = 0 } # REVIEWED 2026-09-26: tracker of a modifier a new record lacks; the first weekly pulse re-adds it
		set_variable = { name = nd_upkeep_mod_on value = 0 } # REVIEWED 2026-09-26: tracker of a modifier a new record lacks; the first weekly pulse re-adds it
		if = { # REVIEWED 2026-09-26: clearing the marker makes the first weekly pulse rebuild the posture modifiers on this record
			limit = { has_variable = nd_posture_on_merged_entry }
			remove_variable = nd_posture_on_merged_entry
		}
		if = {
			limit = { nd_is_armed = yes }
			nd_init_posture = yes
		}
	}

	complete = {
		always = no
	}

	on_complete = {
		if = {
			limit = { var:nuclear_weapons_program_first_nuke_done > 0 }
			custom_tooltip = {
				text = "ADDITIONAL_NUKE_CREATED"
			}
		}
		else = {
			custom_tooltip = {
				text = "FIRST_NUKE_CREATED"
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			# Once per activation, and once for a save made while posture had
			# its own entry.
			nd_rebuild_posture_modifiers = yes

			# Update global nuclear powers leaderboard
			update_nuclear_powers_ranking = yes

			if = {
				limit = {
					modifier:country_nuclear_disarmament_bool = yes
				}
				set_variable = {
					name = nuclear_weapon_stockpile
					value = 0
				}
				set_variable = {
					name = nuclear_weapon_program_progress
					value = 0
				}
				remove_modifier = nuclear_power
			}
			# Self-heal: the first-nuke branch below only grants nuclear_power
			# on the initial build (gated on first_nuke_done = 0), so a
			# disarmed-and-rebuilt country would otherwise be nuclear-armed
			# without the modifier.
			if = {
				limit = {
					NOT = { modifier:country_nuclear_disarmament_bool = yes }
					var:nuclear_weapon_stockpile >= 1
					NOT = { has_modifier = nuclear_power }
				}
				add_modifier = { name = nuclear_power }
			}

			# ==== THE PROGRAMME — only while this country has one ============
			if = {
				limit = { nuclear_program_has_programme = yes }
				nuclear_program_weekly_progress = yes
			}
			# Armed and in the entry, but without the standing to build (a
			# demoted power), or a non-nuclear party to a crisis: nothing is
			# being built, so nothing is paid for. Zeroing the variable is
			# enough — nuclear_program_refresh_state_effect below takes the
			# funding modifier off, as it does for a pause treaty.
			else_if = {
				limit = { var:nuclear_weapons_program_funding > 0 }
				set_variable = {
					name = nuclear_weapons_program_funding
					value = 0
				}
			}

			# The one site that derives the programme's display state. After the
			# programme block, so it sees this week's funding change, this
			# week's warheads and this week's first-device flag. Also owns the
			# funding modifier — see its header in
			# common/scripted_effects/nuclear_weapon_effects.txt.
			nuclear_program_refresh_state_effect = yes

			# ==== POSTURE AND CRISES =========================================
			# After the programme block, so a warhead finished this week is
			# already under a posture.
			nd_weekly_update = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			nd_monthly_update = yes
			nd_crisis_target_watchdog = yes
		}
	}

	on_yearly_pulse = {
		effect = {
			nd_yearly_update = yes
		}
	}

	# One line (two while armed or in a crisis). Unlike the widgets, status_desc
	# is rendered by gui/journal_entry.gui:188 with no IsActive gate, so this is
	# the only surface that can speak for a *deactivated* entry — the state a
	# disarmament settlement puts it into — and for an inactive one shown to a
	# country with the science but not the standing.
	status_desc = je_nuclear_program_status_line

	should_update_on_player_command = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
	# A revolution's winner receives the loser's country variables, keeping
	# its own value wherever it already had one, but none of its modifiers
	# (read from a before/after save pair, 2026-09-25 — scripting_best_practices.md
	# § "What a Civil War's Winner Inherits"). So a winning revolution inherits
	# the arsenal and the posture, and with `no` the winner starts a fresh
	# entry: it auto-activates because the inherited arsenal satisfies
	# `possible`, `immediate` keeps the stockpile and resets the posture
	# trackers, and the first weekly pulse rebuilds the posture modifiers and
	# restores nuclear_power. Two costs, both seen in the saves: up to two
	# weeks with neither the entry nor nuclear_power, and progress toward the
	# next warhead, which `immediate` zeroes (see its header).
	# The original reason for `no` — an inherited entry would carry the loser's
	# modifiers — was wrong: an inherited entry arrives with none and re-runs
	# `immediate` too. So `yes` should differ only by activating at once, closing
	# the two-week gap — but only while the rebels hold no nuclear variables of
	# their own. Once the custody split gives them some, their values win the
	# merge and `immediate` runs against those, so settle the flag together with
	# the split (nuclear_crisis_design.md §0.10, step 2).
	can_revolution_inherit = no
	can_deactivate = yes
}
