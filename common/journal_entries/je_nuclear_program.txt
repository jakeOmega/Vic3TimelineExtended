je_nuclear_program = {
	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	group = je_group_foreign_affairs

	is_shown_when_inactive = {
		has_game_rule = nuclear_weapons_enabled
		has_technology_researched = nuclear_weapons
	}

	# Kept, and kept visible to the AI: these two declarations are the AI's only
	# path into the programme. `is_ai = yes` on their `visible` hides the grid
	# from a human, who uses the two panels below instead. See the header of
	# common/scripted_buttons/nuclear_program_buttons.txt.
	scripted_button = increase_funding_nuclear_program
	scripted_button = decrease_funding_nuclear_program

	# The programme readout sits in container_3, immediately above the native
	# progress bar, so the numbers that explain the bar are next to it; the
	# deterrence and nuclear-powers sections sit in container_4, immediately
	# below it. Panel order in gui/journal_entry.gui: status_desc(188),
	# container_2(207), button grid(215), container_3(236), scripted
	# bars(246), native bar(670), container_4(728), reason(742).
	widget = {
		gui = "gui/journal_entry_widgets/nuclear_program_widget.gui"
		name = "widget_je_nuclear_programme"
		container = "custom_widget_container_3"
	}

	widget = {
		gui = "gui/journal_entry_widgets/nuclear_program_widget.gui"
		name = "widget_je_nuclear_deterrence"
		container = "custom_widget_container_4"
	}

	current_value = {
		value = root.var:nuclear_weapon_program_progress
	}

	goal_add_value = {
		value = nuclear_weapon_program_goal_value
	}

	progressbar = yes

	# Body single-sourced in common/scripted_triggers/nuke_triggers.txt, because
	# the programme's status line has to explain a *deactivated* entry and must
	# ask exactly the same question this does. With can_deactivate = yes below,
	# the entry drops back to inactive the moment this goes false — which is what
	# happens the instant a disarmament settlement or the UN's NPT modifier
	# lands, before the weekly pulse can run.
	possible = {
		nuclear_program_can_run_programme = yes
	}

	immediate = {
		# Runs on *every* activation, not just the first. The stockpile and the
		# first-device flag are therefore created only if they are missing: a
		# great power that drops below `possible` (losing the entry) and is later
		# promoted again used to have its entire arsenal silently deleted here.
		# Progress and funding are still zeroed unconditionally — the goal value
		# is fixed at activation from current_value + goal_add_value, so a
		# re-activation carrying leftover progress would give the native bar a
		# non-zero baseline and a goal of 100 + that leftover, and the bar would
		# read negative after the next warhead wrapped it.
		if = {
			limit = { NOT = { has_variable = nuclear_weapon_stockpile } }
			set_variable = {
				name = nuclear_weapon_stockpile
				value = 0
			}
		}
		if = {
			limit = { NOT = { has_variable = nuclear_weapons_program_first_nuke_done } }
			set_variable = {
				name = nuclear_weapons_program_first_nuke_done
				value = 0
			}
		}
		set_variable = {
			name = nuclear_weapon_program_progress
			value = 0
		}
		set_variable = {
			name = nuclear_weapons_program_funding
			value = 0
		}
		nuclear_program_refresh_state_effect = yes
	}

	complete = {
		always = no
	}

	on_complete = {
		if = {
			limit = { var:nuclear_weapons_program_first_nuke_done > 0 }
			custom_tooltip = {
				text = "ADDITIONAL_NUKE_CREATED"
			}
		}
		else = {
			custom_tooltip = {
				text = "FIRST_NUKE_CREATED"
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			# Update global nuclear powers leaderboard
			update_nuclear_powers_ranking = yes

			if = {
				limit = {
					modifier:country_nuclear_disarmament_bool = yes
				}
				set_variable = {
					name = nuclear_weapon_stockpile
					value = 0
				}
				set_variable = {
					name = nuclear_weapon_program_progress
					value = 0
				}
				remove_modifier = nuclear_power
			}
			# Self-heal: the first-nuke branch below only grants nuclear_power
			# on the initial build (gated on first_nuke_done = 0), so a
			# disarmed-and-rebuilt country would otherwise be nuclear-armed
			# without the modifier.
			if = {
				limit = {
					NOT = { modifier:country_nuclear_disarmament_bool = yes }
					var:nuclear_weapon_stockpile >= 1
					NOT = { has_modifier = nuclear_power }
				}
				add_modifier = { name = nuclear_power }
			}
			# A pause treaty zeroes funding. Taking the funding modifier back off
			# is left to nuclear_program_refresh_state_effect at the tail of this
			# block, the one site that owns that modifier. The remove_modifier
			# that used to sit here ran in *country* scope, while the buttons
			# apply the modifier in `je:je_nuclear_program` scope — so it was a
			# silent no-op and a paused programme went on paying its full weekly
			# innovation bill for ever.
			if = {
				limit = {
					modifier:country_nuclear_program_pause_bool = yes
				}
				set_variable = {
					name = nuclear_weapons_program_funding
					value = 0
				}
			}
			change_variable = {
				name = nuclear_weapon_program_progress
				add = {
					value = nuclear_weapon_program_monthly_progress
					divide = 4
					multiply = {
						value = 1
						add = modifier:country_nuclear_program_progress_mult
					}
				}
			}
			while = {
				limit = {var:nuclear_weapon_program_progress >= nuclear_weapon_program_goal_value}
				change_variable = {
					name = nuclear_weapon_program_progress
					subtract = nuclear_weapon_program_goal_value
				}
				change_variable = {
					name = nuclear_weapon_stockpile
					add = 1
				}
				if = {
					limit = { var:nuclear_weapons_program_first_nuke_done = 0 }
					set_variable = {
						name = nuclear_weapons_program_first_nuke_done
						value = 1
					}
					change_variable = {
						name = nuclear_weapon_program_progress
						multiply = nuclear_weapon_program_additional_nuke_progress_multiplier
					}
					# The year of the first device, for the panel's "First
					# device" row. `year` is a valid script value (see
					# common/script_values/te_history_values.txt).
					set_variable = {
						name = nuclear_program_first_device_year
						value = year
					}
					# Track whether this is the first nuclear power in the world
					if = {
						limit = { NOT = { has_global_variable = world_first_nuclear_weapon } }
						set_variable = { name = is_world_first_nuclear_power value = yes }
					}
					set_global_variable = { name = world_first_nuclear_weapon value = yes }
					trigger_event = { id = nuclear_weapon_events.10 }
					save_scope_as = first_nuke_country
					every_country = {
						limit = {
							NOT = { THIS = scope:first_nuke_country }
						}
						trigger_event = { id = nuclear_weapon_events.9 days = 14 }
					}
				}
			}

			# ---- Development events (only during first nuke, with active funding) ----
			if = {
				limit = {
					var:nuclear_weapons_program_first_nuke_done = 0
					has_variable = nuclear_weapons_program_funding
					var:nuclear_weapons_program_funding >= 1
					var:nuclear_weapon_program_progress >= 5
				}
				random_list = {
					490 = { } # nothing happens
					5 = {
						trigger_event = { id = nuclear_weapon_events.19 }
					}
					5 = {
						trigger_event = { id = nuclear_weapon_events.20 }
					}
				}
			}

			# ---- Proliferation alert (non-first country nearing completion) ----
			if = {
				limit = {
					has_global_variable = world_first_nuclear_weapon
					NOT = { has_variable = is_world_first_nuclear_power }
					var:nuclear_weapons_program_first_nuke_done = 0
					var:nuclear_weapon_program_progress >= 75
					NOT = { has_variable = nuclear_proliferation_alert_sent }
				}
				set_variable = { name = nuclear_proliferation_alert_sent value = 1 }
				save_scope_as = proliferating_country
				# Only countries with a reason to care: great powers, rivals in
				# either direction, and neighbours. Every recipient's reply can
				# shave progress off this programme, so sending it to the whole
				# world let ~90 bystanders each chip at it.
				every_country = {
					limit = {
						NOT = { THIS = ROOT }
						NOT = { is_country_type = decentralized }
						OR = {
							country_rank >= rank_value:great_power
							any_rival_country = { THIS = scope:proliferating_country }
							any_rivaling_country = { THIS = scope:proliferating_country }
							any_neighbouring_state = { owner = scope:proliferating_country }
						}
					}
					trigger_event = { id = nuclear_weapon_events.18 days = 7 }
				}
				# ...and the proliferator learns that the world has noticed.
				trigger_event = { id = nuclear_weapon_events.23 days = 7 }
			}

			# The one site that derives the programme's display state. Last in
			# the block, so it sees this week's funding change, this week's
			# warheads and this week's first-device flag. Also owns the funding
			# modifier — see its header in
			# common/scripted_effects/nuclear_weapon_effects.txt.
			nuclear_program_refresh_state_effect = yes
		}
	}

	# One line. Unlike the widget, status_desc is rendered by
	# gui/journal_entry.gui:188 with no IsActive gate, so this is the only
	# surface that can speak for a *deactivated* entry — the state a disarmament
	# settlement puts the programme into. The eleven triggered_desc entries that
	# used to live here were the global nuclear-powers leaderboard; it is now a
	# collapsed section of the deterrence panel.
	status_desc = je_nuclear_program_status_line

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = yes
}
