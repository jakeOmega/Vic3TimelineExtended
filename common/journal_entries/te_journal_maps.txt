# ============================================================================
# Journal entry map modes: the Map Room (vanilla 1.15; third pass, 2026-11-05)
# ============================================================================
# docs/systems/mod_systems/journal_map_modes.md; the in-game checks in docs/testing/journal-map-modes-playtest.md.
#
# Vanilla 1.15 lets an entry paint the map: map_color (and map_color_second, drawn as stripes) returns a color for each
# state, with the state as root, and map_tooltip prints the hover text. The engine shows that map while the entry's
# page is open (the hidden mm_journal_entry_map_data mode); no .gui calls it. One map per entry, so the Map Room
# carries several on a selector.
#
#   je_te_map_room             the Map Room: a contextless entry nobody is involved in, so it sits in no country's
#                              journal and exists, active, from the game's start. A button opens its page at the first
#                              click (the first pass's hidden world entry proved both, 2026-11-05). Its picks are GUI
#                              buttons calling scripted GUIs (gui/journal_entry_widgets/te_map_room_widget.gui), not
#                              scripted buttons: the engine locks a scripted button for a country not involved in the
#                              entry ("Countries that are not involved in the Journal Entry may not take this action").
#
# The first pass's je_te_map_host (a country entry added by a button) is gone: add_journal_entry was refused every
# time, because every country already holds an inactive copy of an entry with its own activation triggers, and the
# engine activated it on its own check up to two weeks later (scripting_best_practices.md, "Journal Entry Map Modes").
#
# The entry is the player's tool only: no AI country sets anything here, and it has no effect.

je_te_map_room = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_internal_affairs
	context = none

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	transferable = no
	can_revolution_inherit = no

	# Active from the start (possible defaults to yes), involving nobody and shown to nobody: the shape of the first
	# pass's hidden world entry, which the 1837 save held active with no country.
	is_shown_when_inactive = {
		always = yes
	}
	should_be_involved = {
		always = no
	}
	should_show_when_not_involved = {
		always = no
	}

	# The picks, above the description (te_map_room_widget.gui).
	widget = {
		gui = "gui/journal_entry_widgets/te_map_room_widget.gui"
		name = "widget_je_te_map_room"
		container = "custom_widget_container_1"
	}

	# Root: the state being painted. A branch that matches nothing leaves the state unpainted, which is how a map with
	# no pick (selector 0) should look.
	map_color = {
		# 1. Turmoil.
		if = {
			limit = { te_jmap_selector = 1 }
			lerp = {
				min_color = "te_jmap_turmoil_low"
				max_color = "te_jmap_turmoil_high"
				factor = te_jmap_turmoil_factor
			}
		}
		# 2. Literacy.
		else_if = {
			limit = { te_jmap_selector = 2 }
			lerp = {
				min_color = "te_jmap_literacy_low"
				max_color = "te_jmap_literacy_high"
				factor = te_jmap_literacy_factor
			}
		}
		# 3. Life expectancy (Demographics), in two halves at 52.5 years.
		else_if = {
			limit = {
				te_jmap_selector = 3
				has_variable = te_dg_e0
				var:te_dg_e0 > 0
				te_jmap_e0_factor < 0.5
			}
			lerp = {
				min_color = "te_jmap_e0_low"
				max_color = "te_jmap_e0_mid"
				factor = te_jmap_e0_factor_lower
			}
		}
		else_if = {
			limit = {
				te_jmap_selector = 3
				has_variable = te_dg_e0
				var:te_dg_e0 > 0
			}
			lerp = {
				min_color = "te_jmap_e0_mid"
				max_color = "te_jmap_e0_high"
				factor = te_jmap_e0_factor_upper
			}
		}
		# 4. Fertility (Demographics), diverging at replacement.
		else_if = {
			limit = {
				te_jmap_selector = 4
				has_variable = te_dg_tfr
				var:te_dg_tfr > 0
				var:te_dg_tfr < 2.1
			}
			lerp = {
				min_color = "te_jmap_tfr_low"
				max_color = "te_jmap_tfr_mid"
				factor = te_jmap_tfr_factor_lower
			}
		}
		else_if = {
			limit = {
				te_jmap_selector = 4
				has_variable = te_dg_tfr
				var:te_dg_tfr > 0
			}
			lerp = {
				min_color = "te_jmap_tfr_mid"
				max_color = "te_jmap_tfr_high"
				factor = te_jmap_tfr_factor_upper
			}
		}
		# 5. Median age (Demographics), in two halves at 30 years.
		else_if = {
			limit = {
				te_jmap_selector = 5
				has_variable = te_dg_median
				var:te_dg_median > 0
				var:te_dg_median < 30
			}
			lerp = {
				min_color = "te_jmap_median_low"
				max_color = "te_jmap_median_mid"
				factor = te_jmap_median_factor_lower
			}
		}
		else_if = {
			limit = {
				te_jmap_selector = 5
				has_variable = te_dg_median
				var:te_dg_median > 0
			}
			lerp = {
				min_color = "te_jmap_median_mid"
				max_color = "te_jmap_median_high"
				factor = te_jmap_median_factor_upper
			}
		}
		# The Demographics maps, a state with no census yet, no people to count, or the rule off.
		else_if = {
			limit = {
				OR = {
					te_jmap_selector = 3
					te_jmap_selector = 4
					te_jmap_selector = 5
				}
			}
			value = "te_jmap_no_data"
		}
	}

	# Root: the state under the cursor. Never runs as an effect; it only prints.
	map_tooltip = {
		if = {
			limit = { te_jmap_selector = 1 }
			custom_tooltip_no_bullet = te_jmap_tt_turmoil
		}
		else_if = {
			limit = { te_jmap_selector = 2 }
			custom_tooltip_no_bullet = te_jmap_tt_literacy
		}
		# Over 10,000 is a separate branch, not a SelectLocalization in the loc: a key
		# named there loses THIS, and every hover logged "Promote 'THIS' returned
		# nullptr" and printed nothing (2026-11-05).
		else_if = {
			limit = {
				te_jmap_selector = 3
				has_variable = te_dg_e0
				var:te_dg_e0 > 10000
			}
			custom_tooltip_no_bullet = te_jmap_tt_life_expectancy_over
		}
		else_if = {
			limit = {
				te_jmap_selector = 3
				has_variable = te_dg_e0
				var:te_dg_e0 > 0
			}
			custom_tooltip_no_bullet = te_jmap_tt_life_expectancy
		}
		else_if = {
			limit = {
				te_jmap_selector = 4
				has_variable = te_dg_tfr
				var:te_dg_tfr > 0
			}
			custom_tooltip_no_bullet = te_jmap_tt_fertility
		}
		else_if = {
			limit = {
				te_jmap_selector = 5
				has_variable = te_dg_median
				var:te_dg_median > 0
			}
			custom_tooltip_no_bullet = te_jmap_tt_median_age
		}
		else_if = {
			limit = {
				OR = {
					te_jmap_selector = 3
					te_jmap_selector = 4
					te_jmap_selector = 5
				}
			}
			custom_tooltip_no_bullet = te_jmap_tt_no_census
		}
		else = {
			custom_tooltip_no_bullet = te_jmap_tt_no_pick
		}
	}
}
