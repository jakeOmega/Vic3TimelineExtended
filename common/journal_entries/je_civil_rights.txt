		if = {
			limit = { NOT = { has_active_civil_rights_movement = yes } }
			create_political_movement = { type = movement_civil_rights }

		# A fresh run starts clean: stale months counts from an earlier run (one
		# that was invalidated) would corrupt path resolution, and the tier
		# notices re-arm. immediate also runs again, after the variable merge,
		# when a revolution's winner inherits this entry mid-struggle; the
		# loser's cr_run_in_progress then says the run is not fresh, and its
		# counters, tier flags and bar carry on. cr_je_cleanup_effect removes the
		# marker when a run ends. The bar starts at 30 anyway (start_value); the
		# explicit set is kept in case the engine re-uses an invalidated record.
		if = {
			limit = { NOT = { has_variable = cr_run_in_progress } }
			set_variable = cr_run_in_progress
			set_variable = { name = cr_grassroots_months value = 0 }
			set_variable = { name = cr_federal_months value = 0 }
			set_variable = { name = cr_gradualist_months value = 0 }
			set_variable = { name = cr_cooptation_months value = 0 }

			set_variable = { name = cr_segregationist_months value = 0 }

			if = { limit = { has_variable = cr_tier_75_seen } remove_variable = cr_tier_75_seen }
			if = { limit = { has_variable = cr_tier_90_seen } remove_variable = cr_tier_90_seen }

			scope:journal_entry = {
				set_bar_progress = { name = civil_rights_support_bar value = 30 }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			cr_ensure_tracker_vars_effect = yes

			# --- Revolution continuity self-heal (civil_rights_effects.txt) ---
			# A save from before the run marker and the policy mirrors existed
			# gets them here; a revolution's winner that on_civil_war_won did not
			# repair gets its policy modifiers back.

			}
			if = {
				limit = {
					OR = {

						has_variable = cr_policy_gradualist
					}
				}

			}
			if = {

					}
				}

			}
			if = {
				limit = {
					OR = {
						has_modifier = cr_suppression_modifier
						has_variable = cr_policy_suppression
					}
				}
				change_variable = { name = cr_suppression_months add = 1 }
			}
			if = {
				limit = {
					OR = {
						has_modifier = cr_segregationist_modifier
						has_variable = cr_policy_segregationist
					}
				}
				change_variable = { name = cr_segregationist_months add = 1 }
			}

			# --- Self-check, once a game (#1431): the outcomes rest on
			# cr_win_ending_now / cr_fail_ending_now, whose branches compare one
			# script value with another (cr_months_disp_federal >=
			# cr_months_disp_grassroots). Vanilla compares two script values in a
			# journal entry trigger (je_state_and_revolution: revolutionary_bloc_gdp
			# >= revolutionary_bloc_gdp_target); this confirms it inside a script
			# value's limit. Only where the counters differ, so OK cannot be a tie.
			# BROKEN: every ending falls to the generic .200 / .232; rewrite the two
			# chains on the var: counters (docs/testing/vanilla-1-15-playtest.md).
			if = {
				limit = {
					NOT = { has_global_variable = te_cr_ending_chain_checked }
					has_variable = cr_grassroots_months
					has_variable = cr_federal_months
					var:cr_grassroots_months > var:cr_federal_months
				}
				set_global_variable = te_cr_ending_chain_checked
				if = {
					limit = { cr_months_disp_grassroots > cr_months_disp_federal }
					debug_log = "TE_SELF_CHECK OK cr_ending_chain ([THIS.GetCountry.GetTagName] on [TimeKeeper.GetCurrentDate.GetString]: grassroots=[THIS.Var('cr_grassroots_months').GetValue|0] federal=[THIS.Var('cr_federal_months').GetValue|0], and the script values compare the same way)"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN cr_ending_chain ([THIS.GetCountry.GetTagName] on [TimeKeeper.GetCurrentDate.GetString]: grassroots=[THIS.Var('cr_grassroots_months').GetValue|0] federal=[THIS.Var('cr_federal_months').GetValue|0], but cr_months_disp_grassroots > cr_months_disp_federal is false: a script value compared with another fails, so the Civil Rights endings fall to the generic ones; see docs/testing/vanilla-1-15-playtest.md)"
				}
			}

			# --- Cooptation expiry: after 12 months the bar push goes to 0 ---
			# Re-applies cleanly each tick once expired so the marker survives
			# the periodic phase-modifier swaps below. The marker lasts the rest of
			# the run once cooptation has had 12 months, running or not, so the
			# counter alone decides (it only grows while cooptation runs). That also
			# restores it, before a restart can collect the push, in a save where
			# End Cooptation used to remove it and after a civil war whose rebuild
			# left it to this pulse.
			if = {
				limit = {
					var:cr_cooptation_months >= 12
					NOT = { has_modifier = cr_cooptation_expired }
				}
				add_modifier = { name = cr_cooptation_expired }
			}

			# --- Tier-cross threshold events (one-shot via cr_tier_*_seen flags) ---
			# Tier 25
			if = {
				limit = {
					NOT = { has_variable = cr_tier_25_seen }
					scope:journal_entry = { "scripted_bar_progress(civil_rights_support_bar)" >= 25 }
				}
				set_variable = cr_tier_25_seen
				if = {
					limit = { has_severe_discriminatory_minority_law = yes }
					trigger_event = { id = movement_events_te.13 } # Refugee networks (existing)
				}
				else = {
					trigger_event = { id = movement_events_te.301 } # First Mass Rally (new)
				}
			}
			# Tier 50
			if = {
				limit = {
					NOT = { has_variable = cr_tier_50_seen }
					scope:journal_entry = { "scripted_bar_progress(civil_rights_support_bar)" >= 50 }
				}
				set_variable = cr_tier_50_seen
				if = {
					limit = { has_discriminatory_minority_law = yes }
					trigger_event = { id = movement_events_te.15 } # Martyrdom (existing)
				}
				else = {
					trigger_event = { id = movement_events_te.303 } # Trade Union Coalition (new)
				}
			}
			# Tier 75. .16's own trigger needs a discriminatory minority law, and
			# .304's text presumes Federal Protection's two-year commission, so a
			# country with neither gets the law-neutral .302.
			if = {
				limit = {
					NOT = { has_variable = cr_tier_75_seen }
					scope:journal_entry = { "scripted_bar_progress(civil_rights_support_bar)" >= 75 }
				}
				set_variable = cr_tier_75_seen
				if = {
					limit = { var:cr_federal_months > 24 }
					trigger_event = { id = movement_events_te.304 } # Federal Commission (new)
				}
				else_if = {
					limit = { has_discriminatory_minority_law = yes }
					trigger_event = { id = movement_events_te.16 } # Civil Disobedience Campaign (existing)
