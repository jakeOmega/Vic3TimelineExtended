############################################
# Space Race Journal Entries
# Linear milestone competition system
#
# `immediate` runs again: when an entry re-activates (every milestone has
# can_deactivate = yes), and on every entry a revolution's winner inherits,
# after the loser's variables have merged in
# (docs/guides/scripting_best_practices.md § "What a Civil War's Winner
# Inherits"). So it creates only what is missing (progress, funding) and asks
# each choice event once per milestone (sr_<m>_choice_made). Each milestone's
# cost, funding, approach and choice modifiers are declared on its entry
# (scaling_modifiers_while_active, #1404), from variables the winner inherits,
# so nothing puts them back, and so are solar colonization's colony modifiers.
#
# Carrying progress into a new record keeps the progress bar right:
# - Inherited: the new record keeps the loser's bar. Its start date, baseline
#   and goal are copied, not evaluated again (German-revolution saves,
#   2026-09-25: the winner's je_global_warming kept baseline 0.1 and goal 4.0
#   with the anomaly at 1.18).
# - Re-activated before the next monthly pulse, which clears a deactivated
#   milestone's progress (sr_cleanup_inactive_space_race_milestones): the goal
#   is evaluated again, as current_value + goal_add_value, with the progress
#   in current_value. So goal_add_value subtracts the progress, which pins the
#   goal at sr_<m>_goal, and the bar fills from there. This reads like a bug
#   and is not (scripting_best_practices.md § "A Journal Entry's Goal Is
#   Frozen at Activation").
# Solar colonization's bar restarts at every colony, so it keeps its progress
# only when it cannot deactivate: see its `immediate`.
############################################

# 1. SUBORBITAL FLIGHT
# Prereq: rocketry tech, GP/MP rank
# ===========================================
je_space_race_suborbital = {
	icon = "gfx/interface/icons/event_icons/je_space_race_suborbital.dds"
	group = je_group_foreign_affairs

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	# The player-facing surface (gui/journal_entry_widgets/space_race_widget.gui):
	# the overview above the status line, the live sections below it, How the
	# Space Race Works last. The vanilla button grid below is hidden from humans
	# (every space race button carries `is_ai = yes`), so every action it offered
	# is offered in the live sections, through
	# common/scripted_guis/space_race_sguis.txt.
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_suborbital_overview"
		container = "custom_widget_container_1"
	}
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_suborbital"
		container = "custom_widget_container_2"
	}
	# MARKER, not content (gui/journal_entry_widgets/te_je_bars_on_top_marker.gui):
	# the mod's gui/journal_entry.gui hides the entry's own goal bar at the foot
	# of the panel, because the overview above draws the same progress.
	widget = {
		gui = "gui/journal_entry_widgets/te_je_bars_on_top_marker.gui"
		name = "widget_te_je_bars_on_top_marker"
		container = "custom_widget_container_7"
	}

	# 1.15: with the rule off, drop the inactive record for good (a rule never changes mid-game).
	disable_evaluation = {
		NOT = { sr_space_race_on = yes }
	}
	is_shown_when_inactive = {
		sr_space_race_on = yes
		has_technology_researched = rocketry
		is_great_or_major_power = yes
		# The rebels of a running revolution start no programme of their own.
		# They hold none of the country's variables, so they would begin again
		# from suborbital flight. If they won, their own progress and funding
		# would beat the loser's in the variable merge, and a run the loser had
		# already completed would reopen. Only suborbital needs this: every
		# later entry requires a sr_completed_<m> the rebels do not hold. Not in
		# `possible`, which an inherited running entry re-checks
		# (can_deactivate = yes). Vanilla precedent:
		# 02_south_american_national_identity.txt.
		is_revolutionary = no
	}

	possible = {
		has_technology_researched = rocketry
		modifier:country_sr_earth_orbit_program_bool = yes
		is_great_or_major_power = yes
		NOT = { has_variable = sr_completed_suborbital }
	}

	# The milestone's cost and funding (#1404; space_race_values.txt), each
	# while it is billed (sr_milestone_billed_suborbital: while it runs):
	# its cost level (funding plus the approach's overhead, times the era
	# factor), its pace at the effective funding level (soft cap) and its
	# throughput at the level itself. Every running milestone's entry carries
	# its own, so they add up across entries.
	scaling_modifiers_while_active = {
		modifier = sr_space_program_cost
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_suborbital = yes }
				add = sr_suborbital_cost
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_funding_progress
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_suborbital = yes }
				add = sr_funding_suborbital_effective
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_funding_throughput
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_suborbital = yes }
				add = sr_funding_suborbital_level
			}
		}
	}
	# The approach, while its variable is set (sr_effect_safe / sr_effect_ambitious;
	# Standard is neither).
	scaling_modifiers_while_active = {
		modifier = sr_safe_approach
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_safe_suborbital }
				add = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_ambitious_approach
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_ambitious_suborbital }
				add = 1
			}
		}
	}

	immediate = {
		set_variable = { name = sr_active_suborbital value = yes }
		# Its band's offset from the start, so another country's estimate (National Info) never sits on the exact figure
		sr_power_fuzz_first = { MILESTONE = suborbital }
		if = { limit = { NOT = { has_variable = sr_progress_suborbital } } set_variable = { name = sr_progress_suborbital value = 0 } }
		if = { limit = { NOT = { has_variable = sr_funding_suborbital } } set_variable = { name = sr_funding_suborbital value = 1 } }
		sr_refresh_milestone_statuses = yes
	}

	scripted_button = sr_btn_safe_suborbital
	scripted_button = sr_btn_ambitious_suborbital
	scripted_button = sr_btn_fund_up_suborbital
	scripted_button = sr_btn_fund_down_suborbital

	current_value = {
		value = sr_progress_suborbital_value
	}

	# Pins the goal at sr_suborbital_goal: see the file header.
	goal_add_value = {
		value = sr_suborbital_goal
		subtract = sr_progress_suborbital_value
	}

	# Kept for the journal list (the entry's own goal bar is hidden in the panel
	# by the bars-on-top marker above; the overview draws the progress).
	progressbar = yes

	# Only while the entry is inactive. An active entry's overview shows its
	# state, and the reason text below repeats the rest.
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_space_race_suborbital_status
				trigger = { NOT = { has_journal_entry = je_space_race_suborbital } }
			}
		}
	}

	outcome = {
		header = je_space_race_suborbital_completion_header
		effect_header = je_space_race_suborbital_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_space_race_suborbital_complete_conclusion_desc
		trigger = {
			has_variable = sr_progress_suborbital
			var:sr_progress_suborbital >= sr_suborbital_goal
		}
		effect = {
			sr_complete_suborbital_effect = yes
			remove_variable = sr_active_suborbital
			trigger_event = { id = space_race_events.1 }
		}
		# Outcome previews — engine auto-tooltips the effect block; effects do not run.
		event_outcome_effect_desc = {
			header = je_space_race_suborbital_first_outcome_header
			effect = {
				custom_tooltip = je_space_race_suborbital_first_outcome_tt
				add_modifier = { name = sr_first_suborbital }
			}
		}
		event_outcome_effect_desc = {
			header = je_space_race_suborbital_subsequent_outcome_header
			effect = {
				custom_tooltip = je_space_race_suborbital_subsequent_outcome_tt
				add_modifier = { name = sr_suborbital }
			}
		}
	}

	outcome = {
		header = je_space_race_suborbital_failure_header
		effect_header = je_space_race_suborbital_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_space_race_suborbital_fail_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = sr_space_program_incompatible_tt
				NOT = { modifier:country_sr_earth_orbit_program_bool = yes }
			}
		}
		effect = {
			sr_report_program_loss = { MILESTONE = suborbital }
			custom_tooltip = {
				text = sr_milestone_fail_cleanup_tt
				sr_cleanup_milestone_variables_effect = { MILESTONE = suborbital }
			}
		}
		event_outcome_effect_desc = {
			header = je_space_race_suborbital_setback_header
			effect = {
				custom_tooltip = je_space_race_suborbital_setback_tt
			}
		}
	}

	on_monthly_pulse = {
		effect = { sr_monthly_progress_update_effect = { MILESTONE = suborbital } }
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = yes
	should_update_on_player_command = yes
}

	group = je_group_foreign_affairs

	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"

	}
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_interstellar_probe"
		container = "custom_widget_container_2"
	}
	# MARKER, not content (gui/journal_entry_widgets/te_je_bars_on_top_marker.gui):
	# the mod's gui/journal_entry.gui hides the entry's own goal bar at the foot
	# of the panel, because the overview above draws the same progress.
	widget = {
		gui = "gui/journal_entry_widgets/te_je_bars_on_top_marker.gui"
		name = "widget_te_je_bars_on_top_marker"
		container = "custom_widget_container_7"
	}

	# 1.15: with the rule off, drop the inactive record for good (a rule never changes mid-game).
	disable_evaluation = {
		NOT = { sr_space_race_on = yes }
	}
	is_shown_when_inactive = {
		sr_space_race_on = yes
		has_variable = sr_completed_probe
		has_variable = sr_completed_mars_landing
		has_technology_researched = aneutronic_fusion
	}

	possible = {
		has_variable = sr_completed_probe
		has_variable = sr_completed_mars_landing
		has_technology_researched = aneutronic_fusion
		modifier:country_sr_interstellar_mission_program_bool = yes
		NOT = { has_variable = sr_completed_interstellar_probe }
		is_great_or_major_power = yes
	}

	# The milestone's cost and funding (#1404; space_race_values.txt), each
	# while it is billed (sr_milestone_billed_interstellar_probe: while it runs):
	# its cost level (funding plus the approach's overhead, times the era
	# factor), its pace at the effective funding level (soft cap) and its
	# throughput at the level itself. Every running milestone's entry carries
	# its own, so they add up across entries.
	scaling_modifiers_while_active = {
		modifier = sr_space_program_cost
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_interstellar_probe = yes }
				add = sr_interstellar_probe_cost
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_funding_progress
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_interstellar_probe = yes }
				add = sr_funding_interstellar_probe_effective
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_funding_throughput
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_interstellar_probe = yes }
				add = sr_funding_interstellar_probe_level
			}
		}
	}
	# The approach, while its variable is set (sr_effect_safe / sr_effect_ambitious;
	# Standard is neither).
	scaling_modifiers_while_active = {
		modifier = sr_safe_approach
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_safe_interstellar_probe }
				add = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_ambitious_approach
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_ambitious_interstellar_probe }
				add = 1
			}
		}
	}

	immediate = {
		set_variable = { name = sr_active_interstellar_probe value = yes }
		# Its band's offset from the start, so another country's estimate (National Info) never sits on the exact figure
		sr_power_fuzz_first = { MILESTONE = interstellar_probe }
		if = { limit = { NOT = { has_variable = sr_progress_interstellar_probe } } set_variable = { name = sr_progress_interstellar_probe value = 0 } }
		if = { limit = { NOT = { has_variable = sr_funding_interstellar_probe } } set_variable = { name = sr_funding_interstellar_probe value = 1 } }
		sr_refresh_milestone_statuses = yes
	}

	scripted_button = sr_btn_safe_interstellar_probe
	scripted_button = sr_btn_ambitious_interstellar_probe
	scripted_button = sr_btn_fund_up_interstellar_probe
	scripted_button = sr_btn_fund_down_interstellar_probe

	current_value = {
		value = sr_progress_interstellar_probe_value
	}

	# Pins the goal at sr_interstellar_probe_goal: see the file header.
	goal_add_value = {
		value = sr_interstellar_probe_goal
		subtract = sr_progress_interstellar_probe_value
	}

	# Kept for the journal list (the entry's own goal bar is hidden in the panel
	# by the bars-on-top marker above; the overview draws the progress).
	progressbar = yes

	# Only while the entry is inactive. An active entry's overview shows its
	# state, and the reason text below repeats the rest.
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_space_race_interstellar_probe_status
				trigger = { NOT = { has_journal_entry = je_space_race_interstellar_probe } }
			}
		}
	}

	outcome = {
		header = je_space_race_interstellar_probe_completion_header
		effect_header = je_space_race_interstellar_probe_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_space_race_interstellar_probe_complete_conclusion_desc
		trigger = {
			has_variable = sr_progress_interstellar_probe
			var:sr_progress_interstellar_probe >= sr_interstellar_probe_goal
		}
		effect = {
			sr_complete_interstellar_probe_effect = yes
			remove_variable = sr_active_interstellar_probe
			trigger_event = { id = space_race_events.8 }
		}
		event_outcome_effect_desc = {
			header = je_space_race_interstellar_probe_first_outcome_header
			effect = {
				custom_tooltip = je_space_race_interstellar_probe_first_outcome_tt
				add_modifier = { name = sr_first_interstellar_probe }
			}
		}
		event_outcome_effect_desc = {
			header = je_space_race_interstellar_probe_subsequent_outcome_header
			effect = {
				custom_tooltip = je_space_race_interstellar_probe_subsequent_outcome_tt
				add_modifier = { name = sr_interstellar_probe }
			}
		}
	}

	outcome = {
		header = je_space_race_interstellar_probe_failure_header
		effect_header = je_space_race_interstellar_probe_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_space_race_interstellar_probe_fail_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = sr_space_program_incompatible_tt
				NOT = { modifier:country_sr_interstellar_mission_program_bool = yes }
			}
		}
		effect = {
			sr_report_program_loss = { MILESTONE = interstellar_probe }
			custom_tooltip = {
				text = sr_milestone_fail_cleanup_tt
				sr_cleanup_milestone_variables_effect = { MILESTONE = interstellar_probe }
			}
		}
		event_outcome_effect_desc = {
			header = je_space_race_interstellar_probe_setback_header
			effect = {
				custom_tooltip = je_space_race_interstellar_probe_setback_tt
			}
		}
	}

	on_monthly_pulse = {
		effect = { sr_monthly_progress_update_effect = { MILESTONE = interstellar_probe } }
	}

	weight = 90
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = yes
	should_update_on_player_command = yes
}

# ===========================================
# 8. SOLAR SYSTEM COLONIZATION (Staged, Repeatable)
# Prereq: Moon Landing + Solar Colonization PM (Cislunar Infrastructure, era 10);
# worlds beyond the Moon need the Mars Landing too (lunar colonization design § 4)
# Stages 3-5 additionally require Deep Space Exploration PM (era 12)
# 6 stages of colony establishment across the solar system
# Stage 0: The Moon
# Stage 1: Mars & Asteroids
# Stage 2: Jupiter System & Venus
# Stage 3: Mercury & Saturn System
# Stage 4: Uranus & Neptune Systems
# Stage 5: Kuiper Belt & Beyond
# Launch windows (Global Commons § 5): each world but Mercury is reached only
# in its launch window until Asteroid Mining (off-window at a premium) or
# fusion drives (no windows); see sr_launch_window_effects.txt.
# ===========================================
je_space_race_solar_colonization = {
	icon = "gfx/interface/icons/event_icons/je_space_race_solar_colonization.dds"
	group = je_group_foreign_affairs

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	# The player-facing surface (gui/journal_entry_widgets/space_race_widget.gui):
	# the overview above the status line, the live sections below it, How the
	# Space Race Works last. The vanilla button grid below is hidden from humans
	# (every space race button carries `is_ai = yes`), so every action it offered
	# is offered in the live sections, through
	# common/scripted_guis/space_race_sguis.txt.
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_solar_colonization_overview"
		container = "custom_widget_container_1"
	}
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_solar_colonization"
		container = "custom_widget_container_2"
	}
	# MARKER, not content (gui/journal_entry_widgets/te_je_bars_on_top_marker.gui):
	# the mod's gui/journal_entry.gui hides the entry's own goal bar at the foot
	# of the panel, because the overview above draws the same progress.
	widget = {
		gui = "gui/journal_entry_widgets/te_je_bars_on_top_marker.gui"
		name = "widget_te_je_bars_on_top_marker"
		container = "custom_widget_container_7"
	}

	# 1.15: with the rule off, drop the inactive record for good (a rule never changes mid-game).
	disable_evaluation = {
		NOT = { sr_space_race_on = yes }
	}

	# Shown to a country that can colonize, and to any country holding a
	# colony, one given by treaty or inherited from a dead holder included
	# (colony record design § 8): the entry holds the colony's modifiers.
	is_shown_when_inactive = {
		sr_space_race_on = yes
		OR = {
			sr_has_space_colony = yes
			# The Moon Landing and pm_solar_colonization's technology (lunar
			# colonization design § 4): the Moon opens first, the worlds beyond
			# it with the Mars Landing (sr_colonization_beyond_moon).
			AND = {
				has_variable = sr_completed_moon_landing
				has_technology_researched = cislunar_infrastructure
			}
		}
	}

	possible = {
		OR = {
			# Has established colonies — JE stays active to display their modifiers
			# (founded, given by treaty, or inherited)
			custom_tooltip = {
				text = sr_has_existing_colonies_tt
				sr_has_space_colony = yes
			}
			# Can actively colonize: the Moon Landing (lunar colonization design
			# § 4); the Mars Landing gates each world beyond the Moon, not the entry.
			AND = {
				has_variable = sr_completed_moon_landing
				modifier:country_sr_solar_colonization_program_bool = yes
				# Stages 3-5 (outer system) require Deep Space Exploration PM (space_colonization tech, era 12):
				# past stage 2 under races only, once no inner world is free under
				# simple and full (sr_colonization_inner_open). A free lunar world
				# needs neither, at any stage (review I1: Races Only past stage 2
				# with a lunar world free); stages 3-5 stay gated world by world.
				OR = {
					custom_tooltip = {
						text = sr_lunar_world_free_tt
						sr_sites_lunar_open = yes
					}
					custom_tooltip = {
						text = sr_outer_system_requires_deep_space_tt
						OR = {
							sr_colonization_inner_open = yes
							modifier:country_sr_deep_space_exploration_program_bool = yes
						}
					}
				}
				# A world left to claim: never founded, or released by a dead
				# holder with no heir (sr_sites_generated.txt).
				custom_tooltip = {
					text = sr_world_left_to_claim_tt
					sr_any_site_unclaimed = yes
				}
				# With the race over and a lunar world free, only by choice
				# (sr_colonization_by_choice, written so its line shows only
				# where it applies; owner, 2026-10-29).
				trigger_if = {
					limit = { sr_race_over_moon_free = yes }
					custom_tooltip = {
						text = sr_colonization_by_choice_tt
						has_variable = sr_colonization_resumed
					}
				}
				is_great_or_major_power = yes
			}
		}
	}

	# The milestone's cost and funding (#1404; space_race_values.txt), each
	# while it is billed (sr_milestone_billed_solar_colonization: its programme
	# running, not merely its entry open for the colonies):
	# its cost level (funding plus the approach's overhead, times the era
	# factor), its pace at the effective funding level (soft cap) and its
	# throughput at the level itself. Every running milestone's entry carries
	# its own, so they add up across entries.
	scaling_modifiers_while_active = {
		modifier = sr_space_program_cost
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_solar_colonization = yes }
				add = sr_solar_colonization_cost
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_funding_progress
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_solar_colonization = yes }
				add = sr_funding_solar_colonization_effective
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_funding_throughput
		value = {
			value = 0
			if = {
				limit = { sr_milestone_billed_solar_colonization = yes }
				add = sr_funding_solar_colonization_level
			}
		}
	}
	# The approach, while its variable is set (sr_effect_safe / sr_effect_ambitious;
	# Standard is neither).
	scaling_modifiers_while_active = {
		modifier = sr_safe_approach
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_safe_solar_colonization }
				add = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_ambitious_approach
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_ambitious_solar_colonization }
				add = 1
			}
		}
	}

	immediate = {
		# A finished programme stays open to hold the colony modifiers, and a
		# revolution's winner inherits it that way. Running this block again
		# must not re-open it: it would bill its cost until a bar fill closed
		# it a second time, which never comes with the production method off.
		if = {
			limit = {
				OR = {
					NOT = { has_variable = sr_completed_solar_colonization }
					# resumed for the Moon (sr_resume_solar_colonization), whose
					# entry had closed: it runs again by choice
					has_variable = sr_colonization_resumed
				}
			}
			set_variable = { name = sr_active_solar_colonization value = yes }
			# Its band's offset from the start, so another country's estimate (National Info) never sits on the exact figure
			sr_power_fuzz_first = { MILESTONE = solar_colonization }
			# Progress survives only while a colony is held. Then `possible`'s
			# first branch always holds, so the entry never deactivates, and
			# the only re-run is a revolution's inheritance, which keeps the
			# bar (see the file header). With no colony the entry can drop out
			# and come back within a month, carrying its progress in as the
			# bar's baseline; this bar restarts at every colony, so it would
			# read negative for the rest of the game. So a colony-less
			# programme restarts, as it always did, and a revolution costs it
			# the progress towards its first colony. The whole fleets it had
			# staged in orbit are kept apart first and come back with its next
			# launch month (sr_launch_bank_staged_fleets, Global Commons § 5.2).
			if = { # REVIEWED 2026-09-26: deliberate reset of a bar input, see above
				limit = {
					OR = {
						NOT = { has_variable = sr_progress_solar_colonization }
						NOT = { sr_has_space_colony = yes }
					}
				}
				sr_launch_bank_staged_fleets = yes # REVIEWED 2026-10-29: banks the whole fleets staged now, and the reset below always follows, so a re-run moves only what was staged since and never counts a fleet twice. Flagged once the outer guard became NOT completed OR resumed (lunar colonization 2b)
				set_variable = { name = sr_progress_solar_colonization value = 0 }
			}
			if = { limit = { NOT = { has_variable = sr_funding_solar_colonization } } set_variable = { name = sr_funding_solar_colonization value = 1 } }
		}
		# The Moon is stage 0 (lunar colonization design § 3.2).
		if = {
			limit = { NOT = { has_global_variable = sr_colonization_stage } }
			set_global_variable = { name = sr_colonization_stage value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = sr_colony_count } }
			set_variable = { name = sr_colony_count value = 0 }
		}
		sr_refresh_milestone_statuses = yes
	}

	scripted_button = sr_btn_safe_solar_colonization
	scripted_button = sr_btn_ambitious_solar_colonization
	scripted_button = sr_btn_fund_up_solar_colonization
	scripted_button = sr_btn_fund_down_solar_colonization

	current_value = {
		value = sr_progress_solar_colonization_percent
	}

	goal_add_value = {
		value = 100
	}

	# Kept for the journal list (the entry's own goal bar is hidden in the panel
	# by the bars-on-top marker above; the overview draws the progress).
	progressbar = yes

	# Only while the entry is inactive. An active entry's overview shows its
	# state, and the reason text below repeats the rest.
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_space_race_solar_colonization_status
				trigger = { NOT = { has_journal_entry = je_space_race_solar_colonization } }
			}
		}
	}

	# No success outcome: each colony is established in on_monthly_pulse, so the
	# entry stays open for the next one.
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_space_race_solar_colonization_fail_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = sr_space_program_incompatible_tt
				NOT = { modifier:country_sr_solar_colonization_program_bool = yes }
			}
			# Only close if no colonies were established (otherwise JE stays to hold modifiers)
			custom_tooltip = {
				text = sr_no_colonies_to_maintain_tt
				OR = {
					NOT = { has_variable = sr_colony_count }
					var:sr_colony_count < 1
				}
			}
		}
		effect = {
			# Only a programme the country could run: a country that held only a
			# colony it was given (no Moon Landing) has none to lose.
			if = {
				limit = { sr_solar_colonization_qualified = yes }
				sr_report_program_loss = { MILESTONE = solar_colonization }
			}
			custom_tooltip = {
				text = sr_milestone_fail_cleanup_tt
				# Fleets staged in orbit wait for the programme to restart.
				sr_launch_bank_staged_fleets = yes
				sr_cleanup_milestone_variables_effect = { MILESTONE = solar_colonization }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			# Skip colonization logic if completed or PM disabled (passive mode — JE stays for modifiers).
			# The same test as billing and the controls: a programme that would
			# colonize the Moon after the race is over without its country's
			# choice doesn't run (sr_colonization_by_choice).
			if = {
				limit = { sr_solar_colonization_running = yes }
				if = {
					limit = { sr_ai_should_participate = yes }
					# Every approach moves and rolls; Standard is neither variable.
					change_variable = { name = sr_progress_solar_colonization add = sr_progress }
					random = {
						chance = sr_risk_pct_solar_colonization
						set_variable = { name = sr_failed_solar_colonization value = yes }
						sr_count_setback_base = { MILESTONE = solar_colonization }
						if = {
							limit = { has_variable = sr_ambitious_solar_colonization }
							sr_fire_ambitious_failure_event = { MILESTONE = solar_colonization }
						}
						else = {
							sr_fire_safe_setback_event = { MILESTONE = solar_colonization }
						}
					}
					# The month's launch (Global Commons § 5.2, sr_launch_window_effects.txt):
					# the bar fills past the goal with no cap, and progress past it is
					# fleets staged in orbit. One colony a month at most, founded when
					# staged progress covers goal x (1 + premium) for the stage's cheapest
					# site that can depart: in its launch window, any month with fusion
					# drives, or outside it at the premium with Asteroid Mining when the
					# approach's test passes. The overflow is kept, as it always was.
					sr_launch_month = yes
				}
			}
			# Once every world is claimed, the programme is over for every colony
			# holder, not only the one that took the last world: mark it complete
			# and keep the entry open for the modifiers. Checked every month, not
			# on a bar fill, so another holder does not go on paying its funding
			# and approach for a bar that can found nothing (until 2026-10-03 it
			# finished only when that bar next filled, years later), and one with
			# its production method off finishes too. A finished programme is what
			# lets the Commercial Space Industry dividend start
			# (sr_space_race_concluded). A country that holds no colony fails
			# `possible` the moment the last site is claimed, and
			# `can_deactivate = yes` drops the entry back to inactive;
			# sr_cleanup_inactive_space_race_milestones then clears its
			# variables. It is not paid, and not pinned open.
			# A programme resumed for the Moon (sr_colonization_resumed; lunar
			# colonization 2b, owner 2026-10-29) goes on while a lunar world is
			# free, and stops here once the Moon is full; one that didn't choose
			# it finishes as before. A resumed one may already hold its record,
			# and the marker goes. The record is what the entry declares
			# Interplanetary Trade Networks from (sr_solar_system_trade, below).
			if = {
				limit = {
					has_variable = sr_active_solar_colonization
					has_global_variable = sr_all_colonies_complete
					sr_has_space_colony = yes
					OR = {
						NOT = { has_variable = sr_colonization_resumed }
						NOT = { sr_sites_lunar_open = yes }
					}
				}
				set_variable = { name = sr_completed_solar_colonization value = yes }
				if = { limit = { has_variable = sr_colonization_resumed } remove_variable = sr_colonization_resumed }
				remove_variable = sr_active_solar_colonization
				if = { limit = { has_variable = sr_funding_solar_colonization } remove_variable = sr_funding_solar_colonization }
				if = { limit = { has_variable = sr_safe_solar_colonization } remove_variable = sr_safe_solar_colonization }
				if = { limit = { has_variable = sr_ambitious_solar_colonization } remove_variable = sr_ambitious_solar_colonization }
				if = { limit = { has_variable = sr_progress_solar_colonization } remove_variable = sr_progress_solar_colonization }
				sr_complete_milestone_effect = yes
			}
			sr_set_milestone_status_base = { MILESTONE = solar_colonization }
			# Colony population (simple and full; slices 2-3 design § 10): the
			# AI's yearly review, the month's emigrants; colony modules (colony
			# modules design §§ 5, 10): the AI's yearly review, the month's
			# fabrication; autonomy and free colonies (slice 5 design §§ 3, 7):
			# event worlds left stale cleared, the AI's yearly supply review, the
			# month's colony life. A colony holder's entry is always open, so
			# this runs for every holder, programme or not. The entry's modifiers
			# follow what these change by themselves (declared below).
			sr_emig_ai_review = yes
			sr_emig_month = yes
			sr_mod_ai_review = yes
			sr_mod_month = yes
			sr_site_events_clear_stale = yes
			sr_free_ai_review = yes
			sr_life_month = yes
		}
	}

	# The entry's own modifiers (1.15 scaling_modifiers_while_active, #1404;
	# docs/superpowers/specs/2026-11-06-scaling-modifiers-design.md § 4.10). Each
	# value's root is the owning country; each is 0 outside the gate the monthly
	# re-add used to sit behind. They apply only while the entry is active, which
	# a colony holder's always is (`possible`). has_modifier cannot see them: test
	# the inputs. Each world's specializations and discovery outcomes, and the
	# facilities, are declared in common/journal_entries/sr_sites_generated.txt
	# (generated, scripts/generators/gen_solar_sites.py).
	# Interplanetary Trade Networks, once every world is claimed (the monthly
	# pulse above records it).
	scaling_modifiers_while_active = {
		modifier = sr_solar_system_trade
		value = {
			value = 0
			if = {
				limit = { has_variable = sr_completed_solar_colonization }
				add = 1
			}
		}
	}
	# Colony population (simple and full colonies; slices 2-3 design § 10): the
	# emigration programme's upkeep, a unit a level of it.
	scaling_modifiers_while_active = {
		modifier = sr_emigration_programme
		value = {
			value = 0
			if = {
				limit = {
					sr_colonies_contested_enabled = yes
					sr_emig_level_value >= 1
				}
				add = sr_emig_level_value
			}
		}
	}
	# The nation's people off Earth, a unit per 100,000 (capped), nothing below
	# the minimum (design § 10.5).
	scaling_modifiers_while_active = {
		modifier = sr_offworld_population
		value = {
			value = 0
			if = {
				limit = {
					sr_colonies_contested_enabled = yes
					sr_offworld_pop_total >= sr_offworld_bonus_minimum
				}
				add = sr_offworld_units
			}
		}
	}
	# The module levels past the first at the colonies held (colony modules
	# design § 8).
	scaling_modifiers_while_active = {
		modifier = sr_module_upkeep
		value = {
			value = 0
			if = {
				limit = {
					sr_colonies_contested_enabled = yes
					sr_mod_upkeep_levels >= 1
				}
				add = sr_mod_upkeep_levels
			}
		}
	}
	# Autonomy and free colonies (slice 5 design §§ 2-3): a unit a free colony
	# supplied, the trade from them, a unit a garrisoned colony.
	scaling_modifiers_while_active = {
		modifier = sr_free_colony_supply
		value = {
			value = 0
			if = {
				limit = {
					sr_colonies_contested_enabled = yes
					sr_free_supply_count >= 1
				}
				add = sr_free_supply_count
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_free_colony_trade
		value = {
			value = 0
			if = {
				limit = {
					sr_colonies_contested_enabled = yes
					sr_free_trade_units > 0
				}
				add = sr_free_trade_units
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = sr_colonial_garrisons
		value = {
			value = 0
			if = {
				limit = {
					sr_colonies_contested_enabled = yes
					sr_garrison_count >= 1
				}
				add = sr_garrison_count
			}
		}
	}
