############################################
# Space Race Journal Entries
# Linear milestone competition system
#
# `immediate` runs again: when an entry re-activates (every milestone has
# can_deactivate = yes), and on every entry a revolution's winner inherits,
# after the loser's variables have merged in
# (docs/guides/scripting_best_practices.md § "What a Civil War's Winner
# Inherits"). So it creates only what is missing (progress, funding), puts
# back the modifiers the new record lacks (sr_sync_<m>_entry in
# space_race_effects.txt), and asks each choice event once per milestone
# (sr_<m>_choice_made).
#
# Carrying progress into a new record keeps the progress bar right:
# - Inherited: the new record keeps the loser's bar. Its start date, baseline
#   and goal are copied, not evaluated again (German-revolution saves,
#   2026-09-25: the winner's je_global_warming kept baseline 0.1 and goal 4.0
#   with the anomaly at 1.18).
# - Re-activated before the next monthly pulse, which clears a deactivated
#   milestone's progress (sr_cleanup_inactive_space_race_milestones): the goal
#   is evaluated again, as current_value + goal_add_value, with the progress
#   in current_value. So goal_add_value subtracts the progress, which pins the
#   goal at sr_<m>_goal, and the bar fills from there. This reads like a bug
#   and is not (scripting_best_practices.md § "A Journal Entry's Goal Is
#   Frozen at Activation").
# Solar colonization's bar restarts at every colony, so it keeps its progress
# only when it cannot deactivate: see its `immediate`.
############################################

# ===========================================
# 1. SUBORBITAL FLIGHT
# Prereq: rocketry tech, GP/MP rank
# ===========================================
je_space_race_suborbital = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	group = je_group_foreign_affairs

	# The player-facing surface. The vanilla button grid below is hidden from
	# humans (every space race button carries `is_ai = yes`), so every action it
	# offered is offered here, through common/scripted_guis/space_race_sguis.txt.
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_suborbital"
		container = "custom_widget_container_2"
	}

	is_shown_when_inactive = {
		has_game_rule = space_race_enabled
		has_technology_researched = rocketry
		is_great_or_major_power = yes
		# The rebels of a running revolution start no programme of their own.
		# They hold none of the country's variables, so they would begin again
		# from suborbital flight. If they won, their own progress and funding
		# would beat the loser's in the variable merge, and a run the loser had
		# already completed would reopen. Only suborbital needs this: every
		# later entry requires a sr_completed_<m> the rebels do not hold. Not in
		# `possible`, which an inherited running entry re-checks
		# (can_deactivate = yes). Vanilla precedent:
		# 02_south_american_national_identity.txt.
		is_revolutionary = no
	}

	possible = {
		has_technology_researched = rocketry
		modifier:country_sr_earth_orbit_program_bool = yes
		is_great_or_major_power = yes
		NOT = { has_variable = sr_completed_suborbital }
	}

	immediate = {
		set_variable = { name = sr_active_suborbital value = yes }
		set_variable = { name = sr_active_milestone value = yes }
		if = { limit = { NOT = { has_variable = sr_progress_suborbital } } set_variable = { name = sr_progress_suborbital value = 0 } }
		if = { limit = { NOT = { has_variable = sr_funding_suborbital } } set_variable = { name = sr_funding_suborbital value = 1 } }
		sr_sync_suborbital_entry = yes
		sr_recalculate_cost = yes
	}

	scripted_button = sr_btn_safe_suborbital
	scripted_button = sr_btn_ambitious_suborbital
	scripted_button = sr_btn_fund_up_suborbital
	scripted_button = sr_btn_fund_down_suborbital

	current_value = {
		value = sr_progress_suborbital_value
	}

	# Pins the goal at sr_suborbital_goal: see the file header.
	goal_add_value = {
		value = sr_suborbital_goal
		subtract = sr_progress_suborbital_value
	}

	progressbar = yes

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_space_race_suborbital_status
				trigger = { always = yes }
			}
		}
	}

	complete = {
		has_variable = sr_progress_suborbital
		var:sr_progress_suborbital >= sr_suborbital_goal
	}

	on_complete = {
		sr_complete_suborbital_effect = yes
		remove_variable = sr_active_suborbital
		trigger_event = { id = space_race_events.1 }
	}

	fail = {
		custom_tooltip = {
			text = sr_space_program_incompatible_tt
			NOT = { modifier:country_sr_earth_orbit_program_bool = yes }
		}
	}

	on_fail = {
		sr_report_program_loss = { MILESTONE = suborbital }
		custom_tooltip = {
			text = sr_milestone_fail_cleanup_tt
			sr_cleanup_milestone_variables_effect = { MILESTONE = suborbital }
		}
	}

	on_monthly_pulse = {
		effect = { sr_monthly_progress_update_effect = { MILESTONE = suborbital } }
	}

	# Outcome previews — engine auto-tooltips the effect block; effects do not run.
	event_outcome_completed_effect_desc = {
		header = je_space_race_suborbital_first_outcome_header
		effect = {
			custom_tooltip = je_space_race_suborbital_first_outcome_tt
			add_modifier = { name = sr_first_suborbital }
		}
	}
	event_outcome_completed_effect_desc = {
		header = je_space_race_suborbital_subsequent_outcome_header
		effect = {
			custom_tooltip = je_space_race_suborbital_subsequent_outcome_tt
			add_modifier = { name = sr_suborbital }
		}
	}
	event_outcome_failed_effect_desc = {
		header = je_space_race_suborbital_setback_header
		effect = {
			custom_tooltip = je_space_race_suborbital_setback_tt
		}
	}

	custom_completion_header    = je_space_race_suborbital_completion_header
	custom_failure_header       = je_space_race_suborbital_failure_header
	custom_on_completion_header = je_space_race_suborbital_on_completion_header
	custom_on_failure_header    = je_space_race_suborbital_on_failure_header

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = yes
	should_update_on_player_command = yes
}

# 8. SOLAR SYSTEM COLONIZATION (Staged, Repeatable)
# Prereq: Moon Base + Mars Landing + Solar Colonization PM (era 11)
# Stages 3-5 additionally require Deep Space Exploration PM (era 12)
# 5 stages of colony establishment across the solar system
# Stage 1: Mars & Asteroids
# Stage 2: Jupiter System & Venus
# Stage 3: Mercury & Saturn System
# Stage 4: Uranus & Neptune Systems
# Stage 5: Kuiper Belt & Beyond
# ===========================================
je_space_race_solar_colonization = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	group = je_group_foreign_affairs

	# The player-facing surface. The vanilla button grid below is hidden from
	# humans (every space race button carries `is_ai = yes`), so every action it
	# offered is offered here, through common/scripted_guis/space_race_sguis.txt.
	widget = {
		gui = "gui/journal_entry_widgets/space_race_widget.gui"
		name = "widget_je_space_race_solar_colonization"
		container = "custom_widget_container_2"
	}

	is_shown_when_inactive = {
		has_game_rule = space_race_enabled
		has_variable = sr_completed_moon_base
		has_variable = sr_completed_mars_landing
		has_technology_researched = directed_energy_weapons
	}

	possible = {
		has_variable = sr_completed_moon_base
		has_variable = sr_completed_mars_landing
		OR = {
			# Has established colonies — JE stays active to display their modifiers
			custom_tooltip = {
				text = sr_has_existing_colonies_tt
				has_variable = sr_colony_count
				var:sr_colony_count >= 1
			}
			# Can actively colonize (original prerequisites)
			AND = {
				modifier:country_sr_solar_colonization_program_bool = yes
				# Stages 3-5 (outer system) require Deep Space Exploration PM (space_colonization tech, era 12)
				custom_tooltip = {
					text = sr_outer_system_requires_deep_space_tt
					OR = {
						NOT = { has_global_variable = sr_colonization_stage }
						global_var:sr_colonization_stage <= 2
						modifier:country_sr_deep_space_exploration_program_bool = yes
					}
				}
				NOT = { has_global_variable = sr_all_colonies_complete }
				is_great_or_major_power = yes
			}
		}
	}

	immediate = {
		# A finished programme stays open to hold the colony modifiers, and a
		# revolution's winner inherits it that way. Running this block again
		# must not re-open it: it would bill its cost until a bar fill closed
		# it a second time, which never comes with the production method off.
		if = {
			limit = { NOT = { has_variable = sr_completed_solar_colonization } }
			set_variable = { name = sr_active_solar_colonization value = yes }
			set_variable = { name = sr_active_milestone value = yes }
			# Progress survives only while a colony is held. Then `possible`'s
			# first branch always holds, so the entry never deactivates, and
			# the only re-run is a revolution's inheritance, which keeps the
			# bar (see the file header). With no colony the entry can drop out
			# and come back within a month, carrying its progress in as the
			# bar's baseline; this bar restarts at every colony, so it would
			# read negative for the rest of the game. So a colony-less
			# programme restarts, as it always did, and a revolution costs it
			# the progress towards its first colony.
			if = { # REVIEWED 2026-09-26: deliberate reset of a bar input, see above
				limit = {
					OR = {
						NOT = { has_variable = sr_progress_solar_colonization }
						NOT = { sr_has_space_colony = yes }
					}
				}
				set_variable = { name = sr_progress_solar_colonization value = 0 }
			}
			if = { limit = { NOT = { has_variable = sr_funding_solar_colonization } } set_variable = { name = sr_funding_solar_colonization value = 1 } }
		}
		sr_sync_solar_colonization_entry = yes
		if = {
			limit = { NOT = { has_global_variable = sr_colonization_stage } }
			set_global_variable = { name = sr_colonization_stage value = 1 }
		}
		if = {
			limit = { NOT = { has_variable = sr_colony_count } }
			set_variable = { name = sr_colony_count value = 0 }
		}
		sr_recalculate_cost = yes
	}

	scripted_button = sr_btn_safe_solar_colonization
	scripted_button = sr_btn_ambitious_solar_colonization
	scripted_button = sr_btn_fund_up_solar_colonization
	scripted_button = sr_btn_fund_down_solar_colonization

	current_value = {
		value = sr_progress_solar_colonization_percent
	}

	goal_add_value = {
		value = 100
	}

	progressbar = yes

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_space_race_solar_colonization_status
				trigger = { always = yes }
			}
		}
	}

	complete = {
		# Never completes through the bar - completion is handled in on_monthly_pulse
		# This allows the JE to stay open for repeating colony establishment
		always = no
	}

	on_complete = {
		# Display-only: never fires because complete = always = no
		custom_tooltip = {
			text = sr_solar_colony_established_tt
		}
	}

	fail = {
		custom_tooltip = {
			text = sr_space_program_incompatible_tt
			NOT = { modifier:country_sr_solar_colonization_program_bool = yes }
		}
		# Only close if no colonies were established (otherwise JE stays to hold modifiers)
		custom_tooltip = {
			text = sr_no_colonies_to_maintain_tt
			OR = {
				NOT = { has_variable = sr_colony_count }
				var:sr_colony_count < 1
			}
		}
	}

	on_fail = {
		sr_report_program_loss = { MILESTONE = solar_colonization }
		custom_tooltip = {
			text = sr_milestone_fail_cleanup_tt
			sr_cleanup_milestone_variables_effect = { MILESTONE = solar_colonization }
		}
	}

	on_monthly_pulse = {
		effect = {
			# Skip colonization logic if completed or PM disabled (passive mode — JE stays for modifiers)
			if = {
				limit = {
					has_variable = sr_active_solar_colonization
					modifier:country_sr_solar_colonization_program_bool = yes
				}
				if = {
					limit = { sr_ai_should_participate = yes }
					if = {
						limit = {
							OR = {
								has_variable = sr_safe_solar_colonization
								has_variable = sr_ambitious_solar_colonization
							}
						}
						change_variable = { name = sr_progress_solar_colonization add = sr_progress }
						if = {
							limit = { NOT = { sr_has_failure_cooldown = yes } }
							random = {
								chance = sr_risk_pct_solar_colonization
								set_variable = { name = sr_failed_solar_colonization value = yes }
								sr_count_setback_base = { MILESTONE = solar_colonization }
								if = {
									limit = { has_variable = sr_ambitious_solar_colonization }
									sr_fire_ambitious_failure_event = { MILESTONE = solar_colonization }
								}
								else = {
									sr_fire_safe_setback_event = { MILESTONE = solar_colonization }
								}
							}
						}
					}
					else = {
						change_variable = {
							name = sr_progress_solar_colonization
							add = sr_progress_drift
						}
					}
					# Bar-fill detection: when progress reaches goal, establish a colony and reset
					if = {
						limit = {
							has_variable = sr_progress_solar_colonization
							var:sr_progress_solar_colonization >= sr_solar_colonization_goal
						}
						# Save pre-stage-change goal before colony effect (which may advance the stage and raise the goal)
						set_variable = { name = sr_goal_before_colony value = sr_solar_colonization_goal }
						# Establish a colony at a random unclaimed location in the current stage
						sr_establish_colony_effect = yes
						# Reset progress (preserve overflow using the pre-advancement goal)
						change_variable = { name = sr_progress_solar_colonization subtract = var:sr_goal_before_colony }
						remove_variable = sr_goal_before_colony
						# If all stages exhausted, mark complete but keep JE alive for modifiers.
						# Only a country that holds a colony earns the all-colonies reward.
						# One that holds none fails `possible` the moment the last site is
						# claimed, and `can_deactivate = yes` drops the entry back to
						# inactive; sr_cleanup_inactive_space_race_milestones then clears
						# its variables. It is not paid, and not pinned open.
						if = {
							limit = {
								has_global_variable = sr_all_colonies_complete
								has_variable = sr_colony_count
								var:sr_colony_count >= 1
							}
							set_variable = { name = sr_completed_solar_colonization value = yes }
							je:je_space_race_solar_colonization = { add_modifier = { name = sr_solar_system_trade } }
							remove_variable = sr_active_solar_colonization
							if = { limit = { has_variable = sr_funding_solar_colonization } remove_variable = sr_funding_solar_colonization }
							if = { limit = { has_variable = sr_safe_solar_colonization } remove_variable = sr_safe_solar_colonization }
							if = { limit = { has_variable = sr_ambitious_solar_colonization } remove_variable = sr_ambitious_solar_colonization }
							if = { limit = { has_variable = sr_progress_solar_colonization } remove_variable = sr_progress_solar_colonization }
							sr_complete_milestone_effect = yes
						}
					}
				}
			}
			sr_set_milestone_status_base = { MILESTONE = solar_colonization }
		}
	}

	weight = 85
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	can_deactivate = yes
	should_update_on_player_command = yes
}
