				trigger = { NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } } }
			}

				trigger = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
			}
		}

			}
			triggered_desc = {
				desc = je_un_authority_weak

			}
			triggered_desc = {

			}
			triggered_desc = {
				desc = je_un_authority_strong
				trigger = { has_global_variable = un_authority global_var:un_authority >= 60 global_var:un_authority < 85 }
			}
			triggered_desc = {
				desc = je_un_authority_dominant
				trigger = { has_global_variable = un_authority global_var:un_authority >= 85 }
			}
		}

		# 2b. Where authority is heading: the target and this month's step. The
		# full breakdown is in the "Why authority is moving" widget below.
		first_valid = {
			triggered_desc = {

				trigger = { has_global_variable = un_authority_target }
			}
		}

	on_monthly_pulse = {
		effect = {
			# ---- Security Council cleanup (all countries with JE) ----
			# Strip SC seat when leaving the UN. Sub-GP rank no longer immediately
			# strips the seat — permanent members keep it for 10 years sub-GP
			# (eligibility check below). Non-permanent SC seats are no longer
			# created in the new model, but legacy saves may still have them.
			if = {
				limit = {
					je:je_united_nations ?= { has_modifier = un_security_council_modifier }
					NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
				}
				je:je_united_nations ?= { remove_modifier = un_security_council_modifier }
				je:je_united_nations ?= { remove_modifier = un_permanent_member_modifier }
			}
			# Strip non-permanent SC seats from sub-GP holders (legacy save migration).
			if = {
				limit = {
					je:je_united_nations ?= { has_modifier = un_security_council_modifier }
					NOT = { is_un_permanent_member = yes }
					NOT = { country_rank >= rank_value:great_power }
				}
				je:je_united_nations ?= { remove_modifier = un_security_council_modifier }
			}
			# Founding-member SC seat upgrade (legacy save migration).
			# Charter Option A used to grant un_security_council_modifier without
			# un_permanent_member_modifier. Upgrade founding-member GPs with an
			# orphaned SC seat to full permanent member, respecting the 5-seat cap.
			if = {
				limit = {
					je:je_united_nations ?= { has_modifier = un_security_council_modifier }
					je:je_united_nations ?= { has_modifier = un_founding_member_modifier }
					NOT = { is_un_permanent_member = yes }
					country_rank >= rank_value:great_power
					un_permanent_seat_count < 5
				}
				je:je_united_nations ?= { add_modifier = { name = un_permanent_member_modifier } }
			}

			# ---- Treaty obligation auto-enrollment ----
			# GPs auto-enrolled by treaty get a permanent seat only during the
			# founding window (matches the un_join_button rules).
			if = {
				limit = {
					NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
					modifier:country_un_membership_obligation_bool = yes
				}
				je:je_united_nations ?= { add_modifier = { name = un_member_modifier } }
				if = {
					limit = {
						country_rank >= rank_value:great_power
						has_global_variable = un_founding_window_active
						un_permanent_seat_count < 5
					}
					je:je_united_nations ?= { add_modifier = { name = un_security_council_modifier } }
					je:je_united_nations ?= { add_modifier = { name = un_permanent_member_modifier } }
				}
				# Bind the joiner to any conventions already ratified globally.
				un_apply_ratified_conventions = yes
			}

			# ---- Member-only logic ----
			if = {
				limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }

				# Authority moves ONCE globally per month, toward the target computed
				# from the seven pillars, in un_global_authority_on_action
				# (un_authority_monthly_update, un_authority_effects.txt). Nothing
				# per-member is applied here.

				# Refresh contributor-cost cached expense vars so the JE-scoped
				# add_modifier multipliers track current GDP. JE scope can't read
				# country properties like gdp directly inside multiplier blocks.
				if = {
					limit = { je:je_united_nations ?= { has_modifier = un_peacekeeping_contributor_modifier } }
					set_variable = { name = un_peacekeeping_expense_cached value = un_program_expense_value }
				}
				if = {
					limit = { je:je_united_nations ?= { has_modifier = un_development_contributor_modifier } }
					set_variable = { name = un_development_expense_cached value = un_program_expense_value }
				}

				# ---- Apply scaled UN membership modifier ----
				# Modifier stays applied per-member; the multiplier reads the shared
				# global authority. Members of a bloc with the Multilateral Institutions
				# principle scale the benefit further by (1 + country_un_institutional_alignment)
				# -- 0.5 alignment = +50% magnitude. Non-aligned members read 0, so x1.
				if = {
					limit = { global_var:un_authority >= 30 }
					je:je_united_nations ?= { remove_modifier = un_membership_benefits_modifier }
					je:je_united_nations ?= { add_modifier = {
							name = un_membership_benefits_modifier
							multiplier = {
								value = global_var:un_authority
								divide = 50
								multiply = {
									value = 1
									add = modifier:country_un_institutional_alignment
								}
							} }
					}
				}
				else = {
					je:je_united_nations ?= { remove_modifier = un_membership_benefits_modifier }
				}

				# ---- High authority: infamy generation scales for everyone ----
				if = {
					limit = { global_var:un_authority >= 50 }
					je:je_united_nations ?= { remove_modifier = un_high_authority_infamy_modifier }
					je:je_united_nations ?= { add_modifier = {
							name = un_high_authority_infamy_modifier
							multiplier = {
								value = global_var:un_authority
								subtract = 40
								divide = 30
							} }
					}
				}
				else = {
					je:je_united_nations ?= { remove_modifier = un_high_authority_infamy_modifier }
				}

				# ---- NPT enforcement tiers (requires IAEA agency) ----
				# Authority 80+: Non-nuclear member powers face disarmament pressure
				if = {
					limit = {
						has_global_variable = un_agency_iaea
						global_var:un_authority >= 80
						NOT = { has_modifier = nuclear_power }
						NOT = { je:je_united_nations ?= { has_modifier = un_npt_disarmament_modifier } }
					}
					je:je_united_nations ?= { add_modifier = { name = un_npt_disarmament_modifier } }
				}
				# Remove if authority drops or they become nuclear
				if = {
					limit = {
						je:je_united_nations ?= { has_modifier = un_npt_disarmament_modifier }
						OR = {
							global_var:un_authority < 75
							has_modifier = nuclear_power
							NOT = { has_global_variable = un_agency_iaea }
						}
					}
					je:je_united_nations ?= { remove_modifier = un_npt_disarmament_modifier }
				}

				# ---- Permanent Member eligibility re-check ----
				# Permanent seats are granted via un_join_button during the founding
				# window. Here we track sub-GP duration: a permanent member that
				# falls below GP rank for 10+ years (120 months) loses the seat.
				# (Captures real instability — e.g., a former GP whose collapse is
				# permanent eventually loses its veto. See Q2 in the design doc.)
				if = {
					limit = {
						is_un_permanent_member = yes
						NOT = { country_rank >= rank_value:great_power }
					}
					if = {
						limit = { NOT = { has_variable = un_permanent_subgp_months } }
						set_variable = { name = un_permanent_subgp_months value = 0 }
					}
					change_variable = { name = un_permanent_subgp_months add = 1 }
				}
				# Reset counter when permanent member regains GP rank
				if = {
					limit = {
						is_un_permanent_member = yes
						country_rank >= rank_value:great_power
						has_variable = un_permanent_subgp_months
					}
					remove_variable = un_permanent_subgp_months
				}
				# Strip permanent status after sustained sub-GP rank
				if = {
					limit = { un_permanent_member_eligibility_lost = yes }
					je:je_united_nations ?= { remove_modifier = un_permanent_member_modifier }
					je:je_united_nations ?= { remove_modifier = un_security_council_modifier }
					if = {
						limit = { has_variable = un_permanent_subgp_months }
						remove_variable = un_permanent_subgp_months
					}
				}
			}

			# Sync progress bar to the single global authority value for ALL JE holders
			# (members and non-members alike — the bar is now a global fact). Needs
			# scope:journal_entry, which only the per-country pulse has, so it stays
			# here rather than in the global updater.
			if = {
				limit = { has_global_variable = un_authority }
				scope:journal_entry = {
					set_bar_progress = { name = un_authority_bar value = global_var:un_authority }
				}
			}

			# ---- Non-member penalties at high authority (all countries with JE) ----
			if = {
				limit = {
					NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
					has_global_variable = un_founded
					# UN authority (single global value) is high enough to shun outsiders.
					has_global_variable = un_authority
					global_var:un_authority >= 60
				}
				# Apply pariah status: non-members face diplomatic pressure
				if = {
					limit = { NOT = { je:je_united_nations ?= { has_modifier = un_nonmember_pariah_modifier } } }
					je:je_united_nations ?= { add_modifier = { name = un_nonmember_pariah_modifier } }
				}
			}
			# Remove if authority drops or UN not founded or they join
			if = {
				limit = {
					je:je_united_nations ?= { has_modifier = un_nonmember_pariah_modifier }
					OR = {
						je:je_united_nations ?= { has_modifier = un_member_modifier }
						NOT = { has_global_variable = un_founded }
						NOT = {
							AND = {
								has_global_variable = un_authority
								global_var:un_authority >= 55
							}
						}
					}
				}
				je:je_united_nations ?= { remove_modifier = un_nonmember_pariah_modifier }
			}

			# ---- Authority history (the chart in the authority widget) ----
			un_authority_record_history = yes

			# ---- International standing (all countries with the JE) ----
			# The whole national-standing system's monthly work, in one call:
			# seed a new member at the neutral value, advance the rolling year,
			# tick the ongoing-programme and undermining counters, award the
			# capped programme accrual, and swap the tier modifier. This is the
			# SINGLE refresh site for the un_standing_*_modifier family
			# (common/scripted_effects/un_standing_effects.txt). It sits outside
			# the member-only block on purpose: a country that has left still
			# needs its counters dropped and its tier modifier removed.
			un_standing_country_pulse = yes
		}
	}

	event_outcome_activated_effect_desc = {

			add_modifier = { name = un_nonmember_pariah_modifier }
		}
	}
