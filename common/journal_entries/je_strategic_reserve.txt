# ============================================================================
# Strategic Reserve Journal Entry
# ============================================================================
# Activates automatically when a country builds a Strategic Reserve Hub.
# All per-good presentation and control lives in the inventory widget
# (gui/journal_entry_widgets/strategic_reserve_widget.gui): one row per
# unlocked good with its stock, capacity, fill bar, configured signed rate,
# actual net weekly movement, status label and decrease/stop/increase controls.
# Positive rates store goods, negative rates withdraw them. This JE keeps only
# the reserve-wide summary text, the two shared buttons (step size, reset all)
# and the lifecycle entry points into the SR wrapper effects; persistent
# reserve state lives on country vars.
# ============================================================================

je_strategic_reserve = {
	icon = "gfx/interface/icons/building_icons/building_government_administration.dds"
	group = je_group_internal_affairs

	is_shown_when_inactive = {
		has_technology_researched = logistics
	}

	possible = {
		any_scope_building = {
			is_building_type = building_strategic_reserve_hub
		}
	}

	immediate = {
		st_res_je_immediate_effect = yes
	}

	invalid = {
		NOT = {
			any_scope_building = {
				is_building_type = building_strategic_reserve_hub
			}
		}
	}

	on_invalid = {
		st_res_je_invalid_effect = yes
	}

	# Reserve inventory table: one row per unlocked good with its stock, fill
	# bar, configured rate, actual net weekly movement, status and controls.
	# container_2 puts it directly under the summary status text and above the
	# two shared buttons below.
	widget = {
		gui = "gui/journal_entry_widgets/strategic_reserve_widget.gui"
		name = "widget_je_strategic_reserve_inventory"
		container = "custom_widget_container_2"
	}

	# Per-good fill bars and the fourteen per-good rate buttons now live inside
	# the inventory widget above. Only the two shared, reserve-wide actions stay
	# as vanilla-rendered journal-entry buttons.
	scripted_button = st_res_cycle_step_size_button
	scripted_button = st_res_reset_rates_button

	# Summary only. The per-good stock / rate / status detail moved into the
	# inventory widget; status_desc also renders in the journal list, where
	# seven multi-line good blocks were pure noise.
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_strategic_reserve_status_no_hub
				trigger = {
					NOT = {
						any_scope_building = { is_building_type = building_strategic_reserve_hub }
					}
				}
			}
			triggered_desc = {
				desc = je_strategic_reserve_status_deactivated
				trigger = {
					var:st_res_hub_workforce_cached <= 0.99
				}
			}
			triggered_desc = {
				desc = je_strategic_reserve_status_active
				trigger = { always = yes }
			}
		}
		triggered_desc = {
			desc = je_strategic_reserve_profit_line
			trigger = { always = yes }
		}
		triggered_desc = {
			desc = je_strategic_reserve_rate_line
			trigger = { always = yes }
		}
	}

	# Weekly bookkeeping lives on country vars; the JE only provides the pulse.
	on_weekly_pulse = {
		effect = {
			st_res_je_weekly_pulse_effect = yes
		}
	}

	weight = 500
	should_be_pinned_by_default_uninvolved_or_context = yes
	should_update_on_player_command = yes
}
