	# `immediate` runs on the entry's first activation and again on every copy
	# of it that a revolution's winner inherits. That second run comes after the
	# loser's variables have been merged into the winner, and the inherited copy
	# arrives with NO modifiers (scripting_best_practices.md § "What a Civil
	# War's Winner Inherits"; docs/audits/civil_war_inheritance_audit.md E1,
	# E3). The entry cannot deactivate, so nothing else re-runs it.
	immediate = {
		# Cycle state. A country that meets the start test now starts its cycle
		# here (a new game's founders, a released country, a revolution's winner
		# whose loser had none running); the rest stay dormant until the
		# monthly pulse finds them unlocked. A winner that inherits a running
		# cycle already holds finance_cycle_value, so this is skipped and the
		# cycle carries on (#465, audit F6: these sets were once unguarded and
		# turned a panic into a stable 50). banking_cycle_start guards each of
		# the three variables itself.
		if = {
			limit = {
				NOT = { has_variable = finance_cycle_value }
				te_banking_cycle_unlocked = yes
			}
			banking_cycle_start = yes
		}

		# Deliberately UNGUARDED. te_mon_stance_band_applied records which
		# banking_stance_band_* sits on this entry, and at this point the entry
		# carries none: a first activation has no modifiers yet, and an inherited
		# copy arrives with an empty list but with the loser's tracker still
		# naming the loser's band. Left alone, that tracker matched the band,
		# no swap ran, and the band stayed off (audit F7). 0 = none recorded, so
		# next month's banking_cycle_apply_stance_band puts the band back on.
		set_variable = { name = te_mon_stance_band_applied value = 0 } # REVIEWED 2026-09-26: reset on purpose; a fresh or inherited entry holds no stance band, so the tracker must say none (civil_war_inheritance_audit.md F7)

		# An inherited copy can carry banking_bank_holiday_reopening without the
		# holiday it belongs to, which did not survive the revolution. The
		# holiday ended early, so there is no reopening bounce to pay
		# (banking_effect_cb_bank_holiday).
		if = {
			limit = {
				has_variable = banking_bank_holiday_reopening
				banking_tool_bank_holiday_active = no
			}
			remove_variable = banking_bank_holiday_reopening # REVIEWED 2026-10-05: cleared on purpose; the holiday this marker belongs to did not survive the revolution, so no reopening bounce is owed
		}

		# Likewise banking_contagion_pending, the mark that gives a country one
		# contagion event at a time (banking_contagion_mark_pending): the event
		# it stands for belonged to the loser, and went with it. The mark is
		# timed, but whether the merge keeps a timer is unverified, and a mark
		# kept for good would shut the country out of contagion for the game.
		# On a first activation there is none: the spread reaches only countries
		# whose cycle runs (te_banking_cycle_running).
		if = {
			limit = { has_variable = banking_contagion_pending }
			remove_variable = banking_contagion_pending # REVIEWED 2026-10-17: cleared on purpose; the contagion event this mark stands for did not survive the revolution, so the winner may be reached again
		}

		# The four bars, drawn from the variables above by the monthly pulse's
		# own effect (its momentum clamp, and its has_variable test on the
		# monetary system's te_mon_stance_bar_pos). On an inherited copy the
		# saves showed the loser's bar values surviving whatever this block
		# wrote (audit E3); either way the bars now agree with the variables.
		# A dormant entry has no readings, and draws no bars.
		if = {
			limit = { te_banking_cycle_running = yes }
			banking_cycle_update_progress_bars = yes
		}
	}

	# No progress line either (GUI audit BNK-7, 2026-10-27). It read "Free
	# Intervention Points: X/Y", the overview's Budget cell again, out of a
	# total that left out the banking union's point, and it showed while the
	# entry was dormant, when the overview hides the budget on purpose. Its one
	# unique part, the breakdown of the free points by grant and tool, is now
	# in the Budget cell's "free" hover (banking_dash_budget_free_tt).
	progressbar      = no

	# No status text: the overview shows everything it said (the owner's rule:
	# status text that repeats the overview is removed). The phase, momentum and
	# bubble pressure are its icons since play-test round 2, and the three
	# monthly modifier totals, each with the engine's breakdown on hover, moved
	# into the tooltips of the ranges under them (banking_dash_*_change_tt),
	# which the Budget tab shows too. A dormant entry's overview says the cycle
	# has not started, with what it needs on hover, so it needs no line either.

	# A dormant entry's pulses do one thing: the monthly one starts the cycle
	# once te_banking_cycle_unlocked holds (the technology is its first test, so
	# most dormant countries never reach the building scan), and posts the start
	# to a human player. Everything else runs only while the cycle does.
	on_monthly_pulse = {
		effect = {
			if = {
				limit = { te_banking_cycle_running = no }
				if = {
					limit = { te_banking_cycle_unlocked = yes }
					banking_cycle_start = yes
					if = {
						limit = { is_ai = no }
						post_notification = banking_cycle_started_notice
					}
				}
			}
			else = {
				# SAVE MIGRATION — added 2026-09-19, remove after one release.
				# The "Raise Policy Rate" tool was deleted (monetary_policy_design.md
				# §18) and with it the button that switched it off. A save carrying
				# the modifier would otherwise reserve 2 intervention points forever,
				# so strip it here.
				#
				# The points do not come back until next month: add_modifier and
				# remove_modifier results are not visible inside the effect block that
				# issues them, so every `modifier:country_banking_intervention_max_add`
				# read later in THIS pulse still sees the stale -2. The one such read is
				# banking_crash_check_overdrawn_interventions at the end of the block,
				# and the stale -2 makes it MORE likely to shed a tool this month, not
				# less — a one-off cost on the single migrating month, on a save that
				# was already two points down. Nothing else in the pulse reads the
				# modifier, so the position of this block within it is cosmetic.
				#
				# The eventual removal of the static modifier itself is tracked in
				# common/scripted_effects/legacy_modifier_cleanup.txt.
				if = {
					limit = { je:je_banking_cycle ?= { has_modifier = banking_policy_rate_hike } }
					je:je_banking_cycle = { remove_modifier = banking_policy_rate_hike }
				}

				banking_cycle_update_fiscal_policy = yes
				banking_cycle_bank_holiday_reopen = yes
				banking_cycle_advance_variables = yes
				banking_cycle_check_and_execute_crash = yes
				banking_cycle_eliq_wind_down = yes
				# A conditional rescue's reviews (#800, banking_debt_effects.txt).
				banking_rescue_programme_monthly = yes

				# Sample the month's settled cycle state for the history charts.
				# Runs after the crash check so a crash month's bar shows the
				# post-crash values next to its crash marker.
				te_history_record_banking_samples = yes

				set_variable = { name = banking_points_max_from_law value = banking_law_base_points_value }

				banking_cycle_random_event_effect = yes
				banking_cycle_apply_phase_modifiers = yes
				banking_cycle_apply_stance_band = yes
				banking_cycle_update_progress_bars = yes
				banking_cycle_cleanup_capital_controls = yes
				banking_directed_credit_enforce_cap = yes
				banking_cycle_update_ce_pool_balance = yes
				banking_crash_check_overdrawn_interventions = yes
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = { te_banking_cycle_running = yes }
				banking_cycle_weekly_effects = yes
			}
		}
	}
