# ============================================================================
# te_probe_115: the vanilla 1.15 migration's engine probes, the hidden entry (2026-11-05)
# ============================================================================
# What it answers: the open engine questions of docs/audits/vanilla_1_15_prep.md § 2 that script can read with no
# player watching (C3, C5, C7, C19, C20 and three of C21), each as literal `TE_PROBE_115 <name> ...` lines in debug.log;
# which branch fired is the answer. The steps are in common/scripted_effects/te_probe_115_effects.txt, the hidden
# events in events/te_probe_115_events.txt, and what each line means in docs/testing/vanilla-1-15-playtest.md.
#
# Once a game, on one AI major power. The entry picks its own country: the first that is AI, a major power or above,
# at peace and not decentralized activates it, and its immediate claims the probe with a global marker. Whether a
# second country can activate it on the same pass is unknown, so a country whose immediate finds the claim taken
# logs that (itself an answer, c21_activation_pass) and ends at once through the hidden invalid outcome. Once the
# claim is set, disable_evaluation removes every other country's inactive record.
#
# Invisible to players: no player country can hold it (is_ai), it is not pinned, it has no buttons, and its
# events are hidden. Its two modifiers are the nuclear deterrent's display-only terms, whose types the engine and
# the script never read, so the AI country it runs on is unchanged. It takes about ten weeks, then both C3 outcomes
# come true together; te_probe_115.20 reports and removes every variable the probe set. If it stalls, it times out
# after two years and cleans up the same way.
#
# Delete once answered (see docs/testing/vanilla-1-15-playtest.md).

te_probe_115 = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_internal_affairs

	# Not carried over: a revolution's winner or a country the player switches to never takes it.
	can_revolution_inherit = no
	transferable = no

	is_shown_when_inactive = {
		NOT = { has_global_variable = te_probe_115_claimed }
		is_ai = yes
		country_rank >= rank_value:major_power
		NOT = { is_country_type = decentralized }
		exists = capital
		is_at_war = no
	}

	# Once claimed, no country can ever take the entry, so its inactive records go (C21 reads whether they did).
	disable_evaluation = {
		has_global_variable = te_probe_115_claimed
	}

	immediate = {
		if = {
			limit = { NOT = { has_global_variable = te_probe_115_claimed } }
			set_global_variable = { name = te_probe_115_claimed value = yes }
			set_variable = te_probe_115_owner
			set_variable = { name = te_probe_115_step value = 0 }
			set_variable = { name = te_probe_115_c7_scale value = 0 }
			debug_log = "TE_PROBE_115 start (the probe entry activated on the country below; steps follow weekly for about ten weeks)"
			debug_log_scopes = yes
		}
		else = {
			set_variable = te_probe_115_dupe
			debug_log = "TE_PROBE_115 c21_activation_pass DUPE (a second country activated the probe entry after the first had claimed it: every inactive record is checked before any activation's immediate runs; this copy ends through its invalid outcome)"
			debug_log_scopes = yes
		}
	}

	# C7. The control: a constant value, so country_nd_launch_credit_add should read 4 x 0.25 = 1.
	scaling_modifiers_while_active = {
		modifier = nd_authority_terms_2
		value = {
			value = 4
		}
	}
	# C7. The test: the steps' scale, read only behind is_ai, a trigger only a country answers, so it reaches the
	# modifier only if the value's root is the owning country. country_nd_crew_strain_monthly_add reads scale x -6.
	scaling_modifiers_while_active = {
		modifier = nd_readiness_terms_1
		value = {
			value = 0
			if = {
				limit = {
					is_ai = yes
					has_variable = te_probe_115_c7_scale
				}
				add = var:te_probe_115_c7_scale
			}
		}
	}

	# C5: a bar whose script-value minimum says which root it ran in (common/scripted_progress_bars/te_probe_115_bars.txt).
	scripted_progress_bar = te_probe_115_c5_bar

	on_weekly_pulse = {
		effect = {
			te_probe_115_weekly_effect = yes
		}
	}

	# C3: two visible outcomes on one trigger, which step 9 makes true. Declared first, with the failure look: if it
	# runs, the first declared wins, whatever its look.
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = te_probe_115_c3_go
		}
		effect = {
			set_variable = te_probe_115_c3_ran_first
			debug_log = "TE_PROBE_115 c3_precedence FIRST (of two outcomes true on the same update, the first declared ran; c3_both says whether the second ran too)"
			te_probe_115_in_outcome_effect = yes
		}
	}

	# C3: declared second, with the completion look.
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_variable = te_probe_115_c3_go
		}
		effect = {
			set_variable = te_probe_115_c3_ran_second
			debug_log = "TE_PROBE_115 c3_precedence SECOND (of two outcomes true on the same update, the second declared ran; c3_both says whether the first ran too)"
			te_probe_115_in_outcome_effect = yes
		}
	}

	# A second country that activated it on the claim's pass (the immediate's else) ends here, unseen.
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = te_probe_115_dupe
		}
		effect = {
			remove_variable = te_probe_115_dupe
		}
	}

	# A stalled probe: the last `TE_PROBE_115 step` line says where it stopped.
	timeout = 730
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			if = {
				limit = { has_variable = te_probe_115_owner }
				debug_log = "TE_PROBE_115 timeout (the probe stalled for two years; the last step line says where)"
				te_probe_115_cleanup_effect = yes
			}
		}
	}
}
