je_human_augmentation = {
	icon = "gfx/interface/icons/event_icons/je_human_augmentation.dds"
	group = je_group_internal_affairs

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	modifiers_while_active = {
		augmentation_debate_active_modifier
	}

	# 1.15: with the rule off, drop the inactive record for good. A rule can be changed
	# mid-game, from the screen that changes country; an active entry ends through its
	# invalid outcome on the rule.
	disable_evaluation = {
		NOT = { has_game_rule = social_movements_enabled }
	}
	is_shown_when_inactive = {
		has_game_rule = social_movements_enabled
		has_augmentation_tech = yes
		NOT = { is_country_type = decentralized }
		# Rebels do not start their own debate mid-war (and, winning, keep a fresh
		# outcome): vanilla's pattern for historical entries (je_the_two_spains,
		# je_risorgimento). Here, not in possible, so it can never touch an
		# active record, such as one a revolution's winner inherits; the winner
		# stops being revolutionary (vanilla gates power-bloc invites, obligations
		# and coups on it).
		is_revolutionary = no
		# Set by the success, failure and timeout outcomes. A revolution's winner gets a
		# fresh record of an entry the loser had finished, and without this it
		# would re-activate: a completed or failed debate would end again at once,
		# under the laws that ended it, and replay its outcome event; a timed-out
		# one would run its twenty years again. In normal play a finished record
		# never re-activates, so this changes nothing there.
		NOT = { has_variable = human_augmentation_resolved }
	}

	possible = {
		NOT = { has_variable = human_augmentation_resolved }
		has_augmentation_tech = yes
	}

	immediate = {
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			10 = augmentation_events.1
			10 = augmentation_events.2
			10 = augmentation_events.3
			10 = augmentation_events.4
			10 = augmentation_events.5
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_augmentation_status_banned
				trigger = {
					OR = {
						has_law = law_type:law_no_augmentation
						has_law = law_type:law_human_purity
					}
				}
			}
			triggered_desc = {
				desc = je_augmentation_status_medical
				trigger = {
					has_law = law_type:law_medical_augmentation_only
				}
			}
			triggered_desc = {
				desc = je_augmentation_status_regulated
				trigger = {
					has_law = law_type:law_regulated_augmentation_market
				}
			}
			triggered_desc = {
				desc = je_augmentation_status_unrestricted
				trigger = {
					has_law = law_type:law_unrestricted_augmentation
				}
			}
			triggered_desc = {
				desc = je_augmentation_status_mandatory
				trigger = {
					has_law = law_type:law_mandatory_augmentation
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_human_augmentation_complete_conclusion_desc
		trigger = {
			OR = {
				has_law = law_type:law_regulated_augmentation_market
				has_law = law_type:law_mandatory_augmentation
				has_law = law_type:law_unrestricted_augmentation
			}
			# Society has resolved the debate by integrating augmentation.
			# NOTE: law_no_augmentation and law_medical_augmentation_only deliberately do NOT
			# resolve here -- they leave the debate unsettled and route to the timeout outcome's divide
			# (see below). Only law_human_purity fails outright (see the failure outcome).
		}
		effect = {
			set_variable = human_augmentation_resolved
			trigger_event = { id = augmentation_events.200 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_human_augmentation_fail_conclusion_desc
		trigger = {
			has_law = law_type:law_human_purity
			# Society chose to reject augmentation entirely
		}
		effect = {
			set_variable = human_augmentation_resolved
			trigger_event = { id = augmentation_events.100 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_human_augmentation_invalid_conclusion_desc
		trigger = {
			OR = {
				is_country_type = decentralized
				# The rule switched off mid-game: the debate, its modifier and its events end.
				NOT = { has_game_rule = social_movements_enabled }
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = je_human_augmentation_timeout_conclusion_desc
		effect = {
			set_variable = human_augmentation_resolved
			# Reached when the debate never resolves: law_no_augmentation or
			# law_medical_augmentation_only held for the full timeout without integrating
			# (complete) or rejecting outright (fail). The unresolved divide breeds radicals + turmoil.
			add_radicals = {
				value = small_radicals
				strata = lower
			}
			add_radicals = {
				value = small_radicals
				strata = upper
			}
			add_modifier = {
				name = augmentation_divide_modifier
				days = long_modifier_time
				is_decaying = yes
			}
		}
	}

	timeout = 7300

	weight = 75
	should_be_pinned_by_default_uninvolved_or_context = yes
}
