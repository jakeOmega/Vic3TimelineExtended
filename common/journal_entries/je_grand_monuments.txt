# ============================================================================
# The Monuments journal entry (spec §6)
# ============================================================================
# Activates when the country owns a Grand Monument or has a commission on offer
# or open (v2 §2.7); goes when it owns none, has no commission, no
# ledger is still fading (plan decision 3), and no state still carries
# gm_seen (its state pulse has not yet recorded the teardown of a monument
# that is already gone). Comes back with the next monument. Holds the
# national modifiers (gm_apply_national_modifiers), which go with it. No
# `immediate`: nothing here a revolution's winner should lose. The widgets
# show the overview, the national effects, one row per monument and How
# Grand Monuments Work.
#
# Activates two ways: is_shown_when_inactive below auto-activates it in new
# saves the moment a country owns a Grand Monument (gated on the rule too,
# per spec §7's "Disabled -> no JE"), and gm_country_refresh explicitly calls
# add_journal_entry for saves that predate this entry's introduction, where
# is_shown_when_inactive alone never reaches an already-running save
# (scripting_best_practices.md § "JE Auto-Activation Requires BOTH").

je_grand_monuments = {
	icon = "gfx/interface/icons/event_icons/je_grand_monuments.dds"
	group = je_group_internal_affairs

	# The overview above the status description, the live sections below it,
	# How Grand Monuments Work at the foot (grand_monuments_widget.gui).
	widget = {
		gui = "gui/journal_entry_widgets/grand_monuments_widget.gui"
		name = "widget_je_gm_overview"
		container = "custom_widget_container_1"
	}

	widget = {
		gui = "gui/journal_entry_widgets/grand_monuments_widget.gui"
		name = "widget_je_gm_status"
		container = "custom_widget_container_2"
	}

	widget = {
		gui = "gui/journal_entry_widgets/grand_monuments_widget.gui"
		name = "widget_je_gm_reference"
		container = "custom_widget_container_3"
	}

	# A monument, or a commission on offer or open (v2 §2.7): a country with
	# no monument yet has a commission's row to follow.
	is_shown_when_inactive = {
		gm_system_enabled = yes
		OR = {
			any_scope_state = { has_building = building_grand_monument }
			has_variable = gm_com_open
			has_variable = gm_com_offered
		}
	}

	possible = {
		gm_system_enabled = yes
		OR = {
			any_scope_state = { has_building = building_grand_monument }
			has_variable = gm_com_open
			has_variable = gm_com_offered
		}
	}

	invalid = {
		OR = {
			gm_system_enabled = no
			AND = {
				NOT = {
					any_scope_state = { has_building = building_grand_monument }
				}
				gm_ledgers_idle = yes
				NOT = { any_scope_state = { has_variable = gm_seen } }
				NOT = { has_variable = gm_com_open }
				NOT = { has_variable = gm_com_offered }
			}
		}
	}

	# No status_desc: its counts ("Grand Monuments: 3, contested: 1") were the
	# overview's (owner, play-test round 2). No line is kept for the inactive
	# state: `possible` is `is_shown_when_inactive`, so an entry that shows
	# activates at the engine's next check of `possible`.

	weight = 400
	should_update_on_player_command = yes
}
