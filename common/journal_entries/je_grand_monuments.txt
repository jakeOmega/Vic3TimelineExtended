# ============================================================================
# The Monuments journal entry (spec §6)
# ============================================================================
# Activates when the country owns a Grand Monument or has a commission on offer
# or open (v2 §2.7); goes when it owns none, has no commission, no
# ledger is still fading (plan decision 3), and no state still carries
# gm_seen (its state pulse has not yet recorded the teardown of a monument
# that is already gone). Comes back with the next monument. Declares the
# national modifiers (scaling_modifiers_while_active, below), which go with
# it. No `immediate`: nothing here a revolution's winner should lose. The widgets
# show the overview, the national effects, one row per monument and How
# Grand Monuments Work.
#
# Activates two ways. Every country holds an inactive record from the game's
# start, which the engine activates at its own check, fortnightly, once
# is_shown_when_inactive and possible below hold (gated on the rule too, per
# spec §7's "Disabled -> no JE"). And gm_country_refresh and
# gm_com_send_offer call add_journal_entry for a country that holds no record:
# a save that predates this entry, where the engine made none
# (scripting_best_practices.md § "JE Auto-Activation Requires BOTH"), and a
# country whose entry has concluded (the invalid outcome below), whose record
# is gone (same doc, "A concluded entry is gone to script"). Both adds skip a
# country holding a copy, active or inactive: over an inactive copy the add is
# refused with an error, and the engine activates that copy itself (same doc,
# the has_journal_entry bullet; test_add_journal_entry_guard.py, #1416). Were
# a concluded entry to leave a record after all, the engine would activate it
# and the guard would still skip the add.

je_grand_monuments = {
	icon = "gfx/interface/icons/event_icons/je_grand_monuments.dds"
	group = je_group_internal_affairs

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	# The overview above the status description, with the "?" to the Help
	# page at its top, and the live sections below it
	# (grand_monuments_widget.gui).
	widget = {
		gui = "gui/journal_entry_widgets/grand_monuments_widget.gui"
		name = "widget_je_gm_overview"
		container = "custom_widget_container_1"
	}

	widget = {
		gui = "gui/journal_entry_widgets/grand_monuments_widget.gui"
		name = "widget_je_gm_status"
		container = "custom_widget_container_2"
	}

	# 1.15: with the rule off, drop the inactive record for good. A rule can be changed
	# mid-game, from the screen that changes country; an active entry ends through its
	# invalid outcome on the rule.
	disable_evaluation = {
		NOT = { gm_system_enabled = yes }
	}

	# A monument, or a commission on offer or open (v2 §2.7): a country with
	# no monument yet has a commission's row to follow. A trophy held, or one
	# taken from our lands and held abroad (trophies spec §6), has its rows too.
	is_shown_when_inactive = {
		gm_system_enabled = yes
		OR = {
			any_scope_state = { has_building = building_grand_monument }
			any_scope_state = { has_building = building_grand_monument_site }
			has_variable = gm_com_open
			has_variable = gm_com_offered
			gm_tr_country_has_any = yes
			gm_arch_country_has_any = yes
		}
	}

	possible = {
		gm_system_enabled = yes
		OR = {
			any_scope_state = { has_building = building_grand_monument }
			any_scope_state = { has_building = building_grand_monument_site }
			has_variable = gm_com_open
			has_variable = gm_com_offered
			gm_tr_country_has_any = yes
			gm_arch_country_has_any = yes
		}
	}

	# Also the rule switched off mid-game (gm_system_enabled = no): the entry's
	# scaling modifiers go with it, gm_state_rule_off_on_action takes the local
	# ones off the states, and the two cultural-pull values read the rule.
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_grand_monuments_invalid_conclusion_desc
		trigger = {
			OR = {
				gm_system_enabled = no
				AND = {
					NOT = {
						any_scope_state = { has_building = building_grand_monument }
					}
					NOT = {
						any_scope_state = { has_building = building_grand_monument_site }
					}
					gm_ledgers_idle = yes
					NOT = { any_scope_state = { has_variable = gm_seen } }
					NOT = { has_variable = gm_com_open }
					NOT = { has_variable = gm_com_offered }
				}
			}
		}
	}
