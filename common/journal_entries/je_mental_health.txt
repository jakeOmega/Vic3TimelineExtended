je_mental_health_crisis = {
	icon = "gfx/interface/icons/event_icons/je_mental_health_crisis.dds"
	group = je_group_internal_affairs

	# 1.15 asks every entry for this ("missing an activation_description trigger" otherwise);
	# always = yes is vanilla's own default, in 369 of its 449 entries.
	activation_description = {
		always = yes
	}

	modifiers_while_active = {
		mental_health_crisis_active_modifier
	}

	is_shown_when_inactive = {
		has_game_rule = social_movements_enabled
		has_technology_researched = mental_health_awareness
		NOT = { is_country_type = decentralized }
		# Rebels do not start their own debate mid-war (and, winning, keep a fresh
		# outcome): vanilla's pattern for historical entries (je_the_two_spains,
		# je_risorgimento). Here, not in possible, so it can never touch an
		# active record, such as one a revolution's winner inherits; the winner
		# stops being revolutionary (vanilla gates power-bloc invites, obligations
		# and coups on it).
		is_revolutionary = no
		# Set by the success, failure and timeout outcomes. A revolution's winner gets a
		# fresh record of an entry the loser had finished, and without this it
		# would re-activate: a completed or failed debate would end again at once,
		# under the laws that ended it, and replay its outcome event; a timed-out
		# one would run its twenty years again. In normal play a finished record
		# never re-activates, so this changes nothing there.
		NOT = { has_variable = mental_health_crisis_resolved }
	}

	possible = {
		NOT = { has_variable = mental_health_crisis_resolved }
		has_technology_researched = mental_health_awareness
	}

	immediate = {
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			10 = mental_health_events.1
			10 = mental_health_events.2
			10 = mental_health_events.3
			10 = mental_health_events.4
			10 = mental_health_events.5
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_mental_health_status_crisis
				trigger = {
					NOT = {
						has_law = law_type:law_rehabilitation_focused_criminal_justice
					}
				}
			}
			triggered_desc = {
				desc = je_mental_health_status_rehab
				trigger = {
					has_law = law_type:law_rehabilitation_focused_criminal_justice
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_mental_health_crisis_complete_conclusion_desc
		trigger = {
			has_law = law_type:law_rehabilitation_focused_criminal_justice
			institution_investment_level = {
				institution = institution_social_security
				value >= 4
			}
		}
		effect = {
			set_variable = mental_health_crisis_resolved
			trigger_event = { id = mental_health_events.200 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_mental_health_crisis_fail_conclusion_desc
		trigger = {
			# Fail if punitive justice is still in force after a decade of debate:
			# half the timeout, so there is room to reform first. Penal Labor Camps is
			# the harsher punitive law, so it fails too. This was gated on Decline of
			# Organized Religion, an era-10 tech like the one that opens the entry, so
			# it could fail the crisis the day it began (every country starts under
			# Punishment-Focused).
			# One legible line, not the raw age in days (GUI audit JE-8).
			custom_tooltip = {
				text = je_mental_health_fail_tt
				OR = {
					has_law = law_type:law_punishment_focused_criminal_justice
					has_law = law_type:law_penal_labor_camps
				}
				scope:journal_entry.journal_entry_age >= 3650
			}
		}
		effect = {
			set_variable = mental_health_crisis_resolved
			trigger_event = { id = mental_health_events.100 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_mental_health_crisis_invalid_conclusion_desc
		trigger = {
			is_country_type = decentralized
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = je_mental_health_crisis_timeout_conclusion_desc
		effect = {
			set_variable = mental_health_crisis_resolved
			add_radicals = {
				value = small_radicals
				strata = lower
			}
			add_modifier = {
				name = mental_health_neglect_modifier
				days = long_modifier_time
				is_decaying = yes
			}
		}
	}

	timeout = 7300

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
