					has_law = law_type:law_autocracy

	}

# ============================================================================
# RESEARCH COMMONS (docs/systems/mod_systems/power_blocs.md § Research Commons;
# design docs/superpowers/specs/2026-11-16-power-bloc-identities-design.md § 6)
# ============================================================================
# Members stop researching and send the leader 75% of their weekly innovation
# after their own programs; the leader's innovation rises by the sum and its
# cap by a tenth of it; every member is granted every technology any member
# holds. The mechanic is carried by je_research_commons and the
# te_research_commons_*.txt files; the marker below only labels the tooltip,
# and script reads has_identity (te_rc_in_commons).
identity_research_commons = {
	power_bloc_modifier = {
		power_bloc_research_pool_bool = yes
		power_bloc_mandate_progress_per_great_power_member_add = 5
		power_bloc_mandate_progress_per_major_power_member_add = 2
		power_bloc_mandate_progress_per_minor_power_member_add = 1
		power_bloc_mandate_progress_per_insignificant_power_member_add = 1
		power_bloc_cohesion_per_member_add = -1
	}

	leader_modifier = {
	}

	# Members don't research (owner, 2026-11-16): their innovation goes to the
	# leader through je_research_commons, and what they learn comes from the
	# ladder, technology spread and events.
	non_leader_modifier = {
		country_tech_research_speed_mult = -10
	}

	member_modifier = {
	}

	# Hard to leave (owner): the bloc's leverage advantage over the member must
	# be 10 or less, where vanilla's six identities all take 30.
	can_leave = {
		OR = {
			is_power_bloc_leader = yes
			power_bloc = {
				leverage_advantage = {
					target = root
					value <= 10
				}
			}
		}
	}

	# Placeholder icon until the coordinator renders this identity's own.
	icon = "gfx/interface/icons/central_identity_pillars_icons/ideological_union.dds"
	background = "gfx/interface/power_blocs/backgrounds/pb_background_generic.dds"

	# Unlocks with Intergovernmental Organizations (era 6, owner).
	visible = {
		has_dlc_feature = power_bloc_features
		has_technology_researched = intergovernmental_organizations
	}

	# The founder runs a Ministry of Science (owner), whose law needs Nuclear
	# Weapons. The leader can't abolish it afterwards (law_no_ministry_of_science's
	# can_enact). Members need no ministry (design § 6.6).
	possible = {
		has_law = law_type:law_ministry_of_science
	}

	mandate_progress = power_bloc_mandate_progress_by_rank

	on_created = {  }

	cohesion = {
		add = {
			add = 0.2
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			desc = POWER_BLOC_COHESION_BASE
		}

		# The leader must be the bloc's best researcher (owner): its own weekly
		# innovation, before the pool, as a share of the bloc's, at Sovereign
		# Empire's weight for the leader's share of prestige. te_rc_leader_pulse
		# stores the share on the leader each month; the previews of a country
		# joining or leaving read the stored figure unchanged.
		add = {
			value = 0
			if = {
				limit = {
					scope:power_bloc.power_bloc_leader ?= { has_variable = te_rc_leader_share }
				}
				add = scope:power_bloc.power_bloc_leader.var:te_rc_leader_share
			}
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			desc = POWER_BLOC_COHESION_RESEARCH_SHARE
		}

		add = {
			if = {
				limit = { exists = scope:with_country }
				add = "scope:power_bloc.power_bloc_worst_leader_relations_with(scope:with_country)"
			}
			else_if = {
				limit = { exists = scope:without_country }
				add = "scope:power_bloc.power_bloc_worst_leader_relations_without(scope:without_country)"
			}
			else = {
				add = scope:power_bloc.power_bloc_worst_leader_relations
			}
			divide = define:NDiplomacy|MIN_RELATIONS
			multiply = define:NPowerBlocs|COHESION_TARGET_MAX
			multiply = -1
			max = 0
			desc = POWER_BLOC_COHESION_WORST_LEADER_RELATIONS
		}
	}

	ai_weight = {
		value = 100

		if = {
			limit = {
				ig:ig_intelligentsia ?= { is_in_government = yes }
			}
			add = 100
		}

		if = {
			limit = {
				institution_investment_level = {
					institution = institution_ministry_of_science
					value >= 3
				}
			}
			add = 100
		}

		if = {
			limit = { has_global_highest_innovation = yes }
			multiply = 2.0
		}
	}
}
