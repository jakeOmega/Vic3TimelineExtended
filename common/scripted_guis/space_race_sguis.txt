# ============================================================================
# Space Race — Scripted GUIs
# ============================================================================
# The handlers behind the milestone panels in
# gui/journal_entry_widgets/space_race_widget.gui. They own NO rules of their
# own: visibility delegates to the shared gates in
# common/scripted_triggers/space_race_triggers.txt and every action delegates to
# the shared bodies in common/scripted_effects/space_race_effects.txt — the very
# helpers the journal entries' own scripted buttons call. Change an approach or
# funding rule in the helper and both surfaces follow. One gate differs on
# purpose: op 3 asks sr_possible_fund_up (up to six levels past the cap), while
# the AI's increase-funding buttons ask sr_possible_fund_up_within_cap, since
# the AI stays within the cap (lunar colonization design § 5.1, § 15.6).
#
# One scripted GUI per milestone, with the control passed in from the .gui as a
# single saved scope. IMPLICIT CONTRACT with the widget — keep both sides in
# sync (the table is repeated in the .gui header):
#
#   scope:op = 0  ->  select the Safe approach
#   scope:op = 1  ->  select the Ambitious approach
#   scope:op = 2  ->  decrease this milestone's funding level
#   scope:op = 3  ->  increase this milestone's funding level
#   scope:op = 4  ->  select the Standard approach (player only: no AI button)
#
# The milestone itself is NOT encoded in the op code: it arrives free through
# the panel's datacontext, which names one of these eight handlers. Folding it
# into the op would buy a 32-branch chain and save nothing, because variable
# names cannot be built from a scope.
#
# All of these are player-only (`ai_is_valid = { always = no }`). The AI still
# reaches the same helpers through the `ai_chance` blocks on the journal
# entries' scripted buttons, which is its only path into the space race — see
# docs/guides/scripting_best_practices.md, "Journal-Entry Buttons Are the AI's
# Only Path".
#
# The Rivals and Space Powers lists are datamodels of the script lists
# sr_rivals_list_<m> and sr_powers_list (space_race_effects.txt), so their
# "+N more" scripted GUIs went with audit SPC-9 (2026-10-27).
# ============================================================================

sr_milestone_suborbital_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_space_race_suborbital
		sr_controls_shown = { MILESTONE = suborbital }
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			sr_possible_safe = { MILESTONE = suborbital }
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			sr_possible_ambitious = { MILESTONE = suborbital }
		}
		trigger_else_if = {
			limit = { scope:op = 2 }
			sr_possible_fund_down = { MILESTONE = suborbital }
		}
		trigger_else_if = {
			limit = { scope:op = 3 }
			sr_possible_fund_up = { MILESTONE = suborbital }
		}
		trigger_else = {
			sr_possible_standard = { MILESTONE = suborbital }
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 }
			sr_effect_safe = { MILESTONE = suborbital }
		}
		else_if = {
			limit = { scope:op = 1 }
			sr_effect_ambitious = { MILESTONE = suborbital }
		}
		else_if = {
			limit = { scope:op = 2 }
			sr_effect_fund_down = { MILESTONE = suborbital }
		}
		else_if = {
			limit = { scope:op = 3 }
			sr_effect_fund_up = { MILESTONE = suborbital }
		}
		else = {
			sr_effect_standard = { MILESTONE = suborbital }
		}
	}
}
