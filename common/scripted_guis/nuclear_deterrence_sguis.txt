# ============================================================================
# NUCLEAR DETERRENCE — scripted GUIs for
# gui/journal_entry_widgets/nuclear_deterrence_widget.gui
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0 is what shipped).
#
# Two ACTION handlers, parameterized by a saved scope `op` (the Strategic
# Reserve / nuclear programme pattern), and eighteen DISPLAY handlers: the
# seven yes/no questions and names below (nd_umbrella_sgui draws the umbrella
# line, nd_loose_sgui the warheads unaccounted for, nd_crisis_opponent_sgui
# the crisis headline), three that hide the moves our side cannot
# make (nd_crisis_we_issued_sgui, nd_crisis_we_are_target_sgui,
# nd_crisis_private_sgui), and four that print script-built text
# (nd_crisis_pressure_breakdown_sgui, nd_crisis_danger_breakdown_sgui,
# nd_crisis_next_pressure_sgui, nd_risk_breakdown_sgui) — plus the World
# Crises list's gate and row hovers (nd_world_crises_sgui,
# nd_world_crisis_item_detail_sgui), and the At Home list and its footer,
# nd_home_list_sgui and nd_home_footer_sgui. The op
# tables are repeated in the .gui header and in
# docs/systems/journal_entry_systems.md — keep all three in step.
#
# ---- nd_posture_sgui — op table ---------------------------------------------
#   11..15  adopt doctrine 1..5           -> nd_set_doctrine
#   20..23  set readiness target 0..3     -> nd_set_readiness_target
#   31..34  adopt launch authority 1..4   -> nd_set_authority
#   40 / 41 safeguards down / up          -> nd_step_investment
#   50 / 51 hardening down / up           -> nd_step_investment
#   60 / 61 arsenal ceiling down / up     -> nd_taboo_ceiling_lower / _raise
#   62      lift the ceiling              -> nd_taboo_ceiling_lift
#   63      dismantle the arsenal         -> nd_taboo_dismantle_start
#   64      halt the dismantling          -> nd_taboo_dismantle_halt
#
# ---- nd_crisis_action_sgui — op table ---------------------------------------
#   1  go public (issuer)                 -> nd_crisis_act_go_public
#   2  propose a mutual stand-down        -> nd_crisis_act_propose_talks
#   3  a demonstration exercise (armed)   -> nd_crisis_act_exercise
#   4  concede (target)                   -> nd_crisis_act_yield
#   5  let it go (issuer)                 -> nd_crisis_act_back_down
#
# Every `is_valid` delegates to a scripted trigger in
# nuclear_deterrence_triggers.txt and every effect to the same scripted effect
# the AI's path uses (the posture review, the crisis events), so the panel and
# the AI cannot drift apart. The AI never uses these: `ai_is_valid = no`.
#
# `is_shown` on the action handlers is scope-free (gui_modding_guide.md gotcha
# #22): the buttons carry no `visible` from IsShown and are simply greyed by
# is_valid, which says why. The display handlers' is_shown are scope-free
# too, and are what the .gui asks its yes/no questions with.
# ============================================================================

# ---- Names: the crisis headline, and last time's opponent ---------------------
# A country held in a variable is printed from inside its own scope
# ([THIS.GetCountry.GetName]) — gui_modding_guide.md gotchas #11 and #15.
nd_crisis_opponent_sgui = {
	scope = country
	is_shown = {
		nd_in_crisis = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		# The headline (NUC-7): printed in our own scope, so it can name our
		# side and the dispute; the opponent is read through a temporary scope
		# saved only on the branch that prints it (gotcha #25, as
		# nd_world_crisis_row_at reads its target).
		if = {
			limit = {
				nd_in_crisis = yes
				exists = var:nd_crisis_opponent
			}
			var:nd_crisis_opponent = { save_temporary_scope_as = nd_crisis_opp_disp }
			custom_tooltip_no_bullet = nd_crisis_headline
		}
	}
}

# Incident exposure's factors (NUC-17): what multiplies the weekly roll, one
# line each, in the shape of the crisis breakdowns. Multipliers, not a
# probability (nuclear_crisis_design.md §10: no fake precision about hidden
# risk); the band word on the row is the reading. In a crisis its danger adds
# misreadings of its own, said in a word.
nd_risk_breakdown_sgui = {
	scope = country
	is_shown = {
		nd_is_armed = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		custom_tooltip_no_bullet = nd_risk_factor_readiness
		custom_tooltip_no_bullet = nd_risk_factor_doctrine
		custom_tooltip_no_bullet = nd_risk_factor_strain
		custom_tooltip_no_bullet = nd_risk_factor_reliability
		if = {
			limit = { nd_in_crisis = yes }
			custom_tooltip_no_bullet = nd_risk_factor_crisis
		}
		if = {
			limit = { nd_war_danger_band > nd_own_danger_band }
			custom_tooltip_no_bullet = nd_risk_factor_war
		}
	}
}

# ---- At Home: the list's empty states ------------------------------------------
# Each interest group with a view is a table of its own in the panel (round 3,
# 2026-09-30: a datamodel over the country's groups, reading the nd_disp_ig_*
# values in nuclear_deterrence_values.txt). This handler says only what the
# tables cannot: that the groups have not judged us yet, or that none holds a
# view. is_shown is true in just those two states, so the line is not drawn
# above the tables otherwise.
nd_home_list_sgui = {
	scope = country
	is_shown = {
		nd_is_armed = yes
		nd_is_initialized = yes
		OR = {
			NOT = { has_variable = nd_stance_applied }
			NOT = {
				any_interest_group = { has_variable = nd_ig_class }
			}
		}
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		if = {
			limit = { NOT = { has_variable = nd_stance_applied } }
			custom_tooltip_no_bullet = nd_home_list_not_reviewed
		}
		else_if = {
			limit = {
				NOT = {
					any_interest_group = { has_variable = nd_ig_class }
				}
			}
			custom_tooltip_no_bullet = nd_home_list_none
