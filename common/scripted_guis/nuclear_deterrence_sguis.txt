# ============================================================================
# NUCLEAR DETERRENCE — scripted GUIs for
# gui/journal_entry_widgets/nuclear_deterrence_widget.gui
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0 is what shipped).
#
# Two ACTION handlers, parameterized by a saved scope `op` (the Strategic
# Reserve / nuclear programme pattern), and thirteen DISPLAY handlers: the five
# yes/no questions and names below, three that hide the moves our side cannot
# make (nd_crisis_we_issued_sgui, nd_crisis_we_are_target_sgui,
# nd_crisis_private_sgui), and three that print script-built text
# (nd_crisis_pressure_breakdown_sgui, nd_crisis_danger_breakdown_sgui,
# nd_crisis_next_pressure_sgui) — plus the At Home list and its footer,
# nd_home_list_sgui and nd_home_footer_sgui. The op
# tables are repeated in the .gui header and in
# docs/systems/journal_entry_systems.md — keep all three in step.
#
# ---- nd_posture_sgui — op table ---------------------------------------------
#   11..15  adopt doctrine 1..5           -> nd_set_doctrine
#   20..23  set readiness target 0..3     -> nd_set_readiness_target
#   31..33  adopt launch authority 1..3   -> nd_set_authority
#   40 / 41 safeguards down / up          -> nd_step_investment
#   50 / 51 hardening down / up           -> nd_step_investment
#
# ---- nd_crisis_action_sgui — op table ---------------------------------------
#   1  go public (issuer)                 -> nd_crisis_act_go_public
#   2  propose a mutual stand-down        -> nd_crisis_act_propose_talks
#   3  a demonstration exercise (armed)   -> nd_crisis_act_exercise
#   4  concede (target)                   -> nd_crisis_act_yield
#   5  let it go (issuer)                 -> nd_crisis_act_back_down
#
# Every `is_valid` delegates to a scripted trigger in
# nuclear_deterrence_triggers.txt and every effect to the same scripted effect
# the AI's path uses (the posture review, the crisis events), so the panel and
# the AI cannot drift apart. The AI never uses these: `ai_is_valid = no`.
#
# `is_shown` on the action handlers is scope-free (gui_modding_guide.md gotcha
# #22): the buttons carry no `visible` from IsShown and are simply greyed by
# is_valid, which says why. The display handlers' is_shown are scope-free
# too, and are what the .gui asks its yes/no questions with.
# ============================================================================

nd_posture_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_nuclear_program
	}

	is_valid = {
		trigger_if = {
			limit = { exists = scope:op  scope:op = 11 }
			nd_can_set_doctrine = { D = 1 }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 12 }
			nd_can_set_doctrine = { D = 2 }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 13 }
			nd_can_set_doctrine = { D = 3 }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 14 }
			nd_can_set_doctrine = { D = 4 }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 15 }
			nd_can_set_doctrine = { D = 5 }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 20 }
			nd_can_set_readiness = { R = 0 LOCK_OK = yes }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 21 }
			nd_can_set_readiness = { R = 1 LOCK_OK = yes }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 22 }
			nd_can_set_readiness = { R = 2 LOCK_OK = no }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 23 }
			nd_can_set_readiness = { R = 3 LOCK_OK = no }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 31 }
			nd_can_set_authority = { A = 1 }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 32 }
			nd_can_set_authority = { A = 2 }
			custom_tooltip = {
				text = nd_tt_need_radar
				nd_can_adopt_delegation = yes
			}
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 33 }
			nd_can_set_authority = { A = 3 }
			custom_tooltip = {
				text = nd_tt_need_icbms
				nd_can_adopt_launch_on_warning = yes
			}
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 40 }
			nd_can_step_investment_down = { VAR = nd_safeguards }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 41 }
			nd_can_step_investment_up = { VAR = nd_safeguards }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 50 }
			nd_can_step_investment_down = { VAR = nd_hardening }
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 51 }
			nd_can_step_investment_up = { VAR = nd_hardening }
		}
		trigger_else = {
			always = no
		}
	}

	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = { exists = scope:op  scope:op = 11 }
			nd_set_doctrine_1 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 12 }
			nd_set_doctrine_2 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 13 }
			nd_set_doctrine_3 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 14 }
			nd_set_doctrine_4 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 15 }
			nd_set_doctrine_5 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 20 }
			nd_set_readiness_target_0 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 21 }
			nd_set_readiness_target_1 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 22 }
			nd_set_readiness_target_2 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 23 }
			nd_set_readiness_target_3 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 31 }
			nd_set_authority_1 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 32 }
			nd_set_authority_2 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 33 }
			nd_set_authority_3 = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 40 }
			nd_step_investment = { VAR = nd_safeguards OP = subtract }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 41 }
			nd_step_investment = { VAR = nd_safeguards OP = add }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 50 }
			nd_step_investment = { VAR = nd_hardening OP = subtract }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 51 }
			nd_step_investment = { VAR = nd_hardening OP = add }
		}
	}
}

nd_crisis_action_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {

	is_valid = {
		trigger_if = {
			limit = { exists = scope:op  scope:op = 1 }

		}

		}

		}

		}

		}
		trigger_else = {
			always = no
		}
	}

	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = { exists = scope:op  scope:op = 1 }
			nd_crisis_act_go_public = yes
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 2 }

		}

# ---- Names: the opponent, and last time's opponent ---------------------------
# A country held in a variable is printed from inside its own scope
# ([THIS.GetCountry.GetName]) — gui_modding_guide.md gotchas #11 and #15.
nd_crisis_opponent_sgui = {
	scope = country
	is_shown = {
		nd_in_crisis = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		if = {
			limit = { nd_in_crisis = yes }
			var:nd_crisis_opponent ?= {
				custom_tooltip_no_bullet = nd_crisis_opponent_line
			}
		}
	}
}

nd_crisis_next_pressure_sgui = {
	scope = country
	is_shown = {
		nd_in_crisis = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		if = {
			limit = { NOT = { nd_crisis_stage_at_least = { STAGE = 2 } } }
			custom_tooltip_no_bullet = nd_next_pressure_none
		}
		else_if = {
			limit = { nd_crisis_talks_open_either = yes }
			custom_tooltip_no_bullet = nd_next_pressure_paused
		}
		else = {
			custom_tooltip_no_bullet = nd_next_pressure_weeks
		}
	}
}

# ---- At Home: each interest group with a view, and its numbers ----------------
# Script-built text (ExecuteTooltip). Reads the variables nd_refresh_domestic_stance
# stores on each group; the numbers are the ones in the group's approval breakdown.
nd_home_list_sgui = {
	scope = country
	is_shown = {
		nd_is_armed = yes
		nd_is_initialized = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		if = {
			limit = { NOT = { has_variable = nd_stance_applied } }
			custom_tooltip_no_bullet = nd_home_list_not_reviewed
		}
		else_if = {
			limit = {
				any_interest_group = { has_variable = nd_ig_class }
			}
			every_interest_group = {
				limit = { has_variable = nd_ig_class }
				nd_home_line = yes
			}
		}
		else = {
			custom_tooltip_no_bullet = nd_home_list_none
		}
	}
}

# The At Home footer, its own text below the list: a scope's lines print before
# its nested blocks' lines (gui_modding_guide.md gotcha #18), so a closing line
# inside nd_home_list_sgui would print above the groups.
nd_home_footer_sgui = {
	scope = country
	is_shown = {
		nd_is_armed = yes
		nd_is_initialized = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = {
		custom_tooltip_no_bullet = nd_home_list_reviewed
		if = {
			limit = { nd_posture_judged = no }
			custom_tooltip_no_bullet = nd_home_list_waiting
		}
	}
}
