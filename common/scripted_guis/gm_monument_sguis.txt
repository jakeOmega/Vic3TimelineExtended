# ============================================================================
# GRAND MONUMENTS — several monuments in one state: scripted GUIs
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-13-multiple-monuments-per-state-design.md §5.
# ROOT is the player's country; the monument travels as gm_mon (a secondary's
# container, as gm_trophy_sguis.txt's gm_tr). The effect only orders the work,
# and a hidden event does it (monument_trophy_events.10): a tooltip walks the
# effect every frame.

# Carry It Off: one secondary monument in a state we occupy.
gm_mon_take_sgui = {
	scope = country
	saved_scopes = { gm_mon }

	is_shown = {
		exists = scope:gm_mon
		scope:gm_mon = {
			gm_mon_tr_can_take = { WHO = root }
		}
	}

	is_valid = {
		exists = scope:gm_mon
		scope:gm_mon = {
			gm_mon_tr_can_take_tt = { WHO = root }
		}
	}

	effect = {
		scope:gm_mon = {
			custom_tooltip = {
				text = gm_mon_take_tt
				gm_mon_tr_order_take = { WHO = root }
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- A contested monument beside another (ageing spec 2026-10-16 §4.3) ---------------
# Its row's three choices in Our Monuments. Pull Down and Rededicate only
# order the work (gm_mon_order_choice, monument_events.30); Keep as Heritage
# rebuilds nothing and runs here, as gm_preserve_sgui does.
gm_mon_tear_down_sgui = {
	scope = country
	saved_scopes = { gm_mon }

	is_shown = {
		exists = scope:gm_mon
		scope:gm_mon = {
			gm_mon_is_contested = yes
			var:gm_tr_origin ?= { owner = root }
		}
	}

	is_valid = {
		exists = scope:gm_mon
		scope:gm_mon = {
			var:gm_tr_origin ?= { owner = root }
			custom_tooltip = {
				text = gm_sgui_contested_tt
				gm_mon_is_contested = yes
			}
			custom_tooltip = {
				text = gm_mon_sgui_order_pending_tt
				NOT = { has_variable = gm_mon_order }
			}
		}
	}

	effect = {
		scope:gm_mon = {
			custom_tooltip = {
				text = gm_mon_tear_down_tt
				gm_mon_order_choice = { CHOICE = tear_down }
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- Clear Away (owner, 2026-10-31) -------------------------------------------------
# Once six monuments stand in one of our states, the most construction allows,
# the least grand of them may be cleared away to make room for a grander one:
# the button on its row (the state card, the journal entry's Our Monuments, and
# Also Standing Here). Its grandeur and what it gave go with it. Contested
# monuments keep their own three choices. Only orders the work
# (gm_state_order_clear, gm_mon_order_clear); monument_events.33 does it.
# Visibility comes from gm_state_disp_clearable / gm_mon_disp_clearable and the
# owner in the GUI, never from a saved scope reaching is_shown (gui gotcha #22);
# is_valid fails closed.
gm_clear_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			custom_tooltip = {
				text = gm_clear_req_full_tt
				gm_state_monument_count >= 6
			}
			custom_tooltip = {
				text = gm_clear_req_least_tt
				gm_state_primary_is_least = yes
			}
			custom_tooltip = {
				text = gm_clear_req_quiet_tt
				gm_state_clear_quiet = yes
				NOT = { has_variable = gm_clear_by }
			}
		}
	}

	effect = {
		scope:gm_state = {
			custom_tooltip = {
				text = gm_clear_tt
				gm_state_order_clear = yes
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}
