# GRAND MONUMENTS — several monuments in one state: scripted GUIs
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-13-multiple-monuments-per-state-design.md §5.
# ROOT is the player's country; the monument travels as gm_mon (a secondary's
# container, as gm_trophy_sguis.txt's gm_tr). The effect only orders the work,
# and a hidden event does it (monument_trophy_events.10): a tooltip walks the
# effect every frame.

# Carry It Off: one secondary monument in a state we occupy.
gm_mon_take_sgui = {
	scope = country
	saved_scopes = { gm_mon }

	is_shown = {
		exists = scope:gm_mon
		scope:gm_mon = {
			gm_mon_tr_can_take = { WHO = root }
		}
	}

	is_valid = {
		exists = scope:gm_mon
		scope:gm_mon = {
			gm_mon_tr_can_take_tt = { WHO = root }
		}
	}

	effect = {
		scope:gm_mon = {
			custom_tooltip = {
				text = gm_mon_take_tt
				gm_mon_tr_order_take = { WHO = root }
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- A contested monument beside another (ageing spec 2026-10-16 §4.3) ---------------
# Its row's three choices in Our Monuments. Pull Down and Rededicate only
# order the work (gm_mon_order_choice, monument_events.30); Keep as Heritage
# rebuilds nothing and runs here, as gm_preserve_sgui does.
gm_mon_tear_down_sgui = {
	scope = country
	saved_scopes = { gm_mon }

	is_shown = {
		exists = scope:gm_mon
		scope:gm_mon = {
			gm_mon_is_contested = yes
			var:gm_tr_origin ?= { owner = root }
		}
	}

	is_valid = {
		exists = scope:gm_mon
		scope:gm_mon = {
			var:gm_tr_origin ?= { owner = root }
			custom_tooltip = {
				text = gm_sgui_contested_tt
				gm_mon_is_contested = yes
			}
			custom_tooltip = {
				text = gm_mon_sgui_order_pending_tt
				NOT = { has_variable = gm_mon_order }
			}
		}
	}

	effect = {
		scope:gm_mon = {
			custom_tooltip = {
				text = gm_mon_tear_down_tt
				gm_mon_order_choice = { CHOICE = tear_down }
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}
