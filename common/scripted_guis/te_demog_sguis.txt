# ============================================================================
# DEMOGRAPHICS — display gates for the Population panel's Demographics tab
# (gui/te_demographics_widgets.gui; plan Task 13)
# ============================================================================
# A .gui read of a variable that does not exist logs "Failed to fetch variable"
# every frame (gui_modding_guide.md gotchas #14 and #23), and no Scope in GUI
# can ask whether a variable exists. So each family of figures the census
# writes together is shown only under a gate that asks script. The tab's own
# pair (te_system_tab_sguis.txt) covers the census: te_pops_demog_tab_unlock_sgui
# is valid once te_dg_census_year exists, and every figure, band, list and
# label the country census writes exists with it. These cover the rest. Each
# reads only its root (is_shown only: none changes anything, so none is valid
# or does anything). The first eight are country scope, rooted at the player's
# country; the six at the end are state scope, rooted at the state the panel
# shows (the state panel's Demographics subtab, plan Task 14; the last three its
# Causes of Death section, docs/superpowers/plans/2026-10-11-demographics-causes-of-death.md):
# GuiScope.SetRoot( State.MakeScope ).End. Phase 2 step 4's five follow them, three country and two state.
# ============================================================================

# Wealth Concentration's national figures: written together by
# te_demog_wc_national once a state has a score (te_dg_wc_target,
# te_inh_concentration, te_dg_wc_t_*, te_dg_wc_top, te_dg_wc_bottom). A game-start
# census can come before them. The country's Gini is NOT one of them: the same
# call writes it only when a scored state also has wealth bands in the current
# layout (te_dg_gini_layout), so a country can hold the target and no Gini (a save
# from before the bands, a state scored before its first walk). The Gini has its
# own gate below.
te_demog_wealth_ready_sgui = {
	scope = country
	is_shown = { has_variable = te_dg_wc_target }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# The country's Gini (te_dg_gini, te_demog_wc_national). A .gui read of a variable
# the country lacks shows 0, and 0.00 reads as perfect equality, so every place the
# panel shows the country's Gini picks it under this gate and shows a pending text
# otherwise. Written once and never removed, so the figure does not flicker.
te_demog_gini_ready_sgui = {
	scope = country
	is_shown = { has_variable = te_dg_gini }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# The most and least concentrated states (te_dg_wc_top, te_dg_wc_bottom) are
# chosen at the national refresh. Until the next one the state can have been
# ceded, released or merged away (destroyed: has_variable stays true), so each
# row shows only while its state exists and is still the country's.
te_demog_wc_top_sgui = {
	scope = country
	is_shown = {
		has_variable = te_dg_wc_top
		exists = var:te_dg_wc_top
		var:te_dg_wc_top = {
			exists = owner
			owner = ROOT
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

te_demog_wc_bottom_sgui = {
	scope = country
	is_shown = {
		has_variable = te_dg_wc_bottom
		exists = var:te_dg_wc_bottom
		var:te_dg_wc_bottom = {
			exists = owner
			owner = ROOT
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# The state has a census. te_dg_year is written by te_demog_set_scale, then the
# state's figures (te_demog_state_figures ends by writing te_dg_stepped), so
# with both every band and figure the subtab reads exists; the walk's Gini and
# urban share are written before any census. The subtab shows "Census pending"
# for a state without (a state new to the game, or split off since the last
# yearly pulse, or one the rule had not reached).
te_demog_state_census_sgui = {
	scope = state
	is_shown = {
		te_demog_cohorts_run = yes
		te_demog_has_census = yes
		has_variable = te_dg_stepped
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}
