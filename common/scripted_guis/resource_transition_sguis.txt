# ============================================================================
# RESOURCE TRANSITION — Fossil Transition section handlers (issue #660)
# ============================================================================
# Read by te_gw_sec_transition in gui/journal_entry_widgets/global_warming_widget.gui,
# which je_global_warming and the Market panel's Climate tab both show.
#
# The AI never uses these (`ai_is_valid = { always = no }`): it starts and
# stops programmes through the scripted buttons in
# common/scripted_buttons/resource_transition_buttons.txt, which call the same
# rt_possible_* triggers and rt_*_programme effects, so the two surfaces cannot
# drift.
#
# OP CODES, as in global_warming_sguis.txt, so the section reuses its Adopt and
# Repeal button types:
#
#   0   Start  -> rt_possible_start / rt_start_programme
#   1   Stop   -> rt_possible_stop  / rt_stop_programme
#
# An op nobody defined falls through to `trigger_else = { always = no }`.
# `is_shown` never reads scope:op (global_warming_sguis.txt says why); "is it
# running?" is the separate rt_active_<key>_sgui. Each action also requires
# is_player, as the climate policies' do.
#
# READ-ONLY HANDLERS carry `is_valid = { always = no }` and an empty effect:
# the widget asks script a yes/no question instead of re-encoding it in .gui.

# ---------------------------------------------------------------------------
# Programme actions
# ---------------------------------------------------------------------------
rt_programme_coal_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		always = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			rt_possible_start = { KEY = coal BT = building_coal_mine }
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			rt_possible_stop = { KEY = coal }
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 is_player = yes }
			rt_start_programme = { KEY = coal }
		}
		else_if = {
			limit = { scope:op = 1 is_player = yes }
			rt_stop_programme = { KEY = coal }
		}
	}
}
