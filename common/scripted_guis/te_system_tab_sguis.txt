# ============================================================================
# SYSTEM TABS IN VANILLA PANELS — gates
# ============================================================================
# A system's journal-entry panels are also shown as a tab in a vanilla panel
# (docs/systems/system_panels_feasibility.md). Each tab has two handlers here,
# and both read nothing but the root, so they are safe as `visible` / `enabled`
# gates (gotcha #22 in docs/guides/gui_modding_guide.md):
#
#   te_<host>_<system>_tab_sgui         the entry is running. Enables the tab,
#                                       and gates the content that lets its
#                                       GetPlayerJournalEntry datacontext
#                                       resolve. has_journal_entry counts
#                                       active entries only.
#   te_<host>_<system>_tab_unlock_sgui  is_shown puts the tab on the strip
#                                       whenever the system's game rule is on,
#                                       greyed until the entry runs. is_valid is
#                                       the entry's unlock test, rendered with
#                                       IsValidTooltip as the greyed tab's
#                                       checklist, so it calls the same trigger
#                                       the entry does.
#
# The Timeline Extended window (gui/te_systems_window.gui) uses the same pair
# per tab, named te_window_<system>_tab_sgui / _tab_unlock_sgui, plus a gate for
# its launcher and one per Space Race entry (at the foot of this file).
#
# None of these has an effect or is used by the AI.
# ============================================================================

# ---------------------------------------------------------------------------
# Banking: Budget panel (gui/budget_panel.gui), je_banking_cycle.
# ---------------------------------------------------------------------------
te_budget_banking_tab_sgui = {
	scope = country
	is_shown = {
		te_banking_system_on = yes
		has_journal_entry = je_banking_cycle
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

te_budget_banking_tab_unlock_sgui = {
	scope = country
	is_shown = { te_banking_system_on = yes }
	ai_is_valid = { always = no }
	is_valid = { te_banking_entry_unlocked = yes }
	effect = { }
}

# THE TIMELINE EXTENDED WINDOW (gui/te_systems_window.gui): the systems with no
# vanilla home, one tab each (docs/systems/system_panels_feasibility.md §9).

# The launcher under the sidebar shows while any of the three systems' game
# rules is on.

	scope = country
	is_shown = {
		OR = {
			has_game_rule = space_race_enabled
			has_game_rule = decolonization_enabled
			gm_system_enabled = yes
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Space Race: the nine je_space_race_* entries (je_space_race.txt). The tab
# opens while any of them runs; each entry's block has its own gate below,
# because GetPlayerJournalEntry takes one literal key. The checklist is when

	scope = country
