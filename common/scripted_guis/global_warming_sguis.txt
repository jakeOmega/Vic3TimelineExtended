# ============================================================================
# GLOBAL WARMING — journal-entry widget handlers
# ============================================================================
# Read by gui/journal_entry_widgets/global_warming_widget.gui.
#
# The AI never uses these (`ai_is_valid = { always = no }`). AI policy selection
# still runs through the `ai_chance` blocks on je_global_warming's sixteen
# scripted buttons, which are untouched. Every action handler delegates to the
# same `gw_possible_*` / `gw_effect_*` helpers the buttons call, so the two
# surfaces cannot drift.
#
# OP CODES — one handler per policy, two actions, selected by the `op` saved
# scope. Identical for all eight policies, so the widget's row type hard-codes
# them and a row instance carries no op markup of its own:
#
#   0   Adopt   -> gw_possible_<policy>        / gw_effect_<policy>
#   1   Repeal  -> gw_possible_remove_<policy> / gw_effect_remove_<policy>
#
# An op code nobody defined falls through to `trigger_else = { always = no }`,
# so a typo greys the control out instead of running the wrong branch.
#
# WHY `is_shown` NEVER READS `scope:op`. Vanilla declares `saved_scopes`
# alongside `is_shown` exactly once (`je_meiji_restoration_get_faction_sgui`)
# and that `is_shown` does not read the saved scope, so whether the engine
# populates saved scopes on the `IsShown` call path is unverified -- and a `.gui`
# expression that silently returns false would leave a row with no control at
# all. The "is this policy in force?" question therefore comes from the separate
# read-only `gw_active_<policy>_sgui` handlers below, which take no scope.
#
# READ-ONLY HANDLERS carry `is_valid = { always = no }` and an empty effect.
# They exist only so the widget can ask script a yes/no question instead of
# re-encoding a threshold in `.gui`.

# ---------------------------------------------------------------------------
# Policy actions
# ---------------------------------------------------------------------------
gw_policy_carbon_tax_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_global_warming
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			gw_possible_carbon_tax = yes
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			gw_possible_remove_carbon_tax = yes
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 }
			gw_effect_carbon_tax = yes
		}
		else_if = {
			limit = { scope:op = 1 }
			gw_effect_remove_carbon_tax = yes
		}
	}
}
