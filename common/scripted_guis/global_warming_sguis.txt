# ============================================================================
# GLOBAL WARMING — journal-entry widget handlers
# ============================================================================
# Read by gui/journal_entry_widgets/global_warming_widget.gui.
#
# The AI never uses these (`ai_is_valid = { always = no }`). AI policy selection
# still runs through the `ai_chance` blocks on je_global_warming's sixteen
# scripted buttons, which are untouched. Every action handler delegates to the
# same `gw_possible_*` / `gw_effect_*` helpers the buttons call, so the two
# surfaces cannot drift.
#
# OP CODES — one handler per policy, two actions, selected by the `op` saved
# scope. Identical for all eight policies, so the widget's row type hard-codes
# them and a row instance carries no op markup of its own:
#
#   0   Adopt   -> gw_possible_<policy>        / gw_effect_<policy>
#   1   Repeal  -> gw_possible_remove_<policy> / gw_effect_remove_<policy>
#
# An op code nobody defined falls through to `trigger_else = { always = no }`,
# so a typo greys the control out instead of running the wrong branch.
#
# WHY `is_shown` NEVER READS `scope:op`. Not because it cannot work -- because
# of the failure direction if it does not. Vanilla has no `is_shown` that reads a
# saved scope. Strategic Reserve looks like precedent and is not: its `.gui` does
# pass `AddScope('dir', ...)` into `IsShown`, but `st_res_adjust_<good>_sgui`'s
# `is_shown` never reads it -- the `trigger_if` on `scope:dir` a few lines below
# belongs to `is_valid`. The one real in-repo instance is
# `un_chamber_vote_sgui.is_shown`, which gates its veto control on
# `scope:op = 2`; it hedges with `exists = scope:op`, and if the scope never
# arrived that control would merely show too OFTEN.
#
# Here the failure direction is the opposite. These rows are the only surface a
# human has -- the journal entry's button grid is hidden behind `is_ai = yes` --
# so if an op-keyed `is_shown` silently returned false the row would render with
# NO control at all and the system would be unreachable. The "is this policy in
# force?" question therefore comes from the separate, scope-free
# `gw_active_<policy>_sgui` handlers below. `is_valid`, `Execute` and the
# tooltips DO read `scope:op`, which Strategic Reserve proves in shipped code.
# See docs/guides/gui_modding_guide.md gotcha #22.
#
# ANOTHER COUNTRY. The Market panel's Climate tab shows, on another market,
# that market's leader in the same rows (the section's `Country` datacontext),
# with every Adopt and Repeal greyed (Country.IsLocalPlayer in the widget's
# button types). As a second guard, each action's effect also requires
# `is_player = yes`, so a click that reached an AI country's handler would
# change nothing. It is in the branch limits, next to the op test, so the
# tooltip the player sees for their own country is unchanged. (`is_player`
# holds for any human, so in multiplayer the widget's greying is the guard
# against another human's policies.)
#
# READ-ONLY HANDLERS carry `is_valid = { always = no }` and an empty effect.
# They exist only so the widget can ask script a yes/no question instead of
# re-encoding a threshold in `.gui`.

# ---------------------------------------------------------------------------
# Policy actions
# ---------------------------------------------------------------------------
gw_policy_carbon_tax_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_global_warming
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			gw_possible_carbon_tax = yes
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			gw_possible_remove_carbon_tax = yes
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 is_player = yes }
			gw_effect_carbon_tax = yes
		}
		else_if = {
			limit = { scope:op = 1 is_player = yes }
			gw_effect_remove_carbon_tax = yes
		}
	}
}

# Does this country lead its own market? A market member sees the three
# market-wide policies as in force but not repealable, and this is what lets the
# row say "set by market leader" instead of just greying a button.
gw_is_market_leader_sgui = {
	scope = country
	is_shown = { gw_is_market_leader = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Have the yearly emissions figures been taken yet? False before the first
# yearly state pulse of a new game and on a save upgraded mid-year, when the
# widget shows "figures update each January" rather than a row of zeroes.
gw_has_yearly_figures_sgui = {
	scope = country
	is_shown = { gw_has_yearly_figures = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Is warming still below the point where any policy can be adopted? Every
# gw_possible_<policy> needs 0.5 C (gw_cond_warming_mild; divestment 1.0 C), so
# until then every Adopt control is greyed for the same reason. The Mitigation
# Policies section says so once, above the rows (style guide rule 6), instead of
# leaving eight greyed buttons to explain it one hover at a time.
# test_global_warming_layout.py holds this threshold to gw_possible_carbon_tax's.
gw_policies_waiting_sgui = {
	scope = country
	is_shown = { temperature_anomaly_display < 0.5 }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

	effect = { }
}

# ---------------------------------------------------------------------------
# Top Emitters: two tooltip builders, only ever rendered through
# ExecuteTooltip (nothing here runs). The country lists print one line per
# country from inside its own scope, the shape vanilla ships for a country list
# (gui_modding_guide.md gotcha #11), with the intro at the root, above them
# (gotcha #18: root lines print first). The root is a STATE, never a country,
# so no country in a list is also the tooltip's root (gotcha #27: a country's
# lines gather under its first appearance), and each country is listed once.
# ---------------------------------------------------------------------------

# A row's hover: the countries in the market the row's leader leads, the
# leader among them. Root: the leader's capital (the row's State).
gw_te_members_sgui = {
	scope = state
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = {
		custom_tooltip_no_bullet = gw_te_members_intro
		every_country = {
			limit = { market_capital ?= { owner ?= ROOT.owner } }
			custom_tooltip = gw_te_list_entry
		}
	}
}

# "+N more"'s hover: every emitting market, largest first, named by its leader
# with its annual emissions. Root: the viewer's capital (any state would do).
gw_top_emitters_sgui = {
	scope = state
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = {
		custom_tooltip_no_bullet = gw_te_all_intro
		ordered_country = {
			limit = { gw_is_ranked_emitter = yes }
			order_by = gw_disp_own_emis
			max = 200
			check_range_bounds = no
			custom_tooltip = gw_te_all_entry
		}
	}
}
