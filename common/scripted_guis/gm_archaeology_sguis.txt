# Design: docs/superpowers/archive/specs/2026-10-14-grand-monument-archaeology-design.md.
# gui/gm_archaeology_widgets.gui (the journal entry's Archaeological
# Expeditions section). ROOT is the player's country; the site travels as
# gm_arch (a trophy's gm_tr shape). Each effect only orders the work, and a
# hidden event does it (arch_events.10, .11): a tooltip walks the effect every
# frame.

# Launch Excavation by $METHOD$: a site we may dig, with that method's
# technology, greyed while we make less weekly innovation than the dig would
# divert (gm_arch_spares_<method>). Three buttons share one shape, one scripted
# GUI each.
gm_arch_launch_antiquarian_sgui = {
	scope = country
	saved_scopes = { gm_arch }

	is_shown = {
		exists = scope:gm_arch
		scope:gm_arch = {
			gm_arch_site_open = yes
			gm_arch_may_dig_ground = { WHO = root }
		}
		gm_arch_method_antiquarian_ok = yes
	}

	is_valid = {
		exists = scope:gm_arch
		scope:gm_arch = {
			gm_arch_can_dig_tt = { WHO = root }
		}
		custom_tooltip = {
			text = gm_arch_dig_innovation_antiquarian_tt
			gm_arch_spares_antiquarian = yes
		}
	}

	effect = {
		scope:gm_arch = {
			gm_arch_choose_launch = { WHO = root METHOD = antiquarian }
		}
	}

	ai_is_valid = {
		always = no
	}
}

# Suspend Dig: a dig we run. The progress stays.
gm_arch_suspend_sgui = {
	scope = country
	saved_scopes = { gm_arch }

	is_shown = {
		exists = scope:gm_arch
		scope:gm_arch = {
			gm_arch_digs_for = { WHO = root }
		}
	}

	is_valid = {
		exists = scope:gm_arch
		scope:gm_arch = {
			gm_arch_digs_for = { WHO = root }
		}
	}

	effect = {
		scope:gm_arch = {
			custom_tooltip = {
				text = gm_arch_suspend_tt
				gm_arch_order_suspend = { WHO = root }
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

# Display only, never executed: the open row's terms line for a site we would
# dig abroad (rework spec § 4). The row is shared by every country that may dig
# the site, so the line is decided for the viewer here. A concession wins over
# a permit when both are in force, as gm_arch_save_dig_state decides.
gm_arch_abroad_concession_sgui = {
	scope = country
	saved_scopes = { gm_arch }

	is_shown = {
		exists = scope:gm_arch
		scope:gm_arch = {
			gm_arch_would_dig_abroad_by = { WHO = root ARTICLE = archaeological_concession }
		}
	}
	ai_is_valid = { always = no }
}

gm_arch_abroad_permit_sgui = {
	scope = country
	saved_scopes = { gm_arch }

	is_shown = {
		exists = scope:gm_arch
		scope:gm_arch = {
			gm_arch_would_dig_abroad_by = { WHO = root ARTICLE = excavation_permit }
			NOT = { gm_arch_would_dig_abroad_by = { WHO = root ARTICLE = archaeological_concession } }
		}
	}
	ai_is_valid = { always = no }
}
