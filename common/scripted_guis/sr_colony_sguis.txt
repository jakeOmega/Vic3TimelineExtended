# SOLAR SYSTEM COLONIZATION — the colony panels' scripted GUIs
# ============================================================================
# Design: docs/superpowers/specs/2026-10-25-solar-colonization-population-and-transit-design.md
# (§ 3 targets, § 10.7 Settlers) and the colony modules design (§ 11). The buttons in
# gui/journal_entry_widgets/sr_sites_generated_rows.gui (one per world, which
# pass the world's index in gen_solar_sites.SITES as scope:site) and the
# Settlers section of space_race_widget.gui (scope:op). Player-only: the AI
# picks its worlds and sets its programme through the journal entry's pulse

# Solar System: send our next colony fleet to the world, or, on our target,
# let the programme choose again.
sr_colony_target_sgui = {
	scope = country
	saved_scopes = { site }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
	}

	ai_is_valid = {
		always = no
	}

	# Our target can always be cleared; a new one needs the programme running.
	# Each clause is its own line, so a greyed Target says which fails.
	is_valid = {
		trigger_if = {
			limit = { sr_cf_scope_site_is_target = yes }
			always = yes
		}
		trigger_else = {
			custom_tooltip = {
				text = sr_cf_req_running_tt
				sr_solar_colonization_running = yes
			}
			# A world beyond the Moon needs our Mars Landing (lunar colonization
			# design § 4): its own line, so the greyed button says why.
			custom_tooltip = {
				text = sr_cf_req_beyond_moon_tt
				OR = {
					sr_cf_scope_site_is_lunar = yes
					sr_colonization_beyond_moon = yes
				}
			}
			custom_tooltip = {
				text = sr_cf_req_targetable_tt
				sr_cf_scope_site_targetable = yes
			}
		}
	}

	effect = {
		if = {
			limit = { sr_cf_scope_site_is_target = yes }
			custom_tooltip = {
				text = sr_cf_target_clear_tt
				remove_variable = sr_cf_target
			}
		}
		else = {
			custom_tooltip = {
				text = sr_cf_target_set_tt
				sr_cf_set_target_from_scope = yes
			}
		}
	}
}

# The colonization overview's cells (audit SPC-6), display only: never
# Executed. scope:op 1 lists the colonies demanding self-government
# (sr_ov_list_restless, gen_solar_sites), 2 what threatens our colonies
# (sr_ov_list_defense, gen_space_fleets).
sr_colony_overview_sgui = {
	scope = country
	saved_scopes = { op }

	ai_is_valid = {
		always = no
	}

	is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 1 }
			sr_ov_list_restless = yes
		}
		else_if = {
			limit = { exists = scope:op scope:op = 2 }
			sr_ov_list_defense = yes
		}
	}
}

# Settlers: the emigration programme's level, op 0 down and op 1 up.
sr_settlers_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_contested_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			custom_tooltip = {
				text = sr_emig_at_lowest_tt
				sr_emig_level_value >= 1
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = sr_emig_at_highest_tt
				sr_emig_level_value <= 2
			}
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 }
			custom_tooltip = sr_emig_level_down_tt
		}
		else = {
			custom_tooltip = sr_emig_level_up_tt
		}
		hidden_effect = {
			sr_emig_step_level = yes
		}
	}
}

# Colony Modules (colony modules design § 11): scope:op 1-5 queues one module
# of that type at the world (1 its first specialization, 2 its second,
# 3 Habitat, 4 Defences, 5 Spaceport), within its caps; scope:op 0 clears the
# world's queue.
sr_colony_module_sgui = {
	scope = country
	saved_scopes = { site op }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_contested_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			custom_tooltip = {
				text = sr_mod_nothing_queued_tt
				sr_mod_scope_has_queue = yes
			}
		}
		# sr_mod_scope_can_queue's conditions, one line each (the failing one
		# gets the cross).
		trigger_else = {
			custom_tooltip = {
				text = sr_cf_req_held_tt
				sr_cf_scope_site_held = yes
			}
			custom_tooltip = {
				text = sr_mod_req_queue_room_tt
				sr_mq_has_room = yes
			}
			trigger_if = {
				limit = {
					OR = {
						scope:op = 1
						scope:op = 2
					}
				}
				custom_tooltip = {
					text = sr_mod_req_spec_chosen_tt
					sr_mod_scope_spec_chosen = yes
				}
				custom_tooltip = {
					text = sr_mod_req_spec_room_tt
					sr_mod_scope_spec_room = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = sr_mod_req_slot_room_tt
					sr_mod_scope_slot_room = yes
				}
			}
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 }
			custom_tooltip = {
				text = sr_mod_clear_queue_tt
				sr_mod_clear_from_scope = yes
			}
		}
		else_if = {
			limit = { scope:op = 1 }
			custom_tooltip = {
				text = sr_mod_queue_spec_a_tt
				sr_mod_queue_from_scope = yes
			}
		}
		else_if = {
			limit = { scope:op = 2 }
			custom_tooltip = {
				text = sr_mod_queue_spec_b_tt
				sr_mod_queue_from_scope = yes
			}
		}
		else_if = {
			limit = { scope:op = 3 }
			custom_tooltip = {
				text = sr_mod_queue_hab_tt
				sr_mod_queue_from_scope = yes
			}
		}
		else_if = {
			limit = { scope:op = 4 }
			custom_tooltip = {
				text = sr_mod_queue_def_tt
				sr_mod_queue_from_scope = yes
			}
		}
		else = {
			custom_tooltip = {
				text = sr_mod_queue_port_tt
				sr_mod_queue_from_scope = yes
			}
		}
	}
}

# Free Colonies (slice 5 design § 3): start or stop supplying a free colony. The
# entry's modifiers follow at once (Free Colony Supply and Trade).
sr_free_supply_sgui = {
	scope = country
	saved_scopes = { site }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_contested_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	# A supply can always be stopped; starting one needs the means.
	is_valid = {
		trigger_if = {
			limit = { sr_free_scope_site_supplied = yes }
			always = yes
		}
		trigger_else = {
			custom_tooltip = {
				text = sr_free_req_free_tt
				sr_free_scope_site_free = yes
			}
			custom_tooltip = {
				text = sr_free_req_can_supply_tt
				sr_free_can_supply = yes
			}
		}
	}

	effect = {
		if = {
			limit = { sr_free_scope_site_supplied = yes }
			custom_tooltip = {
				text = sr_free_supply_stop_tt
				sr_site_by_scope = { DO = toggle_supply }
			}
		}
		else = {
			custom_tooltip = {
				text = sr_free_supply_start_tt
				sr_site_by_scope = { DO = toggle_supply }
			}
		}
		sr_emig_refresh_modifiers = yes
	}
}
