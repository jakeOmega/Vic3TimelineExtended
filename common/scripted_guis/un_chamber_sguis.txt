# ============================================================================
# UN CHAMBER — SCRIPTED GUIS FOR THE JOURNAL ENTRY WIDGET
# ============================================================================
# Two kinds of entry live here, and the file is split accordingly.
#
# 1. TOOLTIP BUILDERS (everything down to un_chamber_mandates_sgui). Each exists
#    only so gui/journal_entry_widgets/un_chamber_widget.gui can call
#    `[GetScriptedGui('<key>').ExecuteTooltip(GuiScope.SetRoot(
#    JournalEntry.GetCountry.MakeScope).End)]` and render the resulting lines.
#    ExecuteTooltip *renders* an effect without running it — vanilla's
#    documented "Using SGUIs to build lists in loc" pattern; see
#    game/common/scripted_guis/scripted_guis.md and the tooltip builders in
#    game/common/scripted_guis/journal_entry_sguis.txt (which likewise declare
#    nothing but `scope` and `effect`). They contain nothing but scope
#    navigation and custom_tooltip lines (see
#    common/scripted_effects/un_chamber_display_effects.txt).
#
# 2. INTERACTIVE CONTROLS (from un_chamber_vote_sgui down). These do act, and
#    they act by calling the very effects the events and the journal entry's
#    scripted buttons call — see the section header further down.
#
# The pass/fail decision, archiving and history pruning stay entirely with
# un_vote_effects.txt; nothing here touches them.
#
# The active resolution and the bounded history are read in place through
# global_var:un_active_resolution and the un_resolution_history global list —
# no country-side copy of either is kept.
# ============================================================================

# The player's own seat in the organisation: one membership line for every
# state, from no UN at all to a suspended permanent seat. The overview's
# membership icon's tooltip (un_overview_widget.gui); the icon itself is picked
# by un_disp_member_code, which tests the states in the same order. The
# national record (the international-standing score, its tier, withheld
# benefits, the year's programme accrual, the last move) is
# un_chamber_standing_sgui below.
un_chamber_status_sgui = {
	scope = country

	effect = {
		if = {
			limit = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
			# §0.8: a member whose representation is suspended is told so, and
			# whether its overlord carries its seat.
			# Phase 7: the Assembly rejected our credentials, which is not the
			# subject's case above (un_justice_triggers.txt).
			if = {
				limit = { un_credentials_suspended = yes }
				custom_tooltip_no_bullet = je_un_chamber_status_credentials
			}
			else_if = {
				limit = { un_representation_suspended = yes }
				if = {
					limit = { un_seat_carried = yes }
					custom_tooltip_no_bullet = je_un_chamber_status_suspended_carried
				}
				else = {
					custom_tooltip_no_bullet = je_un_chamber_status_suspended
				}
			}
			else_if = {
				limit = { is_un_permanent_member = yes }
				custom_tooltip_no_bullet = je_un_chamber_status_permanent
			}
			else = {
				custom_tooltip_no_bullet = je_un_chamber_status_member
			}
		}
		else_if = {
			limit = { NOT = { has_global_variable = un_founded } }
			custom_tooltip_no_bullet = je_un_ov_status_no_un
		}
		else_if = {
			limit = { NOT = { un_membership_eligible = yes } }
			custom_tooltip_no_bullet = je_un_status_not_eligible
		}
		else = {
			# What joining brings: the journal entry's status text said so until
			# the overview took its place.
			custom_tooltip_no_bullet = je_un_status_not_member
		}
	}
}

# MISSIONS IN THE FIELD, ONE ROW PER DISPLAY SLOT. Every active mission holds a
# slot, 0-7 (un_mission_assign_slot), and the widget draws one row per slot:
# the mission (un_chamber_mission_slot_sgui) and our Join or Withdraw control
# beside it (un_chamber_mission_join_sgui / un_chamber_mission_withdraw_sgui).
# Each passes the slot as `op`. The slot table is repeated in
# un_mission_assign_slot (un_mission_effects.txt) and the widget's rows; keep
# them in step (test_un_chamber_mission_slots.py checks that they are).
#
# A row's visibility, and the choice between Join and Withdraw, read IsValid,
# never IsShown: a saved scope is documented to reach is_valid, and only
# thinly evidenced to reach is_shown (gui_modding_guide.md gotcha #22), which
# here would fail by hiding every row.

# The row: is_valid says whether the slot holds a mission (the row's
# `visible`); the effect renders it (the row's text).
un_chamber_mission_slot_sgui = {
	scope = country
	saved_scopes = { op }

	ai_is_valid = { always = no }

	is_valid = {
		trigger_if = {
			limit = { exists = scope:op scope:op = 0 }
			un_mission_slot_filled = { SLOT = 0 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 1 }
			un_mission_slot_filled = { SLOT = 1 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 2 }
			un_mission_slot_filled = { SLOT = 2 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 3 }
			un_mission_slot_filled = { SLOT = 3 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 4 }
			un_mission_slot_filled = { SLOT = 4 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 5 }
			un_mission_slot_filled = { SLOT = 5 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 6 }
			un_mission_slot_filled = { SLOT = 6 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 7 }
			un_mission_slot_filled = { SLOT = 7 }
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 0 }
			un_chamber_mission_slot_entry = { SLOT = 0 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 1 }
			un_chamber_mission_slot_entry = { SLOT = 1 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 2 }
			un_chamber_mission_slot_entry = { SLOT = 2 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 3 }
			un_chamber_mission_slot_entry = { SLOT = 3 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 4 }
			un_chamber_mission_slot_entry = { SLOT = 4 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 5 }
			un_chamber_mission_slot_entry = { SLOT = 5 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 6 }
			un_chamber_mission_slot_entry = { SLOT = 6 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 7 }
			un_chamber_mission_slot_entry = { SLOT = 7 }
		}
	}
}

# Cast our ballot on the resolution in session, from the chamber card, without
# waiting for the un_vote.1 popup that arrives 30 days in.
#
#   op 0   vote in favour   -> un_vote_cast_yes
#   op 1   vote against     -> un_vote_cast_no
#   op 2   veto             -> un_vote_cast_veto   (permanent seat, binding topic)
#
# The op table is repeated in the widget; keep the two in step. The effects
# themselves are the un_vote.1 option bodies — see
# common/scripted_effects/un_vote_cast_effects.txt — so this control cannot
# apply a different vote from the popup, and the button's ExecuteTooltip shows
# the popup's own consequence text, lobbying keep/break line included.
un_chamber_vote_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		has_global_variable = un_active_resolution
		exists = global_var:un_active_resolution
		global_var:un_active_resolution ?= { has_tag = un_res_voting }
		# The veto control exists only where the veto does. This mirrors
		# un_vote.1 option C's own `trigger`, which is what gates the veto on
		# the event path.
		trigger_if = {
			limit = { exists = scope:op scope:op = 2 }
			is_un_permanent_member = yes
			global_var:un_active_resolution ?= { un_resolution_is_binding = yes }
		}
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		# Asked before "have we voted" because un_resolution_open books the
		# proposer into un_res_yes: without this the country that tabled the
		# motion would be told it had already voted, rather than that its vote
		# IS the proposal.
		custom_tooltip = {
			text = je_un_chamber_vote_not_proposer_tt
			NOT = { un_chamber_is_proposer = yes }
		}
		# A ballot already cast is refused by naming which way it went, rather
		# than with a bare "you have voted" — the direction is otherwise only
		# visible by opening the Recorded Ballot section. Exactly one of these
		# renders, and each is `always = no`, so the line IS the refusal.
		# A veto is checked before an ordinary no vote because it books one as
		# well; a second vetoer overwrites un_res_vetoer, and that case falls
		# through to the no-vote line, which is still true of it.
		trigger_if = {
			limit = { NOT = { un_chamber_is_proposer = yes } }
			trigger_if = {
				limit = {
					global_var:un_active_resolution ?= {
						un_lobby_in_list = { LIST = un_res_yes COUNTRY = ROOT }
					}
				}
				custom_tooltip = {
					text = je_un_chamber_vote_already_yes_tt
					always = no
				}
			}
			trigger_else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						has_tag = un_res_vetoed
						var:un_res_vetoer ?= ROOT
					}
				}
				custom_tooltip = {
					text = je_un_chamber_vote_already_veto_tt
					always = no
				}
			}
			trigger_else_if = {
				limit = {
					global_var:un_active_resolution ?= {
						un_lobby_in_list = { LIST = un_res_no COUNTRY = ROOT }
					}
				}
				custom_tooltip = {
					text = je_un_chamber_vote_already_no_tt
					always = no
				}
			}
			# Nothing on the record: the clause that actually gates the control,
			# and the one green tick a player who can vote sees.
			trigger_else = {
				# §0.8: a member whose representation is suspended has no vote.
				custom_tooltip = {
					text = je_un_chamber_vote_represented_tt
					NOT = { un_representation_suspended = yes }
				}
				# Article 19 (redesign phase 5): two years in arrears on our
				# dues and the vote is gone until we pay.
				custom_tooltip = {
					text = je_un_chamber_vote_article_19_tt
					NOT = { un_dues_vote_suspended = yes }
				}
				custom_tooltip = {
					text = je_un_chamber_vote_not_yet_cast_tt
					un_chamber_ballot_open = yes
				}
			}
		}
	}

	effect = {
		# scope:un_resolution, scope:un_vote_proposer and scope:un_vote_target,
		# exactly as un_vote.1's `immediate` establishes them.
		un_chamber_cast_prepare = yes
		if = {
			limit = { exists = scope:op scope:op = 0 }
			un_vote_cast_yes = yes
		}
		else_if = {
			limit = { exists = scope:op scope:op = 1 }
			un_vote_cast_no = yes
		}
		else_if = {
			limit = { exists = scope:op scope:op = 2 }
			un_vote_cast_veto = yes
		}
	}
}
