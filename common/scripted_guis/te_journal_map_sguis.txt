# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The Map Room's picks and the
# test strip's probe, called from gui/te_map_widgets.gui (the te_jmap_pick_row type, on the Map Room's page and on the
# strip) and from the Demographics tab's map button. The AI never uses these: the map is a player's view.
#
# The picks are scripted GUIs, not the entry's scripted buttons: the Map Room is contextless with nobody involved, and
# the engine locks a scripted button for a country not involved in its entry (play-test check 14, 2026-11-05).
# is_valid is false on the current pick, which the button draws pressed; is_shown hides the three census maps while
# the census doesn't run.

te_jmap_pick_turmoil_sgui = {
	scope = country
	is_shown = { is_player = yes }
	ai_is_valid = { always = no }
	is_valid = { NOT = { te_jmap_selector = 1 } }
	effect = {
		te_jmap_set_selector = { SEL = 1 }
	}
}

# The strip's Potential probe (check 7): toggle the variable that shows je_te_map_potential_probe in Potential.
te_jmap_potential_probe_sgui = {
	scope = country
	is_shown = { is_player = yes }
	ai_is_valid = { always = no }
	is_valid = { is_player = yes }
	effect = {
		if = {
			limit = { has_variable = te_jmap_potential_probe }
			remove_variable = te_jmap_potential_probe
			debug_log = "TE_JMAP potential probe off date=[TimeKeeper.GetCurrentDate.GetString]"
		}
		else = {
			set_variable = te_jmap_potential_probe
			debug_log = "TE_JMAP potential probe on date=[TimeKeeper.GetCurrentDate.GetString] (the entry should show in Potential at the engine's next check)"
		}
	}
}

# Read-only: whether the probe entry exists for the player at all, inactive or active, so the strip's Open button
# asks for it only then.

	is_shown = {
		OR = {
			has_journal_entry = je_te_map_potential_probe
			has_inactive_journal_entry = je_te_map_potential_probe
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}
