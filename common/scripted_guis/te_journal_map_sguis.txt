# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The Map Room's picks, called
# from gui/te_map_widgets.gui (the te_jmap_pick_row type, on the Map Room's page) and from the Demographics tab's map
# button. The AI never uses these: the map is a player's view.
#
# The picks are scripted GUIs, not the entry's scripted buttons: the Map Room is contextless with nobody involved, and
# the engine locks a scripted button for a country not involved in its entry (play-test check 14, 2026-11-05).
# is_valid is false on the current pick, which the button draws pressed; is_shown hides the census maps while the
# census doesn't run. Selector numbers: common/script_values/te_journal_map_values.txt (2 was literacy, retired: vanilla's
# literacy map mode shows it).

te_jmap_pick_turmoil_sgui = {
	scope = country
	is_shown = { is_player = yes }
	ai_is_valid = { always = no }
	is_valid = { NOT = { te_jmap_selector = 1 } }
	effect = {
		te_jmap_set_selector = { SEL = 1 }
	}
}

# The repainter's log line (gui/te_map_widgets.gui, te_jmap_repainter): it runs just before the reopen, so a page that
# closes after a pick and stays closed says whether the reopen ran. Goes with the pick's line when the play-test is done.

	scope = country
	ai_is_valid = { always = no }
	effect = {
		debug_log = "TE_JMAP reopen date=[TimeKeeper.GetCurrentDate.GetString] (the repainter's reopen ran, a tenth of a second after its close)"
	}
}
