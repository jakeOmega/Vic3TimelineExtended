# Journal entry map modes (vanilla 1.15; docs/systems/mod_systems/journal_map_modes.md). The Map Room's options and
# its two helpers, called from gui/te_map_widgets.gui. The AI never uses these: the map is a player's view. The picks,
# one per map, are generated: te_journal_map_picks_generated.txt (scripts/generators/gen_journal_map_picks.py).

# ---------------------------------------------------------------------------
# The Map Room's options (2026-11-14): Area and Outliers, on the page below the picks (te_jmap_options,
# gui/te_map_widgets.gui). Scripted GUIs for the reason the picks are: nobody is involved in the Map Room, so its own
# scripted buttons would be locked. Unlike a pick's, an option's is_valid is not "this isn't the current one": the
# button draws pressed from the option's script value (te_jmap_area, te_jmap_continent, te_jmap_trim), and is_valid
# is kept for what greys it, with the reason in its tooltip (gui_style_guide.md, rule 6): Power Bloc for a country in
# none, and Outliers on a map with no range.
# ---------------------------------------------------------------------------
te_jmap_area_world_sgui = {
	scope = country
	is_shown = { is_player = yes }
	ai_is_valid = { always = no }
	effect = {
		te_jmap_set_area = { AREA = 0 }
	}
}

# Outliers: greyed while the pick has no range to trim (te_jmap_pick_fitted, generated: a map of categories or on a
# fixed scale), and under the Ranked scale, which colors by place in line and reads no range. The setting is kept,
# and acts again on the next fitted pick or the next scale that reads a range.
te_jmap_trim_default_sgui = {
	scope = country
	is_shown = { is_player = yes }
	ai_is_valid = { always = no }
	is_valid = {
		te_jmap_pick_fitted = 1
		NOT = { te_jmap_scale = 3 }
	}
	effect = {
		te_jmap_set_trim = { TRIM = 0 }
	}
}

# The repainter's log line (gui/te_map_widgets.gui, te_jmap_repainter): it runs just before the reopen, so a page that

	scope = country
	ai_is_valid = { always = no }
	effect = {

	}
