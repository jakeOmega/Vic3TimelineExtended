# ============================================================================
# GLOBAL COMMONS — SEABED LICENCES: scripted GUIs (the Seabed tab of the
# Timeline Extended window, gui/te_seabed_widgets.gui)
# ============================================================================
# The policy buttons call te_seabed_set_policy, the one policy change. The AI
# chooses its policy in script, yearly (te_seabed_ai_choose_policy), so none
# of these is the AI's: ai_is_valid = no.
# ============================================================================

te_seabed_policy_1_sgui = {
	scope = country
	is_shown = { te_global_commons_enabled = yes }
	ai_is_valid = { always = no }
	is_valid = { te_seabed_can_choose_survey = yes }
	effect = {
		custom_tooltip = {
			text = te_seabed_policy_effect_1_tt
			te_seabed_set_policy = { POLICY = 1 }
		}
	}
}
