# The policy buttons call te_seabed_set_policy, the one policy change. The AI
# chooses its policy in script, yearly (te_seabed_ai_choose_policy), so none
# of these is the AI's: ai_is_valid = no. Relinquish and Claim Again sit in
# each basin's own row of the rivals list (seabed claims spec § 2.5); the AI
# doesn't relinquish.
# ============================================================================

te_seabed_policy_1_sgui = {
	scope = country
	is_shown = { te_global_commons_enabled = yes }
	ai_is_valid = { always = no }
	is_valid = { te_seabed_can_choose_survey = yes }
	effect = {
		custom_tooltip = {
			text = te_seabed_policy_effect_1_tt
			te_seabed_set_policy = { POLICY = 1 }
		}
	}
}

te_seabed_policy_3_sgui = {
	scope = country
	is_shown = { te_global_commons_enabled = yes }
	ai_is_valid = { always = no }
	is_valid = { te_seabed_can_choose_unilateral = yes }
	effect = {
		custom_tooltip = {
			text = te_seabed_policy_effect_3_tt
			te_seabed_set_policy = { POLICY = 3 }
		}
		# The defection price, shown before the switch: the tooltip above can't
		# show it, because te_seabed_un_on_policy_change charges it only once the
		# new policy holds (te_global_commons_un_triggers.txt).
		if = {
			limit = { te_seabed_defection_would_cost = yes }
			custom_tooltip = te_seabed_defection_tt
			custom_tooltip = te_seabed_defection_infamy_tt
			custom_tooltip = te_seabed_defection_standing_tt
			custom_tooltip = te_seabed_defection_case_tt
		}
	}
}

te_seabed_relinquish_pacific_sgui = {
	scope = country
	is_shown = {
		te_global_commons_enabled = yes
		NOT = { has_variable = te_seabed_relinq_pacific }
	}
	ai_is_valid = { always = no }
	is_valid = {
		custom_tooltip = {
			text = te_seabed_relinquish_has_claims_tt
			has_variable = te_seabed_unbuilt_pacific
			var:te_seabed_unbuilt_pacific > 0
		}
	}
	effect = {
		custom_tooltip = {
			text = te_seabed_relinquish_tt
			te_seabed_relinquish = { B = pacific }
		}
	}
}

# Read-only: a basin's Free figure's hover (seabed claims spec § 3.3), the
# coastal states to build in, op = the basin (1 Pacific .. 5 Southern).
te_seabed_build_in_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		always = yes
	}

	is_valid = {
		always = no
	}

	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 1 }
			te_seabed_build_in_list = { B = pacific N = 1 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 2 }
			te_seabed_build_in_list = { B = atlantic N = 2 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 3 }
			te_seabed_build_in_list = { B = indian N = 3 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 4 }
			te_seabed_build_in_list = { B = arctic N = 4 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 5 }
			te_seabed_build_in_list = { B = southern N = 5 }
		}
	}
}
