# ============================================================================
# LEGISLATED TAX CODE — scripted GUIs (docs/systems/tax_code_schema.md, "Panels")
# ============================================================================
# Display gates for the Budget panel's Tax Code tab and je_tax_code
# (gui/journal_entry_widgets/te_tax_overview_widget.gui). Each answers one
# yes/no question about the player's country through `is_shown` alone: no
# saved scope (gotcha #22 in docs/guides/gui_modding_guide.md), `is_valid`
# always no, an empty effect, nothing for the AI. They read only the code's
# own tokens, each behind has_variable, and a record's payload only through
# the record's own token (te_tax_draft_active, te_tax_bill_active), so opening
# the panel writes nothing and reads no stale payload.
#
# The GUI calls them as
#   GetScriptedGui('<name>').IsShown( GuiScope.SetRoot( GetPlayer.MakeScope ).End )
# rooted on the player, as every tax code panel reads its numbers.
#
# The second half of the file holds the panels' ACTION HANDLERS (plan Task 8):
# each runs one Task 6 command, te_tax_cmd_<c> (te_tax_bill_effects.txt), and
# its is_valid is that command's own trigger, te_tax_can_<c>
# (te_tax_triggers.txt), with the same parameters, so a button's tooltip and its
# click cannot disagree. Every condition of those triggers is a custom_tooltip
# line, so a disabled button's IsValidTooltip lists each one that fails. The
# per-instrument step handlers and per-good toggles are generated
# (te_tax_generated_sguis.txt). test_tax_code_sguis.py checks both halves.
# ============================================================================

# Nothing is in progress: no draft, no bill, no passed bill waiting. A slot's
# test is NAND ("not both: the token exists and is 1"), never a multi-child
# NOT, whose meaning is untested (scripting_best_practices.md, "Write NOR or
# NAND, Never a Multi-Child NOT"): read as NOR it would hide this row for
# every migrated country, which always holds te_tax_p<s>_on.
te_tax_show_no_bill_sgui = {
	scope = country
	is_shown = {
		te_tax_code_in_force = yes
		NOT = { te_tax_draft_active = yes }
		NOT = { te_tax_bill_active = yes }
		NAND = {
			has_variable = te_tax_pa_on
			var:te_tax_pa_on = 1
		}
		NAND = {
			has_variable = te_tax_pb_on
			var:te_tax_pb_on = 1
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# The draft's commencement month. op: 0 a month earlier, 1 a month later.
te_tax_due_sgui = {
	scope = country
	saved_scopes = { op }
	is_shown = { te_tax_code_in_force = yes }
	ai_is_valid = { always = no }
	is_valid = {
		trigger_if = {
			limit = { exists = scope:op scope:op = 0 }
			te_tax_can_draft_due = { DIR = 0 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 1 }
			te_tax_can_draft_due = { DIR = 1 }
		}
		trigger_else = { always = no }
	}
	effect = {
		if = {
			limit = { exists = scope:op scope:op = 0 }
			te_tax_cmd_draft_due = { DIR = 0 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 1 }
			te_tax_cmd_draft_due = { DIR = 1 }
		}
	}
}

# Accepts a moved baseline (the review's warnings). op: the instrument as the
# history ring codes it (te_tax_h<n>_inst: 1 wage, 2 div, 3 land, 4 head,
# 5 cons), 6 the taxed goods.
te_tax_cmd_draft_rebase_sgui = {
	scope = country
	saved_scopes = { op }
	is_shown = { te_tax_code_in_force = yes }
	ai_is_valid = { always = no }
	is_valid = {
		trigger_if = {
			limit = { exists = scope:op scope:op = 1 }
			te_tax_can_draft_rebase = { KEY = wage }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 2 }
			te_tax_can_draft_rebase = { KEY = div }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 3 }
			te_tax_can_draft_rebase = { KEY = land }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 4 }
			te_tax_can_draft_rebase = { KEY = head }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 5 }
			te_tax_can_draft_rebase = { KEY = cons }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 6 }
			te_tax_can_draft_rebase = { KEY = goods }
		}
		trigger_else = { always = no }
	}
	effect = {
		if = {
			limit = { exists = scope:op scope:op = 1 }
			te_tax_cmd_draft_rebase = { KEY = wage }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 2 }
			te_tax_cmd_draft_rebase = { KEY = div }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 3 }
			te_tax_cmd_draft_rebase = { KEY = land }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 4 }
			te_tax_cmd_draft_rebase = { KEY = head }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 5 }
			te_tax_cmd_draft_rebase = { KEY = cons }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 6 }
			te_tax_cmd_draft_rebase = { KEY = goods }
		}
	}
}
