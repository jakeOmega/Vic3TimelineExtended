# ============================================================================
# LEGISLATED TAX CODE — scripted GUIs (docs/systems/tax_code_schema.md, "Panels")
# ============================================================================
# Display gates for the Budget panel's Tax Code tab and je_tax_code
# (gui/journal_entry_widgets/te_tax_overview_widget.gui). Each answers one
# yes/no question about the player's country through `is_shown` alone: no
# saved scope (gotcha #22 in docs/guides/gui_modding_guide.md), `is_valid`
# always no, an empty effect, nothing for the AI. They read only the code's
# own tokens, each behind has_variable, and a record's payload only through
# the record's own token (te_tax_draft_active, te_tax_bill_active), so opening
# the panel writes nothing and reads no stale payload.
#
# The GUI calls them as
#   GetScriptedGui('<name>').IsShown( GuiScope.SetRoot( GetPlayer.MakeScope ).End )
# rooted on the player, as every tax code panel reads its numbers.
#
# Task 8 adds the workbench's action handlers to this file.
# ============================================================================

# ---------------------------------------------------------------------------
# Overview: what is in progress.
# ---------------------------------------------------------------------------

# A draft reform is open.
te_tax_show_draft_sgui = {
	scope = country
	is_shown = {
		te_tax_code_in_force = yes
		te_tax_draft_active = yes
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# A bill is under debate.
te_tax_show_bill_sgui = {
	scope = country
	is_shown = {
		te_tax_code_in_force = yes
		te_tax_bill_active = yes
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Package slot a holds a passed bill (awaiting or held).
te_tax_show_package_a_sgui = {
	scope = country
	is_shown = {
		te_tax_code_in_force = yes
		has_variable = te_tax_pa_on
		var:te_tax_pa_on = 1
		has_variable = te_tax_pa_due
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Nothing is in progress: no draft, no bill, no passed bill waiting.
te_tax_show_no_bill_sgui = {
	scope = country
	is_shown = {
		te_tax_code_in_force = yes
		NOT = { te_tax_draft_active = yes }
		NOT = { te_tax_bill_active = yes }
		NOT = {
			has_variable = te_tax_pa_on
			var:te_tax_pa_on = 1
		}
		NOT = {
			has_variable = te_tax_pb_on
			var:te_tax_pb_on = 1
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# A transition is scheduled (te_tax_next_month, the scheduler's cache).
te_tax_show_next_change_sgui = {
	scope = country
	is_shown = {
		te_tax_code_in_force = yes
		has_variable = te_tax_next_month
		var:te_tax_next_month >= 0
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# The month the code last changed is known (te_tax_view_last_change).
te_tax_show_last_change_sgui = {

	}

}
