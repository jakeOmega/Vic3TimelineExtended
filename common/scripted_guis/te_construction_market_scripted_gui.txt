# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.
#
# Bound from gui/construction_panel.gui to the +/- buttons in the construction
# queue panel. The government purchase is set in one of two modes, chosen with
# the panel's mode buttons (te_construction_market_set_{quantity,budget}_mode):
#
# - Fixed Quantity (the default): each handler nudges the country's
#   public-construction buy target (te_construction_market_public_target) by a
#   fixed amount: 1 (default click), 10 (shift), 100 (ctrl), 1000 (alt), 10000
#   (right-click). On the decrease side, right-click resets to 0 instead of
#   stepping by 10000.
# - Fixed Budget: the te_construction_market_{increase,decrease}_budget_*
#   handlers nudge the weekly budget (te_construction_market_public_budget) by
#   1,000 / 10,000 / 100,000 / 1,000,000, right-click +10,000,000 or reset; the
#   quantity bought follows the budget over a few weeks
#   (te_construction_market_update_budget_quantity).
#
# Every handler is shown only while the construction market runs
# (te_free_market_construction_on): with free_market_construction_rule
# disabled there is nothing to buy, and the panel shows the
# direct-construction read-out instead (te_construction_market_rule_on_sgui).

te_construction_market_increase_target_1 = {

	scope = country

	is_shown = {
		te_free_market_construction_on = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		always = yes
	}

	effect = {
		#custom_tooltip = ""
		te_construction_market_init_target = yes
		change_variable = {
			name = te_construction_market_public_target
			add = 1
		}
		te_construction_market_country_update_private_share = yes
		te_construction_market_apply_per_site_modifier = yes
	}
}

# Read-only: which of the construction panel's two sections to render — the
# market's purchase controls or the direct-construction read-out. The GUI
# cannot test a game rule itself.

	is_shown = {
		te_free_market_construction_on = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		always = no
	}

	effect = {
	}
}

	}

# Mode buttons. Each carries the purchase in force over into the other mode;
# see te_construction_market_switch_to_budget_mode.
te_construction_market_set_quantity_mode = {

	scope = country

	is_shown = {
		te_free_market_construction_on = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		te_construction_market_budget_mode_on = yes
	}

	effect = {
		te_construction_market_switch_to_quantity_mode = yes
	}
}
