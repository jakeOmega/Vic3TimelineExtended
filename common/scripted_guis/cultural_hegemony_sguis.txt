# ============================================================================
# CULTURAL HEGEMONY — SCRIPTED GUIS FOR THE JOURNAL-ENTRY WIDGET
# ============================================================================
# Read by gui/journal_entry_widgets/cultural_hegemony_widget.gui. Eight
# handlers, in two kinds. (The board's three, ch_board_row_sgui,
# ch_board_detail_sgui and ch_board_is_us_sgui, went when the board became a
# datamodel over ch_board_rebuild's rows, CH-2.)
#
# WHY THERE ARE FOUR ch_active_<programme>_sgui HANDLERS THAT TAKE NO SCOPE
# Whether a `saved_scopes` value reaches a handler's `is_shown` is UNPROVEN in
# this mod. Strategic Reserve passes AddScope('dir', …) to IsShown, but its
# is_shown never reads the scope — only is_valid and effect do — so it is not
# evidence; vanilla ships no handler whose is_shown reads a saved scope; and
# vanilla's scripted_guis.md only promises saved scopes "in triggers / effects".
# Whether the player can act at all must not rest on that. So the Enable/Disable
# swap is answered by four scope-free handlers in the banking dashboard's
# play-tested shape, and `op` is used only for is_valid, Execute and the
# tooltips, where Strategic Reserve proves it works. The display handler that
# still branches on `op` for row visibility, ch_show_sgui, FAILS OPEN.
#
# 1. THE ONE INTERACTIVE HANDLER — ch_policy_sgui. It holds no logic of its own:
#    `is_shown` mirrors each journal-entry button's `visible`, `is_valid`
#    delegates to `ch_possible_<button>` and `effect` to `ch_effect_<button>`, so
#    the widget and the AI's buttons run the same script.
#
# 2. DISPLAY-ONLY HANDLERS — ch_show_sgui (and the scope-free yes/no handlers
#    at the foot of the file). All of them carry `is_valid = { always = no }`
#    and `ai_is_valid = { always = no }`, so neither a player nor the AI can
#    execute them: pure questions with an empty effect (the banking
#    dashboard's `banking_dash_phase_*` shape).
#
# ---- ch_policy_sgui op table (keep in step with the .gui header and the doc) --
#   0 / 1   programme funding   decrease / increase
#   2 / 3   International Cultural Outreach   begin / end
#   4 / 5   Cultural Institutes               fund  / defund
#   6 / 7   Global Media Campaign             launch / end
#   8 / 9   Cultural Protectionism            enact / end
#
# ---- ch_show_sgui op table -------------------------------------------------
# Each op reproduces one condition that used to gate a `triggered_desc` on the
# journal entry, so no threshold moves into .gui:
#   0   art row, share at or below the diminishing-returns threshold
#   1   art row, share above it
#   2   prestige row            3   standard-of-living row
#   4   tech-leadership row     5   monuments row
#   6   megaprojects row        7   cultural-pull modifier row
#   8   infamy row              9   instability row
#   10  cultural-pull multiplier row
#   12  exported-model block (the models section's "census taken" gate)
#   (11 was the "Rank N of M" line; the overview and the board's own-rank
#   line now test ch_rank_self_display without a saved scope)
#   31-45  one "Political models" legend row each (op = 30 + ch_model code)
#   (fail open: no scope -> the row shows)
#
# ---- ch_active_<programme>_sgui --------------------------------------------
#   No saved scope at all. is_shown = ch_active_<programme>, the shared trigger
#   the journal-entry buttons' `visible` also reads.
#
# ---- ch_ministry_missing_sgui / ch_benchmark_sgui ---------------------------
#   No saved scope. The first gates the programmes section's one line saying
#   the Ministry of Culture is missing (it greys every Enable control and the
#   funding +, so it is said once above the rows); the second gates the
#   overview's Foreign Cultural Benchmark cell.
# ============================================================================

# ---------------------------------------------------------------------------
# THE POLICY CONTROLS
# ---------------------------------------------------------------------------
ch_policy_sgui = {
	scope = country
	saved_scopes = { op }

	# Mirrors the ten scripted buttons' `visible`. Ops 0/1 were always visible
	# while the entry was up and greyed out through `possible`; the four
	# enable/disable pairs swap on whether the programme is running, so a pair
	# can never both offer "Enable".
	# Scope-free on purpose. Which of an Enable/Disable pair the widget draws is
	# answered by ch_active_<programme>_sgui below, NOT from here, because whether
	# a saved scope reaches `is_shown` is unproven in this mod: Strategic Reserve
	# passes AddScope('dir', …) to IsShown but its is_shown never reads it, and
	# vanilla has no handler whose is_shown reads a saved scope. This block used to
	# branch on scope:op and end in `trigger_else = { always = no }`, which would
	# have hidden all ten controls if the scope never arrived — and with the button
	# grid hidden from humans that leaves the player unable to act at all.
	is_shown = {
		has_journal_entry = je_cultural_hegemony
	}

	ai_is_valid = {
		always = no
	}

	# Every branch delegates. The custom_tooltip wording lives in the triggers,
	# phrased as conditions, because IsValidTooltip renders each with a tick or a
	# cross.
	is_valid = {
		trigger_if = {
			limit = { exists = scope:op scope:op = 0 }
			ch_possible_decrease_program_funding = yes
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			ch_possible_increase_program_funding = yes
		}
		trigger_else_if = {
			limit = { scope:op = 2 }
			ch_possible_world_exposition = yes
		}
		trigger_else_if = {
			limit = { scope:op = 3 }
			ch_possible_disable_world_exposition = yes
		}
		trigger_else_if = {
			limit = { scope:op = 4 }
			ch_possible_cultural_institutes = yes
		}
		trigger_else_if = {
			limit = { scope:op = 5 }
			ch_possible_disable_cultural_institutes = yes
		}
		trigger_else_if = {
			limit = { scope:op = 6 }
			ch_possible_global_media_campaign = yes
		}
		trigger_else_if = {
			limit = { scope:op = 7 }
			ch_possible_disable_global_media_campaign = yes
		}
		trigger_else_if = {
			limit = { scope:op = 8 }
			ch_possible_cultural_protectionism = yes
		}
		trigger_else_if = {
			limit = { scope:op = 9 }
			ch_possible_disable_cultural_protectionism = yes
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 0 }
			ch_effect_decrease_program_funding = yes
		}
		else_if = {
			limit = { scope:op = 1 }
			ch_effect_increase_program_funding = yes
		}
		else_if = {
			limit = { scope:op = 2 }
			ch_effect_world_exposition = yes
		}
		else_if = {
			limit = { scope:op = 3 }
			ch_effect_disable_world_exposition = yes
		}
		else_if = {
			limit = { scope:op = 4 }
			ch_effect_cultural_institutes = yes
		}
		else_if = {
			limit = { scope:op = 5 }
			ch_effect_disable_cultural_institutes = yes
		}
		else_if = {
			limit = { scope:op = 6 }
			ch_effect_global_media_campaign = yes
		}
		else_if = {
			limit = { scope:op = 7 }
			ch_effect_disable_global_media_campaign = yes
		}
		else_if = {
			limit = { scope:op = 8 }
			ch_effect_cultural_protectionism = yes
		}
		else_if = {
			limit = { scope:op = 9 }
			ch_effect_disable_cultural_protectionism = yes
		}
	}
}

# ---------------------------------------------------------------------------
# ROW VISIBILITY — one question per conditional row
# ---------------------------------------------------------------------------
# Every branch is the trigger the matching `triggered_desc` used to carry, moved
# here verbatim so the .gui never compares a number and no threshold is copied.
ch_show_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		# FAIL OPEN. If the saved scope never reaches is_shown (unproven in this
		# mod — see ch_policy_sgui), every row shows rather than every row hiding.
		# The worst case is then a cosmetic zero-valued line, never a reading the
		# player cannot see.
		trigger_if = {
			limit = { NOT = { exists = scope:op } }
			always = yes
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 0 }
			has_variable = ch_art
			var:ch_art > 0
			country_art_share_pct <= cultural_pull_art_dr_threshold
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 1 }
			has_variable = ch_art
			var:ch_art > 0
			country_art_share_pct > cultural_pull_art_dr_threshold
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 2 }
			prestige > 0
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 3 }
			has_variable = ch_sol
			# Either sign: a standard of living below the world average takes pull
			# away, and the row must show it so the lines add up to the Raw Score.
			NOT = { var:ch_sol = 0 }
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 4 }
			has_variable = ch_tech_firsts_recent
			var:ch_tech_firsts_recent > 0
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 5 }
			has_variable = ch_monuments
			var:ch_monuments >= 1
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 6 }
			has_variable = ch_megaprojects
			var:ch_megaprojects >= 1
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 7 }
			modifier:country_cultural_pull_add > 0
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 8 }
			infamy > 0
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 9 }
			OR = {
				country_turmoil > 0.01
				any_civil_war = { }
			}
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 10 }
			modifier:country_cultural_pull_mult > 0
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 12 }
			has_global_variable = ch_rank_1_ideology
		}
		# 31-45: one "Political models" legend row per model, op = 30 + the
		# model's ch_model code. A row shows once its model holds at least half a
		# percent of world culture, so the legend lists only what the pie shows.
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 31 }
			has_global_variable = ch_ideology_communist_share
			global_var:ch_ideology_communist_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 32 }
			has_global_variable = ch_ideology_anarchist_share
			global_var:ch_ideology_anarchist_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 33 }
			has_global_variable = ch_ideology_fascist_share
			global_var:ch_ideology_fascist_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 34 }
			has_global_variable = ch_ideology_corporatist_share
			global_var:ch_ideology_corporatist_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 35 }
			has_global_variable = ch_ideology_socialist_share
			global_var:ch_ideology_socialist_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 36 }
			has_global_variable = ch_ideology_royalist_absolutist_share
			global_var:ch_ideology_royalist_absolutist_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 37 }
			has_global_variable = ch_ideology_royalist_constitutional_share
			global_var:ch_ideology_royalist_constitutional_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 38 }
			has_global_variable = ch_ideology_religious_share
			global_var:ch_ideology_religious_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 39 }
			has_global_variable = ch_ideology_reactionary_share
			global_var:ch_ideology_reactionary_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 40 }
			has_global_variable = ch_ideology_military_junta_share
			global_var:ch_ideology_military_junta_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 41 }
			has_global_variable = ch_ideology_liberal_share
			global_var:ch_ideology_liberal_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 42 }
			has_global_variable = ch_ideology_other_share
			global_var:ch_ideology_other_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 43 }
			has_global_variable = ch_ideology_technocratic_share
			global_var:ch_ideology_technocratic_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 44 }
			has_global_variable = ch_ideology_republican_share
			global_var:ch_ideology_republican_share >= 0.5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 45 }
			has_global_variable = ch_ideology_developmentalist_junta_share
			global_var:ch_ideology_developmentalist_junta_share >= 0.5
		}
		trigger_else = {
			always = no
		}
	}

	is_valid = {
		always = no
	}

	ai_is_valid = {
		always = no
	}

	effect = { }
}

# IS A PROGRAMME RUNNING? — scope-free, for the widget's Enable/Disable swap

# ability to act depends on cannot be broken by a scope that fails to arrive.

# ---------------------------------------------------------------------------
# PANEL GATES — scope-free yes/no questions for the widget's layout
# ---------------------------------------------------------------------------
# Same shape as the ch_active_<programme>_sgui handlers: is_shown asks one
# question of the root, is_valid is always no, the effect is empty.

# The programmes section's "Requires Ministry of Culture" line. The law is the
# first condition of ch_possible_increase_program_funding and of all four
# Enable controls, so without it every one of them is greyed for the same
# reason; the line says it once, above the rows (gui_style_guide.md rule 6).
ch_ministry_missing_sgui = {
	scope = country

	is_shown = {
		NOT = { has_law = law_type:law_ministry_of_culture }
	}

	is_valid = {
		always = no
	}

	ai_is_valid = {
		always = no
	}

	effect = { }
}

# The overview's Foreign Cultural Benchmark cell: shown while the country
# carries the modifier je_cultural_hegemony's monthly pulse applies to
# countries trailing the leading cultural power.
ch_benchmark_sgui = {
	scope = country

	is_shown = {
		has_modifier = cultural_hegemony_foreign_benchmark
	}

	is_valid = {
		always = no
	}

	ai_is_valid = {
		always = no
	}

	effect = { }
}
