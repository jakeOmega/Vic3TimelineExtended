# ============================================================================
# CULTURAL HEGEMONY — SCRIPTED GUIS FOR THE JOURNAL-ENTRY WIDGET
# ============================================================================
# Read by gui/journal_entry_widgets/cultural_hegemony_widget.gui. Ten handlers,
# in two kinds.
#
# WHY THERE ARE FOUR ch_active_<programme>_sgui HANDLERS THAT TAKE NO SCOPE
# Whether a `saved_scopes` value reaches a handler's `is_shown` is UNPROVEN in
# this mod. Strategic Reserve passes AddScope('dir', …) to IsShown, but its
# is_shown never reads the scope — only is_valid and effect do — so it is not
# evidence; vanilla ships no handler whose is_shown reads a saved scope; and
# vanilla's scripted_guis.md only promises saved scopes "in triggers / effects".
# Whether the player can act at all must not rest on that. So the Enable/Disable
# swap is answered by four scope-free handlers in the banking dashboard's
# play-tested shape, and `op` is used only for is_valid, Execute and the
# tooltips, where Strategic Reserve proves it works. The two display handlers
# that still branch on `op` for row visibility FAIL OPEN, and the only
# fail-closed one is the "is this row us" marker.
#
# 1. THE ONE INTERACTIVE HANDLER — ch_policy_sgui. It holds no logic of its own:
#    `is_shown` mirrors each journal-entry button's `visible`, `is_valid`
#    delegates to `ch_possible_<button>` and `effect` to `ch_effect_<button>`, so
#    the widget and the AI's buttons run the same script.
#
# 2. DISPLAY-ONLY HANDLERS — ch_show_sgui, ch_board_row_sgui,
#    ch_board_detail_sgui, ch_history_marker_sgui. All four carry
#    `is_valid = { always = no }` and `ai_is_valid = { always = no }`, so neither
#    a player nor the AI can execute them. Two are pure questions with an empty
#    effect (the banking dashboard's `banking_dash_phase_*` shape); two emit
#    nothing but `custom_tooltip` lines for `ExecuteTooltip` to render — vanilla's
#    documented "using SGUIs to build lists in loc" pattern, see
#    game/common/scripted_guis/scripted_guis.md.
#
# ---- ch_policy_sgui op table (keep in step with the .gui header and the doc) --
#   0 / 1   programme funding   decrease / increase
#   2 / 3   International Cultural Outreach   begin / end
#   4 / 5   Cultural Institutes               fund  / defund
#   6 / 7   Global Media Campaign             launch / end
#   8 / 9   Cultural Protectionism            enact / end
#
# ---- ch_show_sgui op table -------------------------------------------------
# Each op reproduces one condition that used to gate a `triggered_desc` on the
# journal entry, so no threshold moves into .gui:
#   0   art row, share at or below the diminishing-returns threshold
#   1   art row, share above it
#   2   prestige row            3   standard-of-living row
#   4   tech-leadership row     5   monuments row
#   6   megaprojects row        7   cultural-pull modifier row
#   8   infamy row              9   instability row
#   10  cultural-pull multiplier row
#   11  "Rank N of M" line      12  exported-model block
#   31-45  one "Political models" legend row each (op = 30 + ch_model code)
#   (fail open: no scope -> the row shows)
#
# ---- ch_board_row_sgui / ch_board_detail_sgui / ch_board_is_us_sgui --------
#   op = board rank, 1 to 10. The first two fail open, the last fails closed.
#
# ---- ch_active_<programme>_sgui --------------------------------------------
#   No saved scope at all. is_shown = ch_active_<programme>, the shared trigger
#   the journal-entry buttons' `visible` also reads.
# ============================================================================

# ---------------------------------------------------------------------------
# THE POLICY CONTROLS
# ---------------------------------------------------------------------------
ch_policy_sgui = {
	scope = country
	saved_scopes = { op }

	# Mirrors the ten scripted buttons' `visible`. Ops 0/1 were always visible
	# while the entry was up and greyed out through `possible`; the four
	# enable/disable pairs swap on whether the programme is running, so a pair
	# can never both offer "Enable".
	# Scope-free on purpose. Which of an Enable/Disable pair the widget draws is
	# answered by ch_active_<programme>_sgui below, NOT from here, because whether
	# a saved scope reaches `is_shown` is unproven in this mod: Strategic Reserve
	# passes AddScope('dir', …) to IsShown but its is_shown never reads it, and
	# vanilla has no handler whose is_shown reads a saved scope. This block used to
	# branch on scope:op and end in `trigger_else = { always = no }`, which would
	# have hidden all ten controls if the scope never arrived — and with the button
	# grid hidden from humans that leaves the player unable to act at all.
	is_shown = {
		has_journal_entry = je_cultural_hegemony
	}

	ai_is_valid = {
		always = no
	}

	# Every branch delegates. The custom_tooltip wording lives in the triggers,
	# phrased as conditions, because IsValidTooltip renders each with a tick or a
	# cross.
	is_valid = {
		trigger_if = {
			limit = { exists = scope:op scope:op = 0 }
			ch_possible_decrease_program_funding = yes
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			ch_possible_increase_program_funding = yes
		}
		trigger_else_if = {
			limit = { scope:op = 2 }
			ch_possible_world_exposition = yes
		}
		trigger_else_if = {
			limit = { scope:op = 3 }
			ch_possible_disable_world_exposition = yes
		}
		trigger_else_if = {
			limit = { scope:op = 4 }
			ch_possible_cultural_institutes = yes
		}
		trigger_else_if = {
			limit = { scope:op = 5 }
			ch_possible_disable_cultural_institutes = yes
		}
		trigger_else_if = {
			limit = { scope:op = 6 }
			ch_possible_global_media_campaign = yes
		}
		trigger_else_if = {
			limit = { scope:op = 7 }
			ch_possible_disable_global_media_campaign = yes
		}
		trigger_else_if = {
			limit = { scope:op = 8 }
			ch_possible_cultural_protectionism = yes
		}
		trigger_else_if = {
			limit = { scope:op = 9 }
			ch_possible_disable_cultural_protectionism = yes
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 0 }
			ch_effect_decrease_program_funding = yes
		}
		else_if = {
			limit = { scope:op = 1 }
			ch_effect_increase_program_funding = yes
		}
		else_if = {
			limit = { scope:op = 2 }
			ch_effect_world_exposition = yes
		}
		else_if = {
			limit = { scope:op = 3 }
			ch_effect_disable_world_exposition = yes
		}
		else_if = {
			limit = { scope:op = 4 }
			ch_effect_cultural_institutes = yes
		}
		else_if = {
			limit = { scope:op = 5 }
			ch_effect_disable_cultural_institutes = yes
		}
		else_if = {
			limit = { scope:op = 6 }
			ch_effect_global_media_campaign = yes
		}
		else_if = {
			limit = { scope:op = 7 }
			ch_effect_disable_global_media_campaign = yes
		}
		else_if = {
			limit = { scope:op = 8 }
			ch_effect_cultural_protectionism = yes
		}
		else_if = {
			limit = { scope:op = 9 }
			ch_effect_disable_cultural_protectionism = yes
		}
	}
}

ch_show_sgui = {
	scope = country

	is_shown = {
		# FAIL OPEN. If the saved scope never reaches is_shown (unproven in this

			limit = { NOT = { exists = scope:op } }

			limit = { exists = scope:op scope:op = 0 }
			has_variable = ch_art

			limit = { exists = scope:op scope:op = 1 }

			country_art_share_pct > cultural_pull_art_dr_threshold
		}

			limit = { exists = scope:op scope:op = 2 }
			prestige > 0
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 3 }
			has_variable = ch_sol
			var:ch_sol > 0

			limit = { exists = scope:op scope:op = 4 }

# LEADERBOARD — one row's line
# ---------------------------------------------------------------------------
# `is_shown` doubles as the row's visibility: a slot that the yearly pass left
# empty, or whose country has since ceased to exist, hides its whole row rather
# than printing a placeholder.
#
# The effect enters the ranked country's scope and prints one line, so the loc
# key can use [THIS.GetCountry.GetName] — the chain vanilla itself uses for a
# country list inside a scripted GUI. The engine prefixes a line printed inside
# a country scope change with that country's flag for free, so the keys add none.
ch_board_row_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		# FAIL OPEN, like ch_show_sgui: a missing scope leaves ten empty rows,
		# not a missing leaderboard. The effect below is guarded separately with
		# `?=`, so an unfilled slot prints nothing either way — which matters
		# because ExecuteTooltip ignores is_shown (gotcha #16).
		trigger_if = {
			limit = { NOT = { exists = scope:op } }
			always = yes
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 1 }
			has_global_variable = ch_rank_1
			exists = global_var:ch_rank_1
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 2 }
			has_global_variable = ch_rank_2
			exists = global_var:ch_rank_2
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 3 }
			has_global_variable = ch_rank_3
			exists = global_var:ch_rank_3
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 4 }
			has_global_variable = ch_rank_4
			exists = global_var:ch_rank_4
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 5 }
			has_global_variable = ch_rank_5
			exists = global_var:ch_rank_5
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 6 }
			has_global_variable = ch_rank_6
			exists = global_var:ch_rank_6
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 7 }
			has_global_variable = ch_rank_7
			exists = global_var:ch_rank_7
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 8 }
			has_global_variable = ch_rank_8
			exists = global_var:ch_rank_8
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 9 }
			has_global_variable = ch_rank_9
			exists = global_var:ch_rank_9
		}
		trigger_else_if = {
			limit = { exists = scope:op scope:op = 10 }
			has_global_variable = ch_rank_10
			exists = global_var:ch_rank_10
		}
		trigger_else = {
			always = no
		}
	}

	is_valid = {
		always = no
	}

	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 1 }
			global_var:ch_rank_1 ?= { custom_tooltip_no_bullet = ch_board_row_1 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 2 }
			global_var:ch_rank_2 ?= { custom_tooltip_no_bullet = ch_board_row_2 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 3 }
			global_var:ch_rank_3 ?= { custom_tooltip_no_bullet = ch_board_row_3 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 4 }

		}
		else_if = {
			limit = { exists = scope:op scope:op = 5 }
			global_var:ch_rank_5 ?= { custom_tooltip_no_bullet = ch_board_row_5 }
		}
		else_if = {
			limit = { exists = scope:op scope:op = 6 }

		}
