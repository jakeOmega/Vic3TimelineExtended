# ============================================================================
# COVERT WARFARE — SCRIPTED GUIS FOR THE JOURNAL-ENTRY WIDGETS
# ============================================================================
# Read from gui/journal_entry_widgets/covert_operations_widget.gui. Five kinds
# of entry live here.
#
# 1. THE FUNDING CONTROL (covert_funding_sgui). Delegates to the same
#    covert_possible_* trigger and covert_effect_* effect the journal entry's
#    scripted buttons call, so the grid and the widget cannot diverge.
#
#      op | control            | delegates to
#      ---+--------------------+------------------------------------------
#       0 | funding stepper -  | covert_possible/effect_decrease_funding
#       1 | funding stepper +  | covert_possible/effect_increase_funding
#
#    The op table is restated in the .gui header and in
#    docs/systems/journal_entry_systems.md — keep all three in step.
#
# 2. A TOOLTIP BUILDER (covert_funding_ladder_sgui). It exists only so the .gui
#    can call `[GetScriptedGui('…').ExecuteTooltip(GuiScope.SetRoot(
#    JournalEntry.GetCountry.MakeScope).End)]` and render the resulting lines.
#    ExecuteTooltip *renders* an effect without running it — vanilla's
#    documented "Using SGUIs to build lists in loc" pattern; see
#    game/common/scripted_guis/scripted_guis.md.
#
# 3. YES/NO QUESTIONS (covert_ops_dormant_sgui, covert_last_exposed_known_sgui,
#    covert_ops_phase_ready_sgui, covert_ops_priority_ready_sgui). `.gui` has no
#    way to ask script a question, so each of these is an is_shown-only entry
#    read as `[GetScriptedGui('…').IsShown(…)]`. They exist so the widget never
#    re-encodes a threshold or an existence check itself. See
#    docs/guides/scripting_best_practices.md § "Asking Script a Yes/No Question
#    From `.gui`".
#
# 4. THE STAND-DOWN HANDLERS (covert_stand_down_<type>_sgui). One per
#    operation type, each ending the operation a widget row is showing. See
#    their own section header for why there is one per type rather than one
#    with a type op code, and for the single unproven link in the mechanism.
#
# 5. THE PRIORITY STEPPER (covert_priority_up_sgui, covert_priority_down_sgui).
#    One handler per direction; the row's container arrives as the saved scope
#    iw_op. See their section header.
#
# Every entry declares `ai_is_valid = { always = no }`: the AI's path into
# covert funding is je_covert_warfare's monthly pulse, and it must never
# double-dip through a widget handler.
# ============================================================================

# ---- THE FUNDING CONTROL ----------------------------------------------------
# One handler behind both stepper buttons, parameterized by the op code above.
# `is_valid` doubles as the "you are here" indicator: at level 0 the minus
# button greys itself, at iw_funding_level_max the plus does, each with the
# condition that failed as its tooltip.
covert_funding_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		always = yes
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			covert_possible_decrease_funding = yes
		}
		trigger_else = {
			covert_possible_increase_funding = yes
		}
	}

	effect = {
		if = {
			limit = { scope:op = 0 }
			covert_effect_decrease_funding = yes
		}
		else = {
			covert_effect_increase_funding = yes
		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- TOOLTIP BUILDER: the funding ladder -----------------------------------
# Six rows, one per funding level, each carrying that level's detection
# reduction and counterintelligence capacity from the covert_funding_*_at_N
# script values — which are sums of the tuning constants, so the ladder cannot
# drift from the detection formula. The row for the level in force is rendered
# by its own _current key, which is both the highlight and the "you are here"
# marker.
#
# Cheap enough to leave expanded: every line reads constants plus one country
# variable, and there is no iteration.
covert_funding_ladder_sgui = {
	scope = country

	effect = {
		if = {
			limit = {
				OR = {
					NOT = { has_variable = iw_funding_level }
					var:iw_funding_level < 1
				}
			}
			custom_tooltip_no_bullet = je_iw_ladder_row_0_current
		}
		else = {
			custom_tooltip_no_bullet = je_iw_ladder_row_0
		}
		if = {
			limit = {
				has_variable = iw_funding_level
				var:iw_funding_level = 1
			}
			custom_tooltip_no_bullet = je_iw_ladder_row_1_current
		}
		else = {
			custom_tooltip_no_bullet = je_iw_ladder_row_1
		}
		if = {
			limit = {
				has_variable = iw_funding_level
				var:iw_funding_level = 2
			}
			custom_tooltip_no_bullet = je_iw_ladder_row_2_current
		}
		else = {
			custom_tooltip_no_bullet = je_iw_ladder_row_2
		}
		if = {
			limit = {
				has_variable = iw_funding_level
				var:iw_funding_level = 3
			}
			custom_tooltip_no_bullet = je_iw_ladder_row_3_current
		}
		else = {
			custom_tooltip_no_bullet = je_iw_ladder_row_3
		}
		if = {
			limit = {
				has_variable = iw_funding_level
				var:iw_funding_level = 4
			}
			custom_tooltip_no_bullet = je_iw_ladder_row_4_current
		}
		else = {
			custom_tooltip_no_bullet = je_iw_ladder_row_4
		}
		if = {
			limit = {
				has_variable = iw_funding_level
				var:iw_funding_level >= 5
			}
			custom_tooltip_no_bullet = je_iw_ladder_row_5_current
		}
		else = {
			custom_tooltip_no_bullet = je_iw_ladder_row_5
		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- YES/NO: is every operation dormant? -----------------------------------
# True while funding is 0. At that level nothing runs (the journal entry's pulse
# skips covert_ops_apply_all_phase_effects and the detection roll) and no new
# operation can be launched, so the panel says so once and every row repeats it.
covert_ops_dormant_sgui = {
	scope = country

	is_shown = {
		OR = {
			NOT = { has_variable = iw_funding_level }
			var:iw_funding_level < 1
		}
	}

	is_valid = {
		always = no
	}

	effect = { }

	ai_is_valid = {
		always = no
	}
}

# ---- TOOLTIP BUILDER: what our counterintelligence last exposed -------------
# The only thing this journal entry says about hostile operations, and it says it
# only about one that was actually caught: covert_op_burn writes the record at
# the moment the target's counterintelligence exposed the operation.
#
# The attacker's name is produced by navigating into the stored country in script
# and printing [THIS.GetCountry.GetName] on a line inside that scope —
# gui_modding_guide.md gotcha #11 workaround (b), because Var().GetCountry.GetName
# renders blank in this mod. Storing the country rather than its capital is what
# makes annexation safe: `exists` fails and the anonymous line renders, instead of
# a months-old capital state naming whoever conquered it.
covert_last_exposed_sgui = {
	scope = country

	effect = {
		if = {
			limit = {
				has_variable = iw_last_exposed_type
				exists = var:iw_last_exposed_country
			}
			var:iw_last_exposed_country = {
				custom_tooltip = je_iw_last_exposed_named
			}
		}
		else_if = {
			limit = { has_variable = iw_last_exposed_type }
			custom_tooltip_no_bullet = je_iw_last_exposed_unknown
		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- YES/NO: is there a recent exposure worth showing? ----------------------
# Ten years is a display cut-off, not a mechanic: past that the record is history
# rather than intelligence, and the panel drops the line instead of reporting
# "412 months ago".
covert_last_exposed_known_sgui = {
	scope = country

	is_shown = {
		has_variable = iw_last_exposed_type
		has_variable = iw_last_exposed_age
		var:iw_last_exposed_age < 120
	}

	is_valid = {
		always = no
	}

	effect = { }

	ai_is_valid = {
		always = no
	}
}

# ---- YES/NO: do all operations carry their display phase yet? ---------------
# iw_phase / iw_phase_months_left are written by covert_op_refresh_phase, which
# is new. A save made before this widget shipped has containers without them
# until the first monthly pulse after loading, and a .gui that reads a variable
# that does not exist logs "Failed to fetch variable" every frame. The widget
# hides the phase lines until this answers yes.
covert_ops_phase_ready_sgui = {
	scope = country

	is_shown = {
		OR = {
			NOT = { has_variable_list = iw_ops }
			NOT = {
				any_in_list = {
					variable = iw_ops
					NOT = { has_variable = iw_phase }
				}
			}
		}
	}

	is_valid = {
		always = no
	}

	effect = { }

	ai_is_valid = {
		always = no
	}
}

# ---- STAND DOWN: one handler per operation type ----------------------------
# A widget row can end the operation it is showing. One scripted GUI per type,
# rather than one with a type op code, for a specific reason: the row must tell
# script BOTH which type and which target, and the type is already settled by the
# row's own per-tag markup, which lets the target travel alone in a SINGLE
# AddScope. Chaining two AddScope calls type-checks but has no vanilla precedent
# (docs/guides/gui_modding_guide.md, "One scripted GUI, several buttons"), and
# .gui errors only surface in-game, so the shape vanilla demonstrates wins.
#
# The remaining unproven link is that an object scope passed through AddScope
# arrives in a scripted GUI's saved_scopes. Passing an object that way is proven
# in this mod for a script value (gui/market_panel.gui reads scope:base_market),
# but not yet for a scripted GUI. covert_possible_stand_down is therefore
# fail-closed: if the scope arrives unset the control greys out. The worst case
# is a permanently disabled button, never a wrong operation ended — and the
# diplomacy UI's own break-pact path is untouched either way.
#
# Every one of these delegates to covert_possible_stand_down /
# covert_effect_stand_down; the only thing that differs is the two parameters.

covert_stand_down_election_interference_sgui = {
	scope = country
	saved_scopes = { iw_tgt }

	is_shown = {
		always = yes
	}

	is_valid = {
		covert_possible_stand_down = { TYPE = election_interference }
	}

	effect = {
		covert_effect_stand_down = { TYPE = election_interference ACTION = covert_election_interference_action }
	}

	ai_is_valid = {
		always = no
	}
}

# ---- PRIORITY: the row stepper ----------------------------------------------
# Two handlers, up and down, so the row's container can travel alone in a
# SINGLE AddScope — the shape the stand-down handlers use. Both delegate to
# covert_possible_priority_* / covert_effect_priority_*; the eligibility is
# fail-closed, so an unset or foreign scope greys the button instead of
# changing the wrong operation.
covert_priority_up_sgui = {
	scope = country
	saved_scopes = { iw_op }

	is_shown = {
		always = yes
	}

	is_valid = {
		covert_possible_priority_up = yes
	}

	effect = {
		covert_effect_priority_up = yes
	}

	ai_is_valid = {
		always = no
	}
}

covert_priority_down_sgui = {

# ---- YES/NO: do all operations carry a priority yet? -------------------------
# iw_priority is new in slice 3. A save made before it has containers without
# it until the first monthly sync backfills them, and a .gui that reads a
# missing variable logs every frame. The row hides its priority stepper and
# line until this answers yes. Same shape as covert_ops_phase_ready_sgui.
covert_ops_priority_ready_sgui = {
	scope = country

	is_shown = {
		OR = {
			NOT = { has_variable_list = iw_ops }
			NOT = {
				any_in_list = {
					variable = iw_ops
					NOT = { has_variable = iw_priority }
				}
			}
		}
	}

	is_valid = {
		always = no
	}

	effect = { }

	ai_is_valid = {
		always = no
	}
}
