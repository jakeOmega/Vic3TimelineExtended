# ============================================================================
# Strategic Reserve — Scripted GUIs
# ============================================================================
# Thin validation wrappers for the per-good controls in the Strategic Reserve
# inventory widget (gui/journal_entry_widgets/strategic_reserve_widget.gui).
# They own NO gameplay rules of their own: every branch delegates to the shared
# rate effects in common/scripted_effects/st_res_effects.txt, and visibility
# delegates to the per-good unlock triggers in
# common/scripted_triggers/st_res_triggers.txt.
#
# One scripted GUI per good, with the button's direction passed in from the
# .gui as a saved scope. IMPLICIT CONTRACT with the widget — keep both sides in
# sync:
#
#   scope:dir = 0  ->  decrease this good's weekly rate by the shared step
#   scope:dir = 1  ->  stop  (set this good's weekly rate to exactly 0)
#   scope:dir = 2  ->  increase this good's weekly rate by the shared step
#
# The widget passes it as
#   AddScope( 'dir', MakeScopeValue( '(CFixedPoint)<n>' ) )
# (the vanilla je_meiji_restoration_get_faction_sgui pattern).
#
# `is_valid` phrases its custom tooltips as CONDITIONS, not complaints, because
# ScriptedGui.IsValidTooltip renders them with a tick or a cross either way.
#
# MANUAL TAKEOVER. All three directions run through the rate bases, which call
# st_res_switch_to_manual_base first. Touching a rate by hand therefore drops
# that good out of any automated reserve policy, visibly, on the same click.
# That is deliberate, and it is the only such path besides "Reset Reserve
# Rates" — the automatic stockpile clamp must NOT count as intervention.
#
# All of these are player-only (`ai_is_valid = { always = no }`), matching the
# `ai_chance = { value = 0 }` on every Strategic Reserve scripted button: these
# wrappers exist to validate player clicks. Since reserve policies were added
# the AI DOES drive reserve rates, but it does so through
# st_res_ai_seed_policies_effect and the shared weekly evaluator in
# common/scripted_effects/st_res_effects.txt — a script path, never this one.
# ============================================================================

st_res_adjust_grain_sgui = {
	scope = country
	saved_scopes = { dir }

	is_shown = {
		has_journal_entry = je_strategic_reserve
		st_res_grain_unlocked_trigger = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:dir = 0 }
			custom_tooltip = {
				text = "st_res_rate_adjustment_possible"
				var:st_res_grain_rate > st_res_grain_max_withdrawable
				var:st_res_grain_rate > st_res_neg_weekly_base_rate_cap
			}
		}
		trigger_else_if = {
			limit = { scope:dir = 2 }
			custom_tooltip = {
				text = "st_res_rate_adjustment_possible"
				var:st_res_grain_rate < st_res_grain_max_storable
				var:st_res_grain_rate < st_res_weekly_base_rate_cap
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = "st_res_rate_stop_possible"
				NOT = { var:st_res_grain_rate = 0 }
			}
		}
	}

	effect = {
		if = {
			limit = { scope:dir = 0 }
			st_res_decrease_grain_rate_effect = yes
		}
		else_if = {
			limit = { scope:dir = 2 }
			st_res_increase_grain_rate_effect = yes
		}
		else = {
			st_res_stop_grain_rate_effect = yes
		}
	}
}

# ============================================================================
# Reserve policy controls — one scripted GUI per good
# ============================================================================
# The second scripted GUI a good owns (the first is st_res_adjust_<good>_sgui
# above). It backs every control in the row's policy panel: the four policy
# buttons, the three presets and the eight settings steppers. Like its sibling it
# owns NO gameplay rules — every branch delegates to a shared `$GOOD$` effect in
# common/scripted_effects/st_res_effects.txt, and every branch ends in
# st_res_policy_apply_good_effect so the change is re-evaluated and the hub
# refreshed on the same click.
#
# IMPLICIT CONTRACT with gui/journal_entry_widgets/strategic_reserve_widget.gui
# — keep both sides in sync. The widget passes a single integer as
#   AddScope( 'op', MakeScopeValue( '(CFixedPoint)<n>' ) )
# because one saved scope is the shape vanilla demonstrates
# (je_meiji_restoration_get_faction_sgui), so the op code encodes both the
# action and which setting it acts on:
#
#   op  | action
#   ----+--------------------------------------------------------------------
#    0  | select policy Manual
#    1  | select policy Buy When Cheap
#    2  | select policy Release When Expensive
#    3  | select policy Stabilize Prices
#   10  | apply the Conservative preset
#   11  | apply the Standard preset
#   12  | apply the Aggressive preset
#   20  | purchase threshold  -  (st_res_policy_thr_step)
#   21  | purchase threshold  +
#   22  | release threshold   -
#   23  | release threshold   +
#   24  | maximum weekly flow -  (st_res_policy_flow_step)
#   25  | maximum weekly flow +
#   26  | protected stockpile -  (st_res_policy_pct_step, % of capacity)
#   27  | protected stockpile +
#   28  | target stockpile    -  (st_res_policy_pct_step, % of capacity)
#   29  | target stockpile    +
#   30  | weekly budget       -  (st_res_policy_budget_step)
#   31  | weekly budget       +
#   32  | price memory        -  (st_res_policy_memory_step, weeks averaged)
#   33  | price memory        +
#   34  | response ramp       -  (st_res_policy_ramp_step, pp past a threshold)
#   35  | response ramp       +
#   44  | maximum weekly flow -  ten steps (shift-click)
#   45  | maximum weekly flow +  ten steps (shift-click)
#   50  | weekly budget       -  ten steps (shift-click)
#   51  | weekly budget       +  ten steps (shift-click)
#   64  | maximum weekly flow -  a hundred steps (ctrl-click)
#   65  | maximum weekly flow +  a hundred steps (ctrl-click)
#   70  | weekly budget       -  a hundred steps (ctrl-click)
#   71  | weekly budget       +  a hundred steps (ctrl-click)
#
# The modifier-click ops are the plain op + 20 (shift) and + 40 (ctrl). They
# share the plain op's is_valid branch — the check is "any room left", and the
# step base clamps an oversized step to the bound — so the widget binds
# `enabled` to the plain op alone. The flow ceiling (st_res_policy_flow_max)
# and each good's budget ceiling (st_res_<good>_policy_budget_max) follow the
# hub's weekly flow cap rather than being constants; see st_res_script_values.
#
# `is_valid` carries the whole validation surface, so a disabled stepper always
# explains itself in the tooltip and no rule lives only in a .gui expression:
#   - a policy button is invalid when that policy is already selected;
#   - a stepper is invalid at its absolute bound;
#   - the purchase/release threshold steppers are additionally invalid when one
#     more click would let the two bands overlap, and the floor/ceiling
#     steppers when one more click would cross the stock band.
# The relative limits are read as `var:X <= st_res_<good>_policy_*_limit`
# script values, because a Paradox trigger needs a `var:` on its left side.
#
# Phrased as CONDITIONS, not complaints, because ScriptedGui.IsValidTooltip
# renders them with a tick or a cross either way — same convention as above.
#
# `ai_is_valid = { always = no }`: these wrappers exist to validate player
# clicks. The AI reaches the very same policies through
# st_res_ai_seed_policies_effect and the shared weekly evaluator instead, which
# is a script path, not a UI path.
# ============================================================================

st_res_policy_grain_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_strategic_reserve
		st_res_grain_unlocked_trigger = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:op = 0 }
			custom_tooltip = {
				text = "st_res_policy_select_possible"
				NOT = { var:st_res_grain_policy = 0 }
			}
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			custom_tooltip = {
				text = "st_res_policy_select_possible"
				NOT = { var:st_res_grain_policy = 1 }
			}
		}
		trigger_else_if = {
			limit = { scope:op = 2 }
			custom_tooltip = {
				text = "st_res_policy_select_possible"
				NOT = { var:st_res_grain_policy = 2 }
			}
		}
		trigger_else_if = {
			limit = { scope:op = 3 }
			custom_tooltip = {
				text = "st_res_policy_select_possible"
				NOT = { var:st_res_grain_policy = 3 }
			}
		}
		trigger_else_if = {
			# Presets 10-12 are always legal: every preset is internally
			# consistent by construction. Listed explicitly rather than as a
			# range so an op code nobody defined falls through to the
			# fail-closed trigger_else at the bottom.
			limit = {
				OR = {
					scope:op = 10
					scope:op = 11
					scope:op = 12
				}
			}
			always = yes
		}
		trigger_else_if = {
			limit = { scope:op = 20 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_buy_thr > st_res_policy_thr_floor
			}
		}
		trigger_else_if = {
			limit = { scope:op = 21 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_buy_thr < 0
			}
			custom_tooltip = {
				text = "st_res_policy_band_possible"
				var:st_res_grain_buy_thr <= st_res_grain_policy_buy_up_limit
			}
		}
		trigger_else_if = {
			limit = { scope:op = 22 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_sell_thr > 0
			}
			custom_tooltip = {
				text = "st_res_policy_band_possible"
				var:st_res_grain_sell_thr >= st_res_grain_policy_sell_down_limit
			}
		}
		trigger_else_if = {
			limit = { scope:op = 23 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_sell_thr < st_res_policy_thr_ceiling
			}
		}
		trigger_else_if = {
			# 24 one step, 44 ten (shift-click), 64 a hundred (ctrl-click).
			limit = {
				OR = {
					scope:op = 24
					scope:op = 44
					scope:op = 64
				}
			}
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_max_flow > st_res_policy_flow_min
			}
		}
		trigger_else_if = {
			# 25 one step, 45 ten (shift-click), 65 a hundred (ctrl-click).
			limit = {
				OR = {
					scope:op = 25
					scope:op = 45
					scope:op = 65
				}
			}
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_max_flow < st_res_policy_flow_max
			}
		}
		trigger_else_if = {
			limit = { scope:op = 26 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_floor_pct > st_res_policy_stock_min
			}
		}
		trigger_else_if = {
			limit = { scope:op = 27 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_floor_pct < st_res_policy_stock_max
			}
			custom_tooltip = {
				text = "st_res_policy_stock_band_possible"
				var:st_res_grain_floor_pct <= st_res_grain_policy_floor_up_limit
			}
		}
		trigger_else_if = {
			limit = { scope:op = 28 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_ceil_pct > st_res_policy_stock_min
			}
			custom_tooltip = {
				text = "st_res_policy_stock_band_possible"
				var:st_res_grain_ceil_pct >= st_res_grain_policy_ceil_down_limit
			}
		}
		trigger_else_if = {
			limit = { scope:op = 29 }
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_ceil_pct < st_res_policy_stock_max
			}
		}
		trigger_else_if = {
			# 30 one step, 50 ten (shift-click), 70 a hundred (ctrl-click).
			limit = {
				OR = {
					scope:op = 30
					scope:op = 50
					scope:op = 70
				}
			}
			custom_tooltip = {
				text = "st_res_policy_range_possible"
				var:st_res_grain_budget > st_res_policy_budget_min
			}
