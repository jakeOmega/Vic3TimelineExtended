# ============================================================================
# NUCLEAR PROGRAMME — scripted GUIs for gui/journal_entry_widgets/nuclear_program_widget.gui
# ============================================================================
# Three handlers, and that is deliberately all: the programme panel's rows are
# ordinary localization reading display-only script values and
# common/customizable_localization/nuclear_program_custom_loc.txt, which is
# cheaper per frame than ExecuteTooltip and needs no handler at all. A scripted
# GUI is used only where nothing else can do the job.
#
# ---- nuclear_program_has_programme_sgui — display only ---------------------
# Scope-free yes/no: does this country have a programme? The programme panel's
# root asks it, because the entry is also active for an armed power without
# the standing to build and for a non-nuclear party to a crisis, and neither
# has anything to fund.
#
# ---- nuclear_program_funding_sgui — op table -------------------------------
# Keep in step with the .gui header and docs/systems/journal_entry_systems.md.
#   0   step programme funding DOWN   -> nuclear_program_effect_decrease_funding
#   1   step programme funding UP     -> nuclear_program_effect_increase_funding
#
# Both ops delegate `is_valid` to the same scripted trigger and `effect` to the
# same scripted effect that the journal entry's scripted buttons call, so the
# panel and the AI's grid cannot drift apart. Neither carries a custom_tooltip:
# IsValidTooltip renders the trigger's own clauses with a tick or a cross each
# ("innovation above 100", "no disarmament settlement", "no pause treaty"), which
# is both more informative than a hand-written sentence and impossible to let
# drift away from the real condition.
#
# ---- nuclear_program_powers_sgui — op table --------------------------------
#   op = leaderboard rank, 1 to 10.
# One ExecuteTooltip per GUI row rather than one block listing all ten, because
# a scope's own tooltip lines print before the lines produced inside nested
# scope changes (gui_modding_guide.md gotcha #18) — separate rows make that
# impossible. No flag icon in the row text: the engine already prefixes a
# country-scope tooltip line with that country's flag (gotcha #19).
#
# ---- WHY NEITHER `is_shown` DEPENDS ON `scope:op` --------------------------
# Whether a saved scope passed as `IsShown( … AddScope('op', …) … )` actually
# reaches a handler's `is_shown` is UNPROVEN in this mod. Vanilla's
# scripted_guis.md says saved scopes are available "in triggers / effects", and
# `is_valid` demonstrably receives them (the st_res_* policy handlers depend on
# it), but no vanilla handler reads a saved scope inside `is_shown`, and the one
# in-mod case is written fail-open and may never have been exercised. With this
# journal entry's button grid hidden from humans, a control whose `is_shown`
# quietly evaluated false would leave a player no way to act at all. So:
#
#   * nuclear_program_funding_sgui — the ACTION handler — has a scope-free
#     `is_shown`, and the .gui does not drive the stepper buttons' `visible`
#     from it at all. Eligibility is entirely `is_valid`'s job, which is the
#     proven path, and which greys the button and says why.
#   * nuclear_program_powers_sgui — DISPLAY only — is written FAIL-OPEN: with no
#     `scope:op` it returns yes, and emptiness is guarded inside the effect
#     instead. ExecuteTooltip ignores `is_shown` anyway (gotcha #16), so the
#     worst case is a blank line inside a section that is collapsed by default,
#     never a list that refuses to appear.
#
# `effect` reading `scope:op` to pick which rank to render IS the proven path.
# ============================================================================

# ---------------------------------------------------------------------------
# IS THERE A PROGRAMME?
# ---------------------------------------------------------------------------
nuclear_program_has_programme_sgui = {
	scope = country
	is_shown = {
		nuclear_program_has_programme = yes
	}
	is_valid = { always = no }
	ai_is_valid = { always = no }
	effect = { }
}

# ---------------------------------------------------------------------------
# THE FUNDING STEPPER
# ---------------------------------------------------------------------------
nuclear_program_funding_sgui = {
	scope = country
	saved_scopes = { op }

	# Deliberately scope-free — see "WHY NEITHER `is_shown` DEPENDS ON `scope:op`"
	# in the header. In practice this block is now vestigial: the .gui never calls
	# IsShown on the stepper buttons, and IsValid does not consult is_shown. What
	# actually keeps nuclear_program_possible_decrease_funding — which reads a
	# variable only an activated entry has created — from running too early is the
	# widget root's `visible = "[JournalEntry.IsActive]"`. It is kept as the honest
	# declaration of when this handler means anything, and as the guard should some
	# future caller ask IsShown. Which of the two ops is allowed right now is
	# `is_valid`'s business, not this block's.
	is_shown = {
		has_journal_entry = je_nuclear_program
	}

	is_valid = {
		trigger_if = {
			limit = {
				exists = scope:op
				scope:op = 0
			}
			nuclear_program_possible_decrease_funding = yes
		}
		trigger_else_if = {
			limit = { scope:op = 1 }
			nuclear_program_possible_increase_funding = yes
		}
		trigger_else = {
			always = no
		}
	}

	# The AI reaches the programme through the journal entry's scripted buttons
	# and their ai_chance blocks, never through here.
	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = {
				exists = scope:op
				scope:op = 0
			}
			nuclear_program_effect_decrease_funding = yes
		}
		else_if = {
			limit = { scope:op = 1 }
			nuclear_program_effect_increase_funding = yes
		}
	}
}

# THE NUCLEAR POWERS LEADERBOARD — display only
# ---------------------------------------------------------------------------
# The nuke_rank_N globals hold *countries*, and the GUI chain
# Var('x').GetCountry.GetName is recorded as not working in this mod
# (gui_modding_guide.md gotcha #11), with no verified way to list countries in a
# .gui at all (gotcha #15). So the row is entered in script and the line prints
# [THIS.GetCountry.GetName], which is vanilla's own shape for this
# (je_hispanoamerica_not_recognized_countries_sgui). The ranking itself is not
# recomputed here: update_nuclear_powers_ranking already writes these globals
# once a week from a single ordered pass, and this only renders them. Disclosure
# is unchanged from the eleven triggered_desc lines this replaces — the top ten
# nuclear powers and their stockpiles, nothing more.
nuclear_program_powers_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		# FAIL-OPEN first branch: if the saved scope never arrives, every row
		# returns yes and the effect below simply renders nothing for the ranks
		# that are empty. See the header for why this path is not trusted.
		trigger_if = {
			limit = { NOT = { exists = scope:op } }
			always = yes
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 1 }
			has_global_variable = nuke_rank_1
			exists = global_var:nuke_rank_1
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 2 }
			has_global_variable = nuke_rank_2
			exists = global_var:nuke_rank_2
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 3 }
			has_global_variable = nuke_rank_3
			exists = global_var:nuke_rank_3
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 4 }
			has_global_variable = nuke_rank_4
			exists = global_var:nuke_rank_4
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 5 }
			has_global_variable = nuke_rank_5
			exists = global_var:nuke_rank_5
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 6 }
			has_global_variable = nuke_rank_6
			exists = global_var:nuke_rank_6
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 7 }
			has_global_variable = nuke_rank_7
			exists = global_var:nuke_rank_7
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 8 }
			has_global_variable = nuke_rank_8
			exists = global_var:nuke_rank_8
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 9 }
			has_global_variable = nuke_rank_9
			exists = global_var:nuke_rank_9
		}
		trigger_else_if = {
			limit = { exists = scope:op  scope:op = 10 }
			has_global_variable = nuke_rank_10
			exists = global_var:nuke_rank_10
		}
		trigger_else = {
			always = no
		}
	}

	is_valid = {
		always = no
	}

	ai_is_valid = {
		always = no
	}

	effect = {
		if = {
			limit = {
				exists = scope:op
				scope:op = 1
				has_global_variable = nuke_rank_1
			}
			global_var:nuke_rank_1 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_1 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 2  has_global_variable = nuke_rank_2 }
			global_var:nuke_rank_2 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_2 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 3  has_global_variable = nuke_rank_3 }
			global_var:nuke_rank_3 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_3 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 4  has_global_variable = nuke_rank_4 }
			global_var:nuke_rank_4 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_4 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 5  has_global_variable = nuke_rank_5 }
			global_var:nuke_rank_5 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_5 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 6  has_global_variable = nuke_rank_6 }
			global_var:nuke_rank_6 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_6 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 7  has_global_variable = nuke_rank_7 }
			global_var:nuke_rank_7 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_7 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 8  has_global_variable = nuke_rank_8 }
			global_var:nuke_rank_8 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_8 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 9  has_global_variable = nuke_rank_9 }
			global_var:nuke_rank_9 ?= { custom_tooltip_no_bullet = nuclear_program_power_row_9 }
		}
		else_if = {
			limit = { exists = scope:op  scope:op = 10  has_global_variable = nuke_rank_10 }
