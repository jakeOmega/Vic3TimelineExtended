# Design: docs/superpowers/archive/specs/2026-10-26-solar-colonization-fleets-design.md
# § 16. The buttons of the Fleets rows (gui/journal_entry_widgets/
# sr_fleets_generated_rows.gui, one per slot, which pass the slot as scope:op)
# and of Space Command's build mix (scope:op the role: 1 warships, 2 rapid
# response, 3 siege ships, 4 assault carriers, 5 tenders), and of a fleet's
# Transfer popup (scope:op the slot, with scope:pick and scope:step; under
# Transfer below). Full Colonies only;
# player-only: the AI orders its fleets through the same effects from its
# monthly review (sr_fl_ai_review). Each is_valid is the trigger its effect's
# dispatcher checks (sr_fl_<n>_can_go and the rest, generated). Two have no
# effect: sr_fleet_pick_sgui and sr_fleet_target_sgui, the gates of the buttons
# that open the destination and target popups, read only for the buttons'
# greyed state and reason (never executed).

# A − or + in the popup: scope:pick the role (1 warships, 2 rapid response, 3
# siege ships, 4 assault carriers, 5 tenders, 6 marines), scope:step its code
# (gen_space_fleets.XF_STEPS: 1-4 fewer by 1, 10, 100, 1,000, 5 none; 6-9 more,
# 10 all), the construction market's keys. The amount is then held to 0 and
# what the slot has (sr_fl_xfer_step_role). Three chained AddScopes: unproven
# in game (TE_SELF_CHECK xfer_step_scope). The marines' grey while every ship
# goes, since every marine then goes too, and each at the bound the assault
# carriers' room sets (sr_fl_<n>_xf_marines_least, _most), which the amount is
# held to.
sr_fleet_xfer_step_sgui = {
	scope = country
	saved_scopes = { op pick step }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		custom_tooltip = {
			text = sr_fl_not_at_location_tt
			sr_fl_op_check = { TEST = at_location }
		}
		# The marines (6): with every ship going, every marine goes too, and the
		# row shows it (sr_disp_fl_<n>_xf_marines).
		trigger_if = {
			limit = { scope:pick = 6 }
			custom_tooltip = {
				text = sr_fl_xf_marines_follow_tt
				sr_fl_op_check = { TEST = xf_some_stays }
			}
		}
		# Else − at the fewest the carriers that stay leave room for, + at the
		# most the fleet it goes into has room for. The button's greyed state
		# and hover are its plain click's (step 1 or 6).
		trigger_if = {
			limit = {
				scope:pick = 6
				scope:step < 6
				sr_fl_op_check = { TEST = xf_some_stays }
			}
			custom_tooltip = {
				text = sr_fl_xf_marines_fewer_tt
				sr_fl_op_check = { TEST = xf_marines_fewer }
			}
		}
		trigger_if = {
			limit = {
				scope:pick = 6
				scope:step > 5
				sr_fl_op_check = { TEST = xf_some_stays }
			}
			custom_tooltip = {
				text = sr_fl_xf_marines_more_tt
				sr_fl_op_check = { TEST = xf_marines_more }
			}
		}
	}

	effect = {
		if = {
			limit = { scope:step < 6 }
			custom_tooltip = sr_fl_xf_less_tt
		}
		else = {
			custom_tooltip = sr_fl_xf_more_tt
		}
		hidden_effect = {
			sr_fl_by_op = { DO = xfer_step }
		}
	}
}

# The posture buttons (UI pass design § 7): scope:op the slot, scope:pick the
# posture (1 Station, 2 Stand off, 3 Close in, 4 Interdict), set at once with
# what taking it brings (sr_fl_set_posture_at: a target, a price). Each rule is
# its own line: never Stand off or Interdict at Earth (lunar colonization
# §§ 8.1, 8.3), Close in on a hostile world here (at Earth, a lunar one),
# Interdict on a world here of a country at peace with us and not our ally.
sr_fleet_set_posture_sgui = {
	scope = country
	saved_scopes = { op pick }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		# True wherever the panel shows the button (the posture line shows only
		# at a location, and the posture held is lit, not a button): checked,
		# not listed, so the hover keeps to what can fail (under 500 characters).
		hidden_trigger = {
			sr_fl_op_check = { TEST = at_location }
			NOT = { sr_fl_op_check = { TEST = pick_is_posture } }
		}
		# Never Stand off at Earth: listed only there, where it fails.
		trigger_if = {
			limit = {
				scope:pick = 2
				sr_fl_op_check = { TEST = at_earth }
			}
			custom_tooltip = {
				text = sr_fl_req_standoff_not_earth_tt
				NOT = { sr_fl_op_check = { TEST = at_earth } }
			}
		}
		trigger_if = {
			limit = { scope:pick = 3 }
			custom_tooltip = {
				text = sr_fl_req_hostile_world_tt
				sr_fl_op_check = { TEST = has_hostile_world }
			}
		}
		# Never Interdict at Earth (where no world may be interdicted); elsewhere
		# a world here it may interdict.
		trigger_if = {
			limit = { scope:pick = 4 }
			trigger_if = {
				limit = { sr_fl_op_check = { TEST = at_earth } }
				custom_tooltip = {
					text = sr_fl_req_interdict_not_earth_tt
					NOT = { sr_fl_op_check = { TEST = at_earth } }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = sr_fl_req_interdict_world_tt
					sr_fl_op_check = { TEST = has_interdict_world }
				}
			}
		}
		# UI pass design § 10: no Space Command, no fleet orders (the short
		# line: the hover goes on to what the posture does and costs).
		custom_tooltip = {
			text = sr_fl_req_space_command_short_tt
			sr_sc_operational = yes
		}
	}

	effect = {
		if = {
			limit = { scope:pick = 1 }
			if = {
				limit = { sr_fl_op_check = { TEST = at_earth } }
				custom_tooltip = sr_fl_set_posture_1_earth_tt
			}
			else = {
				custom_tooltip = sr_fl_set_posture_1_tt
			}
		}
		else_if = {
			limit = { scope:pick = 2 }
			custom_tooltip = sr_fl_set_posture_2_tt
		}
		else_if = {
			limit = { scope:pick = 3 }
			if = {
				limit = { sr_fl_op_check = { TEST = at_earth } }
				custom_tooltip = sr_fl_set_posture_3_earth_tt
			}
			else = {
				custom_tooltip = sr_fl_set_posture_3_tt
			}
			# The free colony's price, on the world it will close in on: a
			# hostile target stays its target; else the hostile step's pick
			# (sr_fl_set_posture_at).
			if = {
				limit = {
					OR = {
						AND = {
							sr_fl_op_check = { TEST = target_hostile }
							sr_fl_op_check = { TEST = closes_on_free }
						}
						AND = {
							NOT = { sr_fl_op_check = { TEST = target_hostile } }
							sr_fl_op_check = { TEST = hostile_step_free }
						}
					}
				}
				custom_tooltip = sr_fl_free_price_tt
			}
		}
		else = {
			custom_tooltip = sr_fl_set_posture_4_tt
			custom_tooltip = sr_fl_interdict_price_tt
		}
		hidden_effect = {
			sr_fl_by_op = { DO = set_posture }
		}
	}
}

sr_fleet_set_target_sgui = {
	scope = country
	saved_scopes = { op pick }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		# True wherever the popup shows the button (its rows show only for the
		# worlds where the fleet is): checked, not listed, so the hover keeps
		# to what can fail (under 500 characters).
		hidden_trigger = {
			sr_fl_op_check = { TEST = at_location }
			sr_fl_op_check = { TEST = pick_world_here }
		}
		trigger_if = {
			limit = { sr_fl_op_check = { TEST = interdicting } }
			custom_tooltip = {
				text = sr_fl_req_pick_interdictable_tt
				sr_fl_pick_interdictable = yes
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = sr_fl_req_pick_hostile_tt
				sr_fl_pick_hostile = yes
			}
		}
		# UI pass design § 10: no Space Command, no fleet orders (the short
		# line, as the posture buttons').
		custom_tooltip = {
			text = sr_fl_req_space_command_short_tt
			sr_sc_operational = yes
		}
	}

	effect = {
		if = {
			limit = { sr_fl_op_check = { TEST = interdicting } }
			custom_tooltip = sr_fl_set_icpt_target_tt
			custom_tooltip = sr_fl_interdict_price_tt
		}
		else = {
			custom_tooltip = sr_fl_set_target_tt
			if = {
				limit = {
					sr_fl_op_check = { TEST = closing_in }
					sr_fl_pick_free = yes
				}
				custom_tooltip = sr_fl_free_price_tt
			}
		}
		hidden_effect = {
			sr_fl_by_op = { DO = set_target }
		}
	}
}

# Space Command's build mix (the owner, 2026-11-25: - and + in the table's Mix
# column; one button a role until then, stepping 0, 1, 2, 3, 0): scope:op the
# role (gen_space_fleets.ROLE_CODE), scope:pick 1 a step off its weight, 2 a
# step on (MIX_STEPS), held to 0 and sr_sc_mix_max. is_shown reads no saved
# scope (gotcha #22).
sr_sc_mix_sgui = {
	scope = country
	saved_scopes = { op pick }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = { scope:pick = 2 }
			custom_tooltip = {
				text = sr_sc_mix_below_max_tt
				sr_sc_mix_op_value < sr_sc_mix_max
			}
		}
		trigger_else_if = {
			limit = { scope:pick = 1 }
			custom_tooltip = {
				text = sr_sc_mix_above_zero_tt
				sr_sc_mix_op_value > 0
			}
		}
		trigger_else = {
			always = no
		}
	}

	# Each role's tooltip names it, its weight and its share; the roles' rules
	# are the concept's (concept_sr_ship_role).
	effect = {
		if = { limit = { scope:op = 1 } custom_tooltip = sr_sc_mix_step_war_tt }
		else_if = { limit = { scope:op = 2 } custom_tooltip = sr_sc_mix_step_rapid_tt }
		else_if = { limit = { scope:op = 3 } custom_tooltip = sr_sc_mix_step_siege_tt }
		else_if = { limit = { scope:op = 4 } custom_tooltip = sr_sc_mix_step_carrier_tt }
		else = { custom_tooltip = sr_sc_mix_step_tender_tt }
		if = {
			limit = { scope:pick = 2 }
			custom_tooltip = sr_sc_mix_up_tt
		}
		else = {
			custom_tooltip = sr_sc_mix_down_tt
		}
		hidden_effect = {
			sr_sc_mix_by_op = yes
		}
	}
}

# Space Command's orders (the owner, 2026-11-25; the table's Order column):
# scope:op the role, scope:pick the step (gen_space_fleets.ORDER_STEPS: 1 to 3
# take 1, 10 or 100 hulls off its order, 4 to 6 add them; ORDER_CLEAR, 7,
# withdraws it). A click is the plain step, shift-click ten, ctrl-click a
# hundred, which share the plain step's check (the button's enabled and hover
# are its plain click's), the effect holding the order to 0 and sr_sc_order_max.
sr_sc_order_sgui = {
	scope = country
	saved_scopes = { op pick }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = {
				scope:pick >= 4
				scope:pick <= 6
			}
			custom_tooltip = {
				text = sr_sc_order_below_max_tt
				sr_sc_order_op_value < sr_sc_order_max
			}
		}
		trigger_else_if = {
			limit = {
				scope:pick >= 1
				scope:pick <= 7
			}
			custom_tooltip = {
				text = sr_sc_order_some_tt
				sr_sc_order_op_value > 0
			}
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = { limit = { scope:op = 1 } custom_tooltip = sr_sc_order_war_tt }
		else_if = { limit = { scope:op = 2 } custom_tooltip = sr_sc_order_rapid_tt }
		else_if = { limit = { scope:op = 3 } custom_tooltip = sr_sc_order_siege_tt }
		else_if = { limit = { scope:op = 4 } custom_tooltip = sr_sc_order_carrier_tt }
		else = { custom_tooltip = sr_sc_order_tender_tt }
		if = {
			limit = { scope:pick = 7 }
			custom_tooltip = sr_sc_order_clear_tt
		}
		else_if = {
			limit = { scope:pick >= 4 }
			custom_tooltip = sr_sc_order_more_tt
		}
		else = {
			custom_tooltip = sr_sc_order_less_tt
		}
		hidden_effect = {
			sr_sc_order_by_op = yes
		}
	}
}

# Space Command's three modes (the owner, 2026-11-25; one Pause / Resume /
# Resuming button until 2026-10-31's): scope:op 1 Full, 2 Orders Only, 3
# Paused. The panel draws the mode in force selected, not as a button
# (sr_disp_sc_mode), so a click always changes the mode; choosing Paused while
# a resume waits calls it off. Paused, the yards build no fleet strength and
# buy no inputs; fleets, the reserve and Auto-launch carry on. Leaving a pause
# waits for the next monthly step, which builds nothing (sr_sc_month). The AI
# pauses through Paused's effect (sr_sc_ai_pause_review).
sr_sc_mode_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		always = yes
	}

	effect = {
		if = {
			limit = { scope:op = 1 }
			custom_tooltip = sr_sc_mode_full_tt
			if = {
				limit = { sr_sc_is_paused = yes }
				custom_tooltip = sr_sc_mode_resume_tt
			}
			hidden_effect = {
				sr_sc_mode_full = yes
			}
		}
		else_if = {
			limit = { scope:op = 2 }
			custom_tooltip = sr_sc_mode_orders_tt
			if = {
				limit = {
					sr_sc_is_paused = yes
					sr_sc_order_total > 0
				}
				custom_tooltip = sr_sc_mode_resume_tt
			}
			hidden_effect = {
				sr_sc_mode_orders = yes
			}
		}
		else_if = {
			limit = { scope:op = 3 }
			custom_tooltip = sr_sc_mode_paused_tt
			hidden_effect = {
				sr_sc_mode_paused = yes
			}
		}
	}
}
