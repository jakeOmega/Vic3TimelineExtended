# A − or + in the popup: scope:pick the role (1 warships, 2 rapid response, 3
# siege ships, 4 assault carriers, 5 tenders, 6 marines), scope:step its code
# (gen_space_fleets.XF_STEPS: 1-4 fewer by 1, 10, 100, 1,000, 5 none; 6-9 more,
# 10 all), the construction market's keys. The amount is then held to 0 and
# what the slot has (sr_fl_xfer_step_role). Three chained AddScopes: unproven
# in game (TE_SELF_CHECK xfer_step_scope). The marines' grey while every ship
# goes, since every marine then goes too.
sr_fleet_xfer_step_sgui = {
	scope = country
	saved_scopes = { op pick step }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		custom_tooltip = {
			text = sr_fl_not_at_location_tt
			sr_fl_op_check = { TEST = at_location }
		}
		# The marines (6): with every ship going, every marine goes too, and the
		# row shows it (sr_disp_fl_<n>_xf_marines).
		trigger_if = {
			limit = { scope:pick = 6 }
			custom_tooltip = {
				text = sr_fl_xf_marines_follow_tt
				sr_fl_op_check = { TEST = xf_some_stays }
			}
		}
	}

	effect = {
		if = {
			limit = { scope:step < 6 }

		else = {
			custom_tooltip = sr_fl_xf_more_tt
		}
		hidden_effect = {
			sr_fl_by_op = { DO = xfer_step }
		}
	}
}

# The posture buttons (UI pass design § 7): scope:op the slot, scope:pick the
# posture (1 Station, 2 Stand off, 3 Close in, 4 Interdict), set at once with
# what taking it brings (sr_fl_set_posture_at: a target, a price). Each rule is
# its own line: never Stand off or Interdict at Earth (lunar colonization
# §§ 8.1, 8.3), Close in on a hostile world here (at Earth, a lunar one),
# Interdict on a world here of a country at peace with us and not our ally.
sr_fleet_set_posture_sgui = {
	scope = country
	saved_scopes = { op pick }

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		# True wherever the panel shows the button (the posture line shows only
		# at a location, and the posture held is lit, not a button): checked,
		# not listed, so the hover keeps to what can fail (under 500 characters).
		hidden_trigger = {
			sr_fl_op_check = { TEST = at_location }
			NOT = { sr_fl_op_check = { TEST = pick_is_posture } }
		}
		# Never Stand off at Earth: listed only there, where it fails.
		trigger_if = {
			limit = {
				scope:pick = 2
				sr_fl_op_check = { TEST = at_earth }
			}
			custom_tooltip = {
				text = sr_fl_req_standoff_not_earth_tt
				NOT = { sr_fl_op_check = { TEST = at_earth } }
			}
		}
		trigger_if = {
			limit = { scope:pick = 3 }
			custom_tooltip = {
				text = sr_fl_req_hostile_world_tt
				sr_fl_op_check = { TEST = has_hostile_world }
			}
		}
		# Never Interdict at Earth (where no world may be interdicted); elsewhere
		# a world here it may interdict.
		trigger_if = {
			limit = { scope:pick = 4 }
			trigger_if = {
				limit = { sr_fl_op_check = { TEST = at_earth } }
				custom_tooltip = {
					text = sr_fl_req_interdict_not_earth_tt
					NOT = { sr_fl_op_check = { TEST = at_earth } }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = sr_fl_req_interdict_world_tt
					sr_fl_op_check = { TEST = has_interdict_world }
				}
			}
		}
		# UI pass design § 10: no Space Command, no fleet orders (the short
		# line: the hover goes on to what the posture does and costs).
		custom_tooltip = {
			text = sr_fl_req_space_command_short_tt
			sr_sc_operational = yes
		}
	}

	effect = {
		if = {
			limit = { scope:pick = 1 }
			if = {
				limit = { sr_fl_op_check = { TEST = at_earth } }
				custom_tooltip = sr_fl_set_posture_1_earth_tt
			}
			else = {
				custom_tooltip = sr_fl_set_posture_1_tt
			}
		}
		else_if = {
			limit = { scope:pick = 2 }
			custom_tooltip = sr_fl_set_posture_2_tt
		}
		else_if = {
			limit = { scope:pick = 3 }
			if = {
				limit = { sr_fl_op_check = { TEST = at_earth } }
				custom_tooltip = sr_fl_set_posture_3_earth_tt
			}
			else = {
				custom_tooltip = sr_fl_set_posture_3_tt
			}
			# The free colony's price, on the world it will close in on: a
			# hostile target stays its target; else the hostile step's pick
			# (sr_fl_set_posture_at).
			if = {
				limit = {
					OR = {
						AND = {
							sr_fl_op_check = { TEST = target_hostile }
							sr_fl_op_check = { TEST = closes_on_free }
						}
						AND = {
							NOT = { sr_fl_op_check = { TEST = target_hostile } }
							sr_fl_op_check = { TEST = hostile_step_free }
						}
					}
				}
				custom_tooltip = sr_fl_free_price_tt
			}
		}
		else = {
			custom_tooltip = sr_fl_set_posture_4_tt
			custom_tooltip = sr_fl_interdict_price_tt
		}
		hidden_effect = {
			sr_fl_by_op = { DO = set_posture }
		}
	}
}

# month, blockade or not (design § 15). Only an occupied launch site stops it.
sr_reserve_launch_sgui = {
	scope = country

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		custom_tooltip = {
			text = sr_reserve_empty_tt
			sr_reserve_total > 0
		}
		custom_tooltip = {
			text = sr_launch_site_lost_tt
			NOT = { sr_launch_site_lost = yes }
		}
		custom_tooltip = {
			text = sr_reserve_no_slot_tt
			sr_reserve_may_launch = yes
		}
	}

	effect = {
		custom_tooltip = sr_reserve_launch_tt
		hidden_effect = {
			sr_reserve_launch = yes
		}
	}
}

# 6d: the reserve's toggle: on (the default), it launches itself each month
# while no hostile fleet is at Earth; off, it waits for Launch.
sr_reserve_auto_sgui = {
	scope = country

	is_shown = {
		has_journal_entry = je_space_race_solar_colonization
		sr_colonies_full_enabled = yes
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		always = yes
	}

	effect = {
		if = {
			limit = { sr_reserve_auto_on = yes }
			custom_tooltip = sr_reserve_auto_off_tt
		}
		else = {
			custom_tooltip = sr_reserve_auto_on_tt
		}
		hidden_effect = {
			sr_reserve_toggle_auto = yes
		}
	}
}
