# gui/journal_entry_widgets/grand_monuments_widget.gui. The three row buttons
# pass the row's state as the saved scope gm_state; which action is chosen by
# which scripted GUI the button points at (the covert rows' shape). A State
# object in saved_scopes has no vanilla precedent (gui gotcha #22), so every
# is_valid fails closed: the scope exists, we own it, it is contested.
# monument_events.17 is the proven path to the same three choices.

gm_tear_down_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			custom_tooltip = {
				text = gm_sgui_contested_tt
				gm_state_is_contested = yes
			}
		}
	}

	effect = {
		scope:gm_state = { gm_state_choose_tear_down = yes }
		hidden_effect = { gm_country_refresh = yes }
	}

	ai_is_valid = {
		always = no
	}
}

# ---- Dedicate (owner, 2026-10-18) ------------------------------------------------
# A monument waits undedicated while it is built, as long as we like
# (gm_state_awaits_dedication); this opens its dedication ceremony
# (monument_events.2), whose answer finishes it at the levels built. Not while
# any of its levels are under construction (owner: pausing is fine, building
# in parallel is not): dedicate once they finish or are cancelled. Levels
# built after the dedication, a level queued while the ceremony asks among
# them, raise a new monument beside it. On the state view's monument card and
# on the rows of Awaiting Dedication (gm_dedicate_button), which shows it on
# our own monument while gm_state_disp_awaiting holds: its visibility never
# rests on the saved scope reaching is_shown (gui gotcha #22), and is_valid
# fails closed as this file's others do. The AI's ceremony asks by itself once
# nothing more is queued.
gm_dedicate_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			gm_state_awaits_dedication = yes
			custom_tooltip = {
				text = gm_sgui_dedicate_built_tt
				gm_state_site_level >= 1
			}
			custom_tooltip = {
				text = gm_sgui_dedicate_building_tt
				b:building_grand_monument_site ?= { is_under_construction = no }
			}
			custom_tooltip = {
				text = gm_sgui_dedicate_asking_tt
				NOT = { has_variable = gm_ceremony_pending }
			}
			custom_tooltip = {
				text = gm_sgui_dedicate_rebuilding_tt
				NOT = { has_variable = gm_restore_pm }
				NOT = { has_variable = gm_rebuilding }
			}
			# The monument standing here moves beside the new one: not while
			# its unveiling or naming runs (gm_state_finish_site).
			custom_tooltip = {
				text = gm_sgui_dedicate_naming_tt
				NOT = { has_variable = gm_unveil_pending }
				NOT = { has_variable = gm_name_choice_open }
				NOT = { has_variable = gm_being_named }
			}
		}
	}

	effect = {
		scope:gm_state = {
			custom_tooltip = {
				text = gm_dedicate_effect_tt
				gm_state_start_ceremony = yes
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

# Raise It (usability §1): the missing levels queued as government construction
# where gm_com_pick_build_state puts them, through monument_events.27 (the
# button's tooltip walks this effect without writing, and the queue sets a
# variable and reads it back). Greyed while one of our monuments is under
# construction, so a second press can't queue the same levels twice.

	scope = country

	is_shown = {
		has_variable = gm_com_open
	}

		}
	}

	ai_is_valid = {
		always = no
	}
}

# ---- The monument policy (v2 §5; policy spec 2026-10-23 § 3) ---------------------
# One button per policy in National Effects. Valid while it is not the current
# policy and no change has been made in the last 60 months (the first choice
# starts no wait). The national totals change at once; each monument's tourism
# at its state's next monthly pulse. Cultural Promotion is the default
# (STANDARD = yes): a country that never chose one reads as it.
gm_policy_open_sgui = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		custom_tooltip = {
			text = gm_policy_current_tt
			NOT = { gm_policy_is = { P = open STANDARD = no } }
		}
		custom_tooltip = {
			text = gm_policy_wait_tt
			gm_policy_can_change = yes
		}
	}

	effect = {
		custom_tooltip = {
			text = gm_policy_open_effect_tt
			gm_policy_set = { P = open }
		}
		hidden_effect = { gm_country_refresh = yes }
	}

	ai_is_valid = {
		always = no
	}
}

# ---- Typed names (v2 §3.5): the company carrier ---------------------------------
# A name typed in the base game's company-rename popup, on a placeholder company
# that exists only while the player names (company_gm_name_carrier):
#   gm_name_start_sgui     adds the company (and a slot for it) and marks the
#                          state; the row's naming line then opens the popup
#   gm_name_mark_old_sgui  stores the company's starting name, so the row's
#                          watcher can see when it changes
#   gm_name_use_sgui       stores the company's name as the state's gm_name
#                          (MakeScopeFlag in the .gui), then removes it
#   gm_name_cancel_sgui    removes it without storing anything
#   gm_clear_name_sgui     drops a typed name; the chosen name shows again
# One naming at a time: the carrier is one company type. Players only.
gm_name_start_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			custom_tooltip = {
				text = gm_sgui_rename_upheld_tt
				gm_state_status_fits = yes
			}
			custom_tooltip = {
				text = gm_sgui_landmark_name_tt
				gm_state_has_historical_name = no
			}
		}
		custom_tooltip = {
			text = gm_name_one_at_a_time_tt
			NOT = { has_variable = gm_name_carrier }
		}
	}

	effect = {
		custom_tooltip = {
			text = gm_name_start_tt
			scope:gm_state = { set_variable = gm_being_named }
			# Before add_company: the company type's potential reads it.
			set_variable = gm_naming_active
			# The sweep's clock (gm_name_sweep): game time runs while the box
			# is open, so a year, restarted when the naming line appears.
			set_variable = { name = gm_naming_open value = yes days = 365 }
			add_modifier = { name = gm_name_carrier_slot }
			add_company = company_type:company_gm_name_carrier
			capital = {
				state_region = { save_scope_as = gm_name_carrier_hq }
			}
			company:company_gm_name_carrier = {
				set_company_state_region = scope:gm_name_carrier_hq
			}
			set_variable = { name = gm_name_carrier value = company:company_gm_name_carrier }
		}
	}

	ai_is_valid = {
		always = no
	}
}

# The watcher can send this twice before the first has landed, so the effect
# repeats the validity check.
gm_name_use_sgui = {
	scope = country
	saved_scopes = { gm_state gm_name }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		exists = scope:gm_name
		scope:gm_state = {
			owner = root
			has_variable = gm_being_named
		}
	}

	effect = {
		custom_tooltip = {
			text = gm_name_use_tt
			if = {
				limit = {
					scope:gm_state = { has_variable = gm_being_named }
				}
				scope:gm_state = {
					set_variable = { name = gm_name value = scope:gm_name }
				}
				gm_name_carrier_close = yes
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}
