# gui/journal_entry_widgets/grand_monuments_widget.gui. The three row buttons
# pass the row's state as the saved scope gm_state; which action is chosen by
# which scripted GUI the button points at (the covert rows' shape). A State
# object in saved_scopes has no vanilla precedent (gui gotcha #22), so every
# is_valid fails closed: the scope exists, we own it, it is contested.
# monument_events.17 is the proven path to the same three choices.

gm_tear_down_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			custom_tooltip = {
				text = gm_sgui_contested_tt
				gm_state_is_contested = yes
			}
		}
	}

	effect = {
		scope:gm_state = { gm_state_choose_tear_down = yes }
		hidden_effect = { gm_country_refresh = yes }
	}

	ai_is_valid = {
		always = no
	}
}

# PROTOTYPE (v2 spec §3.4, test build 9): name a monument through the base
# game's army-rename popup, with an empty army as the carrier
# (gm_name_carrier_effects.txt has why).
#   gm_name_start_sgui     creates the army (gm_name_carrier_army on the
#                          country) and marks the monument's state
#   gm_name_mark_old_sgui  stores the army's starting name (saved scope
#                          gm_old_name) as its flag gm_name_old; the row sends
#                          it as the naming opens, then watches for a change
#   gm_name_use_sgui       gets the army's name as the saved scope gm_name
#                          (MakeScopeFlag in the .gui) and stores it as the
#                          monument state's gm_name
#   gm_name_cancel_sgui    ends the naming without storing anything
# The .gui disbands the army alongside use and cancel. One naming at a time.
gm_name_start_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
		}
		custom_tooltip = {
			text = gm_name_one_at_a_time_tt
			NOT = { has_variable = gm_name_carrier_army }
		}
	}

	effect = {
		gm_name_carrier_open = yes
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_mark_old_sgui = {
	scope = country
	saved_scopes = { gm_old_name }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_old_name
		exists = var:gm_name_carrier_army
	}

	effect = {
		if = {
			limit = {
				exists = scope:gm_old_name
				exists = var:gm_name_carrier_army
			}
			var:gm_name_carrier_army = {
				set_variable = {
					name = gm_name_old
					value = scope:gm_old_name
				}
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_use_sgui = {
	scope = country
	saved_scopes = { gm_state gm_name }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		exists = scope:gm_name
		scope:gm_state = {
			owner = root
			has_variable = gm_being_named
		}
	}

	# The guard repeats is_valid's: the row's watcher may send this twice
	# before the first has closed the naming.
	effect = {
		if = {
			limit = {
				scope:gm_state = { has_variable = gm_being_named }
			}
			scope:gm_state = {
				set_variable = {
					name = gm_name
					value = scope:gm_name
				}
			}
			gm_name_carrier_close = yes
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_cancel_sgui = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		has_variable = gm_name_carrier_army
	}

	effect = {
		gm_name_carrier_close = yes
	}

	ai_is_valid = {
		always = no
	}
}
