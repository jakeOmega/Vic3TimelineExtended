# gui/journal_entry_widgets/grand_monuments_widget.gui. The three row buttons
# pass the row's state as the saved scope gm_state; which action is chosen by
# which scripted GUI the button points at (the covert rows' shape). A State
# object in saved_scopes has no vanilla precedent (gui gotcha #22), so every
# is_valid fails closed: the scope exists, we own it, it is contested.
# monument_events.17 is the proven path to the same three choices.

gm_tear_down_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			custom_tooltip = {
				text = gm_sgui_contested_tt
				gm_state_is_contested = yes
			}
		}
	}

	effect = {
		scope:gm_state = { gm_state_choose_tear_down = yes }
		hidden_effect = { gm_country_refresh = yes }
	}

	ai_is_valid = {
		always = no
	}
}

# PROTOTYPE (v2 spec §3.4, test build 5): name a monument through the base
# game's state-rename popup. A mod editbox's text can't reach script
# (gui_modding_guide.md § "Typed text can't reach script"), but a name the
# popup saves can: State.GetForestryHubName returns it as a CString. Build 4
# proved the route with a placeholder company; this one types into a logging
# town instead, which every state that can hold a monument has.
#   gm_name_start_here_sgui   carrier = the monument's own state
#   gm_name_start_tonga_sgui  carrier = Tonga's state (test: can we rename a
#                             state we don't own?)
#   gm_name_use_sgui          gets the logging town's name as the saved scope
#                             gm_name (MakeScopeFlag in the .gui), stores it as
#                             the monument state's gm_name, closes the naming
#   gm_name_cancel_sgui       closes it without storing anything
# The carrier state carries gm_carries_name (value gm_is_name_carrier), which
# switches the popup (states_panel.gui) to show only its logging-town field.
# One naming at a time.
gm_name_start_here_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
		}
		custom_tooltip = {
			text = gm_name_one_at_a_time_tt
			NOT = { has_variable = gm_name_carrier_state }
		}
	}

	effect = {
		scope:gm_state = {
			set_variable = gm_being_named
			set_variable = gm_carries_name
		}
		set_variable = {
			name = gm_name_carrier_state
			value = scope:gm_state
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_start_tonga_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
		}
		custom_tooltip = {
			text = gm_name_one_at_a_time_tt
			NOT = { has_variable = gm_name_carrier_state }
		}
		custom_tooltip = {
			text = gm_name_no_tonga_tt
			s:STATE_TONGA = {
				any_scope_state = {
					NOT = { has_variable = gm_carries_name }
				}
			}
		}
	}

	effect = {
		scope:gm_state = {
			set_variable = gm_being_named
		}
		s:STATE_TONGA = {
			random_scope_state = {
				save_scope_as = gm_name_tonga
			}
		}
		scope:gm_name_tonga = {
			set_variable = gm_carries_name
		}
		set_variable = {
			name = gm_name_carrier_state
			value = scope:gm_name_tonga
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_use_sgui = {
	scope = country
	saved_scopes = { gm_state gm_name }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		exists = scope:gm_name
		scope:gm_state = {
			owner = root
			has_variable = gm_being_named
		}
	}

	effect = {
		scope:gm_state = {
			set_variable = {
				name = gm_name
				value = scope:gm_name
			}
		}
		gm_name_carrier_close = yes
	}

	ai_is_valid = {
		always = no
	}
}
