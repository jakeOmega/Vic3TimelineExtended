# gui/journal_entry_widgets/grand_monuments_widget.gui. The three row buttons
# pass the row's state as the saved scope gm_state; which action is chosen by
# which scripted GUI the button points at (the covert rows' shape). A State
# object in saved_scopes has no vanilla precedent (gui gotcha #22), so every
# is_valid fails closed: the scope exists, we own it, it is contested.
# monument_events.17 is the proven path to the same three choices.

gm_tear_down_sgui = {
	scope = country
	saved_scopes = { gm_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		scope:gm_state = {
			owner = root
			custom_tooltip = {
				text = gm_sgui_contested_tt
				gm_state_is_contested = yes
			}
		}
	}

	effect = {
		scope:gm_state = { gm_state_choose_tear_down = yes }
		hidden_effect = { gm_country_refresh = yes }
	}

	ai_is_valid = {
		always = no
	}
}

# PROTOTYPE (v2 spec §3.4, test build 8): name a monument by typing over a
# state's name in the base game's state-rename popup (gm_name_carrier_effects
# has why, and how the name is put back). The .gui picks the carrier, because
# only it can compare names: the capital while it has its map name, else the
# monument's state, whose custom name (if any) the revert loses (the red
# tooltip).
#   gm_name_start_sgui         marks the naming; stores the carrier's name at
#                              the start (saved scope gm_old_name) as the flag
#                              gm_name_old, which the row watches
#   gm_name_use_sgui           gets the carrier's name as the saved scope
#                              gm_name (MakeScopeFlag in the .gui), stores it as
#                              the monument state's gm_name, hands the carrier
#                              to the popup to revert. The row runs it as soon
#                              as the carrier's name differs from gm_name_old,
#                              or by hand.
#   gm_name_cancel_sgui        ends the naming from the row
#   gm_name_popup_closed_sgui  the same from the popup's close button, when the
#                              popup's state is the carrier
#   gm_name_revert_done_sgui   the popup has confirmed the empty name: drop
#                              the carrier's flags
# One naming at a time.
gm_name_start_sgui = {
	scope = country
	saved_scopes = { gm_state gm_carrier gm_old_name }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		exists = scope:gm_carrier
		exists = scope:gm_old_name
		scope:gm_state = {
			owner = root
		}
		scope:gm_carrier = {
			owner = root
			NOT = { has_variable = gm_carries_name }
		}
		custom_tooltip = {
			text = gm_name_one_at_a_time_tt
			NOT = { has_variable = gm_name_carrier_state }
		}
	}

	effect = {
		gm_name_carrier_open = yes
		scope:gm_carrier = {
			set_variable = {
				name = gm_name_old
				value = scope:gm_old_name
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_use_sgui = {
	scope = country
	saved_scopes = { gm_state gm_name }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_state
		exists = scope:gm_name
		scope:gm_state = {
			owner = root
			has_variable = gm_being_named
		}
	}

	# The guard repeats is_valid's: the row's watcher may send this twice
	# before the first has closed the naming.
	effect = {
		if = {
			limit = {
				scope:gm_state = { has_variable = gm_being_named }
			}
			scope:gm_state = {
				set_variable = {
					name = gm_name
					value = scope:gm_name
				}
			}
			gm_name_carrier_hand_over = yes
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_cancel_sgui = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		has_variable = gm_name_carrier_state
	}

	effect = {

	}

	ai_is_valid = {
		always = no
	}
}

# Every close of the state-rename popup by its close button sends this
# (states_panel.gui), so it acts only when the popup's state is a carrier.
gm_name_popup_closed_sgui = {
	scope = country
	saved_scopes = { gm_popup_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_popup_state
		scope:gm_popup_state = {
			has_variable = gm_carries_name
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:gm_popup_state
				scope:gm_popup_state = { has_variable = gm_carries_name }
			}
			gm_name_carrier_close = yes
			scope:gm_popup_state = {
				gm_name_carrier_clear_state = yes
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}

gm_name_revert_done_sgui = {
	scope = country
	saved_scopes = { gm_popup_state }

	is_shown = {
		always = yes
	}

	is_valid = {
		exists = scope:gm_popup_state
	}

	effect = {
		if = {
			limit = { exists = scope:gm_popup_state }
			scope:gm_popup_state = {
				gm_name_carrier_clear_state = yes
			}
		}
	}

	ai_is_valid = {
		always = no
	}
}
