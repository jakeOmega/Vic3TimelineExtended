# ============================================================================
# MONETARY POLICY — SCRIPTED GUIS FOR THE BANKING DASHBOARD'S MONETARY BLOCK
# ============================================================================
# Read by gui/journal_entry_widgets/banking_dashboard_widget.gui. Spec:
# docs/systems/monetary_policy_design.md §3 (what the block shows), §5.2 and §6
# (who may touch what), §16.4 (the widget shape).
#
# These sit beside common/scripted_guis/banking_dashboard_scripted_gui.txt
# rather than inside it because the monetary system keeps its own file per layer
# — te_monetary_effects.txt, te_monetary_triggers.txt,
# te_monetary_script_values.txt — and this is that system's GUI layer.
#
# THE AI NEVER USES ANY OF THESE (`ai_is_valid = { always = no }`). The AI's
# monetary behaviour is not a button at all: te_monetary_update_target simply
# runs the mandate formula for `is_player = no` every month (§6, a documented
# exception recorded in that effect's header).
#
# TWO KINDS OF HANDLER, and the split is gotcha 22 of
# docs/guides/gui_modding_guide.md.
#
# 1. ONE INTERACTIVE HANDLER — banking_mon_control_sgui, parameterised by an
#    `op` saved scope. `is_valid`, `Execute` and the two tooltips are the places
#    a saved scope is PROVEN to arrive (Strategic Reserve, cultural hegemony),
#    so all the op branching lives there. Every branch delegates to a shared
#    trigger in te_monetary_triggers.txt and a shared effect in
#    te_monetary_effects.txt, so the tooltip and the click can never disagree.
#
# 2. SIX SCOPE-FREE DISPLAY HANDLERS — they answer "which control applies here"
#    with `is_shown` alone. Whether a saved scope reaches `is_shown` is thinly
#    evidenced, and the failure direction here is the bad one: if the scope went
#    missing, an op-branching `is_shown` would hide the whole monetary block and
#    leave the player unable to set a rate at all. All six carry
#    `is_valid = { always = no }` and an empty effect, so neither a player nor
#    the AI can execute them.
#
# `is_shown` on the interactive handler is scope-free for the same reason, and
# carries `is_player = yes`: the block only ever renders in the owner's own
# journal-entry panel, and putting the test here rather than in `is_valid` keeps
# a line that is always ticked out of the player's tooltip.
#
# ---- banking_mon_control_sgui op table -------------------------------------
# (keep in step with the .gui header and docs/systems/monetary_policy_design.md)
#   0   policy rate target   step down one point
#   1   policy rate target   step up one point
#   2   delegation           hand the dial to the central bank
#   3   delegation           take the dial back
#   4   mandate              price stability
#   5   mandate              growth
#   6   mandate              peg defence (gold standard only)
#   7   monetisation level   step down one level
#   8   monetisation level   step up one level
# ============================================================================

banking_mon_control_sgui = {
	scope = country
	saved_scopes = { op }

	is_shown = {
		has_journal_entry = je_banking_cycle
		is_player = yes
	}

	ai_is_valid = {
		always = no
	}

	# Every clause is a `custom_tooltip` phrased as a CONDITION, not a complaint:
	# IsValidTooltip renders each one with a tick when it holds and a cross when
	# it does not, so "Your country sets its own policy rate" reads correctly in
	# both states while "You cannot set a rate" would not.
	#
	# `exists = scope:op` guards the first branch so a missing scope falls
	# through to `trigger_else = { always = no }` — every control greys out
	# rather than acting on the wrong op.
	is_valid = {
		# --- 0 / 1: the target stepper (§5.2) ---
		# A dial needs a national bank and a discretionary regime; independence
		# binds the target to the mandate; delegation does the same by choice.
		trigger_if = {
			# 0 a whole point, 9 a tenth (ctrl-click), 11 to the floor (shift-click).
			limit = {
				exists = scope:op
				OR = {
					scope:op = 0
					scope:op = 9
					scope:op = 11
				}
			}
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_not_cbi
				te_mon_is_cbi = no
			}
			custom_tooltip = {
				text = banking_mon_tt_not_delegated
				te_mon_is_delegated = no
			}
			# The bound is te_mon_target_min either way; only the reason for it
			# differs. While §12.3's Defend option runs, the floor is the world
			# rate plus four, which is worth naming — unlike the digital floor,
			# the player cannot read it off the law that imposes it. Commodity
			# money's floor (world rate - 1) is the second such case and gets
			# the same treatment for the same reason: no law tooltip can state a
			# bound that moves with the world rate. (The phase-1 gold band that
			# used to be named here went when gold flows arrived: on gold the
			# dial is now free, and the reserves pay.)
			trigger_if = {
				limit = {
					te_mon_peg_is_defending = yes
					te_mon_is_on_gold = yes
				}
				custom_tooltip = {
					text = banking_mon_tt_peg_defend_floor
					te_mon_target_can_step_down = yes
				}
			}
			trigger_else_if = {
				limit = { te_mon_has_narrow_dial = yes }
				custom_tooltip = {
					text = banking_mon_tt_commodity_floor
					te_mon_target_can_step_down = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = banking_mon_tt_above_floor
					te_mon_target_can_step_down = yes
				}
			}
		}
		trigger_else_if = {
			# 1 a whole point, 10 a tenth, 12 to the ceiling.
			limit = {
				OR = {
					scope:op = 1
					scope:op = 10
					scope:op = 12
				}
			}
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_not_cbi
				te_mon_is_cbi = no
			}
			custom_tooltip = {
				text = banking_mon_tt_not_delegated
				te_mon_is_delegated = no
			}
			# Same split as the floor above: "the highest rate your monetary law
			# allows" is a true but useless sentence when the ceiling is world
			# rate + 1 and moves every month, so the narrow dial names its own.
			trigger_if = {
				limit = { te_mon_has_narrow_dial = yes }
				custom_tooltip = {
					text = banking_mon_tt_commodity_ceiling
					te_mon_target_can_step_up = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = banking_mon_tt_below_ceiling
					te_mon_target_can_step_up = yes
				}
			}
		}

		# --- 2 / 3: delegation (§6) ---
		# Free and reversible for any dial country that is not independent. A
		# country under central bank independence is permanently mandate-bound,
		# which is the whole point of the law, so the toggle is not offered.
		trigger_else_if = {
			limit = { scope:op = 2 }
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_not_cbi
				te_mon_is_cbi = no
			}
			custom_tooltip = {
				text = banking_mon_tt_not_yet_delegated
				te_mon_is_delegated = no
			}
		}
		trigger_else_if = {
			limit = { scope:op = 3 }
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_not_cbi
				te_mon_is_cbi = no
			}
			custom_tooltip = {
				text = banking_mon_tt_currently_delegated
				te_mon_is_delegated = yes
			}
		}

		# --- 4 / 5 / 6: the mandate (§6) ---
		# Pickable while the mandate actually drives the dial — delegated, or
		# bound by independence. The mandate already in force is deliberately
		# INVALID: greying it out is the "you are here" marker, and it saves a
		# parallel highlight path.
		trigger_else_if = {
			limit = { scope:op = 4 }
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_mandate_binds
				te_mon_mandate_binds = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_mandate_other
				OR = {
					NOT = { has_variable = te_mon_mandate }
					NOT = { var:te_mon_mandate = 1 }
				}
			}
		}
		trigger_else_if = {
			limit = { scope:op = 5 }
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_mandate_binds
				te_mon_mandate_binds = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_mandate_other
				OR = {
					NOT = { has_variable = te_mon_mandate }
					NOT = { var:te_mon_mandate = 2 }
				}
			}
		}
		trigger_else_if = {
			limit = { scope:op = 6 }
			custom_tooltip = {
				text = banking_mon_tt_needs_dial
				te_mon_has_dial = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_gold_only
				te_mon_is_on_gold = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_mandate_binds
				te_mon_mandate_binds = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_mandate_other
				OR = {
					NOT = { has_variable = te_mon_mandate }
					NOT = { var:te_mon_mandate = 3 }
				}
			}
		}

		# --- 7 / 8: monetising the deficit (§11) ---
		# TWO LINES, not the dial's three. Eligibility here has five separate
		# causes (no national bank, a currency the state cannot create,
		# central bank independence, a command economy, dollarisation) and
		# listing all five as conditions would grey a button behind a wall of
		# ticks. banking_mon_tt_can_monetise is ONE line that names the binding
		# cause and nothing else — the answer te_mon_no_dial_reason gives for
		# the dial, through the same customizable_localization machinery
		# (te_mon_monetise_condition,
		# common/customizable_localization/banking_dash_custom_loc.txt).
		trigger_else_if = {
			limit = { scope:op = 7 }
			custom_tooltip = {
				text = banking_mon_tt_can_monetise
				te_mon_can_monetise = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_monetise_above_floor
				te_mon_monetise_can_step_down = yes
			}
		}
		trigger_else_if = {
			limit = { scope:op = 8 }
			custom_tooltip = {
				text = banking_mon_tt_can_monetise
				te_mon_can_monetise = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_monetise_below_ceiling
				te_mon_monetise_can_step_up = yes
			}
		}
		# --- 13: recapitalise the bank (§0.5, the hybrid model) ---
		# Treasury -> vault, one way. One custom_tooltip per cause, phrased as
		# conditions like the rest.
		trigger_else_if = {
			limit = { scope:op = 13 }
			custom_tooltip = {
				text = banking_mon_tt_recap_has_flows
				te_mon_has_gold_flows = yes
			}
			custom_tooltip = {
				text = banking_mon_tt_recap_room
				te_mon_bank_recap_amount > 0
			}
			custom_tooltip = {
				text = banking_mon_tt_recap_cash
				gold_reserves >= te_mon_bank_recap_amount
			}
			te_mon_bank_can_recapitalise = yes
		}
		trigger_else = {
			always = no
		}
	}

	effect = {
		if = {
			limit = { exists = scope:op scope:op = 0 }
			te_mon_effect_target_step_down = yes
		}
		else_if = {
			limit = { scope:op = 1 }
			te_mon_effect_target_step_up = yes
		}
		else_if = {
			limit = { scope:op = 2 }
			te_mon_effect_delegate_on = yes
		}
		else_if = {
			limit = { scope:op = 3 }
			te_mon_effect_delegate_off = yes
		}
		else_if = {
			limit = { scope:op = 4 }
			te_mon_effect_set_mandate = { MANDATE = 1 TT = banking_mon_mandate_set_tt }
		}
		else_if = {
			limit = { scope:op = 5 }
			te_mon_effect_set_mandate = { MANDATE = 2 TT = banking_mon_mandate_set_tt }
		}
		else_if = {
			limit = { scope:op = 6 }
			# Peg defence gets its own line: its rule points outside the country,
			# at the world reference rate the dashboard now shows as a row.
			te_mon_effect_set_mandate = { MANDATE = 3 TT = banking_mon_mandate_peg_set_tt }
		}
		else_if = {
			limit = { scope:op = 7 }
			te_mon_effect_monetisation_step_down = yes
		}
		else_if = {
			limit = { scope:op = 8 }
			te_mon_effect_monetisation_step_up = yes
		}
		else_if = {
			limit = { scope:op = 9 }
			te_mon_effect_target_fine_down = yes
		}
		else_if = {
			limit = { scope:op = 10 }
			te_mon_effect_target_fine_up = yes
		}
		else_if = {
			limit = { scope:op = 11 }
			te_mon_effect_target_to_min = yes
		}
		else_if = {
			limit = { scope:op = 12 }
			te_mon_effect_target_to_max = yes
		}
		else_if = {
			limit = { scope:op = 13 }
			te_mon_effect_recapitalise_bank = yes
		}
	}
}


# ---------------------------------------------------------------------------
# DISPLAY-ONLY HANDLERS — one scope-free question each
# ---------------------------------------------------------------------------

# Has this country's first monthly monetary update run? Every `var:` read in the
# monetary block hides behind this one question, because a .gui read of a
# missing variable logs "Failed to fetch variable" once per frame
# (gui_modding_guide.md gotcha 14) and je_banking_cycle is
# is_shown_when_inactive.
banking_mon_display_ready = {
	scope = country
	is_shown = { te_mon_display_ready = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Does this country set its own policy rate at all? Drives the whole
# interactive half of the block: a bankless, commodity-money, crypto or
# command-economy country reads the block but cannot touch it, and gets a
# cause-specific line saying why (te_mon_no_dial_reason).
banking_mon_has_dial = {
	scope = country
	is_shown = { te_mon_has_dial = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Is the delegation toggle offered at all? Central bank independence is
# permanently mandate-bound (§5.2), so it gets the explanation instead of the
# button.
banking_mon_can_delegate = {
	scope = country
	is_shown = {
		te_mon_has_dial = yes
		te_mon_is_cbi = no
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Which half of the delegate / take-back pair to draw.
banking_mon_is_delegated = {
	scope = country
	is_shown = { te_mon_is_delegated = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Peg defence is a gold-standard mandate only (§6), so the third mandate button
# only exists on gold.
#
# te_mon_is_on_gold, not the law: a country that has suspended convertibility
# (§12.3) has no peg to defend, and step 2 would bounce the mandate straight
# back to price stability.
banking_mon_is_gold = {
	scope = country
	is_shown = { te_mon_is_on_gold = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Is the policy rate allowed to go below zero? Only a digital currency abolishes
# the cash that makes a negative rate unenforceable
# (country_policy_rate_floor_add = -0.03 on law_digital_currency), so only a
# digital country needs a history chart whose axis reaches under the zero line.
# Read by gui/journal_entry_widgets/banking_history_widget.gui, which draws one
# of two policy-rate charts on the answer.
#
# The test is on the law rather than on the modifier because a .gui cannot read
# `modifier:` — and inside this system the two are the same question.
banking_mon_is_digital = {
	scope = country
	is_shown = { has_law_or_variant = law_type:law_digital_currency }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Phase 3 (§12): does the dashboard show the gold rows — World Rate Gap, Gold
# Flow, Peg Confidence? A gold-standard LAW with a dial, suspended or not: a
# suspended country still wants to see its frozen hot money and the clock, and
# the rows' own customizable localization says "suspended" where that is the
# answer. A gold country with no national bank has no gap and no flow (§12.2),
# so it has no rows.
banking_mon_shows_gold = {
	scope = country
	is_shown = {
		has_law_or_variant = law_type:law_gold_standard
		te_mon_has_dial = yes
		has_variable = te_gold_flow_gap
		has_variable = te_gold_flow
		has_variable = te_gold_hot_money
		has_variable = te_bank_gold
		has_variable = te_gold_carry
		has_variable = te_peg_confidence
		has_variable = te_mon_world_rate_now
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Is the World Rate row ready to print? te_mon_world_rate_now is a phase-3
# variable, so a save from before it has none until its first pulse.
banking_mon_world_rate_ready = {
	scope = country
	is_shown = { has_variable = te_mon_world_rate_now }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}


# ----------------------------------------------------------------------------
# Phase 5 (§15A): international monetary arrangements
# ----------------------------------------------------------------------------
# Does the dashboard show the arrangements block? Any role at all — anchored, an
# anchor, either side of a swap line or a guarantee. te_mon_arr_scan is exactly
# that flag (the discovery sets it), and the has_variable list is the block's
# displayed contract, the banking_mon_shows_gold pattern.
banking_mon_shows_arrangements = {
	scope = country
	is_shown = {
		has_variable = te_mon_arr_scan
		var:te_mon_arr_scan = 1
		has_variable = te_mon_anchor_kind
		has_variable = te_mon_anchor_rate
		has_variable = te_mon_arr_recipient_pts
		has_variable = te_mon_arr_provider_pts
		has_variable = te_mon_arr_standing_pts
		has_variable = te_mon_peg_parity_offset
		OR = {
			var:te_mon_anchor_kind > 0
			var:te_mon_arr_recipient_pts > 0
			var:te_mon_arr_provider_pts > 0
			var:te_mon_arr_standing_pts > 0
		}
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# ...and the treaty pegger's Peg Confidence row and meter?
banking_mon_shows_anchor_peg = {
	scope = country
	is_shown = {
		te_mon_has_anchor_peg = yes
		has_variable = te_peg_confidence
		has_variable = te_mon_overvaluation
	}
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}
