# ============================================================================
# UN AUTHORITY — SCRIPTED GUIS FOR THE "WHY AUTHORITY IS MOVING" WIDGET
# ============================================================================
# Tooltip builders only, in the chamber's pattern (un_chamber_sguis.txt, part
# 1): gui/journal_entry_widgets/un_authority_widget.gui calls
# `[GetScriptedGui('<key>').ExecuteTooltip(GuiScope.SetRoot(
# JournalEntry.GetCountry.MakeScope).End)]`, which renders the effect without
# running it. The effects are in
# common/scripted_effects/un_authority_display_effects.txt and write nothing.
#
# Split into four so that each widget textbox gets lines in the order it
# expects (root-level lines print before nested ones), and so the collapsed
# help text costs nothing while it is closed.
# ============================================================================

# Whether the pillars exist yet: the gate for the widget's pillar table,
# which reads the snapshots un_authority_monthly_update writes. Never executed.
un_authority_ready_sgui = {
	scope = country

	is_shown = {
		has_global_variable = un_founded
		has_global_variable = un_authority_target
	}
	ai_is_valid = { always = no }
}

# The empty states (not founded; no monthly update yet). Prints nothing once
# the pillar table can be drawn.
un_authority_summary_sgui = {
	scope = country

	effect = {
		un_authority_summary_block = yes
	}
}

# Champions, underminers and great powers outside the UN, one line each.
un_authority_powers_sgui = {
	scope = country

	effect = {
		un_authority_powers_block = yes
	}
}

# The twelve most recent ledger entries, newest first.
un_authority_log_sgui = {
	scope = country

	effect = {
		un_authority_log_block = yes
	}
}

# How the model works. Static.
un_authority_help_sgui = {
	scope = country

	effect = {
		un_authority_help_block = yes
	}
}
