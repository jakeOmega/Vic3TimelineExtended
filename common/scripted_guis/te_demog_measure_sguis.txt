# ============================================================================
# DEMOGRAPHICS — the Family Policy measures' buttons (phase 2 step 3a)
# (docs/superpowers/archive/specs/2026-10-12-demographics-phase2-consequences-design.md §3.3, §3.7; the panel:
# te_demog_sec_family in gui/te_demographics_widgets.gui)
# ============================================================================
# One scripted GUI a measure, rooted at the player's country (GuiScope.SetRoot( GetPlayer.MakeScope ).End):
#   is_shown  the law in force allows the measure: the panel draws its row with this button; while it doesn't, the
#             measure is a dimmed row under "Other Laws", naming the laws that allow it
#   is_valid  greyed with the failing reason only (gui_style_guide.md rule 6): the rule isn't Full ("Family policy
#             measures act only under the Full Demographics rule"), the law, the five-year term with the year it ends,
#             the Crash Campaign's run and bar, the Rural Exemption's need for Birth Limit, and for a measure with an
#             authority upkeep that upkeep free to start (Birth Limit's further 100 to tighten), as a decree's cost.
#             Each reason is a trigger_if that shows only when it fails, so a pressable button lists none
#   effect    one press (te_demog_ms_press; Birth Limit's cycles off, two, one): what it does in words, the switching
#             hidden (its modifiers show on the row's name)
# The AI never presses them (ai_is_valid); its own yearly pulse is te_demog_ms_ai_choose.
# ============================================================================

# The rule is Full: the panel says once, above the rows, that the measures act only under it (rule 6).
te_demog_ms_full_sgui = {
	scope = country
	is_shown = { te_demog_effects_run = yes }
	ai_is_valid = { always = no }
	is_valid = { always = no }
	effect = { }
}

# Child Allowance
te_demog_ms_child_allowance_sgui = {
	scope = country
	is_shown = {
		te_demog_ms_allowed_child_allowance = yes
	}
	is_valid = {
		trigger_if = {
			limit = { NOT = { te_demog_effects_run = yes } }
			custom_tooltip = {
				text = te_demog_ms_rule_tt
				te_demog_effects_run = yes
			}
		}
		trigger_if = {
			limit = { NOT = { te_demog_ms_allowed_child_allowance = yes } }
			custom_tooltip = {
				text = te_demog_ms_child_allowance_law_tt
				te_demog_ms_allowed_child_allowance = yes
			}
		}
		trigger_if = {
			limit = {
				has_variable = te_dg_ms_child_allowance
				NOT = { te_demog_ms_term_run = { M = child_allowance } }
			}
			custom_tooltip = {
				text = te_demog_ms_child_allowance_term_tt
				te_demog_ms_term_run = { M = child_allowance }
			}
		}
	}
	effect = {
		if = {
			limit = { NOT = { has_variable = te_dg_ms_child_allowance } }
			custom_tooltip = te_demog_ms_start_tt
		}
		else = {
			custom_tooltip = te_demog_ms_end_tt
		}
		hidden_effect = {
			te_demog_ms_press = { M = child_allowance }
		}
	}
	ai_is_valid = { always = no }
}

# Birth Limit
te_demog_ms_birth_limit_sgui = {
	scope = country
	is_shown = {
		te_demog_ms_allowed_birth_limit = yes
	}
	is_valid = {
		trigger_if = {
			limit = { NOT = { te_demog_effects_run = yes } }
			custom_tooltip = {
				text = te_demog_ms_rule_tt
				te_demog_effects_run = yes
			}
		}
		trigger_if = {
			limit = { NOT = { te_demog_ms_allowed_birth_limit = yes } }
			custom_tooltip = {
				text = te_demog_ms_birth_limit_law_tt
				te_demog_ms_allowed_birth_limit = yes
			}
		}
		# the term binds only the step back to off, from one child
		trigger_if = {
			limit = {
				has_variable = te_dg_ms_birth_limit_one
				NOT = { te_demog_ms_term_run = { M = birth_limit } }
			}
			custom_tooltip = {
				text = te_demog_ms_birth_limit_term_tt
				te_demog_ms_term_run = { M = birth_limit }
			}
		}
		# the upkeep free, to start (as a decree's cost; the rebalance's review, 2026-10-16), and the one child's further
		# upkeep, to tighten
		trigger_if = {
			limit = {
				NOT = { has_variable = te_dg_ms_birth_limit }
				authority < te_demog_ms_authority_birth_limit
			}
			custom_tooltip = {
				text = te_demog_ms_birth_limit_authority_tt
				authority >= te_demog_ms_authority_birth_limit
			}
		}
		trigger_if = {
			limit = {
				has_variable = te_dg_ms_birth_limit
				NOT = { has_variable = te_dg_ms_birth_limit_one }
				authority < te_demog_ms_authority_birth_limit_tighten
			}
			custom_tooltip = {
				text = te_demog_ms_birth_limit_tighten_authority_tt
				authority >= te_demog_ms_authority_birth_limit_tighten
			}
		}
	}
	effect = {
		if = {
			limit = { NOT = { has_variable = te_dg_ms_birth_limit } }
			custom_tooltip = te_demog_ms_birth_limit_start_tt
		}
		else_if = {
			limit = { NOT = { has_variable = te_dg_ms_birth_limit_one } }
			custom_tooltip = te_demog_ms_birth_limit_tighten_tt
		}
		else = {
			custom_tooltip = te_demog_ms_birth_limit_end_tt
		}
		hidden_effect = {
			te_demog_ms_press_birth_limit = yes
		}
	}
	ai_is_valid = { always = no }
}
