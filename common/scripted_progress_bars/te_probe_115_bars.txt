# What it answers: C5 of docs/audits/vanilla_1_15_prep.md § 2, the scope a scripted progress bar's script-value
# min_value runs in. 1.15's readme says only "evaluated in the journal entry's scope whenever the bar is clamped or
# shown"; whether root there is the entry or its owner is open. The minimum is 40 when the root holds the probe
# country's variable (the owner), 20 when journal_entry_age answers (the entry), 5 when neither does. The probe entry
# (common/journal_entries/te_probe_115_probes.txt) carries the bar; steps 2 and 3 read where it was clamped
# (common/scripted_effects/te_probe_115_c5_effects.txt). The probe entry is hidden, so no player sees the bar.
# A load-time scope error on one of the two triggers below names the scope the engine expects there.
#
# Delete once answered (see docs/testing/vanilla-1-15-playtest.md).

te_probe_115_c5_bar = {
	name = "te_probe_115_c5_bar_name"
	desc = "te_probe_115_c5_bar_text"

	default = yes

	start_value = 0
	min_value = {
		value = 5
		if = {
			limit = { has_variable = te_probe_115_owner }
			value = 40
		}
		else_if = {
			limit = { journal_entry_age >= 0 }
			value = 20
		}
	}
	max_value = 100
}
