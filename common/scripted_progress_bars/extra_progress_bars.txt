# A bar between two poles (1.15, #1434): collapse at 0, where the entry fails,
# and integration at 100, where a five-year hold completes it. Green fill
# (stability held) over the red background (what is lost), with the sliding
# marker, as vanilla's Great Game bar is drawn; no end icons, as Banking's.
# While the entry is active the panel does not draw this look: the overview
# draws its own five-layer bar from the same value (colonial_empire_widget.gui,
# te_ce_ov_bar_row) and the bars-on-top marker hides the stock bar block. The
# look shows wherever the stock block draws the entry's bars without it.
colonial_stability_bar = {
	name = "colonial_stability_bar_name"
	desc = "colonial_stability_bar_desc"
	second_desc = "colonial_stability_bar_second"

	progress_texture = "gfx/interface/progressbar/progressbar_green.dds"
	background_texture = "gfx/interface/progressbar/progressbar_bad.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_green_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_green_glow_line.dds"
	marker = yes

	start_value = 50
	min_value = 0
	max_value = 100

	# ========================================================================
	# Every term of this formula lives as a named LEAF SCRIPT VALUE in
	# common/script_values/colonial_empire_values.txt
	# (`colonial_stability_term_*`, country scope). The bar consumes the leaves
	# one-for-one, each keeping its own `desc` key so the player's hover
	# breakdown still shows all 22 terms individually; the block ends with the
	# monthly cap's clamp (max / min). The journal-entry widget
	# and the history charts consume the group sums (`colonial_stability_drift_*`)
	# and `colonial_stability_drift_total`, which are built from the SAME leaves.
	#
	# DO NOT put a number in this block. Retune the leaf instead, and the bar,
	# the widget, the charts and every tooltip move together. Adding a term =
	# one leaf + one group entry + one `add` line here with a `desc` key.
	#
	# The gate that used to wrap each `add` in an `if = { limit = { owner = … } }`
	# now lives inside the leaf, which returns 0 when the gate is false. The
	# adds below are therefore unconditional; a term that does not apply
	# contributes exactly 0, as it did before.
	#
	# Contributions that are NOT listed here reach the bar through
	# `country_colonial_stability_drift_add` (laws, the decolonization/
	# globalization/knowledge_economy techs, and the timed
	# colonial_stability_positive_event / _negative_event modifiers) and surface
	# in the `colonial_law_aggregate_tt` line's own breakdown.
	# ========================================================================
	monthly_progress = {
		# Soft natural drift; the governance + IG terms below dominate
		add = {
			desc = "colonial_base_drift_tt"
			value = owner.colonial_stability_term_base
		}

		# Laws, techs and timed event modifiers, aggregated via
		# country_colonial_stability_drift_add. Each contributing law declares
		# its colonial-stability impact in its own modifier block, so the law
		# tooltip shows the contribution directly. See:
		#   common/laws/colonial_empire_law_injections.txt
		#   common/modifier_type_definitions/colonial_empire_modifier_types.txt
		add = {
			desc = "colonial_law_aggregate_tt"
			value = owner.colonial_stability_term_laws
		}

		# IG composition — the empire's domestic backers
		add = {
			desc = "colonial_ig_landowners_strong_tt"
			value = owner.colonial_stability_term_ig_landowners
		}
		add = {
			desc = "colonial_ig_armed_forces_strong_tt"
			value = owner.colonial_stability_term_ig_armed_forces
		}
		add = {
			desc = "colonial_ig_intelligentsia_strong_tt"
			value = owner.colonial_stability_term_ig_intelligentsia
		}
		add = {
			desc = "colonial_ig_unions_strong_tt"
			value = owner.colonial_stability_term_ig_unions
		}

		# Imperial overreach — penalty when colonial holdings exceed ~1.5x
		# metropole size. Belgium with Congo: no penalty. UK with global
		# empire: heavy penalty. Scaled by metropole size, not raw count.
		add = {
			desc = "colonial_overreach_tt"
			value = owner.colonial_stability_term_overreach
		}

		# Rank standing
		add = {
			desc = "colonial_gp_rank_bonus_tt"
			value = owner.colonial_stability_term_gp_rank
		}
		add = {
			desc = "colonial_non_gp_penalty_tt"
			value = owner.colonial_stability_term_non_gp
		}

		# GP condemnation pressure (linear + tier escalations), and GP support
		add = {
			desc = "colonial_gp_condemnation_tt"
			value = owner.colonial_stability_term_gp_condemnation
		}
		add = {
			desc = "colonial_gp_high_pressure_tt"
			value = owner.colonial_stability_term_gp_high_pressure
		}
		add = {
			desc = "colonial_gp_extreme_pressure_tt"
			value = owner.colonial_stability_term_gp_extreme_pressure
		}
		add = {
			desc = "colonial_gp_support_tt"
			value = owner.colonial_stability_term_gp_support
		}

		# Programme effectiveness. Cruelty boosts Garrison; nice laws cripple
		# it. Investment is the benevolent-empire primary tool; cruelty
		# undermines it. Assimilation needs the matching minority law to be
		# effective. Each leaf reads the matching
		# country_colonial_*_effectiveness_add aggregate while its programme is
		# active; baselines live in `base_values` and law contributions in each
		# affected law's modifier block, so the aggregate's
		# GetValueWithBreakdownFor tooltip shows base + each law in one place.
		add = {
			desc = "colonial_garrison_aggregate_tt"
			value = owner.colonial_stability_term_garrison
		}
		add = {
			desc = "colonial_assim_aggregate_tt"
			value = owner.colonial_stability_term_assim
		}
		add = {
			desc = "colonial_invest_aggregate_tt"
			value = owner.colonial_stability_term_invest
		}

		# Acceptance: per-state penalties / bonuses
		add = {
			desc = "colonial_low_acceptance_tt"
			value = owner.colonial_stability_term_low_acceptance
		}
		add = {
			desc = "colonial_high_acceptance_tt"
			value = owner.colonial_stability_term_high_acceptance
		}

		# Domestic disruption: war, revolution, and radicalization at home.
		# country_turmoil is a 0-1 fraction; the leaf's threshold gate keeps
		# trace turmoil out. Scaled: turmoil 0.25 -> -0.5/mo, 0.5 -> -1.0/mo.
		add = {
			desc = "colonial_war_penalty_tt"
			value = owner.colonial_stability_term_war
		}
		add = {
			desc = "colonial_independence_war_penalty_tt"
			value = owner.colonial_stability_term_independence_war
		}
		add = {
			desc = "colonial_revolution_penalty_tt"
			value = owner.colonial_stability_term_revolution
		}
		add = {
			desc = "colonial_turmoil_penalty_tt"
			value = owner.colonial_stability_term_turmoil
		}

		# Monthly cap: the net change is clamped to +/- colonial_stability_drift_cap.
		# MUST stay last, after every add, so it clamps the sum. A clamp, as
		# vanilla's 00_rlit_progress_bars.txt ends its bars, not a 23rd leaf: that
		# leaf (capped minus uncapped total) evaluated the 22 leaves twice more,
		# 641 ms of the bar's 1,035 ms in 30 days of a 1931 game
		# (docs/audits/late_game_performance.md). colonial_stability_drift_total
		# clamps with the same two values. See the MONTHLY CAP section of
		# colonial_empire_values.txt.
		max = colonial_stability_drift_cap
		min = colonial_stability_drift_floor
	}
}

# ============================================================================
# CIVIL RIGHTS MOVEMENT SUPPORT BAR
# Drives je_civil_rights' progress 0-100. Drift sources:
#  - base decay (movement fades without effort)
#  - tech tier bonuses (social_justice_movements accelerates)
#  - current minority law (discriminatory laws sap; progressive laws boost)
#  - per low-acceptance state (grievances feed recruitment)
#  - radical fraction (radicalization feeds the movement)
#  - active button policy modifiers (the dominant push, see civil_rights_buttons.txt)
#  - low-bar floor relief (prevents permanent stuck-at-0 before the failure outcome can resolve)
# ============================================================================
civil_rights_support_bar = {
	name = "civil_rights_support_bar_name"
	desc = "civil_rights_support_bar_desc"

	# A tug of war between collapse at 0 and victory at 100 (#1434): green
	# support over a red background, with 1.15's sliding marker, in the
	# fields that replaced the ready-made look (default_green before).
	progress_texture = "gfx/interface/progressbar/progressbar_green.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_green_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_green_glow_line.dds"
	background_texture = "gfx/interface/progressbar/progressbar_bad.dds"
	marker = yes

	start_value = 30
	min_value = 0
	max_value = 100

	# Every conditional factor is a progress_condition (1.15, #1441): it counts
	# like an if, and the journal panel's Advance and Retreat lists show it,
	# ticked while its limit holds and unticked otherwise, so the player sees
	# what is moving the bar and what is left. No format = inside one: it
	# would override the unticked styling.
	monthly_progress = {
		# Civil rights movements are generational: the real-world arc from
		# Brown v. Board to the Fair Housing Act ran ~14 years. The numbers
		# below are calibrated against ~70 points of bar progress (start = 30,
		# complete = 100): a maxed-out push run aims for ~5 years; a dedicated
		# but not all-in run ~7-10 years; a passive-bystander run leaves the
		# bar drifting near zero. Suppression options can still pull the bar
		# down rapidly — repression remains the faster path.

		# Base: movement fades when no one is pushing it
		add = {
			desc = "cr_base_drift_tt"
			value = -0.4
		}

		# Tech tier: social_justice_movements is the maturity gate
		progress_condition = {
			limit = { owner = { has_technology_researched = social_justice_movements } }
			add = {
				desc = "cr_social_justice_tech_tt"
				value = 0.4
			}
		}

		# Severe discriminatory law actively suppresses organizing
		progress_condition = {
			limit = { owner = { has_severe_discriminatory_minority_law = yes } }
			add = {
				desc = "cr_severe_discrim_law_tt"
				value = -0.5
			}
		}

		# Progressive minority law accelerates absorption
		progress_condition = {
			limit = {
				owner = {
					OR = {
						has_law = law_type:law_minority_rights_protection
						has_law = law_type:law_minority_rights_affirmative_action
					}
				}
			}
			add = {
				desc = "cr_progressive_law_tt"
				value = 0.3
			}
		}

		# Per low-acceptance state, capped via max so a 50-state empire doesn't auto-win
		progress_condition = {
			limit = { owner = { cr_low_acceptance_count > 0 } }
			add = {
				desc = "cr_low_acceptance_states_tt"
				value = owner.cr_low_acceptance_count
				multiply = 0.0125
				max = 0.5
			}
		}

		# Radicalization boost: high turmoil feeds recruits
		progress_condition = {
			limit = { owner = { radical_fraction = { value >= 0.10 } } }
			add = {
				desc = "cr_radicals_tier1_tt"
				value = 0.15
			}
		}
		progress_condition = {
			limit = { owner = { radical_fraction = { value >= 0.20 } } }
			add = {
				desc = "cr_radicals_tier2_tt"
				value = 0.15
			}
		}

		# Pro-movement policies (cumulative — stack all three at once for max push)
		progress_condition = {
			limit = { owner = { has_modifier = cr_grassroots_modifier } }
			add = {
				desc = "cr_grassroots_push_tt"
				value = 0.2
			}
		}
		progress_condition = {
			limit = { owner = { has_modifier = cr_federal_protection_modifier } }
			add = {
				desc = "cr_federal_push_tt"
				value = 0.3
			}
		}
		progress_condition = {
			limit = { owner = { has_modifier = cr_gradualist_modifier } }
			add = {
				desc = "cr_gradualist_push_tt"
				value = 0.1
			}
		}

		# Cooptation: short bump (handled by adding cr_cooptation_expired modifier
		# from the JE on_monthly_pulse once cr_cooptation_months >= 12)
		progress_condition = {
			limit = {
				owner = {
					has_modifier = cr_cooptation_modifier
					NOT = { has_modifier = cr_cooptation_expired }
				}
			}
			add = {
				desc = "cr_cooptation_push_tt"
				value = 0.15
			}
		}

		# Anti-movement policies (push the bar DOWN). Repression remains the
		# faster path — suppression and segregationist crackdowns substantially
		# outpace any single pro-movement modifier.
		progress_condition = {
			limit = { owner = { has_modifier = cr_suppression_modifier } }
			add = {
				desc = "cr_suppression_push_tt"
				value = -0.6
			}
		}
		progress_condition = {
			limit = { owner = { has_modifier = cr_segregationist_modifier } }
			add = {
				desc = "cr_segregationist_push_tt"
				value = -0.9
			}
		}

		# Floor relief: pause decay while bar is below 5 so the failure outcomes resolve
		# from sustained suppression, not from random decay drift
		progress_condition = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(civil_rights_support_bar)" < 5
				}
			}
			add = {
				desc = "cr_floor_relief_tt"
				value = 0.4
			}
		}
	}
}
