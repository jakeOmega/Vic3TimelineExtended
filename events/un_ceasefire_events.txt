namespace = un_ceasefire

# ============================================================================
# THE BINDING CEASEFIRE — NOTICES (redesign phase 7, §0.12 item 6)
# ============================================================================
# Notices only: every consequence is applied where it happens, in
# common/scripted_effects/un_order_effects.txt (un_vote.2's carried branch,
# and the order's month), and the options here only say what it was.
#   .1  the aggressor, when the resolution carries (an order, or the vetoed
#       call for a ceasefire)
#   .2  each member the aggressor is fighting, when an order is imposed
#   .3  the war ended inside the order and the aggressor gained nothing from
#       it: the aggressor, the victim and the proposer
#   .4  the order ran out with the war still on: the same three
#   .5  the war ended with the aggressor holding gains from the victim (its
#       former capital, its land, or the victim as its subject;
#       un_order_ceasefire_aggressor_gained): the aggressor, the victim if it
#       still stands, and the proposer
# An order that lapses as moot, or whose victim is gone but not into the
# aggressor's hands, ends with no notice (un_ceasefire_order_month).
# Saved scopes: un_cf_aggressor (the aggressor), un_cf_victim_scope and
# un_cf_proposer_scope where they exist; .3 also un_msn_new when a mission
# opened.
# ============================================================================

# Ordered to cease fire (or called on to). Fired on the aggressor.
un_ceasefire.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = {
		first_valid = {
			triggered_desc = {
				desc = un_ceasefire.1.t_order
				trigger = { has_variable = un_cf_order_months }
			}
			desc = un_ceasefire.1.t_call
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				desc = un_ceasefire.1.d_accepted
				trigger = {
					has_variable = un_cf_order_months
					has_variable = un_cf_order_accepted
				}
			}
			triggered_desc = {
				desc = un_ceasefire.1.d_order
				trigger = { has_variable = un_cf_order_months }
			}
			desc = un_ceasefire.1.d_call
		}
	}
	flavor = un_ceasefire.1.f

	duration = 3

	trigger = {
		has_global_variable = un_founded
	}

	option = {
		name = un_ceasefire.1.a
		default_option = yes
		if = {
			limit = { has_variable = un_cf_order_months }
			custom_tooltip = un_ceasefire_order_terms_tt
			if = {
				limit = { has_variable = un_cf_order_accepted }
				custom_tooltip = un_ceasefire_fight_on_accepted_tt
			}
			else = {
				custom_tooltip = un_ceasefire_fight_on_tt
			}
		}
		else = {
			custom_tooltip = un_ceasefire_call_done_tt
		}
	}
}
