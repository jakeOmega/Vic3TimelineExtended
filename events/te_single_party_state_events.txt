namespace = te_single_party_state

# Single-Party State's dominant party, for the mod's three parties (#1908).
#   .1  sent, as a popup, when a Single-Party State enactment starts with a
#       Green, Populist or Technocratic group in government, and monthly while
#       it runs (common/on_actions/te_single_party_state_on_actions.txt, which
#       says when, and how the event and vanilla's never both take an answer)
#
# Vanilla 1.15.2's single_party_state.8 asks the same question for its eleven
# parties; this event is its copy for the mod's three, and calls the same
# set_sps_dominant_party_effect, so the law's on_activate, the dominance
# amendments (common/amendments/te_sps_party_dominance.txt) and the parties'
# join_weight terms work as they do for vanilla's parties. Its last option
# leaves the choice to vanilla's event when a vanilla party governs as well,
# and a checkpoint that fires vanilla's event with a vanilla party to offer
# closes this one (te_sps_left_to_vanilla).

# ======================================================================
#  1 – The Party-State  (Single-Party State is being enacted)
# ======================================================================
te_single_party_state.1 = {
	type = country_event
	placement = root

	title = te_single_party_state.1.t
	desc = te_single_party_state.1.d
	flavor = te_single_party_state.1.f

	category = enactment

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3
	is_popup = yes

	# Undelayed from the senders, which test te_sps_dominance_pending.
	trigger = {
		is_enacting_law = law_type:law_single_party_state
		NOT = { has_variable = sps_dominant_party_ig_var }
		te_sps_mod_party_in_government = yes
	}

	# The enactment ended; vanilla's event made the choice, or has it now
	# (it fired at a checkpoint with a vanilla party to offer); or none of
	# this event's options is left.
	cancellation_trigger = {
		OR = {
			NOT = { is_enacting_law = law_type:law_single_party_state }
			has_variable = sps_dominant_party_ig_var
			AND = {
				has_variable = te_sps_left_to_vanilla
				te_sps_vanilla_party_in_government = yes
			}
			NOR = {
				te_sps_twin_option_valid = { PARTY = green }
				te_sps_twin_option_valid = { PARTY = populist }
				te_sps_twin_option_valid = { PARTY = technocratic }
			}
		}
	}

	immediate = {
		set_variable = {
			name = te_sps_dominance_pending
			days = 100
		}
		# Each party's largest group, as single_party_state.8 saves them:
		# set_sps_dominant_party_effect reads scope:sps_<PARTY>_ig and
		# scope:sps_<PARTY>_party.
		ordered_interest_group = {
			limit = {
				party ?= {
					is_party_type = green_party
				}
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = sps_green_ig
			party = {
				save_scope_as = sps_green_party
			}
		}
		ordered_interest_group = {
			limit = {
				party ?= {
					is_party_type = populist_party
				}
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = sps_populist_ig
			party = {
				save_scope_as = sps_populist_party
			}
		}
		ordered_interest_group = {
			limit = {
				party ?= {
					is_party_type = technocratic_party
				}
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = sps_technocratic_ig
			party = {
				save_scope_as = sps_technocratic_party
			}
		}
	}

	after = {
		if = {
			limit = { has_variable = te_sps_dominance_pending }
			remove_variable = te_sps_dominance_pending
		}
		hidden_effect = {
			update_party_support = yes
		}
	}

	option = { # the Green Party
		name = te_single_party_state.1.a
		default_option = yes
		trigger = {
			te_sps_twin_option_valid = { PARTY = green }
		}
		show_as_unavailable = {
			exists = scope:sps_green_ig
			any_interest_group = {
				is_in_government = yes
				party ?= {
					is_party_type = green_party
					party_has_insurrectionary_member = yes
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ruler ?= {
						interest_group ?= {
							party ?= {
								is_party_type = green_party
							}
						}
					}
				}
				add = 100
			}
		}
		set_sps_dominant_party_effect = { PARTY = green }
	}

	option = { # the Populist Party
		name = te_single_party_state.1.b
		default_option = yes
		trigger = {
			te_sps_twin_option_valid = { PARTY = populist }
		}
		show_as_unavailable = {
			exists = scope:sps_populist_ig
			any_interest_group = {
				is_in_government = yes
				party ?= {
					is_party_type = populist_party
					party_has_insurrectionary_member = yes
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ruler ?= {
						interest_group ?= {
							party ?= {
								is_party_type = populist_party
							}
						}
					}
				}
				add = 100
			}
		}
		set_sps_dominant_party_effect = { PARTY = populist }
	}

	option = { # the Technocratic Party
		name = te_single_party_state.1.c
		default_option = yes
		trigger = {
			te_sps_twin_option_valid = { PARTY = technocratic }
		}
		show_as_unavailable = {
			exists = scope:sps_technocratic_ig
			any_interest_group = {
				is_in_government = yes
				party ?= {
					is_party_type = technocratic_party
					party_has_insurrectionary_member = yes
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ruler ?= {
						interest_group ?= {
							party ?= {
								is_party_type = technocratic_party
							}
						}
					}
				}
				add = 100
			}
		}
		set_sps_dominant_party_effect = { PARTY = technocratic }
	}

	option = { # a vanilla party: single_party_state.8 asks at a checkpoint
		name = te_single_party_state.1.e
		trigger = {
			te_sps_vanilla_party_in_government = yes
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ruler ?= {
						interest_group ?= {
							party ?= {
								te_sps_is_mod_party = no
							}
						}
					}
				}
				add = 100
			}
		}
		custom_tooltip = te_single_party_state_another_party_tt
		hidden_effect = {
			set_variable = te_sps_left_to_vanilla
		}
	}
}
