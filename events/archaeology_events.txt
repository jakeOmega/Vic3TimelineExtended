# ============================================================================
# GRAND MONUMENTS — archaeology (issue #939)
# ============================================================================
# Design: docs/superpowers/archive/specs/2026-10-14-grand-monument-archaeology-design.md,
# reworked by 2026-10-16-grand-monument-archaeology-rework-design.md (§ 2).
# .1 and .2 are the events a player reads: a site found or located, and a dig
# finished. Each names its site, its place and state, the find and a
# historical note: the effect that fires it saves the state under the site's
# own name (gm_arch_ev_mark, scope:gm_arch_ev_<key>), and the description picks
# the site's lines by that scope (gm_arch_site_<key>_found / _dug, which embed
# _place and _history). An event cannot reach a container's custom
# localization, and a country variable would be overwritten by the next
# popup; a saved scope travels with each. .10 and .11 are hidden: they do the
# work a scripted GUI or an option ordered (gm_archaeology_effects.txt),
# outside any tooltip's walk.

# ============================================================================
# Event 10: a dig ordered (hidden)
# ============================================================================
# ROOT = the sponsor, scope:gm_arch_pick = the site (gm_arch_order_launch).
# Checks again (a month can pass between an offer and its answer), the weekly
# innovation the method diverts too (gm_arch_order_spared), starts the dig by
# the method ordered, then the lists and the refresh that applies the cost.
arch_events.10 = {
	type = country_event
	hidden = yes

	trigger = {
		gm_system_enabled = yes
		exists = scope:gm_arch_pick
	}

	immediate = {
		if = {
			limit = {
				scope:gm_arch_pick = {
					gm_arch_can_dig = { WHO = root }
					has_variable = gm_site_order_method
					gm_arch_order_spared = { WHO = root }
				}
			}
			scope:gm_arch_pick = {
				gm_arch_start_dig = { WHO = root }
			}
		}
		gm_arch_pass = yes
		trigger_event = { id = monument_events.20 }
	}
}

# ============================================================================
# Event 11: a dig suspended (hidden)
# ============================================================================
# ROOT = the sponsor, scope:gm_arch_pick = the site (gm_arch_order_suspend).
arch_events.11 = {
	type = country_event
	hidden = yes

	trigger = {

		exists = scope:gm_arch_pick

			gm_arch_stop_dig = yes
		}
		gm_arch_pass = yes
		trigger_event = { id = monument_events.20 }
	}
}
