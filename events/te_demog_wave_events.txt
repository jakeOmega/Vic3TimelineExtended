# ============================================================================
# DEMOGRAPHICS — THE EVENT WAVE (phase 3)
# Spec: docs/superpowers/specs/2026-10-08-demographics-design.md §9: the census's milestones, each firing when the
# census crosses its threshold, at most once a generation per country, under the Demographics rule's Full setting only.
# The dispatcher and its conditions: common/scripted_effects/te_demog_wave_effects.txt,
# common/scripted_triggers/te_demog_wave_triggers.txt; figures and thresholds: te_demog_wave_values.txt; modifiers:
# te_demog_wave_modifiers.txt. The console fires each on the player (`event te_debug_demog.1`, option w).
#
#    1   The Demographic Dividend   the working-age share 10% above 1836's reference (yearly)
#    2   The Baby Boom              a war of two years or more with mass mobilisation ends (monthly)
#    3   The Lost Generation        war dead pass a twentieth of the men of fighting age (monthly; a harm menu)
#    4   The Greying of the Nation  pensioners (or the people past the old-age threshold) pass the bar (yearly)
#    5   The Youth Bulge            a fifth of the people aged 20-29 while a tenth of the workforce is idle (yearly)
#    6   Missing Girls              110 boys born per 100 girls (Birth Limit after Microprocessor) (yearly)
#    7   Bare Branches              110 men per 100 women aged 20-59 (yearly)
#    8   The Empty Villages         a farm state old and shrinking (yearly; a harm menu)
#    9   The Overspill              a Primate or Dominant country's largest city at Migration Crowding's ceiling (yearly)
#    10  The Last Generation?       Biological Immortality (at once; once a game)
#
# No event has a trigger of its own: the dispatcher holds every condition, and a trigger would block the console. Each
# immediate picks what it names (the village state, the city) itself, with fallbacks for the console.
# ============================================================================

# ======================================================================
#  1 – The Demographic Dividend

#  dominant. Ten years each, the window a dividend lasts before the workers age.

	type = country_event
	placement = ROOT

	title = te_demog_wave.1.t

		name = te_demog_wave.1.a
		default_option = yes

			base = 2

# ======================================================================
#  4 – The Greying of the Nation
#  §9's four answers (raise the pension age, cut benefits, raise contributions, open the borders) under Old Age
#  Pension, and keeping the pension as it is; with no pension law, the borders, almshouses or the family. Raising the
#  age uses the pension setting itself (te_demog_pension_choose), so its five-year term binds as the buttons' does.
# ======================================================================
te_demog_wave.4 = {
	type = country_event
	placement = ROOT

	title = te_demog_wave.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = te_demog_wave.4.d_pension
				trigger = { te_demog_has_pension_law = yes }
			}
			triggered_desc = {
				desc = te_demog_wave.4.d
				trigger = { always = yes }
			}
		}
	}
	flavor = te_demog_wave.4.f

	event_image = { texture = "gfx/event_pictures/budget_hearing_congress.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	# Raise the pension age (Old Age Pension; a position may be chosen; not already raised or linked)
	option = {
		name = te_demog_wave.4.a
		trigger = {
			te_demog_wave_pension_free = yes
			NOT = { te_demog_wave_pension_raised = yes }
		}
		show_as_unavailable = { te_demog_has_pension_law = yes }
		custom_tooltip = {
			text = te_demog_pension_raised_effect_tt
			te_demog_pension_choose = { POS = raised }
		}
		show_as_tooltip = {
			add_modifier = { name = te_demog_pension_pos_raised }
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = { te_demog_pension_bill_share > 0.2 }
				add = 2
			}
			modifier = {
				trigger = { te_demog_country_unemployment > 0.1 }
				add = -1
			}
		}
	}

	# Cut benefits (a pension bill is being paid)
	option = {
		name = te_demog_wave.4.b
		trigger = { te_demog_wave_has_pension_bill = yes }
		add_modifier = { name = te_demog_wave_pension_cut days = long_modifier_time }
		add_modifier = { name = te_demog_wave_pension_cut_income days = long_modifier_time multiplier = te_demog_wave_pension_cut_flow }
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					OR = {
						in_default = yes
						scaled_debt >= 0.5
					}
				}
				add = 3
			}
		}
	}

	# Raise contributions (a pension bill is being paid)
	option = {
		name = te_demog_wave.4.c
		trigger = { te_demog_wave_has_pension_bill = yes }
		add_modifier = { name = te_demog_wave_contributions days = long_modifier_time }
		add_modifier = { name = te_demog_wave_contributions_income days = long_modifier_time multiplier = te_demog_wave_contributions_flow }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { scaled_debt >= 0.25 }
				add = 2
			}
		}
	}

	# Open the borders (any country)
	option = {
		name = te_demog_wave.4.d_opt
		add_modifier = { name = te_demog_wave_open_borders days = long_modifier_time is_decaying = yes }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { ig:ig_industrialists ?= { is_powerful = yes } }
				add = 1
			}
		}
	}

	# Almshouses (no pension law)
	option = {
		name = te_demog_wave.4.f_opt
		trigger = {
			NOT = { te_demog_has_pension_law = yes }
		}
		add_modifier = { name = te_demog_wave_almshouses days = long_modifier_time }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { has_law = law_type:law_public_health_insurance }
				add = 1
			}
		}
	}

	# The default, one option whatever the law: keep the promise (Old Age Pension), or the family will provide (none)
	option = {
		name = {
			trigger = { te_demog_has_pension_law = yes }
			text = te_demog_wave.4.e
		}
		name = {
			trigger = { always = yes }
			text = te_demog_wave.4.g
		}
		default_option = yes
		if = {
			limit = { te_demog_has_pension_law = yes }
			add_modifier = { name = te_demog_wave_pension_kept days = normal_modifier_time is_decaying = yes }
		}
		else = {
			add_modifier = { name = te_demog_wave_family_care days = long_modifier_time }
		}
		ai_chance = {
			base = 2
		}
	}
}

# ======================================================================
#  9 – The Overspill (on the metropolis)
#  §9: new towns, a green belt, or let it sprawl. The trigger is the spec's: a Metropolis-pattern state (the per-state
#  metropolitan share, §5.2, at 60 or more) at Migration Crowding's ceiling (te_demog_wave_cond_overspill); the event
#  sits on the most urban such state. New towns also switch the country's Settlement Authorities to Planned Towns,
#  where it runs any and knows Modern Urban Planning (Internal Resettlement).
# ======================================================================
te_demog_wave.9 = {
	type = country_event
	placement = scope:te_wave_state

	title = te_demog_wave.9.t
	desc = te_demog_wave.9.d
	flavor = te_demog_wave.9.f

	event_image = { texture = "gfx/event_pictures/humanitarian_crisis.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	immediate = {
		te_demog_wave_pick_city = yes
	}

	option = {
		name = te_demog_wave.9.a
		scope:te_wave_state = {
			add_modifier = { name = te_demog_wave_overspill_new_towns days = long_modifier_time is_decaying = yes }
		}
		add_treasury = te_demog_wave_cost_medium
		if = {
			limit = { te_demog_wave_can_plan_towns = yes }
			custom_tooltip = {
				text = te_demog_wave_tt_planned_towns
				te_demog_wave_plan_towns = yes
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					in_default = no
					scaled_debt < 0.25
				}
				add = 2
			}
		}
	}

	option = {
		name = te_demog_wave.9.b
		scope:te_wave_state = {
			add_modifier = { name = te_demog_wave_overspill_green_belt days = very_long_modifier_time }
		}
		add_modifier = { name = te_demog_wave_overspill_green_belt_politics days = normal_modifier_time is_decaying = yes }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { ig:ig_landowners ?= { is_powerful = yes } }
				add = 2
			}
		}
	}

	option = {
		name = te_demog_wave.9.c
		default_option = yes
		scope:te_wave_state = {
			add_modifier = { name = te_demog_wave_overspill_sprawl days = very_long_modifier_time }
		}
		add_modifier = { name = te_demog_wave_overspill_sprawl_politics days = normal_modifier_time is_decaying = yes }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { ig:ig_industrialists ?= { is_powerful = yes } }
				add = 2
			}
		}
	}
}
