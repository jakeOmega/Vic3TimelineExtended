namespace = te_debug_demog

# .130: um_nexus_scaling. A week after the first Nexus's goods modifiers under Full at a utilisation of 0.9 or less with
# its advanced materials figure at 1 or more (te_debug_demog_um_selfcheck_start, from te_demog_um_nexus_refresh): the
# building's digital assets output and advanced materials input modifiers against what the refresh expected, everything
# else on the good plus its own multiplier (1 + those) x (utilisation x factor - 1). OK: both within 0.05, so the engine
# applies a building's runtime goods mult axes from a state's pulse at the state's multiplier, the registered mod-good
# input axis too, and a read of the building's axis carries every source the refresh offsets. BROKEN with a read at the
# good's other bonuses: the modifier didn't reach the building (or the axis isn't registered); near -1: its multiplier
# read 0. The fallback is in the play-test list's Self-checks table. A state that lost its Nexus within the week clears
# the check for the next.
te_debug_demog.130 = {
	type = state_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_variable = te_dg_um_chk_da
				has_building = building_mind_upload_nexus
			}
			b:building_mind_upload_nexus = {
				root = {
					set_variable = { name = te_dg_um_chk_da_got value = prev.modifier:goods_output_digital_assets_mult }
					set_variable = { name = te_dg_um_chk_am_got value = prev.modifier:goods_input_advanced_materials_mult }
				}
			}
			if = {
				limit = {
					te_debug_demog_um_chk_da_gap < 0.05
					te_debug_demog_um_chk_da_gap > -0.05
					te_debug_demog_um_chk_am_gap < 0.05
					te_debug_demog_um_chk_am_gap > -0.05
				}
				debug_log = "TE_SELF_CHECK OK um_nexus_scaling util=[THIS.Var('te_dg_um_chk_util').GetValue|3] digital_assets=[THIS.Var('te_dg_um_chk_da_got').GetValue|3] expected=[THIS.Var('te_dg_um_chk_da').GetValue|3] advanced_materials=[THIS.Var('te_dg_um_chk_am_got').GetValue|3] expected=[THIS.Var('te_dg_um_chk_am').GetValue|3] state=[THIS.GetState.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN um_nexus_scaling util=[THIS.Var('te_dg_um_chk_util').GetValue|3] digital_assets=[THIS.Var('te_dg_um_chk_da_got').GetValue|3] expected=[THIS.Var('te_dg_um_chk_da').GetValue|3] advanced_materials=[THIS.Var('te_dg_um_chk_am_got').GetValue|3] expected=[THIS.Var('te_dg_um_chk_am').GetValue|3] (a read at the good's other bonuses: the modifier didn't reach the building, or the axis isn't registered; near -1: its multiplier read 0; another figure: a source changed within the week, so read the building's goods tooltip. docs/testing/demographics-playtest.md, Self-checks) state=[THIS.GetState.GetNameNoFormatting] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			debug_log_scopes = yes
			remove_variable = te_dg_um_chk_da
			remove_variable = te_dg_um_chk_am
			remove_variable = te_dg_um_chk_util
			remove_variable = te_dg_um_chk_da_got
			remove_variable = te_dg_um_chk_am_got
		}
		else = {
			remove_global_variable = te_demog_selfcheck_um_done
			if = {
				limit = { has_variable = te_dg_um_chk_da }
				remove_variable = te_dg_um_chk_da
				remove_variable = te_dg_um_chk_am
				remove_variable = te_dg_um_chk_util
			}
		}
	}
}
