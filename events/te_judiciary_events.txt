# The court overhaul (spec § 3.5). Fired a week after a ceiling law changes
# (te_jud_law_activated), on the country, so ROOT is the country and the
# multiplier resolves here, and the new law's modifier sums are in place. The
# gap is stored once: a script value would drift as the score decays. A
# same-name add_modifier would stack a second copy, and every copy re-reads
# the multiplier, so a second shock replaces the first, and only for a deeper
# cut than the one being paid for: a sideways or upward move during a shock
# adds nothing.
te_judiciary_events.1 = {
	type = country_event
	hidden = yes

	trigger = {
		has_variable = te_jud_seeded
		has_variable = te_judicial_independence
		te_jud_overhaul_gap >= 20
		OR = {
			NOT = { has_modifier = te_jud_court_overhaul }
			AND = {
				has_variable = te_jud_overhaul_mult
				te_jud_overhaul_mult_value > var:te_jud_overhaul_mult
			}
		}
	}

	immediate = {
		if = {
			limit = { has_modifier = te_jud_court_overhaul }
			remove_modifier = te_jud_court_overhaul
		}
		set_variable = {
			name = te_jud_overhaul_mult
			value = te_jud_overhaul_mult_value
		}
		add_modifier = {
			name = te_jud_court_overhaul
			years = 5
			is_decaying = yes
			multiplier = var:te_jud_overhaul_mult
		}
		# The protests (phase 2, spec § 14.1): Lawyers in the Streets, in
		# place of phase 1's bare middle-strata radicals.
		if = {
			limit = { te_jud_overhaul_gap >= 40 }
			trigger_event = { id = te_judiciary_events.6 }
		}
	}
}

# ======================================================================
#  PHASE 2: DEFYING THE COURT (spec § 14.1)
#  te_jud_challenge_monthly files a forced law with the court once it is in
#  force, and .2 fires 180 days later.
# ======================================================================

# The ruling (hidden). The court must still review forced laws, and the law
# must still stand; otherwise the case lapses. A court whose score is at the
# threshold for how hard the law was forced strikes it down (.3); a weaker one
# upholds it (.4).
te_judiciary_events.2 = {
	type = country_event
	hidden = yes

	trigger = {
		has_variable = te_jud_case_law
	}

	immediate = {
		if = {
			limit = {
				te_jud_court_can_strike_laws = yes
				has_law = var:te_jud_case_law
				has_variable = te_judicial_independence
				has_variable = te_jud_case_tier
			}
			if = {
				limit = { var:te_judicial_independence >= te_jud_strike_threshold }
				trigger_event = { id = te_judiciary_events.3 }
			}
			else = {
				trigger_event = { id = te_judiciary_events.4 }
			}
		}
		else = {
			te_jud_close_case = yes
		}
	}
}

# The Court Strikes Down the Act. Comply (the law in force before it comes

# ======================================================================
#  PHASE 3: AMENDMENT DEBATES (spec § 14.2)
#  The scripted ways in for three Judiciary amendments while a host court
#  law is enacted, pooled in on_law_checkpoint_debate. Each trigger
#  restates its amendment's `possible`, since add_amendment silently does
#  nothing when it fails. When the law in force carries the amendment, the

# Who Judges the Judges? (Judicial Elections), while Statutory Courts or
# Constitutional Review is enacted under an elected government.
te_judiciary_events.20 = {
	type = country_event
	placement = root

	title = te_judiciary_events.20.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = te_judiciary_events.20.d_carry
				trigger = {
					active_law:lawgroup_judiciary ?= {
						has_amendment = amendment_type:amendment_judicial_elections
					}
				}
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = te_judiciary_events.20.d
			}
		}
	}
	flavor = te_judiciary_events.20.f

	category = enactment

	event_image = { video = "southamerica_election" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	cooldown = { days = normal_modifier_time }

	trigger = {
		OR = {
			is_enacting_law = law_type:law_statutory_courts
			is_enacting_law = law_type:law_constitutional_review
		}
		# amendment_judicial_elections's `possible`
		te_jud_elected_government = yes
		NOT = {
			currently_enacting_law = {
				has_amendment = amendment_type:amendment_judicial_elections
			}
		}
		any_interest_group = {
			ig_counts_as_marginal = no
			OR = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_jacksonian_democrat
						has_ideology = ideology:ideology_populist
					}
				}
				law_stance = {
					law = law_type:law_universal_suffrage
					value > approve
				}
			}
		}
	}

	immediate = {
		currently_enacting_law = {
			save_scope_as = current_law_scope
		}
		ordered_interest_group = {
			limit = {
				ig_counts_as_marginal = no
				OR = {
					leader ?= {
						OR = {
							has_ideology = ideology:ideology_jacksonian_democrat
							has_ideology = ideology:ideology_populist
						}
					}
					law_stance = {
						law = law_type:law_universal_suffrage
						value > approve
					}
				}
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = amendment_sponsor_ig
		}
	}

	cancellation_trigger = {
		NOT = { currently_enacting_law = scope:current_law_scope }
	}

	option = { # Let the voters choose the judges
		name = te_judiciary_events.20.a
		trigger = { exists = scope:amendment_sponsor_ig }
		currently_enacting_law = {
			add_amendment = {
				type = amendment_judicial_elections
				sponsor = scope:amendment_sponsor_ig
				cooldown = 120
			}
		}
		add_enactment_modifier = {
			name = amendment_compromise_modifier
		}
		ig:ig_intelligentsia ?= {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
	}

	option = { # The bench is no place for candidates
		name = te_judiciary_events.20.b
		default_option = yes
		add_enactment_modifier = {
			name = amendment_purist_modifier
		}
		scope:amendment_sponsor_ig ?= {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }

# The Free Soil Case: a constitutional court rules on slavery while slavery
# is legal and abolitionists organize. Dred Scott v. Sandford (1857) held
# that the enslaved were property the constitution protected; Somerset v.
# Stewart (1772) that English law knew no slavery. A court sure of its
# standing (60) can take the Somerset road.

	type = country_event
	placement = root

		NOT = { has_variable = te_jud_heard_free_soil }
		te_jud_court_can_strike_laws = yes
		has_variable = te_judicial_independence
		var:te_judicial_independence >= 30
		NOT = { has_law = law_type:law_slavery_banned }

			is_political_movement_type = movement_anti_slavery

		set_variable = te_jud_heard_free_soil

			var:te_judicial_independence >= 60
		}
		add_modifier = { name = te_jud_case_free_soil years = 10 }
