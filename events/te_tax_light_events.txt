# ============================================================================
# LIGHT TAX CODE — internal country events
# ============================================================================
# Hidden, no loc. Raised by common/on_actions/te_tax_on_actions.txt, by
# te_tax_on_uprising / te_tax_repair_after_civil_war
# (common/scripted_effects/te_tax_civil_war_effects.txt) and by
# te_tax_light_adopt_copied_code (te_tax_light.4 raises te_tax_light.3). Each
# runs in the country it fires on; the relief and reaction modifiers are
# je_tax_code's scaling modifiers, which read the variables the sync writes. Logic:
# common/scripted_effects/te_tax_light_effects.txt;
# design: docs/superpowers/archive/specs/2026-10-09-tax-code-light-design.md.
# ============================================================================
namespace = te_tax_light

# The monthly processor: the running reactions fade, then one sync. Raised for
# every migrated country by the global pulse on the 1st.
te_tax_light.1 = {
	type = country_event
	hidden = yes

	trigger = { te_tax_code_light_on = yes }

	immediate = {
		te_tax_light_process_month = yes
	}
}

# Migration: every tax at this country's taxation law's rate for its native
# tax level. Raised for every country after the lobby, for each new country and
# by the monthly dispatch for any country still unmigrated. Firing it again
# (console: event te_tax_light.2) does nothing once the country is migrated.
te_tax_light.2 = {
	type = country_event
	hidden = yes

	trigger = { te_tax_code_light_on = yes }

	immediate = {
		te_tax_light_migrate_country = yes
	}
}

# Sync: raised a moment after a taxation law is activated (on_law_activated),
# so it sees the new law, for a civil war's winner, and a day after an
# uprising adopts a copied code (te_tax_light.4). Firing it again
# (console: event te_tax_light.3) changes nothing.
te_tax_light.3 = {
	type = country_event
	hidden = yes

	trigger = { te_tax_code_light_on = yes }

	immediate = {
		te_tax_light_sync = yes
	}
}

# Adoption of a copied code (#1469): on the rebels of an uprising, raised by
# te_tax_on_uprising once te_tax_light_copy_code has given them the original's
# code. The tax level is set to the code's and the first sync follows the next
# day (te_tax_light.3). Firing it again (console: event te_tax_light.4) puts the
# tax level back where the code last set it and syncs again.
te_tax_light.4 = {
	type = country_event
	hidden = yes

	trigger = { te_tax_code_light_on = yes }

	immediate = {
		te_tax_light_adopt_copied_code = yes
	}
}
