namespace = nuclear_custody

# ============================================================================
# NUCLEAR CUSTODY EVENTS
# ============================================================================
# Design: docs/superpowers/specs/2026-09-26-nuclear-custody-design.md; what
# shipped is docs/systems/nuclear_crisis_design.md §0.10. The logic is in
# common/scripted_effects/nuclear_custody_effects.txt.
#
#   .1   Who Holds the Button? — an armed country at the outbreak of a civil
#        war decides what becomes of its warheads
#   .2   (hidden) warheads have arrived: the rebels' share, or an ended
#        arsenal's settlement. Runs what needs ROOT = the new holder
#   .3   The Arsenal Changes Hands — a human's notice of warheads received
#   .4   The Old Regime's Arsenal — a human new regime's notice
#   .5   Deny Them the Bomb — a government losing a revolution while armed
#   .6   A Nuclear Power Divided — a watching power takes a side, offers to
#        secure the arsenal, or stays out
#   .7   An Offer to Secure Our Arsenal — the government answers .6's offer
#   .8   the helper hears the answer
#   .9   (hidden) the settlement, a day after a hook where the old owner may
#        still have been alive (nd_custody_reconcile_soon)
# ============================================================================

# ---- .1 Who Holds the Button? -----------------------------------------------
# Sent by nd_custody_on_civil_war_start with scope:nd_cw_rebel saved. The split
# is rolled once in immediate, so each option shows the numbers it applies.
# France tested its bomb early during the 1961 Generals' putsch in Algiers, so
# the device could not fall to the plotters; South Africa dismantled its six
# before the transfer of power. The default is to hold the line: what would
# have happened had nobody decided.
nuclear_custody.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_standoff_tension.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	title = nuclear_custody.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = nuclear_custody.1.d_secession
				trigger = { scope:nd_cw_rebel ?= { is_secessionist = yes } }
			}
			triggered_desc = {
				desc = nuclear_custody.1.d
			}
		}
	}
	flavor = nuclear_custody.1.f

	duration = 3

	trigger = {
		nd_is_armed = yes
		exists = scope:nd_cw_rebel
		scope:nd_cw_rebel = { is_country_alive = yes }
	}

	immediate = {
		nd_cw_roll_split = yes
	}

	# Hold the line: the rebels keep what they can reach.
	option = {
		name = nuclear_custody.1.a
		default_option = yes
		nd_cw_apply_split = { MODE = hold TT = nd_tt_cw_split_hold }
		ai_chance = {
			base = 2
			modifier = {
				trigger = { var:nd_cw_hold_taken < 1 }
				add = 6
			}
		}
	}

	# Pull them back: into storage for the whole war.
	option = {
		name = nuclear_custody.1.b
		nd_cw_pull_back = yes
		ai_chance = {
			base = 2
			modifier = {
				trigger = { var:nd_cw_hold_taken >= 1 }
				add = 6
			}
			modifier = {
				trigger = {
					has_variable = nd_safeguards
					var:nd_safeguards >= 3
				}
				add = -1
			}
		}
	}

	# Dismantle them: if the rebels win, nothing of ours is left for them.
	option = {
		name = nuclear_custody.1.c
		nd_cw_dismantle = yes
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					ruler_is_cautious = yes
					nd_cw_rebel_population_share >= 0.5
				}
				add = 3
			}
		}
	}
}

# ---- .5 Deny Them the Bomb ---------------------------------------------------------
# Sent by nd_cw_monthly_check, once per civil war, to a government losing a
# revolution while it still holds warheads, with scope:nd_cw_rebel saved. The
# winner would inherit what it holds. The loss of a hasty dismantling is
# rolled once in immediate, so the option shows it.
nuclear_custody.5 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_bunker_life.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	title = nuclear_custody.5.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = nuclear_custody.5.d_custodian
				trigger = { nd_cw_custodian_can_supervise = yes }
			}
			triggered_desc = {
				desc = nuclear_custody.5.d
			}
		}
	}
	flavor = nuclear_custody.5.f

	duration = 3

	trigger = {
		nd_is_armed = yes
		exists = scope:nd_cw_rebel
		scope:nd_cw_rebel = { is_country_alive = yes }
	}

	immediate = {
		nd_cw_deny_roll = yes
		var:nd_cw_custodian ?= { save_scope_as = nd_cw_helper }
	}

	# Take them apart now, in haste.
	option = {
		name = nuclear_custody.5.a
		nd_cw_deny_dismantle = yes
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 3
			}
			modifier = {
				trigger = { enemy_occupation >= 0.5 }
				add = 3
			}
		}
	}

	# Let the custodian take them apart: nothing goes missing.
	option = {
		name = nuclear_custody.5.b
		trigger = { nd_cw_custodian_can_supervise = yes }
		nd_cw_deny_supervised = yes
		ai_chance = {
			base = 8
		}
	}

	# Keep them: the last card.
	option = {
		name = nuclear_custody.5.c
		default_option = yes
		custom_tooltip = nd_tt_cw_deny_keep
		hidden_effect = {
			nd_cw_deny_clear = yes
		}
		ai_chance = {
			base = 3
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 2
			}
		}
	}
}

# ---- .6 A Nuclear Power Divided ----------------------------------------------------
# To a power that watches closely (nd_cw_would_watch), a week after the
# outbreak of a civil war in an armed country (nd_cw_notify_world). The pair
# is read from the observer's own variables, set for a month.
nuclear_custody.6 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_diplomacy_talks.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = nuclear_custody.6.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = nuclear_custody.6.d_rebels_armed
				trigger = { scope:nd_cwr_reb ?= { nd_is_armed = yes } }
			}
			triggered_desc = {
				desc = nuclear_custody.6.d
			}
		}
	}
	flavor = nuclear_custody.6.f

	duration = 3

	trigger = {
		nd_cwr_pair_at_war = yes
	}

	immediate = {
		var:nd_cwr_origin = { save_scope_as = nd_cwr_gov }
		var:nd_cwr_rebel = { save_scope_as = nd_cwr_reb }
	}

	# Back the government.
	option = {
		name = nuclear_custody.6.a
		nd_cwr_back = { SIDE = gov OTHER = reb TT = nd_tt_cwr_back_gov }
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					OR = {
						has_attitude = { who = scope:nd_cwr_gov attitude = protective }
						has_attitude = { who = scope:nd_cwr_gov attitude = genial }
						has_attitude = { who = scope:nd_cwr_gov attitude = loyal }
					}
				}
				add = 6
			}
			modifier = {
				trigger = { is_in_same_power_bloc = scope:nd_cwr_gov }
				add = 4
			}
			modifier = {
				trigger = { scope:nd_cwr_gov.relations:root >= relations_threshold:cordial }
				add = 2
			}
		}
	}

	# Back the rebels.
	option = {
		name = nuclear_custody.6.b
		nd_cwr_back = { SIDE = reb OTHER = gov TT = nd_tt_cwr_back_rebels }
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					OR = {
						has_diplomatic_pact = { who = scope:nd_cwr_gov type = rivalry }
						has_attitude = { who = scope:nd_cwr_gov attitude = antagonistic }
						has_attitude = { who = scope:nd_cwr_gov attitude = belligerent }
						has_attitude = { who = scope:nd_cwr_gov attitude = domineering }
					}
				}
				add = 5
			}
		}
	}

	# Offer to secure their arsenal.
	option = {
		name = nuclear_custody.6.c
		trigger = {
			nd_is_armed = yes
			scope:nd_cwr_gov = {
				nd_is_armed = yes
				NOT = { has_variable = nd_cw_custodian }
				NOT = { has_variable = nd_cw_offer_from }
			}
			NOR = {
				has_attitude = { who = scope:nd_cwr_gov attitude = antagonistic }
				has_attitude = { who = scope:nd_cwr_gov attitude = belligerent }
				has_attitude = { who = scope:nd_cwr_gov attitude = domineering }
			}
		}
		nd_cwr_offer_custody = yes
		ai_chance = {
			base = 0
			modifier = {
				trigger = { scope:nd_cwr_gov = { nd_stockpile >= 5 } }
				add = 4
			}
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 2
			}
		}
	}

	# Stay out.
	option = {
		name = nuclear_custody.6.e
		default_option = yes
		ai_chance = {
			base = 5
		}
	}

	after = {
		if = {
			limit = { has_variable = nd_cwr_origin }
			remove_variable = nd_cwr_origin
		}
		if = {
			limit = { has_variable = nd_cwr_rebel }
			remove_variable = nd_cwr_rebel
		}
	}
}
