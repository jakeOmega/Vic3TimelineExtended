namespace = covert_warfare

# ============================================================================
# EVENT 1: Covert Operation Detected (Attacker's Perspective)
# ============================================================================
# Fired by covert_ops_roll_detection_all with the burned operation in
# scope:iw_burned_op. The pact and container are removed in after; the
# attacker gains infamy and the target is notified.
# after also charges the agency's Tradecraft by the burned operation's tier.
# ============================================================================

covert_warfare.1 = {
	type = country_event
	placement = root

	title = covert_warfare.1.t
	desc = covert_warfare.1.d
	flavor = covert_warfare.1.f

	event_image = { texture = "gfx/event_pictures/espionage_dead_drop.dds" }
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		exists = scope:iw_burned_op
	}

	immediate = {
		# The burned operation arrives as scope:iw_burned_op from
		# covert_ops_roll_detection_all. Read it HERE and only here: the options
		# and `after` run when the player clicks, up to `duration` days later, by
		# which time stand-down or the monthly sync may have destroyed it.
		# Event loc cannot reach a container either, so the type code is copied
		# onto ROOT (scripting_best_practices.md, container rule 10). The phase
		# is copied for the same reason as the code — the options fire days
		# later and read only country variables. The war state is copied for
		# the same reason again, and because the war can begin between the
		# event firing and the player clicking: a wartime operation started
		# during the diplomatic play may be caught before the war and clicked
		# through after it starts, so the state has to be pinned here too.
		# Two instances open at once would share iw_burned_type_code (the second
		# immediate overwrites it); not reachable at the monthly pulse's one-burn
		# cadence with duration = 3.
		scope:iw_burned_op = {
			var:iw_target = { save_scope_as = detected_by_country }
			ROOT = {
				set_variable = { name = iw_burned_type_code value = PREV.var:iw_type_code }
				set_variable = { name = iw_burned_phase value = PREV.var:iw_phase }
				if = {
					limit = { has_war_with = scope:detected_by_country }
					set_variable = { name = iw_burned_at_war value = 1 }
				}
				else = {
					set_variable = { name = iw_burned_at_war value = 0 }
				}
			}
		}
		# UN redesign phase 3: an exposed severe operation is a record the
		# General Assembly can hold against us (un_dossier_effects.txt), unless
		# the exposure is costless (a wartime operation caught at war).
		if = {
			limit = {
				covert_code_tier_severe = { VAR = iw_burned_type_code }
				NOT = { covert_exposure_is_costless = yes }
			}
			un_dossier_record = { RECORD = covert POINTS = un_dos_points_covert }
			# ...and a grievance the target may bring before the Assembly
			# (redesign phase 4, un_docket_effects.txt).
			un_docket_note_grievance = { KIND = covert VICTIM = scope:detected_by_country }
			# An exposed attempt to overthrow a government (regime change,
			# code 9) is a crime the International Criminal Court may indict
			# our ruler for (UN redesign phase 5).
			if = {
				limit = { var:iw_burned_type_code = 9 }
				un_regime_icc_note_crime = { CRIME = 2 }
			}
		}
	}

	option = {
		name = covert_warfare.1.a
		default_option = yes

		# Acknowledge — more infamy, smaller relations hit, clean break
		custom_tooltip = covert_op_burned_tt
		if = {
			limit = { NOT = { covert_exposure_is_costless = yes } }
			change_infamy = { value = covert_exposure_infamy_acknowledge }
		}
		if = {
			limit = {
				exists = scope:detected_by_country
				NOT = { covert_exposure_is_costless = yes }
			}
			change_relations = {
				country = scope:detected_by_country
				value = covert_exposure_relations_acknowledge
			}
		}
		if = {
			limit = {
				exists = scope:detected_by_country
				covert_code_tier_severe = { VAR = iw_burned_type_code }
			}
			covert_exposure_third_party_blowback = yes
		}

		ai_chance = {
			base = 10
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 5
			}
		}
	}

	option = {
		name = covert_warfare.1.b

		# Deny — less infamy, more relations hit, but target hardens defenses
		custom_tooltip = covert_op_burned_tt
		if = {
			limit = { NOT = { covert_exposure_is_costless = yes } }
			change_infamy = { value = covert_exposure_infamy_deny }
		}
		if = {
			limit = {
				exists = scope:detected_by_country
				NOT = { covert_exposure_is_costless = yes }
			}
			change_relations = {
				country = scope:detected_by_country
				value = covert_exposure_relations_deny
			}
		}
		if = {
			limit = { exists = scope:detected_by_country }
			scope:detected_by_country = {
				add_modifier = {
					name = covert_op_heightened_vigilance_modifier
					days = normal_modifier_time
					is_decaying = yes
				}
			}
		}
		if = {
			limit = {
				exists = scope:detected_by_country
				covert_code_tier_severe = { VAR = iw_burned_type_code }
			}
			covert_exposure_third_party_blowback = yes
		}

		ai_chance = {
			base = 5
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 5
			}
		}
	}

	after = {
		# Always burn the operation (pact + container) and notify the target,
		# whichever option was chosen. Branch on the code copied in `immediate`;
		# covert_op_destroy is idempotent, so a container that already vanished
		# is harmless.
		if = {
			limit = {
				exists = scope:detected_by_country
				has_variable = iw_burned_type_code
			}
			if = {
				limit = { var:iw_burned_type_code = 0 }
				covert_op_burn = { TYPE = election_interference ACTION = covert_election_interference_action TARGET = scope:detected_by_country CODE = 0 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 1 }
				covert_op_burn = { TYPE = financial_subversion ACTION = covert_financial_subversion_action TARGET = scope:detected_by_country CODE = 1 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 2 }
				covert_op_burn = { TYPE = infrastructure_sabotage ACTION = covert_infrastructure_sabotage_action TARGET = scope:detected_by_country CODE = 2 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 3 }
				covert_op_burn = { TYPE = comms_disruption ACTION = covert_comms_disruption_action TARGET = scope:detected_by_country CODE = 3 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 4 }
				covert_op_burn = { TYPE = industrial_espionage ACTION = covert_industrial_espionage_action TARGET = scope:detected_by_country CODE = 4 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 5 }
				covert_op_burn = { TYPE = military_espionage ACTION = covert_military_espionage_action TARGET = scope:detected_by_country CODE = 5 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 6 }
				covert_op_burn = { TYPE = influence_campaign ACTION = covert_influence_campaign_action TARGET = scope:detected_by_country CODE = 6 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 7 }
				covert_op_burn = { TYPE = ideological_subversion ACTION = covert_ideological_subversion_action TARGET = scope:detected_by_country CODE = 7 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 8 }
				covert_op_burn = { TYPE = destabilization ACTION = covert_destabilization_action TARGET = scope:detected_by_country CODE = 8 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 9 }
				covert_op_burn = { TYPE = regime_change ACTION = covert_regime_change_action TARGET = scope:detected_by_country CODE = 9 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 10 }
				covert_op_burn = { TYPE = nuclear_sabotage ACTION = covert_nuclear_sabotage_action TARGET = scope:detected_by_country CODE = 10 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 11 }
				covert_op_burn = { TYPE = space_espionage ACTION = covert_space_espionage_action TARGET = scope:detected_by_country CODE = 11 }
			}
			else_if = {
				limit = { var:iw_burned_type_code = 12 }
				covert_op_burn = { TYPE = cultivate_assets ACTION = covert_cultivate_assets_action TARGET = scope:detected_by_country CODE = 12 }
			}

			# The agency's experience takes the hit (slice 5), by tier. Here,
			# while iw_burned_type_code still exists; removed further down.
			covert_tradecraft_burn_loss = yes

			scope:detected_by_country = {
				trigger_event = { id = covert_warfare.2 }
			}
		}
		if = {
			limit = { has_variable = iw_burned_type_code }
			remove_variable = iw_burned_type_code
		}
		if = {
			limit = { has_variable = iw_burned_phase }
			remove_variable = iw_burned_phase
		}
		if = {
			limit = { has_variable = iw_burned_at_war }
			remove_variable = iw_burned_at_war
		}
	}
}

# ============================================================================
# EVENT 2: Foreign Covert Operation Detected (Defender's Perspective)
# ============================================================================
# Fired on the target country when its counterintelligence has exposed an
# operation run against it: only from covert_warfare.1's `after`, straight
# after covert_op_burn has written the target's exposure record
# (iw_last_exposed_country / _type / _phase / _age). The record is the
# trigger, so a burn reaches the defender in every phase, preparation
# included, and exactly once per burn (#430). The text says whether the
# operation was stopped before it took effect (iw_last_exposed_phase = 1).
# ============================================================================

covert_warfare.2 = {
	type = country_event
	placement = root

	title = covert_warfare.2.t
	desc = {
		first_valid = {
			# Caught during preparation: nothing had happened to us yet.
			triggered_desc = {
				trigger = {
					has_variable = iw_last_exposed_phase
					var:iw_last_exposed_phase = 1
				}
				desc = covert_warfare.2.d_preparatory
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = covert_warfare.2.d
			}
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = iw_last_exposed_phase
					var:iw_last_exposed_phase = 1
				}
				desc = covert_warfare.2.f_preparatory
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = covert_warfare.2.f
			}
		}
	}

	event_image = { texture = "gfx/event_pictures/espionage_dead_drop.dds" }
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		# Only fire when someone's operation against us has just been exposed
		# (#158): the record covert_op_burn writes. It used to be one
		# has_modifier line per operation type — the modifier each type leaves
		# on its target — but those are applied only once an operation is
		# established, so a burn during preparation never reached the defender
		# (#430). The record covers every type and every phase.
		has_variable = iw_last_exposed_type
		NOT = { has_variable = iw_defender_event_cooldown }
	}

	immediate = {
		# Cooldown decays via the JE monthly pulse: iw_defender_event_cooldown_age
		# is incremented each month, and both variables are cleared after 24 months.
		# Without this, the event can only fire once per defender per game.
		set_variable = { name = iw_defender_event_cooldown value = yes }
		set_variable = { name = iw_defender_event_cooldown_age value = 0 }
		# Who ran it, for option b and its text. The record holds the country
		# itself (covert_op_burn), written on the burn's own tick; option b
		# checks the perpetrator is still alive when it is clicked, so a
		# country annexed while the popup is open closes it.
		var:iw_last_exposed_country ?= { save_scope_as = iw_exposed_by }
	}

	option = {
		name = covert_warfare.2.a
		default_option = yes

		# Harden defenses: temporary IC boost
		add_modifier = {
			name = intelligence_capacity_defense
			days = normal_modifier_time
			is_decaying = yes
			multiplier = 3
		}

		ai_chance = {
			base = 10
		}
	}

	# Retaliate in kind (#430): our own service turns the exposure into a
	# foothold inside theirs — a network in the perpetrator's country
	# (covert_retaliation_seed_network), the part of the covert system that
	# makes the next operation against them start further along and stay
	# hidden longer. Until #430 this option only took 3 infamy off us, and no
	# retaliation happened; that effect is option c now. Needs a service of
	# our own: networks tick only on the covert journal entry's pulse.
	option = {
		name = covert_warfare.2.b
		trigger = {
			exists = scope:iw_exposed_by
			scope:iw_exposed_by = { is_country_alive = yes }
			NOT = { this = scope:iw_exposed_by }
			custom_tooltip = {
				text = covert_warfare.2.b_needs_service_tt
				has_game_rule = covert_warfare_enabled
				has_journal_entry = je_covert_warfare
			}
		}
		show_as_unavailable = {
			exists = scope:iw_exposed_by
			scope:iw_exposed_by = { is_country_alive = yes }
			NOT = { this = scope:iw_exposed_by }
		}
		custom_tooltip = covert_warfare.2.b_tt
		hidden_effect = {
			covert_retaliation_seed_network = { TARGET = scope:iw_exposed_by }
		}

		ai_chance = {
			base = 5
		}
	}

	# Make the evidence public: the exposure buys us diplomatic credibility.
	# The effect "Retaliate in kind" carried until #430, under its honest name.
	option = {
		name = covert_warfare.2.c

		change_infamy = -3

		ai_chance = {
			base = 5
		}
	}
}
