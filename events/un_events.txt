# Event 1: The United Nations Charter - Formation Event (fires once)
un_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/un_charter_signing.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = un_events.1.t
	desc = {
		first_valid = {
			triggered_desc = { desc = un_events.1.d_council trigger = { has_law = law_type:law_council_republic } }
			triggered_desc = { desc = un_events.1.d_authoritarian trigger = { has_law = law_type:law_single_party_state } }
			triggered_desc = { desc = un_events.1.d_theocratic trigger = { has_law = law_type:law_theocracy } }
			desc = un_events.1.d
		}
	}
	flavor = un_events.1.f

	duration = 3

	trigger = {
		# Fired globally by on_action when 3+ great powers have the tech
		NOT = { has_variable = un_charter_signed }
		NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		is_subject = no
		NOT = { is_country_type = decentralized }
	}

	immediate = {
		set_variable = { name = un_charter_signed value = 1 }
	}

	# Option A: Sign the charter as a founding member
	option = {
		name = un_events.1.a
		default_option = yes
		ai_chance = {
			base = 8
			modifier = {
				trigger = { has_law = law_type:law_isolationism }
				add = -6
			}
		}
		# First signatory hosts UN Headquarters
		if = {
			limit = { NOT = { has_global_variable = un_hq_country } }
			set_global_variable = { name = un_hq_country value = ROOT }
			je:je_united_nations ?= { add_modifier = { name = un_headquarters_modifier } }
		}
		# Great powers get Security Council permanent seats — capped at 5
		# (matches un_join_button behavior). The 5th GP to sign Option A
		# claims the last seat; subsequent GPs join the UN without a seat.
		if = {
			limit = {
				country_rank >= rank_value:great_power
				un_permanent_seat_count < 5
			}
			je:je_united_nations ?= { add_modifier = { name = un_security_council_modifier } }
			je:je_united_nations ?= { add_modifier = { name = un_permanent_member_modifier } }
		}
		je:je_united_nations ?= { add_modifier = { name = un_member_modifier } }
		je:je_united_nations ?= { add_modifier = { name = un_founding_member_modifier } }
		# No conventions are ratified at UN founding, so this is a no-op today;
		# kept for symmetry with the other join paths in case the founding event
		# is ever fired after conventions exist (alternate timeline starts, etc.).
		un_apply_ratified_conventions = yes
		ig_approval_effect = { IG = ig_intelligentsia MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
	}

	# Option B: Observe but don't join yet
	option = {
		name = un_events.1.b
		ai_chance = {
			base = 2
			modifier = {
				trigger = { has_law = law_type:law_isolationism }
				add = 5
			}
		}
		add_modifier = {
			name = un_observer_status_modifier
			days = long_modifier_time
			is_decaying = yes
		}
	}

	# Option C: Reject this internationalist nonsense
	option = {
		name = un_events.1.c
		ai_chance = {
			base = 1
			modifier = {
				trigger = { has_law = law_type:law_total_war }
				add = 3
			}
		}
		add_modifier = {
			name = un_rejectionist_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		ig_approval_effect = { IG = ig_armed_forces MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
	}
}

# Event 2: General Assembly Crisis - A rival's aggression is condemned
un_events.2 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = un_events.2.t
	desc = un_events.2.d
	flavor = un_events.2.f

	duration = 3

	cooldown = { days = long_modifier_time }

	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		is_at_war = no
		any_country = {
			has_diplomatic_pact = { who = ROOT type = rivalry }
			is_at_war = yes
			is_diplomatic_play_initiator = yes
		}
		NOT = { has_modifier = un_ga_resolution_modifier }
		NOT = { has_global_variable = un_vote_active }
	}

	immediate = {
		random_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				is_at_war = yes
				is_diplomatic_play_initiator = yes
			}
			save_scope_as = condemned_country
		}
		random_country = {
			limit = {
				NOT = { this = ROOT }
				has_war_with = scope:condemned_country
				is_diplomatic_play_initiator = no
			}
			save_scope_as = defender_country
		}
		# Reserve the vote slot immediately so no other country can start one
		un_vote_reserve = yes
	}

	# Option A: Propose condemnation resolution to the General Assembly
	option = {
		name = un_events.2.a
		default_option = yes
		ai_chance = {
			base = 5
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 3
			}
		}
		custom_tooltip = un_vote_initiated_tt
		# Proposer takes a personal stance
		add_modifier = {
			name = un_ga_resolution_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		# Open the vote: every other UN member votes, results in a year
		hidden_effect = {
			scope:condemned_country = { save_scope_as = un_vote_target }
			un_resolution_open = { TOPIC = condemn }
		}
	}

	# Option B: Vote in favor but don't lead
	option = {
		name = un_events.2.b
		ai_chance = {
			base = 5
		}
		add_modifier = {
			name = un_ga_resolution_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		custom_tooltip = un_ledger_tt_credibility_up_0_5
		un_ledger_actor_entry = { PILLAR = credibility POINTS = 0.5 REASON = 102 }
		change_relations = { country = scope:condemned_country value = -5 }
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = root
			SECOND = scope:condemned_country
			PRO_AMOUNT = 1
			ANTI_AMOUNT = -1
			FACTOR = appeasement_relations_decreased
		}
		change_relations = { country = scope:defender_country value = 5 }
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = root
			SECOND = scope:defender_country
			PRO_AMOUNT = 1
			ANTI_AMOUNT = -1
			FACTOR = appeasement_relations_increased
		}
		# Release the vote slot since we didn't start one
		un_vote_release = yes
	}

	# Option C: Abstain - avoid entanglements
	option = {
		name = un_events.2.c
		ai_chance = {
			base = 2
			modifier = {
				trigger = { has_law = law_type:law_isolationism }
				add = 5
			}
		}
		custom_tooltip = un_ledger_tt_credibility_down_0_5
		un_ledger_actor_entry = { PILLAR = credibility POINTS = -0.5 REASON = 102 }
		ig_approval_effect = { IG = ig_armed_forces MODIFIER = ig_approval_positive_modifier DAYS = short_modifier_time }
		# Release the vote slot since we didn't start one
		un_vote_release = yes
	}
}

# Event 4: Peacekeeping Mission Deployed
un_events.4 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/peacekeeping_deployment.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = un_events.4.t
	desc = un_events.4.d
	flavor = un_events.4.f

	duration = 3

	cooldown = { days = long_modifier_time }

	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		je:je_united_nations ?= { has_modifier = un_peacekeeping_contributor_modifier }
		country_rank >= rank_value:major_power
		# A host that is not our own enemy and has no peacekeepers yet. The
		# host carries un_peacekeeping_welcomed/resented_modifier (un_events.102);
		# un_peacekeeping_deployed_modifier sits on the CONTRIBUTOR, so testing
		# it here never excluded anyone. Keep in step with the immediate below.
		any_country = {
			NOT = { this = ROOT }
			is_at_war = yes
			NOT = { has_war_with = ROOT }
			NOT = { has_modifier = un_peacekeeping_welcomed_modifier }
			NOT = { has_modifier = un_peacekeeping_resented_modifier }
		}
	}

	immediate = {
		random_country = {
			limit = {
				NOT = { this = ROOT }
				is_at_war = yes
				NOT = { has_war_with = ROOT }
				NOT = { has_modifier = un_peacekeeping_welcomed_modifier }
				NOT = { has_modifier = un_peacekeeping_resented_modifier }
			}
			save_scope_as = peacekeeping_target
		}
	}

	# Option A: Deploy a full peacekeeping force - costly but effective
	option = {
		name = un_events.4.a
		default_option = yes
		ai_chance = {
			base = 4
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 3
			}
		}
		add_modifier = {
			name = un_peacekeeping_deployed_modifier
			days = long_modifier_time
			is_decaying = yes
		}
		custom_tooltip = un_ledger_tt_delivery_up_1_5
		un_ledger_actor_entry = { PILLAR = delivery POINTS = 1.5 REASON = 104 }
		# International standing: an actual deployment, not a pledge. Option B
		# (observers only) is the token gesture and deliberately pays NOTHING —
		# the whole point of standing is that only delivery counts. One payout
		# per firing; the event is gated on the target not already carrying
		# un_peacekeeping_deployed_modifier and on a five-year cooldown, and
		# the relations gain with the host is already in this option.
		custom_tooltip = {
			text = un_standing_peacekeeping_delivered_tt
			un_standing_gain = { AMOUNT = un_standing_peacekeeping_delivered_gain REASON = 3 }
		}
		change_relations = { country = scope:peacekeeping_target value = 20 }
		create_diplomatic_catalyst = { type = catalyst_event_positive target = scope:peacekeeping_target }
		ig_approval_effect = { IG = ig_intelligentsia MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
		# Military costs
		add_radicals = {
			value = small_radicals
			strata = upper
		}
		# Notify the target country
		save_scope_as = peacekeeping_donor
		scope:peacekeeping_target = {
			trigger_event = { id = un_events.102 days = 14 }
		}
	}

	# Option B: Send observers only - cheaper but less impactful
	option = {
		name = un_events.4.b
		ai_chance = {
			base = 5
		}
		add_modifier = {
			name = un_observer_mission_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		custom_tooltip = un_ledger_tt_delivery_up_0_5
		un_ledger_actor_entry = { PILLAR = delivery POINTS = 0.5 REASON = 104 }
		change_relations = { country = scope:peacekeeping_target value = 5 }
	}

	# Option C: Decline to participate - save resources
	option = {
		name = un_events.4.c
		ai_chance = {
			base = 2
			modifier = {
				trigger = { is_at_war = yes }
				add = 5
			}
		}
		custom_tooltip = un_ledger_tt_credibility_down_1
		un_ledger_actor_entry = { PILLAR = credibility POINTS = -1 REASON = 104 }
		add_loyalists = {
			value = small_radicals
			strata = upper
		}
	}
}

# Event 6: UN Reform Debate - Security Council expansion or restructuring
un_events.6 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = un_events.6.t
	desc = {
		first_valid = {
			triggered_desc = { desc = un_events.6.d_council trigger = { has_law = law_type:law_council_republic } }
			triggered_desc = { desc = un_events.6.d_authoritarian trigger = { has_law = law_type:law_single_party_state } }
			desc = un_events.6.d
		}
		triggered_desc = {
			trigger = { exists = scope:un_hq_state }
			desc = un_events.6.d_hq
		}
	}
	flavor = un_events.6.f

	duration = 3

	cooldown = { days = long_modifier_time }

	# Redesign phase 2: the reform on the table is the charter's next stage
	# (Reform I raises the ceiling on authority from 70 to 85, Reform II to
	# 100), and it can be tabled only once the organisation has outgrown its
	# charter — the same gate as the chamber and the button
	# (un_charter_reform_ripe, un_ladder_triggers.txt).
	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		country_rank >= rank_value:great_power
		un_charter_reform_ripe = yes
		NOT = { un_resolution_topic_on_cooldown = { TOPIC = reform } }
		NOT = { has_modifier = un_reform_champion_modifier }
		NOT = { has_modifier = un_reform_opponent_modifier }
		NOT = { has_global_variable = un_vote_active }
	}

	immediate = {
		# Reserve the vote slot immediately so no other country can start one
		un_vote_reserve = yes
		if = {
			limit = {
				any_country = {
					any_scope_state = { has_building = building_un_headquarters }
				}
			}
			random_country = {
				limit = { any_scope_state = { has_building = building_un_headquarters } }
				random_scope_state = {
					limit = { has_building = building_un_headquarters }
					save_scope_as = un_hq_state
				}
			}
		}
	}

	# Option A: Propose reform resolution to the General Assembly
	option = {
		name = un_events.6.a
		default_option = yes
		ai_chance = {
			base = 4
			modifier = {
