namespace = un_events

# ============================================================================
# UNITED NATIONS EVENTS
# Gated by intergovernmental_organizations technology (Era 6+).
#
# Redesign phase 4 (docs/systems/un_redesign_design.md §8, §0.4): nothing here
# is rolled at random any more. Each event is raised by the docket
# (common/scripted_effects/un_docket_effects.txt) for a real situation and sent
# to the countries it concerns, or fired where the thing it answers happens:
#   docket    2 (an appeal: nuclear strike, aggression, covert operation),
#             3, 6, 9, 12, 14, 16, 17, 18, 19, 21, 22 (topics to table),
#             4 (a collapsing state), 7 (a famine), 15 (an embargo)
#   hooks     5 (sanctions adopted), 8 (a court case from event 2),
#             10 (the crisis opens), 11 (a veto), 20 (a great power leaves),
#             30-32 (dissolution and refounding), 101-103 (notifications)
# Event 13 (an ideological confrontation) was retired: no situation drives it.
# Every option costs something; a proposer event's decline hands the item to
# another delegation (un_docket_offer_topic_next) and earns nothing.
# ============================================================================

# Event 1: The United Nations Charter - Formation Event (fires once)
un_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/un_charter_signing.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = un_events.1.t
	desc = {
		first_valid = {
			triggered_desc = { desc = un_events.1.d_council trigger = { has_law = law_type:law_council_republic } }
			triggered_desc = { desc = un_events.1.d_authoritarian trigger = { has_law = law_type:law_single_party_state } }
			triggered_desc = { desc = un_events.1.d_theocratic trigger = { has_law = law_type:law_theocracy } }
			desc = un_events.1.d
		}
	}
	flavor = un_events.1.f

	duration = 3

	trigger = {
		# Fired globally by on_action when 3+ great powers have the tech
		NOT = { has_variable = un_charter_signed }
		NOT = { je:je_united_nations ?= { has_modifier = un_member_modifier } }
		is_subject = no
		NOT = { is_country_type = decentralized }
	}

	immediate = {
		set_variable = { name = un_charter_signed value = 1 }
	}

	# Option A: Sign the charter as a founding member
	option = {
		name = un_events.1.a
		default_option = yes
		ai_chance = {
			base = 8
			modifier = {
				trigger = { has_law = law_type:law_isolationism }
				add = -6
			}
		}
		# First signatory hosts UN Headquarters
		if = {
			limit = { NOT = { has_global_variable = un_hq_country } }
			set_global_variable = { name = un_hq_country value = ROOT }
			je:je_united_nations ?= { add_modifier = { name = un_headquarters_modifier } }
		}
		# Signing seats nobody. The charter goes to every country at once and
		# the AI answers the moment it arrives, so seats given to the first
		# great powers to sign always went to the AI. A great power that signs
		# stands for a permanent seat, filled by prestige when the signing
		# period closes (un_seat_effects.txt); the tooltip says so.
		un_seat_join_tooltip = yes
		je:je_united_nations ?= { add_modifier = { name = un_member_modifier } }
		je:je_united_nations ?= { add_modifier = { name = un_founding_member_modifier } }
		# No conventions are ratified at UN founding, so this is a no-op today;
		# kept for symmetry with the other join paths in case the founding event
		# is ever fired after conventions exist (alternate timeline starts, etc.).
		un_apply_ratified_conventions = yes
		ig_approval_effect = { IG = ig_intelligentsia MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
	}

	# Option B: Observe but don't join yet
	option = {
		name = un_events.1.b
		ai_chance = {
			base = 2
			modifier = {
				trigger = { has_law = law_type:law_isolationism }
				add = 5
			}
		}
		add_modifier = {
			name = un_observer_status_modifier
			days = long_modifier_time
			is_decaying = yes
		}
	}

	# Option C: Reject this internationalist nonsense
	option = {
		name = un_events.1.c
		ai_chance = {
			base = 1
			modifier = {
				trigger = { has_law = law_type:law_total_war }
				add = 3
			}
		}
		add_modifier = {
			name = un_rejectionist_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		ig_approval_effect = { IG = ig_armed_forces MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
	}
}

# Event 2: An Appeal to the Assembly (redesign phase 4)
# Raised by the docket (un_docket_effects.txt, items 1-3) against
# scope:condemned_country: a nuclear strike, a war of aggression against a
# member, or a severe covert operation exposed (var:un_dkt_cause on the accused:
# 1, 2, 3). Offered first to the aggrieved party (scope:un_dkt_victim); if it
# lets the matter rest, to one member with a stake (un_docket_appeal_second);
# then the item lapses. Holds the floor while it is answered. Every answer
# costs something: tabling a censure costs relations and the tabling marker, a
# court case costs relations, letting it rest costs prestige or credibility.
un_events.2 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = {
		first_valid = {
			triggered_desc = {
				desc = un_events.2.t_ours
				trigger = { scope:un_dkt_victim ?= ROOT }
			}
			desc = un_events.2.t
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				desc = un_events.2.d_nuclear_ours
				trigger = {
					scope:un_dkt_victim ?= ROOT
					scope:condemned_country = { un_docket_cause_is = { CAUSE = 1 } }
				}
			}
			triggered_desc = {
				desc = un_events.2.d_war_ours
				trigger = {
					scope:un_dkt_victim ?= ROOT
					scope:condemned_country = { un_docket_cause_is = { CAUSE = 2 } }
				}
			}
			triggered_desc = {
				desc = un_events.2.d_covert_ours
				trigger = {
					scope:un_dkt_victim ?= ROOT
					scope:condemned_country = { un_docket_cause_is = { CAUSE = 3 } }
				}
			}
			triggered_desc = {
				desc = un_events.2.d_nuclear
				trigger = { scope:condemned_country = { un_docket_cause_is = { CAUSE = 1 } } }
			}
			triggered_desc = {
				desc = un_events.2.d_war
				trigger = { scope:condemned_country = { un_docket_cause_is = { CAUSE = 2 } } }
			}
			triggered_desc = {
				desc = un_events.2.d_covert
				trigger = { scope:condemned_country = { un_docket_cause_is = { CAUSE = 3 } } }
			}
			desc = un_events.2.d
		}
		triggered_desc = {
			desc = un_events.2.d_place
			trigger = { exists = scope:un_dkt_place }
		}
		desc = un_events.2.d_grounds
	}
	flavor = un_events.2.f

	duration = 3

	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		exists = scope:condemned_country
		NOT = { this = scope:condemned_country }
		NOT = { has_global_variable = un_vote_active }
	}

	immediate = {
		# Hold the floor while we answer.
		un_vote_reserve = yes
	}

	# Option A: Table a condemnation — if the accused's record gives grounds
	option = {
		name = un_events.2.a
		trigger = {
			scope:condemned_country = { un_case_supports = { TOPIC = condemn } }
			NOT = { un_resolution_topic_on_cooldown = { TOPIC = condemn } }
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { scope:un_dkt_victim ?= ROOT }
				add = 5
			}
			modifier = {
				trigger = { has_diplomatic_pact = { who = scope:condemned_country type = rivalry } }
				add = 5
			}
		}
		custom_tooltip = un_vote_initiated_tt
		add_modifier = {
			name = un_ga_resolution_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		change_relations = { country = scope:condemned_country value = -20 }
		hidden_effect = {
			scope:condemned_country = { save_scope_as = un_vote_target }
			un_resolution_open = { TOPIC = condemn }
		}
	}

	# Option B: Call for sanctions — heavier grounds, and a major power to lead them
	option = {
		name = un_events.2.b
		trigger = {
			country_rank >= rank_value:major_power
			NOT = { has_modifier = un_sanctions_enforcer_modifier }
			NOT = { un_resolution_topic_on_cooldown = { TOPIC = sanctions } }
			scope:condemned_country = {
				un_case_supports = { TOPIC = sanctions }
				NOT = { has_modifier = un_sanctions_target_modifier }
			}
		}
		ai_chance = {
			base = 4
			modifier = {
				trigger = { scope:condemned_country = { un_docket_cause_is = { CAUSE = 1 } } }
				add = 6
			}
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 3
			}
		}
		custom_tooltip = un_vote_initiated_tt
		add_modifier = {
			name = un_ga_resolution_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		add_modifier = {
			name = un_request_cooldown
			days = long_modifier_time
			is_decaying = yes
		}
		change_relations = { country = scope:condemned_country value = -30 }
		hidden_effect = {
			scope:condemned_country = { save_scope_as = un_vote_target }
			un_resolution_open = { TOPIC = sanctions }
		}
	}

	# Option C: Ask for peacekeepers on our own soil — the attacked member only
	option = {
		name = un_events.2.c
		trigger = {
			scope:un_dkt_victim ?= ROOT
			scope:condemned_country = { un_docket_cause_is = { CAUSE = 2 } }
			NOT = { un_resolution_topic_on_cooldown = { TOPIC = peacekeeping_request } }
			NOT = { has_modifier = un_peacekeeping_welcomed_modifier }
			NOT = { has_modifier = un_request_cooldown }
		}
		ai_chance = {
			base = 6
		}
		un_propose_peacekeepers_effect = yes
	}

	# Option D: Bring a case before the World Court — no grounds needed, no vote
	option = {
		name = un_events.2.court
		trigger = {
			scope:condemned_country = {
				je:je_united_nations ?= { has_modifier = un_member_modifier }
				NOT = { has_modifier = un_court_compliance_modifier }
				NOT = { has_modifier = un_court_defiance_modifier }
			}
		}
		ai_chance = {
			base = 3
			modifier = {
				trigger = {
					NOT = { scope:condemned_country = { un_case_supports = { TOPIC = condemn } } }
				}
				add = 8
			}
		}
		custom_tooltip = un_events.2.court_tt
		change_relations = { country = scope:condemned_country value = -15 }
		un_vote_release = yes
		hidden_effect = {
			save_scope_as = court_plaintiff
			scope:condemned_country = { trigger_event = { id = un_events.8 days = 30 } }
		}
	}

	# Option E (the aggrieved party): Leave it to the Council
	option = {
		name = un_events.2.e
		trigger = { scope:un_dkt_victim ?= ROOT }
		default_option = yes
		ai_chance = {
			base = 2
			modifier = {
				trigger = {
					OR = {
						has_treaty_alliance_with = { TARGET = scope:condemned_country }
						is_in_same_power_bloc = scope:condemned_country
					}
				}
				add = 10
			}
		}
		add_modifier = {
			name = un_appeal_withheld_modifier
			days = short_modifier_time
			is_decaying = yes
		}
		custom_tooltip = un_events.2.pass_tt
		un_vote_release = yes
		hidden_effect = {
			un_docket_pass_appeal = yes
		}
	}

	# Option G (a member with a stake): Not our quarrel — the item lapses
	option = {
		name = un_events.2.g
		trigger = { NOT = { scope:un_dkt_victim ?= ROOT } }
		default_option = yes
		ai_chance = {
			base = 4
		}
		custom_tooltip = un_ledger_tt_credibility_down_0_5
		un_ledger_actor_entry = { PILLAR = credibility POINTS = -0.5 REASON = 102 }
		if = {
			limit = { exists = scope:un_dkt_victim }
			change_relations = { country = scope:un_dkt_victim value = -10 }
		}
		un_vote_release = yes
	}
}

# Event 4: Peacekeeping Mission Deployed
un_events.4 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/peacekeeping_deployment.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = un_events.4.t
	desc = un_events.4.d
	flavor = un_events.4.f

	duration = 3

	# Redesign phase 4: no longer rolled against a random war anywhere. The
	# docket (un_docket_take_up_collapse, item 4) sends it to up to three
	# major-power members, those running the peacekeeping programme first,
	# when a state collapses: scope:peacekeeping_target is the host. The host
	# carries un_peacekeeping_welcomed/resented_modifier (un_events.102) once
	# peacekeepers are there.
	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		country_rank >= rank_value:major_power
		exists = scope:peacekeeping_target
		NOT = { this = scope:peacekeeping_target }
		NOT = { has_war_with = scope:peacekeeping_target }
		scope:peacekeeping_target = {
			NOT = { has_modifier = un_peacekeeping_welcomed_modifier }
			NOT = { has_modifier = un_peacekeeping_resented_modifier }
		}
	}

	# Option A: Deploy a full peacekeeping force - costly but effective
	option = {
		name = un_events.4.a
		default_option = yes
		ai_chance = {
			base = 4
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 3
			}
		}
		add_modifier = {
			name = un_peacekeeping_deployed_modifier
			days = long_modifier_time
			is_decaying = yes
		}
		custom_tooltip = un_ledger_tt_delivery_up_1_5
		un_ledger_actor_entry = { PILLAR = delivery POINTS = 1.5 REASON = 104 }
		# International standing: an actual deployment, not a pledge. Option B
		# (observers only) is the token gesture and deliberately pays NOTHING —
		# the whole point of standing is that only delivery counts. One payout
		# per firing; the event is gated on the target not already carrying
		# un_peacekeeping_deployed_modifier and on a five-year cooldown, and
		# the relations gain with the host is already in this option.
		custom_tooltip = {
			text = un_standing_peacekeeping_delivered_tt
			un_standing_gain = { AMOUNT = un_standing_peacekeeping_delivered_gain REASON = 3 }
		}
		change_relations = { country = scope:peacekeeping_target value = 20 }
		create_diplomatic_catalyst = { type = catalyst_event_positive target = scope:peacekeeping_target }
		ig_approval_effect = { IG = ig_intelligentsia MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
		# Military costs
		add_radicals = {
			value = small_radicals
			strata = upper
		}
		# The mission (redesign phase 6): a stabilisation mission in the
		# collapsing country's most populous state, which we join or open.
		if = {
			limit = { scope:peacekeeping_target = { un_mission_can_join_host = { TYPE = stabilisation } } }
			custom_tooltip = un_mission_stab_joined_tt
		}
		hidden_effect = {
			scope:peacekeeping_target = { save_scope_as = un_msn_host }
			un_mission_join_host = { TYPE = stabilisation CODE = 3 ORDER = un_docket_state_pop_order }
			un_mission_add_contributor = yes
		}
		# Notify the target country
		save_scope_as = peacekeeping_donor
		scope:peacekeeping_target = {
			trigger_event = { id = un_events.102 days = 14 }
		}
	}

	# Option B: Send observers only - cheaper but less impactful
	option = {
		name = un_events.4.b
		ai_chance = {
			base = 5
		}
		add_modifier = {
			name = un_observer_mission_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		# Redesign phase 4: observers are not free. A quarter of what a full
		# programme costs, for as long as the mission runs.
		add_modifier = {
			name = un_observer_mission_cost
			multiplier = un_docket_token_expense_value
			days = normal_modifier_time
			is_decaying = yes
		}
		custom_tooltip = un_ledger_tt_delivery_up_0_5
		un_ledger_actor_entry = { PILLAR = delivery POINTS = 0.5 REASON = 104 }
		change_relations = { country = scope:peacekeeping_target value = 5 }
	}

	# Option C: Decline to participate - save resources
	option = {
		name = un_events.4.c
		ai_chance = {
			base = 2
			modifier = {
				trigger = { is_at_war = yes }
				add = 5
			}
		}
		custom_tooltip = un_ledger_tt_credibility_down_1
		un_ledger_actor_entry = { PILLAR = credibility POINTS = -1 REASON = 104 }
		add_loyalists = {
			value = small_radicals
			strata = upper
		}
	}
}

# Event 5: Sanctions Regime - Economic pressure on a pariah state
un_events.5 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/sanctions_cargo_inspection.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	title = un_events.5.t
	desc = un_events.5.d
	flavor = un_events.5.f

	duration = 3

	# Redesign phase 4: no longer rolled. un_vote.2 fires it to the proposer of
	# adopted sanctions (the enforcer) three months on, with
	# scope:sanctioned_country the target. A queued event from an older save
	# has no saved target and picks one, as the rolled event used to.
	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		has_modifier = un_sanctions_enforcer_modifier
		NOT = { has_modifier = un_sanctions_enforcement_modifier }
		# A regime must actually be in force. The enforcer modifier outlives a
		# target that accepted its censure (no target modifier is applied) or
		# that no longer exists, and every option below acts on the target.
		OR = {
			scope:sanctioned_country ?= { has_modifier = un_sanctions_target_modifier }
			AND = {
