# ======================================================================
#  INHERITANCE EVENTS
#  Spec: docs/superpowers/specs/2026-10-08-inheritance-laws-design.md
#  Docs: docs/systems/mod_systems.md § Inheritance.
#
#    1       hidden: every country's score set at game start
#    8       hidden state event: a state's Wealth Concentration modifiers
#            at game start (ROOT the state, for the multiplier)
#    2-5     generation events (§5, first wave), rolled yearly from
#            te_inh_generation_* (te_inheritance_on_actions.txt) once the
#            law has been held for a generation; 25-year cooldowns
#    6-7     Posthumous Title (§6), from Mind Backups or Biological
#            Immortality, once per country
#    10-13   amendment debate events (§2), pooled in
#            on_law_checkpoint_debate (law_events_on_actions.txt). Each
#            trigger restates its amendment's `possible`, since
#            add_amendment silently does nothing when it fails. When the
#            law in force carries the amendment, the event offers to carry
#            it over to the incoming law (a law's amendments leave with it)
#
#  Options that move the concentration score call
#  te_inh_shift_concentration inside a custom_tooltip, since a variable
#  change prints nothing (silent_variable_audit).
# ======================================================================

# ======================================================================
#  1 – Game start (hidden)
#  on_game_started has no ROOT, so the score and its modifiers are set
#  here, where ROOT is the country (te_inh_on_game_started). The laws'
#  on_activate may have started the generation clock while history set
#  them; the 1836 laws are old, so it is cleared.
# ======================================================================
inheritance_events.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = te_inh_law_recent }
			remove_variable = te_inh_law_recent
		}
		te_inh_game_start = yes
		# The state modifiers' multipliers resolve against ROOT, which is the
		# country here: each state refreshes its own in .8.
		every_scope_state = {
			trigger_event = { id = inheritance_events.8 }
		}
	}
}

# ======================================================================
#  8 – Game start, per state (hidden)
#  ROOT is the state, so the state modifiers' multipliers read its own
#  score (te_inh_refresh_wc_state_effects). Fired by .1 after the states
#  are seeded; from then on each state's yearly pulse refreshes them.
# ======================================================================
inheritance_events.8 = {
	type = state_event
	hidden = yes

	immediate = {
		te_inh_refresh_wc_state_effects = yes
	}
}

		}

		add_loyalists = {
			value = very_small_radicals
			pop_type = peasants
		}
		ai_chance = {
			base = 1
		}
	}

	option = { # Sell to a neighbour
		name = inheritance_events.3.c
		custom_tooltip = {
			text = inheritance_events.3.c.tt
			te_inh_shift_concentration = { AMOUNT = 5 }
		}
		ig:ig_landowners ?= {
			add_modifier = { name = ig_approval_positive_modifier days = normal_modifier_time is_decaying = yes }
		}
		add_radicals = {
			value = very_small_radicals
			pop_type = peasants
		}
		ai_chance = {
			base = 1

			add_amendment = {
				type = amendment_undivided_farm_succession

		}

			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}

			base = 1
			modifier = {
				trigger = { scope:rural_folk_ig ?= { is_in_government = yes } }
				add = 2
			}
		}
	}

# ======================================================================
#  10 – Death Duties  (Inheritance – Amendment)
#  The scripted way in for amendment_estate_duties while a law with private
#  estates is enacted: a group that backs graduated taxation asks for a
#  duty on estates, or to keep the one the outgoing law carries. Pooled in
#  on_law_checkpoint_debate.
# ======================================================================
inheritance_events.10 = {
	type = country_event
	placement = root

	title = inheritance_events.10.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = inheritance_events.10.d_carry
				trigger = {
					active_law:lawgroup_inheritance ?= {
						has_amendment = amendment_type:amendment_estate_duties
					}
				}
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = inheritance_events.10.d
			}
		}
	}
	flavor = inheritance_events.10.f

	category = enactment

	event_image = { video = "unspecific_politicians_arguing" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	cooldown = { days = normal_modifier_time }

	trigger = {
		te_inh_enacting_private_estates_law = yes
		# amendment_estate_duties's `possible`
		has_technology_researched = political_agitation
		NOT = {
			active_law:lawgroup_inheritance ?= {
				has_amendment = amendment_type:amendment_birthright_endowment
			}
		}
		NOT = {
			currently_enacting_law = {
				OR = {
					has_amendment = amendment_type:amendment_estate_duties
					has_amendment = amendment_type:amendment_birthright_endowment
				}
			}
		}
		any_interest_group = {
			ig_counts_as_marginal = no
			OR = {
				is_interest_group_type = ig_intelligentsia
				law_stance = {
					law = law_type:law_graduated_taxation
					value > neutral
				}
			}
		}
	}

	immediate = {
		currently_enacting_law = {
			save_scope_as = current_law_scope
		}
		ordered_interest_group = {
			limit = {
				ig_counts_as_marginal = no
				OR = {
					is_interest_group_type = ig_intelligentsia
					law_stance = {
						law = law_type:law_graduated_taxation
						value > neutral
					}
				}
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = duty_ig
		}
	}

	cancellation_trigger = {
		NOT = { currently_enacting_law = scope:current_law_scope }
	}

	option = { # Write the duty into the bill
		name = inheritance_events.10.a
		trigger = { exists = scope:duty_ig }
		currently_enacting_law = {
			add_amendment = {
				type = amendment_estate_duties
				sponsor = scope:duty_ig
				cooldown = 120
			}
		}
		add_enactment_modifier = {
			name = amendment_compromise_modifier
		}
		ig:ig_landowners ?= {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
		add_radicals = {
			value = very_small_radicals
			pop_type = aristocrats
		}
	}

	option = { # Leave the estates untaxed
		name = inheritance_events.10.b
		default_option = yes
		add_enactment_modifier = {
			name = amendment_purist_modifier
		}
		scope:duty_ig ?= {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
	}
}
