# ============================================================================
# UN TEST CONSOLE — fired only from the console: `event te_debug_un.1`
# ============================================================================
# Vic3's console has no `effect` command (that is CK3); `event <id>` is the way
# to run script by hand. Nothing in the mod fires this event. It calls the
# console-only helpers in common/scripted_effects/te_debug_un_effects.txt so
# the chamber, mandates, standing and lobbying can be tried in minutes.
# ============================================================================
namespace = te_debug_un

te_debug_un.1 = { # REVIEWED 2026-09-18: console-only test event (`event te_debug_un.1`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_un.1.t
	desc = te_debug_un.1.d
	flavor = te_debug_un.1.f

	# Found the UN; seat the player and every great and major power.
	option = {
		name = te_debug_un.1.a
		default_option = yes
		te_debug_un_setup = yes
	}

	# A mandate against a neighbour, for one of its states bordering us.
	option = {
		name = te_debug_un.1.b
		trigger = {
			any_neighbouring_state = { NOT = { owner = ROOT } }
		}
		random_neighbouring_state = {
			limit = { NOT = { owner = ROOT } }
			save_scope_as = te_debug_mnd_state
			owner = { save_scope_as = te_debug_mnd_target }
		}
		if = {
			limit = {
				exists = scope:te_debug_mnd_state
				exists = scope:te_debug_mnd_target
			}
			te_debug_un_grant_mandate = {
				TARGET = scope:te_debug_mnd_target
				REGION = scope:te_debug_mnd_state.state_region
			}
		}
	}

	# Another member tables a condemnation of a third member, so the player can
	# vote on it from the chamber at once (binding: a permanent member also gets
	# the veto). Replaces whatever is on the floor.
	option = {
		name = te_debug_un.1.g
		trigger = {
			any_country = {
				count >= 2
				NOT = { this = ROOT }
				je:je_united_nations ?= { has_modifier = un_member_modifier }
			}
		}
		random_country = {
			limit = {
				NOT = { this = ROOT }
				je:je_united_nations ?= { has_modifier = un_member_modifier }
			}
			save_scope_as = te_debug_res_proposer
		}
		random_country = {
			limit = {
				NOT = { this = ROOT }
				NOT = { this = scope:te_debug_res_proposer }
				je:je_united_nations ?= { has_modifier = un_member_modifier }
			}
			save_scope_as = te_debug_res_target
		}
		if = {
			limit = {
				exists = scope:te_debug_res_proposer
				exists = scope:te_debug_res_target
			}
			te_debug_un_open_resolution = {
				PROPOSER = scope:te_debug_res_proposer
				TOPIC = condemn
				TARGET = scope:te_debug_res_target
			}
		}
	}

	# Throw away whatever is on the floor and give back the vote lock.
	option = {
		name = te_debug_un.1.h
		trigger = { has_global_variable = un_vote_active }
		te_debug_un_clear_vote = yes
	}

	option = {
		name = te_debug_un.1.c
		te_debug_un_set_standing = { VALUE = 85 }
	}

	option = {
		name = te_debug_un.1.e
		te_debug_un_set_standing = { VALUE = 10 }
	}

	option = {
		name = te_debug_un.1.z
	}
}
