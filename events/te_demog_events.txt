namespace = te_demog_events

# ======================================================================
#  DEMOGRAPHICS EVENTS
#  Spec: docs/superpowers/specs/2026-10-08-demographics-design.md
#  Effects: common/scripted_effects/te_demog_effects.txt.
#
#    1       hidden: every state's first census at game start
#    2-3     hidden: fast mode's census clock (console only): each state's
#            step, then each country's census a day later
#    4       hidden: a state's engine lines refreshed (law, technology,
#            new game)
#    5       hidden: .4 again a day later in each of a country's states
# ======================================================================

# ======================================================================
#  1 – Game start (hidden)
#  on_game_started has no ROOT, so the census is seeded here, where ROOT
#  is the country (te_demog_on_game_started).
# ======================================================================
te_demog_events.1 = {
	type = country_event
	hidden = yes

	immediate = {
		te_demog_country_game_start = yes
	}
}

# ======================================================================
#  2 – Fast mode: a state's step (hidden)
#  The census clock's step (te_demog_clock_tick, te_demog_fast_effects.txt;
#  started only from the console) fires this in every state, so ROOT is the
#  state and the fast rate modifiers' multipliers read it.
# ======================================================================
te_demog_events.2 = {
	type = state_event
	hidden = yes

	trigger = {
		te_demog_clock_on = yes
	}

	immediate = {
		te_demog_state_clock_step = yes
	}
}

# ======================================================================
#  3 – Fast mode: a country's census (hidden)
#  A day after the clock's step, once every state has stepped; ROOT is the
#  country (te_demog_country_census writes its bands to it).
# ======================================================================
te_demog_events.3 = {
	type = country_event
	hidden = yes

	trigger = { # REVIEWED 2026-11-13 (delayed_trigger): te_demog_clock_tick runs only while the clock's global is set; a filter
		te_demog_clock_on = yes
	}

	immediate = {
		te_demog_country_clock_census = yes
	}
}

# ======================================================================
#  4 – The engine's moved birth and mortality lines (hidden)
#  Fired in each state by the law, technology and global hooks (te_demog_on_actions.txt: a law's
#  activation, a technology, a new game), so ROOT is the state, as te_demog_lines_refresh's multipliers need.
# ======================================================================
te_demog_events.4 = {
	type = state_event
	hidden = yes

	immediate = {
		te_demog_lines_refresh = yes
		te_demog_workforce_lines_refresh = yes
		te_demog_workforce_heal = yes
		te_demog_rates_heal = yes
	}
}

# ======================================================================
#  5 – The same a day later, through the country (hidden)
#  The law and technology hooks refresh each state again a day on, in case the engine hadn't re-summed the families
#  on the day. A state event triggered with a delay never runs (the delayed_state_event self-check, BROKEN in the
#  2026-11-08 new game), so the day waits on this country event, which runs .4 in each of its states undelayed.
# ======================================================================
te_demog_events.5 = {
	type = country_event
	hidden = yes

	immediate = {
		every_scope_state = {
			trigger_event = { id = te_demog_events.4 }
		}
	}
}
