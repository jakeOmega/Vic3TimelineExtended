# mean_time_to_happen nor a random_events registration is reached only by
# trigger_event anyway (see events/te_tax_internal_events.txt).
# ============================================================================
namespace = te_grid

# The monthly balancing of every grid. Firing it again in the same month
# (console: event te_grid.1) balances again from the flows just set, which
# moves them the damping's step further toward their targets.
te_grid.1 = {
	type = country_event
	hidden = yes

	immediate = {
		te_grid_world_update_effect = yes
	}
}
