# ============================================================================
# SPACE RACE TEST CONSOLE — fired only from the console: `event te_debug_space_race.1`
# ============================================================================
# Vic3's console has no `effect` command (that is CK3); `event <id>` is the way
# to run script by hand. Nothing in the mod fires this event. It calls the
# console-only helpers in
# common/scripted_effects/te_debug_space_race_effects.txt so every state the
# milestone panel can show can be reached in a couple of clicks.
# ============================================================================
namespace = te_debug_space_race

	title = te_debug_space_race.1.t

	flavor = te_debug_space_race.1.f

	# Two other powers report the same milestone within the notification delay.

		name = te_debug_space_race.1.k

		name = te_debug_space_race.1.e

		te_debug_sr_clear_cooldown = yes

		name = te_debug_space_race.1.g
		te_debug_sr_seed_rivals = yes
	}

	# Take "the first" off the table for every milestone we have not won.
	option = {
		name = te_debug_space_race.1.h
		te_debug_sr_claim_firsts = yes
	}

	# Give it back.
	option = {
		name = te_debug_space_race.1.i
		te_debug_sr_reset_globals = yes
	}

	# No programme, to try the Found a Space Program decision.
	option = {
		name = te_debug_space_race.1.l
		trigger = { sr_has_space_program = yes }
		te_debug_sr_remove_program = yes
	}

	option = {
		name = te_debug_space_race.1.z
	}
}

	title = te_debug_space_race.2.t
	desc = te_debug_space_race.2.d
	flavor = te_debug_space_race.2.f

	# Every world's record, one TE_SR_SITES line each.
	option = {
		name = te_debug_space_race.2.a
		default_option = yes
		sr_sites_debug_log = yes
	}

	# The first unclaimed world, ours, with its first specialization.
	option = {
		name = te_debug_space_race.2.b
		trigger = { sr_any_site_unclaimed = yes }
		sr_sites_debug_claim_first_free = yes
	}

	# Our first colony to the largest other great power, through Cede Colony's
	# own transfer (the notice and the receiver's entry included).
	option = {
		name = te_debug_space_race.2.c
		trigger = { sr_has_space_colony = yes }
		ordered_country = {
			limit = {
				NOT = { this = root }
				is_country_alive = yes
				is_great_or_major_power = yes
			}
			order_by = gdp
			max = 1
			save_scope_as = sr_debug_receiver
		}
		if = {
			limit = { exists = scope:sr_debug_receiver }
			sr_sites_debug_give_first_held = yes
		}
	}

	# Our first colony released, as a dead holder with no heir would leave it.
	option = {
		name = te_debug_space_race.2.r
		trigger = { sr_has_space_colony = yes }
		sr_sites_debug_release_first_held = yes
	}

	# The monthly step now (migration on its first run, the dead-holder rule,
	# adoption or release, the projection, every player's rows).
	option = {
		name = te_debug_space_race.2.e
		sr_sites_global_monthly = yes
	}
}

# The retired Moon Base's migration (lunar colonization design § 10;
# sr_lunar_migrate, sr_site_effects.txt): its self-check again, or an old
# save's state staged in a new game, so the migration and the carry can be
# watched without one.
te_debug_space_race.8 = { # REVIEWED 2026-10-29: console-only test event (`event te_debug_space_race.8`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/lunar_base_dome.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	title = te_debug_space_race.8.t
	desc = te_debug_space_race.8.d
	flavor = te_debug_space_race.8.f

	# The lunar_migration self-check now, whatever it logged before:
	# TE_SELF_CHECK OK or BROKEN lunar_migration. Read after a real migration:
	# later, a country that founded a lunar colony and ceded it reads BROKEN.
	option = {
		name = te_debug_space_race.8.a
		default_option = yes
		sr_lunar_selfcheck_migration = yes
	}

	# As if we had finished the old Moon Base as a scientific outpost: the
	# migration runs again next month and gives us Daedalus Crater (or the next
	# free lunar world). It skips any country already holding one.
	option = {
		name = te_debug_space_race.8.b
		trigger = { NOT = { sr_holds_lunar_world = yes } }
		set_variable = { name = sr_completed_moon_base value = yes }
		if = {
			limit = { NOT = { sr_moon_base_choice_made = yes } }
			set_variable = { name = sr_moon_base_scientific value = yes }
		}
		if = {
			limit = { has_global_variable = sr_lunar_initialized }
			remove_global_variable = sr_lunar_initialized
		}
	}

	# As if an old Moon Base were open with 120 progress: next month's pulse
	# carries it onto Solar System Colonization (TE_SELF_CHECK OK lunar_migration
	# carried=120), and the programme's next launch month adds it to the bar.
	option = {
		name = te_debug_space_race.8.c
		trigger = { NOT = { has_variable = sr_completed_moon_base } }
		set_variable = { name = sr_progress_moon_base value = 120 }
		set_variable = { name = sr_funding_moon_base value = 1 }
		if = {
			limit = { has_global_variable = sr_lunar_carry_ok_logged }
			remove_global_variable = sr_lunar_carry_ok_logged
		}
		# the carry logs OK only on a country's first carry (review M3)
		if = {
			limit = { has_variable = sr_lunar_carried }
			remove_variable = sr_lunar_carried
		}
		if = {
			limit = { has_variable = sr_lunar_carry_grace }
			remove_variable = sr_lunar_carry_grace
		}
	}
}
