namespace = un_vote

# ============================================================================
# UN GENERAL ASSEMBLY VOTING EVENTS
# Three-phase vote on a UN resolution.
#
# Each resolution is a script container (tags, variables and lifecycle in
# common/scripted_effects/un_vote_effects.txt). un_resolution_open creates it
# from a proposer event (un_events) or a JE button, and every event here gets
# it as scope:un_resolution.
#
# Phase 1: un_vote.1 fires to every other HUMAN member 30 days later; the AI
#          votes in script by the itemised lean (un_vote.4, fired by un_vote.5
#          at 30 days, redesign phase 3). Votes go into the resolution's
#          un_res_yes / un_res_no lists.
# Phase 2: un_vote.2 fires to the proposer 365 days after opening. Its
#          immediate closes the vote; its option applies the outcome.
# Phase 3: un_vote.3 notifies every other member, who comply or refuse.
#
# Vote topics (tag un_topic_<topic> on the resolution):
#   condemn               - Condemn aggressor (event 2 / button)
#   human_rights          - Human Rights Declaration (event 3)
#   reform                - UN Institutional Reform (event 6)
#   heritage              - Cultural Heritage Program (event 9)
#   decolonization        - Decolonization Resolution (event 12)
#   npt                   - Non-Proliferation Treaty (event 14)
#   pandemic              - Global Pandemic Response (event 16)
#   climate               - Climate Accord (event 17)
#   refugee               - International Refugee Crisis (event 18)
#   space                 - International Space Cooperation (event 19)
#   law_of_sea            - Convention on the Law of the Sea (event 21)
#   icc                   - International Criminal Court (event 22)
#   peacekeeping_request  - Peacekeeping deployment (request button)
#   aid_request           - Humanitarian aid (request button)
#   sanctions             - International sanctions (sanctions button)
#   expulsion             - Strip a permanent member's seat (expulsion button)
#   military_mandate      - Authorize one member to recover one claimed state
#                           region from one censured state (mandate button).
#                           Binding and vetoable, with no graduated form: a
#                           veto blocks the authorization outright, exactly as
#                           it does for reform. The resolution carries two
#                           extra variables, un_res_mandate_region and
#                           un_res_mandate_beneficiary; carrying it un-vetoed
#                           creates the mandate container
#                           (common/scripted_effects/un_mandate_effects.txt).
# ============================================================================

					}
					hidden_effect = {
						scope:un_vote_target = { save_scope_as = sanctioned_country }
						trigger_event = { id = un_events.5 days = 90 }
					}
				}
				# Apply sanctions to target
				if = {
					limit = { exists = scope:un_vote_target }
					scope:un_vote_target = {
						if = {
							limit = { NOT = { scope:un_resolution ?= { has_tag = un_res_target_accepted } } }
							# Scaled by the UN's enforcement (redesign phase 2).
							# un_enforcement reads only globals, so the stored
							# multiplier is sound whenever the engine re-reads it.
							add_modifier = {
								name = un_sanctions_target_modifier
								multiplier = un_enforcement
							}
							# Teeth by tier (redesign phase 5, §5.2): from Strong
							# the members who voted for it embargo the target; at
							# Supranational every major-power member does.
							if = {
								limit = { un_tier_is_supranational = yes }
								custom_tooltip = un_teeth_embargo_all_tt
								hidden_effect = { un_teeth_sanctions_embargo = yes }
							}
							else_if = {
								limit = { un_tier_at_least_strong = yes }
								custom_tooltip = un_teeth_embargo_voters_tt
								hidden_effect = { un_teeth_sanctions_embargo = yes }
							}
							custom_tooltip = {
								text = un_standing_sanctioned_tt
								un_standing_loss = { AMOUNT = un_standing_sanctioned_loss REASON = 24 }
							}
						}
						else = {
							custom_tooltip = {
								text = un_standing_sanctioned_accepted_tt
								un_standing_loss = { AMOUNT = un_standing_sanctioned_partial_loss REASON = 24 }
							}
						}
					}
					change_relations = { country = scope:un_vote_target value = -30 }
				}
			}
			else_if = {
				limit = { scope:un_resolution ?= { has_tag = un_topic_sanctions } }
				custom_tooltip = un_vote_sanctions_passed_vetoed_tt
				un_ledger_entry = { PILLAR = credibility POINTS = 0.5 REASON = 2 }
				if = {
					limit = { exists = scope:un_vote_target }
					scope:un_vote_target = {
						if = {
							limit = { NOT = { scope:un_resolution ?= { has_tag = un_res_target_accepted } } }
							add_modifier = {
								name = un_sanctions_voluntary_target_modifier
								multiplier = un_enforcement
								days = normal_modifier_time
								is_decaying = yes
							}
							custom_tooltip = {
								text = un_standing_sanctioned_partial_tt
								un_standing_loss = { AMOUNT = un_standing_sanctioned_partial_loss REASON = 25 }
							}
						}
					}
					change_relations = { country = scope:un_vote_target value = -10 }
				}
			}

			# ---- Law of the Sea: Maritime cooperation framework ----
			if = {
				limit = { scope:un_resolution ?= { has_tag = un_topic_law_of_sea } }
				custom_tooltip = un_vote_law_of_sea_passed_tt
				un_ledger_entry = { PILLAR = credibility POINTS = 1 REASON = 1 }
				set_global_variable = un_agency_itlos
			}

			# ---- Military Mandate: authorize one narrow objective ----
			# Like reform, this binding topic has NO graduated form. There is
			# no weaker version of a licence to use force, so a veto blocks it
			# outright and no mandate is issued. un_mandate_create re-validates
			# before issuing: a year has passed since the vote opened, and the
			# target may have lost the territory, we may have left the UN, or
			# we may already hold another authorization — in any of those cases
			# the resolution carries but grants nothing.
			if = {
				limit = {
					scope:un_resolution ?= { has_tag = un_topic_military_mandate }
					NOT = { scope:un_resolution ?= { has_tag = un_res_vetoed } }
				}
				custom_tooltip = un_vote_military_mandate_passed_tt
				un_ledger_entry = { PILLAR = credibility POINTS = 1 REASON = 1 }
				# Hidden: create_container / set_variable have no player-facing
				# tooltip worth showing, and letting them render would push the
				# container's internals in under the custom_tooltip above. Same
				# reasoning as the un_vote_apply_treaty_modifier split — this
				# file controls what the option tooltip says.
				hidden_effect = {
					un_mandate_create = yes
				}
			}
			else_if = {
				limit = { scope:un_resolution ?= { has_tag = un_topic_military_mandate } }
				custom_tooltip = un_vote_military_mandate_passed_vetoed_tt
				# No effect — flat block, as with reform.
			}

			# ---- International Criminal Court: Global criminal justice ----
			# Vetoed: no agency established, just a symbolic censure (proposer
			# gets un_court_contested_modifier as a small accountability gesture).
			if = {
				limit = {
					scope:un_resolution ?= { has_tag = un_topic_icc }
					NOT = { scope:un_resolution ?= { has_tag = un_res_vetoed } }
				}
				custom_tooltip = un_vote_icc_passed_tt
				un_ledger_entry = { PILLAR = credibility POINTS = 1 REASON = 1 }
				set_global_variable = un_agency_icc
			}
			else_if = {
				limit = { scope:un_resolution ?= { has_tag = un_topic_icc } }
				custom_tooltip = un_vote_icc_passed_vetoed_tt
				un_ledger_entry = { PILLAR = credibility POINTS = 0.5 REASON = 2 }
				add_modifier = {
					name = un_court_contested_modifier
					days = normal_modifier_time
					is_decaying = yes
				}
			}

			# The proposer ratifies on passage, like every other member
			# (redesign phase 5, §5.3): it no longer takes the convention's
			# modifier up front when it tables the resolution. After the
			# per-topic branches above, so the agency they found is in force
			# when the convention's terms are read.
			un_vote_apply_treaty_modifier = yes

			# Influence reward for leading a successful resolution.
			# Skipped for vetoed binding resolutions — only the graduated form
			# went through, so the proposer didn't fully succeed.
			if = {
				limit = { NOT = { scope:un_resolution ?= { has_tag = un_res_vetoed } } }
				add_modifier = {
					name = un_vote_success_reward
					days = normal_modifier_time
					is_decaying = yes
				}
			}

			# Infamy reduction: successfully leading a humanitarian resolution rehabilitates image
			if = {
				limit = {
					OR = {
						scope:un_resolution ?= { has_tag = un_topic_human_rights }
						scope:un_resolution ?= { has_tag = un_topic_refugee }
						scope:un_resolution ?= { has_tag = un_topic_pandemic }
						scope:un_resolution ?= { has_tag = un_topic_aid_request }
						scope:un_resolution ?= { has_tag = un_topic_peacekeeping_request }
					}
					infamy > 0
				}
				change_infamy = -3
			}

			# Power bloc leverage: successful proposer gains leverage over bloc members who voted
			if = {
				limit = { is_power_bloc_leader = yes }
				power_bloc ?= {
					every_power_bloc_member = {
						limit = {
							je:je_united_nations ?= { has_modifier = un_member_modifier }
							NOT = { THIS = ROOT }
						}
						root = {
							power_bloc ?= {
								add_leverage = { target = prev value = 3 }
							}
						}
					}
				}
			}
		}

		# ---- RESOLUTION FAILED ----
		if = {
			limit = {
				scope:un_resolution ?= {
					has_tag = un_res_failed
					NOT = { has_tag = un_topic_expulsion }
				}
			}
			custom_tooltip = un_vote_failed_tt
			un_ledger_entry = { PILLAR = credibility POINTS = -1 REASON = 3 }
			add_modifier = {
				name = un_vote_failure_penalty
				days = normal_modifier_time
				is_decaying = yes
			}
			# Embarrassment: failed humanitarian resolution costs credibility
			if = {
				limit = {
					OR = {
						scope:un_resolution ?= { has_tag = un_topic_human_rights }
						scope:un_resolution ?= { has_tag = un_topic_refugee }
						scope:un_resolution ?= { has_tag = un_topic_pandemic }
					}
				}
				change_infamy = 2
			}
			# Power bloc leverage loss: failed proposer loses credibility with bloc
			if = {
				limit = { is_power_bloc_leader = yes }
				power_bloc ?= {
					every_power_bloc_member = {
						limit = {
							je:je_united_nations ?= { has_modifier = un_member_modifier }
							NOT = { THIS = ROOT }
						}
						root = {
							power_bloc ?= {
								add_leverage = { target = prev value = -3 }
							}
						}
					}
				}
			}
			# Revoke unfulfilled pledges from voters
			hidden_effect = {
				if = {
					limit = { scope:un_resolution ?= { has_tag = un_topic_peacekeeping_request } }
					every_country = {
						limit = {
							je:je_united_nations ?= { has_modifier = un_peacekeeping_contributor_modifier }
							NOT = { je:je_united_nations ?= { has_modifier = un_peacekeeping_contributor_cost } }
						}
						je:je_united_nations ?= { remove_modifier = un_peacekeeping_contributor_modifier }
					}
				}
				if = {
					limit = { scope:un_resolution ?= { has_tag = un_topic_aid_request } }
					every_country = {
						limit = {
							has_modifier = un_humanitarian_aid_modifier
							NOT = { has_modifier = un_humanitarian_aid_cost }
						}
						remove_modifier = un_humanitarian_aid_modifier
					}
				}
			}
		}

		# ---- Notify all other UN members of the result ----
		hidden_effect = {
			every_country = {
				limit = {
					je:je_united_nations ?= { has_modifier = un_member_modifier }
					NOT = { THIS = ROOT }
				}
				trigger_event = { id = un_vote.3 }
			}
		}

		ai_chance = { base = 100 }
	}
}

# ---- Phase 3: Notification to all other UN members ----
un_vote.3 = {
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = un_vote.3.t

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:un_resolution ?= { has_tag = un_res_passed } }
				desc = un_vote.3.d_passed
			}
			desc = un_vote.3.d_failed
		}
		# We voted against a resolution that carried: say what options B and C
		# actually decide, which depends on the kind of resolution. A
		# convention binds only the members that ratify it; a request for aid
		# or peacekeepers asks each member to take part; anything else is a
		# decision of the whole Assembly that stands whatever we answer.
		first_valid = {
			triggered_desc = {
				trigger = { un_vote_3_we_opposed_passed = { KIND = convention } }
				desc = un_vote.3.d_choice_convention
			}
			triggered_desc = {
				trigger = { un_vote_3_we_opposed_passed = { KIND = pledge } }
				desc = un_vote.3.d_choice_pledge
			}
			triggered_desc = {
				trigger = { un_vote_3_we_opposed_passed = { KIND = decision } }
				desc = un_vote.3.d_choice_decision
			}
		}
	}

	duration = 3

	# scope:un_resolution is the closed resolution, passed on from un_vote.2.
	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		exists = scope:un_resolution
	}

	immediate = {
		# Vote counts as country vars for loc display (loc has no accessor
		# for a script container held in a saved scope)
		set_variable = {
			name = un_display_support
			value = scope:un_resolution.var:un_res_support
		}
		set_variable = {
			name = un_display_oppose
			value = scope:un_resolution.var:un_res_oppose
		}
		scope:un_resolution = {
			var:un_res_proposer ?= { save_scope_as = un_vote_proposer }
			var:un_res_target ?= { save_scope_as = un_vote_target }
		}
	}

	# Option A: Catch-all "acknowledge" — vote failed, you voted yes, or you
	# joined the UN after vote.1 fired and never had a chance to vote. In the
	# last case the country still gets bound by the passed resolution (default
	# acceptance) since the apply blocks below already gate on the passed tag.
	option = {
		name = un_vote.3.a
		default_option = yes
		trigger = {
			OR = {
				NOT = { scope:un_resolution ?= { has_tag = un_res_passed } }
				NOT = { scope:un_resolution ?= { un_resolution_has_vote = { LIST = un_res_no } } }
			}
		}

		# Apply the persistent treaty/declaration modifier directly in the
		# option block so the engine renders each add_modifier in the option
		# tooltip — players can see exactly what they're ratifying before
		# clicking acknowledge. (Previously these were buried inside
		# un_vote_apply_resolution_compliance, where the tooltip nesting
		# made the modifier name and fields easy to miss.)
		if = {
			limit = { scope:un_resolution ?= { has_tag = un_res_passed } }
			un_vote_apply_treaty_modifier = yes
			un_vote_apply_resolution_compliance = yes
		}
		ai_chance = { base = 100 }
	}

	# Option B: Vote passed AND you voted no - accept the resolution anyway
	option = {
		name = {
			trigger = { scope:un_resolution ?= { un_resolution_is_convention = yes } }
			text = un_vote.3.b_convention
		}
		name = {
			trigger = { always = yes }
			text = un_vote.3.b
		}
		trigger = {
			scope:un_resolution ?= {
				has_tag = un_res_passed
				un_resolution_has_vote = { LIST = un_res_no }
			}
		}

		# Same effects as yes voters - you accept the majority decision
		un_vote_apply_treaty_modifier = yes
		un_vote_apply_resolution_compliance = yes

		# International standing: complying with a BINDING decision you voted
		# against is the one act in this event that is not merely voting — the
		# member is bound by something it argued against and submits anyway.
		# Voting with the majority pays nothing (option A above is untouched).
		# Restricted to binding resolutions because only those bind: yielding
		# on a recommendation costs nothing and so earns nothing.
		# Once per member per resolution — un_vote.3 fires once per member, and
		# an event runs exactly one option.
		#
		# peacekeeping_request is excluded because it is the ONE topic that is
		# both binding and a pledge topic: un_vote_apply_resolution_compliance
		# just paid this country the contribution award for taking on a share
		# of the deployment, and paying the compliance award on top would make
		# a no-voter who complies worth twice a yes-voter who does the same
		# thing. The pledge is the compliance, and it has already been paid.
		if = {
			limit = {
				scope:un_resolution ?= { un_resolution_is_binding = yes }
				NOT = { scope:un_resolution ?= { has_tag = un_topic_peacekeeping_request } }
			}
			custom_tooltip = {
				text = un_standing_complied_tt
				un_standing_gain = { AMOUNT = un_standing_compliance_gain REASON = 5 }
			}
		}
		ai_chance = { base = 50 }
	}

	# Option C: Vote passed AND you voted no - refuse to comply. What that
	# means depends on the resolution, and the name and first tooltip line say
	# which: stay outside a convention, take no part in a request for aid or
	# peacekeepers, or — for every decision of the whole Assembly (a mandate,
	# sanctions, a condemnation, an expulsion, a reform) — denounce it on the
	# record. Refusing never stops or weakens the resolution itself; the
	# effects below are the whole of it.
	option = {
		name = {
			trigger = { scope:un_resolution ?= { un_resolution_is_convention = yes } }
			text = un_vote.3.c_convention
		}
		name = {
			trigger = { scope:un_resolution ?= { un_resolution_target_is_beneficiary = yes } }
			text = un_vote.3.c_pledge
		}
		name = {
			trigger = { always = yes }
			text = un_vote.3.c_decision
		}
		trigger = {
			scope:un_resolution ?= {
				has_tag = un_res_passed
				un_resolution_has_vote = { LIST = un_res_no }
			}
		}

		if = {
			limit = { scope:un_resolution ?= { un_resolution_is_convention = yes } }
			custom_tooltip = un_vote.3.c_convention_tt
		}
		else_if = {
			limit = { scope:un_resolution ?= { un_resolution_target_is_beneficiary = yes } }
			custom_tooltip = un_vote.3.c_pledge_tt
		}
		else = {
			custom_tooltip = un_vote.3.c_decision_tt
		}

		# Diplomatic penalties for defying the international community
		if = {
			limit = { exists = scope:un_vote_proposer }
			change_relations = { country = scope:un_vote_proposer value = -20 }
			create_diplomatic_catalyst = { type = catalyst_event_negative target = scope:un_vote_proposer }
		}
		change_infamy = 3

		# International standing: openly refusing a BINDING decision is the
		# defiance the brief asks to punish. Refusing a recommendatory one is
		# legal dissent — the same reasoning that leaves the veto free — so it
		# costs nothing here.
		if = {
			limit = { scope:un_resolution ?= { un_resolution_is_binding = yes } }
			custom_tooltip = {
				text = un_standing_defied_tt
				un_standing_loss = { AMOUNT = un_standing_defiance_loss REASON = 26 }
			}
			# The dossier (redesign phase 3): refusing a binding decision is
			# a record the Assembly can later hold against us.
			custom_tooltip = un_dossier_tt_defiance
			un_dossier_record = { RECORD = defiance POINTS = un_dos_points_defiance }
		}

		# Power bloc leverage: defying bloc decision
		if = {
			limit = {
				exists = scope:un_vote_proposer
				scope:un_vote_proposer = { is_power_bloc_leader = yes }
				is_in_same_power_bloc = scope:un_vote_proposer
			}
			scope:un_vote_proposer = {
				power_bloc ?= {
					add_leverage = { target = root value = -15 }
				}
			}
		}

		ai_chance = {
			base = 20
			# Countries that are rivals of proposer more likely to refuse
			modifier = {
				trigger = {
					exists = scope:un_vote_proposer
					has_diplomatic_pact = {
						who = scope:un_vote_proposer
						type = rivalry
					}
				}
				add = 40
			}
			# Great powers refuse more — sovereignty concerns
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 20
			}
			# NPT: nuclear powers refuse rather than disarm
			modifier = {
				trigger = {
					scope:un_resolution ?= { has_tag = un_topic_npt }
					has_variable = nuclear_weapon_stockpile
					var:nuclear_weapon_stockpile >= 1
				}
				add = 50
			}
			# Climate: heavy industry nations refuse binding targets
			modifier = {
				trigger = {
					scope:un_resolution ?= { has_tag = un_topic_climate }
					ig:ig_industrialists ?= {
						is_powerful = yes
					}
				}
				add = 25
			}
		}
	}
}

# ---- The ballot in script (redesign phase 3) ----
# docs/systems/un_redesign_design.md §6.2. The AI no longer answers un_vote.1:
# it votes here, by the itemised lean (common/script_values/un_dossier_values.txt),
# so the chamber's explanation of a vote is the rule that decided it.

# One member's ballot. Hidden. Fired by un_vote_dispatch_ai_ballots to every AI
# member (which then casts), and by un_vote_refresh_human_leans to every human
# member (which only has its lean written, for the chamber). ROOT is the voter,
# which the relations terms of the lean need.
# Both senders save scope:un_resolution first. If it did not reach the event,
# the open resolution (global_var:un_active_resolution) stands in, so a lost
# scope costs the chamber nothing and the AI still votes.
un_vote.4 = {
	type = country_event
	hidden = yes

	trigger = {
		je:je_united_nations ?= { has_modifier = un_member_modifier }
		OR = {
			scope:un_resolution ?= { has_tag = un_res_voting }
			AND = {
				NOT = { exists = scope:un_resolution }
				global_var:un_active_resolution ?= { has_tag = un_res_voting }
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:un_resolution } }
			global_var:un_active_resolution = { save_scope_as = un_resolution }
		}
		scope:un_resolution = {
			var:un_res_proposer ?= { save_scope_as = un_vote_proposer }
			var:un_res_target ?= { save_scope_as = un_vote_target }
		}
		un_lean_snapshot = yes
		if = {
			limit = {
				is_ai = yes
				un_vote_can_cast_ballot = yes
			}
			un_vote_ai_cast = yes
		}
	}
}

# Thirty days after a resolution opens: send the AI members their ballots.
# Hidden; fired to the proposer by un_resolution_open. If the proposer is gone
# the monthly pulse sends them instead (un_resolutions_monthly_update). The
# same stand-in as un_vote.4 if scope:un_resolution did not arrive.
un_vote.5 = {
	type = country_event
	hidden = yes

	trigger = {
		OR = {
			scope:un_resolution ?= {
				has_tag = un_res_voting
				NOT = { has_variable = un_res_ai_cast }
			}
			AND = {
				NOT = { exists = scope:un_resolution }
				global_var:un_active_resolution ?= {
					has_tag = un_res_voting
					NOT = { has_variable = un_res_ai_cast }
				}
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:un_resolution } }
			global_var:un_active_resolution = { save_scope_as = un_resolution }
		}
		un_vote_dispatch_ai_ballots = yes
	}
}
