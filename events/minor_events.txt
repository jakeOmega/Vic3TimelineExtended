	# country variable contagion_crash: a second copy open at once overwrites that with its own roll. The
	# engine reads the title when the event is queued, before its `immediate`. Since the roll moved ahead of
	# it (the hidden event .9, banking_contagion_decide), a crash has saved contagion_crashed by then and
	# the title reads right; a copy an older build sent holds no saved scope, or the upstream country's,
	# which is never ROOT, so it reads the plain title. If the engine also keeps that early title there, a

	title = {
		first_valid = {
			triggered_desc = { trigger = { banking_contagion_own_roll_crashed = yes } desc = minor_events_timelineextended.7.t_cascade }

			triggered_desc = { trigger = { banking_contagion_own_roll_crashed = yes scope:banking_crash_origin_country = { var:crash_severity >= 60 } } desc = minor_events_timelineextended.7.d_cascade_severe }
			triggered_desc = { trigger = { banking_contagion_own_roll_crashed = yes } desc = minor_events_timelineextended.7.d_cascade }
			triggered_desc = { trigger = { scope:banking_crash_origin_country = { var:crash_severity >= 80 } } desc = minor_events_timelineextended.7.d_systemic }
			triggered_desc = { trigger = { scope:banking_crash_origin_country = { var:crash_severity >= 60 } } desc = minor_events_timelineextended.7.d_panic }

	immediate = {

		banking_contagion_selfcheck_open = yes
		# The roll was made ahead of this event, by the hidden event .9 that sent it
		# (banking_contagion_decide), so a scare inside the two-year cooldown never opens. A copy an
		# older build sent has not rolled (no contagion_crash): it rolls here, as before.
		if = {
			limit = { NOT = { has_variable = contagion_crash } }
			banking_contagion_crash_check = yes

		# origin's crash in the same pulse re-runs banking_crisis_wave_init, which clears the reached marks), and the
		# first one's `after` removes contagion_crash before the second resolves, so the text, options and `after`
		# read this saved scope, not the variable (observer run, 2026-10-12: three unset reads in a second). It is
		# the country, and banking_contagion_own_roll_crashed asks that it be this one: an inherited scope is the
		# upstream country's, and clear_saved_scope did not remove it (2026-10-16, the European Union's scare read
		# as a crash). The save below does replace an inherited one: in the 2026-10-17 observer run's autosave,
		# cascade targets two and three tiers out had taken the crash-only options.

		banking_contagion_selfcheck_close = yes
		# Cascade: if this event's own roll crashed the country, spread contagion from it. Not
		# `exists = scope:contagion_crashed`: a target reached by a cascade inherits the upstream country's.
		if = {
			limit = { banking_contagion_own_roll_crashed = yes }
			# Track this crash in the crisis wave (Great Depression mechanic)
			banking_crisis_wave_increment = yes
			save_scope_as = banking_crash_origin_country
			banking_cycle_spread_contagion = yes
			# Check if enough countries have crashed for Great Depression
			banking_crisis_wave_check_great_depression = yes
		}
		banking_cycle_post_event_refresh = yes
		# The outcome is this event's alone. Removed every time, even while another copy is open: the
		# self-check takes a missing value to mean a copy opened and closed meanwhile. Guarded: a copy sent
		# while this one held no mark (an older build opened it) clears its leftovers (banking_contagion_clear_stale).
		if = {
			limit = { has_variable = contagion_crash }
			remove_variable = contagion_crash
		}
		# Answered: the spread may send this country a contagion event again.
		if = {
			limit = { has_variable = banking_contagion_pending }
			remove_variable = banking_contagion_pending
		}
	}
}

# Great Depression event: fired for all countries when a cascading crash
# wave hits 10+ countries covering 50%+ of world GDP — at most once per game
# (great_depression_occurred, set in banking_crisis_wave_check_great_depression).
# Mostly notificational with long-lasting momentum penalty.
# Isolated countries can avoid the worst effects.
# banking_crisis_wave_crashed marks a country whose banks crashed in the wave,
# and banking_crisis_wave_scared one the wave reached that only took a scare
# (see banking_crisis_wave_init). A country reached but not yet decided — its
# contagion event fires seven days after it was reached, this one three days
# after the threshold — carries only banking_crisis_wave_hit: it reads the
# general text and takes the bystander (or isolated) option, as a country the
# wave never reached does. Most contagion rolls are scares, so that is the
# likelier outcome; before #428 an undecided country was told it had crashed.

	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/foreign_investment_opportunity.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	title = {
		first_valid = {
			triggered_desc = { trigger = { has_variable = banking_crisis_wave_crashed } desc = minor_events_timelineextended.8.t_crashed }
			triggered_desc = { trigger = { always = yes } desc = minor_events_timelineextended.8.t }
		}
	}
	desc = {
		first_valid = {
			triggered_desc = { trigger = { has_variable = banking_crisis_wave_crashed } desc = minor_events_timelineextended.8.d_crashed }
			triggered_desc = {
				trigger = {
					OR = {
						has_law = law_type:law_isolationism
						has_law = law_type:law_command_economy
					}
				}
				desc = minor_events_timelineextended.8.d_isolated
			}
			# The contagion reached us and our banks held: only a scare (#428).
			# _scared, not _hit: a country reached in the last few days has not
			# had its own contagion roll yet, and reads the general text below.
			triggered_desc = { trigger = { has_variable = banking_crisis_wave_scared } desc = minor_events_timelineextended.8.d_scare }
			triggered_desc = { trigger = { always = yes } desc = minor_events_timelineextended.8.d }

	flavor = {

			triggered_desc = { trigger = { has_variable = banking_crisis_wave_crashed } desc = minor_events_timelineextended.8.f_crashed }

	}

			value = small_radicals
			strata = middle
		}
		ai_chance = {
			base = 100
		}
	}

	# Option B: Isolated/command economy countries — largely shielded
	option = {
		name = minor_events_timelineextended.8.b
		trigger = {
			NOT = { has_variable = banking_crisis_wave_crashed }
