############################################
# Space Race Colony Events
# Staged solar system colonization events
# Stage 1: Mars & Asteroids (.1-.10)
# Stage 2: Jupiter System & Venus (.11-.17)
# Stage 3: Mercury & Saturn System (.18-.23)
# Stage 4: Uranus & Neptune Systems (.24-.29)
# Stage 5: Kuiper Belt & Beyond (.30-.34)
#
# Each option's specialisation goes on the solar colonization entry through
# sr_grant_colony_modifier, which also records it (<modifier>_held) so a
# revolution's winner gets it back: "What a revolution's winner keeps" in
# space_race_effects.txt.
############################################

namespace = space_race_colony_events

##############################################
# STAGE 1: MARS & ASTEROIDS
##############################################

# Colony 1: Mars - Valles Marineris
space_race_colony_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/colony_valles_marineris.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	title = space_race_colony_events.1.t
	desc = space_race_colony_events.1.d
	flavor = space_race_colony_events.1.f

	duration = 3
	trigger = { }
	immediate = {
		create_character = {
			save_scope_as = flavor_speaker
			on_created = { place_character_in_void = 6 }
		}
	}

	option = {
		name = space_race_colony_events.1.a
		default_option = yes
		ai_chance = { base = 5 }
		# Geological research base in the canyon
		sr_grant_colony_modifier = { MODIFIER = sr_colony_valles_marineris_research }
		add_loyalists = { value = medium_radicals pop_type = academics }
		add_loyalists = { value = small_radicals }
	}

	option = {
		name = space_race_colony_events.1.b
		ai_chance = { base = 5 }
		# Industrial mining of canyon wall minerals
		sr_grant_colony_modifier = { MODIFIER = sr_colony_valles_marineris_mining }
		add_loyalists = { value = small_radicals }
		add_loyalists = { value = small_radicals pop_type = capitalists }
	}

	after = {
		hidden_effect = {
			if = {
				limit = { exists = scope:flavor_speaker }
				scope:flavor_speaker = { kill_character = { hidden = yes } }
			}
		}
	}
}
