namespace = nuclear_crisis

# ============================================================================
# NUCLEAR CRISIS EVENTS
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md (§0.3 is what shipped).
#
# Every decision event here is sent by nd_crisis_send_event with the crisis's
# saved scopes (scope:nd_issuer, scope:nd_target) and a token; its trigger is
# nd_crisis_event_matches, and every option runs its act through
# nd_crisis_option, so a popup left open while the crisis ended or changed
# explains itself instead of acting on another crisis.
#
# AI weights read the target's nd_yield_pressure (0–100), the same number the
# player's crisis panel shows (§9: one explanation for both).
#
#   .1   the target receives a warning or an ultimatum
#   .2   the issuer hears the answer: rejected, or met with a counter-threat
#   .3   the other side proposes a mutual stand-down
#   .4   the issuer's deadline arrives
#   .5   the target is pressed again
#   .6   outcome notice, to both sides
#   .7   a public ultimatum was defied by war
#   .20  a guarantor is asked to honour its guarantee
#   .21  the beneficiary hears its guarantor's answer
#   .99  hidden: an AI party reviews its posture (nd_crisis_ai_review)
#   .100 a human's first posture notice (upkeep has begun)
# ============================================================================

# ---- .1 A warning or an ultimatum arrives -----------------------------------
# To the target. Its concession, if it yields, depends on the dispute:
# nd_crisis_demand_text (custom loc on the issuer) names it.
nuclear_crisis.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/diplomatic_crisis_ultimatum.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	title = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:nd_issuer = { nd_crisis_is_public = yes } }
				desc = nuclear_crisis.1.t_public
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_crisis.1.t
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:nd_issuer = { nd_crisis_is_public = yes } }
				desc = nuclear_crisis.1.d_public
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_crisis.1.d
			}
		}
	}
	flavor = nuclear_crisis.1.f

	duration = 3

	trigger = {
		nd_crisis_event_matches = yes
	}

	# Concede.
	option = {
		name = nuclear_crisis.1.a
		nd_crisis_option = { ACT = yield }
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					has_variable = nd_yield_pressure
					var:nd_yield_pressure >= 40
				}
				add = 2
			}
			modifier = {
				trigger = {
					has_variable = nd_yield_pressure
					var:nd_yield_pressure >= 60
				}
				add = 4
			}
			modifier = {
				trigger = {
					has_variable = nd_yield_pressure
					var:nd_yield_pressure >= 80
				}
				add = 8
			}
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 2
			}
		}
	}

	# Refuse.
	option = {
		name = nuclear_crisis.1.b
		default_option = yes
		nd_crisis_option = { ACT = reject }
		ai_chance = {
			base = 4
			modifier = {
				trigger = {
					has_variable = nd_yield_pressure
					var:nd_yield_pressure < 40
				}
				add = 4
			}
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 3
			}
			modifier = {
				trigger = { scope:nd_issuer = { nd_crisis_is_public = yes } }
				add = 1
			}
		}
	}

	# Answer threat with threat. Armed targets only.
	option = {
		name = nuclear_crisis.1.c
		trigger = { nd_is_armed = yes }
		nd_crisis_option = { ACT = counter_threat }
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					has_variable = nd_survivability
					var:nd_survivability >= 50
				}
				add = 3
			}
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 3
			}
			modifier = {
				trigger = { nd_regime_is_militarist = yes }
				add = 2
			}
			modifier = {
				trigger = { ruler_is_cautious = yes }
				factor = 0.25
			}
		}
	}

	# Open a line and propose that both sides step back.
	option = {
		name = nuclear_crisis.1.e
		nd_crisis_option = { ACT = propose_talks }
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 3
			}
			modifier = {
				trigger = {
					has_variable = nd_yield_pressure
					var:nd_yield_pressure >= 40
					var:nd_yield_pressure < 80
				}
				add = 2
			}
			modifier = {
				trigger = { nd_is_armed = yes }
				add = 1
			}
		}
	}
}

				desc = nuclear_crisis.6.t_yielded
			}

			}
			triggered_desc = {

			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_crisis.6.t

		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					nd_crisis_last_outcome_is = { O = 1 }
					nd_crisis_last_role_is = { R = 1 }
				}
				desc = nuclear_crisis.6.d_yielded_issuer
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 1 } }
				desc = nuclear_crisis.6.d_yielded_target
			}
			triggered_desc = {
				trigger = {
					nd_crisis_last_outcome_is = { O = 4 }
					nd_crisis_last_role_is = { R = 1 }
				}
				desc = nuclear_crisis.6.d_native_win_issuer
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 4 } }
				desc = nuclear_crisis.6.d_native_win_target
			}
			triggered_desc = {
				trigger = {
					nd_crisis_last_outcome_is = { O = 2 }
					nd_crisis_last_role_is = { R = 1 }
				}
				desc = nuclear_crisis.6.d_backed_down_issuer
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 2 } }

			}
			triggered_desc = {
				trigger = {
					nd_crisis_last_outcome_is = { O = 5 }
					nd_crisis_last_role_is = { R = 1 }
				}
				desc = nuclear_crisis.6.d_native_loss_issuer
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 5 } }
				desc = nuclear_crisis.6.d_native_loss_target
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 3 } }
				desc = nuclear_crisis.6.d_standdown
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 6 } }
				desc = nuclear_crisis.6.d_war_ended
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 7 } }
				desc = nuclear_crisis.6.d_use
			}
			triggered_desc = {
				trigger = { nd_crisis_last_outcome_is = { O = 8 } }
				desc = nuclear_crisis.6.d_expired
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_crisis.6.d_neutral
			}
		}
	}
	flavor = nuclear_crisis.6.f

	duration = 3

	trigger = {
		has_variable = nd_crisis_last_outcome
	}

	# The other party of THIS notice, from the scopes it was sent with. The
	# option applies the pending record only if it is still this pair's.
	immediate = {
		if = {
			limit = {
				has_variable = nd_crisis_pending_role
				var:nd_crisis_pending_role = 1
				exists = scope:nd_target
			}
			scope:nd_target = { save_scope_as = nd_outcome_other }
		}
		else_if = {
			limit = { exists = scope:nd_issuer }
			scope:nd_issuer = { save_scope_as = nd_outcome_other }
		}
	}

	option = {
		name = nuclear_crisis.6.a
		default_option = yes
		# Apply this notice's record: the pending record is this pair's, or
		# the other party no longer exists (annexed after conceding a war), in
		# which case no newer record about it can have replaced this one.
		if = {
			limit = {
				has_variable = nd_crisis_pending_outcome
				OR = {
					NOT = { exists = scope:nd_outcome_other }
					NOT = { exists = var:nd_crisis_pending_opponent }
					var:nd_crisis_pending_opponent ?= scope:nd_outcome_other
				}
			}
			nd_crisis_apply_outcome_side = yes
		}

		name = nuclear_crisis.20.c
		nd_guarantee_act_abandon = yes

			base = 1
			modifier = {

					var:nd_survivability < 30
				}

			}
		}
	}

# A day after "answer in kind" (.20.b, nd_guarantor_schedule_answer): one
# warhead on the attacker named in nd_answer_strike_target, booked as
# retaliation (nuclear_response_strike, not a first strike) — but only if we

	type = country_event

				nd_forces_assembled = yes
			}

			var:nd_answer_strike_target = {

				}
			}

		}
		remove_variable = nd_answer_strike_target
	}
