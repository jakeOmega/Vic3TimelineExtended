namespace = amendment_petition_events

# ======================================================================
#  AMENDMENT PETITIONS
#
#  Rare events in which an interest group asks for one of the mod's
#  amendments on a law that is already in force, because the country's
#  situation gives it a case. The enactment events (extra_law_events.28–34,
#  .87–.90) offer amendments while a law is debated; these reach a law
#  years after it passed.
#
#  Shape of every event:
#    trigger    te_ap_window_open (one petition per five years), the law in
#               force qualifies (te_ap_law_ok_<amendment>, which restates
#               the amendment's allowed_laws), the amendment's `possible`
#               restated (add_amendment silently fails otherwise), a sponsor
#               that is not marginal, and the situation that justifies it.
#    immediate  sets te_ap_recent_petition, saves petition_law (the law in
#               force) and petition_sponsor_ig.
#    option a   attaches the amendment (cooldown 60 months before it can be
#               repealed) and pleases the sponsor; the groups and pops that
#               oppose it react.
#    option b   (default) declines: the sponsor is displeased, its
#               opponents relieved.
#  Pool: common/on_actions/amendment_petition_on_actions.txt (yearly).
#  Triggers: common/scripted_triggers/amendment_petition_triggers.txt.
#  Docs: docs/systems/mod_systems.md § Amendment Petitions.
# ======================================================================

# ======================================================================
#  1 – The Old Letters  (Minority Script Preservation)
#  A restive cultural minority; the Intelligentsia ask that its script be
#  kept beside the national one.
# ======================================================================
amendment_petition_events.1 = {
	type = country_event
	placement = root

	title = amendment_petition_events.1.t
	desc = amendment_petition_events.1.d
	flavor = amendment_petition_events.1.f

	event_image = { texture = "gfx/event_pictures/language_reform_classroom.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	cooldown = { days = very_long_modifier_time }

	trigger = {
		te_ap_window_open = yes
		te_ap_law_ok_minority_script_preservation = yes
		# amendment_minority_script_preservation's `possible`
		any_scope_state = {
			any_scope_culture = {
				NOT = { is_primary_culture_of = root }
			}
		}
		te_ap_minority_restive = yes
		ig:ig_intelligentsia ?= { is_marginal = no }
	}

	immediate = {
		set_variable = { name = te_ap_recent_petition years = 5 }
		active_law:lawgroup_language_policy ?= { save_scope_as = petition_law }
		ig:ig_intelligentsia ?= { save_scope_as = petition_sponsor_ig }
	}

	cancellation_trigger = {
		NOT = { te_ap_law_ok_minority_script_preservation = yes }
	}

	option = { # Keep the minority scripts
		name = amendment_petition_events.1.a
		scope:petition_law = {
			add_amendment = {
				type = amendment_minority_script_preservation
				sponsor = scope:petition_sponsor_ig
				cooldown = 60
			}
		}
		scope:petition_sponsor_ig = {
			add_modifier = { name = ig_approval_positive_modifier days = normal_modifier_time is_decaying = yes }
		}
		ig:ig_armed_forces ?= {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
		discriminated_pops_loyalty_effect = { VALUE = small_radicals }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { scope:petition_sponsor_ig = { is_in_government = yes } }
				add = 2
			}
			modifier = {
				trigger = { has_any_secessionists_growing = yes }
				add = 2
			}
		}
	}

	option = { # One script for the whole nation
		name = amendment_petition_events.1.b
		default_option = yes
		scope:petition_sponsor_ig = {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
		discriminated_pops_radicalism_effect = { VALUE = small_radicals }
		add_loyalists = {
			value = very_small_radicals
			strata = middle
		}
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ig:ig_armed_forces ?= { is_in_government = yes } }
				add = 1
			}
		}
	}
}

# ======================================================================
#  2 – Spoken Like a Citizen  (Language Requirement for Naturalization)
#  Nativist pressure under an official-language law; the Petite
#  Bourgeoisie (or the Rural Folk) ask for a language test for citizenship.
# ======================================================================
amendment_petition_events.2 = {
	type = country_event
	placement = root

	title = amendment_petition_events.2.t
	desc = amendment_petition_events.2.d
	flavor = amendment_petition_events.2.f

	event_image = { video = "unspecific_steam_ship" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	cooldown = { days = very_long_modifier_time }

	trigger = {
		te_ap_window_open = yes
		te_ap_law_ok_language_naturalization_requirement = yes
		# amendment_language_naturalization_requirement's `possible`
		OR = {
			has_law = law_type:law_civic_monolingualism
			has_law = law_type:law_state_led_language_reform
			has_law = law_type:law_linguistic_purity
		}
		te_ap_nativist_pressure = yes
		OR = {
			ig:ig_petty_bourgeoisie ?= { is_marginal = no }
			ig:ig_rural_folk ?= { is_marginal = no }
		}
	}

	immediate = {
		set_variable = { name = te_ap_recent_petition years = 5 }
		active_law:lawgroup_citizenship ?= { save_scope_as = petition_law }
		if = {
			limit = { ig:ig_petty_bourgeoisie ?= { is_marginal = no } }
			ig:ig_petty_bourgeoisie = { save_scope_as = petition_sponsor_ig }
		}
		else = {
			ig:ig_rural_folk ?= { save_scope_as = petition_sponsor_ig }
		}
	}

	cancellation_trigger = {
		NOT = { te_ap_law_ok_language_naturalization_requirement = yes }

	option = { # Require the language for citizenship
		name = amendment_petition_events.2.a
		scope:petition_law = {
			add_amendment = {
				type = amendment_language_naturalization_requirement
				sponsor = scope:petition_sponsor_ig

			}
		}
		scope:petition_sponsor_ig = {
			add_modifier = { name = ig_approval_positive_modifier days = normal_modifier_time is_decaying = yes }
		}
		ig:ig_intelligentsia ?= {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
		discriminated_pops_radicalism_effect = { VALUE = small_radicals }
		ai_chance = {
			base = 1
			modifier = {
				trigger = { scope:petition_sponsor_ig = { is_in_government = yes } }
				add = 2
			}
			modifier = {
				trigger = { scope:petition_sponsor_ig = { is_powerful = yes } }
				add = 1
			}
		}
	}

	option = { # No grammar test for citizenship
		name = amendment_petition_events.2.b
		default_option = yes
		scope:petition_sponsor_ig = {
			add_modifier = { name = ig_approval_negative_modifier days = normal_modifier_time is_decaying = yes }
		}
		add_radicals = {
			value = very_small_radicals
			interest_group = scope:petition_sponsor_ig
		}
		discriminated_pops_loyalty_effect = { VALUE = very_small_radicals }
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ig:ig_intelligentsia ?= { is_in_government = yes } }
				add = 1
