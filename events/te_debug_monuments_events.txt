# ============================================================================
# GRAND MONUMENTS — console test event: `event te_debug_monuments.1`
# ============================================================================
# Each option writes TE_MONUMENTS: lines to debug.log (gm_debug_log). The
# options together answer the in-game checklist: a ceremony, a monthly pass,
# a forced contest, vanity backlash, and a print of every total.

te_debug_monuments.1 = { # REVIEWED 2026-09-27: console-only test event (`event te_debug_monuments.1`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_monuments.1.t
	desc = te_debug_monuments.1.desc
	flavor = te_debug_monuments.1.flavor

	# Raise a level-20 monument in the capital and hold the ceremony.
	option = {
		name = te_debug_monuments.1.a
		default_option = yes
		capital = {
			if = {
				limit = {
					NOT = { has_building = building_grand_monument }
				}
				create_building = { building = building_grand_monument level = 20 }
			}
			gm_state_start_ceremony = yes
		}
	}

	# Run this month's pass now (contests, ledgers, totals, the log).
	option = {
		name = te_debug_monuments.1.b
		gm_country_monthly = yes
	}

	# Contest our tallest regime, ruler or faith monument now, as if it had
	# fallen, and open its notice. gm_country_refresh would run gm_check_contests
	# again immediately, see the just-forced contest still fitting, and lift it
	# right back. gm_notify_new_contests only fires monument_events.16 (or, for
	# the AI, clears gm_new_contest) without re-deriving contests, so the forced
	# contest lasts until the player decides it or the next monthly pulse lifts
	# it (the monument still fits): test the notice and the row buttons within
	# the month.
	option = {
		name = te_debug_monuments.1.c
		ordered_scope_state = {
			limit = {
				gm_state_is_dedicated = yes
				gm_state_is_bound = yes
				gm_state_is_contested = no
			}
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			gm_state_contest = yes
		}
		gm_notify_new_contests = yes
	}

	# Vanity backlash in the capital's monument, as if a level had finished in hard times.
	option = {
		name = te_debug_monuments.1.d
		capital = {
			if = {
				limit = { has_building = building_grand_monument }
				gm_state_vanity_backlash = yes
			}
		}
		gm_country_refresh = yes
	}

	# Print every total.
	option = {
		name = te_debug_monuments.1.e
		gm_debug_log = yes
	}
}

# ============================================================================
# COMMISSIONS (v2 §2) — console test event: `event te_debug_monuments.2`
# ============================================================================
# Each offer option forgets any commission first, writes the record as its
# source would (through the source's own effect where it can) and opens the
# offer (monument_events.21). The last options close the open commission each
# way and print it.
te_debug_monuments.2 = { # REVIEWED 2026-10-05: console-only test event (`event te_debug_monuments.2`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_monuments.2.t
	desc = te_debug_monuments.2.desc
	flavor = te_debug_monuments.2.flavor

	# Victory over a random rival, as if a war leader's war of a year had been won.
	option = {
		name = te_debug_monuments.2.a
		default_option = yes
		gm_com_clear = yes
		random_country = {
			limit = {
				NOT = { this = root }
				exists = capital
			}
			save_scope_as = enemy_country
		}
		gm_com_war_victory = yes
	}

	# Defeat, as if such a war had been lost.
	option = {
		name = te_debug_monuments.2.b
		gm_com_clear = yes
		gm_com_war_defeat = yes
	}

	# A centenary this year.
	option = {
		name = te_debug_monuments.2.c
		gm_com_clear = yes
		set_variable = { name = gm_founded_year value = { value = year subtract = 100 } }
		gm_remove_state_var = { VAR = gm_com_centenary_100 }
		set_variable = { name = gm_com_age value = 100 }
		gm_com_track_centenary_n = { N = 100 }
		remove_variable = gm_com_age
	}

	# The ruler's death, as if after a long reign (the ruler lives on).
	option = {
		name = te_debug_monuments.2.d
		gm_com_clear = yes
		if = {
			limit = { exists = ruler }
			ruler = { save_scope_as = gm_late_ruler }
			gm_com_ruler_death = yes
		}
	}

	# A new regime: forget last month's, so this month's reads as new.
	option = {
		name = te_debug_monuments.2.e
		gm_com_clear = yes
		set_variable = { name = gm_regime_last value = flag:none }
		gm_com_track_regime = yes
	}

	# A space first (the orbital one), without touching the space race.
	option = {
		name = te_debug_monuments.2.f
		gm_com_clear = yes
		gm_com_begin = { SOURCE = space DEDICATION = civic NAMESAKE = occasion FORM = obelisk }
		set_variable = { name = gm_com_occasion value = flag:space_orbital }
		set_variable = { name = gm_com_ig value = flag:intelligentsia }
		gm_com_fix_to_capital = yes
		gm_com_send_offer = yes
	}

	# A petition, past the monthly roll and the cooldown.
	option = {
		name = te_debug_monuments.2.g
		gm_com_clear = yes
		gm_remove_state_var = { VAR = gm_com_cooldown }
		gm_com_offer_petition = yes
	}

	# Meet the open commission now.
	option = {
		name = te_debug_monuments.2.h
		if = {
			limit = { has_variable = gm_com_open }
			gm_com_fulfil = yes
			gm_country_refresh = yes
		}
	}

	# Miss it: the clock runs out this month.
	option = {
		name = te_debug_monuments.2.i
		if = {
			limit = { has_variable = gm_com_open }
			set_variable = { name = gm_com_months_left value = 1 }
			gm_com_monthly = yes
			gm_country_refresh = yes
		}
	}

	# Lapse it.
	option = {
		name = te_debug_monuments.2.j
		if = {
			limit = { has_variable = gm_com_open }
			gm_com_lapse = yes
			gm_country_refresh = yes
		}
	}

	# Print the commission.
	option = {
		name = te_debug_monuments.2.k
		gm_debug_log = yes
	}
}

# ============================================================================
# THE DEDICATION LOCK — console test event: `event te_debug_monuments.3`
# ============================================================================
# Option a answers whether activate_production_method can move a monument
# between dedications past the self-referencing lock (grand_monument_pms.txt).
# Undedicated stays selectable so that script can always switch a new
# monument off a dedication it copied from its siblings; if script can
# override the lock as well, Undedicated could be locked again. It tries our
# tallest monument on To the Nation (the War Memorial, if it is To the Nation
# already), neither of which a technology gates, and
# te_debug_monuments.4 reads the result a day later and puts the dedication
# back. The other options re-open one monument's dedication: its records go
# and the ceremony asks again, from Undedicated. They are for a monument that
# kept a copied dedication before gm_state_reset_copied_dedication existed.
te_debug_monuments.3 = { # REVIEWED 2026-10-08: console-only test event (`event te_debug_monuments.3`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_monuments.3.t
	desc = te_debug_monuments.3.desc
	flavor = te_debug_monuments.3.flavor

	# Our three tallest monuments, named in the re-open options.
	immediate = {
		ordered_scope_state = {
			limit = { has_building = building_grand_monument }
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			save_scope_as = gm_dbg_m1
		}
		ordered_scope_state = {
			limit = { has_building = building_grand_monument }
			order_by = gm_state_grandeur
			position = 1
			check_range_bounds = no
			save_scope_as = gm_dbg_m2
		}
		ordered_scope_state = {
			limit = { has_building = building_grand_monument }
			order_by = gm_state_grandeur
			position = 2
			check_range_bounds = no
			save_scope_as = gm_dbg_m3
		}
	}

	# Probe the lock on our tallest dedicated monument.
	option = {
		name = te_debug_monuments.3.a
		default_option = yes
		ordered_scope_state = {
			limit = {
				gm_state_is_dedicated = yes
				has_variable = gm_dedication
			}
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			# Holds gm_state_keep_dedication off until te_debug_monuments.4.
			set_variable = { name = gm_pm_probe days = 5 }
			if = {
				limit = {
					can_activate_production_method = {
						building_type = building_grand_monument
						production_method = pm_monument_undedicated
					}
				}
				debug_log = "TE_MONUMENTS: lock probe: Undedicated can be activated (expected: it is never locked)"
			}
			else = {
				debug_log = "TE_MONUMENTS: lock probe: Undedicated can NOT be activated (unexpected: it is never locked)"
			}
			if = {
				limit = { gm_state_has_pm = { PM = pm_monument_civic } }
				set_variable = { name = gm_pm_probe_target value = flag:war_memorial }
				if = {
					limit = {
						can_activate_production_method = {
							building_type = building_grand_monument
							production_method = pm_monument_war_memorial
						}
					}
					debug_log = "TE_MONUMENTS: lock probe: the engine calls the War Memorial available (unexpected: the lock should hide it)"
				}
				else = {
					debug_log = "TE_MONUMENTS: lock probe: the engine calls the War Memorial unavailable, as the lock should"
				}
				activate_production_method = {
					building_type = building_grand_monument
					production_method = pm_monument_war_memorial
				}
			}
			else = {
				set_variable = { name = gm_pm_probe_target value = flag:civic }
				if = {
					limit = {
						can_activate_production_method = {
							building_type = building_grand_monument
							production_method = pm_monument_civic
						}
					}
					debug_log = "TE_MONUMENTS: lock probe: the engine calls To the Nation available (unexpected: the lock should hide it)"
				}
				else = {
					debug_log = "TE_MONUMENTS: lock probe: the engine calls To the Nation unavailable, as the lock should"
				}
				activate_production_method = {
					building_type = building_grand_monument
					production_method = pm_monument_civic
				}
			}
		}
		trigger_event = { id = te_debug_monuments.4 days = 1 }
	}

	# Re-open the dedication of one monument.
	option = {
		name = te_debug_monuments.3.b
		trigger = { exists = scope:gm_dbg_m1 }
		scope:gm_dbg_m1 = {
			gm_clear_state = yes
			gm_state_start_ceremony = yes
		}
		trigger_event = { id = monument_events.20 }
	}

	option = {
		name = te_debug_monuments.3.c
		trigger = { exists = scope:gm_dbg_m2 }
		scope:gm_dbg_m2 = {
			gm_clear_state = yes
			gm_state_start_ceremony = yes
		}
		trigger_event = { id = monument_events.20 }
	}

	option = {
		name = te_debug_monuments.3.d
		trigger = { exists = scope:gm_dbg_m3 }
		scope:gm_dbg_m3 = {
			gm_clear_state = yes
			gm_state_start_ceremony = yes
		}
		trigger_event = { id = monument_events.20 }
	}
}

# Option a's result, a day after its switch: did the monument move to the
# locked dedication? Then the dedication goes back (gm_state_keep_dedication,
# through Undedicated).
te_debug_monuments.4 = {
	type = country_event
	hidden = yes

	immediate = {
		every_scope_state = {
			limit = { has_variable = gm_pm_probe }
			if = {
				limit = {
					OR = {
						AND = {
							var:gm_pm_probe_target = flag:civic
							gm_state_has_pm = { PM = pm_monument_civic }
						}
						AND = {
							var:gm_pm_probe_target = flag:war_memorial
							gm_state_has_pm = { PM = pm_monument_war_memorial }
						}
					}
				}
				debug_log = "TE_MONUMENTS: lock probe result: MOVED. Script overrides the lock; Undedicated could be locked again"
			}
			else_if = {
				limit = { gm_state_has_pm = { PM = pm_monument_undedicated } }
				debug_log = "TE_MONUMENTS: lock probe result: the monument went to Undedicated, not the target"
			}
			else = {
				debug_log = "TE_MONUMENTS: lock probe result: KEPT. Script cannot override the lock; Undedicated must stay unlocked"
			}
			remove_variable = gm_pm_probe
			remove_variable = gm_pm_probe_target
			gm_state_keep_dedication = yes
		}
	}
}
