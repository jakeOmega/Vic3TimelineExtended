# ============================================================================
# GRAND MONUMENTS — console test event: `event te_debug_monuments.1`
# ============================================================================
# Each option writes TE_MONUMENTS: lines to debug.log (gm_debug_log). The
# options together answer the in-game checklist: a ceremony, a monthly pass,
# a forced contest, vanity backlash, and a print of every total.

te_debug_monuments.1 = { # REVIEWED 2026-09-27: console-only test event (`event te_debug_monuments.1`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_monuments.1.t
	desc = te_debug_monuments.1.desc
	flavor = te_debug_monuments.1.flavor

	# Raise a 20-level construction site in the capital (sites spec
	# 2026-10-18): its Dedicate button holds the ceremony.
	option = {
		name = te_debug_monuments.1.a
		default_option = yes
		capital = {
			if = {
				limit = {
					NOT = { has_building = building_grand_monument_site }
				}
				create_building = { building = building_grand_monument_site level = 20 }
			}
		}
	}

	# Run this month's pass now (contests, ledgers, totals, the log).
	option = {
		name = te_debug_monuments.1.b
		gm_country_monthly = yes
	}

	# Contest our tallest regime, ruler or faith monument now, as if it had
	# fallen, and open its notice. gm_country_refresh would run gm_check_contests
	# again immediately, see the just-forced contest still fitting, and lift it
	# right back. gm_notify_new_contests only fires monument_events.16 (or, for
	# the AI, clears gm_new_contest) without re-deriving contests, so the forced
	# contest lasts until the player decides it or the next monthly pulse lifts
	# it (the monument still fits): test the notice and the row buttons within
	# the month.
	option = {
		name = te_debug_monuments.1.c
		ordered_scope_state = {
			limit = {
				gm_state_is_dedicated = yes
				gm_state_is_bound = yes
				gm_state_is_contested = no
			}
			order_by = gm_state_grandeur
			position = 0
			check_range_bounds = no
			gm_state_contest = yes
		}
		gm_notify_new_contests = yes
	}

	# Vanity backlash in the capital's monument, as if a level had finished in hard times.
	option = {
		name = te_debug_monuments.1.d
		capital = {
			if = {
				limit = { has_building = building_grand_monument }
				gm_state_vanity_backlash = yes
			}
		}
		gm_country_refresh = yes
	}

	# Print every total.
	option = {
		name = te_debug_monuments.1.e
		gm_debug_log = yes
	}
}

# ============================================================================
# OPPORTUNITIES — console test event: `event te_debug_monuments.2`
# ============================================================================
# Each offer option forgets any opportunity first, writes the record as its
# moment would (through the moment's own effect where it can) and opens the
# offer (monument_events.21), past the rank and the once-per-game marks. The
# last options close the open opportunity each way and print it. "Complete it
# now" runs the gift (monument_events.26) and the boon as a real completion.
# Docs: docs/superpowers/archive/specs/2026-10-12-grand-monument-opportunities-design.md.
te_debug_monuments.2 = { # REVIEWED 2026-10-12: console-only test event (`event te_debug_monuments.2`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_monuments.2.t
	desc = te_debug_monuments.2.desc
	flavor = te_debug_monuments.2.flavor

	# Victory over a random rival, as if a war leader's war of a year had been won.
	option = {
		name = te_debug_monuments.2.a
		default_option = yes
		gm_com_clear = yes
		random_country = {
			limit = {
				NOT = { this = root }
				exists = capital
			}
			save_scope_as = enemy_country
		}
		gm_com_war_victory = yes
	}

	# Defeat, as if such a war had been lost.
	option = {
		name = te_debug_monuments.2.b
		gm_com_clear = yes
		gm_com_war_defeat = yes
	}

	# A centenary this year.
	option = {
		name = te_debug_monuments.2.c
		gm_com_clear = yes
		set_variable = { name = gm_founded_year value = { value = year subtract = 100 } }
		gm_remove_state_var = { VAR = gm_com_centenary_100 }
		set_variable = { name = gm_com_age value = 100 }
		gm_com_track_centenary_n = { N = 100 }
		remove_variable = gm_com_age
	}

	# The ruler's death, as if after a long reign (the ruler lives on).
	option = {
		name = te_debug_monuments.2.d
		gm_com_clear = yes
		if = {
			limit = { exists = ruler }
			ruler = { save_scope_as = gm_late_ruler }
			gm_com_ruler_death = yes
		}
	}

	# A new order: forget last month's regime, so this month's reads as new.
	option = {
		name = te_debug_monuments.2.e
		gm_com_clear = yes
		set_variable = { name = gm_regime_last value = flag:none }
		gm_com_track_regime = yes
	}

	# A space first (the orbital one), without touching the space race.
	option = {
		name = te_debug_monuments.2.f
		gm_com_clear = yes
		gm_com_begin = { SOURCE = space DEDICATION = civic NAMESAKE = occasion FORM = obelisk }
		set_variable = { name = gm_com_occasion value = flag:space_orbital }
		set_variable = { name = gm_com_ig value = flag:intelligentsia }
		gm_com_fix_to_capital = yes
		gm_com_send_offer = yes
	}

	# A new reign, for the ruler we have (To the Crown or To the Leader, if
	# the government can raise either).
	option = {
		name = te_debug_monuments.2.g
		gm_com_clear = yes
		if = {
			limit = { exists = ruler }
			ruler = { save_scope_as = gm_com_tracked_ruler }
			gm_com_try_reign = yes
		}
	}

	# Among the Great Powers, whatever our rank.
	option = {
		name = te_debug_monuments.2.h
		gm_com_clear = yes
		gm_com_begin = { SOURCE = great_power DEDICATION = civic NAMESAKE = honours FORM = column }
		gm_com_ig_for_the_nation = yes
		gm_com_fix_to_capital = yes
		gm_com_send_offer = yes
	}

	# A new faith, as if the state religion had just changed.
	option = {
		name = te_debug_monuments.2.i
		gm_com_clear = yes
		gm_com_begin = { SOURCE = faith DEDICATION = religious NAMESAKE = honours FORM = default }
		set_variable = { name = gm_com_ig value = flag:devout }
		gm_com_send_offer = yes
	}

	# A technology's: Romanticism's Grand Opera House.
	option = {
		name = te_debug_monuments.2.j
		gm_com_clear = yes
		gm_com_begin = { SOURCE = tech_artistic DEDICATION = artistic NAMESAKE = city FORM = default }
		set_variable = { name = gm_com_ig value = flag:intelligentsia }
		gm_com_send_offer = yes
	}

	# Complete the open opportunity now: the gift, the approval and the boon.
	option = {
		name = te_debug_monuments.2.k
		if = {
			limit = { has_variable = gm_com_open }
			gm_com_fulfil = yes
			gm_country_refresh = yes
		}
	}

	# Run it out: the clock ends this month.
	option = {
		name = te_debug_monuments.2.l
		if = {
			limit = { has_variable = gm_com_open }
			set_variable = { name = gm_com_months_left value = 1 }
			gm_com_monthly = yes
			gm_country_refresh = yes
		}
	}

	# Lapse it.
	option = {
		name = te_debug_monuments.2.m
		if = {
			limit = { has_variable = gm_com_open }
			gm_com_lapse = yes
			gm_country_refresh = yes
		}
	}

	# Print the opportunity.
	option = {
		name = te_debug_monuments.2.n
		gm_debug_log = yes
	}
}

# Option a carries off the tallest monument in the world that is not ours, as
# if our armies held its state (no war or province test); option b our own
# tallest, as if from an unincorporated state (no incorporation or builder
# test); c returns every trophy we hold; d runs the monthly pass now and logs
# our trophies on TE_MONUMENTS: lines.
te_debug_monuments.6 = { # REVIEWED 2026-10-12: console-only test event (`event te_debug_monuments.6`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_monuments.6.t
	desc = te_debug_monuments.6.desc
	flavor = te_debug_monuments.6.flavor

	# Each option does its work behind a click-only marker
	# (scripting_best_practices.md § "An Option's Tooltip Walks Its Effects"):
	# a taking or a return reads back what it writes, which the tooltip's walk
	# would log every frame.
	option = {
		name = te_debug_monuments.6.a
		custom_tooltip = {
			text = te_debug_monuments.6.a.tt
			set_variable = gm_tr_debug_click
			if = {
				limit = { has_variable = gm_tr_debug_click }
				remove_variable = gm_tr_debug_click
				ordered_state = {
					limit = {
						gm_state_tr_takeable = yes
						NOT = { owner = root }
					}
					order_by = gm_state_grandeur
					position = 0
					check_range_bounds = no
					gm_state_tr_take = { WHO = root ROUTE = occupation }
				}
				gm_tr_finish = yes
			}
		}
	}

	option = {
		name = te_debug_monuments.6.b
		custom_tooltip = {
			text = te_debug_monuments.6.b.tt
			set_variable = gm_tr_debug_click
			if = {
				limit = { has_variable = gm_tr_debug_click }
				remove_variable = gm_tr_debug_click
				ordered_scope_state = {
					limit = { gm_state_tr_takeable = yes }
					order_by = gm_state_grandeur
					position = 0
					check_range_bounds = no
					gm_state_tr_take = { WHO = root ROUTE = own }
				}
				gm_tr_finish = yes
			}
		}
	}

	option = {
		name = te_debug_monuments.6.c
		custom_tooltip = {
			text = te_debug_monuments.6.c.tt
			set_variable = gm_tr_debug_click
			if = {
				limit = {
					has_variable = gm_tr_debug_click
					has_global_variable_list = gm_trophies
				}
				every_in_global_list = {
					variable = gm_trophies
					limit = { gm_tr_held_by = { WHO = root } }
					add_to_temporary_list = gm_tr_debug_returning
				}
				every_in_list = {
					list = gm_tr_debug_returning
					gm_tr_return_home = { RELATIONS = yes }
				}
				gm_tr_finish = yes
			}
			if = {
				limit = { has_variable = gm_tr_debug_click }
				remove_variable = gm_tr_debug_click
			}
		}
	}

	option = {
		name = te_debug_monuments.6.d
		default_option = yes
		custom_tooltip = {
			text = te_debug_monuments.6.d.tt
			set_variable = gm_tr_debug_click
			if = {

				remove_variable = gm_tr_debug_click
				gm_trophy_monthly = yes
				gm_tr_debug_log = yes
			}
		}
	}
