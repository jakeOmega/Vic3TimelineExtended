# ============================================================================
# DEMOGRAPHICS TEST CONSOLE — fired only from the console: `event te_debug_demog.1`
# ============================================================================
# Vic3's console has no `effect` command (that is CK3); `event <id>` is the way
# to run script by hand. Nothing in the mod fires this event. Every option writes
# to debug.log; the effects are in common/scripted_effects/te_debug_demog_effects.txt.
# debug.log rolls over at 512 KB: copy the lines out as they appear
# (docs/guides/scripting_best_practices.md, "Runtime Debugging with debug_log").
#
# What each option is for:
#   a    replay: step the capital's census one year now and log it (TE_DEMOG_REPLAY);
#        `scripts/analysis/demographics_harness.py replay debug.log` checks it against
#        the model. A second run in the same year steps again (the census year goes
#        back by one first) and folds this year's newborns into the over-150 pool:
#        use a save you can throw away, or re-seed after (c)
#   b    start or stop the census log: each 31 December a TE_DEMOG_CENSUS: line per
#        country of a million people or more (scripts/analysis/
#        demographics_observer_report.py reads them), and a TE_DEMOG_PULSE line at the
#        player's capital's yearly state pulse (one step per calendar year?)
#   c    re-seed this country's census from today's population, then take its census
#   d    benchmark (spec §14 Q9's late-game re-run): every state's step in the world,
#        three passes, between TE_DEMOG_BENCH lines; then, untimed, every census is
#        seeded again (the passes leave them wrong)
#   e    war dead (spec §14 Q4): the player's dead in each war, its monthly tally and
#        the states' pending dead (TE_DEMOG_WARDEAD)
#   f/g  skew check (spec §14 Q11): f logs the capital's pops (TE_DEMOG_SKEW) and adds
#        a 60-day +0.05 working-adult ratio modifier; g logs them again a month later
#   h-k  fast mode (te_demog_fast_effects.txt): demography, research and construction
#        at 4x (h), 6x (i) or 12x (j), for iterating on the model; k goes back to normal
#        speed. A test tool, never for play: the first switch re-seeds every census, and
#        the census year then runs on its own clock. Fire it on day 1, then `observe`.
#   l    the census's mortality and fertility modifier types as the capital and an
#        unincorporated state read them (TE_DEMOG_MODS)
#   m    the shock terms (probe P1): every devastated or polluted state's reads
#        (TE_DEMOG_SHOCK)
#   n    the census's rates (probe P3): the capital's births and deaths terms, the
#        on-top scales of its last step and its reads (TE_DEMOG_RATES)
#   o    the engine-scaling probe (Task 9 of the late-marriage plan): a script-only type on
#        the capital's state modifier at multiplier 0.62 (TE_DEMOG_PROBE), logged before and
#        a week after by the hidden te_debug_demog.2
#   q    the owner-change probe (the late-marriage plan, Stop B's I1): one of our states, alone
#        in its region, given to a neighbour and logged before, in on_state_owner_change and a
#        day after by the hidden te_debug_demog.4 (TE_DEMOG_PROBE_OWNER). It really changes
#        hands: throwaway games only. (p and te_debug_demog.3 are phase 2 step 1's.)
#
# An option's tooltip walks its effects each time the open event renders it, testing
# the limits and writing nothing. Options a, c, d and e walk a census step, the walks
# or every state and war, and their tooltips wrote about 1,200 error lines a second
# while the event was open (2026-10-09). Each sets te_dg_dbg_click and does its work
# only under has_variable = te_dg_dbg_click: the tooltip pass never sets it, so it
# skips the work, and the event now logs nothing while open (confirmed 2026-10-09;
# docs/guides/scripting_best_practices.md, "An Option's Tooltip Walks Its Effects").
# ============================================================================
namespace = te_debug_demog

te_debug_demog.1 = { # REVIEWED 2026-10-09: console-only test event (`event te_debug_demog.1`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/medical_breakthrough.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_demog.1.t
	desc = te_debug_demog.1.desc
	flavor = te_debug_demog.1.flavor

	# ---- a: replay ----------------------------------------------------------
	option = {
		name = te_debug_demog.1.a
		trigger = {
			te_demog_cohorts_run = yes
			capital ?= { te_demog_has_census = yes }
		}
		custom_tooltip = {
			text = te_debug_demog.1.a.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				capital = { te_debug_demog_replay = yes }
			}
		}
	}

	# ---- b: the census log --------------------------------------------------
	option = {
		name = te_debug_demog.1.b
		trigger = {
			NOT = { has_global_variable = te_demog_census_log }
		}
		custom_tooltip = {
			text = te_debug_demog.1.b.tt
			set_global_variable = te_demog_census_log
			debug_log = "TE_DEMOG_LOG census log on, date=[TimeKeeper.GetCurrentDate.GetString]"
		}
	}

	option = {
		name = te_debug_demog.1.b_off
		trigger = {
			has_global_variable = te_demog_census_log
		}
		remove_global_variable = te_demog_census_log
		debug_log = "TE_DEMOG_LOG census log off, date=[TimeKeeper.GetCurrentDate.GetString]"
	}

	# ---- c: re-seed ---------------------------------------------------------
	# The walks first, as at game start (te_demog_country_game_start), so each seed
	# reads this year's literacy, urban share and wealth.
	option = {
		name = te_debug_demog.1.c
		trigger = {
			te_demog_cohorts_run = yes
		}
		custom_tooltip = {
			text = te_debug_demog.1.c.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				every_scope_state = {
					te_demog_walks = yes
					# off the states' pulses: each seeds again at its next one
					te_demog_seed_off_pulse = yes
				}
				te_demog_country_census = yes
			}
		}
	}

	# ---- d: benchmark -------------------------------------------------------
	option = {
		name = te_debug_demog.1.d
		trigger = {
			te_demog_cohorts_run = yes
		}
		custom_tooltip = {
			text = te_debug_demog.1.d.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_debug_demog_benchmark = yes
			}
		}
	}

	# ---- e: war dead --------------------------------------------------------
	option = {
		name = te_debug_demog.1.e
		custom_tooltip = {
			text = te_debug_demog.1.e.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_debug_demog_war_dead = yes
			}
		}
	}

	# ---- f and g: the skew check --------------------------------------------
	option = {
		name = te_debug_demog.1.f
		trigger = {
			exists = capital
		}
		custom_tooltip = {
			text = te_debug_demog.1.f.tt
			capital = {
				debug_log = "TE_DEMOG_SKEW f: before the test modifier"
				te_debug_demog_skew_lines = yes
				add_modifier = { name = te_demog_debug_working_adults days = 60 }
			}
		}
	}

	option = {
		name = te_debug_demog.1.g
		trigger = {
			exists = capital
		}
		custom_tooltip = {
			text = te_debug_demog.1.g.tt
			capital = {
				debug_log = "TE_DEMOG_SKEW g: later"
				te_debug_demog_skew_lines = yes
			}
		}
	}

	# ---- h-k: fast mode -----------------------------------------------------
	option = {
		name = te_debug_demog.1.h
		custom_tooltip = {
			text = te_debug_demog.1.h.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_demog_fast_set = { MONTHS = 3 }
			}
		}
	}

	option = {
		name = te_debug_demog.1.i
		custom_tooltip = {
			text = te_debug_demog.1.i.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_demog_fast_set = { MONTHS = 2 }
			}
		}
	}

	option = {
		name = te_debug_demog.1.j
		custom_tooltip = {
			text = te_debug_demog.1.j.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_demog_fast_set = { MONTHS = 1 }
			}
		}
	}

	option = {
		name = te_debug_demog.1.k
		trigger = {
			te_demog_fast_k > 1
		}
		custom_tooltip = {
			text = te_debug_demog.1.k.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_demog_fast_set = { MONTHS = 12 }
			}
		}
	}

	# ---- l: the mortality modifier types -----------------------------------
	option = {
		name = te_debug_demog.1.l
		trigger = {
			exists = capital
		}
		custom_tooltip = {
			text = te_debug_demog.1.l.tt
			te_debug_demog_modifier_reads = yes
		}
	}

	# ---- m: the shock terms (probe P1) -------------------------------------
	option = {
		name = te_debug_demog.1.m
		custom_tooltip = {
			text = te_debug_demog.1.m.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_debug_demog_shock_lines = yes
			}
		}
	}

	# ---- n: the census's rates (probe P3) -----------------------------------
	option = {
		name = te_debug_demog.1.n
		trigger = {
			exists = capital
			te_demog_cohorts_run = yes
		}
		custom_tooltip = {
			text = te_debug_demog.1.n.tt
			set_variable = te_dg_dbg_click

	}

	# ---- o: the engine-scaling probe (the late-marriage plan, Task 9) ------------
	option = {
		name = te_debug_demog.1.o
		trigger = {
			exists = capital
		}
		custom_tooltip = {
			text = te_debug_demog.1.o.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				capital = { te_debug_demog_probe_start = yes }
			}
		}
	}

	# ---- q: the owner-change probe (the late-marriage plan, Stop B's I1) ----------
	option = {
		name = te_debug_demog.1.q
		trigger = {
			exists = capital
		}
		custom_tooltip = {
			text = te_debug_demog.1.q.tt
			set_variable = te_dg_dbg_click
			if = {
				limit = { has_variable = te_dg_dbg_click }
				remove_variable = te_dg_dbg_click
				te_debug_demog_probe_owner_start = yes
			}
		}
	}

	option = {
		name = te_debug_demog.1.z
		default_option = yes
	}
}

# The probe's second half (option o): a week after the add, the state's sum and the probe off again.
te_debug_demog.2 = {
	type = state_event
	hidden = yes

	immediate = {
		debug_log = "TE_DEMOG_PROBE after absorbed_b=[THIS.ScriptValue('te_debug_demog_line_absorbed_b')|5] date=[TimeKeeper.GetCurrentDate.GetString]"
		debug_log_scopes = yes
		if = {
			limit = { has_modifier = te_debug_demog_scaled_probe }
			remove_modifier = te_debug_demog_scaled_probe
		}
	}
}

# The owner-change probe's last line (option q): a day after the transfer, every state now in the stamped region, then
# the stamps and the hook's flag off again.
te_debug_demog.4 = {
	type = country_event
	hidden = yes

	immediate = {
		every_state = {
			limit = {
				state_region = { has_variable = te_probe_owner_region }
			}
			debug_log = "TE_DEMOG_PROBE_OWNER after name=[THIS.GetState.GetNameNoFormatting] id=[THIS.GetState.GetID] owner=[THIS.GetState.GetOwner.GetTag] var=[THIS.ScriptValue('te_debug_demog_probe_owner_var')|0] year=[THIS.ScriptValue('te_debug_demog_probe_owner_year')|0] lm=[THIS.ScriptValue('te_debug_demog_probe_owner_lm')|3] date=[TimeKeeper.GetCurrentDate.GetString]"
			debug_log_scopes = yes
			if = {
				limit = { has_variable = te_probe_owner_var }
				remove_variable = te_probe_owner_var
			}
		}
		every_state = {
			limit = {
				state_region = { has_variable = te_probe_owner_region }
			}
			state_region = { remove_variable = te_probe_owner_region }
		}
		remove_global_variable = te_probe_owner_on
		remove_global_variable = te_probe_owner_to
	}
}

# The Primogeniture self-check's second half (te_debug_demog_pr_selfcheck_start): runs once a game, in every game, a
# week after the first fresh Primogeniture line. OK: the engine summed the script-only type scaled by the agrarian
# share. BROKEN: swap Primogeniture to Plan B (docs/superpowers/plans/2026-10-11-demographics-late-marriage.md, Task 9).
te_debug_demog.5 = {
	type = state_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = te_inh_pr_chk_before }
			if = {
				limit = { te_debug_demog_pr_chk_gap < 0.002 }
				debug_log = "TE_SELF_CHECK OK primogeniture_scaled added=[THIS.ScriptValue('te_debug_demog_pr_chk_added')|4] expected=[THIS.ScriptValue('te_debug_demog_pr_chk_expected')|4] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN primogeniture_scaled added=[THIS.ScriptValue('te_debug_demog_pr_chk_added')|4] expected=[THIS.ScriptValue('te_debug_demog_pr_chk_expected')|4] (0.1 or 0: the engine doesn't scale a script-only type on a state modifier; take Plan B, late-marriage plan Task 9) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			debug_log_scopes = yes
			remove_variable = te_inh_pr_chk_before
		}
	}
}

# The measures' self-check's second half (te_debug_demog_ms_selfcheck_start): runs once a game, under Full, a week
# after the first measure that moves desired family size switched on. added: the capital's read of
# state_desired_fertility_mult since then; expected: the change in the measures' lines its owner holds
# (te_demog_ms_desired_on). OK: a script-only state type on a country static modifier reaches the state. BROKEN: it
# doesn't, so the census will never see the measures' types; put them on a state modifier in each state instead
# (docs/testing/demographics-playtest.md, Self-checks). No verdict when the state changed hands in the week.
te_debug_demog.10 = {
	type = state_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_variable = te_dg_ms_chk_before
				exists = owner
				owner = var:te_dg_ms_chk_owner
			}
			owner = {
				set_local_variable = { name = te_dg_ms_chk_e1 value = te_demog_ms_desired_on }
			}
			# kept on the state for the line (a debug_log line reads THIS.Var, not a local)
			set_variable = {
				name = te_dg_ms_chk_added
				value = {
					value = modifier:state_desired_fertility_mult
					subtract = var:te_dg_ms_chk_before
				}
			}
			set_variable = {
				name = te_dg_ms_chk_expected
				value = {
					value = local_var:te_dg_ms_chk_e1
					subtract = var:te_dg_ms_chk_e0
				}
			}
			set_local_variable = {
				name = te_dg_ms_chk_gap
				value = {
					value = var:te_dg_ms_chk_added
					subtract = var:te_dg_ms_chk_expected
				}
			}
			if = {
				limit = { local_var:te_dg_ms_chk_gap < 0 }
				change_local_variable = { name = te_dg_ms_chk_gap multiply = -1 }
			}
			if = {
				limit = { local_var:te_dg_ms_chk_gap < 0.002 }
				debug_log = "TE_SELF_CHECK OK measure_country_types added=[THIS.Var('te_dg_ms_chk_added').GetValue|4] expected=[THIS.Var('te_dg_ms_chk_expected').GetValue|4] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN measure_country_types added=[THIS.Var('te_dg_ms_chk_added').GetValue|4] expected=[THIS.Var('te_dg_ms_chk_expected').GetValue|4] (a script-only state type on a country static modifier doesn't reach the state: put the measures' census types on a state modifier in each state) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			debug_log_scopes = yes
			remove_variable = te_dg_ms_chk_added
			remove_variable = te_dg_ms_chk_expected
		}
		if = {
			limit = { has_variable = te_dg_ms_chk_before }
			remove_variable = te_dg_ms_chk_before
			remove_variable = te_dg_ms_chk_e0
			remove_variable = te_dg_ms_chk_owner
		}
	}
}
