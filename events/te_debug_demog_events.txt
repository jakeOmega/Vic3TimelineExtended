# ============================================================================
# DEMOGRAPHICS TEST CONSOLE — fired only from the console: `event te_debug_demog.1`
# ============================================================================
# Vic3's console has no `effect` command (that is CK3); `event <id>` is the way
# to run script by hand. Nothing in the mod fires this event. Every option writes
# to debug.log; the effects are in common/scripted_effects/te_debug_demog_effects.txt.
# debug.log rolls over at 512 KB: copy the lines out as they appear
# (docs/guides/scripting_best_practices.md, "Runtime Debugging with debug_log").
#
# What each option is for:
#   a    replay: step the capital's census one year now and log it (TE_DEMOG_REPLAY);
#        `scripts/analysis/demographics_harness.py replay debug.log` checks it against
#        the model. A second run in the same year steps again (the census year goes
#        back by one first) and folds this year's newborns into the over-150 pool:
#        use a save you can throw away, or re-seed after (c)
#   b    start or stop the census log: each 31 December a TE_DEMOG_CENSUS: line per
#        country of a million people or more (scripts/analysis/
#        demographics_observer_report.py reads them), and a TE_DEMOG_PULSE line at the
#        player's capital's yearly state pulse (one step per calendar year?)
#   c    re-seed this country's census from today's population, then take its census
#   d    benchmark (spec §14 Q9's late-game re-run): every state's step in the world,
#        three passes, between TE_DEMOG_BENCH lines; then, untimed, every census is
#        seeded again (the passes leave them wrong)
#   e    war dead (spec §14 Q4): the player's dead in each war, its monthly tally and
#        the states' pending dead (TE_DEMOG_WARDEAD)
#   f/g  skew check (spec §14 Q11): f logs the capital's pops (TE_DEMOG_SKEW) and adds
#        a 60-day +0.05 working-adult ratio modifier; g logs them again a month later
#   h-k  fast mode (te_demog_fast_effects.txt): demography, research and construction
#        at 4x (h), 6x (i) or 12x (j), for iterating on the model; k goes back to normal
#        speed. A test tool, never for play: the first switch re-seeds every census, and
#        the census year then runs on its own clock. Fire it on day 1, then `observe`.
#   l    the census's mortality and fertility modifier types as the capital and an
#        unincorporated state read them (TE_DEMOG_MODS), and phase 2 step 1's work figures,
#        working-adult families and participation inputs (TE_DEMOG_WORK)
#   m    the shock terms (probe P1): every devastated or polluted state's reads
#        (TE_DEMOG_SHOCK)
#   n    the census's rates (probe P3): the capital's births and deaths terms, the
#        on-top scales of its last step and its reads (TE_DEMOG_RATES)
#   o    the engine-scaling probe (Task 9 of the late-marriage plan): a script-only type on
#        the capital's state modifier at multiplier 0.62 (TE_DEMOG_PROBE), logged before and
#        a week after by the hidden te_debug_demog.2
#   q    the owner-change probe (the late-marriage plan, Stop B's I1): one of our states, alone
#        in its region, given to a neighbour and logged before, in on_state_owner_change and a
#        day after by the hidden te_debug_demog.4 (TE_DEMOG_PROBE_OWNER). It really changes
#        hands: throwaway games only.
#   p    call 3's carriers (phase 2 step 1): Punishment-Focused Criminal Justice if needed with Community Service,
#        Woman Soldiers at half strength and the Trade Unions pushed past loyal, all for 60 days; the capital's
#        TE_DEMOG_MODS and TE_DEMOG_WORK lines now, a day and a month later (the hidden te_debug_demog.3)
# Hidden, fired by script in every game: te_debug_demog.5 (the Primogeniture self-check) and .6, .7 (phase 2 step 1's
# workforce self-checks; te_debug_demog_wf_selfchecks). Each writes one TE_SELF_CHECK line a game.
#
# An option's tooltip walks its effects each time the open event renders it, testing
# the limits and writing nothing. Options a, c, d and e walk a census step, the walks
# or every state and war, and their tooltips wrote about 1,200 error lines a second
# while the event was open (2026-10-09). Each sets te_dg_dbg_click and does its work
# only under has_variable = te_dg_dbg_click: the tooltip pass never sets it, so it
# skips the work, and the event now logs nothing while open (confirmed 2026-10-09;
# docs/guides/scripting_best_practices.md, "An Option's Tooltip Walks Its Effects").
# ============================================================================
namespace = te_debug_demog

		custom_tooltip = {

			set_variable = te_dg_dbg_click

		}
	}

		}
		custom_tooltip = {
			text = te_debug_demog.1.f.tt

				te_debug_demog_skew_lines = yes

		trigger = {

		}

			capital = {

			}
		}
	}

	# ---- h-k: fast mode -----------------------------------------------------
	option = {

		custom_tooltip = {
			text = te_debug_demog.1.h.tt

			set_variable = te_dg_dbg_click

# The probe's second half (option o): a week after the add, the state's sum and the probe off again.
te_debug_demog.2 = {
	type = state_event
	hidden = yes

	immediate = {
		debug_log = "TE_DEMOG_PROBE after absorbed_b=[THIS.ScriptValue('te_debug_demog_line_absorbed_b')|5] date=[TimeKeeper.GetCurrentDate.GetString]"
		debug_log_scopes = yes
		if = {
			limit = { has_modifier = te_debug_demog_scaled_probe }
			remove_modifier = te_debug_demog_scaled_probe
		}
	}
}

# Option p's later reads (phase 2 step 1): the capital's TE_DEMOG_MODS and TE_DEMOG_WORK lines a day after the setup
# (the engine has summed the new modifiers; the pair waits for its refresh) and a month after (the monthly refresh has
# run). Fired only by te_debug_demog_call3_setup.
te_debug_demog.3 = {
	type = country_event
	hidden = yes

	immediate = {
		capital ?= { te_debug_demog_mods_line = yes }
	}
}

# The owner-change probe's last line (option q): a day after the transfer, every state now in the stamped region, then
# the stamps and the hook's flag off again.
te_debug_demog.4 = {
	type = country_event
	hidden = yes

	immediate = {
		every_state = {
			limit = {
				state_region = { has_variable = te_probe_owner_region }
			}
			debug_log = "TE_DEMOG_PROBE_OWNER after name=[THIS.GetState.GetNameNoFormatting] id=[THIS.GetState.GetID] owner=[THIS.GetState.GetOwner.GetTag] var=[THIS.ScriptValue('te_debug_demog_probe_owner_var')|0] year=[THIS.ScriptValue('te_debug_demog_probe_owner_year')|0] lm=[THIS.ScriptValue('te_debug_demog_probe_owner_lm')|3] date=[TimeKeeper.GetCurrentDate.GetString]"
			debug_log_scopes = yes
			if = {
				limit = { has_variable = te_probe_owner_var }
				remove_variable = te_probe_owner_var
			}
		}
		every_state = {
			limit = {
				state_region = { has_variable = te_probe_owner_region }
			}
			state_region = { remove_variable = te_probe_owner_region }
		}
		remove_global_variable = te_probe_owner_on
		remove_global_variable = te_probe_owner_to
	}
}

# The Primogeniture self-check's second half (te_debug_demog_pr_selfcheck_start): runs once a game, in every game, a
# week after the first fresh Primogeniture line. OK: the engine summed the script-only type scaled by the agrarian
# share. BROKEN: swap Primogeniture to Plan B (docs/superpowers/plans/2026-10-11-demographics-late-marriage.md, Task 9).
te_debug_demog.5 = {
	type = state_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = te_inh_pr_chk_before }
			if = {
				limit = { te_debug_demog_pr_chk_gap < 0.002 }
				debug_log = "TE_SELF_CHECK OK primogeniture_scaled added=[THIS.ScriptValue('te_debug_demog_pr_chk_added')|4] expected=[THIS.ScriptValue('te_debug_demog_pr_chk_expected')|4] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN primogeniture_scaled added=[THIS.ScriptValue('te_debug_demog_pr_chk_added')|4] expected=[THIS.ScriptValue('te_debug_demog_pr_chk_expected')|4] (0.1 or 0: the engine doesn't scale a script-only type on a state modifier; take Plan B, late-marriage plan Task 9) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			debug_log_scopes = yes
			remove_variable = te_inh_pr_chk_before
		}
	}
}

# Call 3's carrier-kind self-checks' second halves (te_debug_demog_wf_selfchecks): once a game, in every game where the
# census runs, a week after a country first held the carrier, on its capital. A carrier gone in the week re-arms the
# check for a later month.
# .6: Community Service or Mandatory Minimums. OK: the engine sums a script-only type from an amendment into a state's
# read. BROKEN: read equals laws_only, so it doesn't; revert both amendments' blocks to their engine field
# (docs/testing/demographics-playtest.md, "Self-checks").
te_debug_demog.6 = {
	type = state_event
	hidden = yes

	immediate = {
		if = {
			limit = { owner = { te_debug_demog_wf_amendment_held = yes } }
			if = {
				limit = { te_debug_demog_wf_amend_gap < 0.002 }
				debug_log = "TE_SELF_CHECK OK workforce_amendment_type read=[THIS.ScriptValue('te_debug_demog_wf_part_m_read')|3] expected=[THIS.ScriptValue('te_debug_demog_wf_amend_expected')|3] laws_only=[THIS.ScriptValue('te_debug_demog_wf_amend_laws_only')|3] date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN workforce_amendment_type read=[THIS.ScriptValue('te_debug_demog_wf_part_m_read')|3] expected=[THIS.ScriptValue('te_debug_demog_wf_amend_expected')|3] laws_only=[THIS.ScriptValue('te_debug_demog_wf_amend_laws_only')|3] (read = laws_only: the engine doesn't sum a script-only type from an amendment; revert Community Service's and Mandatory Minimums' blocks, docs/testing/demographics-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
			}
			debug_log_scopes = yes
		}
		else = {
			remove_global_variable = te_demog_selfcheck_amend_done
		}
	}
}
