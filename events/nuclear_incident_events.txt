namespace = nuclear_incident

# ============================================================================
# NUCLEAR INCIDENTS — the dangerous-posture chains
# ============================================================================
# Design: docs/systems/nuclear_crisis_design.md §8 (§0.4 is what shipped).
#
# One incident roll per armed country per month (nd_roll_incident); the family
# is drawn in nd_fire_incident from those the country is eligible for. Most
# incidents end in expense, lost readiness, scandal or alarm. A launch is a
# conditional terminal branch that needs weapons, a war with the victim, and a
# prior choice of authority — or an explicit order. Outside a war, a launch
# branch is an intercepted order (nd_launch_or_intercept).
#
#   .1   Conflicting Indications       the government decides (central authority)
#   .2   The Order Nobody Passed       launch on warning / delegation: someone halted it
#   .3   The Inquiry                   what caused it, and what we make of it
#   .4   After the Launch              a launch happened: struck, or recalled
#   .5   Their Launchers Were Ready    to the country that was nearly struck
#   .10  The Exercise They Mistook     Able Archer
#   .11  Is This the Real Thing?       .10's player opponent chooses its own alert
#   .12  They Showed Us the Timetable  .10's invitation, to the crisis issuer
#   .20  Silence from the Capital      the isolated commander (Arkhipov and B-59)
#   .30  The Cost of Permanent Alert   an accident with weapons aboard
#   .40  A Routine Mishap              the fallback: klaxons, bears, tornadoes
#   .50  The Monopoly Window           nobody can answer us in this war
#   .60  A Secret Comes Out            a concealed incident is exposed
#
# Historical references inform the structure and flavour, never tags or dates
# (§14): nothing here requires a particular country, bloc or year.
# ============================================================================

# ---- .1 Conflicting Indications ------------------------------------------------
# The early-warning network reports launches from scope:nd_warning_suspect; a
# second channel has not corroborated them, and the duty officer says so. The
# text never calls it a false alarm: that is for the inquiry (.3).
# nd_warning_cause is set but not revealed.
nuclear_incident.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_false_alarm_panic.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	title = nuclear_incident.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					nd_in_crisis = yes
					var:nd_crisis_opponent ?= scope:nd_warning_suspect
				}
				desc = nuclear_incident.1.d_crisis
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_incident.1.d
			}
		}
	}
	flavor = nuclear_incident.1.f

	duration = 3

	trigger = {
		exists = scope:nd_warning_suspect
	}

	# Trust the officer: wait for corroboration.
	option = {
		name = nuclear_incident.1.a
		default_option = yes
		custom_tooltip = nuclear_incident.1.a.tt
		trigger_event = { id = nuclear_incident.3 days = 3 }
		ai_chance = {
			base = 5
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 3
			}
		}
	}

	# Scramble the bombers and call them.
	option = {
		name = nuclear_incident.1.b
		add_modifier = {
			name = nd_exercise_cost
			multiplier = sv_money_flow_event_medium
			days = 60
		}
		custom_tooltip = {
			text = nd_tt_strain_up_10
			change_variable = { name = nd_strain add = 10 }
			clamp_variable = { name = nd_strain min = 0 max = 100 }
		}
		scope:nd_warning_suspect = {
			change_relations = { country = ROOT value = -10 }
		}
		if = {
			limit = {
				nd_in_crisis = yes
				var:nd_crisis_opponent ?= scope:nd_warning_suspect
			}
			custom_tooltip = nd_tt_crisis_turns_acute
			hidden_effect = {
				nd_crisis_save_parties = yes
				scope:nd_issuer ?= {
					set_variable = { name = nd_crisis_stage value = 3 }
				}
			}
		}
		trigger_event = { id = nuclear_incident.3 days = 5 }
		ai_chance = {
			base = 3
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 2
			}
			modifier = {
				trigger = { nd_in_crisis = yes }
				add = 2
			}
		}
	}

	# Execute the warning-response orders. The only way a warning under central
	# authority ends in a launch: somebody has to give the order.
	option = {
		name = nuclear_incident.1.c
		trigger = { nd_is_armed = yes }
		highlighted_option = yes
		set_variable = { name = nd_warning_launched value = 1 }
		nd_launch_or_intercept = { VICTIM = scope:nd_warning_suspect NARRATE = yes KIND = warning }
		trigger_event = { id = nuclear_incident.3 days = 10 }
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					nd_regime_is_militarist = yes
					ruler_is_aggressive = yes
					has_war_with = scope:nd_warning_suspect
				}
				add = 1
			}
		}
	}
}

# ---- .2 The Order Nobody Passed ------------------------------------------------
# Launch on warning (or delegated authority in a war with the suspect): the
# warning reached the launch crews before it reached the government, and the
# standing orders said fire. Someone in the chain refused. There was no
# decision for the government to make — nd_hold_chance was the protection.
nuclear_incident.2 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_false_alarm_panic.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	title = nuclear_incident.2.t
	desc = nuclear_incident.2.d
	flavor = nuclear_incident.2.f

	duration = 3

	trigger = {
		exists = scope:nd_warning_suspect
	}

	# Leave the standing orders as they are.
	option = {
		name = nuclear_incident.2.a
		default_option = yes
		trigger_event = { id = nuclear_incident.3 days = 3 }
		ai_chance = {
			base = 3
			modifier = {
				trigger = { nd_regime_is_militarist = yes }
				add = 3
			}
		}
	}

	# Take the decision back to the capital: central authority, now.
	option = {
		name = nuclear_incident.2.b
		nd_set_authority_1 = yes
		every_interest_group = {
			limit = { nd_ig_class_warfighting = yes }
			add_modifier = {
				name = ig_approval_negative_modifier
				days = short_modifier_time
				is_decaying = yes
			}
		}

		ai_chance = {
			base = 3
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 3
			}
		}
	}
}

		first_valid = {
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 1 } }
				desc = nuclear_incident.3.d_moonrise
			}
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 2 } }
				desc = nuclear_incident.3.d_tape
			}
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 3 } }
				desc = nuclear_incident.3.d_chip
			}
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 4 } }
				desc = nuclear_incident.3.d_relay
			}
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 5 } }
				desc = nuclear_incident.3.d_clouds
			}
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 6 } }
				desc = nuclear_incident.3.d_solar
			}
			triggered_desc = {
				trigger = { nd_warning_cause_is = { C = 7 } }
				desc = nuclear_incident.3.d_rocket
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_incident.3.d_coincidence
			}
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = nd_warning_launched
					var:nd_warning_launched = 1
				}
				desc = nuclear_incident.3.f_launched

			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_incident.3.f
			}
		}
	}

	duration = 3

	trigger = {
		has_variable = nd_warning_cause
	}

	# Honour the scepticism.
	option = {
		name = nuclear_incident.3.a
		default_option = yes
		nd_inquiry_reward_skeptic = yes
		ai_chance = {
			base = 4
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 2
			}
			modifier = {
				trigger = { country_is_democratic = yes }
				add = 2
			}
		}
	}

	# Bury it.
	option = {
		name = nuclear_incident.3.b
		nd_inquiry_conceal = yes
		ai_chance = {
			base = 2
			modifier = {
				trigger = { country_is_democratic = no }
				add = 2
			}
		}
	}

	# Orders are orders.
	option = {
		name = nuclear_incident.3.c
		nd_inquiry_punish_skeptic = yes
		ai_chance = {
			base = 1
			modifier = {
				trigger = { nd_regime_is_militarist = yes }
				add = 3
			}
		}
	}

	after = {
		remove_variable = nd_warning_cause
		if = {
			limit = { has_variable = nd_warning_launched }
			remove_variable = nd_warning_launched
		}
	}
}

# ---- .5 Their Launchers Were Ready ----------------------------------------------
# To scope:nd_scare_victim: another country's launch was recalled at the last
# moment, and it was aimed at us. scope:nd_interceptor is that country.
# An armed victim that is free to may answer with a private warning (option c,
# nd_crisis_open with itself as issuer). nd_intercepted_launch used to open
# that crisis by itself; the AI still picks it most of the time.
nuclear_incident.5 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_proliferation_threat.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/mushroom_cloud.dds"

	title = nuclear_incident.5.t
	desc = nuclear_incident.5.d
	flavor = nuclear_incident.5.f

	duration = 3

	trigger = {
		exists = scope:nd_interceptor
	}

	option = {
		name = nuclear_incident.5.a
		default_option = yes
	}

	# Armed: answer the alert with ours.
	option = {
		name = nuclear_incident.5.b
		trigger = {
			nd_is_armed = yes
			nd_readiness_locked = no
			has_variable = nd_readiness_target
			var:nd_readiness_target < 3
		}
		change_variable = { name = nd_readiness_target add = 1 }
		custom_tooltip = nd_tt_readiness_step_up
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 2
			}
		}
	}

	# Armed and free: warn them privately — stand down, or else. The same
	# private crisis nd_nuclear_warning_action opens, minus its dispute test:
	# the recalled launch is the grievance (it classifies as an alert dispute
	# when nothing weightier applies), and minus its two-year cooldown, which
	# guards against farming a chosen action, not against a new near-launch.
	option = {
		name = nuclear_incident.5.c
		trigger = {
			nd_is_armed = yes
			nd_crisis_parties_free = { TARGET = scope:nd_interceptor }
			NOT = { nd_has_nonuse_pledge_with = { TARGET = scope:nd_interceptor } }
		}
		nd_crisis_open = { TARGET = scope:nd_interceptor PUBLIC = no }
		ai_chance = {
			base = 12
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 4
			}
		}
	}
}

# ---- .10 The Exercise They Mistook ----------------------------------------------
# During a crisis past the warning stage, our exercise or alert looked to the
# other side like the cover for a first strike. Scaling back does not erase
# their alarm at once.
#
# The other side's readiness is its own choice. An AI opponent goes to its
# highest alert here, as its posture allows; a player opponent is asked
# (.11), and the desc says it is deciding. The desc claims their alert only
# when it is really at High Alert; an opponent that is unarmed or locked
# (a recent stand-down) is alarmed but does not move. Telling them in advance (option a)
# opens talks on our own record if we issued the crisis; if they did, it
# invites them to (.12) rather than writing their record for them.
nuclear_incident.10 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/joint_military_exercises.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = nuclear_incident.10.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:nd_exercise_opponent ?= { te_ea_nd_opponent_decides_alert = yes } }
				desc = nuclear_incident.10.d_deciding
			}
			triggered_desc = {
				trigger = {
					scope:nd_exercise_opponent ?= {
						has_variable = nd_readiness_target
						var:nd_readiness_target = 3
					}
				}
				desc = nuclear_incident.10.d
			}
			# Unarmed, or its readiness locked by a stand-down: alarm, no alert.
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_incident.10.d_alarmed
			}
		}
	}
	flavor = nuclear_incident.10.f

	duration = 3

	trigger = {
		nd_in_crisis = yes
		exists = var:nd_crisis_opponent
	}

	immediate = {
		nd_crisis_save_parties = yes
		save_scope_as = nd_exerciser
		var:nd_crisis_opponent = {
			save_scope_as = nd_exercise_opponent
		}
		# They respond as their posture allows: an AI at once, a player by
		# choice (.11, keyed to this crisis).
		if = {
			limit = { scope:nd_exercise_opponent = { te_ea_nd_opponent_decides_alert = yes } }
			nd_crisis_send_event = { WHO = scope:nd_exercise_opponent EVENT = nuclear_incident.11 }
		}
		else = {
			scope:nd_exercise_opponent = {
				if = {
					limit = {
						is_ai = yes
						nd_is_armed = yes
						nd_readiness_locked = no
						has_variable = nd_readiness_target
					}
					set_variable = { name = nd_readiness_target value = 3 }
				}
			}
		}
		scope:nd_issuer ?= {
			set_variable = { name = nd_crisis_stage value = 3 }
		}
	}

	# Tell them in advance; invite observers.
	option = {
		name = nuclear_incident.10.a
		default_option = yes
		scope:nd_exercise_opponent = {
			change_relations = { country = ROOT value = 10 }
		}
		# Our own record, if the crisis is ours; theirs is theirs to open (.12).
		if = {
			limit = { nd_is_crisis_owner = yes }
			custom_tooltip = nd_tt_exercise_notified
			hidden_effect = {
				set_variable = { name = nd_crisis_talks value = 1 }
			}
		}
		else_if = {
			limit = {
				scope:nd_issuer ?= {
					has_variable = nd_crisis_talks
					var:nd_crisis_talks = 0
				}
			}
			custom_tooltip = nd_tt_exercise_invited
			hidden_effect = {
				nd_crisis_send_event = { WHO = scope:nd_issuer EVENT = nuclear_incident.12 }
			}
		}
		else = {
			custom_tooltip = nd_tt_exercise_notified_talks_open
		}
		every_interest_group = {
			limit = { nd_ig_class_warfighting = yes }
			add_modifier = {
				name = ig_approval_negative_modifier
				days = short_modifier_time
				is_decaying = yes
			}
		}
		ai_chance = {
			base = 4
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 3
			}
		}
	}

	# Scale it back.
	option = {
		name = nuclear_incident.10.b
		nd_readiness_step_down = yes
		custom_tooltip = {
			text = nd_tt_credibility_down_3
			nd_change_credibility = { AMOUNT = -3 }
		}
		ai_chance = {
			base = 3
		}
	}

	# Carry on: it plays well at home, and it keeps them guessing.
	option = {
		name = nuclear_incident.10.c
		every_interest_group = {
			limit = { nd_ig_class_warfighting = yes }
			add_modifier = {
				name = ig_approval_positive_modifier
				days = short_modifier_time
				is_decaying = yes
			}
		}
		hidden_effect = {
			scope:nd_issuer ?= {
				set_variable = { name = nd_crisis_exercise_weeks value = 6 }
			}
			# Only the threatening side's own exercise presses the target.
			if = {
				limit = { nd_is_crisis_owner = yes }
				scope:nd_issuer ?= { set_variable = { name = nd_crisis_exercise_by_issuer value = 1 } }
			}
			else = {
				scope:nd_issuer ?= { set_variable = { name = nd_crisis_exercise_by_issuer value = 0 } }
			}
		}
		custom_tooltip = nd_tt_exercise_continued
		ai_chance = {
			base = 2
			modifier = {
				trigger = { nd_regime_is_militarist = yes }
				add = 3
			}
		}
	}

	# Use their fear: press our demands while they believe we might strike.
	option = {
		name = nuclear_incident.10.e
		trigger = { nd_is_crisis_owner = yes }
		highlighted_option = yes
		custom_tooltip = {
			text = nd_tt_credibility_up_5
			nd_change_credibility = { AMOUNT = 5 }
		}
		change_infamy = 5
		hidden_effect = {
			set_variable = { name = nd_crisis_exercise_weeks value = 8 }
			set_variable = { name = nd_crisis_exercise_by_issuer value = 1 }
			set_variable = { name = nd_crisis_pressure_weeks value = nd_crisis_pressure_interval_weeks }
		}
		custom_tooltip = nd_tt_exercise_exploited
		ai_chance = {
			base = 1
			modifier = {
				trigger = { nd_doctrine_at_least_compellence = yes }
				add = 3
			}
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 2
			}
		}
	}
}

# ---- .11 Is This the Real Thing? ----------------------------------------------
# To a PLAYER crisis opponent of a country whose exercise we took for cover
# for a first strike (.10, sent through nd_crisis_send_event, so it lapses if
# the crisis ends first). An AI opponent goes to high alert in .10 itself;
# a player decides. scope:nd_exerciser is the country exercising.
nuclear_incident.11 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_standoff_tension.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = nuclear_incident.11.t
	desc = nuclear_incident.11.d
	flavor = nuclear_incident.11.f

	duration = 3

	trigger = {
		exists = scope:nd_exerciser
		nd_crisis_event_matches = yes
	}

	# Go to our highest alert.
	option = {
		name = nuclear_incident.11.a
		trigger = {
			nd_is_armed = yes
			nd_readiness_locked = no
		}
		if = {
			limit = { nd_crisis_event_matches = yes }

		}
		else = {
			custom_tooltip = nd_tt_crisis_stale
		}
		ai_chance = {
			base = 3
		}
	}

	# Hold our posture and ask them what they are doing.
	option = {
		name = nuclear_incident.11.b
		default_option = yes
		custom_tooltip = nd_tt_alert_held
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 2
			}
		}
	}
}

# ---- .12 They Showed Us the Timetable ------------------------------------------
# To the crisis ISSUER, when its target answered .10 by sending the exercise

	type = country_event
	placement = root

# ---- .20 Silence from the Capital ------------------------------------------------
# Delegated authority, in a war or an acute crisis. A unit with weapons aboard
# (a submarine, a bomber wing or a missile battery, by technology) has lost
# contact, is under attack or believes it is, and its commander thinks the
# war has begun. What happened is rolled in `immediate` with no further veto:
# the player delegated this (§8.3). Held — a second officer refused; not held —
# the weapon was used (in a war) or the order was recalled (outside one).
nuclear_incident.20 = {
	type = country_event
	placement = root

	event_image = {
		trigger = {
			has_variable = nd_ic_unit
			var:nd_ic_unit = 1

	event_image = { texture = "gfx/event_pictures/nuclear_standoff_tension.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = nd_launch_held
					var:nd_launch_held = 1
				}
				desc = nuclear_incident.20.t_held
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_incident.20.t_fired
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = nd_launch_held
					var:nd_launch_held = 1
					var:nd_ic_unit = 1
				}
				desc = nuclear_incident.20.d_held_sub
			}
			triggered_desc = {
				trigger = {
					has_variable = nd_launch_held
					var:nd_launch_held = 1
				}
				desc = nuclear_incident.20.d_held
			}
			triggered_desc = {
				trigger = {
					has_variable = nd_last_launch_kind
					var:nd_last_launch_kind = 1

				trigger = { always = yes }
				desc = nuclear_incident.20.d_fired_recalled

			triggered_desc = {

				}

			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_incident.20.f_fired

		nd_is_armed = yes
		nd_authority_delegated_or_warning = yes

			limit = { has_technology_researched = advanced_submarine_technology }
			set_variable = { name = nd_ic_unit value = 1 }
		}
		else_if = {
			limit = { has_technology_researched = ICBMs }
			set_variable = { name = nd_ic_unit value = 3 }
		}
		else = {
			set_variable = { name = nd_ic_unit value = 2 }
		}

		# opponent.

			limit = {
				any_country = { has_war_with = scope:nd_ic_self }
			}
			random_country = {
				limit = { has_war_with = scope:nd_ic_self }
				save_scope_as = nd_ic_enemy
			}

		else_if = {
			limit = { nd_in_crisis = yes }

			nd_launch_or_intercept = { VICTIM = scope:nd_ic_enemy NARRATE = no KIND = commander }
		}

			set_variable = { name = nd_launch_held value = 1 }

			base = 3

				add = 2
			}
		}
	}

	option = {
		name = nuclear_incident.20.b
		trigger = {
			var:nd_launch_held = 1
			var:nd_safeguards < 3
		}
		nd_step_investment = { VAR = nd_safeguards OP = add }
		ai_chance = {
			base = 2
		}
	}

	# Held: bury it.
	option = {
		name = nuclear_incident.20.c
		trigger = { var:nd_launch_held = 1 }
		nd_inquiry_conceal = yes
		ai_chance = {
			base = 1
			modifier = {
				trigger = { country_is_democratic = no }
				add = 2
			}
		}
	}

	# Fired: take the weapons back under central control.
	option = {
		name = nuclear_incident.20.e
		trigger = { var:nd_launch_held = 0 }
		default_option = yes
		nd_set_authority_1 = yes
		ai_chance = {
			base = 3
		}
	}

	# Fired: stand behind the commander.
	option = {
		name = nuclear_incident.20.g
		trigger = { var:nd_launch_held = 0 }
		every_interest_group = {
			limit = { nd_ig_class_warfighting = yes }
			add_modifier = {
				name = ig_approval_positive_modifier
				days = short_modifier_time
				is_decaying = yes
			}
		}
		every_interest_group = {
			limit = { nd_ig_class_dove = yes }
			add_modifier = {
				name = ig_approval_negative_modifier
				days = normal_modifier_time
				is_decaying = yes
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = { nd_regime_is_militarist = yes }
				add = 3
			}
		}
	}
