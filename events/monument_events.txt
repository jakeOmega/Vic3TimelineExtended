# ============================================================================
# GRAND MONUMENTS
# ============================================================================
# Event 1       - a level finished (hidden building_event, fires from
#                 on_building_built): on a construction site, the AI's
#                 ceremony once nothing more is queued, then Event 20.
# Event 2       - the dedication ceremony (state event): a player's from the
#                 Dedicate button (gm_dedicate_sgui, owner 2026-10-18), the
#                 AI's once nothing more is queued; Event 11 - the skin,
#                 players only (§1.1).
# Events 3-10,
# 12-15         - recurring anniversary flavour, dispatched from
#                 monument_events_on_actions.txt.
# Events 16-17  - a fallen order's monuments: the contested notice, then one
#                 event per monument (§4.5).
# Event 20      - refresh (hidden plumbing).
# Event 21      - an opportunity offered (opportunities spec §2), from
#                 gm_com_send_offer.
# Events 24-25  - the monument's form and namesake (v2 §3.3), players only.
# Event 26      - an opportunity's gift: its monument rebuilt taller (hidden,
#                 opportunities spec §3.1).
# Event 27      - Raise It: an opportunity's missing levels queued (hidden).
# Event 28      - an opportunity seized: the monument, its gift and the boon,
#                 players only.
# Event 29      - a monument's own work: building beside, the grandest standing
#                 first (hidden, ageing spec 2026-10-16 §3), and a revolt's
#                 split piece (its identity, its return).
# Event 30      - choices on a monument beside another (hidden, §4.3).
# Event 31      - a merged state's monuments adopted from their shadows
#                 (hidden, issue #936).
# Event 32      - an older save's monument migrated to records and a level-1
#                 building, or made a construction site (hidden, sites spec
#                 2026-10-18 §3.5).
#
# POPUPS (owner, 2026-10-19): the dedication's own chain -- the ceremony (2),
# the style (11), the form (24), the namesake (25) and the seized
# opportunity's notice (28) -- opens at once instead of waiting in the
# notification list: each defines `is_popup = yes` and every trigger_event
# that fires one carries `popup = yes`. A player who pressed Dedicate is
# standing at the monument; a card to click first was a step too many.
# Anniversaries (3-10, 12-15) and offers (21) stay quiet cards. How and why:
# docs/guides/event_creation_guide.md § "Making an Event Open at Once".
# ============================================================================

# Event 1: a level finished (hidden plumbing). ROOT is the BUILDING, so
# nothing here adds a multiplier modifier: it hops to the state for vanity
# backlash (§5) and the AI's ceremony, then refreshes the nation through the
# hidden country event.
monument_events.1 = {
	type = building_event
	hidden = yes

	immediate = {
		# A construction site's level (sites spec 2026-10-18 §3.1): the AI's
		# ceremony once nothing more is queued (a player's waits for its
		# Dedicate button), then a refresh. Not for script's own building
		# (gm_rebuilding, one day).
		if = {
			limit = {
				is_building_type = building_grand_monument_site
				exists = owner
			}
			state = {
				if = {
					limit = {
						gm_system_enabled = yes
						NOT = { has_variable = gm_rebuilding }
					}
					if = {
						limit = {
							gm_state_ai_dedicates_now = yes
							NOT = { has_variable = gm_ceremony_pending }
						}
						gm_state_start_ceremony = yes
					}
					owner = { trigger_event = { id = monument_events.20 } }
				}
			}
		}
		# A finished monument's building: script makes it at level 1 (under
		# gm_rebuilding); an older save's may still be finishing queued levels.
		else_if = {
			limit = {
				is_building_type = building_grand_monument
				exists = owner
			}
			state = {
				if = {
					limit = { gm_system_enabled = yes }
					# A gifted monument's dedication back (opportunities spec
					# §3.1), before anything reads it.
					gm_state_restore_dedication = yes
					# Vanity backlash is the construction site's monthly count's
					# (gm_state_count_site_levels, sites spec 2026-10-18): this
					# hook missed levels that finished on the same day.
					# A part of this region merged into this state (issue #936):
					# the monuments that stood there are adopted first, so their
					# levels found no monument beside and start no ceremony.
					if = {
						limit = { gm_state_merge_pending = yes }
						gm_state_adopt_orphans = yes
					}
					# The monument an opportunity completed by this level names
					# (opportunities spec §3).
					owner = { set_variable = { name = gm_com_last_state value = prev } }
					owner = { trigger_event = { id = monument_events.20 } }
				}
			}
		}
	}
}

# Event 2: The Dedication (§3). A STATE event placed on the monument's state;
# ROOT is that state, and every country-level gate goes through `owner`. Each
# option dedicates through a gm_state_dedicate_* wrapper (gm_effects.txt),
# which writes the records, finishes the monument at the levels built
# (gm_levels: a level queued while it asks raises a monument beside it),
# offers the skins that fit (Event 11) and refreshes the nation (Event 20). A
# player's opens from the monument's Dedicate button (gm_dedicate_sgui), only
# with nothing under construction, or from Rededicate; the AI's once nothing
# more is queued.
#
# OPTION GATING: a `trigger` hides a dedication the country cannot raise (a
# crown it does not wear, a faith under State Atheism, an untaught technology);
# ai_chance tilts the rest.
#
# TOOLTIPS: every number is read from the gm_step_* script values

# loc holds no hand-kept numbers.
