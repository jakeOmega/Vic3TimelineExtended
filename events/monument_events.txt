namespace = monument_events

# ============================================================================
# GRAND MONUMENTS
# ============================================================================
# Event 1  - hidden building_event, fires from on_building_built. Hands off to the
#            monument's own STATE, because that is where the ceremony belongs and
#            because building scope supports neither variables nor modifiers.
# Event 2  - the dedication ceremony, a state_event placed on the monument's state.
#            ROOT is that state, so activate_production_method targets it directly
#            and every country-level gate goes through `owner`.
# Events 3+ - recurring flavour, dispatched from monument_events_on_actions.txt.
# ============================================================================

# Event 1: construction finished (hidden plumbing).
# ROOT is the newly-built BUILDING. Hop to its state and let the ceremony fire there.
monument_events.1 = {
	type = building_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_building_group = bg_grand_monuments
				exists = owner
			}
			state = { trigger_event = { id = monument_events.2 } }
		}
	}
}

# Event 2: The Dedication. A STATE event -- the ceremony happens where the monument
# stands, not in the capital, and the event window is placed on that state.
#
# Every option locks its choice in permanently via the unlocking_production_methods
# ratchet (see common/production_methods/grand_monument_pms.txt).
#
# OPTION GATING. Two kinds, and they are not interchangeable:
#   * `trigger` on an option HIDES it. Reserved for dedications the country has no
#     business offering at all -- a state-atheist republic consecrating a shrine, an
#     industry-banned agrarian state glorifying the factory, or a technology it has
#     not researched.
#   * `ai_chance` TILTS it. Everything a country could plausibly want stays visible
#     and is steered by weight instead, so the player always sees the full menu of
#     what is actually possible.
# The civic dedication is never hidden: it is offered under one of three wordings
# depending on who the monument would be raised to. See
# common/scripted_triggers/monument_triggers.txt.
#
# TOOLTIPS. Each option wraps its activate_production_method in a custom_tooltip so
# the player reads what the dedication actually does before committing to it, rather
# than relying on whatever the engine emits for that effect. The .tt strings live in
# localization/english/te_events_l_english.yml and are hand-kept in sync with
# grand_monument_pms.txt; change a value there, change it here.
monument_events.2 = {
	type = state_event
	placement = ROOT

	event_image = { video = "unspecific_world_fair" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = monument_events.2.t
	desc = monument_events.2.d
	flavor = monument_events.2.f

	duration = 3

	# Only for a monument that has not been dedicated yet. `has_building` alone would
	# re-open the question every time an existing monument gains a level.
	trigger = {
		any_scope_building = {
			is_building_type = building_grand_monument
			has_active_production_method = pm_monument_undedicated
		}
	}

	# Named in the description and in every option tooltip.
	immediate = {
		save_scope_as = monument_state
	}

	# Leave it undedicated -- the monument keeps pm_monument_undedicated and the
	# player can still dedicate it by hand in the building panel later.
	option = {
		name = monument_events.2.a
		default_option = yes
		ai_chance = { base = 1 }

		custom_tooltip = monument_events.2.a.tt
	}

	# Civic, republican wording.
	option = {
		name = monument_events.2.b
		trigger = {
			owner = { monument_government_is_republican = yes }
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.b.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_civic
			}
		}
	}

	# Civic, dynastic wording. Same production method as the republican option.
	option = {
		name = monument_events.2.l
		trigger = {
			owner = { monument_government_is_crowned = yes }
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.l.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_civic
			}
		}
	}

	# Civic, neutral wording -- theocracies, corporate states, colonial
	# administrations, neocameralist states, and any governance principle added later.
	option = {
		name = monument_events.2.m
		trigger = {
			owner = {
				monument_government_is_republican = no
				monument_government_is_crowned = no
			}
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.m.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_civic
			}
		}
	}

	# Religious. Hidden only under state atheism, where there is no faith to
	# consecrate it to; a secular state may still build a cathedral, it is just
	# an unlikely use of the money.
	option = {
		name = monument_events.2.c
		trigger = {
			owner = {
				NOT = { has_law_or_variant = law_type:law_state_atheism }
			}
		}

		ai_chance = {
			base = 3
			modifier = {
				trigger = {
					owner = { has_law_or_variant = law_type:law_total_separation }
				}
				add = -2
			}
			modifier = {
				trigger = {
					owner = { has_law_or_variant = law_type:law_state_religion }
				}
				add = 2
			}
		}

		custom_tooltip = {
			text = monument_events.2.c.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_religious
			}
		}
	}

	# War memorial. Every country has fallen soldiers to name, so this is never
	# hidden -- but a country that has just been through a war reaches for it.
	option = {
		name = monument_events.2.e

		ai_chance = {
			base = 3
			modifier = {
				trigger = {
					owner = { is_at_war = yes }
				}
				add = 3
			}
		}

		custom_tooltip = {
			text = monument_events.2.e.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_war_memorial
			}
		}
	}

	# Artistic.
	option = {
		name = monument_events.2.g
		trigger = {
			owner = { has_technology_researched = romanticism }
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.g.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_artistic
			}
		}
	}

	# Naturalist.
	option = {
		name = monument_events.2.h
		trigger = {
			owner = { has_technology_researched = romanticism }
		}

		ai_chance = {
			base = 3
			# The state itself has industry worth planting over.
			modifier = {
				trigger = {
					pollution_generation >= 200
				}
				add = 3
			}
		}

		custom_tooltip = {
			text = monument_events.2.h.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_naturalist
			}
		}
	}

	# Scientific.
	option = {
		name = monument_events.2.i
		trigger = {
			owner = { has_technology_researched = empiricism }
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.i.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_scientific
			}
		}
	}

	# Industrial. Hidden where industry itself is outlawed -- a great exhibition
	# hall is not something an industry-banned state puts up.
	option = {
		name = monument_events.2.j
		trigger = {
			owner = {
				has_technology_researched = marketing_research
				NOT = { has_law_or_variant = law_type:law_industry_banned }
			}
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.j.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_industrial
			}
		}
	}

	# Athletic.
	option = {
		name = monument_events.2.k
		trigger = {
			owner = { has_technology_researched = television_broadcasting }
		}

		ai_chance = { base = 3 }

		custom_tooltip = {
			text = monument_events.2.k.tt
			activate_production_method = {
				building_type = building_grand_monument
				production_method = pm_monument_athletic
			}
		}
	}

}

		}

	}
}
