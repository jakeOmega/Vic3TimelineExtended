# Construction-market subsystem.
# Based on the third-party "Free Market Construction" mod. See README.md.

namespace = te_construction_market_recalc_events

# Border / geopolitical recalc — fires on country-shape changes (market
# merge/create, country released, country formed, capitulation, diplo-play
# back-down, revolution/secession, war end). Re-runs the per-country state
# update so construction-site placements track the new border layout.
te_construction_market_recalc_events.1 = {
	type = country_event
	hidden = yes

	immediate = {
		te_construction_market_on_border_changes = yes
	}
}

# One-shot per-country setup — runs once at game start (history file);
# on_country_formed fires .1, the border recalc. Initializes the public buy-target variable and creates
# the construction-site building in the capital if the country has none.
te_construction_market_recalc_events.2 = {
	type = country_event
	hidden = yes

	immediate = {
		te_construction_market_country_setup = yes
	}
}

# Construction Cost Scaling at game start (te_cc_game_start, from on_game_started_after_lobby): the small-economy
# rule's factor for this country, from the first day. A country event so that the modifier's multiplier resolves
# against the country (ROOT is none in the on_action). No trigger: it also queues itself a week on for a country
# whose GDP reads 0, and a delayed event with a false trigger is dropped (#1596).
te_construction_market_recalc_events.3 = {
	type = country_event
	hidden = yes

	immediate = {
		te_cc_game_start_country = yes
	}
}
