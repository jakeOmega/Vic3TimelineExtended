nuclear_weapon_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_city_destruction.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	title = nuclear_weapon_events.1.t
	desc = nuclear_weapon_events.1.d
	flavor = nuclear_weapon_events.1.f

	duration = 3

	trigger = {
		# triggered by nuclear strike
	}

	immediate = {
		post_notification = industrial_nuke_succeeds_defender
	}

	option = {
		name = nuclear_weapon_events.1.a
		default_option = yes
		ai_chance = {
			base = 1
		}
	}

	# Retaliate (#430): the gates and bookkeeping nuke_diplo_action puts on any
	# launch — an arsenal no disarmament binds, still at war with the striker
	# (retaliation is permitted under every doctrine), no pledge in the way —
	# and the use recorded once per salvo with nd_record_nuclear_use. The
	# strike stays nuclear_response_strike: nd_dispatch_strategic_strike would
	# book a retaliation as a first strike (+25 infamy, the first-strike UN
	# dossier and ICC entries). This event is triggered in the struck state's
	# scope, so every gate names the retaliator explicitly: scope:target_country
	# (the struck country), answering scope:attacking_country. The AI retaliates
	# only when nd_ai_nuclear_use_justified agrees, like the strike actions.
	option = {
		name = nuclear_weapon_events.1.b
		trigger = {
			scope:target_country = {
				has_modifier = nuclear_power
				var:nuclear_weapon_stockpile > 0
				nd_retaliation_permitted = { ENEMY = scope:attacking_country }
			}
		}
		show_as_unavailable = {
			scope:target_country = {
				has_modifier = nuclear_power
				var:nuclear_weapon_stockpile > 0
			}
		}
		highlighted_option = yes

		ai_chance = {
			base = 8
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 8
			}
			modifier = {
				trigger = {
					has_variable = nuclear_weapon_stockpile
					var:nuclear_weapon_stockpile <= nuclear_ai_desired_stockpile
				}
				add = 10
			}
			modifier = {
				trigger = { var:nuclear_weapon_stockpile < 6 }
				add = 6
			}
			modifier = {
				trigger = {
					ruler_is_aggressive = yes
					var:nuclear_weapon_stockpile > nuclear_ai_desired_stockpile
				}
				add = -6
			}
			# Last: a factor applies to what the modifiers above have added.
			modifier = {
				trigger = {
					NOT = { scope:target_country = { nd_ai_nuclear_use_justified = { ENEMY = scope:attacking_country } } }
				}
				factor = 0
			}
		}
		scope:attacking_country = {
			ordered_scope_state = {
				order_by = nuclear_industrial_strike_target_score
				position = 0
				nuclear_response_strike = yes
			}
		}
		scope:target_country = {
			nd_record_nuclear_use = { VICTIM = scope:attacking_country }
		}
	}

	# Full retaliation: as option b, three warheads, one record.
	option = {
		name = nuclear_weapon_events.1.c
		trigger = {
			scope:target_country = {
				has_modifier = nuclear_power
				var:nuclear_weapon_stockpile > 2
				nd_retaliation_permitted = { ENEMY = scope:attacking_country }
			}
		}
		show_as_unavailable = {
			scope:target_country = {
				has_modifier = nuclear_power
				var:nuclear_weapon_stockpile > 2
			}
		}
		highlighted_option = yes

		ai_chance = {
			base = 3
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 10
			}
			modifier = {
				trigger = { is_at_war = yes }
				add = 8
			}
			modifier = {
				trigger = { var:nuclear_weapon_stockpile > nuclear_ai_desired_stockpile }
				add = 12
			}
			modifier = {
				trigger = { var:nuclear_weapon_stockpile > 10 }
				add = 8
			}
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = -10
			}
			modifier = {
				trigger = { var:nuclear_weapon_stockpile <= nuclear_ai_desired_stockpile }
				add = -8
			}
			# Last: a factor applies to what the modifiers above have added.
			modifier = {
				trigger = {
					NOT = { scope:target_country = { nd_ai_nuclear_use_justified = { ENEMY = scope:attacking_country } } }
				}
				factor = 0
			}
		}

		scope:attacking_country = {
			ordered_scope_state = {
				order_by = nuclear_industrial_strike_target_score
				position = 0
				nuclear_response_strike = yes
			}

			ordered_scope_state = {
				order_by = nuclear_industrial_strike_target_score
				position = 1
				nuclear_response_strike = yes
			}

			ordered_scope_state = {
				order_by = nuclear_industrial_strike_target_score
				position = 2
				nuclear_response_strike = yes
			}
		}
		scope:target_country = {
			nd_record_nuclear_use = { VICTIM = scope:attacking_country }
		}
	}
}

nuclear_weapon_events.2 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/tactical_nuclear_strike.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	title = nuclear_weapon_events.2.t
	# A launch out of an incident (nd_launch_or_intercept) was nobody's
	# considered retribution: the unconfirmed warning, or a commander cut off
	# from the capital. That path saves scope:nd_incident_launch_<kind> on us
	# before the strike fires this event (#430). Saved scopes ride the whole
	# chain, so a deliberate counter-strike later in it — the victim retaliates
	# (.1), we answer (.11 b/c), nuclear_response_response_strike fires this
	# event on us again — would inherit it; .11 marks its answer with
	# scope:nd_deliberate_response, which the variants exclude.
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:nd_incident_launch_warning ?= ROOT
					NOT = { exists = scope:nd_deliberate_response }
				}
				desc = nuclear_weapon_events.2.d_warning
			}
			triggered_desc = {
				trigger = {
					scope:nd_incident_launch_commander ?= ROOT
					NOT = { exists = scope:nd_deliberate_response }
				}
				desc = nuclear_weapon_events.2.d_commander
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_weapon_events.2.d
			}
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						scope:nd_incident_launch_warning ?= ROOT
						scope:nd_incident_launch_commander ?= ROOT
					}
					NOT = { exists = scope:nd_deliberate_response }
				}
				desc = nuclear_weapon_events.2.f_incident
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = nuclear_weapon_events.2.f
			}
		}
	}

	duration = 3

	trigger = {
		# triggered by nuclear strike
	}

	immediate = {
		post_notification = industrial_nuke_succeeds_attacker
	}

	option = {
		name = nuclear_weapon_events.2.a
		default_option = yes
	}
}

# ============================================================================
# EVENT 21: A Moment of Terror — Nuclear Close Call
# ============================================================================
# Rare random event for nuclear powers. A false alarm, training exercise
# malfunction, or near-accidental launch. Based on real Cold War incidents
# (Petrov, Able Archer, Norwegian rocket incident, etc.).
# ============================================================================

nuclear_weapon_events.21 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_diplomacy_talks.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = nuclear_weapon_events.21.t
	desc = nuclear_weapon_events.21.d
	flavor = nuclear_weapon_events.21.f

	duration = 3
	cooldown = { days = very_long_modifier_time }

	# Outside the deterrence system only: an armed country's warning incidents
	# (the Unconfirmed Warning family, nuclear_incident.1-.5) are rolled by
	# nd_roll_incident from its posture and readiness; this one reads neither.
	# The deterrence half of je_nuclear_program is gated on nd_is_armed.
	trigger = {
		has_modifier = nuclear_power
		any_rival_country = { has_modifier = nuclear_power }
		NOT = { nd_is_armed = yes }
	}

	immediate = {
	}

	# "Review our safety procedures." — small prestige loss, stability
	option = {
		name = nuclear_weapon_events.21.a
		default_option = yes
		add_modifier = { name = prestige_loss_small days = normal_modifier_time is_decaying = yes }
		add_modifier = { name = nuclear_safety_review days = normal_modifier_time is_decaying = yes }
		ai_chance = {
			base = 5
			modifier = { trigger = { ruler_is_cautious = yes } add = 4 }
			modifier = { trigger = { radical_population_fraction > 0.1 } add = 4 }
		}
	}

	# "Tighten military oversight." — armed forces happy, intelligentsia unhappy
	option = {
		name = nuclear_weapon_events.21.b
		every_interest_group = {
			limit = { is_interest_group_type = ig_armed_forces }
			add_modifier = { name = ig_approval_positive_modifier days = normal_modifier_time is_decaying = yes }
		}
		every_interest_group = {
			limit = { is_interest_group_type = ig_intelligentsia }
			add_modifier = { name = ig_approval_negative_modifier days = short_modifier_time is_decaying = yes }
		}
		ai_chance = {
			base = 3
			modifier = { trigger = { ruler_is_aggressive = yes } add = 3 }
			modifier = {
				trigger = {
					has_variable = nuclear_weapon_stockpile
					any_rival_country = {
						has_modifier = nuclear_power
						# The rival's arsenal as the world estimates it, not its secret
						# (docs/systems/nuclear_crisis_design.md §0.6).
						has_variable = nd_public_estimate
						var:nd_public_estimate > ROOT.var:nuclear_weapon_stockpile
					}
				}
				add = 5
			}
			modifier = { trigger = { is_at_war = yes } add = 3 }
		}
	}

	# "This must never be spoken of." — no cost, but no benefit
	option = {
		name = nuclear_weapon_events.21.c
		ai_chance = {
			base = 2
			modifier = { trigger = { ruler_is_aggressive = yes } add = 2 }
			modifier = { trigger = { is_at_war = yes } add = 4 }
			modifier = { trigger = { radical_population_fraction < 0.05 } add = 2 }
		}
	}
}
