# A Second Living World (europa); 10% founding roll, 1% per eligible month.
space_race_discovery_events.1 = {
	type = country_event
	placement = root
	event_image = { texture = "gfx/event_pictures/colony_europa.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	title = space_race_discovery_events.1.t
	desc = space_race_discovery_events.1.d
	flavor = space_race_discovery_events.1.f
	duration = 3
	trigger = {
		has_game_rule = space_race_enabled
		sr_has_space_colony = yes
		has_variable = sr_discovery_europa_life_pending
		NOT = { has_variable = sr_discovery_europa_life_done }
		NOT = { has_variable = sr_discovery_europa_life_survey }
	}
	immediate = {
		hidden_effect = { sr_finish_colony_discovery = { DISCOVERY = europa_life } }
	}
	option = {
		name = space_race_discovery_events.1.a
		default_option = yes
		ai_chance = { base = 5 }
		sr_grant_colony_modifier = { MODIFIER = sr_discovery_europa_life_a }
	}
	option = {
		name = space_race_discovery_events.1.b
		ai_chance = { base = 5 }
		sr_grant_colony_modifier = { MODIFIER = sr_discovery_europa_life_b }
	}
}
