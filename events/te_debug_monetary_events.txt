# ============================================================================
# MONETARY POLICY TEST CONSOLE — fired only from the console:
# `event te_debug_monetary.1`
# ============================================================================
# Vic3's console has no `effect` command (that is CK3); `event <id>` is the way
# to run script by hand. Nothing in the mod fires this event.
#
# It is a read-out, not a cheat menu: the description prints every variable
# te_monetary_monthly_update owns, for the country that fires it, so a pulse can
# be inspected without a debugger. The two options re-run the monthly update and
# the yearly GDP snapshot, so the drift, the hysteresis and the random walks can
# be stepped one month at a time instead of waiting for the calendar.
#
# HIDDEN STATE. te_neutral_rate, te_neutral_error, te_neutral_walk and
# te_mon_stance_gap are hidden from the player everywhere else (§8 "The rule").
# Printing them HERE is deliberate and allowed — this event is console-only, it
# is never surfaced by any journal entry, button or on-action, and reading r*
# is the whole point of a debug read-out. Nothing in this file may be reused in
# player-facing loc.
# ============================================================================
namespace = te_debug_monetary

te_debug_monetary.1 = { # REVIEWED 2026-09-19: console-only test event (`event te_debug_monetary.1`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/stock_exchange_frenzy.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	title = te_debug_monetary.1.t
	desc = te_debug_monetary.1.d
	flavor = te_debug_monetary.1.f

	# Step one month: target, drift, premium, rate paid, stance, modifiers.
	option = {
		name = te_debug_monetary.1.a
		default_option = yes
		te_monetary_monthly_update = yes
	}

	# Recompute the neutral rate's growth term against this instant's GDP,
	# instead of waiting for the next yearly pulse.
	option = {
		name = te_debug_monetary.1.b
		te_monetary_yearly_growth_snapshot = yes
		te_monetary_monthly_update = yes
	}

	# Re-read the world reference rate, then pulse.
	option = {
		name = te_debug_monetary.1.c
		te_monetary_refresh_world_era_base = yes
		te_monetary_monthly_update = yes
	}

	option = {
		name = te_debug_monetary.1.z
	}
}


# ============================================================================
# THE INTEREST-MODIFIER MULTIPLIER EXPERIMENT — .2, .3, .4
# ============================================================================
# THE BUG. On a fresh 1836 game, day one, the budget interest tooltip reads
# "+20.0% Base Value, -19.0% Timed Modifiers" for every country: the
# te_monetary_base_offset half of the pair is there (-0.2 = -20%), and
# te_monetary_rate_paid is there too but at a multiplier of exactly 1
# (0.01 = +1%), even though te_rate_paid_applied holds the right number
# (12.5 for Russia, 3.7 for Britain — te_debug_monetary.1 prints it). No script
# error is logged from any monetary file. The first application runs from
# on_game_started -> every_country (common/on_actions/te_monetary_on_actions.txt),
# and afterwards step 9 skips the re-apply while |pts - applied| < 0.05, so a
# bad first write sticks for the whole game.
#
# The three events below run the SAME remove-then-add-the-pair block that step 9
# runs (te_monetary_apply_interest_modifiers), mid-game, from country scope, and
# differ only in what goes in the `multiplier =` slot:
#
#   .2  multiplier = var:te_rate_paid_applied      the production form
#   .3  multiplier = te_mon_rate_paid_multiplier   a named script value wrapping
#                                                  that same variable
#   .4  multiplier = 7                             a literal — the control
#
# Reading the result: Timed Modifiers = -20% (the offset) + the rate the
# multiplier actually produced. So for a country paying 12.5pp, a working
# multiplier reads about -7.5%, a multiplier of 1 reads -19.0%, the .3 fallback
# reads -17.0%, and the .4 control reads -13.0%.
#
# WHAT EACH OUTCOME MEANS. If .2 works mid-game the var form is fine and the
# fault is the on_game_started call site (ROOT there is `none`, and
# scripting_best_practices.md:403-404 documents multiplier expressions being
# resolved in the OUTER scope of an iterator, which would leave `var:` reading
# against nothing and defaulting to 1.0). If .2 fails but .3 works, the var form
# is the wrong form for this modifier. If .4 also fails, the modifier does not
# scale at all and the whole approach has to change.
#
# ORDER OF USE. Fire .2, then .3, then .4 — .4 leaves the country paying 7%
# with both modifiers present, which step 9's 0.05 threshold will never correct
# on its own. te_debug_monetary.2's second option is the undo: it zeroes
# te_rate_paid_applied and runs the real monthly update, which sees the whole
# gap at once and rewrites the pair through the production path.
#
# HOVER BEFORE CLICKING. The option tooltip previews the resolved modifier, so
# it is worth reading the tooltip as well as the budget panel afterwards.
#
# None of these events is fired by anything in the mod; they exist for
# `event te_debug_monetary.2` and friends typed into the console.
# ============================================================================

te_debug_monetary.2 = { # REVIEWED 2026-09-19: console-only diagnostic (`event te_debug_monetary.2`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/stock_exchange_frenzy.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	title = te_debug_monetary.2.t
	desc = te_debug_monetary.2.d
	flavor = te_debug_monetary.2.f

	# The production form, character for character with step 9's write.
	option = {
		name = te_debug_monetary.2.a
		default_option = yes
		if = {
			limit = { has_modifier = te_monetary_base_offset }
			remove_modifier = te_monetary_base_offset
		}
		if = {
			limit = { has_modifier = te_monetary_rate_paid }
			remove_modifier = te_monetary_rate_paid
		}
		add_modifier = { name = te_monetary_base_offset }
		add_modifier = {
			name = te_monetary_rate_paid
			multiplier = var:te_rate_paid_applied
		}
	}

	# The undo, and the cleanest test of the production path itself: zeroing the
	# applied value forces step 9 past its 0.05 threshold, so the monthly update
	# rewrites the pair exactly as it would on any other month. Note that the
	# update is not idempotent — it also drifts the policy rate a third of a
	# point and draws both hidden random walks, same as any pulse.
	option = {
		name = te_debug_monetary.2.b
		set_variable = { name = te_rate_paid_applied value = 0 }
		te_monetary_monthly_update = yes
	}

	option = {
		name = te_debug_monetary.2.z
	}
}
