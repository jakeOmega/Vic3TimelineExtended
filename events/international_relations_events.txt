# Event 1: Arms Race Escalation (rival powers building up military)
# Fired only by .201.a, the rival's own choice to build up against us: the
# rival arrives as scope:arms_race_rival. Its old pool slot and trigger moved
# to .201 (te_ea_ir_arms_race_target), so the buildup it reports is real.
international_relations_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/arms_race_factories.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = international_relations_events.1.t
	desc = international_relations_events.1.d
	flavor = international_relations_events.1.f

	duration = 3

	trigger = {
		exists = scope:arms_race_rival
	}

	immediate = {
		save_scope_as = arms_race_instigator
	}

	option = {
		name = international_relations_events.1.a
		default_option = yes
		ai_chance = {
			base = 3
			modifier = {
				trigger = { country_rank >= rank_value:great_power }
				add = 2
			}
		}
		# Match their buildup - military boost but expensive
		add_modifier = {
			name = arms_race_escalation_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		change_relations = {
			country = scope:arms_race_rival
			value = -10
		}
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = root
			SECOND = scope:arms_race_rival
			PRO_AMOUNT = -1
			ANTI_AMOUNT = 1
			FACTOR = appeasement_relations_decreased
		}
		custom_tooltip = IR_ARMS_RACE_RIVAL_PRESSURE_TT
		scope:arms_race_rival = {
			trigger_event = { id = international_relations_events.105 days = 3 }
		}
		# Armed forces approve, but raises military costs
		ig_approval_effect = { IG = ig_armed_forces MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
	}

	option = {
		name = international_relations_events.1.b
		ai_chance = {
			base = 2
			modifier = {
				trigger = {
					has_attitude = { who = scope:arms_race_rival attitude = cautious }
				}
				add = 1
			}
		}
		# Seek diplomatic channels - risky, may be rebuffed
		random_list = {
			60 = {
				# Diplomacy succeeds
				add_modifier = {
					name = diplomatic_overture_modifier
					days = normal_modifier_time
					is_decaying = yes
				}
				change_relations = {
					country = scope:arms_race_rival
					value = 10
				}
				add_lobby_appeasement_from_diplomacy_unidirectional = {
					FIRST = root
					SECOND = scope:arms_race_rival
					PRO_AMOUNT = 1
					ANTI_AMOUNT = -1
					FACTOR = appeasement_relations_increased
				}
			}
			40 = {
				# Rival rebuffs us - humiliation
				add_modifier = {
					name = diplomatic_overture_rebuffed_modifier
					days = normal_modifier_time
					is_decaying = yes
				}
				change_relations = {
					country = scope:arms_race_rival
					value = -15
				}
				add_lobby_appeasement_from_diplomacy_unidirectional = {
					FIRST = root
					SECOND = scope:arms_race_rival
					PRO_AMOUNT = -1
					ANTI_AMOUNT = 1
					FACTOR = appeasement_relations_decreased
				}
			}
		}
		# Armed forces disapprove of diplomatic approach
		ig_approval_effect = { IG = ig_armed_forces MODIFIER = ig_approval_negative_modifier DAYS = normal_modifier_time }
	}

	option = {
		name = international_relations_events.1.c
		ai_chance = {
			base = 1
			modifier = {
				trigger = { country_rank < rank_value:great_power }
				add = 2
			}
		}
		# Ignore the threat - save money on military costs, but weaker and lose face
		add_modifier = {
			name = arms_race_ignored_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		change_relations = {
			country = scope:arms_race_rival
			value = -20
		}
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = root
			SECOND = scope:arms_race_rival
			PRO_AMOUNT = -1
			ANTI_AMOUNT = 1
			FACTOR = appeasement_relations_decreased

# Event 2: Proxy Conflict (supporting insurgency in rival's sphere)
# Covert-disabled fallback (the PR #418 shape of .6): with covert warfare
# enabled, funding a rival's opposition is the destabilization and
# regime-change operations, whose exposure the covert system narrates.

	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/proxy_war_map.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = international_relations_events.2.t
	desc = international_relations_events.2.d
	flavor = international_relations_events.2.f

	duration = 3

	cooldown = { days = normal_modifier_time }

	trigger = {
		# The dedicated system owns this outcome when enabled.
		has_game_rule = covert_warfare_disabled
		NOR = {
			has_modifier = proxy_conflict_funding_modifier
			has_modifier = proxy_conflict_avoided_modifier
		}
		has_technology_researched = intergovernmental_organizations
		country_rank >= rank_value:major_power
		any_rival_country = {
			country_rank >= rank_value:major_power
			any_subject_or_below = { always = yes }
		}
	}

	immediate = {
		save_scope_as = proxy_instigator
		random_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				country_rank >= rank_value:major_power

			country = scope:proxy_rival
			value = -30
		}
		create_diplomatic_catalyst = { type = catalyst_event_negative target = scope:proxy_rival }
		custom_tooltip = IR_PROXY_CONFLICT_RIVAL_EFFECT_TT
