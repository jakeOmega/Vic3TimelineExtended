# ----------------------------------------------------------------------------
# te_lolr.1 — the guarantee is called (§15A.2)
# ----------------------------------------------------------------------------
# Fires on the GUARANTOR (ROOT) when a country it stands behind defaults:
# te_monetary_on_country_default (common/on_actions/te_monetary_on_actions.txt)
# saves the ward as scope:te_lolr_ward and sets the ward's five-year cooldown
# itself, so there is no `trigger` block — the te_peg.1 rule.
#
# HONOUR pays 5% of the WARD'S GDP across and keeps the article. RENEGE leaves
# the treaty, costs prestige and infamy, and starts te_mon_lolr_suspended_months
# on the guarantor — which te_mon_lolr_is_effective reads from every OTHER
# ward's side: guarantees are only worth the last one honoured.
#
# [6a] THE WARD'S RECORD (§15C.1, ruling H3). The on-action increments
# te_mon_lolr_calls before dispatching, so the honour cost here is 5% x
# (1 + 0.5 x PRIOR calls) to 15%, and the odds walk from 10:2 to 1:8 as a ward
# defaults again and again. The other two schedules the counter drives — the
# relief taper and the ward's own cooldown — are elsewhere; this event pays for
# the call in front of it.
#
# TWO KINDS OF GUARANTOR. The on-action sends this to a treaty guarantor when
# the ward's lender_of_last_resort article is in force, and otherwise to the
# leader of a tier-3 monetary union the ward has adopted — who signed no
# article. The desc and Renege's tooltip say which (a union leader has no
# treaty to leave; Renege suspends its union backstop instead), and the ward
# hears the answer either way (te_lolr.2).
te_lolr.1 = {
	type = country_event
	placement = ROOT

	title = te_lolr.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { te_ea_lolr_called_by_treaty = yes }
				desc = te_lolr.1.d
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = te_lolr.1.d_union
			}
		}
	}
	flavor = te_lolr.1.f

	event_image = { texture = "gfx/event_pictures/banking_boardroom.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	duration = 3

	immediate = {
		save_scope_as = te_lolr_guarantor
	}

	option = {
		name = te_lolr.1.a
		default_option = yes
		custom_tooltip = te_lolr_calls_tt

		add_treasury = {
			value = scope:te_lolr_ward.te_mon_lolr_honour_cost
			multiply = -1
		}
		scope:te_lolr_ward = {
			hidden_effect = {
				set_variable = { name = te_lolr_paid value = te_mon_lolr_honour_cost days = 120 }
				set_variable = { name = te_lolr_answer value = 1 days = 120 }
			}
			add_treasury = te_mon_lolr_honour_cost
			change_relations = {
				country = root
				value = 25
			}
			hidden_effect = { trigger_event = { id = te_lolr.2 } }
		}

		ai_chance = {
			base = 10
			# A guarantor that cannot pay its own way does not pay another's.
			modifier = {
				trigger = { scaled_debt >= 0.5 }
				add = -4
			}
			modifier = {
				trigger = { scaled_debt >= 0.9 }
				add = -4
			}
			modifier = {
				trigger = { is_in_same_power_bloc = scope:te_lolr_ward }
				add = 3
			}
			# [6a] ...and a ward that has been here before. Three tiers, so the
			# base walks 10 / 7 / 4 / 1 and then SATURATES: a fifth call is as bad
			# as a fourth, never worse than certain refusal.
			modifier = {
				trigger = { scope:te_lolr_ward = { te_mon_lolr_calls_prior >= 1 } }
				add = -3
			}
			modifier = {
				trigger = { scope:te_lolr_ward = { te_mon_lolr_calls_prior >= 2 } }
				add = -3
			}
			modifier = {
				trigger = { scope:te_lolr_ward = { te_mon_lolr_calls_prior >= 3 } }
				add = -3
			}
		}
	}

	option = {
		name = te_lolr.1.b
		if = {
			limit = { te_ea_lolr_called_by_treaty = yes }
			custom_tooltip = te_lolr.1.b.tt
		}
		else = {
			custom_tooltip = te_lolr.1.b.union.tt
		}

		hidden_effect = {
			set_variable = { name = te_mon_lolr_suspended_months value = te_mon_lolr_suspension_months }
			every_scope_treaty = {
				limit = {
					binds = scope:te_lolr_ward
					any_scope_article = {
						has_type = lender_of_last_resort
						source_country = root
					}
				}
				withdraw = { country = root }
			}
		}
		change_infamy = { value = 5 }
		add_modifier = {
			name = te_mon_lolr_reneged
			days = 1825
		}
		scope:te_lolr_ward = {
			change_relations = {
				country = root
				value = -40
			}
			hidden_effect = {
				set_variable = { name = te_lolr_answer value = 2 days = 120 }
			}
			hidden_effect = { trigger_event = { id = te_lolr.2 } }
		}

		ai_chance = {
			base = 2
			modifier = {
				trigger = { is_at_war = yes }
				add = 3
			}
			# [6a] The mirror of Honour's three tiers: 2 / 4 / 6 / 8.
			modifier = {
				trigger = { scope:te_lolr_ward = { te_mon_lolr_calls_prior >= 1 } }
				add = 2
			}
			modifier = {
				trigger = { scope:te_lolr_ward = { te_mon_lolr_calls_prior >= 2 } }
				add = 2
			}
			modifier = {
				trigger = { scope:te_lolr_ward = { te_mon_lolr_calls_prior >= 3 } }
				add = 2
			}
		}
	}
}


# ----------------------------------------------------------------------------
# te_lolr.2 — the ward hears the answer
# ----------------------------------------------------------------------------
# Fired on the defaulting ward by either te_lolr.1 option, so the treasury
# transfer or the broken promise (and the relations swing either way) arrives
# with an explanation. te_lolr.1 saves the guarantor and writes, on the ward,
# te_lolr_answer (1 honoured, 2 refused) and, when honoured, te_lolr_paid.
te_lolr.2 = {
	type = country_event
	placement = ROOT

	title = {
		first_valid = {
			triggered_desc = {
				trigger = { te_ea_lolr_answer_honoured = yes }
				desc = te_lolr.2.t_honoured
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = te_lolr.2.t_refused
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { te_ea_lolr_answer_honoured = yes }
				desc = te_lolr.2.d_honoured
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = te_lolr.2.d_refused
			}
		}
	}
	flavor = te_lolr.2.f

	event_image = { texture = "gfx/event_pictures/banking_boardroom.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	duration = 3

	trigger = {
		exists = scope:te_lolr_guarantor
		has_variable = te_lolr_answer
	}

	option = {
		name = {
			trigger = { te_ea_lolr_answer_honoured = yes }
			text = te_lolr.2.a_honoured
		}
		name = {
			trigger = { te_ea_lolr_answer_honoured = no }
			text = te_lolr.2.a_refused
		}
		default_option = yes
	}
}


# ============================================================================
# te_union.1 — The Question of the Common Currency (§15A.3)
# ============================================================================
namespace = te_union

# Put to a PRESSED holdout by te_monetary_update_union (step 9b), and only to a
# PLAYER: an AI holdout resolves on te_mon_union_adopt_score directly, with the
# same two outcomes. It fires when the answer could have changed — pressure
# newly applied, the criteria newly met, the bloc's tier changed — never on a
# timer, and no more than once in five years (the caller sets
# te_mon_union_ask_cooldown itself, so there is no `trigger` block).
#
# ADOPT is te_mon_effect_union_adopt behind its own gate, re-asked here because
# three days can pass between the pulse and the click. REFUSE builds something:
# +250 leverage resistance for ten years, refreshed not stacked — and costs the
# leader cohesion, which is why pressing a bloc of refusers is a losing trade.
te_union.1 = {
	type = country_event
	placement = ROOT

	title = te_union.1.t
	desc = te_union.1.d
	flavor = te_union.1.f

	event_image = { texture = "gfx/event_pictures/banking_boardroom.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	duration = 3

	immediate = {
		if = {
			limit = { exists = power_bloc }
			power_bloc = { power_bloc_leader = { save_scope_as = te_union_leader } }
		}
	}

	option = {
		name = te_union.1.a
		trigger = { te_mon_union_can_adopt = yes }
		custom_tooltip = te_mon_union_adopt_effect_tt

		hidden_effect = {
			te_mon_effect_union_adopt = yes
		}
		if = {
			limit = { exists = scope:te_union_leader }
			scope:te_union_leader = {
				change_relations = {
					country = root
					value = 15
				}
			}
		}

		ai_chance = {
			base = 1
		}
	}

	option = {
		name = te_union.1.b
		default_option = yes
		custom_tooltip = te_union.1.b.tt

		te_mon_effect_union_refuse = yes

		ai_chance = {
			base = 1
		}
	}
}
