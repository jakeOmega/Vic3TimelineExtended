# ============================================================================
# What they answer (docs/audits/vanilla_1_15_prep.md § 2): .1 samples C7's scaling modifier day by day; .10 and .11
# are C20's fire_only_once and fire_only_once_globally; .20 reports C3 and C21 after the probe entry concluded and
# cleans up. All hidden, with no options: nothing shows to any player. The steps that fire them are in
# common/scripted_effects/te_probe_115_effects.txt, the entry in common/journal_entries/te_probe_115_probes.txt.
#
# Delete once answered (see docs/testing/vanilla-1-15-playtest.md). To re-arm the probe in a save that has run it,
# rename te_probe_115_claimed and give .10 and .11 new ids: a fire_only_once event that has fired never fires again.

namespace = te_probe_115

# C7: one daily sample of the scaling modifier, until it shows the phase's scale or 14 days pass.
te_probe_115.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = te_probe_115_c7_sampling }
			change_variable = { name = te_probe_115_c7_days add = 1 }
			te_probe_115_c7_check_effect = yes
			if = {
				limit = { has_variable = te_probe_115_c7_sampling }
				trigger_event = { id = te_probe_115.1 days = 1 }
			}
		}
	}
}

# C20: fire_only_once, behind a trigger the steps switch on. Tried on the probe country with the trigger false
# (step 5), true (step 6) and true again (step 7), and once on another AI country (step 6).
te_probe_115.10 = {
	type = country_event
	hidden = yes
	fire_only_once = yes

	trigger = {
		has_variable = te_probe_115_c20_gate
	}

	immediate = {
		if = {
			limit = { has_variable = te_probe_115_owner }
			change_variable = { name = te_probe_115_c20_once_here add = 1 }
			debug_log = "TE_PROBE_115 c20_once fired on the probe country"
		}
		else_if = {
			limit = { has_variable = te_probe_115_c20_other_mark }
			change_global_variable = { name = te_probe_115_c20_once_other add = 1 }
			debug_log = "TE_PROBE_115 c20_once fired on the other country"
		}
	}
}

# C20: fire_only_once_globally. Tried on the probe country (steps 5 and 7) and on another AI country (step 6).
te_probe_115.11 = {
	type = country_event
	hidden = yes
	fire_only_once_globally = yes # REVIEWED 2026-11-05: a 1.15 event field (Modding-Digests 1.15.0 changes_features.md, One Time Events) only vanilla's commented-out test.122 uses, so the catalog lacks it

	immediate = {
		change_global_variable = { name = te_probe_115_c20_globally_n add = 1 }
		if = {
			limit = { has_variable = te_probe_115_owner }
			debug_log = "TE_PROBE_115 c20_globally fired on the probe country"
		}
		else = {
			debug_log = "TE_PROBE_115 c20_globally fired on the other country"
		}
	}
}

# C3 and C21: three days after the probe entry concluded through one of its C3 outcomes.
te_probe_115.20 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_variable = te_probe_115_c3_ran_first
				has_variable = te_probe_115_c3_ran_second
			}
			debug_log = "TE_PROBE_115 c3_both BOTH (both outcomes' effects ran: every outcome true on the update runs)"
		}
		else = {
			debug_log = "TE_PROBE_115 c3_both ONE (only one outcome's effect ran)"
		}
		if = {
			limit = { has_journal_entry = te_probe_115 }
			debug_log = "TE_PROBE_115 c21_concluded_active YES (has_journal_entry is true for a concluded entry)"
		}
		else = {
			debug_log = "TE_PROBE_115 c21_concluded_active NO (has_journal_entry is false for a concluded entry)"
		}
		if = {
			limit = { exists = je:te_probe_115 }
			debug_log = "TE_PROBE_115 c21_concluded_je YES (je: still finds a concluded entry)"
		}
		else = {
			debug_log = "TE_PROBE_115 c21_concluded_je NO (je: does not find a concluded entry)"
		}
		if = {
			limit = { has_inactive_journal_entry = te_probe_115 }
			debug_log = "TE_PROBE_115 c21_concluded_inactive YES (a concluded entry counts as inactive)"
		}
		else = {
			debug_log = "TE_PROBE_115 c21_concluded_inactive NO (a concluded entry does not count as inactive)"
		}
		te_probe_115_cleanup_effect = yes
		debug_log = "TE_PROBE_115 done (every probe has reported; the claim marker stays so it never runs again in this game)"
	}
}
