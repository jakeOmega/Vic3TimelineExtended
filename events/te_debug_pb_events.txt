# ============================================================================
# POWER BLOC TEST CONSOLE — fired only from the console:
# `event te_debug_pb.1`
# ============================================================================
# Forms a small bloc, fills slots without waiting for mandates, adds or takes
# away a technology slot (te_debug_pb_principle_slots stands in for the four
# slot technologies) and speeds up mandates, so the slot count
# (te_pb_principle_slot_cap) and the panel can be checked in minutes. Nothing in
# the mod fires this event.
#
# Mandate test (#1442): Add runs add_power_bloc_mandate = 1, Remove runs
# add_power_bloc_mandate = -1. The script docs promise only adding, and the mod
# wants to know whether a negative value takes one away before it builds ways to
# spend mandates. Add is the control: if Add fails too, the test is broken and
# says nothing about removing. Each click logs the count straight after the
# effect, and te_debug_pb.2 compares it with the count before a day later, logs
# a TE_DEBUG_PB_MANDATE verdict and reopens this console with it in the flavor
# text. Both are hidden while fast mandates are on, since a mandate earned
# during the day would spoil the count. Steps:
# docs/testing/vanilla-1-15-playtest.md, "Power bloc mandates".
# ============================================================================
namespace = te_debug_pb

te_debug_pb.1 = { # REVIEWED 2026-10-06: console-only test event (`event te_debug_pb.1`); never fired by script on purpose
	type = country_event
	placement = ROOT

	event_image = { texture = "gfx/event_pictures/un_assembly_hall_vote.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = te_debug_pb.1.t
	desc = te_debug_pb.1.d
	# The last mandate test's verdict, set by te_debug_pb.2.
	flavor = {
		first_valid = {
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 1 } desc = te_debug_pb.1.f_add_ok }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 2 } desc = te_debug_pb.1.f_add_none }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 3 } desc = te_debug_pb.1.f_add_cap }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 4 } desc = te_debug_pb.1.f_remove_ok }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 5 } desc = te_debug_pb.1.f_remove_none }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 6 } desc = te_debug_pb.1.f_remove_floor }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 7 } desc = te_debug_pb.1.f_remove_negative }
			triggered_desc = { trigger = { var:te_debug_pb_mandate_verdict ?= 8 } desc = te_debug_pb.1.f_unexpected }
			triggered_desc = { trigger = { always = yes } desc = te_debug_pb.1.f }
		}
	}

	# The option tooltips read var:te_debug_pb_tech_slots every frame; set it
	# here so they never read a missing variable.
	immediate = {
		if = {
			limit = { NOT = { has_variable = te_debug_pb_tech_slots } }
			set_variable = {
				name = te_debug_pb_tech_slots
				value = 0
			}
		}
	}

	# Form a two-member bloc led by us: below the rank and members slot 3 needs.
	option = {
		name = te_debug_pb.1.a
		default_option = yes
		trigger = {
			is_in_power_bloc = no
		}
		ordered_country = {
			limit = {
				NOT = { this = root }
				is_in_power_bloc = no
				is_subject = no
				exists = capital
			}
			order_by = gdp
			position = 0
			save_scope_as = te_debug_pb_member
		}
		create_power_bloc = {
			name = te_debug_pb_bloc_name
			map_color = hsv{ 0.55 0.7 0.9 }
			identity = identity_trade_league
			principle = principle_transport_1
			member = scope:te_debug_pb_member
		}
	}

	# Fill the next free slot with the first of ten unrestricted groups we lack.
	option = {
		name = te_debug_pb.1.b
		trigger = {
			is_power_bloc_leader = yes
		}
		power_bloc = {
			if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_transport } }
				add_principle = principle_transport_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_external_trade } }
				add_principle = principle_external_trade_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_food_standardization } }
				add_principle = principle_food_standardization_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_police_coordination } }
				add_principle = principle_police_coordination_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_military_industry } }
				add_principle = principle_military_industry_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_companies } }
				add_principle = principle_companies_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_navy } }
				add_principle = principle_navy_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_defensive_cooperation } }
				add_principle = principle_defensive_cooperation_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_freedom_of_movement } }
				add_principle = principle_freedom_of_movement_1
			}
			else_if = {
				limit = { NOT = { has_principle_group = principle_group:principle_group_creative_legislature } }
				add_principle = principle_creative_legislature_1
			}
		}
	}

	# One more technology slot (stands in for a slot-granting technology).
	option = {
		name = te_debug_pb.1.c
		trigger = {
			NOT = { var:te_debug_pb_tech_slots ?= 4 }
		}
		change_variable = {
			name = te_debug_pb_tech_slots
			add = 1
		}
		if = {
			limit = { has_modifier = te_debug_pb_principle_slots }
			remove_modifier = te_debug_pb_principle_slots
		}
		add_modifier = {
			name = te_debug_pb_principle_slots
			multiplier = var:te_debug_pb_tech_slots
		}
	}

	# One fewer technology slot.
	option = {
		name = te_debug_pb.1.e
		trigger = {
			has_variable = te_debug_pb_tech_slots
			var:te_debug_pb_tech_slots > 0
		}
		change_variable = {
			name = te_debug_pb_tech_slots
			add = -1
		}
		if = {
			limit = { has_modifier = te_debug_pb_principle_slots }
			remove_modifier = te_debug_pb_principle_slots
		}
		if = {
			limit = { var:te_debug_pb_tech_slots > 0 }
			add_modifier = {
				name = te_debug_pb_principle_slots
				multiplier = var:te_debug_pb_tech_slots
			}
		}
	}

	# Mandates in days: toggle the fast-mandate modifier on the bloc.
	option = {
		name = te_debug_pb.1.l
		trigger = {
			is_power_bloc_leader = yes
		}
		power_bloc = {
			if = {
				limit = { has_modifier = te_debug_pb_fast_mandates }
				remove_modifier = te_debug_pb_fast_mandates
			}
			else = {
				add_modifier = { name = te_debug_pb_fast_mandates }
			}
		}
	}

	# Mandate test: add one. The control for Remove; at the cap of 5 it should
	# change nothing.
	option = {
		name = te_debug_pb.1.g
		trigger = {
			is_power_bloc_leader = yes
			power_bloc = {
				NOT = { has_modifier = te_debug_pb_fast_mandates }
			}
		}
		set_variable = {
			name = te_debug_pb_mandates_before
			value = te_debug_pb_mandates_by_trigger
		}
		set_variable = {
			name = te_debug_pb_mandate_test
			value = 1
		}
		power_bloc = {
			add_power_bloc_mandate = 1
		}
		debug_log = "TE_DEBUG_PB_MANDATE add clicked: before=[SCOPE.ScriptValue('te_debug_pb_mandates_before_view')|0] straight_after=[SCOPE.ScriptValue('te_debug_pb_mandates_by_trigger')|0] link=[SCOPE.ScriptValue('te_debug_pb_mandates_by_link')|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		trigger_event = { id = te_debug_pb.2 days = 1 }
	}

	# Mandate test: take one away with a negative value. Shown with no mandates
	# too, so a click at 0 shows whether the count can go below 0.
	option = {
		name = te_debug_pb.1.h
		trigger = {
			is_power_bloc_leader = yes
			power_bloc = {
				NOT = { has_modifier = te_debug_pb_fast_mandates }
			}
		}
		set_variable = {
			name = te_debug_pb_mandates_before
			value = te_debug_pb_mandates_by_trigger
		}
		set_variable = {
			name = te_debug_pb_mandate_test
			value = 2
		}
		power_bloc = {
			add_power_bloc_mandate = -1
		}
		debug_log = "TE_DEBUG_PB_MANDATE remove clicked: before=[SCOPE.ScriptValue('te_debug_pb_mandates_before_view')|0] straight_after=[SCOPE.ScriptValue('te_debug_pb_mandates_by_trigger')|0] link=[SCOPE.ScriptValue('te_debug_pb_mandates_by_link')|0] date=[TimeKeeper.GetCurrentDate.GetString]"
		trigger_event = { id = te_debug_pb.2 days = 1 }
	}

	option = {
		name = te_debug_pb.1.m
	}
}

# The mandate test's verdict, a day after Add or Remove: compares the trigger's
# count with the one before the click, logs it and reopens the console, whose
# flavor text shows it (verdicts 1-8, te_debug_pb.1's flavor). The reopen is
# safe undelayed: nothing here runs again until the next click.
te_debug_pb.2 = { # REVIEWED 2026-11-06: console-only, fired by te_debug_pb.1 options g and h
	type = country_event
	hidden = yes

	trigger = {
		has_variable = te_debug_pb_mandate_test
	}

	immediate = {
		set_variable = {
			name = te_debug_pb_mandates_after
			value = te_debug_pb_mandates_by_trigger
		}
		set_variable = {
			name = te_debug_pb_mandate_change
			value = var:te_debug_pb_mandates_after
		}
		change_variable = {
			name = te_debug_pb_mandate_change
			subtract = var:te_debug_pb_mandates_before
		}
		if = {
			limit = { var:te_debug_pb_mandate_test = 1 }
			if = {
				limit = { var:te_debug_pb_mandate_change = 1 }
				set_variable = { name = te_debug_pb_mandate_verdict value = 1 }
				debug_log = "TE_DEBUG_PB_MANDATE add OK: one more mandate"
			}
			else_if = {
				limit = {
					var:te_debug_pb_mandate_change = 0
					var:te_debug_pb_mandates_before >= 5
				}
				set_variable = { name = te_debug_pb_mandate_verdict value = 3 }
				debug_log = "TE_DEBUG_PB_MANDATE add AT_CAP: no change at the cap of 5, as expected; remove one and try again"
			}
			else_if = {
				limit = { var:te_debug_pb_mandate_change = 0 }
				set_variable = { name = te_debug_pb_mandate_verdict value = 2 }
				debug_log = "TE_DEBUG_PB_MANDATE add NO_CHANGE: below the cap and nothing changed, so the test or add_power_bloc_mandate is broken; Remove's result means nothing"
			}
			else = {
				set_variable = { name = te_debug_pb_mandate_verdict value = 8 }
				debug_log = "TE_DEBUG_PB_MANDATE add UNEXPECTED"
			}
		}
		else = {
			if = {
				limit = { var:te_debug_pb_mandates_after < 0 }
				set_variable = { name = te_debug_pb_mandate_verdict value = 7 }
				debug_log = "TE_DEBUG_PB_MANDATE remove NEGATIVE: the count went below 0"
			}
			else_if = {
				limit = { var:te_debug_pb_mandate_change = -1 }
				set_variable = { name = te_debug_pb_mandate_verdict value = 4 }
				debug_log = "TE_DEBUG_PB_MANDATE remove OK: a negative value takes a mandate away"
			}
			else_if = {
				limit = {
					var:te_debug_pb_mandate_change = 0
					var:te_debug_pb_mandates_before = 0
				}
				set_variable = { name = te_debug_pb_mandate_verdict value = 6 }
				debug_log = "TE_DEBUG_PB_MANDATE remove FLOOR: with no mandates nothing changed; the count stays at 0"
			}
			else_if = {
				limit = { var:te_debug_pb_mandate_change = 0 }
				set_variable = { name = te_debug_pb_mandate_verdict value = 5 }
				debug_log = "TE_DEBUG_PB_MANDATE remove NO_CHANGE: a negative value is ignored (trust this only if Add read OK)"
			}
			else = {
				set_variable = { name = te_debug_pb_mandate_verdict value = 8 }
				debug_log = "TE_DEBUG_PB_MANDATE remove UNEXPECTED"
			}
		}
		debug_log = "TE_DEBUG_PB_MANDATE counts a day later: before=[SCOPE.ScriptValue('te_debug_pb_mandates_before_view')|0] after=[SCOPE.ScriptValue('te_debug_pb_mandates_by_trigger')|0] link=[SCOPE.ScriptValue('te_debug_pb_mandates_by_link')|0] (after 6 means 6 or more; link should equal after) date=[TimeKeeper.GetCurrentDate.GetString]"
		remove_variable = te_debug_pb_mandate_test
		trigger_event = { id = te_debug_pb.1 }
	}
}
