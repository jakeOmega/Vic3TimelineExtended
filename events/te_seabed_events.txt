# A Lease <Basin> Seabed Claims article came into force a day ago with this
# country as the lessee (te_seabed_lease_start): each lease due starts from
# its article in the country's treaties (te_seabed_lease_take_up).
te_seabed.1 = {
	type = country_event
	hidden = yes

	immediate = {
		te_seabed_lease_take_up = yes
	}
}

# Self-check seabed_pact_read (docs/testing/global-commons-playtest.md): a day
# after its overlord took Administer Seabed Claims (te_seabed_pact_start),
# the subject reads the pact from its own side, as every sharing does
# (te_seabed_pact_stands). OK once a game; BROKEN each time it fails. A subject
# no longer one gives no verdict.
te_seabed.2 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = overlord }
			overlord = { save_scope_as = te_sb_over }
			if = {
				limit = {
					has_diplomatic_pact = { who = scope:te_sb_over type = te_seabed_administer_claims }
				}
				if = {
					limit = { NOT = { has_global_variable = te_seabed_selfcheck_pact_done } }
					set_global_variable = te_seabed_selfcheck_pact_done
					debug_log = "TE_SELF_CHECK OK seabed_pact_read subject=[THIS.GetNameNoFormatting] passed_pacific=[THIS.Var('te_seabed_pass_unused_pacific').GetValue|0] passed_atlantic=[THIS.Var('te_seabed_pass_unused_atlantic').GetValue|0] (the subject sees the pact a day after it began)"
					debug_log_scopes = yes
				}
			}
			else = {
				debug_log = "TE_SELF_CHECK BROKEN seabed_pact_read subject=[THIS.GetNameNoFormatting] (a day after Administer Seabed Claims began, has_diplomatic_pact does not see it from the subject's side, or the overlord ended it within the day)"
				debug_log_scopes = yes
			}
		}
	}
}

# The Seabed Rush Begins (seabed claims spec § 7; an event, not a notification,
# owner 2026-10-17): news, once a game, to every player the month the first
# country can claim (te_seabed_global_monthly), told three ways: to that
# country, to a coastal one and to a landlocked one. With the Global Commons
# rule on it stands in for the research's headline
# (headlines_tech_abyssal_plain_mining, common/on_actions/headlines.txt). The
# first claimant is read back from its global variable in case the saved
# scope doesn't survive the dispatch.
te_seabed.3 = {
	type = country_event
	placement = ROOT

	title = te_seabed.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = te_seabed.3.d_first
				trigger = { scope:te_seabed_first_claimant ?= this }
			}
			triggered_desc = {
				desc = te_seabed.3.d_coastal
				trigger = { any_scope_state = { is_coastal = yes } }
			}
			desc = te_seabed.3.d_landlocked
		}
	}
	flavor = te_seabed.3.f

	event_image = { texture = "gfx/event_pictures/law_of_sea_maritime.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	immediate = {
		global_var:te_seabed_first_claimant ?= { save_scope_as = te_seabed_first_claimant }
	}

	option = {
		name = {
			trigger = { scope:te_seabed_first_claimant ?= this }
			text = te_seabed.3.a_first
		}
		name = {
			trigger = { NOT = { scope:te_seabed_first_claimant ?= this } }
			text = te_seabed.3.a
		}
		default_option = yes
		custom_tooltip = te_seabed.3.a.tt
	}
}
