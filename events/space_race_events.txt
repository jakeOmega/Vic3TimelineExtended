# Event 1: Suborbital Flight Achievement
space_race_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/rocket_ascending_sky.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	title = space_race_events.1.t
	# The world-first text only for the country that really was first
	# (sr_was_first_suborbital is set in on_complete before this event fires).
	desc = {
		first_valid = {
			triggered_desc = {
				desc = space_race_events.1.d
				trigger = { has_variable = sr_was_first_suborbital }
			}
			desc = space_race_events.1.d_later
		}
	}
	flavor = space_race_events.1.f

	duration = 3

	trigger = { }
	immediate = { }

	option = {
		name = space_race_events.1.a
		default_option = yes
		ai_chance = { base = 5 }
		sr_apply_suborbital_reward = yes
		# Celebrate the achievement - loyalists
		add_loyalists = { value = medium_radicals pop_type = academics }
		add_loyalists = { value = small_radicals }
	}

	option = {
		name = space_race_events.1.b
		ai_chance = { base = 3 }
		sr_apply_suborbital_reward = yes
		# Channel momentum into next milestone - innovation boost. Its own
		# modifier: sr_suborbital is the permanent reward of a country that
		# was not first, and adding it again here would collide with it.
		add_modifier = {
			name = sr_suborbital_momentum
			days = short_modifier_time
			is_decaying = yes
		}
	}
}

		OR = {
			sr_has_milestone_notice = { MILESTONE = suborbital }
			sr_has_milestone_notice = { MILESTONE = orbital }

			sr_has_milestone_notice = { MILESTONE = probe }
			sr_has_milestone_notice = { MILESTONE = moon_base }

			sr_has_milestone_notice = { MILESTONE = interstellar_probe }
		}

			on_created = { place_character_in_void = 6 }

		# waiting in the same 14-day window is left for its own popup). Saves
		# sr_achiever, sr_milestone_<m>, and sr_is_first / sr_already_achieved.

			limit = { sr_has_milestone_notice = { MILESTONE = suborbital } }
			sr_take_milestone_notice_base = { MILESTONE = suborbital }

			limit = { sr_has_milestone_notice = { MILESTONE = orbital } }
			sr_take_milestone_notice_base = { MILESTONE = orbital }
		}
		else_if = {
			limit = { sr_has_milestone_notice = { MILESTONE = moon_landing } }
			sr_take_milestone_notice_base = { MILESTONE = moon_landing }
		}
		else_if = {
			limit = { sr_has_milestone_notice = { MILESTONE = probe } }
			sr_take_milestone_notice_base = { MILESTONE = probe }
		}
		else_if = {
			limit = { sr_has_milestone_notice = { MILESTONE = moon_base } }
			sr_take_milestone_notice_base = { MILESTONE = moon_base }
		}
		else_if = {
			limit = { sr_has_milestone_notice = { MILESTONE = mars_landing } }
			sr_take_milestone_notice_base = { MILESTONE = mars_landing }
		}
		else_if = {
			limit = { sr_has_milestone_notice = { MILESTONE = interstellar_probe } }
			sr_take_milestone_notice_base = { MILESTONE = interstellar_probe }
		}
		sr_clear_legacy_milestone_notice = yes
	}

	option = {
		name = space_race_events.20.a
		default_option = yes
		ai_chance = { base = 5 }
		# We must accelerate our own program: every milestone under way gains a
		# little. A country with none running rallies its scientists instead.

			sr_boost_active_milestones = { MONTHS = 1 }
		}
		else = {
			add_loyalists = { value = very_small_radicals pop_type = academics }
		}
	}

	option = {
		name = space_race_events.20.b
		ai_chance = { base = 3 }
		# Congratulate them publicly
		if = {
			limit = { exists = scope:sr_achiever }
			change_relations = { country = scope:sr_achiever value = 10 }
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = { exists = scope:flavor_speaker }
				scope:flavor_speaker = { kill_character = { hidden = yes } }
			}
		}

space_race_events.40 = {

	placement = root

	desc = space_race_events.40.d
	flavor = space_race_events.40.f

	duration = 3
	cooldown = { days = long_modifier_time }

	trigger = {
		OR = {
			has_variable = sr_active_orbital
			has_variable = sr_active_moon_landing
			sr_colonizing_moon_first = yes
		}
		OR = {
			AND = {
				has_variable = sr_active_orbital
				var:sr_progress_orbital >= 25
			}
			AND = {
				has_variable = sr_active_moon_landing
				var:sr_progress_moon_landing >= 25
			}
			AND = {
				sr_colonizing_moon_first = yes
				var:sr_progress_solar_colonization >= 25
			}
		}
		NOT = { has_modifier = sr_mission_failure }
	}

		sr_record_event_pace = yes

	option = {
		name = space_race_events.40.a
		default_option = yes
		ai_chance = { base = 5 }
		# Active magnetic shielding - heavier but more effective
		custom_tooltip = SR_PROGRESS_ADD_1_MONTHS_TT
		if = {
			limit = { has_variable = sr_active_orbital }
			sr_gain_progress_months = { MILESTONE = orbital MONTHS = 1 }
		}
		if = {
			limit = { has_variable = sr_active_moon_landing }
			sr_gain_progress_months = { MILESTONE = moon_landing MONTHS = 1 }
		}
		if = {
			limit = { sr_colonizing_moon_first = yes }
			sr_gain_progress_months = { MILESTONE = solar_colonization MONTHS = 1 }
		}
		add_loyalists = { value = small_radicals pop_type = academics }
	}

	option = {
		name = space_race_events.40.b
		ai_chance = { base = 4 }
		# Passive water shielding - proven physics, adds mass budget

		if = {
			limit = { has_variable = sr_active_orbital }
			sr_gain_progress_months = { MILESTONE = orbital MONTHS = 2 }
		}
		if = {
			limit = { has_variable = sr_active_moon_landing }
			sr_gain_progress_months = { MILESTONE = moon_landing MONTHS = 2 }
		}
		if = {
			limit = { sr_colonizing_moon_first = yes }
			sr_gain_progress_months = { MILESTONE = solar_colonization MONTHS = 2 }
