# ============================================================================
# POWER BLOC SUMMITS (#1442)
# ============================================================================
# A bloc leader spends a spare mandate on one initiative the whole bloc carries
# for ten years: a use for mandates once the principle slots are full.
# te_pb_summit.1 opens from the panel's Convene a Summit button
# (te_pb_summit_sgui, gui/power_bloc_panel.gui) for a player, and from
# te_pb_summit_ai_on_action for an AI leader holding two or more mandates.
# Each option but Adjourn spends one mandate (te_pb_summit_pay) and is offered
# only while the bloc holds one (te_pb_summit_can_pay), since the event can
# wait while the mandate goes on a principle. A timed initiative already
# running is not offered again; Rally the Bloc is, for its immediate cohesion.
# Modifiers: common/static_modifiers/te_power_bloc_summit_modifiers.txt.
# Design: docs/systems/mod_systems/power_blocs.md.
# ============================================================================
namespace = te_pb_summit

te_pb_summit.1 = {
	type = country_event
	placement = ROOT

	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	title = te_pb_summit.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { te_pb_summit_can_pay = no }
				desc = te_pb_summit.1.d_none
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = te_pb_summit.1.d
			}
		}
	}
	flavor = te_pb_summit.1.f

	duration = 3

	trigger = {
		is_power_bloc_leader = yes
	}

	# Rally the Bloc: cohesion now and for ten years. Offered while running too.
	option = {
		name = te_pb_summit.1.a
		trigger = {
			te_pb_summit_can_pay = yes
		}
		te_pb_summit_pay = { KIND = rally }
		power_bloc = {
			add_cohesion_percent = 0.2
		}
		te_pb_summit_grant_bloc_initiative = { MODIFIER = te_pb_summit_rally }
		ai_chance = {
			base = 10
			modifier = {
				trigger = { power_bloc ?= { current_cohesion_percentage < 0.5 } }
				add = 30
			}
			modifier = {
				trigger = { power_bloc ?= { current_cohesion_percentage < 0.25 } }
				add = 30
			}
		}
	}

	# Recruitment Drive: invitations accepted more readily.
	option = {
		name = te_pb_summit.1.b
		trigger = {
			te_pb_summit_can_pay = yes
			power_bloc ?= {
				NOT = { has_modifier = te_pb_summit_recruitment }
			}
		}
		te_pb_summit_pay = { KIND = recruitment }
		te_pb_summit_grant_bloc_initiative = { MODIFIER = te_pb_summit_recruitment }
		ai_chance = {
			base = 5
			modifier = {
				trigger = { power_bloc ?= { num_power_bloc_members < 5 } }
				add = 10
			}
		}
	}

	# Leverage Campaign: leverage generated faster on every country.
	option = {
		name = te_pb_summit.1.c
		trigger = {
			te_pb_summit_can_pay = yes
			power_bloc ?= {
				NOT = { has_modifier = te_pb_summit_leverage }
			}
		}
		te_pb_summit_pay = { KIND = leverage }
		te_pb_summit_grant_bloc_initiative = { MODIFIER = te_pb_summit_leverage }
		ai_chance = {
			base = 10
		}
	}

	# Open Borders Accord: port access and wider migration, on loan. Hidden when
	# the bloc's principles give both.
	option = {
		name = te_pb_summit.1.e
		trigger = {
			te_pb_summit_can_pay = yes
			power_bloc ?= {
				NOT = { has_modifier = te_pb_summit_open_borders }
				te_pb_summit_has_open_borders = no
			}
		}
		te_pb_summit_pay = { KIND = open_borders }
		te_pb_summit_grant_bloc_initiative = { MODIFIER = te_pb_summit_open_borders }
		ai_chance = {
			base = 5
		}
	}

	# A Mandate at Home: legitimacy and authority for the leader.
	option = {
		name = te_pb_summit.1.g
		trigger = {
			te_pb_summit_can_pay = yes
			NOT = { has_modifier = te_pb_summit_home }
		}
		te_pb_summit_pay = { KIND = home }
		hidden_effect = {
			if = {
				limit = { has_modifier = te_pb_summit_home }
				remove_modifier = te_pb_summit_home
			}
		}
		add_modifier = {
			name = te_pb_summit_home
			days = long_modifier_time
		}
		ai_chance = {
			base = 5
			modifier = {
				trigger = { legitimacy < 50 }
				add = 15
			}
		}
	}

	# Adjourn: nothing spent. The AI never takes it while it can pay.
	option = {
		name = te_pb_summit.1.h
		default_option = yes
		custom_tooltip = te_pb_summit_adjourn_tt # REVIEWED 2026-11-06 (no_effect_option): adjourning is the choice; the tooltip says no mandate is spent
		ai_chance = {
			base = 0
		}
	}
}
