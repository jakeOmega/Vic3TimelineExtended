# ============================================================================
# The Creditors' Consortium: a member defaults, and is rescued on terms
# ============================================================================
# Design: docs/superpowers/specs/2026-11-16-power-bloc-identities-design.md § 5.3.
# The flow, the scopes and the helpers: te_creditors_consortium_effects.txt.
#
#   te_cc.1  the leader: rescue on terms, or refuse
#   te_cc.2  the member: accept the terms, or refuse them
#   te_cc.3  the leader hears the member's answer
#   te_cc.4  the member hears that the leader refused
#   te_cc.9  hidden, three days after a rescue: the treaty self-checks
#
# Fired only from te_monetary_on_country_default, which asks te_mon_full_system
# (the banking rule on Full) and te_cred_guarantee_is_effective; each event
# checks again in its trigger, since the rule can change while one waits.
# ============================================================================

namespace = te_cc

# ----------------------------------------------------------------------------
# Every rescue spends all the terms the leader has (te_cred_terms_to_pick: one
# and the bloc's Conditionality tier's figure, or as many as are on offer if
# fewer). Adjustment and privatization take one each; concessions take what is
# left, each term spent on them buying the tier's monopolies (owner,
# 2026-11-17). One option for each way to spend them that the engine allows
# (te_cred_rescue_option_*), so the player sees only rescues of the right size
# and the leader chooses which: at most four at once.
#
# THE AI'S ODDS ARE te_lolr.1's. Every rescue option carries te_lolr.1's
# honour weight in full (10, +3 for a bloc member, -4 at 0.5 and again at 0.9
# of the leader's ceiling, -3 for each prior call up to three) and Refuse its
# refusal weight; where two, three or four rescues show at once
# (te_cred_rescue_split_2 / _3 / _4), each takes a half, a third or a quarter, so the rescues
# together weigh what te_lolr.1's one Honour does. The split is a factor after
# the adds: read as (base + adds) x factor, the sum is exact; read the other
# way it errs toward refusal.

			trigger = { scope:te_cc_debtor.var:te_cc_answer ?= 1 }

# ----------------------------------------------------------------------------
# Fired on the member three days after it accepted (te_cred_apply_rescue).
#
# cc_receivership_treaty (§ 11 fact 1): te_cred_make_receivership made a
# debt_receivership treaty with is_draft = no (te_cred_rcv_check names the
# leader). OK if the member is now the source of an in-force debt_receivership
# article; BROKEN if not, and the receivership is then set directly
# (te_cred_set_direct_receivership), the fallback. Its exit ramp and relief
# follow either way.
#
# cc_monopoly_treaty (§ 11 fact 3): te_cred_demand_monopolies made
# te_cred_mono_made acquire_monopoly_for_company treaties with a company and a
# building type as inputs. OK if at least one such article with the member as
# source is now in force; BROKEN if none is. No fallback runs: the leader may
# still ask for monopolies in the treaty screen (docs/testing/power-bloc-identities-playtest.md).
#
# Each logs once a game, from the first rescue that tries it.
te_cc.9 = {
	type = country_event
	hidden = yes

	immediate = {
		# Off the Full rule nothing is read: the engine has removed any
		# receivership article (requirement_to_maintain), so a check would log a
		# false BROKEN and set a receivership the rule has paused.
		if = {
			limit = {
				te_mon_full_system = yes
				has_variable = te_cred_rcv_check
				exists = var:te_cred_rcv_check
			}
			var:te_cred_rcv_check = { save_scope_as = te_cc_creditor }
			if = {
				limit = { te_mon_is_article_source = { TYPE = debt_receivership } }
				if = {
					limit = { NOT = { has_global_variable = te_cc_sc_receivership_logged } }
					set_global_variable = te_cc_sc_receivership_logged
					debug_log = "TE_SELF_CHECK OK cc_receivership_treaty (create_treaty made [ROOT.GetCountry.GetNameNoFormatting]'s debt_receivership to [SCOPE.sCountry('te_cc_creditor').GetNameNoFormatting] and it is in force three days on; receivership flag=[ROOT.GetCountry.MakeScope.Var('te_mon_receivership').GetValue|0], which the article's on_entry_into_force dispatch sets) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
			else = {
				if = {
					limit = { NOT = { has_global_variable = te_cc_sc_receivership_logged } }
					set_global_variable = te_cc_sc_receivership_logged
					debug_log = "TE_SELF_CHECK BROKEN cc_receivership_treaty (create_treaty with debt_receivership from [ROOT.GetCountry.GetNameNoFormatting] to [SCOPE.sCountry('te_cc_creditor').GetNameNoFormatting] left no article in force three days on: the receivership is set directly instead (te_cred_set_direct_receivership); see docs/testing/power-bloc-identities-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				te_cred_set_direct_receivership = yes
			}
			remove_variable = te_cred_rcv_check
		}

		if = {
			limit = {
				te_mon_full_system = yes
				has_variable = te_cred_mono_made
				var:te_cred_mono_made > 0
			}
			# Only a member that held no monopoly article before the rescue
			# (te_cred_mono_prior, te_cred_demand_monopolies), so an older
			# article can't make it pass.
			if = {
				limit = {
					NOT = { has_global_variable = te_cc_sc_monopoly_logged }
					NOT = { has_variable = te_cred_mono_prior }
				}
				set_global_variable = te_cc_sc_monopoly_logged
				if = {
					limit = {
						any_scope_treaty = {
							any_scope_article = {
								has_type = acquire_monopoly_for_company
								source_country = root
							}
						}
					}
					debug_log = "TE_SELF_CHECK OK cc_monopoly_treaty (create_treaty made [ROOT.GetCountry.MakeScope.Var('te_cred_mono_made').GetValue|0] acquire_monopoly_for_company treaties with company and building-type inputs from [ROOT.GetCountry.GetNameNoFormatting], and one is in force three days on; check the company's monopoly in its panel) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
				else = {
					debug_log = "TE_SELF_CHECK BROKEN cc_monopoly_treaty (create_treaty made [ROOT.GetCountry.MakeScope.Var('te_cred_mono_made').GetValue|0] acquire_monopoly_for_company treaties from [ROOT.GetCountry.GetNameNoFormatting] and none is in force three days on: the inputs are not taken from script; see docs/testing/power-bloc-identities-playtest.md) date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
			remove_variable = te_cred_mono_made
		}
		if = {
			limit = { has_variable = te_cred_rcv_check }
			remove_variable = te_cred_rcv_check
		}
		if = {
			limit = { has_variable = te_cred_mono_made }
			remove_variable = te_cred_mono_made
		}
		if = {
			limit = { has_variable = te_cred_mono_prior }
			remove_variable = te_cred_mono_prior
		}
	}
}
