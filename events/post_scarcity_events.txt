# Event 1: Mass Unemployment Protests
post_scarcity_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/post_scarcity_unemployment.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	title = post_scarcity_events.1.t
	desc = post_scarcity_events.1.d
	flavor = post_scarcity_events.1.f

	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		has_technology_researched = universal_basic_income
		NOT = { has_law = law_type:law_post-scarcity }
	}

	immediate = {
	}

	# Expand UBI programs (only a country with a UBI has programs to expand)
	option = {
		name = post_scarcity_events.1.a
		default_option = yes
		trigger = { has_law = law_type:law_universal_basic_income }
		ai_chance = {
			base = 4
			# welfare/redistributive economies naturally expand UBI in a jobs crisis;
			# a country with the UBI law always gets the +2 it earned here before
		}
		add_modifier = {
			name = ubi_expansion_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		add_treasury = {
			value = yearly_gross_income
			multiply = -0.25
		}
	}

	# Without a UBI: the same relief, as emergency income support
	option = {
		name = post_scarcity_events.1.a_emergency
		default_option = yes
		trigger = {
			NOT = { has_law = law_type:law_universal_basic_income }
		}
		ai_chance = {
			base = 2
			# welfare/redistributive economies naturally reach for income support in a jobs crisis
			modifier = {
				trigger = {
					OR = {
						has_law = law_type:law_wage_subsidies
						has_law = law_type:law_command_economy
						has_law = law_type:law_council_republic
					}
				}
				add = 2
			}
		}
		add_modifier = {
			name = emergency_income_support_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		add_treasury = {
			value = yearly_gross_income
			multiply = -0.25
		}
	}

	# Retrain the workforce
	option = {
		name = post_scarcity_events.1.b
		ai_chance = {
			base = 3
			# intentional flat weight (no strong strategic axis) — pragmatic centrist response
		}
		add_modifier = {
			name = workforce_retraining_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		add_treasury = {
			value = yearly_gross_income
			multiply = -0.15
		}
	}

	# Suppress the protests
	option = {
		name = post_scarcity_events.1.c
		ai_chance = {
			base = 1
			# authoritarian/security states lean toward suppressing unrest
			modifier = {
				trigger = {
					OR = {
						has_law = law_type:law_secret_police
						has_law = law_type:law_autocracy
						has_law = law_type:law_oligarchy
						has_law = law_type:law_single_party_state
					}
				}
				add = 4
			}
		}
		add_radicals = {
			value = small_radicals
			strata = lower
		}
	}
}
