# Event 3: Diplomatic Crisis - a rival great power has started a diplomatic play
# (or the war it became) against a smaller country in our sphere: a fellow power
# bloc member, or one we are already bound to defend. Both are real and named.

	# REVIEWED 2026-09-25 (imputed_foreign_action): tracked in #427 — gate on real pressure

	placement = root

	event_image = { texture = "gfx/event_pictures/diplomatic_crisis_ultimatum.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	title = world_war_events.3.t
	desc = world_war_events.3.d
	flavor = world_war_events.3.f

	duration = 3

	trigger = {
		NOT = { has_variable = ww_active_phase }
		ww_ideological_tension >= 50
		country_rank >= rank_value:great_power
		any_diplomatic_play = {
			ww_play_is_rival_pressure_on_sphere_of = { COUNTRY = ROOT }
		}
	}

	cooldown = { days = normal_modifier_time }

	immediate = {
		random_diplomatic_play = {
			limit = {
				ww_play_is_rival_pressure_on_sphere_of = { COUNTRY = ROOT }
			}
			initiator = { save_scope_as = crisis_aggressor }
			target = { save_scope_as = crisis_victim }

			trigger = { ww_can_guarantee_sovereignty = { GUARANTOR = ROOT TARGET = scope:crisis_victim } }
			text = world_war_events.3.a
		}

			trigger = { always = yes }
			text = world_war_events.3.a_bound
		}

			base = 4
			modifier = {
				trigger = { country_is_democratic = yes }
				add = 2
			}

		change_relations = {
			country = scope:crisis_aggressor
			value = -30

			FIRST = root
			SECOND = scope:crisis_aggressor

			value = 25
		}

			limit = { ww_can_guarantee_sovereignty = { GUARANTOR = ROOT TARGET = scope:crisis_victim } }

	option = {

			base = 2
			modifier = {
				trigger = { je:je_world_war = { has_modifier = ww_appeasement_modifier } }

# ============================================================================
# WAR OUTBREAK EVENT (5) - The brink: launch the world war, or stand down
# Fired from the je_world_war pulse when tension reaches critical. Nothing marks
# either country as a belligerent until the launch option is chosen.
# ============================================================================

world_war_events.5 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/world_war_declaration.dds" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	title = world_war_events.5.t
	desc = world_war_events.5.d
	flavor = world_war_events.5.f

	duration = 3

	trigger = {
		NOT = { has_variable = ww_active_phase }
		world_war_conditions_met = yes
		ww_ideological_tension >= 70
	}

	cooldown = { days = long_modifier_time }

	immediate = {
		# Find the primary rival for the war
		random_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = rivalry }
				country_rank >= rank_value:great_power
				ROOT = { country_has_opposed_ideology = { TARGET = prev } }
			}
			save_scope_as = war_target
		}
		# Hold the outbreak for us while we decide, so no other great power opens a
		# second world war meanwhile (world_war_conditions_met). Both options clear it.
		set_global_variable = { name = ww_outbreak_pending days = 90 }
	}

	# Option A: Launch the war
	option = {
		name = world_war_events.5.a
		default_option = yes
		ai_chance = { base = 10 }
		hidden_effect = {
			remove_global_variable = ww_outbreak_pending
			# A new world war: forget the outcome of any earlier one
			ww_clear_stale_outcome_latches = yes
			# Mark only the two primary belligerents
			set_variable = ww_active_phase
			set_variable = ww_aggressor_side
			set_variable = { name = ww_years_elapsed value = 0 }
			set_variable = { name = ww_months_counter value = 0 }
			scope:war_target = {
				set_variable = ww_active_phase
				set_variable = ww_defender_side
				set_variable = { name = ww_years_elapsed value = 0 }
				set_variable = { name = ww_months_counter value = 0 }
			}
			# Other great powers are NOT auto-marked as participants.
			# They will see the war via the JE and can choose to join
			# through event 20 (late entry) or the join_war button.
		}
		# Create the diplomatic play against the rival
		create_diplomatic_play = {
			target_country = scope:war_target
			type = dp_contain_threat
			escalation = 90
		}
		# Remove leadup modifiers and apply war modifiers
		if = {
			limit = { je:je_world_war = { has_modifier = ww_rising_tensions_modifier } }
			je:je_world_war = { remove_modifier = ww_rising_tensions_modifier }
		}
		if = {
			limit = { je:je_world_war = { has_modifier = ww_appeasement_modifier } }
			je:je_world_war = { remove_modifier = ww_appeasement_modifier }
		}
		# National rally effect
		add_loyalists = {
			value = large_radicals
			pop_type = soldiers
		}
		add_loyalists = {
			value = medium_radicals
			strata = lower
		}
	}

	# Option B: Stand down - no play is launched. The general staff and the rank
	# and file take it badly; this event's cooldown keeps the question from coming
	# straight back.
	option = {
		name = world_war_events.5.b
		ai_chance = {
			base = 1
			modifier = {
				trigger = { je:je_world_war = { has_modifier = ww_appeasement_modifier } }
				add = 3
			}
			modifier = {
				trigger = { country_is_democratic = yes }
				add = 1
			}
		}
		hidden_effect = { remove_global_variable = ww_outbreak_pending }
		ig_approval_effect = { IG = ig_armed_forces MODIFIER = ig_approval_negative_modifier DAYS = normal_modifier_time }
		add_radicals = {
			value = small_radicals
			pop_type = soldiers
		}
		add_loyalists = {
			value = small_radicals
			strata = middle
		}
	}
}
