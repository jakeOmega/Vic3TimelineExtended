# Design: docs/superpowers/specs/2026-09-26-nuclear-loose-warheads-design.md
# §3; what shipped is docs/systems/nuclear_crisis_design.md §0.10 step 5. The
# logic is in common/scripted_effects/nuclear_loose_effects.txt.

#   .1   (hidden) a plot, drawn from the social-tensions terror pool
#   .2   A City Destroyed — the victim, when the device detonates
#   .3   A Device Seized — the victim, when its services find it first

#        to a shortlisted suspect
#   .7   Named in the Investigation — a suspect named in public
#   .8   the victim hears the answer to its demand
#   .9   Loose Material Secured — a warhead unaccounted for is found
# ============================================================================

# ---- .4 Where Did It Come From? -----------------------------------------------------
# A month after a device surfaced. The attribution was rolled then
# (nd_loose_attribute); this says what the investigators found and asks what
# to do with it.
nuclear_loose.4 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/nuclear_proliferation_threat.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = nuclear_loose.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = nuclear_loose.4.d_confirmed
				trigger = { nd_loose_origin_answerable = yes }
			}
			triggered_desc = {
				desc = nuclear_loose.4.d_confirmed_gone
				trigger = { nd_loose_attributed = { RESULT = 1 } }
			}
			triggered_desc = {
				desc = nuclear_loose.4.d_shortlist_three
				trigger = {
					nd_loose_attributed = { RESULT = 2 }
					var:nd_loose_suspect_count >= 3
				}
			}
			triggered_desc = {
				desc = nuclear_loose.4.d_shortlist
				trigger = { nd_loose_attributed = { RESULT = 2 } }
			}
			triggered_desc = {
				desc = nuclear_loose.4.d_unknown
			}
		}
	}
	flavor = nuclear_loose.4.f

	duration = 3

	trigger = {
		has_variable = nd_loose_attribution
	}

	immediate = {
		var:nd_loose_origin ?= { save_scope_as = nd_loose_accused }
		var:nd_loose_suspect_1 ?= { save_scope_as = nd_loose_s1 }
		var:nd_loose_suspect_2 ?= { save_scope_as = nd_loose_s2 }
		var:nd_loose_suspect_3 ?= { save_scope_as = nd_loose_s3 }
		random_rival_country = {
			limit = { is_country_alive = yes }
			save_scope_as = nd_loose_rival
		}
	}

	# Hold them to account: a confirmed origin that still stands.
	option = {
		name = nuclear_loose.4.a
		trigger = { nd_loose_origin_answerable = yes }
		nd_loose_hold_to_account = yes
		ai_chance = {
			base = 8
		}
	}

	# Name every suspect, in public.
	option = {
		name = nuclear_loose.4.b
		trigger = { nd_loose_attributed = { RESULT = 2 } }
		nd_loose_name_suspects = yes
		ai_chance = {
			# An AI names innocent great powers only under a ruler who would
			# rather be seen to act than be right.
			base = 0
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 4
			}
		}
	}

	# Ask each suspect, privately, to cooperate.
	option = {
		name = nuclear_loose.4.c
		trigger = { nd_loose_attributed = { RESULT = 2 } }
		nd_loose_ask_suspects = yes
		ai_chance = {
			base = 6
		}
	}

	# Blame a rival anyway.
	option = {
		name = nuclear_loose.4.e
		trigger = {
			nd_loose_attributed = { RESULT = 3 }
			exists = scope:nd_loose_rival
		}
		nd_loose_blame_rival = yes
		ai_chance = {
			base = 0
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 3
			}
		}
	}

	# Say nothing for now.
	option = {
		name = nuclear_loose.4.g
		default_option = yes
		ai_chance = {
			base = 2
			modifier = {
				trigger = { nd_loose_attributed = { RESULT = 3 } }
				add = 4
			}
		}
	}

	after = {
		nd_loose_clear_record = yes
	}
}

# ---- .6 The Warhead Was Ours? ------------------------------------------------------------
# To a confirmed origin (the full demand) or a shortlisted suspect (a request
# to cooperate), from nd_loose_send_demand.
nuclear_loose.6 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/diplomatic_crisis_ultimatum.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = {
		first_valid = {
			triggered_desc = {
				desc = nuclear_loose.6.t
				trigger = { nd_loose_demand_is_full = yes }
			}
			triggered_desc = {
				desc = nuclear_loose.6.t_request
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				desc = nuclear_loose.6.d
				trigger = { nd_loose_demand_is_full = yes }
			}
			triggered_desc = {
				desc = nuclear_loose.6.d_request
			}
		}
	}
	flavor = nuclear_loose.6.f

	duration = 3

	trigger = {
		var:nd_loose_demand_from ?= { is_country_alive = yes }
		has_variable = nd_loose_demand_kind
	}

	immediate = {
		var:nd_loose_demand_from = { save_scope_as = nd_loose_claimant }
	}

	# Pay compensation (the full demand only).
	option = {
		name = nuclear_loose.6.a
		trigger = { nd_loose_demand_is_full = yes }
		nd_loose_pay = yes
		ai_chance = {
			base = 3
			modifier = {
				trigger = { scope:nd_loose_claimant.relations:root >= relations_threshold:cordial }
				add = 3
			}
			modifier = {
				trigger = { scope:nd_loose_claimant = { country_rank >= rank_value:great_power } }
				add = 2
			}
		}
	}

	# Open our custody to inspection.
	option = {
		name = nuclear_loose.6.b
		nd_loose_open_to_inspection = yes
		ai_chance = {
			base = 4
			modifier = {
				trigger = { ruler_is_cautious = yes }
				add = 3
			}
		}
	}

	# Deny it.
	option = {
		name = nuclear_loose.6.c
		default_option = yes
		nd_loose_deny = yes
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ruler_is_aggressive = yes }
				add = 5
			}
			modifier = {
				trigger = { has_diplomatic_pact = { who = scope:nd_loose_claimant type = rivalry } }
				add = 3
			}
		}
	}
}
