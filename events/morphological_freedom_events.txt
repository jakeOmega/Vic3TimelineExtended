# ============================================================================
# MORPHOLOGICAL FREEDOM DEBATE EVENTS (era 13 spec § 7.2)
# ============================================================================
#
# Pooled from je_morphological_freedom's monthly pulse (240 = 0 beside six
# 10s). All six share a two-year lockout, te_mf_event_recent, so one never
# follows another within two years. .1, .3, .4 and .6 fire once (their
# twenty-year cooldown is the debate's whole length); .2 and .5 can return
# after ten years. A debate that runs its twenty years brings about eight
# events at most. .4 and .6 need clinics working legally, so not under
# Natural Form.
#
# .100 and .200 are the outcomes, fired by the entry's on_timeout (Natural
# Form held to the end) and on_complete.

# Event 1: The First Clinics
morphological_freedom_events.1 = {
	type = country_event
	placement = root

	event_image = { texture = "gfx/event_pictures/neural_interface_consumer.dds" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	title = morphological_freedom_events.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_law = law_type:law_natural_form }
				desc = morphological_freedom_events.1.d_natural
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = morphological_freedom_events.1.d
			}
		}
	}
	flavor = morphological_freedom_events.1.f

	duration = 3
	cooldown = { days = very_long_modifier_time }

	trigger = {
		has_technology_researched = morphological_freedom
		NOT = { has_variable = te_mf_event_recent }
	}

	immediate = {
		set_variable = { name = te_mf_event_recent days = 730 }
	}

	# Inspect them like any other clinic
	option = {
		name = morphological_freedom_events.1.a
		default_option = yes
		ai_chance = {
			base = 3
		}
		add_treasury = { value = sv_treasury_event_small multiply = -1 }
		add_modifier = {
			name = mf_clinic_inspections_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
	}

	# Leave the clinics to their customers
	option = {
		name = morphological_freedom_events.1.b
		ai_chance = {
			base = 2
			modifier = {
				trigger = { ig:ig_industrialists ?= { is_in_government = yes } }
				add = 1
			}
		}
		add_modifier = {
			name = mf_clinic_boom_modifier
			days = normal_modifier_time
			is_decaying = yes
		}
		ig_approval_effect = { IG = ig_industrialists MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
		ig_approval_effect = { IG = ig_devout         MODIFIER = ig_approval_negative_modifier DAYS = normal_modifier_time }
	}

	# Shut them down: only while the law allows no such clinics
	option = {
		name = morphological_freedom_events.1.c
		trigger = {
			has_law = law_type:law_natural_form
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = { ig:ig_devout ?= { is_in_government = yes } }
				add = 2
			}
		}
		add_radicals = {
			value = very_small_radicals
			pop_type = academics
		}
		ig_approval_effect = { IG = ig_devout         MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
		ig_approval_effect = { IG = ig_rural_folk     MODIFIER = ig_approval_positive_modifier DAYS = normal_modifier_time }
		ig_approval_effect = { IG = ig_intelligentsia MODIFIER = ig_approval_negative_modifier DAYS = normal_modifier_time }
