# ======================================================================
#  1 – Siting the New Capital
#  ROOT = the country that took the decision. The immediate saves up to three sites, the largest by people first; each
#  option builds the capital there (te_demog_pc_build: the costs, then the move). Not Now costs nothing, and the
#  decision stays open.
#  AI: the largest site most often (3:2:1), never Not Now: the decision's own rule (te_demog_pc_ai_wants) already said
#  it wants one.
# ======================================================================
te_demog_capital.1 = {
	type = country_event
	placement = root

	title = te_demog_capital.1.t
	desc = te_demog_capital.1.d
	flavor = te_demog_capital.1.f

	event_image = { video = "middleeast_engineer_blueprint" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	immediate = {
		ordered_scope_state = {
			limit = { te_demog_pc_candidate = yes }
			order_by = te_demog_pc_site_order
			position = 0
			check_range_bounds = no
			save_scope_as = te_pc_site_1
		}
		ordered_scope_state = {
			limit = { te_demog_pc_candidate = yes }
			order_by = te_demog_pc_site_order
			position = 1
			check_range_bounds = no
			save_scope_as = te_pc_site_2
		}
		ordered_scope_state = {
			limit = { te_demog_pc_candidate = yes }
			order_by = te_demog_pc_site_order
			position = 2
			check_range_bounds = no
			save_scope_as = te_pc_site_3
		}
	}

	option = {
		name = te_demog_capital.1.a
		trigger = { exists = scope:te_pc_site_1 }
		default_option = yes
		custom_tooltip = te_demog_capital.1.a.tt
		te_demog_pc_build = { SITE = te_pc_site_1 }
		ai_chance = {
			base = 3
		}
	}

	option = {
		name = te_demog_capital.1.b
		trigger = { exists = scope:te_pc_site_2 }
		custom_tooltip = te_demog_capital.1.b.tt
		te_demog_pc_build = { SITE = te_pc_site_2 }
		ai_chance = {
			base = 2
		}
	}

	option = {
		name = te_demog_capital.1.c
		trigger = { exists = scope:te_pc_site_3 }
		custom_tooltip = te_demog_capital.1.c.tt
		te_demog_pc_build = { SITE = te_pc_site_3 }
		ai_chance = {
			base = 1
		}
	}

	option = {
		name = te_demog_capital.1.e
		# REVIEWED 2026-10-12 (no_effect_option): waiting is the choice; the tooltip says nothing is spent and the decision stays open
		custom_tooltip = te_demog_capital.1.e.tt
		ai_chance = {
			base = 0
		}
	}
}
