namespace = te_qa_probe

# Hidden carrier: runs whatever the mailbox effect currently holds.
te_qa_probe.1 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "TE_QA_PROBE carrier fired country=[THIS.GetCountry.GetTagName] date=[TimeKeeper.GetCurrentDate.GetString]"
		te_qa_probe_mailbox = yes
	}
}

	placement = ROOT

	event_image = { video = "africa_desert_expedition" }

	duration = 3
